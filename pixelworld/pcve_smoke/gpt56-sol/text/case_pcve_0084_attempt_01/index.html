+"""Remove the apple from source frame 18 onward.
+
+The source is a locked-off camera shot.  The apple is the small warm/red rigid
+body travelling rightward close to y=86 in frames 18--39 (one-based).  Later
+frames contain the unobstructed plate at the same image coordinates, so those
+frames are used as clean plates.  This preserves the original filmed grain and
+spatial texture rather than synthesising a smooth patch.
+"""
+from pathlib import Path
+import numpy as np
+from scipy.ndimage import binary_dilation, binary_closing, gaussian_filter
+import artifact_io
+
+CASE_DIR = Path(__file__).resolve().parent
+START = 17  # zero-based index of source frame 18
+
+
+def clean_plate(source, index, center_x):
+    """Select a temporally stable apple-free frame for the current patch."""
+    h, w = source.shape[1:3]
+    x0, x1 = max(0, center_x - 15), min(w, center_x + 16)
+    y0, y1 = 75, 96
+    yy, xx = np.mgrid[y0:y1, x0:x1]
+    # Exclude the future replacement area while matching the surrounding grain.
+    core = ((xx - center_x) / 11.0) ** 2 + ((yy - 86.0) / 8.7) ** 2 <= 1.35
+    ring = ~core
+    candidates = list(range(0, 13)) + list(range(39, len(source)))
+    target = source[index, y0:y1, x0:x1].astype(np.int16)
+    score = []
+    for j in candidates:
+        delta = np.abs(target - source[j, y0:y1, x0:x1].astype(np.int16))
+        score.append((float(delta[ring].mean()), j))
+    return min(score)[1]
+
+
+def build_frames():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    frames = source.copy()
+
+    # Apple centers, measured from the color-separated body.  The object exits
+    # through the right edge after source frame 39.
+    centers = [165, 172, 178, 184, 189, 194, 199, 205, 209, 214, 219,
+               223, 227, 230, 235, 238, 242, 245, 248, 251, 253, 255]
+
+    yy, xx = np.mgrid[0:source.shape[1], 0:source.shape[2]]
+    for index, cx in enumerate(centers, start=START):
+        # The visible apple body is red/warm.  Include its dark contour/stem by
+        # closing and dilating the color mask, but constrain edits to a compact
+        # moving ellipse so nearby scenery is untouched.
+        f = source[index].astype(np.int16)
+        warm = ((f[..., 0] > f[..., 1] + 7) &
+                (f[..., 0] > f[..., 2] + 4) & (f[..., 0] > 68))
+        support = (((xx - cx) / 12.0) ** 2 + ((yy - 86.0) / 9.0) ** 2 <= 1.0)
+        seed = warm & support

+        mask &= (((xx - cx) / 13.0) ** 2 + ((yy - 86.0) / 10.0) ** 2 <= 1.0)
+
+        # Feather only the outer 1--2 pixels.  Interior pixels come exactly from
+        # a real clean plate; the narrow blend hides compression-phase changes.
+        alpha = gaussian_filter(mask.astype(np.float32), sigma=0.65)
+        alpha = np.clip(alpha * 1.35, 0.0, 1.0)[..., None]
+        plate_index = clean_plate(source, index, cx)

+        frames[index] = np.clip(np.rint(out), 0, 255).astype(np.uint8)

+    # source pixels exactly is the physically correct continuation.
+    return frames
