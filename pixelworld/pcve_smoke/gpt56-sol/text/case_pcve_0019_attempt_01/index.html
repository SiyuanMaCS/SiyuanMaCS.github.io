+"""Generate the friction-edited PixelWorld clip from the supplied RGB frames.
+
+The incoming star ball is already sliding on the horizontal surface in frame 1.
+Its unimpeded centres in frames 1--8 fit constant friction very closely.  The
+counterfactual therefore keeps its measured initial velocity, multiplies the
+measured deceleration by five, and stops it at the first contact with the
+stationary construction.  The ball's source appearances are motion-retimed
+and aligned before interpolation, preserving its star texture and lighting
+without introducing a second/ghost ball.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def soft_disc(height, width, cx, cy, inner=16.3, outer=18.6):
+    """A softly antialiased mask covering the ball and its immediate shading."""
+    yy, xx = np.ogrid[:height, :width]
+    radius = np.sqrt((xx - cx) ** 2 + (yy - cy) ** 2)
+    return np.clip((outer - radius) / (outer - inner), 0.0, 1.0).astype(np.float32)
+
+
+def translate_layer(layer, dx, dy):
+    matrix = np.float32([[1.0, 0.0, dx], [0.0, 1.0, dy]])
+    return cv2.warpAffine(
+        layer,
+        matrix,
+        (layer.shape[1], layer.shape[0]),
+        flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=0,
+    )
+
+
+def generate():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3) and source.dtype == np.uint8
+    h, w = source.shape[1:3]
+
+    # Measured centres of the freely sliding ball in source frames 1--8.
+    # The eighth source frame is the last frame before visible deformation of
+    # the contacted construction begins.
+    centers_x = np.array([238.0, 222.8, 208.0, 193.3, 178.8, 164.8, 150.8, 136.8])
+    centers_y = np.array([70.0, 69.8, 70.0, 69.4, 69.3, 69.2, 69.6, 69.8])
+    sample_t = np.arange(8, dtype=np.float64)
+
+    # x = q2*t^2 + q1*t + q0, hence source deceleration is 2*q2.
+    q2, q1, q0 = np.polyfit(sample_t, centers_x, 2)
+
+    # A temporal median reveals the surface behind the rapidly moving ball
+    # while retaining the untouched construction in its initial state.
+    revealed = np.median(source[:8].astype(np.float32), axis=0)
+
+    # Keep the exact first-frame scene as the base, replacing only the initial
+    # ball footprint by the revealed surface before compositing the ball.
+    initial_cover = soft_disc(h, w, centers_x[0], centers_y[0], 16.0, 19.3)
+    base = source[0].astype(np.float32)
+    base = base * (1.0 - initial_cover[..., None]) + revealed * initial_cover[..., None]
+
+    # Difference layers retain the ball's real shading and star pattern.  A
+    # soft geometric support excludes compression noise elsewhere in the shot.
+    layers = []
+    for j in range(8):
+        support = soft_disc(h, w, centers_x[j], centers_y[j])
+        difference = source[j].astype(np.float32) - revealed
+        layers.append(difference * support[..., None])
+
+    result = np.empty_like(source)
+    contact_x = centers_x[-1]
+
+    for frame_index in range(96):
+        t = float(frame_index)
+
+        # Preserve the measured initial velocity, but use 5x the fitted
+        # frictional acceleration from frame 1 onward.
+        counter_x = q0 + q1 * t + (5.0 * q2) * t * t
+        counter_x = max(contact_x, counter_x)
+
+        # Find the source-time whose ball position equals the counterfactual
+        # position. This naturally retimes both translation and rotation.
+        if counter_x <= contact_x + 1e-6:
+            source_time = 7.0
+        else:
+            discriminant = max(0.0, q1 * q1 - 4.0 * q2 * (q0 - counter_x))
+            source_time = (-q1 - np.sqrt(discriminant)) / (2.0 * q2)
+            source_time = float(np.clip(source_time, 0.0, 7.0))
+
+        lo = int(np.floor(source_time))
+        hi = min(7, lo + 1)
+        fraction = source_time - lo
+        target_y = float(np.interp(source_time, sample_t, centers_y))
+
+        # Align both neighbouring source appearances to one centre before
+        # blending; this gives smooth subpixel motion without double images.
+        aligned_lo = translate_layer(
+            layers[lo], counter_x - centers_x[lo], target_y - centers_y[lo]
+        )
+        if hi == lo:
+            ball_layer = aligned_lo
+        else:
+            aligned_hi = translate_layer(
+                layers[hi], counter_x - centers_x[hi], target_y - centers_y[hi]
+            )
+            ball_layer = (1.0 - fraction) * aligned_lo + fraction * aligned_hi
+
+        composed = np.clip(base + ball_layer, 0.0, 255.0)
+        result[frame_index] = np.rint(composed).astype(np.uint8)
+
+    # The intervention changes dynamics, not the initial visible state.
+    result[0] = source[0]
+    return result
+
+
+if __name__ == "__main__":
+    frames = generate()
+    artifact_io.save(frames, CASE_DIR)
+
