+"""Generate the requested 0.75x-initial-speed counterfactual video.
+
+The intervention occurs at the first source frame.  In this rigid-body clip the
+only changing pixels belong to the incoming star ball and the objects moved by
+its collision.  Sampling the source physical evolution at 0.75 source frames
+per output frame therefore gives the incoming ball 75% of its original image
+velocity, delays impact, and carries the reduced rate through the consequent
+collision motion.  Bidirectional dense flow is used between source frames so
+the large, roughly 14 pixel/frame ball motion remains a single sharp object
+rather than becoming a cross-faded double image.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+SPEED_SCALE = 0.75
+
+
+def _flow(dis, first_rgb, second_rgb):
+    """Dense numeric RGB-frame correspondence from first to second."""
+    first = cv2.cvtColor(first_rgb, cv2.COLOR_RGB2GRAY)
+    second = cv2.cvtColor(second_rgb, cv2.COLOR_RGB2GRAY)
+    return dis.calc(first, second, None)
+
+
+def _interpolate(first, second, alpha, grid, dis):
+    """Motion-compensated interpolation of two adjacent RGB frames."""
+    forward = _flow(dis, first, second)
+    backward = _flow(dis, second, first)
+
+    # cv2.remap expects, at every output location, the location to sample.
+    # These backward maps move each endpoint toward the intermediate time.
+    map_from_first = grid - np.float32(alpha) * forward
+    map_from_second = grid - np.float32(1.0 - alpha) * backward
+    moved_first = cv2.remap(
+        first,
+        map_from_first,
+        None,
+        cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_REFLECT_101,
+    )
+    moved_second = cv2.remap(
+        second,
+        map_from_second,
+        None,
+        cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_REFLECT_101,
+    )
+
+    # A small endpoint-weighted blend handles occlusion boundaries and retains
+    # the source renderer's natural motion blur while the warps align objects.
+    mixed = (
+        np.float32(1.0 - alpha) * moved_first.astype(np.float32)
+        + np.float32(alpha) * moved_second.astype(np.float32)
+    )
+    return np.clip(np.rint(mixed), 0, 255).astype(np.uint8)
+
+
+def generate():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    if source.shape != (96, 144, 256, 3) or source.dtype != np.uint8:
+        raise ValueError(f"unexpected source array: {source.shape} {source.dtype}")
+
+    height, width = source.shape[1:3]
+    xx, yy = np.meshgrid(
+        np.arange(width, dtype=np.float32),
+        np.arange(height, dtype=np.float32),
+    )
+    grid = np.dstack((xx, yy))
+
+    # Medium DIS at full resolution robustly follows both the fast incoming
+    # ball and the slower tumbling pieces without any learned model.
+    dis = cv2.DISOpticalFlow_create(cv2.DISOPTICAL_FLOW_PRESET_MEDIUM)
+    dis.setFinestScale(0)
+
+    result = np.empty_like(source)
+    for output_index in range(len(result)):
+        source_time = SPEED_SCALE * output_index
+        left = int(np.floor(source_time))
+        alpha = float(source_time - left)
+        if alpha < 1e-8 or left == len(source) - 1:
+            result[output_index] = source[left]
+        else:
+            result[output_index] = _interpolate(
+                source[left], source[left + 1], alpha, grid, dis
+            )
+
+    # The edit begins exactly on frame 1, so its factual initial appearance is
+    # preserved byte-for-byte.
+    result[0] = source[0]
+    return result
+
+
+if __name__ == "__main__":
+    frames = generate()
+    artifact_io.save(frames, CASE_DIR)
+    print(
+        f"saved {len(frames)} frames at 256x144; "
+        f"source-time range 0..{SPEED_SCALE * (len(frames) - 1):.2f}"
+    )
