+"""Generate the 40x-blue-mass counterfactual for this case.
+
+The source is a Newton-cradle-like two-marble event.  Up to contact, changing
+the resting blue marble's mass has no visible effect, so those RGB frames are
+kept verbatim.  At contact we use the one-dimensional elastic-collision
+solution for masses 1 and 40, preserve the measured red-marble acceleration
+on the inclined track, and give the blue marble 1/20.5 of the equal-mass
+transfer seen in the source.  Rendering uses only source pixels and numeric
+RGB operations.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+N, H, W = 96, 144, 256
+
+
+def clean_background(src: np.ndarray) -> np.ndarray:
+    """Recover the static scene after removing the final marble footprint."""
+    # Frame 35 is a sharp, settled view with the incoming marble absent from
+    # the whole ramp.  Its only non-background area is the final pair footprint
+    # in the lower-left corner.  Detect that footprint against frame zero and
+    # inpaint it from the immediately surrounding wood texture.
+    base = src[35].copy()
+    delta = np.max(np.abs(src[35].astype(np.int16) - src[0].astype(np.int16)), axis=2)
+    yy, xx = np.mgrid[:H, :W]
+    roi = (xx < 49) & (yy > 70) & (yy < 119)
+    mask = ((delta > 7) & roi).astype(np.uint8) * 255
+    # Include both silhouettes and their small contact shadows.
+    mask = cv2.morphologyEx(mask, cv2.MORPH_CLOSE, np.ones((5, 5), np.uint8))
+    mask = cv2.dilate(mask, np.ones((5, 5), np.uint8), iterations=1)
+    return cv2.inpaint(base, mask, 5.0, cv2.INPAINT_TELEA)
+
+
+def make_sprite(frame: np.ndarray, cx: float, cy: float, kind: str,
+                reference: np.ndarray | None = None):
+    """Extract a compact source-pixel sprite and a feathered alpha matte."""
+    half = 13
+    x0, y0 = int(round(cx)) - half, int(round(cy)) - half
+    rgb = frame[y0:y0 + 2 * half + 1, x0:x0 + 2 * half + 1].copy()
+    ph, pw = rgb.shape[:2]
+    py, px = np.mgrid[:ph, :pw]
+    lx, ly = cx - x0, cy - y0
+    radial = (px - lx) ** 2 + (py - ly) ** 2
+
+    if kind == "blue":
+        a = rgb.astype(np.int16)
+        chroma = ((a[:, :, 2] > 76) &
+                  (a[:, :, 2] > 1.10 * a[:, :, 1]) &
+                  (a[:, :, 2] > 1.14 * a[:, :, 0]) &
+                  (radial < 10.0 ** 2)).astype(np.uint8)
+        # Fill the dark rim and highlight holes of the blue sphere.
+        chroma = cv2.morphologyEx(chroma, cv2.MORPH_CLOSE,
+                                  np.ones((5, 5), np.uint8))
+        chroma = cv2.dilate(chroma, np.ones((3, 3), np.uint8), iterations=1)
+        alpha = cv2.GaussianBlur(chroma.astype(np.float32), (0, 0), .58)
+    else:
+        assert reference is not None
+        ref = reference[y0:y0 + ph, x0:x0 + pw]
+        diff = np.max(np.abs(rgb.astype(np.int16) - ref.astype(np.int16)), axis=2)
+        matte = ((diff > 6) & (radial < 10.5 ** 2)).astype(np.uint8)
+        matte = cv2.morphologyEx(matte, cv2.MORPH_CLOSE,
+                                 np.ones((3, 3), np.uint8))
+        alpha = cv2.GaussianBlur(matte.astype(np.float32), (0, 0), .55)
+
+    # A geometric falloff makes the one-pixel silhouette antialias stable even
+    # after subpixel translation.
+    edge = np.clip((9.35 - np.sqrt(radial)) / 0.9, 0.0, 1.0)
+    alpha = np.clip(alpha * edge, 0.0, 1.0)
+    return rgb, alpha, (lx, ly)
+
+
+def rotate_sprite(sprite, angle_degrees: float):
+    rgb, alpha, anchor = sprite
+    h, w = alpha.shape
+    mat = cv2.getRotationMatrix2D(anchor, angle_degrees, 1.0)
+    rr = cv2.warpAffine(rgb, mat, (w, h), flags=cv2.INTER_LINEAR,
+                        borderMode=cv2.BORDER_REFLECT_101)
+    aa = cv2.warpAffine(alpha, mat, (w, h), flags=cv2.INTER_LINEAR,
+                        borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+    return rr, aa, anchor
+
+
+def composite(canvas: np.ndarray, sprite, cx: float, cy: float) -> None:
+    """Subpixel alpha-composite a sprite at the requested center."""
+    rgb, alpha, (ax, ay) = sprite
+    h, w = alpha.shape
+    x0, y0 = int(np.floor(cx - ax)), int(np.floor(cy - ay))
+    fx, fy = (cx - ax) - x0, (cy - ay) - y0
+    mat = np.float32([[1, 0, fx], [0, 1, fy]])
+    rgb = cv2.warpAffine(rgb, mat, (w, h), flags=cv2.INTER_LINEAR,
+                         borderMode=cv2.BORDER_REFLECT_101)
+    alpha = cv2.warpAffine(alpha, mat, (w, h), flags=cv2.INTER_LINEAR,
+                           borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+
+    xa, ya, xb, yb = max(0, x0), max(0, y0), min(W, x0 + w), min(H, y0 + h)
+    if xa >= xb or ya >= yb:
+        return
+    sx0, sy0 = xa - x0, ya - y0
+    a = alpha[sy0:sy0 + yb - ya, sx0:sx0 + xb - xa, None]
+    fg = rgb[sy0:sy0 + yb - ya, sx0:sx0 + xb - xa].astype(np.float32)
+    bg = canvas[ya:yb, xa:xb].astype(np.float32)
+    canvas[ya:yb, xa:xb] = np.clip(fg * a + bg * (1.0 - a), 0, 255).astype(np.uint8)
+
+
+def add_contact_shadow(canvas: np.ndarray, cx: float, cy: float, strength=.18):
+    """Small soft shadow consistent with the source's lower-left track light."""
+    if cx < -12 or cx > W + 12:
+        return
+    layer = np.zeros((H, W), np.float32)
+    cv2.ellipse(layer, (int(round(cx + 1)), int(round(cy + 7))),
+                (7, 2), 0, 0, 360, 1.0, -1, cv2.LINE_AA)
+    layer = cv2.GaussianBlur(layer, (0, 0), 1.15)[:, :, None]
+    tint = np.array([39., 29., 25.], np.float32)
+    a = layer * strength
+    canvas[:] = np.clip(canvas.astype(np.float32) * (1 - a) + tint * a,
+                        0, 255).astype(np.uint8)
+
+
+def first_red_motion(d: float):
+    """Post-first-impact red position, fitted to the incoming source arc."""
+    # Contact at source time 21.5: p=(40,80), measured incoming velocity
+    # (-12.94,+3.57) px/frame, acceleration=(-.602,+.166).
+    return np.array([40.0 + 12.31 * d - .301 * d * d,
+                     80.0 - 3.40 * d + .083 * d * d], np.float64)
+
+
+def generate() -> np.ndarray:
+    src = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert src.shape == (N, H, W, 3) and src.dtype == np.uint8
+    plate = clean_background(src)
+
+    red0 = make_sprite(src[0], 179.15, 41.65, "red", src[35])
+    blue0 = make_sprite(src[0], 24.48, 87.68, "blue")
+
+    out = np.empty_like(src)
+    # No force depends on the resting blue mass before contact.  This exact
+    # copy also preserves every source reflection, spin cue and compression
+    # nuance in the factual pre-contact history.
+    out[:22] = src[:22]
+
+    contact_frame = 21.5
+    blue_x0, blue_y0 = 24.48, 87.68
+    # Equal-mass source transfer is about 6.9 px/frame; 2/(1+40) of the
+    # incoming momentum gives the heavy blue marble roughly .34 px/frame.
+    blue_v1 = np.array([-.337, .024], np.float64)
+
+    # Solve the second contact numerically.  The returning red marble meets
+    # the slowly departing heavy blue marble near source frame 63.5.
+    ds = np.linspace(38.0, 45.0, 1401)
+    rpath = np.stack([first_red_motion(float(d)) for d in ds])
+    bpath = np.stack([np.array([blue_x0, blue_y0]) + blue_v1 * d for d in ds])
+    sep = np.linalg.norm(rpath - bpath, axis=1)
+    d2 = float(ds[np.argmin(np.abs(sep - 16.5))])
+    p2r = first_red_motion(d2)
+    p2b = np.array([blue_x0, blue_y0]) + blue_v1 * d2
+
+    vin2 = np.array([12.31 - .602 * d2, -3.40 + .166 * d2])
+    # Elastic 1-D response along the dominant (track) direction.  Applying the
+    # mass equations component-wise is appropriate here because both centers
+    # and velocities are nearly collinear at contact.
+    vblue2 = (2.0 / 41.0) * vin2 + (39.0 / 41.0) * blue_v1
+    vred2 = (-39.0 / 41.0) * vin2 + (80.0 / 41.0) * blue_v1
+
+    for t in range(22, N):
+        canvas = plate.copy()
+        d = t - contact_frame
+
+        if d < d2:
+            rp = first_red_motion(d)
+            bp = np.array([blue_x0, blue_y0]) + blue_v1 * d
+            red_dist = abs(12.31 * d - .301 * d * d)
+        else:
+            e = d - d2
+            rp = p2r + vred2 * e + .5 * np.array([-.602, .166]) * e * e
+            bp = p2b + vblue2 * e
+            red_dist = abs(12.31 * d2 - .301 * d2 * d2) + np.linalg.norm(vred2) * e
+
+        # Shadows are laid down before the opaque source-pixel spheres.
+        add_contact_shadow(canvas, float(bp[0]), float(bp[1]), .17)
+        add_contact_shadow(canvas, float(rp[0]), float(rp[1]), .15)
+
+        # The subtle rotation prevents the rebounding marble from looking like
+        # a translated decal.  Blue rotation is deliberately tiny, matching
+        # its 40x inertia and short travel.
+        red_angle = -red_dist / 8.4 * (180.0 / np.pi)
+        blue_dist = abs(float(bp[0] - blue_x0))
+        blue_angle = blue_dist / 8.2 * (180.0 / np.pi)
+        composite(canvas, rotate_sprite(red0, red_angle), float(rp[0]), float(rp[1]))
+        composite(canvas, rotate_sprite(blue0, blue_angle), float(bp[0]), float(bp[1]))
+        out[t] = canvas
+
+    return out
+
+
+if __name__ == "__main__":
+    frames = generate()
+    artifact_io.save(frames, CASE_DIR)
+    print("saved", frames.shape, frames.dtype,
+          "range", int(frames.min()), int(frames.max()))
