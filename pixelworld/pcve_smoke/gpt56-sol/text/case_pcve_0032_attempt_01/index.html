+"""Generate the complete counterfactual video for this case.
+
+The source uses a fixed camera.  The red sphere first hits the floor around
+source frame 13, then again between frames 32 and 33.  The intervention starts
+at frame 20, while the sphere is airborne, so frames through 32 remain factual.
+At the next impact its measured source restitution (~0.83) is multiplied by
+0.1.  That leaves only a sub-pixel hop before the sphere settles and rolls.
+
+Only local RGB/numeric operations are used.  The source sphere is sampled from
+each factual frame so its material, illumination and rotation remain intact.
+A robust temporal clean plate removes the factual sphere/rebound; a contact
+shadow sampled from a factual impact is transplanted to the new position.
+"""
+from pathlib import Path
+
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+H, W = 144, 256
+N = 96
+
+
+def sphere_centers(frames):
+    """Locate the red sphere from its strong red-over-green chroma."""
+    out = []
+    yy, xx = np.mgrid[:H, :W]
+    for frame in frames:
+        a = frame.astype(np.int16)
+        mask = ((a[..., 0] - a[..., 1] > 50) &
+                (a[..., 0] - a[..., 2] > 50) &
+                (a[..., 0] > 80) &
+                (xx >= 105) & (xx <= 180) & (yy <= 100))
+        # In this constrained ROI the ball is the only component of this size.
+        ys, xs = np.nonzero(mask)
+        # Reject isolated non-ball pixels using the dense 30x30 neighborhood
+        # around the median of the red pixels.
+        mx, my = np.median(xs), np.median(ys)
+        core = (np.abs(xs - mx) < 16) & (np.abs(ys - my) < 16)
+        out.append((xs[core].mean(), ys[core].mean()))
+    return np.asarray(out, dtype=np.float64)
+
+
+def bilinear(image, x, y):
+    """Bilinearly sample an HxWxC image at identically-shaped x/y grids."""
+    x = np.clip(x, 0.0, W - 1.001)
+    y = np.clip(y, 0.0, H - 1.001)
+    x0 = np.floor(x).astype(np.int32)
+    y0 = np.floor(y).astype(np.int32)
+    x1 = x0 + 1
+    y1 = y0 + 1
+    fx = (x - x0)[..., None]
+    fy = (y - y0)[..., None]
+    a = image[y0, x0] * (1.0 - fx) + image[y0, x1] * fx
+    b = image[y1, x0] * (1.0 - fx) + image[y1, x1] * fx
+    return a * (1.0 - fy) + b * fy
+
+
+def clean_plate(frames, centers):
+    """Estimate the static scene while rejecting the moving sphere/shadow."""
+    stack = frames.astype(np.float32)
+    yy, xx = np.mgrid[:H, :W]
+    work = stack.copy()
+    # Exclude the sphere itself.  The upper temporal percentile then rejects
+    # the (darker) moving cast shadow while retaining the textured set.
+    for t, (cx, cy) in enumerate(centers):
+        ball = (xx - cx) ** 2 + (yy - cy) ** 2 < 14.0 ** 2
+        work[t, ball] = np.nan
+    plate = np.nanpercentile(work, 70.0, axis=0)
+    # The ball never visits outside the central strip; ordinary median is less
+    # noisy there and ensures the operation is entirely local to the action.
+    median = np.median(stack, axis=0)
+    roi = (xx >= 100) & (xx <= 200) & (yy <= 120)
+    plate[~roi] = median[~roi]
+    return plate.astype(np.float32)
+
+
+def frame_plate_offsets(frames, plate):
+    """Recover tiny per-frame renderer quantization/illumination offsets."""
+    # Fixed, object-free upper corners.
+    sample = np.concatenate((frames[:, :35, :80], frames[:, :35, 190:]), axis=2)
+    p_sample = np.concatenate((plate[:35, :80], plate[:35, 190:]), axis=1)
+    return np.median(sample.astype(np.float32) - p_sample[None], axis=(1, 2))
+
+
+def target_trajectory(centers):
+    """Counterfactual center path after reducing restitution by 10x."""
+    target = centers.copy()
+
+    # Fits to the factual incoming and outgoing arcs place the second impact at
+    # index 31.58, center y=81.96 px.  Incoming speed is 7.43 px/frame and the
+    # factual rebound speed is 6.18 px/frame (e ~= 0.831).  New rebound speed:
+    collision_t = 31.5805
+    floor_y = 81.963
+    gravity_half = 0.375       # y = y0 + vy*t + 0.5*g*t^2
+    rebound_v = -0.6179        # -(0.1 * 0.831) * 7.43
+
+    # The tiny hop returns to contact after ~1.65 frames.  At sampled frame 33
+    # it is only 0.1 px above the floor; thereafter stable contact is clearer
+    # and physically equivalent at this resolution.
+    settle_t = collision_t + (-rebound_v / gravity_half)
+    for t in range(32, N):
+        dt = t - collision_t
+        if t <= settle_t:
+            target[t, 1] = floor_y + rebound_v * dt + gravity_half * dt * dt
+        else:
+            target[t, 1] = floor_y
+
+    # Tangential impulse is smaller with the reduced normal impulse.  From the
+    # source collision fits, factual vx changes 0.317 -> 0.268 px/frame.  Scale
+    # that friction change by (1+e_new)/(1+e_old), giving ~0.288 px/frame.
+    impact_x = 134.59659267 + 0.31741755 * collision_t
+    rolling_vx = 0.2884
+    for t in range(32, N):
+        target[t, 0] = impact_x + rolling_vx * (t - collision_t)
+    return target
+
+
+def edit_frame(frame, t, centers, target, plate, offsets, shadow_ref=48):
+    """Erase factual ball/shadow, then composite counterfactual shadow/ball."""
+    src = frame.astype(np.float32)
+    bg = np.clip(plate + offsets[t][None, None, :], 0, 255)
+    out = src.copy()
+    yy, xx = np.mgrid[:H, :W]
+    ocx, ocy = centers[t]
+
+    # Erase factual cast shadow.  Its source residual is darker than the robust
+    # clean plate and occupies a compact region down/right of the sphere.
+    relx, rely = xx - ocx, yy - ocy
+    local = ((relx > -31) & (relx < 39) &
+             (rely > -18) & (rely < 49))
+    radius = np.sqrt(relx * relx + rely * rely)
+    darkness = np.mean(bg - src, axis=2)
+    shadow_alpha = np.clip((darkness - 1.7) / 5.0, 0.0, 1.0)
+    shadow_alpha *= local & (radius > 11.2)
+    # Slightly soften numeric thresholding without spatial filtering.
+    shadow_alpha = shadow_alpha * shadow_alpha * (3.0 - 2.0 * shadow_alpha)
+    out = out * (1.0 - shadow_alpha[..., None]) + bg * shadow_alpha[..., None]
+
+    # Erase the factual sphere itself, including its antialiased rim.
+    erase_alpha = np.clip((13.1 - radius) / 0.9, 0.0, 1.0)
+    out = out * (1.0 - erase_alpha[..., None]) + bg * erase_alpha[..., None]
+
+    tcx, tcy = target[t]
+
+    # Add a factual near-contact cast shadow, translated as a residual relative
+    # to the sphere center.  Frame 49 is at the same floor impact height.
+    rcx, rcy = centers[shadow_ref]
+    sx = xx - tcx + rcx
+    sy = yy - tcy + rcy
+    ref_src = bilinear(frames_float[shadow_ref], sx, sy)
+    ref_bg = bilinear(plate, sx, sy)
+    srelx, srely = xx - tcx, yy - tcy
+    slocal = ((srelx > -31) & (srelx < 39) &
+              (srely > -18) & (srely < 49))
+    sradius = np.sqrt(srelx * srelx + srely * srely)
+    sdark = np.mean(ref_bg - ref_src, axis=2)
+    sa = np.clip((sdark - 1.25) / 4.0, 0.0, 1.0)
+    sa *= slocal & (sradius > 11.2)
+    sa = sa * sa * (3.0 - 2.0 * sa)
+    shadow_delta = ref_src - ref_bg
+    # Only the physically dark residual is useful; suppress isolated bright
+    # temporal quantization while preserving the shadow's RGB coloration.
+    useful = np.mean(shadow_delta, axis=2) < -1.0
+    sa *= useful
+    out = np.clip(out + shadow_delta * sa[..., None], 0, 255)
+
+    # Transplant the sphere from this same source frame, preserving material
+    # detail and temporal rotation.  Analytic edge coverage avoids a halo.
+    sx = xx - tcx + ocx
+    sy = yy - tcy + ocy
+    sphere_rgb = bilinear(frames_float[t], sx, sy)
+    tr = np.sqrt((xx - tcx) ** 2 + (yy - tcy) ** 2)
+    sphere_alpha = np.clip((12.25 - tr) / 0.75, 0.0, 1.0)
+    out = out * (1.0 - sphere_alpha[..., None]) + sphere_rgb * sphere_alpha[..., None]
+    return np.clip(np.rint(out), 0, 255).astype(np.uint8)
+
+
+if __name__ == "__main__":
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (N, H, W, 3) and source.dtype == np.uint8
+    frames_float = source.astype(np.float32)  # used by edit_frame's sampler
+
+    centers = sphere_centers(source)
+    plate = clean_plate(source, centers)
+    offsets = frame_plate_offsets(source, plate)
+    target = target_trajectory(centers)
+
+    result = source.copy()
+    # First changed sample is frame 33 (index 32), just after the next impact.
+    for t in range(32, N):
+        result[t] = edit_frame(source[t], t, centers, target, plate, offsets)
+
+    assert np.array_equal(result[:32], source[:32])
+    assert result.shape == (96, 144, 256, 3) and result.dtype == np.uint8
+    artifact_io.save(result, CASE_DIR)

\"\"\"Generate the complete counterfactual video for this case.

The source uses a fixed camera.  The red sphere first hits the floor around
source frame 13, then again between frames 32 and 33.  The intervention starts
at frame 20, while the sphere is airborne, so frames through 32 remain factual.
At the next impact its measured source restitution (~0.83) is multiplied by
0.1.  That leaves only a sub-pixel hop before the sphere settles and rolls.

Only local RGB/numeric operations are used.  The source sphere is sampled from
each factual frame so its material, illumination and rotation remain intact.
A robust temporal clean plate removes the factual sphere/rebound; a contact
shadow sampled from a factual impact is transplanted to the new position.
\"\"\"
from pathlib import Path

import numpy as np

import artifact_io


CASE_DIR = Path(__file__).resolve().parent
H, W = 144, 256
N = 96


def sphere_centers(frames):
    \"\"\"Locate the red sphere from its strong red-over-green chroma.\"\"\"
    out = []
    yy, xx = np.mgrid[:H, :W]
    for frame in frames:
        a = frame.astype(np.int16)
        mask = ((a[..., 0] - a[..., 1] > 50) &
                (a[..., 0] - a[..., 2] > 50) &
                (a[..., 0] > 80) &
                (xx >= 105) & (xx <= 180) & (yy <= 100))
        # In this constrained ROI the ball is the only component of this size.
        ys, xs = np.nonzero(mask)
        # Reject isolated non-ball pixels using the dense 30x30 neighborhood
        # around the median of the red pixels.
        mx, my = np.median(xs), np.median(ys)
        core = (np.abs(xs - mx) < 16) & (np.abs(ys - my) < 16)
        out.append((xs[core].mean(), ys[core].mean()))
    return np.asarray(out, dtype=np.float64)


def bilinear(image, x, y):
    \"\"\"Bilinearly sample an HxWxC image at identically-shaped x/y grids.\"\"\"
    x = np.clip(x, 0.0, W - 1.001)
    y = np.clip(y, 0.0, H - 1.001)
    x0 = np.floor(x).astype(np.int32)
    y0 = np.floor(y).astype(np.int32)
    x1 = x0 + 1
    y1 = y0 + 1
    fx = (x - x0)[..., None]
    fy = (y - y0)[..., None]
    a = image[y0, x0] * (1.0 - fx) + image[y0, x1] * fx
    b = image[y1, x0] * (1.0 - fx) + image[y1, x1] * fx
    return a * (1.0 - fy) + b * fy


def clean_plate(frames, centers):
    \"\"\"Estimate the static scene while rejecting the moving sphere/shadow.\"\"\"
    stack = frames.astype(np.float32)
    yy, xx = np.mgrid[:H, :W]
    work = stack.copy()
    # Exclude the sphere itself.  The upper temporal percentile then rejects
    # the (darker) moving cast shadow while retaining the textured set.
    for t, (cx, cy) in enumerate(centers):
        ball = (xx - cx) ** 2 + (yy - cy) ** 2 < 14.0 ** 2
        work[t, ball] = np.nan
    plate = np.nanpercentile(work, 70.0, axis=0)
    # The ball never visits outside the central strip; ordinary median is less
    # noisy there and ensures the operation is entirely local to the action.
    median = np.median(stack, axis=0)
    roi = (xx >= 100) & (xx <= 200) & (yy <= 120)
    plate[~roi] = median[~roi]
    return plate.astype(np.float32)


def frame_plate_offsets(frames, plate):
    \"\"\"Recover tiny per-frame renderer quantization/illumination offsets.\"\"\"
    # Fixed, object-free upper corners.
    sample = np.concatenate((frames[:, :35, :80], frames[:, :35, 190:]), axis=2)
    p_sample = np.concatenate((plate[:35, :80], plate[:35, 190:]), axis=1)
    return np.median(sample.astype(np.float32) - p_sample[None], axis=(1, 2))


def target_trajectory(centers):
    \"\"\"Counterfactual center path after reducing restitution by 10x.\"\"\"
    target = centers.copy()

    # Fits to the factual incoming and outgoing arcs place the second impact at
    # index 31.58, center y=81.96 px.  Incoming speed is 7.43 px/frame and the
    # factual rebound speed is 6.18 px/frame (e ~= 0.831).  New rebound speed:
    collision_t = 31.5805
    floor_y = 81.963
    gravity_half = 0.375       # y = y0 + vy*t + 0.5*g*t"'^2
    rebound_v = -0.6179        # -(0.1 * 0.831) * 7.43

    # The tiny hop returns to contact after ~1.65 frames.  At sampled frame 33
    # it is only 0.1 px above the floor; thereafter stable contact is clearer
    # and physically equivalent at this resolution.
    settle_t = collision_t + (-rebound_v / gravity_half)
    for t in range(32, N):
        dt = t - collision_t
        if t <= settle_t:
            target[t, 1] = floor_y + rebound_v * dt + gravity_half * dt * dt
        else:
            target[t, 1] = floor_y

    # Tangential impulse is smaller with the reduced normal impulse.  From the
    # source collision fits, factual vx changes 0.317 -> 0.268 px/frame.  Scale
    # that friction change by (1+e_new)/(1+e_old), giving ~0.288 px/frame.
    impact_x = 134.59659267 + 0.31741755 * collision_t
    rolling_vx = 0.2884
    for t in range(32, N):
        target[t, 0] = impact_x + rolling_vx * (t - collision_t)
    return target


def edit_frame(frame, t, centers, target, plate, offsets, shadow_ref=48):
    """Erase factual ball/shadow, then composite counterfactual shadow/ball."""
    src = frame.astype(np.float32)
    bg = np.clip(plate + offsets[t][None, None, :], 0, 255)
    out = src.copy()
    yy, xx = np.mgrid[:H, :W]
    ocx, ocy = centers[t]

    # Erase factual cast shadow.  Its source residual is darker than the robust
    # clean plate and occupies a compact region down/right of the sphere.
    relx, rely = xx - ocx, yy - ocy
    local = ((relx > -31) & (relx < 39) &
             (rely > -18) & (rely < 49))
    radius = np.sqrt(relx * relx + rely * rely)
    darkness = np.mean(bg - src, axis=2)
    shadow_alpha = np.clip((darkness - 1.7) / 5.0, 0.0, 1.0)
    shadow_alpha *= local & (radius > 11.2)
    # Slightly soften numeric thresholding without spatial filtering.
    shadow_alpha = shadow_alpha * shadow_alpha * (3.0 - 2.0 * shadow_alpha)
    out = out * (1.0 - shadow_alpha[..., None]) + bg * shadow_alpha[..., None]

    # Erase the factual sphere itself, including its antialiased rim.
    erase_alpha = np.clip((13.1 - radius) / 0.9, 0.0, 1.0)
    out = out * (1.0 - erase_alpha[..., None]) + bg * erase_alpha[..., None]

    tcx, tcy = target[t]

    # Add a factual near-contact cast shadow, translated as a residual relative
    # to the sphere center.  Frame 49 is at the same floor impact height.
    rcx, rcy = centers[shadow_ref]
    sx = xx - tcx + rcx
    sy = yy - tcy + rcy
    ref_src = bilinear(frames_float[shadow_ref], sx, sy)
    ref_bg = bilinear(plate, sx, sy)
    srelx, srely = xx - tcx, yy - tcy
    slocal = ((srelx > -31) & (srelx < 39) &
              (srely > -18) & (srely < 49))
    sradius = np.sqrt(srelx * srelx + srely * srely)
    sdark = np.mean(ref_bg - ref_src, axis=2)
    sa = np.clip((sdark - 1.25) / 4.0, 0.0, 1.0)
    sa *= slocal & (sradius > 11.2)
    sa = sa * sa * (3.0 - 2.0 * sa)
    shadow_delta = ref_src - ref_bg
    # Only the physically dark residual is useful; suppress isolated bright
    # temporal quantization while preserving the shadow'"'s RGB coloration.
    useful = np.mean(shadow_delta, axis=2) < -1.0
    sa *= useful
    out = np.clip(out + shadow_delta * sa[..., None], 0, 255)

    # Transplant the sphere from this same source frame, preserving material
    # detail and temporal rotation.  Analytic edge coverage avoids a halo.
    sx = xx - tcx + ocx
    sy = yy - tcy + ocy
    sphere_rgb = bilinear(frames_float[t], sx, sy)
    tr = np.sqrt((xx - tcx) ** 2 + (yy - tcy) ** 2)
    sphere_alpha = np.clip((12.25 - tr) / 0.75, 0.0, 1.0)
    out = out * (1.0 - sphere_alpha[..., None]) + sphere_rgb * sphere_alpha[..., None]
    return np.clip(np.rint(out), 0, 255).astype(np.uint8)
