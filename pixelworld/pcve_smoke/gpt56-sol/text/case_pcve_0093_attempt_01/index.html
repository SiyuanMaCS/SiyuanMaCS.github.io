+"""Generate the counterfactual video with the moving target ball removed.
+
+The shot is locked off.  The target travels diagonally across a mostly static
+background, contacts the small stationary patterned ball, and then returns.
+We remove it using clean temporal samples of the same background pixels.  The
+contacted ball is restored from its uncontaminated pre-contact frames so that
+it never changes state in the counterfactual.
+"""
+
+from pathlib import Path
+
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def track_target(source: np.ndarray, plate: np.ndarray) -> np.ndarray:
+    """Track the compact moving ball with local, plate-relative centroids."""
+    h, w = source.shape[1:3]
+    yy, xx = np.mgrid[:h, :w]
+    centers = np.zeros((len(source), 2), dtype=np.float64)
+
+    # Expected positions only disambiguate the target from the stationary ball.
+    # The final location in each frame comes from the source pixels themselves.
+    for t in range(13):
+        expected_x = 85.4 + 3.55 * t - 0.045 * t * t
+        expected_y = 94.45 - 2.13 * t + 0.046 * t * t
+        delta = np.linalg.norm(source[t].astype(np.float32) - plate, axis=2)
+        region = (xx - expected_x) ** 2 + (yy - expected_y) ** 2 < 12.0**2
+        # Do not let the nearby stationary ball pull the centroid.
+        region &= (xx - 126.5) ** 2 + (yy - 71.0) ** 2 > 6.5**2
+        weights = np.maximum(delta - 5.0, 0.0) * region
+        for _ in range(3):
+            total = weights.sum()
+            cx = float((weights * xx).sum() / total)
+            cy = float((weights * yy).sum() / total)
+            region = (xx - cx) ** 2 + (yy - cy) ** 2 < 8.0**2
+            region &= (xx - 126.5) ** 2 + (yy - 71.0) ** 2 > 6.5**2
+            weights = np.maximum(delta - 5.0, 0.0) * region
+        centers[t] = cx, cy
+
+    # From contact onward, follow the same connected moving appearance.  The
+    # target first recedes to the upper right, pauses, then returns on its path.
+    previous = np.array([124.0, 72.0])
+    for t in range(13, len(source)):
+        delta = np.max(np.abs(source[t].astype(np.float32) - plate), axis=2)
+        region = (xx - previous[0]) ** 2 + (yy - previous[1]) ** 2 < 10.5**2
+        if t >= 17:
+            region &= (xx - 126.5) ** 2 + (yy - 71.0) ** 2 > 6.0**2
+        weights = np.maximum(delta - 6.0, 0.0) * region
+        for _ in range(3):
+            total = weights.sum()
+            cx = float((weights * xx).sum() / total)
+            cy = float((weights * yy).sum() / total)
+            region = (xx - cx) ** 2 + (yy - cy) ** 2 < 8.0**2
+            if t >= 17:
+                region &= (xx - 126.5) ** 2 + (yy - 71.0) ** 2 > 6.0**2
+            weights = np.maximum(delta - 6.0, 0.0) * region
+        centers[t] = cx, cy
+        previous = centers[t]
+    return centers
+
+
+def nearest_clean_fill(source: np.ndarray, occupancy: np.ndarray) -> np.ndarray:
+    """For every occupied pixel, take the nearest clean temporal observation."""
+    n, h, w = occupancy.shape
+    fill = source.copy()
+    times = np.arange(n)
+    for t in range(n):
+        ys, xs = np.where(occupancy[t])
+        # The edited footprint is small, so a per-pixel temporal lookup is both
+        # inexpensive and avoids flattening the grass's fine spatial texture.
+        for y, x in zip(ys, xs):
+            clean_times = times[~occupancy[:, y, x]]
+            nearest = clean_times[np.argmin(np.abs(clean_times - t))]
+            fill[t, y, x] = source[nearest, y, x]
+    return fill
+
+
+def make_frames() -> np.ndarray:
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3) and source.dtype == np.uint8
+    n, h, w = source.shape[:3]
+    yy, xx = np.mgrid[:h, :w]
+
+    # A temporal median is used for tracking only.  Since the ball occupies any
+    # one background pixel briefly, this is a reliable clean reference plate.
+    plate = np.median(source.astype(np.float32), axis=0)
+    centers = track_target(source, plate)
+
+    # The target appears larger near the camera, then settles to a six-pixel
+    # footprint in the distance.  Extra margin includes antialiasing and shadow.
+    radii = 9.0 - 3.0 * np.minimum(np.arange(n), 24) / 24.0
+    distance = np.empty((n, h, w), dtype=np.float32)
+    occupancy = np.empty((n, h, w), dtype=bool)
+    for t, ((cx, cy), radius) in enumerate(zip(centers, radii)):
+        distance[t] = np.sqrt((xx - cx) ** 2 + (yy - cy) ** 2)
+        # A larger support mask guarantees that fill samples are truly clean.
+        occupancy[t] = distance[t] <= radius + 1.5
+
+    clean = nearest_clean_fill(source, occupancy)
+    result = source.astype(np.float32)
+    # Feather only the outer margin; the complete original ball is in the solid
+    # interior, while the source remains pixel-identical away from the edit.
+    alpha = np.clip((radii[:, None, None] + 0.75 - distance) / 1.5, 0.0, 1.0)
+    result = result * (1.0 - alpha[..., None]) + clean * alpha[..., None]
+
+    # Restore the stationary ball to its state before target contact.  Choose,
+    # for each output frame, the pre-contact source frame whose nearby grass is
+    # the closest match, minimizing even one-level codec seams at the boundary.
+    bx, by = 126.5, 71.0
+    ball_dist = np.sqrt((xx - bx) ** 2 + (yy - by) ** 2)
+    ball_alpha = np.clip((10.25 - ball_dist) / 1.25, 0.0, 1.0)[..., None]
+    annulus = (ball_dist >= 11.0) & (ball_dist <= 18.0)
+    candidates = range(7)  # target is well clear of this patch in frames 0..6
+    for t in range(n):
+        scores = [
+            np.mean(
+                np.abs(
+                    source[t, annulus].astype(np.int16)
+                    - source[c, annulus].astype(np.int16)
+                )
+            )
+            for c in candidates
+        ]
+        chosen = int(np.argmin(scores))
+        result[t] = (
+            result[t] * (1.0 - ball_alpha)
+            + source[chosen].astype(np.float32) * ball_alpha
+        )
+
+    return np.clip(np.rint(result), 0, 255).astype(np.uint8)
+
+
+if __name__ == "__main__":
+    frames = make_frames()
+    artifact_io.save(frames, CASE_DIR)
+    print(f"saved {frames.shape[0]} frames: {frames.shape}, {frames.dtype}")
