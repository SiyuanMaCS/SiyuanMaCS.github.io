+import numpy as np
+from PIL import Image, ImageFilter
+from scipy import ndimage
+import artifact_io

+CASE_DIR = Path(__file__).resolve().parent
+N, H, W = 96, 144, 256

+def make_clean_backgrounds(src):
+    """Remove the factual dice while retaining the source's subtle temporal texture."""
+    # Late frames are clean above the resting dice; early frames are clean near the floor.
+    clean = np.median(src[32:72], axis=0).astype(np.uint8)
+    clean[54:106, 108:148] = np.median(src[0:4, 54:106, 108:148], axis=0).astype(np.uint8)
+    backgrounds = src.copy()
+    # Restrict replacement to the narrow tube traversed by the original dice and softly
+    # feather its edge. This leaves the entire unaffected scene on its factual timeline.
+    mask = np.zeros((H, W), np.float32)
+    mask[0:104, 110:146] = 1.0
+    mask = ndimage.gaussian_filter(mask, 1.4)[..., None]
+    return np.clip(src * (1.0 - mask) + clean[None] * mask, 0, 255).astype(np.uint8), clean

+def extract_sprite(frame, background, box, kind):
+    """Extract one die (including its pips) with an antialiased silhouette."""
+    x0, y0, x1, y1 = box
+    patch = frame[y0:y1, x0:x1].copy()
+    ref = background[y0:y1, x0:x1]
+    mx, mn = patch.max(2), patch.min(2)
+    if kind == "white":
+        core = (mn > 145) & ((mx - mn) < 78)
+    else:
+        core = (patch[..., 0] > 78) & (patch[..., 0] > 1.22 * patch[..., 1]) & (patch[..., 0] > 1.15 * patch[..., 2])
+    # Keep only the component closest to the patch center, then close over dark pips.
+    labels, count = ndimage.label(core)
+    cy, cx = (np.array(core.shape) - 1) / 2
+    best = None
+    for i in range(1, count + 1):
+        yy, xx = np.where(labels == i)
+        if len(xx) < 4:
+            continue
+        score = len(xx) - 0.8 * ((xx.mean() - cx) ** 2 + (yy.mean() - cy) ** 2)
+        if best is None or score > best[0]:
+            best = (score, i)
+    shape = labels == best[1]
+    shape = ndimage.binary_closing(shape, iterations=2)
+    shape = ndimage.binary_fill_holes(shape)
+    shape = ndimage.binary_dilation(shape, iterations=1)
+    delta = np.max(np.abs(patch.astype(np.int16) - ref.astype(np.int16)), axis=2)
+    alpha = np.clip((delta.astype(np.float32) - 2.0) / 15.0, 0, 1)
+    alpha *= ndimage.gaussian_filter(shape.astype(np.float32), 0.55)
+    alpha[shape & (delta > 22)] = 1.0
+    rgba = np.dstack((patch, np.round(alpha * 255).astype(np.uint8)))
+    return Image.fromarray(rgba, "RGBA")

+def simulate():
+    """Substepped equal-mass rigid-body approximation in image coordinates."""
+    pos = np.array([[132.1, 8.5], [122.7, 20.8]], np.float64)  # white, red
+    # The intervention applies at frame 1: white's measured ~5 px/frame becomes 4x.
+    vel = np.array([[0.0, 20.0], [0.0, 8.45]], np.float64)
+    radius = np.array([7.05, 7.05])
+    floor_y = 95.5
+    gravity = 1.72
+    floor_restitution = 0.48
+    collision_restitution = 0.40
+    out_p = np.empty((N, 2, 2), np.float64)
+    out_v = np.empty_like(out_p)
+    angles = np.empty((N, 2), np.float64)
+    angle = np.array([4.0, -2.0])
+    spin = np.array([23.0, -12.0])
+    substeps = 24
+
+    for frame in range(N):
+        out_p[frame], out_v[frame], angles[frame] = pos, vel, angle
+        for _ in range(substeps):
+            dt = 1.0 / substeps
+            vel[:, 1] += gravity * dt
+            pos += vel * dt
+            angle += spin * dt
+
+            # Die-die collision: normal impulse plus tangential friction/spin transfer.
+            delta = pos[0] - pos[1]
+            dist = float(np.linalg.norm(delta))
+            target = radius.sum()
+            if dist < target:
+                normal = delta / max(dist, 1e-8)
+                overlap = target - dist
+                pos[0] += normal * overlap * 0.5
+                pos[1] -= normal * overlap * 0.5
+                rel = vel[0] - vel[1]
+                vn = float(np.dot(rel, normal))
+                if vn < 0:
+                    impulse = -(1.0 + collision_restitution) * vn / 2.0
+                    vel[0] += impulse * normal
+                    vel[1] -= impulse * normal
+                    tangent = np.array([-normal[1], normal[0]])
+                    vt = float(np.dot(rel, tangent))
+                    jt = np.clip(-vt / 2.0, -0.24 * impulse, 0.24 * impulse)
+                    vel[0] += jt * tangent
+                    vel[1] -= jt * tangent
+                    spin += np.array([-1.0, 1.0]) * jt * 2.5
+
+            # Contact with the tabletop. Small impacts are put to sleep to avoid jitter.
+            for i in range(2):
+                bottom = pos[i, 1] + radius[i]
+                if bottom > floor_y:
+                    pos[i, 1] = floor_y - radius[i]
+                    if vel[i, 1] > 0:
+                        vel[i, 1] *= -floor_restitution
+                        spin[i] += vel[i, 0] * 0.7
+                    vel[i, 0] *= 0.82
+                    spin[i] *= 0.82
+                if pos[i, 1] > floor_y - radius[i] - 0.03 and abs(vel[i, 1]) < 0.55:
+                    vel[i, 1] = 0.0
+                    vel[i, 0] *= 0.90
+                    spin[i] *= 0.88
+            # Very light air drag; most slowing comes from visible bounces/contact.
+            vel *= 0.9995 ** dt
+    return out_p, out_v, angles
+

+def rotated_sprite(sprite, angle):
+    # expand=True preserves every corner while bicubic filtering yields clean subpixel edges.
+    return sprite.rotate(float(angle), resample=Image.Resampling.BICUBIC, expand=True)
+

+def composite_rgba(base, overlay, center):
+    x = int(round(center[0] - overlay.width / 2))
+    y = int(round(center[1] - overlay.height / 2))
+    base.alpha_composite(overlay, (x, y))
+

+def render():
+    src = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    backgrounds, clean = make_clean_backgrounds(src)
+    # Boxes are padded by 2-3 pixels so their source antialiasing is retained.
+    white = extract_sprite(src[0], clean, (123, 0, 142, 19), "white")
+    red = extract_sprite(src[0], clean, (114, 12, 132, 31), "red")
+    pos, vel, angles = simulate()
+    frames = np.empty((N, H, W, 3), np.uint8)
+
+    for t in range(N):
+        canvas = Image.fromarray(backgrounds[t], "RGB").convert("RGBA")
+        # Contact/approach shadows are generated in scene coordinates, behind both dice.
+        shadow_layer = Image.new("RGBA", (W, H), (0, 0, 0, 0))
+        for i in range(2):
+            height = max(0.0, 88.45 - pos[t, i, 1])
+            opacity = int(48 * np.exp(-height / 18.0))
+            if opacity > 1:
+                sw = 13.0 + min(height, 35.0) * 0.16
+                sh = 2.2 + min(height, 35.0) * 0.035
+                blob = Image.new("RGBA", (int(sw + 10), int(sh + 10)), (0, 0, 0, 0))
+                yy, xx = np.ogrid[:blob.height, :blob.width]
+                cx, cy = (blob.width - 1) / 2, (blob.height - 1) / 2
+                q = ((xx - cx) / (sw / 2)) ** 2 + ((yy - cy) / (sh / 2)) ** 2
+                a = np.where(q <= 1, opacity * (1 - q) ** 1.5, 0).astype(np.uint8)
+                arr = np.zeros((blob.height, blob.width, 4), np.uint8)
+                arr[..., 3] = a
+                blob = Image.fromarray(arr, "RGBA").filter(ImageFilter.GaussianBlur(1.0))
+                composite_rgba(shadow_layer, blob, (pos[t, i, 0] + 2.0, 94.0))
+        canvas = Image.alpha_composite(canvas, shadow_layer)
+
+        # Back-to-front ordering is determined by image-space depth (vertical center).
+        for i in np.argsort(pos[t, :, 1]):
+            obj = white if i == 0 else red
+            # Once almost at rest, damp the displayed rotation along with the simulation.
+            rot = rotated_sprite(obj, angles[t, i])
+            composite_rgba(canvas, rot, pos[t, i])
+        frames[t] = np.asarray(canvas.convert("RGB"), dtype=np.uint8)
+    return frames, pos, vel
+
+
+def main():
+    frames, pos, vel = render()
+    assert frames.shape == (96, 144, 256, 3) and frames.dtype == np.uint8
+    artifact_io.save(frames, CASE_DIR)
+    # Small numeric diagnostics keep the counterfactual reproducible and auditable.
+    np.savez_compressed(CASE_DIR / "trajectory_debug.npz", positions=pos, velocities=vel)
+    print("saved", frames.shape, frames.dtype)
+    print("white trajectory keyframes:", np.round(pos[[0, 1, 2, 3, 4, 8, 16, 32, 95], 0], 2).tolist())
+    print("red trajectory keyframes:", np.round(pos[[0, 1, 2, 3, 4, 8, 16, 32, 95], 1], 2).tolist())
+
+
+if __name__ == "__main__":
+    main()

def make_clean_backgrounds(src):
    \"\"\"Remove the factual dice while retaining the source's subtle temporal texture.\"\"\"
    # Late frames are clean above the resting dice; early frames are clean near the floor.
    clean = np.median(src[32:72], axis=0).astype(np.uint8)
    clean[54:106, 108:148] = np.median(src[0:4, 54:106, 108:148], axis=0).astype(np.uint8)
    backgrounds = src.copy()
    # Restrict replacement to the narrow tube traversed by the original dice and softly
    # feather its edge. This leaves the entire unaffected scene on its factual timeline.
    mask = np.zeros((H, W), np.float32)
    mask[0:104, 110:146] = 1.0
    mask = ndimage.gaussian_filter(mask, 1.4)[..., None]
    return np.clip(src * (1.0 - mask) + clean[None] * mask, 0, 255).astype(np.uint8), clean


def extract_sprite(frame, background, box, kind):
    \"\"\"Extract one die (including its pips) with an antialiased silhouette.\"\"\"
    x0, y0, x1, y1 = box
    patch = frame[y0:y1, x0:x1].copy()
    ref = background[y0:y1, x0:x1]
    mx, mn = patch.max(2), patch.min(2)
    if kind == \"white\":
        core = (mn > 145) & ((mx - mn) < 78)
    else:
        core = (patch[..., 0] > 78) & (patch[..., 0] > 1.22 * patch[..., 1]) & (patch[..., 0] > 1.15 * patch[..., 2])
    # Keep only the component closest to the patch center, then close over dark pips.
    labels, count = ndimage.label(core)
    cy, cx = (np.array(core.shape) - 1) / 2
    best = None
    for i in range(1, count + 1):
        yy, xx = np.where(labels == i)
        if len(xx) < 4:
            continue
        score = len(xx) - 0.8 * ((xx.mean() - cx) ** 2 + (yy.mean() - cy) ** 2)
        if best is None or score > best[0]:
            best = (score, i)
    shape = labels == best[1]
    shape = ndimage.binary_closing(shape, iterations=2)
    shape = ndimage.binary_fill_holes(shape)
    shape = ndimage.binary_dilation(shape, iterations=1)
    delta = np.max(np.abs(patch.astype(np.int16) - ref.astype(np.int16)), axis=2)
    alpha = np.clip((delta.astype(np.float32) - 2.0) / 15.0, 0, 1)
    alpha *= ndimage.gaussian_filter(shape.astype(np.float32), 0.55)
    alpha[shape & (delta > 22)] = 1.0
    rgba = np.dstack((patch, np.round(alpha * 255).astype(np.uint8)))
    return Image.fromarray(rgba, \"RGBA\")


def simulate():
    \"\"\"Substepped equal-mass rigid-body approximation in image coordinates.\"\"\"
    pos = np.array([[132.1, 8.5], [122.7, 20.8]], np.float64)  # white, red
    # The intervention applies at frame 1: white's measured ~5 px/frame becomes 4x.
    vel = np.array([[0.0, 20.0], [0.0, 8.45]], np.float64)
    radius = np.array([7.05, 7.05])
    floor_y = 95.5
    gravity = 1.72
    floor_restitution = 0.48
    collision_restitution = 0.40
    out_p = np.empty((N, 2, 2), np.float64)
    out_v = np.empty_like(out_p)
    angles = np.empty((N, 2), np.float64)
    angle = np.array([4.0, -2.0])
    spin = np.array([23.0, -12.0])
    substeps = 24

    for frame in range(N):
        out_p[frame], out_v[frame], angles[frame] = pos, vel, angle
        for _ in range(substeps):
            dt = 1.0 / substeps
            vel[:, 1] += gravity * dt
            pos += vel * dt
            angle += spin * dt

            # Die-die collision: normal impulse plus tangential friction/spin transfer.
            delta = pos[0] - pos[1]
            dist = float(np.linalg.norm(delta))
            target = radius.sum()
            if dist < target:
                normal = delta / max(dist, 1e-8)
                overlap = target - dist
                pos[0] += normal * overlap * 0.5
                pos[1] -= normal * overlap * 0.5
                rel = vel[0] - vel[1]
                vn = float(np.dot(rel, normal))
                if vn < 0:
                    impulse = -(1.0 + collision_restitution) * vn / 2.0
                    vel[0] += impulse * normal
                    vel[1] -= impulse * normal
                    tangent = np.array([-normal[1], normal[0]])
                    vt = float(np.dot(rel, tangent))
                    jt = np.clip(-vt / 2.0, -0.24 * impulse, 0.24 * impulse)
                    vel[0] += jt * tangent
                    vel[1] -= jt * tangent
                    spin += np.array([-1.0, 1.0]) * jt * 2.5

            # Contact with the tabletop. Small impacts are put to sleep to avoid jitter.
            for i in range(2):
                bottom = pos[i, 1] + radius[i]
                if bottom > floor_y:
                    pos[i, 1] = floor_y - radius[i]
                    if vel[i, 1] > 0:
                        vel[i, 1] *= -floor_restitution
                        spin[i] += vel[i, 0] * 0.7
                    vel[i, 0] *= 0.82
                    spin[i] *= 0.82
                if pos[i, 1] > floor_y - radius[i] - 0.03 and abs(vel[i, 1]) < 0.55:
                    vel[i, 1] = 0.0
                    vel[i, 0] *= 0.90
                    spin[i] *= 0.88
            # Very light air drag; most slowing comes from visible bounces/contact.
            vel *= 0.9995 ** dt
    return out_p, out_v, angles


def rotated_sprite(sprite, angle):
    # expand=True preserves every corner while bicubic filtering yields clean subpixel edges.
    return sprite.rotate(float(angle), resample=Image.Resampling.BICUBIC, expand=True)


def composite_rgba(base, overlay, center):
    x = int(round(center[0] - overlay.width / 2))
    y = int(round(center[1] - overlay.height / 2))
    base.alpha_composite(overlay, (x, y))


def render():
    src = np.load(CASE_DIR / \"source_pixels.npz\")[\"rgb\"]
    backgrounds, clean = make_clean_backgrounds(src)
    # Boxes are padded by 2-3 pixels so their source antialiasing is retained.
    white = extract_sprite(src[0], clean, (123, 0, 142, 19), \"white\")
    red = extract_sprite(src[0], clean, (114, 12, 132, 31), \"red\")
    pos, vel, angles = simulate()
    frames = np.empty((N, H, W, 3), np.uint8)

    for t in range(N):
        canvas = Image.fromarray(backgrounds[t], \"RGB\").convert(\"RGBA\")
        # Contact/approach shadows are generated in scene coordinates, behind both dice.
        shadow_layer = Image.new(\"RGBA\", (W, H), (0, 0, 0, 0))
        for i in range(2):
            height = max(0.0, 88.45 - pos[t, i, 1])
            opacity = int(48 * np.exp(-height / 18.0))
            if opacity > 1:
                sw = 13.0 + min(height, 35.0) * 0.16
                sh = 2.2 + min(height, 35.0) * 0.035
                blob = Image.new(\"RGBA\", (int(sw + 10), int(sh + 10)), (0, 0, 0, 0))
                yy, xx = np.ogrid[:blob.height, :blob.width]
                cx, cy = (blob.width - 1) / 2, (blob.height - 1) / 2
                q = ((xx - cx) / (sw / 2)) ** 2 + ((yy - cy) / (sh / 2)) ** 2
                a = np.where(q <= 1, opacity * (1 - q) ** 1.5, 0).astype(np.uint8)

            rot = rotated_sprite(obj, angles[t, i])
            composite_rgba(canvas, rot, pos[t, i])
        frames[t] = np.asarray(canvas.convert(\"RGB\"), dtype=np.uint8)
