+from pathlib import Path
+import numpy as np
+from scipy import ndimage as ndi
+import artifact_io
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def smooth_alpha(mask, sigma=0.7):
+    a = ndi.gaussian_filter(mask.astype(np.float32), sigma)
+    return np.clip(a, 0.0, 1.0)
+
+
+def composite(dst, rgb, alpha, x0, y0):
+    """Alpha composite a possibly partly off-screen premultiplied sprite."""
+    h, w = alpha.shape
+    xa, ya = max(0, x0), max(0, y0)
+    xb, yb = min(dst.shape[1], x0+w), min(dst.shape[0], y0+h)
+    if xa >= xb or ya >= yb:
+        return
+    sx, sy = xa-x0, ya-y0
+    aa = alpha[sy:sy+yb-ya, sx:sx+xb-xa, None]
+    patch = dst[ya:yb, xa:xb].astype(np.float32)
+    fg = rgb[sy:sy+yb-ya, sx:sx+xb-xa].astype(np.float32)
+    dst[ya:yb, xa:xb] = np.clip(patch*(1-aa) + fg*aa, 0, 255).astype(np.uint8)
+
+
+def clean_plate(src):
+    # Before either die arrives, frame 0 supplies the lower table.  A late
+    # frame supplies the upper half after the falling dice have passed it.
+    plate = src[35].astype(np.float32).copy()
+    plate[66:] = src[0, 66:]
+    # Hide the join in an area known to contain no die in either donor.
+    for y in range(62, 70):
+        q = (y-62)/7.0
+        plate[y] = (1-q)*src[35, y] + q*src[0, y]
+    return plate
+
+
+def source_white_center(t):
+    # Sub-pixel centers measured from the white/neutral connected component.
+    key = np.array([
+        [0,132,7], [1,132,13], [2,132,20], [3,132,29],
+        [4,132,40], [5,132,52], [6,132,65], [7,132,78],
+        [8,132,75], [9,132,69], [10,132,65], [11,132,63],
+        [12,132,64], [13,132,67], [14,132,72], [15,132,80],
+        [16,132,78], [17,132,75], [18,132,74], [19,132,75],
+        [20,132,78], [21,132,81], [22,132,79], [24,132,77],
+        [28,132,77], [95,132,77]
+    ], dtype=np.float32)
+    return (float(np.interp(t, key[:,0], key[:,1])),
+            float(np.interp(t, key[:,0], key[:,2])))
+
+
+def make_white_sprite(src, plate):
+    # Frame 4 has the white die fully separated from the red die.
+    x0, y0, x1, y1 = 120, 28, 144, 52
+    rgb = src[4, y0:y1, x0:x1].copy()
+    bg = plate[y0:y1, x0:x1]
+    delta = np.sqrt(np.mean((rgb.astype(np.float32)-bg)**2, axis=2))
+    # Difference segmentation, constrained to the compact die component.
+    m = delta > 7.0
+    m = ndi.binary_closing(m, iterations=1)
+    labels, n = ndi.label(m)
+    cy, cx = 12, 12
+    lab = labels[cy, cx]
+    if lab == 0:
+        sizes = ndi.sum(m, labels, range(1,n+1)); lab = int(np.argmax(sizes))+1
+    m = labels == lab
+    m = ndi.binary_fill_holes(ndi.binary_dilation(m, iterations=1))
+    # A rounded support clips any stray changing background pixels.
+    yy, xx = np.mgrid[:24,:24]
+    support = ((xx-12)/11.5)**4 + ((yy-12)/11.5)**4 < 1.15
+    m &= support
+    return rgb, smooth_alpha(m, 0.6)
+
+
+def erase_factual_white(frame, source_frame, plate, t):
+    cx, cy = source_white_center(t)
+    yy, xx = np.mgrid[:144,:256]
+    box = (np.abs(xx-cx) <= 13) & (np.abs(yy-cy) <= 13)
+    s = source_frame.astype(np.float32)
+    p = plate.astype(np.float32)
+    delta = np.sqrt(np.mean((s-p)**2, axis=2))
+    mx, mn = s.max(axis=2), s.min(axis=2)
+    # Do not remove saturated red pixels belonging to the neighboring red die.
+    red = (s[:,:,0] > 105) & (s[:,:,0] > 1.30*s[:,:,1]) & (s[:,:,0] > 1.15*s[:,:,2])
+    changed = box & (delta > 4.0) & ~red
+    # Include antialiased boundaries, and compact black pips, but stay local.
+    changed = ndi.binary_dilation(changed, iterations=2) & box & ~ndi.binary_dilation(red, iterations=1)
+    alpha = smooth_alpha(changed, 0.8)
+    # Match slow global exposure drift before inserting the clean plate.
+    sample = ((xx < 105) | (xx > 150)) & (yy < 105)
+    offset = np.median(s[sample]-p[sample], axis=0)
+    fill = np.clip(p + offset, 0, 255)
+    a = alpha[...,None]
+    frame[:] = np.clip(frame.astype(np.float32)*(1-a) + fill*a, 0,255).astype(np.uint8)
+
+
+def modified_path():
+    # The measured first contact is frame 7. Gravity is estimated from the
+    # source fall (~1.25 px/frame^2). Original e ~= .58, hence e'=.754.
+    y = np.full(96, np.nan, np.float32)
+    y[:8] = [7,13,20,29,40,52,65,78]
+    pos, vel, floor = 78.0, -9.70, 78.0
+    gravity, restitution = 1.25, 0.754
+    for t in range(8,96):
+        # Substeps make contact times and rebound heights smooth.
+        for _ in range(32):
+            vel += gravity/32.0
+            pos += vel/32.0
+            if pos > floor and vel > 0:
+                pos = floor-(pos-floor)
+                vel = -restitution*vel
+                # Once imperceptible, hold the die still without chatter.
+                if abs(vel) < 0.48:
+                    vel, pos = 0.0, floor
+        y[t] = pos
+    return y
+
+
+def rotate_sprite(rgb, alpha, angle):
+    # Rotate premultiplied RGB to avoid dark/bright edge halos.
+    premul = rgb.astype(np.float32)*alpha[...,None]
+    rp = ndi.rotate(premul, angle, reshape=True, order=1, mode='constant', cval=0.0)
+    ra = ndi.rotate(alpha, angle, reshape=True, order=1, mode='constant', cval=0.0)
+    out = np.zeros_like(rp)
+    np.divide(rp, np.maximum(ra[...,None], 1e-5), out=out)
+    return np.clip(out,0,255).astype(np.uint8), np.clip(ra,0,1)
+
+
+def generate():
+    src = np.load(CASE_DIR/'source_pixels.npz')['rgb']
+    assert src.shape == (96,144,256,3) and src.dtype == np.uint8
+    out = src.copy()
+    plate = clean_plate(src)
+    sprite_rgb, sprite_a = make_white_sprite(src, plate)
+    path = modified_path()
+
+    # Frames through contact remain factually identical. Restitution can only
+    # alter motion after the impulse has occurred.
+    for t in range(8,96):
+        erase_factual_white(out[t], src[t], plate, t)
+
+        # Soft contact shadow: strongest and tightest near the table, weaker
+        # as the die rises. It is kept behind the die and red neighbor.
+        height = max(0.0, 78.0-float(path[t]))
+        shadow_strength = 0.19*np.exp(-height/15.0)
+        sy, sx = np.mgrid[:144,:256]
+        shadow = np.exp(-(((sx-133.0)/(8.0+height*.07))**2 + ((sy-89.0)/2.1)**2))*shadow_strength
+        # Avoid tinting the saturated red die.
+        sf = src[t].astype(np.float32)
+        red = (sf[:,:,0]>105)&(sf[:,:,0]>1.3*sf[:,:,1])&(sf[:,:,0]>1.15*sf[:,:,2])
+        shadow[ndi.binary_dilation(red,iterations=1)] = 0
+        out[t] = np.clip(out[t].astype(np.float32)*(1-shadow[...,None]),0,255).astype(np.uint8)
+
+        # Preserve the source spin rate; impacts reverse a small fraction of
+        # visible roll naturally through continuous rotation.
+        angle = -18.0*(t-4)
+        rr, aa = rotate_sprite(sprite_rgb, sprite_a, angle)
+        cx, cy = 132.0, float(path[t])
+        x0 = int(round(cx-aa.shape[1]/2))
+        y0 = int(round(cy-aa.shape[0]/2))
+        composite(out[t], rr, aa, x0, y0)
+    return out
+
+
+if __name__ == '__main__':
+    frames = generate()
+    artifact_io.save(frames, CASE_DIR)
+    print('saved', frames.shape, frames.dtype,
+          'changed_pixels=', int(np.any(frames != np.load(CASE_DIR/'source_pixels.npz')['rgb'], axis=3).sum()))

from pathlib import Path
import numpy as np
from scipy import ndimage as ndi
import artifact_io

CASE_DIR = Path(__file__).resolve().parent


def smooth_alpha(mask, sigma=0.7):
    a = ndi.gaussian_filter(mask.astype(np.float32), sigma)
    return np.clip(a, 0.0, 1.0)


def composite(dst, rgb, alpha, x0, y0):
    \"\"\"Alpha composite a possibly partly off-screen premultiplied sprite.\"\"\"
    h, w = alpha.shape
    xa, ya = max(0, x0), max(0, y0)
    xb, yb = min(dst.shape[1], x0+w), min(dst.shape[0], y0+h)
    if xa >= xb or ya >= yb:
        return
    sx, sy = xa-x0, ya-y0
    aa = alpha[sy:sy+yb-ya, sx:sx+xb-xa, None]
    patch = dst[ya:yb, xa:xb].astype(np.float32)
    fg = rgb[sy:sy+yb-ya, sx:sx+xb-xa].astype(np.float32)
    dst[ya:yb, xa:xb] = np.clip(patch*(1-aa) + fg*aa, 0, 255).astype(np.uint8)


def clean_plate(src):
    # Before either die arrives, frame 0 supplies the lower table.  A late
    # frame supplies the upper half after the falling dice have passed it.
    plate = src[35].astype(np.float32).copy()
    plate[66:] = src[0, 66:]
    # Hide the join in an area known to contain no die in either donor.
    for y in range(62, 70):
        q = (y-62)/7.0
        plate[y] = (1-q)*src[35, y] + q*src[0, y]
    return plate


def source_white_center(t):
    # Sub-pixel centers measured from the white/neutral connected component.
    key = np.array([
        [0,132,7], [1,132,13], [2,132,20], [3,132,29],
        [4,132,40], [5,132,52], [6,132,65], [7,132,78],
        [8,132,75], [9,132,69], [10,132,65], [11,132,63],
        [12,132,64], [13,132,67], [14,132,72], [15,132,80],
        [16,132,78], [17,132,75], [18,132,74], [19,132,75],
        [20,132,78], [21,132,81], [22,132,79], [24,132,77],
        [28,132,77], [95,132,77]
    ], dtype=np.float32)
    return (float(np.interp(t, key[:,0], key[:,1])),
            float(np.interp(t, key[:,0], key[:,2])))


def make_white_sprite(src, plate):
    # Frame 4 has the white die fully separated from the red die.
    x0, y0, x1, y1 = 120, 28, 144, 52
    rgb = src[4, y0:y1, x0:x1].copy()
    bg = plate[y0:y1, x0:x1]
    delta = np.sqrt(np.mean((rgb.astype(np.float32)-bg)**2, axis=2))
    # Difference segmentation, constrained to the compact die component.
    m = delta > 7.0
    m = ndi.binary_closing(m, iterations=1)
    labels, n = ndi.label(m)
    cy, cx = 12, 12
    lab = labels[cy, cx]
    if lab == 0:
        sizes = ndi.sum(m, labels, range(1,n+1)); lab = int(np.argmax(sizes))+1
    m = labels == lab
    m = ndi.binary_fill_holes(ndi.binary_dilation(m, iterations=1))
    # A rounded support clips any stray changing background pixels.
    yy, xx = np.mgrid[:24,:24]
    support = ((xx-12)/11.5)**4 + ((yy-12)/11.5)**4 < 1.15
    m &= support
    return rgb, smooth_alpha(m, 0.6)


def erase_factual_white(frame, source_frame, plate, t):
    cx, cy = source_white_center(t)
    yy, xx = np.mgrid[:144,:256]
    box = (np.abs(xx-cx) <= 13) & (np.abs(yy-cy) <= 13)
    s = source_frame.astype(np.float32)
    p = plate.astype(np.float32)
    delta = np.sqrt(np.mean((s-p)**2, axis=2))
    mx, mn = s.max(axis=2), s.min(axis=2)
    # Do not remove saturated red pixels belonging to the neighboring red die.
    red = (s[:,:,0] > 105) & (s[:,:,0] > 1.30*s[:,:,1]) & (s[:,:,0] > 1.15*s[:,:,2])
    changed = box & (delta > 4.0) & ~red
    # Include antialiased boundaries, and compact black pips, but stay local.
    changed = ndi.binary_dilation(changed, iterations=2) & box & ~ndi.binary_dilation(red, iterations=1)
    alpha = smooth_alpha(changed, 0.8)
    # Match slow global exposure drift before inserting the clean plate.
    sample = ((xx < 105) | (xx > 150)) & (yy < 105)
    offset = np.median(s[sample]-p[sample], axis=0)
    fill = np.clip(p + offset, 0, 255)
    a = alpha[...,None]
    frame[:] = np.clip(frame.astype(np.float32)*(1-a) + fill*a, 0,255).astype(np.uint8)


def modified_path():
    # The measured first contact is frame 7. Gravity is estimated from the
    # source fall (~1.25 px/frame"'^2). Original e ~= .58, hence e'"'=.754.
    y = np.full(96, np.nan, np.float32)
    y[:8] = [7,13,20,29,40,52,65,78]
    pos, vel, floor = 78.0, -9.70, 78.0
    gravity, restitution = 1.25, 0.754
    for t in range(8,96):
        # Substeps make contact times and rebound heights smooth.
        for _ in range(32):
            vel += gravity/32.0
            pos += vel/32.0
            if pos > floor and vel > 0:
                pos = floor-(pos-floor)
                vel = -restitution*vel
                # Once imperceptible, hold the die still without chatter.
                if abs(vel) < 0.48:
                    vel, pos = 0.0, floor
        y[t] = pos
    return y


def rotate_sprite(rgb, alpha, angle):
    # Rotate premultiplied RGB to avoid dark/bright edge halos.
    premul = rgb.astype(np.float32)*alpha[...,None]
    rp = ndi.rotate(premul, angle, reshape=True, order=1, mode='constant', cval=0.0)
    ra = ndi.rotate(alpha, angle, reshape=True, order=1, mode='constant', cval=0.0)
    out = np.zeros_like(rp)
    np.divide(rp, np.maximum(ra[...,None], 1e-5), out=out)
    return np.clip(out,0,255).astype(np.uint8), np.clip(ra,0,1)


def generate():
    src = np.load(CASE_DIR/'source_pixels.npz')['rgb']
    assert src.shape == (96,144,256,3) and src.dtype == np.uint8
    out = src.copy()
    plate = clean_plate(src)
    sprite_rgb, sprite_a = make_white_sprite(src, plate)
    path = modified_path()

    # Frames through contact remain factually identical. Restitution can only
    # alter motion after the impulse has occurred.
    for t in range(8,96):
        erase_factual_white(out[t], src[t], plate, t)

        # Soft contact shadow: strongest and tightest near the table, weaker
        # as the die rises. It is kept behind the die and red neighbor.
        height = max(0.0, 78.0-float(path[t]))
        shadow_strength = 0.19*np.exp(-height/15.0)
        sy, sx = np.mgrid[:144,:256]
        shadow = np.exp(-(((sx-133.0)/(8.0+height*.07))**2 + ((sy-89.0)/2.1)**2))*shadow_strength
        # Avoid tinting the saturated red die.
        sf = src[t].astype(np.float32)
        red = (sf[:,:,0]>105)&(sf[:,:,0]>1.3*sf[:,:,1])&(sf[:,:,0]>1.15*sf[:,:,2])
        shadow[ndi.binary_dilation(red,iterations=1)] = 0
        out[t] = np.clip(out[t].astype(np.float32)*(1-shadow[...,None]),0,255).astype(np.uint8)

        # Preserve the source spin rate; impacts reverse a small fraction of
        # visible roll naturally through continuous rotation.
        angle = -18.0*(t-4)
        rr, aa = rotate_sprite(sprite_rgb, sprite_a, angle)
        cx, cy = 132.0, float(path[t])
        x0 = int(round(cx-aa.shape[1]/2))
        y0 = int(round(cy-aa.shape[0]/2))
        composite(out[t], rr, aa, x0, y0)
    return out


if __name__ == '__main__':

    print('saved', frames.shape, frames.dtype,
