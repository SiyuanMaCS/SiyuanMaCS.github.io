+from pathlib import Path
+import numpy as np
+import cv2
+import artifact_io

+CASE_DIR = Path(__file__).resolve().parent

+def radial_mask(h, w, cx, cy, radius, feather=1.2):
+    yy, xx = np.mgrid[:h, :w]
+    d = np.sqrt((xx - cx) ** 2 + (yy - cy) ** 2)
+    return np.clip((radius + feather - d) / feather, 0.0, 1.0)
+
+
+def inpaint_objects(frame, centers, radius=7.0):
+    """Remove small balls while retaining the table's smooth spatial lighting."""
+    mask = np.zeros(frame.shape[:2], np.uint8)
+    yy, xx = np.mgrid[:frame.shape[0], :frame.shape[1]]
+    for x, y in centers:
+        mask[((xx - x) ** 2 + (yy - y) ** 2) <= radius ** 2] = 255
+    # TELEA behaves especially well on the nearly textureless pool cloth.
+    return cv2.inpaint(frame, mask, 4.0, cv2.INPAINT_TELEA)
+
+
+def sample_delta(source, clean, center, half=8):
+    """Extract an object's appearance as a feathered additive sprite."""
+    x, y = center
+    x0, y0 = int(round(x)) - half, int(round(y)) - half
+    x1, y1 = x0 + 2 * half + 1, y0 + 2 * half + 1
+    delta = source[y0:y1, x0:x1].astype(np.float32) - clean[y0:y1, x0:x1].astype(np.float32)
+    yy, xx = np.mgrid[-half:half + 1, -half:half + 1]
+    alpha = np.clip((7.2 - np.sqrt(xx * xx + yy * yy)) / 1.8, 0, 1).astype(np.float32)
+    return delta, alpha
+
+
+def place_delta(frame, sprite, center):
+    delta, alpha = sprite
+    h, w = alpha.shape
+    half = h // 2
+    # Sub-pixel shift of the delta sprite avoids visibly quantized motion.
+    ix, iy = int(np.floor(center[0])), int(np.floor(center[1]))
+    fx, fy = center[0] - ix, center[1] - iy
+    M = np.float32([[1, 0, fx], [0, 1, fy]])
+    shifted_d = cv2.warpAffine(delta, M, (w, h), flags=cv2.INTER_LINEAR,
+                               borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+    shifted_a = cv2.warpAffine(alpha, M, (w, h), flags=cv2.INTER_LINEAR,
+                               borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+    x0, y0 = ix - half, iy - half
+    x1, y1 = x0 + w, y0 + h
+    sx0, sy0 = max(0, -x0), max(0, -y0)
+    sx1, sy1 = w - max(0, x1 - frame.shape[1]), h - max(0, y1 - frame.shape[0])
+    x0, y0, x1, y1 = max(0, x0), max(0, y0), min(frame.shape[1], x1), min(frame.shape[0], y1)
+    if x1 <= x0 or y1 <= y0:
+        return
+    roi = frame[y0:y1, x0:x1].astype(np.float32)
+    a = shifted_a[sy0:sy1, sx0:sx1, None]
+    d = shifted_d[sy0:sy1, sx0:sx1]
+    # Delta already includes antialiasing; alpha only tapers the outer fringe.
+    roi += d * a
+    frame[y0:y1, x0:x1] = np.clip(roi, 0, 255).astype(np.uint8)
+
+
+def draw_yellow_one(frame, center):
+    """Numerically shade a small solid-yellow one-ball and its contact shadow."""
+    h, w = frame.shape[:2]
+    cx, cy = center
+    yy, xx = np.mgrid[:h, :w]
+
+    # Soft cloth shadow, offset down-right in agreement with the source balls.
+    dx, dy = xx - (cx + 1.2), yy - (cy + 2.0)
+    shadow = np.exp(-0.5 * ((dx / 5.1) ** 2 + (dy / 2.7) ** 2)) * 0.34
+    shadow *= np.clip((7.5 - np.sqrt(dx * dx + dy * dy)) / 2.0, 0, 1)
+    f = frame.astype(np.float32)
+    f *= (1.0 - shadow[..., None])
+
+    # Antialiased spherical body.
+    dx, dy = xx - cx, yy - cy
+    radius = 5.15
+    rr = np.sqrt(dx * dx + dy * dy)
+    coverage = np.clip(radius + 0.75 - rr, 0, 1)
+    inside = rr < radius
+    z = np.sqrt(np.maximum(0.0, 1.0 - (dx / radius) ** 2 - (dy / radius) ** 2))
+    nx, ny = dx / radius, dy / radius
+    # Light comes from upper left, as on the white source ball.
+    lx, ly, lz = -0.45, -0.55, 0.71
+    diffuse = np.clip(nx * lx + ny * ly + z * lz, 0, 1)
+    shade = 0.46 + 0.58 * diffuse
+    base = np.array([232.0, 186.0, 28.0], np.float32)
+    ball = base[None, None, :] * shade[..., None]
+    rim = np.clip((radius - rr) / 1.3, 0, 1)
+    ball *= (0.82 + 0.18 * rim[..., None])
+    # Compact specular glint.
+    glint = np.exp(-0.5 * (((dx + 1.7) / 0.9) ** 2 + ((dy + 1.9) / 0.75) ** 2))
+    ball = ball * (1 - 0.72 * glint[..., None]) + 250 * (0.72 * glint[..., None])
+    a = (coverage * inside)[..., None]
+    f = f * (1 - a) + ball * a
+
+    # Tiny ivory number medallion on the camera-facing side, with a dark "1".
+    spot = ((dx - 0.9) / 1.65) ** 2 + ((dy + 0.05) / 1.8) ** 2
+    spot_a = np.clip((1.18 - spot) * 2.1, 0, 1) * inside
+    ivory = np.array([236.0, 230.0, 194.0], np.float32)
+    f = f * (1 - spot_a[..., None]) + ivory * spot_a[..., None]
+    one = (np.abs(dx - 0.9) < 0.42) & (dy > -1.05) & (dy < 1.15) & inside
+    one_a = one.astype(np.float32) * 0.82
+    ink = np.array([57.0, 47.0, 27.0], np.float32)
+    f = f * (1 - one_a[..., None]) + ink * one_a[..., None]
+    frame[:] = np.clip(f, 0, 255).astype(np.uint8)
+
+
+def dark_target_tracks(frames):
+    """Track the dark target using its low luminance in the known diagonal lane."""
+    tracks = []
+    for t, rgb in enumerate(frames):
+        if t <= 12:
+            tracks.append((127.0, 71.0))
+            continue
+        lum = rgb.astype(np.float32).mean(axis=2)
+        best = (1e9, 127, 71)
+        for y in range(42, 78):
+            for x in range(120, 177):
+                if abs(y - (143.0 - 0.57 * x)) > 6.0:
+                    continue
+                score = float(lum[y-2:y+3, x-2:x+3].mean())
+                if score < best[0]:
+                    best = (score, x, y)
+        tracks.append((float(best[1]), float(best[2])))
+    return tracks
+
+
+def generate():
+    src = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert src.shape == (96, 144, 256, 3) and src.dtype == np.uint8
+
+    # Measured cue-ball centers before impact; afterward it rests at the target's
+    # old location in the factual source.
+    cue_early = [
+        (85.36, 93.60), (89.00, 91.41), (92.41, 89.54), (96.00, 87.50),
+        (99.00, 85.50), (102.09, 83.69), (105.00, 82.00), (108.00, 80.36),
+        (111.00, 78.74), (113.50, 77.33), (116.00, 75.74), (118.50, 74.33),
+        (120.79, 73.00), (123.00, 71.54), (124.57, 70.82)
+    ]
+    cue_src = [cue_early[min(t, len(cue_early)-1)] if t < 15 else (125.55, 70.10)
+               for t in range(96)]
+    target_src = dark_target_tracks(src)
+
+    # Clean source balls only where needed. Frame 4 also supplies a matching
+    # stopped-cue sprite, and every target phase supplies its own rolling look.
+    clean4 = inpaint_objects(src[4], [cue_src[4]])
+    cue_sprite = sample_delta(src[4], clean4, cue_src[4])
+
+    target_sprites = []
+    target_clean = []
+    for s in range(96):
+        c = inpaint_objects(src[s], [target_src[s]])
+        target_clean.append(c)
+        target_sprites.append(sample_delta(src[s], c, target_src[s]))
+
+    out = src.copy()
+    cue_stop = cue_src[4]  # first equal-mass transfer at the inserted ball
+    yellow_initial = ((cue_src[0][0] + 127.0) / 2.0,
+                      (cue_src[0][1] + 71.0) / 2.0)
+    yellow_stop = (118.4, 75.65)  # contact point with the dark target
+
+    for t in range(96):
+        # Through frame 4, the factual cue and target are still exactly right.
+        if t >= 5:
+            centers = [cue_src[t]]
+            if t >= 9:
+                centers.append(target_src[t])
+            out[t] = inpaint_objects(out[t], centers)
+
+        # Counterfactual target starts four frames earlier after the chain hit.
+        if t >= 9:
+            s = min(95, t + 4)
+            tx, ty = target_src[s]
+            if t + 4 > 95:
+                # Continue the smooth return from the top cushion for the last
+                # four frames rather than freezing at the source endpoint.
+                q = t + 4 - 95
+                tx += -0.70 * q
+                ty += 0.47 * q
+            place_delta(out[t], target_sprites[s], (tx, ty))
+
+        # The cue retains its source rendering until contact, then remains at
+        # the first collision point.
+        if t >= 5:
+            place_delta(out[t], cue_sprite, cue_stop)
+
+        # Yellow ball: stationary from frame 1, then receives the cue's speed,
+        # and stops after transferring that speed to the target.
+        if t <= 4:
+            yp = yellow_initial
+        elif t <= 8:
+            u = (t - 4) / 4.0
+            # Smooth linear transfer with no artificial acceleration.
+            yp = (yellow_initial[0] * (1-u) + yellow_stop[0] * u,
+                  yellow_initial[1] * (1-u) + yellow_stop[1] * u)
+        else:
+            yp = yellow_stop
+        draw_yellow_one(out[t], yp)
+
+    return out.astype(np.uint8)
+
+
+if __name__ == "__main__":
+    frames = generate()
+    artifact_io.save(frames, CASE_DIR)
+    print("saved", frames.shape, frames.dtype,
+          "range", int(frames.min()), int(frames.max()))

from pathlib import Path
import numpy as np
import cv2
import artifact_io

CASE_DIR = Path(__file__).resolve().parent


def radial_mask(h, w, cx, cy, radius, feather=1.2):
    yy, xx = np.mgrid[:h, :w]
    d = np.sqrt((xx - cx) ** 2 + (yy - cy) ** 2)
    return np.clip((radius + feather - d) / feather, 0.0, 1.0)


def inpaint_objects(frame, centers, radius=7.0):
    \"\"\"Remove small balls while retaining the table's smooth spatial lighting.\"\"\"
    mask = np.zeros(frame.shape[:2], np.uint8)
    yy, xx = np.mgrid[:frame.shape[0], :frame.shape[1]]
    for x, y in centers:
        mask[((xx - x) ** 2 + (yy - y) ** 2) <= radius ** 2] = 255
    # TELEA behaves especially well on the nearly textureless pool cloth.
    return cv2.inpaint(frame, mask, 4.0, cv2.INPAINT_TELEA)


def sample_delta(source, clean, center, half=8):
    \"\"\"Extract an object's appearance as a feathered additive sprite.\"\"\"
    x, y = center
    x0, y0 = int(round(x)) - half, int(round(y)) - half
    x1, y1 = x0 + 2 * half + 1, y0 + 2 * half + 1
    delta = source[y0:y1, x0:x1].astype(np.float32) - clean[y0:y1, x0:x1].astype(np.float32)
    yy, xx = np.mgrid[-half:half + 1, -half:half + 1]
    alpha = np.clip((7.2 - np.sqrt(xx * xx + yy * yy)) / 1.8, 0, 1).astype(np.float32)
    return delta, alpha


def place_delta(frame, sprite, center):
    delta, alpha = sprite
    h, w = alpha.shape
    half = h // 2
    # Sub-pixel shift of the delta sprite avoids visibly quantized motion.
    ix, iy = int(np.floor(center[0])), int(np.floor(center[1]))
    fx, fy = center[0] - ix, center[1] - iy
    M = np.float32([[1, 0, fx], [0, 1, fy]])
    shifted_d = cv2.warpAffine(delta, M, (w, h), flags=cv2.INTER_LINEAR,
                               borderMode=cv2.BORDER_CONSTANT, borderValue=0)
    shifted_a = cv2.warpAffine(alpha, M, (w, h), flags=cv2.INTER_LINEAR,
                               borderMode=cv2.BORDER_CONSTANT, borderValue=0)
    x0, y0 = ix - half, iy - half
    x1, y1 = x0 + w, y0 + h
    sx0, sy0 = max(0, -x0), max(0, -y0)
    sx1, sy1 = w - max(0, x1 - frame.shape[1]), h - max(0, y1 - frame.shape[0])
    x0, y0, x1, y1 = max(0, x0), max(0, y0), min(frame.shape[1], x1), min(frame.shape[0], y1)
    if x1 <= x0 or y1 <= y0:
        return
    roi = frame[y0:y1, x0:x1].astype(np.float32)
    a = shifted_a[sy0:sy1, sx0:sx1, None]
    d = shifted_d[sy0:sy1, sx0:sx1]
    # Delta already includes antialiasing; alpha only tapers the outer fringe.
    roi += d * a
    frame[y0:y1, x0:x1] = np.clip(roi, 0, 255).astype(np.uint8)


def draw_yellow_one(frame, center):
    \"\"\"Numerically shade a small solid-yellow one-ball and its contact shadow.\"\"\"
    h, w = frame.shape[:2]
    cx, cy = center
    yy, xx = np.mgrid[:h, :w]

    # Soft cloth shadow, offset down-right in agreement with the source balls.
    dx, dy = xx - (cx + 1.2), yy - (cy + 2.0)
    shadow = np.exp(-0.5 * ((dx / 5.1) ** 2 + (dy / 2.7) ** 2)) * 0.34
    shadow *= np.clip((7.5 - np.sqrt(dx * dx + dy * dy)) / 2.0, 0, 1)
    f = frame.astype(np.float32)
    f *= (1.0 - shadow[..., None])

    # Antialiased spherical body.
    dx, dy = xx - cx, yy - cy
    radius = 5.15
    rr = np.sqrt(dx * dx + dy * dy)
    coverage = np.clip(radius + 0.75 - rr, 0, 1)
    inside = rr < radius
    z = np.sqrt(np.maximum(0.0, 1.0 - (dx / radius) ** 2 - (dy / radius) ** 2))
    nx, ny = dx / radius, dy / radius
    # Light comes from upper left, as on the white source ball.
    lx, ly, lz = -0.45, -0.55, 0.71
    diffuse = np.clip(nx * lx + ny * ly + z * lz, 0, 1)
    shade = 0.46 + 0.58 * diffuse
    base = np.array([232.0, 186.0, 28.0], np.float32)
    ball = base[None, None, :] * shade[..., None]
    rim = np.clip((radius - rr) / 1.3, 0, 1)
    ball *= (0.82 + 0.18 * rim[..., None])
    # Compact specular glint.
    glint = np.exp(-0.5 * (((dx + 1.7) / 0.9) ** 2 + ((dy + 1.9) / 0.75) ** 2))
    ball = ball * (1 - 0.72 * glint[..., None]) + 250 * (0.72 * glint[..., None])
    a = (coverage * inside)[..., None]
    f = f * (1 - a) + ball * a

    # Tiny ivory number medallion on the camera-facing side, with a dark \"1\".
    spot = ((dx - 0.9) / 1.65) ** 2 + ((dy + 0.05) / 1.8) ** 2
    spot_a = np.clip((1.18 - spot) * 2.1, 0, 1) * inside
    ivory = np.array([236.0, 230.0, 194.0], np.float32)
    f = f * (1 - spot_a[..., None]) + ivory * spot_a[..., None]

    one_a = one.astype(np.float32) * 0.82
    ink = np.array([57.0, 47.0, 27.0], np.float32)
    f = f * (1 - one_a[..., None]) + ink * one_a[..., None]
    frame[:] = np.clip(f, 0, 255).astype(np.uint8)

    \"\"\"Track the dark target using its low luminance in the known diagonal lane.\"\"\"
    tracks = []
    for t, rgb in enumerate(frames):

            tracks.append((127.0, 71.0))

                if score < best[0]:
