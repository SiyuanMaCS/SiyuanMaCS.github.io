+from pathlib import Path
+import numpy as np
+import artifact_io
+
+CASE_DIR = Path(__file__).resolve().parent
+

+def smoothstep(x):
+    x = np.clip(x, 0.0, 1.0)
+    return x * x * (3.0 - 2.0 * x)
+
+
+def factual_tracks():
+    """Subpixel centers measured from the supplied RGB sequence."""
+    bx_head = np.array([
+        224.5, 218.0, 212.0, 205.5, 199.0, 193.0, 187.0, 180.5,
+        175.0, 169.0, 163.0, 157.0, 152.0, 146.5, 141.5, 137.0,
+        133.0, 129.0, 125.0, 122.0, 118.5, 115.5, 112.5, 110.0,
+        107.5, 105.5, 103.5, 102.0, 100.5, 99.5, 99.0, 98.5,
+        98.0
+    ], dtype=np.float64)
+    bx = np.r_[bx_head, np.full(96 - len(bx_head), 98.0)]
+    sx_head = np.array([
+        131.2, 131.2, 131.2, 131.2, 131.2, 131.2, 131.2, 131.2,
+        131.2, 131.2, 131.2, 131.2, 131.2, 130.9, 128.1, 119.9,
+        111.6, 103.6, 95.8, 88.5, 81.1, 74.3, 67.6, 61.4,
+        55.5, 50.0, 44.9, 39.9, 35.4, 31.4, 27.5, 24.2,
+        21.2, 18.6, 16.5, 14.7, 13.3, 12.5, 11.8, 11.7
+    ], dtype=np.float64)
+    sx = np.r_[sx_head, np.full(96 - len(sx_head), 11.7)]
+    by = 69.5 + 0.0350 * (224.5 - bx)
+    sy = 72.25 + 0.0250 * (131.2 - sx)
+    return bx, by, sx, sy
+
+
+def geometry_alpha(h, w, cx, cy, rx, ry, feather=1.5):
+    yy, xx = np.mgrid[:h, :w]
+    q = np.sqrt(((xx - cx) / rx) ** 2 + ((yy - cy) / ry) ** 2)
+    # 1 inside, smoothly falling over approximately `feather` pixels.
+    edge = feather / max(rx, ry)
+    return smoothstep((1.0 + edge - q) / (2.0 * edge))
+
+
+def erase_factual(frame, background, centers):
+    """Remove the two factual marbles, but retain all unrelated source pixels."""
+    bx, by, sx, sy = centers
+    h, w = frame.shape[:2]
+    # Broad bounds include contact shadow and the anti-aliased portions seen
+    # through/along the front lip of the track.
+    gb = geometry_alpha(h, w, bx, by + 1.0, 17.0, 17.5, 2.0)
+    gs = geometry_alpha(h, w, sx, sy + 0.4, 8.5, 9.0, 1.5)
+    geom = np.maximum(gb, gs)
+    diff = np.max(np.abs(frame.astype(np.int16) - background.astype(np.int16)), axis=2)
+    # Noise in this clean render is 0--3 levels; changes above that belong to
+    # the marbles, their contact shadows, or pixels revealed by their motion.
+    changed = smoothstep((diff.astype(np.float64) - 2.5) / 7.0)
+    a = (geom * changed)[..., None]
+    out = frame.astype(np.float64) * (1.0 - a) + background.astype(np.float64) * a
+    return np.clip(np.rint(out), 0, 255).astype(np.uint8)
+
+
+def sprite_layer(source_frame, background, cx, cy, rx, ry, kind):
+    """Extract a pre-rendered, track-occluded marble layer from a source frame."""
+    h, w = source_frame.shape[:2]
+    geom = geometry_alpha(h, w, cx, cy + (0.5 if kind == 'small' else 1.0),
+                          rx, ry, 1.4)
+    diff = np.max(np.abs(source_frame.astype(np.int16) - background.astype(np.int16)), axis=2)
+    evidence = smoothstep((diff.astype(np.float64) - 2.0) / 6.0)
+    alpha = geom * evidence
+
+    if kind == 'small':
+        # Suppress any warm pixels of the neighboring large marble at contact,
+        # while preserving neutral highlights in the small marble's core.
+        rgb = source_frame.astype(np.float64)
+        cyan = ((rgb[..., 1] > rgb[..., 0] * 1.10) |
+                (rgb[..., 2] > rgb[..., 0] * 1.10))
+        yy, xx = np.mgrid[:h, :w]
+        core = (((xx - cx) / 4.8) ** 2 + ((yy - cy) / 4.8) ** 2) < 1.0
+        alpha *= np.where(cyan | core, 1.0, 0.18)
+    return source_frame, alpha
+
+
+def translate_and_composite(canvas, rgb, alpha, dx, dy):
+    """Integer-pixel translation; source anti-aliasing remains untouched."""
+    ix, iy = int(np.rint(dx)), int(np.rint(dy))
+    h, w = canvas.shape[:2]
+    x0s, x1s = max(0, -ix), min(w, w - ix)
+    y0s, y1s = max(0, -iy), min(h, h - iy)
+    if x1s <= x0s or y1s <= y0s:
+        return canvas
+    x0d, x1d = x0s + ix, x1s + ix
+    y0d, y1d = y0s + iy, y1s + iy
+    a = alpha[y0s:y1s, x0s:x1s, None]
+    dst = canvas[y0d:y1d, x0d:x1d].astype(np.float64)
+    src = rgb[y0s:y1s, x0s:x1s].astype(np.float64)
+    canvas[y0d:y1d, x0d:x1d] = np.clip(np.rint(dst * (1.0 - a) + src * a),
+                                                    0, 255).astype(np.uint8)
+    return canvas
+
+
+def build_video():
+    source = np.load(CASE_DIR / 'source_pixels.npz')['rgb']
+    assert source.shape == (96, 144, 256, 3) and source.dtype == np.uint8
+
+    # In the first 40 frames every pixel along either path is unobscured for
+    # most of the samples. Their temporal median therefore recovers a clean,
+    # source-native plate without inventing track texture.
+    background = np.median(source[:40].astype(np.float32), axis=0)
+    background = np.rint(background).astype(np.uint8)
+
+    bx, by, sx, sy = factual_tracks()
+    out = source.copy()
+
+    # Collision starts between source frames 13 and 14 (zero based). Before
+    # that instant, changing mass has no effect on gravity or free rolling.
+    # Effective source fit: M/m ~= 5.7, restitution ~= 0.63. Therefore after
+    # M <- M/8, the large marble has only ~0.27 px/frame leftward velocity,
+    # while the small marble leaves at ~4.0 px/frame. Rolling resistance is
+    # measured at about 0.36 px/frame^2 from the factual small-marble track.
+    big_final_x = 142.35
+    small_x0 = 129.30                 # partial first post-contact frame
+    small_v0 = 4.00
+    rolling_a = 0.36
+    stop_t = small_v0 / rolling_a
+    small_final_x = small_x0 - small_v0 * stop_t + 0.5 * rolling_a * stop_t**2
+
+    # Precompute source-native sprites. The big marble has essentially stopped
+    # after impact, so its frame-13 texture/orientation remains fixed. The small
+    # marble's references are selected by travelled distance, retaining roll.
+    big_ref_f = 13
+    big_ref_x, big_ref_y = bx[big_ref_f], by[big_ref_f]
+    big_rgb, big_alpha = sprite_layer(source[big_ref_f], background,
+                                        big_ref_x, big_ref_y, 15.0, 15.8, 'big')
+
+    small_ref_frames = np.array([12, 14, 15, 16, 17])
+    small_ref_disp = 131.2 - sx[small_ref_frames]
+    small_layers = {}
+    for rf in small_ref_frames:
+        small_layers[int(rf)] = sprite_layer(source[rf], background,
+                                               sx[rf], sy[rf], 7.2, 7.6, 'small')
+
+    for f in range(14, 96):
+        base = erase_factual(source[f], background, (bx[f], by[f], sx[f], sy[f]))
+
+        # Large marble: mixed pre/post-contact movement places it at 142.5 in
+        # frame 14; residual motion is extinguished by rolling resistance.
+        if f == 14:
+            nbx = 142.55
+        else:
+            nbx = big_final_x
+        nby = 69.5 + 0.0350 * (224.5 - nbx)
+
+        # Small marble: uniformly resisted roll, then true rest (no creeping).
+        t = float(f - 14)
+        if t < stop_t:
+            nsx = small_x0 - small_v0 * t + 0.5 * rolling_a * t * t
+        else:
+            nsx = small_final_x
+        nsy = 72.25 + 0.0250 * (131.2 - nsx)
+
+        # During separation the small marble is in front on the left. Draw the
+        # large one first, matching the source's collision ordering.
+        translate_and_composite(base, big_rgb, big_alpha,
+                                nbx - big_ref_x, nby - big_ref_y)
+
+        disp = 131.2 - nsx
+        ri = int(np.argmin(np.abs(small_ref_disp - disp)))
+        rf = int(small_ref_frames[ri])
+        srgb, salpha = small_layers[rf]
+        translate_and_composite(base, srgb, salpha,
+                                nsx - sx[rf], nsy - sy[rf])
+        out[f] = base
+
+    return out
+
+
+if __name__ == '__main__':
+    frames = build_video()
+    artifact_io.save(frames, CASE_DIR)
+    print('saved', frames.shape, frames.dtype,
+          'range', int(frames.min()), int(frames.max()))

def smoothstep(x):
    x = np.clip(x, 0.0, 1.0)
    return x * x * (3.0 - 2.0 * x)


def factual_tracks():

def geometry_alpha(h, w, cx, cy, rx, ry, feather=1.5):
    yy, xx = np.mgrid[:h, :w]
    q = np.sqrt(((xx - cx) / rx) ** 2 + ((yy - cy) / ry) ** 2)
    # 1 inside, smoothly falling over approximately "'`feather` pixels.
    edge = feather / max(rx, ry)
    return smoothstep((1.0 + edge - q) / (2.0 * edge))


def erase_factual(frame, background, centers):
    """Remove the two factual marbles, but retain all unrelated source pixels."""
    bx, by, sx, sy = centers
    h, w = frame.shape[:2]
    # Broad bounds include contact shadow and the anti-aliased portions seen
    # through/along the front lip of the track.
    gb = geometry_alpha(h, w, bx, by + 1.0, 17.0, 17.5, 2.0)
    gs = geometry_alpha(h, w, sx, sy + 0.4, 8.5, 9.0, 1.5)
    geom = np.maximum(gb, gs)
    diff = np.max(np.abs(frame.astype(np.int16) - background.astype(np.int16)), axis=2)
    # Noise in this clean render is 0--3 levels; changes above that belong to
    # the marbles, their contact shadows, or pixels revealed by their motion.
    changed = smoothstep((diff.astype(np.float64) - 2.5) / 7.0)
    a = (geom * changed)[..., None]
    out = frame.astype(np.float64) * (1.0 - a) + background.astype(np.float64) * a
    return np.clip(np.rint(out), 0, 255).astype(np.uint8)


def sprite_layer(source_frame, background, cx, cy, rx, ry, kind):
    """Extract a pre-rendered, track-occluded marble layer from a source frame."""
    h, w = source_frame.shape[:2]
    geom = geometry_alpha(h, w, cx, cy + (0.5 if kind == '"'small' else 1.0),
                          rx, ry, 1.4)
    diff = np.max(np.abs(source_frame.astype(np.int16) - background.astype(np.int16)), axis=2)
    evidence = smoothstep((diff.astype(np.float64) - 2.0) / 6.0)
    alpha = geom * evidence

    if kind == 'small':
        # Suppress any warm pixels of the neighboring large marble at contact,
        # while preserving neutral highlights in the small marble's core.
        rgb = source_frame.astype(np.float64)
        cyan = ((rgb[..., 1] > rgb[..., 0] * 1.10) |
                (rgb[..., 2] > rgb[..., 0] * 1.10))
        yy, xx = np.mgrid[:h, :w]
        core = (((xx - cx) / 4.8) ** 2 + ((yy - cy) / 4.8) ** 2) < 1.0
        alpha *= np.where(cyan | core, 1.0, 0.18)
    return source_frame, alpha


def translate_and_composite(canvas, rgb, alpha, dx, dy):
    \"\"\"Integer-pixel translation; source anti-aliasing remains untouched.\"\"\"
    ix, iy = int(np.rint(dx)), int(np.rint(dy))
    h, w = canvas.shape[:2]
    x0s, x1s = max(0, -ix), min(w, w - ix)
    y0s, y1s = max(0, -iy), min(h, h - iy)
    if x1s <= x0s or y1s <= y0s:
        return canvas
    x0d, x1d = x0s + ix, x1s + ix
    y0d, y1d = y0s + iy, y1s + iy
    a = alpha[y0s:y1s, x0s:x1s, None]
    dst = canvas[y0d:y1d, x0d:x1d].astype(np.float64)
    src = rgb[y0s:y1s, x0s:x1s].astype(np.float64)
    canvas[y0d:y1d, x0d:x1d] = np.clip(np.rint(dst * (1.0 - a) + src * a),
                                                    0, 255).astype(np.uint8)
    return canvas

    source = np.load(CASE_DIR / 'source_pixels.npz')['rgb']
    assert source.shape == (96, 144, 256, 3) and source.dtype == np.uint8

    # In the first 40 frames every pixel along either path is unobscured for
    # most of the samples. Their temporal median therefore recovers a clean,
    # source-native plate without inventing track texture.
    background = np.median(source[:40].astype(np.float32), axis=0)
    background = np.rint(background).astype(np.uint8)

    bx, by, sx, sy = factual_tracks()
    out = source.copy()

    # that instant, changing mass has no effect on gravity or free rolling.
    # Effective source fit: M/m ~= 5.7, restitution ~= 0.63. Therefore after
    # M <- M/8, the large marble has only ~0.27 px/frame leftward velocity,
    # while the small marble leaves at ~4.0 px/frame. Rolling resistance is
    # measured at about 0.36 px/frame"'^2 from the factual small-marble track.
    big_final_x = 142.35
    small_x0 = 129.30                 # partial first post-contact frame
    small_v0 = 4.00
    rolling_a = 0.36
    stop_t = small_v0 / rolling_a
    small_final_x = small_x0 - small_v0 * stop_t + 0.5 * rolling_a * stop_t**2

    # Precompute source-native sprites. The big marble has essentially stopped
    # after impact, so its frame-13 texture/orientation remains fixed. The small
    # marble'"'s references are selected by travelled distance, retaining roll.
    big_ref_f = 13

                                        big_ref_x, big_ref_y, 15.0, 15.8, 'big')

                                               sx[rf], sy[rf], 7.2, 7.6, 'small')

        translate_and_composite(base, srgb, salpha,
                                nsx - sx[rf], nsy - sy[rf])
        out[f] = base

    return out
