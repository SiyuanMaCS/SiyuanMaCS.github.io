+from pathlib import Path
+import cv2
+import numpy as np
+import artifact_io
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def component_tracks(frames):
+    """Measure the two dynamic bodies directly from RGB values."""
+    n, h, w, _ = frames.shape
+    ball = np.zeros((n, 2), np.float32)
+    apple = np.full((n, 2), np.nan, np.float32)
+
+    # The ball is the large, nearly achromatic circle in the lower middle.
+    for t, im in enumerate(frames):
+        hsv = cv2.cvtColor(im, cv2.COLOR_RGB2HSV)
+        m = ((hsv[:, :, 2] > 170) & (hsv[:, :, 1] < 70)).astype(np.uint8)
+        m[:65] = 0
+        m[:, 190:] = 0
+        count, _, stats, centers = cv2.connectedComponentsWithStats(m, 8)
+        choices = []
+        for i in range(1, count):
+            x, y, ww, hh, area = stats[i]
+            if area > 110 and 65 <= y <= 90 and 15 <= ww <= 25 and 15 <= hh <= 25:
+                choices.append((area, centers[i]))
+        if not choices:
+            raise RuntimeError(f"silver ball was not found in frame {t}")
+        ball[t] = max(choices, key=lambda z: z[0])[1]
+
+    # The apple is the small warm/red body that drops in and departs to the right.
+    yy = np.arange(h)[:, None]
+    xx = np.arange(w)[None, :]
+    previous = None
+    for t in range(12, 40):
+        im = frames[t].astype(np.float32)
+        r, g, b = im[:, :, 0], im[:, :, 1], im[:, :, 2]
+        warm = (r > 70) & (r > 1.15 * g) & (r > 1.20 * b) & (yy > 58) & (xx > 124)
+        count, _, stats, centers = cv2.connectedComponentsWithStats(warm.astype(np.uint8), 8)
+        choices = []
+        for i in range(1, count):
+            x, y, ww, hh, area = stats[i]
+            if area >= 7 and ww <= 10 and hh <= 10:
+                c = centers[i]
+                if previous is None or (c[0] >= previous[0] - 2 and np.linalg.norm(c - previous) < 20):
+                    choices.append((area, c))
+        if choices:
+            # Prefer the coherent right-moving component, not warm details in scenery.
+            c = min(choices, key=lambda z: abs(z[1][0] - (previous[0] if previous is not None else 134)))[1]
+            apple[t] = c
+            previous = c
+
+    valid = np.flatnonzero(np.isfinite(apple[:, 0]))
+    if len(valid) < 20:
+        raise RuntimeError("apple trajectory extraction failed")
+    return ball, apple
+
+
+def ellipse_mask(h, w, cx, cy, rx, ry, feather=0.8):
+    y, x = np.mgrid[:h, :w]
+    q = np.sqrt(((x - cx) / rx) ** 2 + ((y - cy) / ry) ** 2)
+    return np.clip((1.0 + feather / max(rx, ry) - q) * max(rx, ry) / feather, 0, 1).astype(np.float32)
+
+
+def occupancy_masks(shape, ball, apple):
+    n, h, w, _ = shape
+    occ = np.zeros((n, h, w), bool)
+    for t in range(n):
+        # Include the ball, contact edge, and its low soft shadow.
+        bm = ellipse_mask(h, w, ball[t, 0], ball[t, 1] + 1.0, 13.0, 14.0, 1.0)
+        occ[t] |= bm > 0
+        if np.isfinite(apple[t, 0]):
+            am = ellipse_mask(h, w, apple[t, 0], apple[t, 1] + 0.4, 7.0, 7.0, 1.0)
+            occ[t] |= am > 0
+    return occ
+
+
+def clean_background(frames, occ):
+    """Reveal covered grass using the temporally nearest unobscured source pixel."""
+    n, h, w, _ = frames.shape
+    previous = np.full((n, h, w), -1000, np.int16)
+    following = np.full((n, h, w), 1000, np.int16)
+    last = np.full((h, w), -1000, np.int16)
+    for t in range(n):
+        last[~occ[t]] = t
+        previous[t] = last
+    nxt = np.full((h, w), 1000, np.int16)
+    for t in range(n - 1, -1, -1):
+        nxt[~occ[t]] = t
+        following[t] = nxt
+
+    bg = frames.copy()
+    yy, xx = np.mgrid[:h, :w]
+    for t in range(n):
+        m = occ[t]
+        use_prev = (t - previous[t]) <= (following[t] - t)
+        indices = np.where(use_prev, previous[t], following[t])
+        indices = np.clip(indices, 0, n - 1)
+        bg[t][m] = frames[indices[m], yy[m], xx[m]]
+    return bg
+
+
+def shift_layer(rgb, alpha, dx, dy):
+    """Subpixel translate a premultiplied RGB object layer."""
+    h, w = alpha.shape
+    matrix = np.float32([[1, 0, dx], [0, 1, dy]])
+    premul = rgb.astype(np.float32) * alpha[:, :, None]
+    moved_rgb = cv2.warpAffine(premul, matrix, (w, h), flags=cv2.INTER_LINEAR,
+                               borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+    moved_a = cv2.warpAffine(alpha, matrix, (w, h), flags=cv2.INTER_LINEAR,
+                             borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+    return moved_rgb, moved_a
+
+
+def composite(canvas, premul, alpha):
+    return canvas * (1.0 - alpha[:, :, None]) + premul
+
+
+def make_video():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    if source.shape != (96, 144, 256, 3) or source.dtype != np.uint8:
+        raise ValueError((source.shape, source.dtype))
+    n, h, w, _ = source.shape
+    ball, apple = component_tracks(source)
+    occ = occupancy_masks(source.shape, ball, apple)
+    clean = clean_background(source, occ)
+
+    # Object alpha follows the actual silhouettes while retaining soft antialiasing.
+    ball_alpha = []
+    apple_alpha = []
+    for t in range(n):
+        # Solid circular silver body plus a difference-selected lower contact shadow.
+        body = ellipse_mask(h, w, ball[t, 0], ball[t, 1], 10.7, 10.7, 1.1)
+        diff = np.max(np.abs(source[t].astype(np.int16) - clean[t].astype(np.int16)), axis=2)
+        vicinity = ellipse_mask(h, w, ball[t, 0] + 0.5, ball[t, 1] + 1.5, 13.0, 14.0, 1.3)
+        detail = np.clip((diff.astype(np.float32) - 7.0) / 18.0, 0, 1) * vicinity
+        ball_alpha.append(np.maximum(body, detail))
+
+        if np.isfinite(apple[t, 0]):
+            body_a = ellipse_mask(h, w, apple[t, 0], apple[t, 1], 4.15, 4.15, 0.9)
+            near_a = ellipse_mask(h, w, apple[t, 0], apple[t, 1], 6.1, 6.4, 1.0)
+            detail_a = np.clip((diff.astype(np.float32) - 6.0) / 16.0, 0, 1) * near_a
+            apple_alpha.append(np.maximum(body_a, detail_a))
+        else:
+            apple_alpha.append(np.zeros((h, w), np.float32))
+
+    output = source.copy().astype(np.float32)
+    contact = 12  # first visible physical contact (zero-based source index)
+    ball_origin = ball[contact].copy()
+    apple_origin = apple[contact].copy()
+
+    # A twice-as-massive apple delivers substantially more collision impulse.
+    # The source uses damped rolling; scaling displacement preserves the measured
+    # deceleration profile and yields a physically legible farther stopping point.
+    BALL_MOMENTUM_SCALE = 1.70
+    APPLE_CARRY_SCALE = 1.12
+
+    for t in range(contact, n):
+        # First remove both factual dynamic bodies from their old locations.
+        removal = occ[t].astype(np.float32)
+        frame = output[t] * (1.0 - removal[:, :, None]) + clean[t].astype(np.float32) * removal[:, :, None]
+
+        # Ball: unchanged at contact, then rolls left 1.7x the original displacement.
+        desired_ball = ball_origin + BALL_MOMENTUM_SCALE * (ball[t] - ball_origin)
+        bp, ba = shift_layer(source[t], ball_alpha[t],
+                             float(desired_ball[0] - ball[t, 0]),
+                             float(desired_ball[1] - ball[t, 1]))
+        frame = composite(frame, bp, ba)
+
+        # Apple: preserve free-fall through contact; after impact its greater inertia
+        # carries it somewhat farther right before it exits the image.
+        if np.isfinite(apple[t, 0]):
+            if t <= contact:
+                desired_apple = apple[t]
+            else:
+                desired_apple = apple_origin + APPLE_CARRY_SCALE * (apple[t] - apple_origin)
+            ap, aa = shift_layer(source[t], apple_alpha[t],
+                                 float(desired_apple[0] - apple[t, 0]),
+                                 float(desired_apple[1] - apple[t, 1]))
+            frame = composite(frame, ap, aa)
+
+        output[t] = frame
+
+    return np.clip(np.rint(output), 0, 255).astype(np.uint8)
+
+
+if __name__ == "__main__":
+    frames = make_video()
+    artifact_io.save(frames, CASE_DIR)
+    print("saved", frames.shape, frames.dtype)

from pathlib import Path
import cv2
import numpy as np
import artifact_io

CASE_DIR = Path(__file__).resolve().parent


def component_tracks(frames):
    \"\"\"Measure the two dynamic bodies directly from RGB values.\"\"\"
    n, h, w, _ = frames.shape
    ball = np.zeros((n, 2), np.float32)
    apple = np.full((n, 2), np.nan, np.float32)

    # The ball is the large, nearly achromatic circle in the lower middle.
    for t, im in enumerate(frames):
        hsv = cv2.cvtColor(im, cv2.COLOR_RGB2HSV)
        m = ((hsv[:, :, 2] > 170) & (hsv[:, :, 1] < 70)).astype(np.uint8)
        m[:65] = 0
        m[:, 190:] = 0
        count, _, stats, centers = cv2.connectedComponentsWithStats(m, 8)
        choices = []
        for i in range(1, count):
            x, y, ww, hh, area = stats[i]
            if area > 110 and 65 <= y <= 90 and 15 <= ww <= 25 and 15 <= hh <= 25:
                choices.append((area, centers[i]))
        if not choices:
            raise RuntimeError(f\"silver ball was not found in frame {t}\")
        ball[t] = max(choices, key=lambda z: z[0])[1]

    # The apple is the small warm/red body that drops in and departs to the right.
    yy = np.arange(h)[:, None]
    xx = np.arange(w)[None, :]
    previous = None
    for t in range(12, 40):
        im = frames[t].astype(np.float32)
        r, g, b = im[:, :, 0], im[:, :, 1], im[:, :, 2]
        warm = (r > 70) & (r > 1.15 * g) & (r > 1.20 * b) & (yy > 58) & (xx > 124)
        count, _, stats, centers = cv2.connectedComponentsWithStats(warm.astype(np.uint8), 8)
        choices = []
        for i in range(1, count):
            x, y, ww, hh, area = stats[i]
            if area >= 7 and ww <= 10 and hh <= 10:
                c = centers[i]
                if previous is None or (c[0] >= previous[0] - 2 and np.linalg.norm(c - previous) < 20):
                    choices.append((area, c))
        if choices:
            # Prefer the coherent right-moving component, not warm details in scenery.
            c = min(choices, key=lambda z: abs(z[1][0] - (previous[0] if previous is not None else 134)))[1]
            apple[t] = c
            previous = c

    valid = np.flatnonzero(np.isfinite(apple[:, 0]))
    if len(valid) < 20:
        raise RuntimeError(\"apple trajectory extraction failed\")
    return ball, apple


def ellipse_mask(h, w, cx, cy, rx, ry, feather=0.8):
    y, x = np.mgrid[:h, :w]
    q = np.sqrt(((x - cx) / rx) ** 2 + ((y - cy) / ry) ** 2)
    return np.clip((1.0 + feather / max(rx, ry) - q) * max(rx, ry) / feather, 0, 1).astype(np.float32)


def occupancy_masks(shape, ball, apple):
    n, h, w, _ = shape
    occ = np.zeros((n, h, w), bool)
    for t in range(n):
        # Include the ball, contact edge, and its low soft shadow.
        bm = ellipse_mask(h, w, ball[t, 0], ball[t, 1] + 1.0, 13.0, 14.0, 1.0)
        occ[t] |= bm > 0
        if np.isfinite(apple[t, 0]):
            am = ellipse_mask(h, w, apple[t, 0], apple[t, 1] + 0.4, 7.0, 7.0, 1.0)
            occ[t] |= am > 0
    return occ


def clean_background(frames, occ):
    \"\"\"Reveal covered grass using the temporally nearest unobscured source pixel.\"\"\"
    n, h, w, _ = frames.shape
    previous = np.full((n, h, w), -1000, np.int16)
    following = np.full((n, h, w), 1000, np.int16)
    last = np.full((h, w), -1000, np.int16)
    for t in range(n):
        last[~occ[t]] = t
        previous[t] = last
    nxt = np.full((h, w), 1000, np.int16)
    for t in range(n - 1, -1, -1):
        nxt[~occ[t]] = t
        following[t] = nxt

    bg = frames.copy()
    yy, xx = np.mgrid[:h, :w]
    for t in range(n):
        m = occ[t]
        use_prev = (t - previous[t]) <= (following[t] - t)
        indices = np.where(use_prev, previous[t], following[t])
        indices = np.clip(indices, 0, n - 1)
        bg[t][m] = frames[indices[m], yy[m], xx[m]]
    return bg


def shift_layer(rgb, alpha, dx, dy):
    \"\"\"Subpixel translate a premultiplied RGB object layer.\"\"\"
    h, w = alpha.shape
    matrix = np.float32([[1, 0, dx], [0, 1, dy]])
    premul = rgb.astype(np.float32) * alpha[:, :, None]
    moved_rgb = cv2.warpAffine(premul, matrix, (w, h), flags=cv2.INTER_LINEAR,
                               borderMode=cv2.BORDER_CONSTANT, borderValue=0)
    moved_a = cv2.warpAffine(alpha, matrix, (w, h), flags=cv2.INTER_LINEAR,
                             borderMode=cv2.BORDER_CONSTANT, borderValue=0)
    return moved_rgb, moved_a


def composite(canvas, premul, alpha):
    return canvas * (1.0 - alpha[:, :, None]) + premul


def make_video():
    source = np.load(CASE_DIR / \"source_pixels.npz\")[\"rgb\"]
    if source.shape "'!= (96, 144, 256, 3) or source.dtype != np.uint8:
        raise ValueError((source.shape, source.dtype))
    n, h, w, _ = source.shape
    ball, apple = component_tracks(source)
    occ = occupancy_masks(source.shape, ball, apple)
    clean = clean_background(source, occ)

    # Object alpha follows the actual silhouettes while retaining soft antialiasing.
    ball_alpha = []
    apple_alpha = []
    for t in range(n):
        # Solid circular silver body plus a difference-selected lower contact shadow.
        body = ellipse_mask(h, w, ball[t, 0], ball[t, 1], 10.7, 10.7, 1.1)
        diff = np.max(np.abs(source[t].astype(np.int16) - clean[t].astype(np.int16)), axis=2)
        vicinity = ellipse_mask(h, w, ball[t, 0] + 0.5, ball[t, 1] + 1.5, 13.0, 14.0, 1.3)
        detail = np.clip((diff.astype(np.float32) - 7.0) / 18.0, 0, 1) * vicinity
        ball_alpha.append(np.maximum(body, detail))

        if np.isfinite(apple[t, 0]):
            body_a = ellipse_mask(h, w, apple[t, 0], apple[t, 1], 4.15, 4.15, 0.9)
            near_a = ellipse_mask(h, w, apple[t, 0], apple[t, 1], 6.1, 6.4, 1.0)
            detail_a = np.clip((diff.astype(np.float32) - 6.0) / 16.0, 0, 1) * near_a
            apple_alpha.append(np.maximum(body_a, detail_a))
        else:
            apple_alpha.append(np.zeros((h, w), np.float32))

    output = source.copy().astype(np.float32)
    contact = 12  # first visible physical contact (zero-based source index)
    ball_origin = ball[contact].copy()
    apple_origin = apple[contact].copy()

    # A twice-as-massive apple delivers substantially more collision impulse.
    # The source uses damped rolling; scaling displacement preserves the measured
    # deceleration profile and yields a physically legible farther stopping point.
    BALL_MOMENTUM_SCALE = 1.70
    APPLE_CARRY_SCALE = 1.12

    for t in range(contact, n):
        # First remove both factual dynamic bodies from their old locations.
        removal = occ[t].astype(np.float32)
        frame = output[t] * (1.0 - removal[:, :, None]) + clean[t].astype(np.float32) * removal[:, :, None]

        # Ball: unchanged at contact, then rolls left 1.7x the original displacement.
        desired_ball = ball_origin + BALL_MOMENTUM_SCALE * (ball[t] - ball_origin)
        bp, ba = shift_layer(source[t], ball_alpha[t],
                             float(desired_ball[0] - ball[t, 0]),
                             float(desired_ball[1] - ball[t, 1]))
        frame = composite(frame, bp, ba)

        # Apple: preserve free-fall through contact; after impact its greater inertia
        # carries it somewhat farther right before it exits the image.
        if np.isfinite(apple[t, 0]):
            if t <= contact:
                desired_apple = apple[t]
            else:
                desired_apple = apple_origin + APPLE_CARRY_SCALE * (apple[t] - apple_origin)
            ap, aa = shift_layer(source[t], apple_alpha[t],
                                 float(desired_apple[0] - apple[t, 0]),
                                 float(desired_apple[1] - apple[t, 1]))
            frame = composite(frame, ap, aa)

        output[t] = frame

    return np.clip(np.rint(output), 0, 255).astype(np.uint8)


if __name__ == "__main__":
