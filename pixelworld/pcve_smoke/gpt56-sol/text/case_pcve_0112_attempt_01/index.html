+"""Generate the counterfactual tennis-ball video.
+
+The source ball is airborne until it first meets the horizontal surface at frame
+28 (zero-based).  Friction cannot alter the free-flight arc.  For Coulomb
+friction, multiplying mu by three multiplies the tangential deceleration by
+three.  Thus, after contact, the corresponding point on the original rolling
+trajectory is reached three times sooner, while displacement is divided by
+three:
+
+    x_3mu(t) = x_contact + (x_mu(3*t) - x_contact) / 3.
+
+The code estimates the tiny ball's center from RGB differences, removes its
+original post-contact rasterization against a clean temporal background, then
+moves a source ball residual (including its antialiasing/contact shadow) to the
+counterfactual position.  Everything outside that local footprint is retained
+verbatim from the source.
+"""
+from pathlib import Path
+import numpy as np
+
+import artifact_io
+
+CASE_DIR = Path(__file__).resolve().parent
+N, H, W = 96, 144, 256
+CONTACT = 28                 # zero-based: first frame resting on the surface
+FRICTION_MULTIPLIER = 3.0
+
+
+def track_ball(frames):
+    """Track the compact moving yellow-green component without external tools."""
+    temporal_bg = np.median(frames, axis=0).astype(np.uint8)
+    centers = []
+    previous = np.array([28.0, 59.0])  # clear ball location in source frame 0
+    yy, xx = np.indices((H, W))
+    for frame in frames:
+        difference = np.max(
+            np.abs(frame.astype(np.int16) - temporal_bg.astype(np.int16)), axis=2
+        )
+        local = ((xx - previous[0]) ** 2 + (yy - previous[1]) ** 2) < 100.0
+        weights = np.where(local & (difference > 12), difference, 0).astype(float)
+        total = weights.sum()
+        if total == 0:
+            center = previous
+        else:
+            center = np.array([(weights * xx).sum() / total,
+                               (weights * yy).sum() / total])
+        centers.append(center)
+        previous = center
+    return np.asarray(centers)
+
+
+def predicted_clean_patch(frames, clean_reference, frame_index, center, radius=6):
+    """Return bounds and a clean, illumination-adjusted background patch."""
+    cx, cy = center
+    x0 = max(0, int(np.floor(cx)) - radius)
+    x1 = min(W, int(np.floor(cx)) + radius + 1)
+    y0 = max(0, int(np.floor(cy)) - radius)
+    y1 = min(H, int(np.floor(cy)) + radius + 1)
+
+    # Estimate the small frame-to-frame render/encoding offset in an annulus.
+    ax0, ax1 = max(0, x0 - 5), min(W, x1 + 5)
+    ay0, ay1 = max(0, y0 - 5), min(H, y1 + 5)
+    current = frames[frame_index, ay0:ay1, ax0:ax1].astype(np.int16)
+    reference = clean_reference[ay0:ay1, ax0:ax1].astype(np.int16)
+    ring = np.ones(current.shape[:2], dtype=bool)
+    ring[y0-ay0:y1-ay0, x0-ax0:x1-ax0] = False
+    delta = np.median((current - reference)[ring], axis=0)
+    clean = np.clip(
+        clean_reference[y0:y1, x0:x1].astype(float) + delta, 0, 255
+    )
+    return (x0, y0, x1, y1), clean
+
+
+def ball_residual(frames, clean_reference, frame_index, center):
+    """Extract an additive RGB residual for ball, edge AA, and contact shadow."""
+    bounds, clean = predicted_clean_patch(
+        frames, clean_reference, frame_index, center, radius=6
+    )
+    x0, y0, x1, y1 = bounds
+    observed = frames[frame_index, y0:y1, x0:x1].astype(float)
+    raw = observed - clean
+    strength = np.max(np.abs(raw), axis=2)
+
+    # Keep the component compact and suppress tiny background/codec variation.
+    yy, xx = np.indices(strength.shape)
+    geometric = ((xx + x0 - center[0]) ** 2 +
+                 (yy + y0 - center[1]) ** 2) <= 25.0
+    feather = np.clip((strength - 3.0) / 6.0, 0.0, 1.0) * geometric
+    residual = raw * feather[:, :, None]
+    return bounds, residual
+
+
+def place_shifted_residual(frame, bounds, residual, dx, dy):
+    """Place a small residual with bilinear subpixel translation."""
+    x0, y0, x1, y1 = bounds
+    rh, rw = residual.shape[:2]
+    # Destination support, enlarged by one pixel for bilinear filtering.
+    ox0 = max(0, int(np.floor(x0 + dx)) - 1)
+    ox1 = min(W, int(np.ceil(x1 + dx)) + 1)
+    oy0 = max(0, int(np.floor(y0 + dy)) - 1)
+    oy1 = min(H, int(np.ceil(y1 + dy)) + 1)
+    if ox0 >= ox1 or oy0 >= oy1:
+        return
+
+    oy, ox = np.indices((oy1 - oy0, ox1 - ox0), dtype=float)
+    sx = ox + ox0 - dx - x0
+    sy = oy + oy0 - dy - y0
+    ix = np.floor(sx).astype(int)
+    iy = np.floor(sy).astype(int)
+    fx = sx - ix
+    fy = sy - iy
+
+    sampled = np.zeros((oy1 - oy0, ox1 - ox0, 3), dtype=float)
+    for j, wy in ((0, 1.0 - fy), (1, fy)):
+        for i, wx in ((0, 1.0 - fx), (1, fx)):
+            px, py = ix + i, iy + j
+            valid = (px >= 0) & (px < rw) & (py >= 0) & (py < rh)
+            if np.any(valid):
+                sampled[valid] += residual[py[valid], px[valid]] * \
+                                  (wx[valid] * wy[valid])[:, None]
+    frame[oy0:oy1, ox0:ox1] += sampled
+
+
+def generate():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (N, H, W, 3) and source.dtype == np.uint8
+    centers = track_ball(source)
+
+    # Frames 0..19 contain no ball near the eventual floor path. Their median
+    # is a stable, texture-preserving estimate of that occluded background.
+    clean_reference = np.median(source[:20], axis=0).astype(np.uint8)
+    output = source.astype(float).copy()
+
+    contact_x, contact_y = centers[CONTACT]
+    # The source has fully settled by frame 80; averaging the final tracked
+    # centers avoids fitting to subpixel raster noise.
+    settled_x = float(np.median(centers[80:, 0]))
+    settled_y = float(np.median(centers[80:, 1]))
+
+    for t in range(CONTACT + 1, N):
+        # Remove the factual ball from this frame, retaining all other pixels.
+        old_bounds, old_residual = ball_residual(
+            source, clean_reference, t, centers[t]
+        )
+        x0, y0, x1, y1 = old_bounds
+        output[t, y0:y1, x0:x1] -= old_residual
+
+        elapsed = t - CONTACT
+        source_time = CONTACT + FRICTION_MULTIPLIER * elapsed
+        if source_time < 80.0:
+            # Interpolate the source center at the accelerated physical time.
+            lo = int(np.floor(source_time))
+            hi = min(95, lo + 1)
+            f = source_time - lo
+            sampled_center = centers[lo] * (1.0 - f) + centers[hi] * f
+            new_x = contact_x + (sampled_center[0] - contact_x) / \
+                    FRICTION_MULTIPLIER
+            new_y = contact_y + (sampled_center[1] - contact_y) / \
+                    FRICTION_MULTIPLIER
+            sprite_frame = int(round(source_time))
+        else:
+            # At 3x friction the ball has stopped; retain a fixed final phase.
+            new_x = contact_x + (settled_x - contact_x) / FRICTION_MULTIPLIER
+            new_y = contact_y + (settled_y - contact_y) / FRICTION_MULTIPLIER
+            sprite_frame = 85
+
+        sprite_bounds, sprite_residual = ball_residual(
+            source, clean_reference, sprite_frame, centers[sprite_frame]
+        )
+        place_shifted_residual(
+            output[t], sprite_bounds, sprite_residual,
+            new_x - centers[sprite_frame, 0],
+            new_y - centers[sprite_frame, 1],
+        )
+
+    output = np.clip(np.rint(output), 0, 255).astype(np.uint8)
+    # Frame 28 is intentionally exact: it is the instant of first contact.
+    assert np.array_equal(output[:CONTACT + 1], source[:CONTACT + 1])
+    return output
+
+
+if __name__ == "__main__":
+    frames = generate()
+    artifact_io.save(frames, CASE_DIR)
+    print("saved", frames.shape, frames.dtype,
+          "to", CASE_DIR / "pixels.npz")

\"\"\"Generate the counterfactual tennis-ball video.

The source ball is airborne until it first meets the horizontal surface at frame
28 (zero-based).  Friction cannot alter the free-flight arc.  For Coulomb
friction, multiplying mu by three multiplies the tangential deceleration by
three.  Thus, after contact, the corresponding point on the original rolling
trajectory is reached three times sooner, while displacement is divided by
three:

    x_3mu(t) = x_contact + (x_mu(3*t) - x_contact) / 3.

The code estimates the tiny ball's center from RGB differences, removes its
original post-contact rasterization against a clean temporal background, then
moves a source ball residual (including its antialiasing/contact shadow) to the
counterfactual position.  Everything outside that local footprint is retained
verbatim from the source.
\"\"\"
from pathlib import Path
import numpy as np

import artifact_io

CASE_DIR = Path(__file__).resolve().parent
N, H, W = 96, 144, 256
CONTACT = 28                 # zero-based: first frame resting on the surface
FRICTION_MULTIPLIER = 3.0


def track_ball(frames):
    \"\"\"Track the compact moving yellow-green component without external tools.\"\"\"
    temporal_bg = np.median(frames, axis=0).astype(np.uint8)
    centers = []
    previous = np.array([28.0, 59.0])  # clear ball location in source frame 0
    yy, xx = np.indices((H, W))
    for frame in frames:
        difference = np.max(
            np.abs(frame.astype(np.int16) - temporal_bg.astype(np.int16)), axis=2
        )
        local = ((xx - previous[0]) ** 2 + (yy - previous[1]) ** 2) < 100.0
        weights = np.where(local & (difference > 12), difference, 0).astype(float)
        total = weights.sum()
        if total == 0:
            center = previous
        else:
            center = np.array([(weights * xx).sum() / total,
                               (weights * yy).sum() / total])
        centers.append(center)
        previous = center
    return np.asarray(centers)


def predicted_clean_patch(frames, clean_reference, frame_index, center, radius=6):
    \"\"\"Return bounds and a clean, illumination-adjusted background patch.\"\"\"
    cx, cy = center
    x0 = max(0, int(np.floor(cx)) - radius)
    x1 = min(W, int(np.floor(cx)) + radius + 1)
    y0 = max(0, int(np.floor(cy)) - radius)
    y1 = min(H, int(np.floor(cy)) + radius + 1)

    # Estimate the small frame-to-frame render/encoding offset in an annulus.
    ax0, ax1 = max(0, x0 - 5), min(W, x1 + 5)
    ay0, ay1 = max(0, y0 - 5), min(H, y1 + 5)
    current = frames[frame_index, ay0:ay1, ax0:ax1].astype(np.int16)
    reference = clean_reference[ay0:ay1, ax0:ax1].astype(np.int16)
    ring = np.ones(current.shape[:2], dtype=bool)
    ring[y0-ay0:y1-ay0, x0-ax0:x1-ax0] = False
    delta = np.median((current - reference)[ring], axis=0)
    clean = np.clip(
        clean_reference[y0:y1, x0:x1].astype(float) + delta, 0, 255
    )
    return (x0, y0, x1, y1), clean


def ball_residual(frames, clean_reference, frame_index, center):
    \"\"\"Extract an additive RGB residual for ball, edge AA, and contact shadow.\"\"\"
    bounds, clean = predicted_clean_patch(
        frames, clean_reference, frame_index, center, radius=6
    )
    x0, y0, x1, y1 = bounds
    observed = frames[frame_index, y0:y1, x0:x1].astype(float)
    raw = observed - clean
    strength = np.max(np.abs(raw), axis=2)

    # Keep the component compact and suppress tiny background/codec variation.
    yy, xx = np.indices(strength.shape)
    geometric = ((xx + x0 - center[0]) ** 2 +
                 (yy + y0 - center[1]) ** 2) <= 25.0
    feather = np.clip((strength - 3.0) / 6.0, 0.0, 1.0) * geometric
    residual = raw * feather[:, :, None]
    return bounds, residual


def place_shifted_residual(frame, bounds, residual, dx, dy):
    \"\"\"Place a small residual with bilinear subpixel translation.\"\"\"
    x0, y0, x1, y1 = bounds
    rh, rw = residual.shape[:2]
    # Destination support, enlarged by one pixel for bilinear filtering.
    ox0 = max(0, int(np.floor(x0 + dx)) - 1)
    ox1 = min(W, int(np.ceil(x1 + dx)) + 1)
    oy0 = max(0, int(np.floor(y0 + dy)) - 1)
    oy1 = min(H, int(np.ceil(y1 + dy)) + 1)
    if ox0 >= ox1 or oy0 >= oy1:
        return

    oy, ox = np.indices((oy1 - oy0, ox1 - ox0), dtype=float)
    sx = ox + ox0 - dx - x0
    sy = oy + oy0 - dy - y0
    ix = np.floor(sx).astype(int)
    iy = np.floor(sy).astype(int)
    fx = sx - ix
    fy = sy - iy

    sampled = np.zeros((oy1 - oy0, ox1 - ox0, 3), dtype=float)
    for j, wy in ((0, 1.0 - fy), (1, fy)):
        for i, wx in ((0, 1.0 - fx), (1, fx)):
            px, py = ix + i, iy + j
            valid = (px >= 0) & (px < rw) & (py >= 0) & (py < rh)
            if np.any(valid):
                sampled[valid] += residual[py[valid], px[valid]] * \\
                                  (wx[valid] * wy[valid])[:, None]
    frame[oy0:oy1, ox0:ox1] += sampled


def generate():
    source = np.load(CASE_DIR / \"source_pixels.npz\")[\"rgb\"]
    assert source.shape == (N, H, W, 3) and source.dtype == np.uint8
    centers = track_ball(source)

    # Frames 0..19 contain no ball near the eventual floor path. Their median
    # is a stable, texture-preserving estimate of that occluded background.
    clean_reference = np.median(source[:20], axis=0).astype(np.uint8)
    output = source.astype(float).copy()

    contact_x, contact_y = centers[CONTACT]
    # The source has fully settled by frame 80; averaging the final tracked
    # centers avoids fitting to subpixel raster noise.
    settled_x = float(np.median(centers[80:, 0]))
    settled_y = float(np.median(centers[80:, 1]))

    for t in range(CONTACT + 1, N):
        # Remove the factual ball from this frame, retaining all other pixels.
        old_bounds, old_residual = ball_residual(
            source, clean_reference, t, centers[t]
        )
        x0, y0, x1, y1 = old_bounds
        output[t, y0:y1, x0:x1] -= old_residual

        elapsed = t - CONTACT
        source_time = CONTACT + FRICTION_MULTIPLIER * elapsed
        if source_time < 80.0:
            # Interpolate the source center at the accelerated physical time.
            lo = int(np.floor(source_time))
            hi = min(95, lo + 1)
            f = source_time - lo
            sampled_center = centers[lo] * (1.0 - f) + centers[hi] * f
            new_x = contact_x + (sampled_center[0] - contact_x) / \\
                    FRICTION_MULTIPLIER
            new_y = contact_y + (sampled_center[1] - contact_y) / \\
                    FRICTION_MULTIPLIER
            sprite_frame = int(round(source_time))
        else:
            # At 3x friction the ball has stopped; retain a fixed final phase.
            new_x = contact_x + (settled_x - contact_x) / FRICTION_MULTIPLIER
            new_y = contact_y + (settled_y - contact_y) / FRICTION_MULTIPLIER
            sprite_frame = 85

        sprite_bounds, sprite_residual = ball_residual(
            source, clean_reference, sprite_frame, centers[sprite_frame]
        )
        place_shifted_residual(
            output[t], sprite_bounds, sprite_residual,
            new_x - centers[sprite_frame, 0],
            new_y - centers[sprite_frame, 1],
        )

    output = np.clip(np.rint(output), 0, 255).astype(np.uint8)
    # Frame 28 is intentionally exact: it is the instant of first contact.
    assert np.array_equal(output[:CONTACT + 1], source[:CONTACT + 1])
    return output
