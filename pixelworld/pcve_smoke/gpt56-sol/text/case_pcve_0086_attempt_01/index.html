+"""Generate the 10x-target-mass counterfactual for this PixelWorld case.
+
+The light cue ball and dark target ball collide at about source frame 14.  Before
+contact, changing mass has no visible consequence, so those frames are retained
+verbatim.  For a one-dimensional elastic collision with masses m and 10m, the
+cue and target velocities are respectively -9/11 and 2/11 of the incoming cue
+velocity.  The source target's post-impact image-space velocity is used as the
+calibrated incoming velocity, preserving the source camera projection.
+
+Only local RGB operations are used: temporal-median background reconstruction,
+small masked inpainting, source-sprite extraction, and bilinear translation.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+N, H, W = 96, 144, 256
+
+
+def disk_mask(cx, cy, radius, shape=(H, W)):
+    yy, xx = np.ogrid[: shape[0], : shape[1]]
+    return ((xx - cx) ** 2 + (yy - cy) ** 2 <= radius**2).astype(np.uint8)
+
+
+def extract_sprite(frame, center, radius=11):
+    """Extract a softly antialiased ball/shadow sprite from one source frame."""
+    cx, cy = center
+    seed = disk_mask(cx, cy, radius)
+    clean = cv2.inpaint(frame, seed, 4, cv2.INPAINT_TELEA)
+    difference = np.max(np.abs(frame.astype(np.int16) - clean.astype(np.int16)), axis=2)
+
+    # Keep only changed pixels connected to the ball center.  This captures the
+    # rendered ball, its antialiased rim, and its compact contact shadow without
+    # carrying an opaque square of the old table background.
+    candidate = ((difference > 5) & (seed != 0)).astype(np.uint8)
+    candidate = cv2.morphologyEx(candidate, cv2.MORPH_CLOSE, np.ones((3, 3), np.uint8))
+    count, labels, _, _ = cv2.connectedComponentsWithStats(candidate, 8)
+    label_at_center = labels[int(round(cy)), int(round(cx))]
+    if label_at_center == 0:
+        # White-ball highlights can occasionally split at the center; select the
+        # largest component inside the tightly bounded extraction disk instead.
+        sizes = np.bincount(labels.ravel())
+        label_at_center = int(np.argmax(sizes[1:]) + 1) if count > 1 else 0
+    object_mask = (labels == label_at_center).astype(np.uint8)
+    object_mask = cv2.dilate(object_mask, np.ones((3, 3), np.uint8), iterations=1)
+    alpha = cv2.GaussianBlur(object_mask.astype(np.float32), (0, 0), 0.55)
+    alpha = np.clip(alpha, 0.0, 1.0)
+
+    # Use a full-frame premultiplied canvas.  This makes arbitrary fractional
+    # translations simple and avoids one-pixel stepping for the heavy target.
+    premultiplied = frame.astype(np.float32) * alpha[..., None]
+    return premultiplied, alpha
+
+
+def shifted_sprite(premultiplied, alpha, source_center, destination_center):
+    dx = float(destination_center[0] - source_center[0])
+    dy = float(destination_center[1] - source_center[1])
+    matrix = np.array([[1.0, 0.0, dx], [0.0, 1.0, dy]], dtype=np.float32)
+    warped_rgb = cv2.warpAffine(
+        premultiplied,
+        matrix,
+        (W, H),
+        flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=0,
+    )
+    warped_alpha = cv2.warpAffine(
+        alpha,
+        matrix,
+        (W, H),
+        flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=0,
+    )
+    return warped_rgb, np.clip(warped_alpha, 0.0, 1.0)
+
+
+def composite(frame, sprite_rgb, sprite_alpha):
+    a = sprite_alpha[..., None]
+    return frame * (1.0 - a) + sprite_rgb
+
+
+def factual_black_center(t):
+    """Subpixel center of the source's dark target after factual impact."""
+    if t < 14:
+        return np.array([127.7, 70.5], dtype=np.float32)
+    if t <= 45:
+        keys = np.array([14, 20, 30, 40, 45], dtype=np.float32)
+        xs = np.array([129.6, 139.3, 154.2, 164.5, 170.2], dtype=np.float32)
+        ys = np.array([69.9, 63.4, 55.2, 48.4, 46.2], dtype=np.float32)
+    else:
+        keys = np.array([46, 50, 54, 60, 64, 70, 75, 80, 86, 90, 95], dtype=np.float32)
+        xs = np.array([169.3, 166.3, 163.3, 159.0, 156.0, 152.0, 148.2, 144.3, 140.0, 137.6, 134.1], dtype=np.float32)
+        ys = np.array([46.5, 47.7, 49.2, 52.0, 54.0, 56.2, 58.3, 60.4, 63.0, 64.8, 66.7], dtype=np.float32)
+    return np.array([np.interp(t, keys, xs), np.interp(t, keys, ys)], dtype=np.float32)
+
+
+def clean_background(source):
+    """Build a clean, source-textured table for filling factual ball locations."""
+    background = np.median(source, axis=0).astype(np.uint8)
+    # The light ball is stationary after factual contact and therefore survives
+    # the temporal median.  Remove it once from the background plate.
+    occupied = disk_mask(125.5, 70.2, 11)
+    background = cv2.inpaint(background, occupied, 5, cv2.INPAINT_TELEA)
+    return background
+
+
+def erase_factual_objects(frame, background, t):
+    """Replace the two factual post-contact balls with clean source pixels."""
+    mask = disk_mask(125.5, 70.2, 10).astype(np.float32)
+    bx, by = factual_black_center(t)
+    mask = np.maximum(mask, disk_mask(float(bx), float(by), 10).astype(np.float32))
+
+    # A narrow feather keeps the source's subtle compression/noise variation at
+    # the patch boundary while fully removing ball cores and their shadows.
+    feather = cv2.GaussianBlur(mask, (0, 0), 0.8)
+    feather = np.clip(feather, 0.0, 1.0)
+
+    # Match the clean plate's local table tint to this source frame.  The source
+    # has tiny global illumination changes, so color correction avoids a visible
+    # static patch while retaining median-recovered rails/obstacles.
+    ring_outer = cv2.dilate((mask > 0).astype(np.uint8), np.ones((9, 9), np.uint8))
+    ring_inner = cv2.dilate((mask > 0).astype(np.uint8), np.ones((3, 3), np.uint8))
+    ring = (ring_outer != 0) & (ring_inner == 0)
+    if np.any(ring):
+        delta = np.median(frame[ring].astype(np.float32) - background[ring].astype(np.float32), axis=0)
+    else:
+        delta = np.zeros(3, dtype=np.float32)
+    fill = np.clip(background.astype(np.float32) + delta, 0, 255)
+    return frame.astype(np.float32) * (1.0 - feather[..., None]) + fill * feather[..., None]
+
+
+def generate():
+    source = np.load(ROOT / "source_pixels.npz")["rgb"]
+    assert source.shape == (N, H, W, 3) and source.dtype == np.uint8
+
+    # Source-calibrated sprite anchors.
+    white_anchor = np.array([85.4, 93.5], dtype=np.float32)
+    black_anchor = np.array([127.7, 70.5], dtype=np.float32)
+    white_sprite = extract_sprite(source[0], white_anchor, radius=12)
+    black_sprite = extract_sprite(source[0], black_anchor, radius=11)
+    background = clean_background(source)
+
+    output = source.copy()
+    contact_frame = 14
+    white_contact = np.array([124.0, 71.2], dtype=np.float32)
+    black_contact = np.array([128.2, 70.4], dtype=np.float32)
+
+    # Average factual target velocity before it meets the far bumper.  It is the
+    # source-projected incoming cue velocity for this near-collinear collision.
+    incoming = np.array([1.33, -0.79], dtype=np.float32)
+    white_velocity = (-9.0 / 11.0) * incoming
+    black_velocity = (2.0 / 11.0) * incoming
+
+    for t in range(contact_frame, N):
+        dt = float(t - contact_frame)
+        white_position = white_contact + white_velocity * dt
+        black_position = black_contact + black_velocity * dt
+
+        frame = erase_factual_objects(source[t], background, t)
+        wrgb, wa = shifted_sprite(*white_sprite, white_anchor, white_position)
+        brgb, ba = shifted_sprite(*black_sprite, black_anchor, black_position)
+
+        # The upper ball is painted first; the lower ball is closer to camera.
+        if white_position[1] >= black_position[1]:
+            frame = composite(frame, brgb, ba)
+            frame = composite(frame, wrgb, wa)
+        else:
+            frame = composite(frame, wrgb, wa)
+            frame = composite(frame, brgb, ba)
+        output[t] = np.clip(np.rint(frame), 0, 255).astype(np.uint8)
+
+    return output
+
+
+if __name__ == "__main__":
+    frames = generate()
+    artifact_io.save(frames, ROOT)
+    print("saved", frames.shape, frames.dtype, "to", ROOT)
