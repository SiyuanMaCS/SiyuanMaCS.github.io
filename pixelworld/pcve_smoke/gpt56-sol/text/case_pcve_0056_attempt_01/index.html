+#!/usr/bin/env python3
+"""Generate the requested counterfactual RGB video.
+
+The edit is entirely numeric.  The prominent blue rigid body is segmented from
+its blue chroma, completed behind its original red occluder, removed from its
+factual pose, and translated so that its centre is 25% of the way from the
+initial yellow-stone centre to the central red-stone centre.  Native source
+texture and frame-to-frame illumination are retained.
+"""
+from pathlib import Path
+import cv2
+import numpy as np
+import artifact_io
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def largest_filled(mask: np.ndarray) -> np.ndarray:
+    """Filled external contour of the largest connected region."""
+    contours, _ = cv2.findContours(mask.astype(np.uint8), cv2.RETR_EXTERNAL,
+                                   cv2.CHAIN_APPROX_SIMPLE)
+    if not contours:
+        raise RuntimeError("expected object contour was not found")
+    contour = max(contours, key=cv2.contourArea)
+    out = np.zeros(mask.shape, np.uint8)
+    cv2.drawContours(out, [contour], -1, 255, cv2.FILLED)
+    return out
+
+
+def component_near(mask: np.ndarray, point, largest=False) -> np.ndarray:
+    """Return a filled chroma component, selected by size or a nearby point."""
+    n, labels, stats, centers = cv2.connectedComponentsWithStats(
+        mask.astype(np.uint8), 8)
+    ids = [i for i in range(1, n) if stats[i, cv2.CC_STAT_AREA] >= 3]
+    if not ids:
+        raise RuntimeError("expected reference stone was not found")
+    if largest:
+        idx = max(ids, key=lambda i: stats[i, cv2.CC_STAT_AREA])
+    else:
+        px, py = point
+        idx = min(ids, key=lambda i: (centers[i, 0]-px)**2 +
+                                      (centers[i, 1]-py)**2)
+    raw = (labels == idx).astype(np.uint8)
+    return largest_filled(raw)
+
+
+def centroid(mask: np.ndarray) -> np.ndarray:
+    m = cv2.moments(mask.astype(np.float32))
+    if m["m00"] == 0:
+        raise RuntimeError("empty object mask")
+    return np.array([m["m10"] / m["m00"], m["m01"] / m["m00"]],
+                    np.float32)
+
+
+def feather(mask: np.ndarray, sigma: float) -> np.ndarray:
+    a = mask.astype(np.float32) / 255.0
+    return cv2.GaussianBlur(a, (0, 0), sigma).clip(0, 1)
+
+
+def background_fill(frame: np.ndarray, object_mask: np.ndarray) -> np.ndarray:
+    """Reconstruct the smooth set behind the old large blue body.
+
+    The set is nearly planar but has a mild per-row illumination gradient.
+    Robust side-strip samples preserve that gradient without copying either
+    moving small stone into the repaired region.
+    """
+    h, w = object_mask.shape
+    ys, xs = np.where(object_mask > 0)
+    x0, x1 = int(xs.min()), int(xs.max())
+    y0, y1 = int(ys.min()), int(ys.max())
+    result = frame.astype(np.float32).copy()
+    f = frame.astype(np.float32)
+
+    left0, left1 = max(0, x0-15), max(1, x0-5)
+    right0, right1 = min(w-1, x1+6), min(w, x1+16)
+    # Gentle deterministic texture prevents a conspicuously flat repair.
+    texture = np.array([[-0.55, 0.15, 0.45, -0.10],
+                        [ 0.25,-0.35, 0.05, 0.35],
+                        [ 0.05, 0.40,-0.40, 0.00]], np.float32)
+    for y in range(max(0, y0-2), min(h, y1+3)):
+        lstrip = f[max(0, y-1):min(h, y+2), left0:left1]
+        rstrip = f[max(0, y-1):min(h, y+2), right0:right1]
+        left = np.median(lstrip.reshape(-1, 3), axis=0)
+        right = np.median(rstrip.reshape(-1, 3), axis=0)
+        xx = np.arange(x0, x1+1, dtype=np.float32)
+        q = ((xx-x0) / max(1, x1-x0))[:, None]
+        row = left[None, :] * (1-q) + right[None, :] * q
+        # The broad background is smooth; use sub-code-value dither only.
+        row += texture[:, (np.arange(len(xx)) + y) % 4].T
+        result[y, x0:x1+1] = row
+
+    alpha = feather(object_mask, 1.15)[..., None]
+    return f * (1-alpha) + result * alpha
+
+
+def foreground_masks(frame: np.ndarray):
+    """Masks for the red/yellow bodies which visually sit above blue."""
+    a = frame.astype(np.int16)
+    red_chroma = ((a[..., 0]-a[..., 1] > 38) &
+                  (a[..., 0]-a[..., 2] > 30))
+    yellow_chroma = ((a[..., 0]-a[..., 2] > 28) &
+                     (a[..., 1]-a[..., 2] > 28) &
+                     (np.abs(a[..., 0]-a[..., 1]) < 48))
+
+    # Central red body is the largest component.  Filling its external contour
+    # includes its neutral specular highlight.
+    central_red = component_near(red_chroma, (128, 75), largest=True)
+
+    # Small moving saturated red body lies to the left initially.  If it has
+    # merged visually with the central red body, its pixels are already covered
+    # by central_red and no separate mask is needed.
+    n, lab, stats, centers = cv2.connectedComponentsWithStats(
+        red_chroma.astype(np.uint8), 8)
+    small_red = np.zeros(red_chroma.shape, np.uint8)
+    candidates = [i for i in range(1, n)
+                  if 3 <= stats[i, cv2.CC_STAT_AREA] < 120 and centers[i, 0] < 125]
+    if candidates:
+        idx = max(candidates, key=lambda i: stats[i, cv2.CC_STAT_AREA])
+        small_red = largest_filled((lab == idx).astype(np.uint8))
+
+    yellow = component_near(yellow_chroma, (205, 76), largest=False)
+
+    # A one-pixel expansion captures antialiasing and contact shadow fringe.
+    k = np.ones((3, 3), np.uint8)
+    combined = cv2.dilate(np.maximum.reduce([central_red, small_red, yellow]), k)
+    return combined, central_red, yellow
+
+
+def build_frames(source: np.ndarray) -> np.ndarray:
+    first = source[0]
+    a0 = first.astype(np.int16)
+
+    # Strong blue chroma cleanly identifies the large blue body.  Its filled
+    # external contour restores the area hidden by the central red body.
+    strong_blue = ((a0[..., 2]-a0[..., 0] > 52) &
+                   (a0[..., 1]-a0[..., 0] > 28))
+    blue_shape = largest_filled(strong_blue)
+    blue_shape = cv2.morphologyEx(blue_shape, cv2.MORPH_CLOSE,
+                                  np.ones((5, 5), np.uint8))
+    blue_center = centroid(blue_shape)
+
+    _, red0, yellow0 = foreground_masks(first)
+    red_center = centroid(red0)
+    yellow_center = centroid(yellow0)
+    target = yellow_center + 0.25 * (red_center-yellow_center)
+    delta = target-blue_center
+    transform = np.array([[1.0, 0.0, float(delta[0])],
+                          [0.0, 1.0, float(delta[1])]], np.float32)
+
+    h, w = first.shape[:2]
+    out = np.empty_like(source)
+    blue_alpha0 = feather(blue_shape, 0.75)
+
+    for t, frame in enumerate(source):
+        foreground, _, _ = foreground_masks(frame)
+
+        # Rebuild the old footprint first.
+        base = background_fill(frame, blue_shape)
+
+        # Complete the blue texture behind all non-blue foreground occluders.
+        # Telea works well here because the rigid body is smoothly lit.
+        repair = cv2.dilate(foreground, np.ones((3, 3), np.uint8))
+        repair = cv2.bitwise_and(repair, blue_shape)
+        texture = cv2.inpaint(frame, repair, 7.0, cv2.INPAINT_TELEA)
+
+        moved_rgb = cv2.warpAffine(texture, transform, (w, h),
+                                   flags=cv2.INTER_LINEAR,
+                                   borderMode=cv2.BORDER_REFLECT_101)
+        moved_alpha = cv2.warpAffine(blue_alpha0, transform, (w, h),
+                                     flags=cv2.INTER_LINEAR,
+                                     borderMode=cv2.BORDER_CONSTANT,
+                                     borderValue=0).clip(0, 1)
+        composed = base * (1-moved_alpha[..., None]) + \
+                   moved_rgb.astype(np.float32) * moved_alpha[..., None]
+
+        # Preserve exact native rendering of all reference stones in front.
+        fg_alpha = feather(foreground, 0.55)[..., None]
+        composed = composed * (1-fg_alpha) + frame.astype(np.float32) * fg_alpha
+        out[t] = np.clip(np.rint(composed), 0, 255).astype(np.uint8)
+
+    print("blue_center", blue_center.tolist())
+    print("red_center", red_center.tolist())
+    print("yellow_center", yellow_center.tolist())
+    print("target_center", target.tolist())
+    print("translation", delta.tolist())
+    return out
+
+
+def main():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3) and source.dtype == np.uint8
+    frames = build_frames(source)
+    artifact_io.save(frames, CASE_DIR)
+    print("saved", frames.shape, frames.dtype, "to", CASE_DIR)
+
+
+if __name__ == "__main__":
+    main()

#"'!/usr/bin/env python3
"""Generate the requested counterfactual RGB video.

The edit is entirely numeric.  The prominent blue rigid body is segmented from
its blue chroma, completed behind its original red occluder, removed from its
factual pose, and translated so that its centre is 25% of the way from the
initial yellow-stone centre to the central red-stone centre.  Native source
texture and frame-to-frame illumination are retained.
"""
from pathlib import Path
import cv2
import numpy as np
import artifact_io

CASE_DIR = Path(__file__).resolve().parent


def largest_filled(mask: np.ndarray) -> np.ndarray:
    """Filled external contour of the largest connected region."""
    contours, _ = cv2.findContours(mask.astype(np.uint8), cv2.RETR_EXTERNAL,
                                   cv2.CHAIN_APPROX_SIMPLE)
    if not contours:
        raise RuntimeError("expected object contour was not found")
    contour = max(contours, key=cv2.contourArea)
    out = np.zeros(mask.shape, np.uint8)
    cv2.drawContours(out, [contour], -1, 255, cv2.FILLED)
    return out


def component_near(mask: np.ndarray, point, largest=False) -> np.ndarray:
    """Return a filled chroma component, selected by size or a nearby point."""
    n, labels, stats, centers = cv2.connectedComponentsWithStats(
        mask.astype(np.uint8), 8)
    ids = [i for i in range(1, n) if stats[i, cv2.CC_STAT_AREA] >= 3]
    if not ids:
        raise RuntimeError("expected reference stone was not found")
    if largest:
        idx = max(ids, key=lambda i: stats[i, cv2.CC_STAT_AREA])
    else:
        px, py = point
        idx = min(ids, key=lambda i: (centers[i, 0]-px)**2 +
                                      (centers[i, 1]-py)**2)
    raw = (labels == idx).astype(np.uint8)
    return largest_filled(raw)


def centroid(mask: np.ndarray) -> np.ndarray:
    m = cv2.moments(mask.astype(np.float32))
    if m["m00"] == 0:
        raise RuntimeError("empty object mask")
    return np.array([m["m10"] / m["m00"], m["m01"] / m["m00"]],
                    np.float32)


def feather(mask: np.ndarray, sigma: float) -> np.ndarray:
    a = mask.astype(np.float32) / 255.0
    return cv2.GaussianBlur(a, (0, 0), sigma).clip(0, 1)


def background_fill(frame: np.ndarray, object_mask: np.ndarray) -> np.ndarray:
    """Reconstruct the smooth set behind the old large blue body.

    The set is nearly planar but has a mild per-row illumination gradient.
    Robust side-strip samples preserve that gradient without copying either
    moving small stone into the repaired region.
    """
    h, w = object_mask.shape
    ys, xs = np.where(object_mask > 0)
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    result = frame.astype(np.float32).copy()
    f = frame.astype(np.float32)

    left0, left1 = max(0, x0-15), max(1, x0-5)
    right0, right1 = min(w-1, x1+6), min(w, x1+16)
    # Gentle deterministic texture prevents a conspicuously flat repair.
    texture = np.array([[-0.55, 0.15, 0.45, -0.10],
                        [ 0.25,-0.35, 0.05, 0.35],
                        [ 0.05, 0.40,-0.40, 0.00]], np.float32)
    for y in range(max(0, y0-2), min(h, y1+3)):
        lstrip = f[max(0, y-1):min(h, y+2), left0:left1]
        rstrip = f[max(0, y-1):min(h, y+2), right0:right1]
        left = np.median(lstrip.reshape(-1, 3), axis=0)
        right = np.median(rstrip.reshape(-1, 3), axis=0)
        xx = np.arange(x0, x1+1, dtype=np.float32)
        q = ((xx-x0) / max(1, x1-x0))[:, None]
        row = left[None, :] * (1-q) + right[None, :] * q
        # The broad background is smooth; use sub-code-value dither only.
        row += texture[:, (np.arange(len(xx)) + y) % 4].T
        result[y, x0:x1+1] = row

    alpha = feather(object_mask, 1.15)[..., None]
    return f * (1-alpha) + result * alpha


def foreground_masks(frame: np.ndarray):
    """Masks for the red/yellow bodies which visually sit above blue."""
    a = frame.astype(np.int16)
    red_chroma = ((a[..., 0]-a[..., 1] > 38) &
                  (a[..., 0]-a[..., 2] > 30))
    yellow_chroma = ((a[..., 0]-a[..., 2] > 28) &
                     (a[..., 1]-a[..., 2] > 28) &
                     (np.abs(a[..., 0]-a[..., 1]) < 48))

    # Central red body is the largest component.  Filling its external contour
    # includes its neutral specular highlight.
    central_red = component_near(red_chroma, (128, 75), largest=True)

    # Small moving saturated red body lies to the left initially.  If it has
    # merged visually with the central red body, its pixels are already covered
    # by central_red and no separate mask is needed.
    n, lab, stats, centers = cv2.connectedComponentsWithStats(
        red_chroma.astype(np.uint8), 8)
    small_red = np.zeros(red_chroma.shape, np.uint8)
    candidates = [i for i in range(1, n)
                  if 3 <= stats[i, cv2.CC_STAT_AREA] < 120 and centers[i, 0] < 125]
    if candidates:
        idx = max(candidates, key=lambda i: stats[i, cv2.CC_STAT_AREA])
        small_red = largest_filled((lab == idx).astype(np.uint8))

    yellow = component_near(yellow_chroma, (205, 76), largest=False)

    # A one-pixel expansion captures antialiasing and contact shadow fringe.
    k = np.ones((3, 3), np.uint8)
    combined = cv2.dilate(np.maximum.reduce([central_red, small_red, yellow]), k)
    return combined, central_red, yellow


def build_frames(source: np.ndarray) -> np.ndarray:
    first = source[0]
    a0 = first.astype(np.int16)

    # Strong blue chroma cleanly identifies the large blue body.  Its filled
    # external contour restores the area hidden by the central red body.
    strong_blue = ((a0[..., 2]-a0[..., 0] > 52) &
                   (a0[..., 1]-a0[..., 0] > 28))
    blue_shape = largest_filled(strong_blue)
    blue_shape = cv2.morphologyEx(blue_shape, cv2.MORPH_CLOSE,
                                  np.ones((5, 5), np.uint8))
    blue_center = centroid(blue_shape)

    _, red0, yellow0 = foreground_masks(first)
    red_center = centroid(red0)
    yellow_center = centroid(yellow0)
    target = yellow_center + 0.25 * (red_center-yellow_center)
    delta = target-blue_center
    transform = np.array([[1.0, 0.0, float(delta[0])],
                          [0.0, 1.0, float(delta[1])]], np.float32)

    h, w = first.shape[:2]
    out = np.empty_like(source)
    blue_alpha0 = feather(blue_shape, 0.75)

    for t, frame in enumerate(source):
        foreground, _, _ = foreground_masks(frame)

        # Rebuild the old footprint first.
        base = background_fill(frame, blue_shape)
