+"""Generate the requested counterfactual pool-table video.
+
+The source contains a cue ball travelling directly toward a small dark target
+ball.  The intervention inserts the yellow 1-ball one quarter of the way down
+that line at frame 1.  Equal-mass, nearly head-on impacts therefore produce a
+three-stage motion: cue -> 1-ball -> target, followed by the target's cushion
+return and a second target -> 1-ball impact late in the shot.
+
+Only source pixels and small numerical RGB operations are used.  Existing
+balls are removed with clean temporal patches (or a local polynomial fill at
+the one location that is occupied in every source frame), and source ball
+sprites are resampled with subpixel placement to retain the original look.
+"""

+def interp_keys(keys, t):
+    """Piecewise-linear 2-D interpolation, with linear extrapolation."""
+    k = np.asarray(keys, dtype=np.float64)
+    if t <= k[0, 0]:
+        return k[0, 1:].copy()
+    if t >= k[-1, 0]:
+        a, b = k[-2], k[-1]
+        f = (t - b[0]) / (b[0] - a[0])
+        return b[1:] + f * (b[1:] - a[1:])
+    j = int(np.searchsorted(k[:, 0], t))
+    a, b = k[j - 1], k[j]
+    f = (t - a[0]) / (b[0] - a[0])
+    return a[1:] * (1.0 - f) + b[1:] * f
+

+# the cue is about seven pixels wide and the dark target about six.

+def bilinear(image, xs, ys):
+    """Sample an RGB image at arrays of floating point coordinates."""
+    xs = np.clip(xs, 0.0, image.shape[1] - 1.001)
+    ys = np.clip(ys, 0.0, image.shape[0] - 1.001)
+    x0 = np.floor(xs).astype(np.int32)
+    y0 = np.floor(ys).astype(np.int32)
+    x1 = x0 + 1
+    y1 = y0 + 1
+    fx = (xs - x0)[..., None]
+    fy = (ys - y0)[..., None]
+    a = image[y0, x0].astype(np.float64)
+    b = image[y0, x1].astype(np.float64)
+    c = image[y1, x0].astype(np.float64)
+    d = image[y1, x1].astype(np.float64)
+    return (a * (1 - fx) + b * fx) * (1 - fy) + (c * (1 - fx) + d * fx) * fy
+
+
+def paste_shifted(dst, donor, src_center, dst_center, radius, solid_radius=None):
+    """Paste a feathered, subpixel-translated circular source patch."""
+    sx, sy = map(float, src_center)
+    dx, dy = map(float, dst_center)
+    pad = int(math.ceil(radius + 2.0))
+    x0, x1 = max(0, int(math.floor(dx)) - pad), min(W, int(math.floor(dx)) + pad + 1)
+    y0, y1 = max(0, int(math.floor(dy)) - pad), min(H, int(math.floor(dy)) + pad + 1)
+    yy, xx = np.mgrid[y0:y1, x0:x1].astype(np.float64)
+    rr = np.sqrt((xx - dx) ** 2 + (yy - dy) ** 2)
+    inner = radius - 1.15 if solid_radius is None else solid_radius
+    alpha = np.clip((radius - rr) / max(0.01, radius - inner), 0.0, 1.0)[..., None]
+    sample = bilinear(donor, sx + xx - dx, sy + yy - dy)
+    old = dst[y0:y1, x0:x1].astype(np.float64)
+    dst[y0:y1, x0:x1] = np.clip(old * (1.0 - alpha) + sample * alpha, 0, 255).astype(np.uint8)
+
+
+def local_polynomial_fill(frame, center, radius):
+    """Fill a tiny occupied disk from a quadratic fit to its surrounding ring."""
+    cx, cy = map(float, center)
+    outer = radius + 4.5
+    x0, x1 = max(0, int(cx - outer - 1)), min(W, int(cx + outer + 2))
+    y0, y1 = max(0, int(cy - outer - 1)), min(H, int(cy + outer + 2))
+    yy, xx = np.mgrid[y0:y1, x0:x1].astype(np.float64)
+    dx, dy = xx - cx, yy - cy
+    rr = np.sqrt(dx * dx + dy * dy)
+    ring = (rr >= radius + 1.0) & (rr <= outer)
+    # Quadratic terms retain the mild table-lighting gradient without making
+    # a conspicuous flat green disk.
+    X = np.stack([np.ones(ring.sum()), dx[ring], dy[ring],
+                  dx[ring] ** 2, dx[ring] * dy[ring], dy[ring] ** 2], axis=1)
+    target = frame[y0:y1, x0:x1].astype(np.float64)
+    filled = target.copy()
+    inside = rr <= radius + 0.7
+    Xi = np.stack([np.ones(inside.sum()), dx[inside], dy[inside],
+                   dx[inside] ** 2, dx[inside] * dy[inside], dy[inside] ** 2], axis=1)
+    for ch in range(3):
+        coef = np.linalg.lstsq(X, target[..., ch][ring], rcond=None)[0]
+        filled[..., ch][inside] = Xi @ coef
+    # Feather only the last 1.2 pixels to preserve source texture outside.
+    a = np.clip((radius + 0.7 - rr) / 1.2, 0.0, 1.0)[..., None]
+    mixed = target * (1.0 - a) + filled * a
+    frame[y0:y1, x0:x1] = np.clip(mixed, 0, 255).astype(np.uint8)
+
+
+def source_cue(t):
+    return interp_keys(CUE_KEYS, float(t))
+
+
+def source_target(t):
+    return interp_keys(TARGET_KEYS, float(t))

+def remove_source_ball(frame, source, center, radius, cue_track, target_track):
+    """Replace a source ball by the cleanest same-location temporal patch."""
+    cx, cy = center
+    best_t, best_clearance = 0, -1.0
+    # All source frames are available; a sparse search is enough and favors a
+    # temporally nearby clean patch when clearances tie.
+    for tt in range(N):
+        clearance = min(np.linalg.norm(np.asarray((cx, cy)) - cue_track[tt]),
+                        np.linalg.norm(np.asarray((cx, cy)) - target_track[tt]))
+        if clearance > best_clearance:
+            best_t, best_clearance = tt, clearance
+    if best_clearance >= radius + 2.2:
+        paste_shifted(frame, source[best_t], center, center, radius + 0.9)
+    else:
+        local_polynomial_fill(frame, center, radius)
+
+
+def smoothstep(x):
+    x = float(np.clip(x, 0.0, 1.0))
+    return x * x * (3.0 - 2.0 * x)
+

+def draw_one_ball(frame, center, frame_index):
+    """Numerically shade an antialiased yellow one-ball and its felt shadow."""
+    cx, cy = map(float, center)
+    pad = 7
+    x0, x1 = max(0, int(cx) - pad), min(W, int(cx) + pad + 1)
+    y0, y1 = max(0, int(cy) - pad), min(H, int(cy) + pad + 1)
+    region = frame[y0:y1, x0:x1].astype(np.float64)
+
+    # Soft contact shadow, offset toward the lower right as in the source.
+    yy, xx = np.mgrid[y0:y1, x0:x1].astype(np.float64)
+    ex = (xx - (cx + 0.8)) / 4.3
+    ey = (yy - (cy + 1.75)) / 2.15
+    e = ex * ex + ey * ey
+    shadow = np.clip(1.0 - e, 0.0, 1.0) ** 1.7 * 0.31
+    shadow_tint = region * np.array([0.55, 0.67, 0.56])[None, None, :]
+    region = region * (1.0 - shadow[..., None]) + shadow_tint * shadow[..., None]
+
+    # Four-by-four supersampling is inexpensive for this seven-pixel ball and
+    # gives a considerably cleaner silhouette than a raster ellipse.
+    accum = np.zeros_like(region)
+    cover = np.zeros(region.shape[:2], dtype=np.float64)
+    r = 3.55
+    light = np.array([-0.47, -0.58, 0.66], dtype=np.float64)
+    light /= np.linalg.norm(light)
+    offsets = (np.arange(4) + 0.5) / 4.0 - 0.5
+    for oy in offsets:
+        for ox in offsets:
+            dx = xx + ox - cx
+            dy = yy + oy - cy
+            q = (dx * dx + dy * dy) / (r * r)
+            inside = q <= 1.0
+            if not np.any(inside):
+                continue
+            nz = np.sqrt(np.clip(1.0 - q, 0.0, 1.0))
+            nx, ny = dx / r, dy / r
+            lambert = np.clip(nx * light[0] + ny * light[1] + nz * light[2], 0.0, 1.0)
+            shade = 0.48 + 0.52 * lambert
+            base = np.stack([239.0 * shade, 190.0 * shade, 24.0 * shade], axis=2)
+            # Broad warm reflection and compact white specular highlight.
+            spec = np.exp(-((dx + 1.25) ** 2 + (dy + 1.25) ** 2) / 0.72)
+            base += spec[..., None] * np.array([25.0, 46.0, 76.0])[None, None, :]
+
+            # Small white number medallion with a dark vertical numeral.  Its
+            # warm white is deliberately not pure, matching the source grade.
+            px, py = 0.42, 0.22
+            badge = ((dx - px) ** 2 + (dy - py) ** 2) <= 1.18 ** 2
+            badge_col = np.stack([236.0 * shade + 13.0,
+                                  232.0 * shade + 12.0,
+                                  201.0 * shade + 10.0], axis=2)
+            base = np.where(badge[..., None], badge_col, base)
+            numeral = badge & (np.abs(dx - (px + 0.08)) < 0.19) & (dy > py - 0.58) & (dy < py + 0.66)
+            base = np.where(numeral[..., None], np.array([42.0, 38.0, 27.0]), base)
+            accum += np.where(inside[..., None], base, 0.0)
+            cover += inside.astype(np.float64)
+
+    alpha = cover / 16.0
+    ball_rgb = accum / np.maximum(cover[..., None], 1.0)
+    result = region * (1.0 - alpha[..., None]) + ball_rgb * alpha[..., None]
+    frame[y0:y1, x0:x1] = np.clip(result, 0, 247).astype(np.uint8)
+

+def generate():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (N, H, W, 3) and source.dtype == np.uint8
+
+    cue_track = np.stack([source_cue(t) for t in range(N)])
+    target_track = np.stack([source_target(t) for t in range(N)])
+
+    cue0 = source_cue(0)
+    target0 = source_target(0)
+    line = target0 - cue0
+    line_len = float(np.linalg.norm(line))
+    direction = line / line_len
+
+    # Exact requested intervention point: one quarter from cue to target.
+    yellow0 = cue0 + 0.25 * line
+    contact_distance = 6.20
+    cue_stop = yellow0 - contact_distance * direction
+    yellow_stop = target0 - contact_distance * direction
+
+    first_yellow_contact = 1.45
+    first_target_contact = 10.0
+    target_time_scale = 1.20
+    source_return_contact = 104.1
+    second_contact = first_target_contact + (source_return_contact - 13.0) / target_time_scale
+    return_velocity = np.array([-0.71, 0.42]) * target_time_scale
+
+    out = np.empty_like(source)
+    for t in range(N):
+        frame = source[t].copy()
+
+        # Remove both factual moving objects before laying down their
+        # counterfactual positions.  Target first avoids preserving collision
+        # overlap from the factual frame 13-14 transition.
+        remove_source_ball(frame, source, target_track[t], 5.25, cue_track, target_track)
+        remove_source_ball(frame, source, cue_track[t], 5.80, cue_track, target_track)
+
+        # Counterfactual cue: factual motion until its early contact with the
+        # inserted ball, then rest at the equal-mass transfer position.
+        if t <= 1:
+            cue_pos = source_cue(t)
+            cue_donor_t = t
+        else:
+            cue_pos = cue_stop
+            cue_donor_t = 1
+        paste_shifted(frame, source[cue_donor_t], source_cue(cue_donor_t), cue_pos, 6.35)
+
+        # Counterfactual target.  It rests through the first chain collision,
+        # follows the source cushion trajectory at the slightly greater speed
+        # retained by the nearer 1-ball, then stops when it transfers momentum
+        # back to the 1-ball on the return leg.
+        if t <= first_target_contact:
+            target_pos = target0
+            donor_t = 0
+            donor_pos = target0
+        elif t < second_contact:
+            virtual_t = 13.0 + target_time_scale * (t - first_target_contact)
+            target_pos = source_target(virtual_t)
+            donor_t = min(95, max(0, int(round(virtual_t))))
+            donor_pos = source_target(donor_t)
+        else:
+            target_pos = target0
+            donor_t = 0
+            donor_pos = target0
+        paste_shifted(frame, source[donor_t], donor_pos, target_pos, 7.10)
+
+        # Yellow 1-ball: stationary at insertion, moving between the first two
+        # impacts, resting while the target visits the cushion, then moving
+        # back down-table after the late return impact.
+        if t <= first_yellow_contact:
+            yellow_pos = yellow0
+        elif t <= first_target_contact:
+            f = (t - first_yellow_contact) / (first_target_contact - first_yellow_contact)
+            # Mild deceleration while rolling; normalized so contact is exact.
+            f = 1.0 - (1.0 - np.clip(f, 0.0, 1.0)) ** 1.12
+            yellow_pos = yellow0 * (1.0 - f) + yellow_stop * f
+        elif t <= second_contact:
+            yellow_pos = yellow_stop
+        else:
+            yellow_pos = yellow_stop + return_velocity * (t - second_contact)
+        draw_one_ball(frame, yellow_pos, t)
+
+        out[t] = frame
+
+    return out
+
