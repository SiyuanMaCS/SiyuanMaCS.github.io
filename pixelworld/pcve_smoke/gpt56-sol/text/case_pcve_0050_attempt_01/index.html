+"""Generate the counterfactual clip for this case.
+
+The source shot is static apart from two small moving stones.  The red stone
+travels from the left and reaches the central obstacle at about source frame
+30.  To double only its initial speed, its free-flight appearance is sampled
+at twice source time.  Once it reaches the obstacle, the left-hand contact
+sequence is advanced by the corresponding 15 frames while the yellow stone
+and the remainder of the shot stay on their original clock.
+
+All compositing is done from source RGB pixels; in particular, the gently
+changing size, shading, and contact shadow of the red stone are retained.
+"""
+
+from pathlib import Path
+
+import numpy as np
+from scipy.ndimage import gaussian_filter
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def red_center(frame_number: float) -> tuple[float, float]:
+    """Center of the incoming red-stone sprite on the 256x144 raster.
+
+    This is a least-squares description of its center in source frames 0..29.
+    The small vertical component reflects the perspective of the surface.
+    """
+
+    return 47.35 + 1.985 * frame_number, 67.75 + 0.132 * frame_number
+
+
+def ellipse_alpha(height: int, width: int, cx: float, cy: float,
+                  rx: float, ry: float) -> np.ndarray:
+    """Feathered ellipse which includes both the stone and contact shadow."""
+
+    yy, xx = np.mgrid[:height, :width]
+    radius = np.sqrt(((xx - cx) / rx) ** 2 + ((yy - cy) / ry) ** 2)
+    # Opaque over the object, then a short cosine-like (Gaussian-smoothed)
+    # transition over the surrounding source pixels.
+    alpha = np.clip((1.14 - radius) / 0.27, 0.0, 1.0)
+    return gaussian_filter(alpha.astype(np.float32), 0.45)
+
+
+def left_contact_alpha(height: int, width: int) -> np.ndarray:
+    """Soft region containing only the red stone's side of the collision."""
+
+    yy, xx = np.mgrid[:height, :width]
+    # Interior spans the incoming path and the left half of the central stone.
+    ax = np.minimum(np.clip((xx - 94.0) / 3.0, 0, 1),
+                    np.clip((131.0 - xx) / 3.0, 0, 1))
+    ay = np.minimum(np.clip((yy - 60.0) / 3.0, 0, 1),
+                    np.clip((90.0 - yy) / 3.0, 0, 1))
+    alpha = np.minimum(ax, ay).astype(np.float32)
+    return gaussian_filter(alpha, 0.55)
+
+
+def blend(dst: np.ndarray, src: np.ndarray, alpha: np.ndarray) -> np.ndarray:
+    a = alpha[..., None]
+    return dst * (1.0 - a) + src * a
+
+
+def build_frames() -> np.ndarray:
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3)
+
+    # The first 20 source frames provide a clean, robust plate at every point
+    # along the incoming path: a moving stone occupies any one pixel for only
+    # a few frames, while the central object has not yet been touched.
+    clean_plate = np.median(source[:20].astype(np.float32), axis=0)
+    out = source.astype(np.float32).copy()
+    h, w = source.shape[1:3]
+
+    # Frame zero has the same position in both histories.  For frames 1..14,
+    # erase the factual red stone and paste its source appearance from 2*t.
+    # This gives exactly twice the translational velocity without redrawing or
+    # rescaling the shaded sprite.
+    for t in range(1, 15):
+        old_cx, old_cy = red_center(t)
+        old_a = ellipse_alpha(h, w, old_cx, old_cy, 9.0, 6.6)
+        out[t] = blend(out[t], clean_plate, old_a)
+
+        sampled_t = 2 * t
+        new_cx, new_cy = red_center(sampled_t)
+        new_a = ellipse_alpha(h, w, new_cx, new_cy, 9.3, 6.7)
+        out[t] = blend(out[t], source[sampled_t].astype(np.float32), new_a)
+
+    # At output frame 15 the faster stone reaches the obstacle.  Remove the
+    # slower factual in-flight stone through frame 29.  Its replacement is the
+    # contact state below, so no duplicate red stone remains in the scene.
+    for t in range(15, 30):
+        old_cx, old_cy = red_center(t)
+        old_a = ellipse_alpha(h, w, old_cx, old_cy, 9.2, 6.7)
+        out[t] = blend(out[t], clean_plate, old_a)
+
+    # Advance only the left-hand contact history by 15 frames.  The feathered
+    # right edge lies at the center of the obstacle, preserving the yellow
+    # stone's original approach and impact on the right.  After source frame
+    # 95, the settled final contact appearance is held.
+    contact_a = left_contact_alpha(h, w)
+    for t in range(15, 96):
+        advanced = min(t + 15, 95)
+        out[t] = blend(out[t], source[advanced].astype(np.float32), contact_a)
+
+    # Preserve the factual initial image exactly and quantize once at the end.
+    out[0] = source[0]
+    return np.clip(np.rint(out), 0, 255).astype(np.uint8)
+
+
+if __name__ == "__main__":
+    frames = build_frames()
+    artifact_io.save(frames, CASE_DIR)
+    print(f"saved {len(frames)} frames to {CASE_DIR}")
