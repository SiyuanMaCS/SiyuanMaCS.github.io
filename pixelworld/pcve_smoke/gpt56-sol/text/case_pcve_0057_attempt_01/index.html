+"""Generate the 4x-mass cola-can counterfactual video.
+
+The source depicts a one-dimensional, strongly inelastic chain collision.  The
+red cola can is the incoming object.  Before its first contact its mass cannot
+change its kinematics, so frames 0--4 are preserved exactly.  After contact we
+retime the source's rendered collision states using the momentum-conserving
+speed ratios for a four-unit incoming can versus the original one-unit can.
+
+For n unit-mass objects in the moving cluster, source speed is u/n.  With the
+cola can changed from mass 1 to mass 4, speed is 4u/(n+3), hence the ratios
+n*4/(n+3): 1.6, 2.0, 16/7, and 2.5 for clusters of 2--5 objects.  Using source
+frames themselves preserves geometry, lighting, occlusion, labels, and all
+unaffected scene appearance without introducing compositing seams.
+"""
+from pathlib import Path
+import numpy as np
+import artifact_io
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def source_time(output_frame: int) -> float:
+    """Map an output-frame time to the corresponding rendered source state."""
+    if output_frame <= 4:
+        return float(output_frame)
+
+    # Source-state intervals begin at collisions with successive stationary
+    # objects.  Advance through each interval at its 4x-mass speed ratio.
+    s = 4.0
+    remaining = float(output_frame - 4)
+    phases = (
+        (10.0, 8.0 / 5.0),   # moving mass: source 2, counterfactual 5
+        (17.0, 2.0),         # source 3, counterfactual 6
+        (21.0, 16.0 / 7.0),  # source 4, counterfactual 7
+        (34.0, 5.0 / 2.0),   # source 5, counterfactual 8
+    )
+    for end, rate in phases:
+        output_duration = (end - s) / rate
+        if remaining <= output_duration:
+            return s + remaining * rate
+        remaining -= output_duration
+        s = end
+
+    # The source has fully settled by source frame 34.  Its final rendered
+    # state is therefore held, just as in the factual video's long static tail.
+    return 34.0
+
+
+def generate() -> np.ndarray:
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3) and source.dtype == np.uint8
+
+    mapped = np.array([source_time(t) for t in range(96)])
+    # Nearest rendered state avoids double images/ghosting during the fast
+    # impacts; the physical displacement between output frames remains smooth.
+    indices = np.clip(np.floor(mapped + 0.5).astype(np.int64), 0, 95)
+    frames = np.ascontiguousarray(source[indices])
+    assert frames.shape == (96, 144, 256, 3) and frames.dtype == np.uint8
+    return frames
+
+
+if __name__ == "__main__":
+    result = generate()
+    artifact_io.save(result, CASE_DIR)
+    print("saved", result.shape, result.dtype)
+    print("source frame map:", [int(np.floor(source_time(i) + 0.5)) for i in range(20)])
