+"""Generate the 20x-red-ball-mass counterfactual video.
+
+The intervention changes inertia, not gravitational acceleration, so the source is
+preserved exactly through the approach.  At the low-restitution ball collision the
+20:1 red/blue mass ratio leaves the red ball with about 94% of its incident speed;
+both balls consequently travel through the left boundary instead of the red ball
+coming to rest at x ~= 22.
+
+Only local RGB/numeric operations are used.  The source ball pixels are removed by
+small-mask inpainting, and source-derived ball sprites are composited over that
+reconstructed patch with contact shadows.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+N, H, W = 96, 144, 256
+
+
+def disk_mask(shape, cx, cy, radius, value=255):
+    """Return an antialiased-independent uint8 disk mask for source cleanup."""
+    yy, xx = np.ogrid[: shape[0], : shape[1]]
+    return (((xx - cx) ** 2 + (yy - cy) ** 2) <= radius**2).astype(np.uint8) * value
+
+
+def add_cleanup_mask(mask, cx, cy):
+    """Cover a 10 px ball plus its compact contact/cast shadow."""
+    if cx < -18 or cx > W + 18:
+        return
+    cv2.circle(mask, (int(round(cx)), int(round(cy))), 12, 255, -1, cv2.LINE_AA)
+    # The original renderer has a dark shadow extending roughly 17 px below the
+    # sphere.  Masking it prevents a stationary 'ghost' after relocation.
+    cv2.ellipse(
+        mask,
+        (int(round(cx + 1)), int(round(cy + 13))),
+        (13, 9),
+        0,
+        0,
+        360,
+        255,
+        -1,
+        cv2.LINE_AA,
+    )
+
+
+def extract_ball_template(frame, cx, cy, radius=12):
+    """Extract a square RGB template centered on a cleanly visible source ball."""
+    cx, cy = int(round(cx)), int(round(cy))
+    return frame[cy - radius : cy + radius + 1, cx - radius : cx + radius + 1].copy()
+
+
+def draw_contact_shadow(frame, cx, cy, strength=0.58):
+    """Draw a soft, warm contact shadow beneath a ball in linear blend form."""
+    if cx < -20 or cx > W + 20:
+        return
+    yy, xx = np.mgrid[0:H, 0:W]
+    # Tight dark contact core plus a softer lower cast shadow.
+    core = np.exp(-(((xx - cx) / 6.2) ** 2 + ((yy - (cy + 8.0)) / 3.0) ** 2) * 0.5)
+    cast = np.exp(-(((xx - (cx + 1.5)) / 9.0) ** 2 + ((yy - (cy + 13.0)) / 6.0) ** 2) * 0.5)
+    alpha = np.clip(strength * core + 0.24 * cast, 0.0, 0.72)[..., None]
+    # The scene's ambient shadows are dark brown rather than neutral black.
+    shadow_color = np.array([49.0, 35.0, 29.0], dtype=np.float32)
+    out = frame.astype(np.float32)
+    out[:] = out * (1.0 - alpha) + shadow_color * alpha
+    frame[:] = np.clip(out, 0, 255).astype(np.uint8)
+
+
+def paste_ball(frame, template, cx, cy, radius=10.6):
+    """Composite a source-derived sphere at a possibly subpixel center."""
+    if cx < -radius - 2 or cx > W + radius + 2:
+        return
+
+    half = template.shape[0] // 2
+    # Integer destination bounds large enough for the antialiased sphere.
+    x0 = max(0, int(np.floor(cx - half)))
+    x1 = min(W, int(np.ceil(cx + half + 1)))
+    y0 = max(0, int(np.floor(cy - half)))
+    y1 = min(H, int(np.ceil(cy + half + 1)))
+    if x0 >= x1 or y0 >= y1:
+        return
+
+    ys, xs = np.mgrid[y0:y1, x0:x1].astype(np.float32)
+    # Relative sprite coordinates, sampled bilinearly to avoid one-pixel judder.
+    map_x = (xs - cx + half).astype(np.float32)
+    map_y = (ys - cy + half).astype(np.float32)
+    sampled = cv2.remap(
+        template,
+        map_x,
+        map_y,
+        interpolation=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_REPLICATE,
+    ).astype(np.float32)
+
+    dist = np.sqrt((xs - cx) ** 2 + (ys - cy) ** 2)
+    alpha = np.clip(radius + 0.55 - dist, 0.0, 1.0)[..., None]
+    dst = frame[y0:y1, x0:x1].astype(np.float32)
+    frame[y0:y1, x0:x1] = np.clip(dst * (1.0 - alpha) + sampled * alpha, 0, 255).astype(np.uint8)
+
+
+def source_red_center(t):
+    """Source red-ball center during/after the collision, in output pixels."""
+    # Full-sphere centers (the saturated-color centroid sits about 2 px higher).
+    keys = np.array(
+        [
+            [22, 39.4, 88.3],
+            [23, 34.9, 88.5],
+            [24, 31.1, 88.7],
+            [25, 28.1, 88.8],
+            [26, 25.8, 88.9],
+            [27, 23.8, 88.9],
+            [28, 22.6, 89.1],
+            [29, 22.0, 89.0],
+            [30, 21.9, 89.0],
+            [95, 21.9, 89.0],
+        ],
+        dtype=np.float32,
+    )
+    return tuple(np.interp(t, keys[:, 0], keys[:, i]) for i in (1, 2))
+
+
+def source_blue_center(t):
+    """Source blue-ball center while it is still visible after impact."""
+    keys = np.array(
+        [
+            [22, 21.6, 88.6],
+            [23, 14.6, 89.2],
+            [24, 7.8, 89.5],
+            [25, 4.1, 90.1],
+            [26, 1.7, 90.9],
+        ],
+        dtype=np.float32,
+    )
+    return tuple(np.interp(t, keys[:, 0], keys[:, i]) for i in (1, 2))
+
+
+def target_centers(t):
+    """Counterfactual post-contact centers from a 20:1 low-e collision."""
+    # At frame 22 the collision is only just resolving.  Thereafter the red ball
+    # has v ~= -11 px/frame (19.8/21 of its incident velocity), while the blue
+    # ball has v ~= -13.5 px/frame.  A measured ~0.6 px/frame^2 rolling drag is
+    # applied to both while visible.
+    dt = float(t - 22)
+    red_x = 39.4 - 11.0 * dt + 0.30 * dt * dt
+    blue_x = 20.5 - 13.5 * dt + 0.30 * dt * dt
+    red_y = 88.3 + min(dt, 1.0) * 0.7
+    blue_y = 88.7 + min(dt, 1.0) * 0.3
+    return (red_x, red_y), (blue_x, blue_y)
+
+
+def generate():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (N, H, W, 3) and source.dtype == np.uint8
+
+    # Clean source examples: red is fully visible after settling; blue is fully
+    # visible before contact.  Their shading therefore remains scene-authentic.
+    red_template = extract_ball_template(source[30], 22, 89)
+    blue_template = extract_ball_template(source[0], 25, 89)
+
+    frames = source.copy()
+
+    # Frames 0..21 are identical: changing mass does not alter free-fall/rolling
+    # acceleration, and contact has not yet occurred.
+    for t in range(22, N):
+        cleanup = np.zeros((H, W), dtype=np.uint8)
+        rcx, rcy = source_red_center(t)
+        add_cleanup_mask(cleanup, rcx, rcy)
+        if t <= 26:
+            bcx, bcy = source_blue_center(t)
+            add_cleanup_mask(cleanup, bcx, bcy)
+
+        # A tiny dilation catches compressed antialias/shadow fringes.  Telea's
+        # local numeric fill preserves the surrounding rock texture and lighting.
+        cleanup = cv2.dilate(cleanup, np.ones((3, 3), np.uint8), iterations=1)
+        clean = cv2.inpaint(source[t], cleanup, 5.0, cv2.INPAINT_TELEA)
+
+        red_pos, blue_pos = target_centers(t)
+        # Shadows are behind both spheres.  Draw blue first, then red; this matches
+        # their slight overlap at the first resolving frame.
+        draw_contact_shadow(clean, blue_pos[0], blue_pos[1], strength=0.53)
+        draw_contact_shadow(clean, red_pos[0], red_pos[1], strength=0.60)
+        paste_ball(clean, blue_template, *blue_pos)
+        paste_ball(clean, red_template, *red_pos)
+        frames[t] = clean
+
+    assert frames.shape == (96, 144, 256, 3)
+    assert frames.dtype == np.uint8
+    return frames
+
+
+if __name__ == "__main__":
+    result = generate()
+    artifact_io.save(result, CASE_DIR)
+    print(f"saved {result.shape[0]} RGB frames to {CASE_DIR}")
+
