+"""Generate the low-restitution counterfactual for this PixelWorld case.
+
+The new ball does not touch anything until the end of frame 21, so the source
+is retained exactly up to that point.  At contact, reducing restitution to a
+tenth changes the energetic two-ball rebound into an almost inelastic impact.
+The two equal balls therefore continue left at similar speeds, lose speed to
+the same surface friction visible in the source, and settle near one another.
+
+All rendering below is made from RGB source pixels.  In particular, the ball
+appearance, antialiasing, contact shadow, and illumination are extracted as
+signed RGB differences against clean observations of the same background.
+"""
+
+from pathlib import Path
+
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def tennis_centres(frames):
+    """Locate the two grounded tennis balls after the source collision.
+
+    Their green/yellow chroma is unusually strong in this small floor region.
+    A fixed split at x=68 cleanly separates the left and right balls from
+    source frame 22 onward (zero-based index 21 is the contact frame).
+    """
+    centres = {}
+    yy, xx = np.mgrid[0:frames.shape[1], 0:frames.shape[2]]
+    for t in range(22, len(frames)):
+        a = frames[t].astype(np.int16)
+        chroma = (
+            (a[:, :, 1] > a[:, :, 0] + 8)
+            & (a[:, :, 1] > a[:, :, 2] + 25)
+            & (a[:, :, 0] > 80)
+            & (a[:, :, 1] > 100)
+            & (yy >= 83)
+            & (yy <= 103)
+            & (xx >= 10)
+            & (xx <= 120)
+        )
+        pair = []
+        for side in (xx < 68, xx >= 68):
+            m = chroma & side
+            pair.append((float(xx[m].mean()), float(yy[m].mean())))
+        centres[t] = pair
+    return centres
+
+
+def observed_effect(frames, t, centre, donor_t, radius=9):
+    """Return a soft, signed RGB sprite effect and its source patch origin."""
+    h, w = frames.shape[1:3]
+    cx, cy = centre
+    ix, iy = int(round(cx)), int(round(cy))
+    x0, x1 = max(0, ix - radius), min(w, ix + radius + 1)
+    y0, y1 = max(0, iy - radius), min(h, iy + radius + 1)
+
+    current = frames[t, y0:y1, x0:x1].astype(np.float32)
+    clean = frames[donor_t, y0:y1, x0:x1].astype(np.float32)
+    raw = current - clean
+
+    # Ignore tiny temporal raster/illumination fluctuations while preserving
+    # antialiased edges and the faint contact shadow.  Smoothstep avoids a
+    # visible cutout contour.
+    magnitude = np.max(np.abs(raw), axis=2)
+    confidence = np.clip((magnitude - 2.0) / 5.0, 0.0, 1.0)
+    confidence = confidence * confidence * (3.0 - 2.0 * confidence)
+    return raw * confidence[:, :, None], x0, y0
+
+
+def add_shifted(image, effect, x0, y0, dx, dy, scale=1.0):
+    """Add a signed patch after a subpixel translation using bilinear splats."""
+    h, w = image.shape[:2]
+    fx, fy = int(np.floor(dx)), int(np.floor(dy))
+    ax, ay = float(dx - fx), float(dy - fy)
+    weights = (
+        (fx, fy, (1.0 - ax) * (1.0 - ay)),
+        (fx + 1, fy, ax * (1.0 - ay)),
+        (fx, fy + 1, (1.0 - ax) * ay),
+        (fx + 1, fy + 1, ax * ay),
+    )
+    ph, pw = effect.shape[:2]
+    for ox, oy, weight in weights:
+        if weight <= 0.0:
+            continue
+        tx0, ty0 = x0 + ox, y0 + oy
+        tx1, ty1 = tx0 + pw, ty0 + ph
+        sx0, sy0 = max(0, -tx0), max(0, -ty0)
+        sx1, sy1 = pw - max(0, tx1 - w), ph - max(0, ty1 - h)
+        if sx1 > sx0 and sy1 > sy0:
+            image[
+                ty0 + sy0 : ty0 + sy1,
+                tx0 + sx0 : tx0 + sx1,
+            ] += effect[sy0:sy1, sx0:sx1] * (scale * weight)
+
+
+def low_restitution_paths():
+    """Counterfactual centres from contact onward.
+
+    The pre-impact horizontal speed is about -5.9 px/frame.  For equal masses
+    and e~=0.08 (one tenth of the source ball's approximately 0.8 restitution),
+    the ideal post-impact speeds are -3.2 and -2.7 px/frame.  Source rolling
+    speed decays by roughly 0.82 each frame.  The very small vertical rebound
+    has under 0.1 px height (bounce height scales with e squared), so both
+    centres remain at their observed contact height.
+    """
+    old = np.zeros((96, 2), dtype=np.float32)
+    new = np.zeros((96, 2), dtype=np.float32)
+
+    # Contact configuration at the end of zero-based frame 21.
+    old[21] = (65.5, 94.15)
+    new[21] = (75.7, 94.00)
+    old_v, new_v = -3.20, -2.72
+    drag = 0.82
+    for t in range(22, 96):
+        old[t] = (old[t - 1, 0] + old_v, 94.15)
+        new[t] = (new[t - 1, 0] + new_v, 94.00)
+        old_v *= drag
+        new_v *= drag
+    return old, new
+
+
+def generate():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3) and source.dtype == np.uint8
+
+    centres = tennis_centres(source)
+    old_path, new_path = low_restitution_paths()
+    result = source.copy()
+
+    # The right/new ball has fully settled by frame 39.  Its stable source
+    # sprite gives the low-e result a correct ground-contact shadow immediately
+    # rather than reusing the source's several-pixel rebound sprite.
+    stable_t = 39
+    stable_new_centre = centres[stable_t][1]
+    stable_new_effect, snx0, sny0 = observed_effect(
+        source, stable_t, stable_new_centre, donor_t=0
+    )
+
+    for t in range(22, 96):
+        canvas = source[t].astype(np.float32)
+        old_centre, source_new_centre = centres[t]
+
+        # At x>45 frame 39 is a clean observation under the moving left ball;
+        # farther left frame 0 is clean.  Frame 0 is also clean under the full
+        # post-collision trajectory of the right/new ball.
+        old_donor = 39 if old_centre[0] > 45.0 else 0
+        old_effect, ox0, oy0 = observed_effect(
+            source, t, old_centre, donor_t=old_donor
+        )
+        new_effect, nx0, ny0 = observed_effect(
+            source, t, source_new_centre, donor_t=0
+        )
+
+        # Erase the factual high-restitution trajectories.
+        add_shifted(canvas, old_effect, ox0, oy0, 0.0, 0.0, scale=-1.0)
+        add_shifted(canvas, new_effect, nx0, ny0, 0.0, 0.0, scale=-1.0)
+
+        # Restore the affected old ball on its reduced-energy trajectory,
+        # retaining its source rolling appearance for the current instant.
+        add_shifted(
+            canvas,
+            old_effect,
+            ox0,
+            oy0,
+            old_path[t, 0] - old_centre[0],
+            old_path[t, 1] - old_centre[1],
+        )
+
+        # Place the new ball using its grounded appearance.  Reusing the exact
+        # source effect preserves material, lighting, edge AA, and soft shadow.
+        add_shifted(
+            canvas,
+            stable_new_effect,
+            snx0,
+            sny0,
+            new_path[t, 0] - stable_new_centre[0],
+            new_path[t, 1] - stable_new_centre[1],
+        )
+        result[t] = np.clip(np.rint(canvas), 0, 255).astype(np.uint8)
+
+    return result
+
+
+if __name__ == "__main__":
+    frames = generate()
+    artifact_io.save(frames, CASE_DIR)
+    print(
+        "saved", frames.shape, frames.dtype,
+        "range", (int(frames.min()), int(frames.max())),
+    )
