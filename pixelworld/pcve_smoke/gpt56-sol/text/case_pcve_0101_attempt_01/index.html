+"""Generate the 5x-mass counterfactual edit for this PixelWorld case.
+
+The camera is static.  The worn tennis ball is the muted green ball resting at
+(67, 94).  The bright ball reaches it on source frame 22.  Before contact,
+mass has no visible effect.  After contact, the worn ball receives about one
+third of the equal-mass collision velocity and, under the source's floor
+friction, stops near x=63 instead of crossing the floor and hitting the gray
+ball at x=25.  The bright ball's already-observed rightward rebound and its
+right-side consequences are retained from the source.
+"""
+from pathlib import Path
+import numpy as np
+from scipy.ndimage import gaussian_filter
+import artifact_io
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def clean_plate(frames):
+    """Make a source-only clean plate for the pale floor.
+
+    At every pixel choose the temporal sample with the highest minimum RGB
+    channel.  All moving objects in this area are darker or chromatic, while
+    the floor is neutral and bright, so this removes balls and their shadows
+    without introducing synthetic texture.
+    """
+    score = frames.min(axis=3)
+    which = score.argmax(axis=0)
+    yy, xx = np.indices(which.shape)
+    return frames[which, yy, xx]
+
+
+def soft_ellipse(h, w, cx, cy, rx, ry, feather=1.0):
+    yy, xx = np.mgrid[:h, :w]
+    q = np.sqrt(((xx - cx) / rx) ** 2 + ((yy - cy) / ry) ** 2)
+    return np.clip((1.0 + feather / min(rx, ry) - q) * min(rx, ry) / feather,
+                   0.0, 1.0)
+
+
+def make_sprite(source, plate, box, center, radii):
+    """Extract a ball and its contact shadow with a soft source-derived matte."""
+    x0, y0, x1, y1 = box
+    patch = source[y0:y1, x0:x1].astype(np.float32)
+    back = plate[y0:y1, x0:x1].astype(np.float32)
+    h, w = patch.shape[:2]
+    cx, cy = center[0] - x0, center[1] - y0
+
+    # Solid spherical silhouette.  The actual RGB values (highlight, worn
+    # mottling, and shading) all come directly from source frame 1.
+    sphere = soft_ellipse(h, w, cx, cy, radii[0], radii[1], 1.15)
+
+    # Preserve the soft cast/contact shadow extending down-right.  Its alpha
+    # comes from measured departure from the clean source plate and is softly
+    # restricted to the physical shadow neighborhood.
+    diff = np.max(np.abs(patch - back), axis=2)
+    yy, xx = np.mgrid[:h, :w]
+    shadow_region = (((xx - (cx + radii[0] * .72)) / (radii[0] * 1.35)) ** 2 +
+                     ((yy - (cy + radii[1] * .68)) / (radii[1] * .52)) ** 2) < 1
+    shadow = np.clip((diff - 1.5) / 16.0, 0, 0.78) * shadow_region
+    shadow = gaussian_filter(shadow, .65)
+    alpha = np.maximum(sphere, shadow).astype(np.float32)
+    return patch, alpha, (cx, cy)
+
+
+def composite(frame, sprite, alpha, anchor, center):
+    """Alpha-composite a source sprite at a floating-point center."""
+    patch = sprite
+    cx, cy = center
+    ax, ay = anchor
+    x0 = int(round(cx - ax))
+    y0 = int(round(cy - ay))
+    h, w = alpha.shape
+    sx0, sy0 = max(0, -x0), max(0, -y0)
+    sx1, sy1 = min(w, frame.shape[1] - x0), min(h, frame.shape[0] - y0)
+    if sx1 <= sx0 or sy1 <= sy0:
+        return
+    dst = frame[y0 + sy0:y0 + sy1, x0 + sx0:x0 + sx1].astype(np.float32)
+    src = patch[sy0:sy1, sx0:sx1]
+    a = alpha[sy0:sy1, sx0:sx1, None]
+    frame[y0 + sy0:y0 + sy1, x0 + sx0:x0 + sx1] = np.clip(
+        src * a + dst * (1.0 - a), 0, 255).astype(np.uint8)
+
+
+def build_frames():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3) and source.dtype == np.uint8
+    result = source.copy()
+    plate = clean_plate(source)
+
+    # Both source sprites are taken from the untouched first frame.
+    gray_rgb, gray_a, gray_anchor = make_sprite(
+        source[0], plate, (12, 84, 43, 107), (25, 94), (8.0, 6.2))
+    old_rgb, old_a, old_anchor = make_sprite(
+        source[0], plate, (55, 84, 83, 107), (67, 94), (6.7, 6.2))
+
+    # Smooth edit matte for the left floor.  It fully removes the source's
+    # now-invalid old-ball/gray-ball interaction, but ends before the bright
+    # rebound ball at x~=76 on frame 23.
+    yy, xx = np.mgrid[:144, :256]
+    mx = np.where(xx <= 72, 1.0, np.clip((78 - xx) / 6.0, 0, 1))
+    my_top = np.clip((yy - 83) / 4.0, 0, 1)
+    my_bot = np.clip((108 - yy) / 5.0, 0, 1)
+    edit_alpha = (mx * my_top * my_bot).astype(np.float32)[..., None]
+
+    # Contact geometry is unchanged through source frame 22 (index 21).
+    # Perfectly elastic 1:5 transfer gives the heavy old ball ~1/3 of the
+    # original equal-mass speed; measured source friction then arrests it in
+    # four frames.  Coordinates are indexed by output frame.
+    old_x = {
+        22: 65.1,   # frame 23
+        23: 63.8,
+        24: 63.0,
+        25: 62.7,
+    }
+    settled_x = 62.7
+
+    for t in range(22, 96):
+        f = result[t]
+        # Restore the invalidated left-side action to an object-free plate.
+        a = edit_alpha
+        f[:] = np.clip(plate.astype(np.float32) * a + f.astype(np.float32) * (1-a),
+                       0, 255).astype(np.uint8)
+
+        # The left gray ball is no longer reached, so it remains exactly in
+        # its initial location and appearance.  The heavier old tennis ball
+        # moves only a few pixels before settling.
+        composite(f, gray_rgb, gray_a, gray_anchor, (25.0, 94.0))
+        x = old_x.get(t, settled_x)
+        composite(f, old_rgb, old_a, old_anchor, (x, 94.0))
+
+    return result
+
+
+if __name__ == "__main__":
+    frames = build_frames()
+    artifact_io.save(frames, CASE_DIR)
+    print("saved", frames.shape, frames.dtype, "to", CASE_DIR)

    src = patch[sy0:sy1, sx0:sx1]
    a = alpha[sy0:sy1, sx0:sx1, None]
    frame[y0 + sy0:y0 + sy1, x0 + sx0:x0 + sx1] = np.clip(
        src * a + dst * (1.0 - a), 0, 255).astype(np.uint8)


def build_frames():
    source = np.load(CASE_DIR / \"source_pixels.npz\")[\"rgb\"]
    assert source.shape == (96, 144, 256, 3) and source.dtype == np.uint8
    result = source.copy()
    plate = clean_plate(source)

    # Both source sprites are taken from the untouched first frame.
    gray_rgb, gray_a, gray_anchor = make_sprite(
        source[0], plate, (12, 84, 43, 107), (25, 94), (8.0, 6.2))
    old_rgb, old_a, old_anchor = make_sprite(
        source[0], plate, (55, 84, 83, 107), (67, 94), (6.7, 6.2))

    # Smooth edit matte for the left floor.  It fully removes the source's
    # now-invalid old-ball/gray-ball interaction, but ends before the bright
    # rebound ball at x~=76 on frame 23.
    yy, xx = np.mgrid[:144, :256]
    mx = np.where(xx <= 72, 1.0, np.clip((78 - xx) / 6.0, 0, 1))
    my_top = np.clip((yy - 83) / 4.0, 0, 1)
    my_bot = np.clip((108 - yy) / 5.0, 0, 1)
    edit_alpha = (mx * my_top * my_bot).astype(np.float32)[..., None]

    # Contact geometry is unchanged through source frame 22 (index 21).
    # Perfectly elastic 1:5 transfer gives the heavy old ball ~1/3 of the
    # original equal-mass speed; measured source friction then arrests it in
    # four frames.  Coordinates are indexed by output frame.
    old_x = {
        22: 65.1,   # frame 23
        23: 63.8,
        24: 63.0,
        25: 62.7,
    }
    settled_x = 62.7

    for t in range(22, 96):
        f = result[t]
        # Restore the invalidated left-side action to an object-free plate.
        a = edit_alpha
        f[:] = np.clip(plate.astype(np.float32) * a + f.astype(np.float32) * (1-a),
                       0, 255).astype(np.uint8)

        # The left gray ball is no longer reached, so it remains exactly in
        # its initial location and appearance.  The heavier old tennis ball
        # moves only a few pixels before settling.
        composite(f, gray_rgb, gray_a, gray_anchor, (25.0, 94.0))
        x = old_x.get(t, settled_x)
        composite(f, old_rgb, old_a, old_anchor, (x, 94.0))

    return result


if __name__ == \"__main__\":
    frames = build_frames()
    artifact_io.save(frames, CASE_DIR)
    print(\"saved\", frames.shape, frames.dtype, \"to\", CASE_DIR)
