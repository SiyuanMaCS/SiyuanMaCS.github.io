+"""Generate the complete mass-counterfactual PixelWorld clip.
+
+The factual clip is unchanged until the first contact: changing mass does not
+change gravitational acceleration.  Starting with that contact, a small rigid
+body model gives the yellow marble five times the purple marble's mass.  The
+original RGB tensor supplies the set, marble shading, and cast shadows; no
+external renderer or image asset is used.
+"""
+from pathlib import Path
+import numpy as np
+
+import artifact_io
+
+CASE_DIR = Path(__file__).resolve().parent
+N, H, W = 96, 144, 256
+CENTER_X = 127.75

+def largest_component_bbox(mask):
+    """Return (xmin,ymin,xmax,ymax) of the largest 8-connected component."""
+    seen = np.zeros(mask.shape, dtype=bool)
+    best = None
+    best_n = 0
+    hh, ww = mask.shape
+    for y, x in zip(*np.nonzero(mask)):
+        if seen[y, x]:
+            continue
+        stack = [(int(y), int(x))]
+        seen[y, x] = True
+        points = []
+        while stack:
+            yy, xx = stack.pop()
+            points.append((yy, xx))
+            for dy in (-1, 0, 1):
+                for dx in (-1, 0, 1):
+                    if dx == 0 and dy == 0:
+                        continue
+                    ny, nx = yy + dy, xx + dx
+                    if (0 <= ny < hh and 0 <= nx < ww and mask[ny, nx]
+                            and not seen[ny, nx]):
+                        seen[ny, nx] = True
+                        stack.append((ny, nx))
+        if len(points) > best_n:
+            yy = np.fromiter((p[0] for p in points), dtype=np.int16)
+            xx = np.fromiter((p[1] for p in points), dtype=np.int16)
+            best = (int(xx.min()), int(yy.min()), int(xx.max()), int(yy.max()))
+            best_n = len(points)
+    return best
+

+def source_tracks(source):
+    """Locate both colored marbles from RGB values, without masks/assets."""
+    yellow = np.empty((N, 2), np.float64)
+    purple = np.empty((N, 2), np.float64)
+    yy = np.arange(H)[:, None]
+    xx = np.arange(W)[None, :]
+    region = (yy >= 45) & (yy <= 84)
+    for f, im8 in enumerate(source):
+        im = im8.astype(np.float64) / 255.0
+        mx = im.max(axis=2)
+        mn = im.min(axis=2)
+        d = mx - mn
+        hue = np.zeros((H, W), np.float64)
+        nz = d > 0
+        m = (mx == im[:, :, 0]) & nz
+        hue[m] = ((im[:, :, 1] - im[:, :, 2])[m] / d[m]) % 6.0
+        m = (mx == im[:, :, 1]) & nz
+        hue[m] = (im[:, :, 2] - im[:, :, 0])[m] / d[m] + 2.0
+        m = (mx == im[:, :, 2]) & nz
+        hue[m] = (im[:, :, 0] - im[:, :, 1])[m] / d[m] + 4.0
+        hue /= 6.0
+        sat = np.divide(d, mx, out=np.zeros_like(d), where=mx > 0)
+        ymask = region & (xx < 200) & (hue > .10) & (hue < .20) \
+                & (sat > .35) & (mx > .40)
+        pmask = region & (hue > .70) & (hue < .96) & (sat > .25) \
+                & (mx > .25)
+        for mask, out in ((ymask, yellow), (pmask, purple)):
+            box = largest_component_bbox(mask)
+            if box is None:
+                raise RuntimeError("Unable to locate a source marble")
+            x0, y0, x1, y1 = box
+            out[f] = ((x0 + x1) * .5, (y0 + y1) * .5)
+    return yellow, purple

+def simulate_heavy_collision():
+    """Simulate centers after first contact, in output pixels / source frame.
+
+    Linear velocity and the rolling-equivalent angular velocity q are tracked.
+    Coulomb slip transitions into rolling without slip.  This matters here:
+    the heavy yellow marble keeps its original spin at impact, so it briefly
+    speeds back up even though linear momentum was transferred to purple.
+    """
+    # Source-derived constants: rolling acceleration is 0.23 px/frame^2 and
+    # pre-impact speed at contact is 6.25 px/frame.
+    gravity = .322       # along-track acceleration before the 5/7 rolling term
+    friction = .28       # kinetic slip acceleration
+    restitution = .60   # source first-collision rebound is about 0.58
+    wall_restitution = .45
+    radius_contact = 15.0
+    left_stop, right_stop = 27.5, 228.0
+    masses = np.array([5.0, 1.0])
+
+    # Output frame 34 is the common contact configuration and is still exactly
+    # the source image. Simulation rows begin at that instant.
+    x = np.array([120.25, 135.25], np.float64)
+    incoming = np.array([6.25, -6.25], np.float64)
+    q = incoming.copy()
+    e = restitution
+    v = np.array([
+        (masses[0] * incoming[0] + masses[1] * incoming[1]
+         - masses[1] * e * (incoming[0] - incoming[1])) / masses.sum(),
+        (masses[0] * incoming[0] + masses[1] * incoming[1]
+         + masses[0] * e * (incoming[0] - incoming[1])) / masses.sum(),
+    ])
+    rolling = np.array([False, False])
+    centers = np.empty((N, 2), np.float64)
+    centers[:35] = np.nan
+    centers[34] = x
+
+    dt = .005
+    steps = round(1.0 / dt)
+    for frame in range(35, N):
+        for _ in range(steps):
+            for i in range(2):
+                g = gravity if x[i] < CENTER_X else -gravity
+                if rolling[i]:
+                    v[i] += (5.0 / 7.0) * g * dt
+                    q[i] = v[i]
+                else:
+                    slip = v[i] - q[i]
+                    sign = np.sign(slip)
+                    av = g - friction * sign
+                    aq = 2.5 * friction * sign
+                    next_slip = slip + (av - aq) * dt
+                    if sign != 0 and slip * next_slip <= 0:
+                        # Conserve translational + rotational tangential
+                        # momentum while imposing the no-slip constraint.
+                        common = (v[i] + .4 * q[i]) / 1.4
+                        v[i] = common
+                        q[i] = common
+                        rolling[i] = True
+                    else:
+                        v[i] += av * dt
+                        q[i] += aq * dt
+                x[i] += v[i] * dt
+
+                # The raised ends of the physical track are hard stops.
+                if x[i] < left_stop:
+                    x[i] = left_stop
+                    v[i] = -wall_restitution * v[i]
+                    rolling[i] = False
+                elif x[i] > right_stop:
+                    x[i] = right_stop
+                    v[i] = -wall_restitution * v[i]
+                    rolling[i] = False
+
+            # Collision formula works both for the first left-to-right impact
+            # and when rebounding purple later catches yellow from behind.
+            if x[1] - x[0] < radius_contact and v[0] > v[1]:
+                overlap = radius_contact - (x[1] - x[0])
+                x[0] -= overlap * .5
+                x[1] += overlap * .5
+                u0, u1 = v.copy()
+                v[0] = (masses[0] * u0 + masses[1] * u1
+                        - masses[1] * e * (u0 - u1)) / masses.sum()
+                v[1] = (masses[0] * u0 + masses[1] * u1
+                        + masses[0] * e * (u0 - u1)) / masses.sum()
+                rolling[:] = False
+        centers[frame] = x
+    return centers
+

+def bilinear(image, x, y):
+    """Vectorized bilinear RGB sampler."""
+    h, w = image.shape[:2]
+    x = np.clip(x, 0, w - 1.001)
+    y = np.clip(y, 0, h - 1.001)
+    x0 = np.floor(x).astype(np.int32)
+    y0 = np.floor(y).astype(np.int32)
+    x1 = np.minimum(x0 + 1, w - 1)
+    y1 = np.minimum(y0 + 1, h - 1)
+    fx = (x - x0)[..., None]
+    fy = (y - y0)[..., None]
+    return ((1-fx)*(1-fy)*image[y0, x0] + fx*(1-fy)*image[y0, x1]
+            + (1-fx)*fy*image[y1, x0] + fx*fy*image[y1, x1])
+
+
+def track_height(x):
+    """Source-fitted centerline of the symmetric curved rail."""
+    return 72.25 - .1375 * np.abs(x - CENTER_X)
+
+
+def render_sprite(canvas, target_x, target_y, color_index, tracks,
+                  source, clean_background, shadow_only):
+    """Transport one source marble (or just its shadow) to a new center.
+
+    If a marble crosses to the opposite half, its source sprite is taken from
+    the symmetric rail position and reflected.  Thus highlights still face the
+    central window and cast shadows still point outward.
+    """
+    track = tracks[color_index]
+    home_left = color_index == 0
+    crossed = (target_x > CENTER_X) if home_left else (target_x < CENTER_X)
+    reference_x = (2 * CENTER_X - target_x) if crossed else target_x
+    # Only pre-impact samples are used as uncontaminated sprite exemplars.
+    sample_frame = int(np.argmin(np.abs(track[:34, 0] - reference_x)))
+    source_x, source_y = track[sample_frame]
+
+    x0 = max(0, int(np.floor(target_x - 15)))
+    x1 = min(W, int(np.ceil(target_x + 16)))
+    y0 = max(0, int(np.floor(target_y - 10)))
+    y1 = min(H, int(np.ceil(target_y + 17)))
+    gy, gx = np.mgrid[y0:y1, x0:x1]
+    dx = gx - target_x
+    dy = gy - target_y
+    sx = source_x - dx if crossed else source_x + dx
+    sy = source_y + dy
+
+    src = bilinear(source[sample_frame].astype(np.float64), sx, sy)
+    src_bg = bilinear(clean_background.astype(np.float64), sx, sy)
+    delta = src - src_bg
+    change = np.max(np.abs(delta), axis=2)
+
+    # The source object is about 12 px across. Keep its exact antialiasing in
+    # the interior and softly end the transported patch outside the silhouette.
+    r = np.sqrt(dx * dx + (dy - .15) * (dy - .15))
+    core = np.clip((6.9 - r) / .85, 0.0, 1.0)
+
+    outward = -1.0 if target_x < CENTER_X else 1.0
+    shadow_r = ((dx - outward * 2.4) / 9.0) ** 2 + ((dy - 8.0) / 4.3) ** 2
+    shadow_shape = np.clip((1.18 - shadow_r) / .30, 0.0, 1.0)
+    # Reject codec-level deltas while retaining source penumbra gradients.
+    shadow_strength = np.clip((change - 1.5) / 5.0, 0.0, 1.0)
+    shadow = shadow_shape * shadow_strength * (dy > 3.2)
+
+    patch = canvas[y0:y1, x0:x1].astype(np.float64)
+    if shadow_only:
+        # Cast shadows are already composited in source, so transporting the
+        # source-background delta preserves their softness exactly.
+        a = shadow[..., None]
+        patch = patch + delta * a
+    else:
+        a = core[..., None]
+        # Absolute source color preserves marble saturation and speculars.
+        patch = patch * (1.0 - a) + src * a
+    canvas[y0:y1, x0:x1] = np.clip(np.rint(patch), 0, 255).astype(np.uint8)
+
+
+def generate():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    if source.shape != (N, H, W, 3) or source.dtype != np.uint8:
+        raise ValueError(f"Unexpected source tensor: {source.shape}, {source.dtype}")
+    yellow_track, purple_track = source_tracks(source)
+    tracks = (yellow_track, purple_track)
+
+    # During the first 30 frames each location is uncovered in a large
+    # majority of samples. Their temporal median is therefore a clean,
+    # source-authentic set plate without either moving marble or its shadow.
+    clean = np.median(source[:30].astype(np.float32), axis=0)
+    clean = np.clip(np.rint(clean), 0, 255).astype(np.uint8)
+
+    dynamics = simulate_heavy_collision()
+    frames = np.empty_like(source)
+    # Frame 34 is the instant of contact. Up to and including it, mass cannot
+    # affect position, so preserve the factual RGB history pixel-for-pixel.
+    frames[:35] = source[:35]
+    for f in range(35, N):
+        frame = clean.copy()
+        xs = dynamics[f]
+        ys = track_height(xs)
+        # Shadows first, then opaque bodies, matching the source's layer order.
+        for i in range(2):
+            render_sprite(frame, xs[i], ys[i], i, tracks, source, clean, True)
+        for i in range(2):
+            render_sprite(frame, xs[i], ys[i], i, tracks, source, clean, False)
+        frames[f] = frame
+    return frames
+
+
+if __name__ == "__main__":
+    result = generate()
+    # Required output call: writes pixels.npz, all PNG frames, inspection
+    # sheets, and the self-contained HTML player into the case directory.
+    artifact_io.save(result, CASE_DIR)

\"\"\"Generate the complete mass-counterfactual PixelWorld clip.

The factual clip is unchanged until the first contact: changing mass does not
change gravitational acceleration.  Starting with that contact, a small rigid
body model gives the yellow marble five times the purple marble's mass.  The
original RGB tensor supplies the set, marble shading, and cast shadows; no
external renderer or image asset is used.

from pathlib import Path
import numpy as np

import artifact_io

CASE_DIR = Path(__file__).resolve().parent
N, H, W = 96, 144, 256
CENTER_X = 127.75


def largest_component_bbox(mask):
    \"\"\"Return (xmin,ymin,xmax,ymax) of the largest 8-connected component.\"\"\"
    seen = np.zeros(mask.shape, dtype=bool)
    best = None
    best_n = 0
    hh, ww = mask.shape
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]
        seen[y, x] = True
        points = []
        while stack:
            yy, xx = stack.pop()
            points.append((yy, xx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if dx == 0 and dy == 0:
                        continue
                    ny, nx = yy + dy, xx + dx
                    if (0 <= ny < hh and 0 <= nx < ww and mask[ny, nx]
                            and not seen[ny, nx]):
                        seen[ny, nx] = True
                        stack.append((ny, nx))
        if len(points) > best_n:
            yy = np.fromiter((p[0] for p in points), dtype=np.int16)
            xx = np.fromiter((p[1] for p in points), dtype=np.int16)
            best = (int(xx.min()), int(yy.min()), int(xx.max()), int(yy.max()))
            best_n = len(points)
    return best


def source_tracks(source):
    \"\"\"Locate both colored marbles from RGB values, without masks/assets.\"\"\"
    yellow = np.empty((N, 2), np.float64)
    purple = np.empty((N, 2), np.float64)
    yy = np.arange(H)[:, None]
    xx = np.arange(W)[None, :]
    region = (yy >= 45) & (yy <= 84)
    for f, im8 in enumerate(source):
        im = im8.astype(np.float64) / 255.0
        mx = im.max(axis=2)
        mn = im.min(axis=2)
        d = mx - mn
        hue = np.zeros((H, W), np.float64)
        nz = d > 0
        m = (mx == im[:, :, 0]) & nz
        hue[m] = ((im[:, :, 1] - im[:, :, 2])[m] / d[m]) % 6.0
        m = (mx == im[:, :, 1]) & nz
        hue[m] = (im[:, :, 2] - im[:, :, 0])[m] / d[m] + 2.0
        m = (mx == im[:, :, 2]) & nz
        hue[m] = (im[:, :, 0] - im[:, :, 1])[m] / d[m] + 4.0
        hue /= 6.0
        sat = np.divide(d, mx, out=np.zeros_like(d), where=mx > 0)
        ymask = region & (xx < 200) & (hue > .10) & (hue < .20) \\
                & (sat > .35) & (mx > .40)
        pmask = region & (hue > .70) & (hue < .96) & (sat > .25) \\
                & (mx > .25)
        for mask, out in ((ymask, yellow), (pmask, purple)):
            box = largest_component_bbox(mask)
            if box is None:
                raise RuntimeError(\"Unable to locate a source marble\")
            x0, y0, x1, y1 = box
            out[f] = ((x0 + x1) * .5, (y0 + y1) * .5)
    return yellow, purple


def simulate_heavy_collision():
    \"\"\"Simulate centers after first contact, in output pixels / source frame.

    Linear velocity and the rolling-equivalent angular velocity q are tracked.
    Coulomb slip transitions into rolling without slip.  This matters here:
    the heavy yellow marble keeps its original spin at impact, so it briefly
    speeds back up even though linear momentum was transferred to purple.
    \"\"\"
    # Source-derived constants: rolling acceleration is 0.23 px/frame"'^2 and
    # pre-impact speed at contact is 6.25 px/frame.
    gravity = .322       # along-track acceleration before the 5/7 rolling term
    friction = .28       # kinetic slip acceleration
    restitution = .60   # source first-collision rebound is about 0.58
    wall_restitution = .45
    radius_contact = 15.0
    left_stop, right_stop = 27.5, 228.0
    masses = np.array([5.0, 1.0])

    # Output frame 34 is the common contact configuration and is still exactly
    # the source image. Simulation rows begin at that instant.
    x = np.array([120.25, 135.25], np.float64)
    incoming = np.array([6.25, -6.25], np.float64)
    q = incoming.copy()
    e = restitution
    v = np.array([
        (masses[0] * incoming[0] + masses[1] * incoming[1]
         - masses[1] * e * (incoming[0] - incoming[1])) / masses.sum(),
        (masses[0] * incoming[0] + masses[1] * incoming[1]
         + masses[0] * e * (incoming[0] - incoming[1])) / masses.sum(),
    ])
    rolling = np.array([False, False])
    centers = np.empty((N, 2), np.float64)
    centers[:35] = np.nan
    centers[34] = x

    dt = .005
    steps = round(1.0 / dt)
    for frame in range(35, N):
        for _ in range(steps):
            for i in range(2):
                g = gravity if x[i] < CENTER_X else -gravity
                if rolling[i]:
                    v[i] += (5.0 / 7.0) * g * dt
                    q[i] = v[i]
                else:
                    slip = v[i] - q[i]
                    sign = np.sign(slip)
                    av = g - friction * sign
                    aq = 2.5 * friction * sign
                    next_slip = slip + (av - aq) * dt
                    if sign != 0 and slip * next_slip <= 0:
                        # Conserve translational + rotational tangential
                        # momentum while imposing the no-slip constraint.
                        common = (v[i] + .4 * q[i]) / 1.4
                        v[i] = common
                        q[i] = common
                        rolling[i] = True
                    else:
                        v[i] += av * dt
                        q[i] += aq * dt
                x[i] += v[i] * dt

                # The raised ends of the physical track are hard stops.
                if x[i] < left_stop:
                    x[i] = left_stop
                    v[i] = -wall_restitution * v[i]
                    rolling[i] = False
                elif x[i] > right_stop:
                    x[i] = right_stop
                    v[i] = -wall_restitution * v[i]
                    rolling[i] = False

            # Collision formula works both for the first left-to-right impact
            # and when rebounding purple later catches yellow from behind.
            if x[1] - x[0] < radius_contact and v[0] > v[1]:
                overlap = radius_contact - (x[1] - x[0])
                x[0] -= overlap * .5
                x[1] += overlap * .5
                u0, u1 = v.copy()
                v[0] = (masses[0] * u0 + masses[1] * u1
                        - masses[1] * e * (u0 - u1)) / masses.sum()
                v[1] = (masses[0] * u0 + masses[1] * u1
                        + masses[0] * e * (u0 - u1)) / masses.sum()
                rolling[:] = False
        centers[frame] = x
    return centers


def bilinear(image, x, y):
    """Vectorized bilinear RGB sampler."""
    h, w = image.shape[:2]
    x = np.clip(x, 0, w - 1.001)
    y = np.clip(y, 0, h - 1.001)
    x0 = np.floor(x).astype(np.int32)
    y0 = np.floor(y).astype(np.int32)
    x1 = np.minimum(x0 + 1, w - 1)
    y1 = np.minimum(y0 + 1, h - 1)
    fx = (x - x0)[..., None]
    fy = (y - y0)[..., None]
    return ((1-fx)*(1-fy)*image[y0, x0] + fx*(1-fy)*image[y0, x1]
            + (1-fx)*fy*image[y1, x0] + fx*fy*image[y1, x1])


def track_height(x):
    """Source-fitted centerline of the symmetric curved rail."""
    return 72.25 - .1375 * np.abs(x - CENTER_X)


def render_sprite(canvas, target_x, target_y, color_index, tracks,
                  source, clean_background, shadow_only):
    """Transport one source marble (or just its shadow) to a new center.

    If a marble crosses to the opposite half, its source sprite is taken from
    the symmetric rail position and reflected.  Thus highlights still face the
    central window and cast shadows still point outward.
    """
    track = tracks[color_index]
    home_left = color_index == 0
    crossed = (target_x > CENTER_X) if home_left else (target_x < CENTER_X)
    reference_x = (2 * CENTER_X - target_x) if crossed else target_x
    # Only pre-impact samples are used as uncontaminated sprite exemplars.
    sample_frame = int(np.argmin(np.abs(track[:34, 0] - reference_x)))
    source_x, source_y = track[sample_frame]

    x0 = max(0, int(np.floor(target_x - 15)))
    x1 = min(W, int(np.ceil(target_x + 16)))
    y0 = max(0, int(np.floor(target_y - 10)))
    y1 = min(H, int(np.ceil(target_y + 17)))
    gy, gx = np.mgrid[y0:y1, x0:x1]
    dx = gx - target_x
    dy = gy - target_y
    sx = source_x - dx if crossed else source_x + dx
    sy = source_y + dy

    src = bilinear(source[sample_frame].astype(np.float64), sx, sy)
    src_bg = bilinear(clean_background.astype(np.float64), sx, sy)
    delta = src - src_bg
    change = np.max(np.abs(delta), axis=2)

    # The source object is about 12 px across. Keep its exact antialiasing in
    # the interior and softly end the transported patch outside the silhouette.
    r = np.sqrt(dx * dx + (dy - .15) * (dy - .15))
    core = np.clip((6.9 - r) / .85, 0.0, 1.0)

    outward = -1.0 if target_x < CENTER_X else 1.0
    shadow_r = ((dx - outward * 2.4) / 9.0) ** 2 + ((dy - 8.0) / 4.3) ** 2
    shadow_shape = np.clip((1.18 - shadow_r) / .30, 0.0, 1.0)
    # Reject codec-level deltas while retaining source penumbra gradients.
    shadow_strength = np.clip((change - 1.5) / 5.0, 0.0, 1.0)
    shadow = shadow_shape * shadow_strength * (dy > 3.2)

    patch = canvas[y0:y1, x0:x1].astype(np.float64)
    if shadow_only:
        # Cast shadows are already composited in source, so transporting the
        # source-background delta preserves their softness exactly.
        a = shadow[..., None]
        patch = patch + delta * a
    else:
        a = core[..., None]
        # Absolute source color preserves marble saturation and speculars.
        patch = patch * (1.0 - a) + src * a
    canvas[y0:y1, x0:x1] = np.clip(np.rint(patch), 0, 255).astype(np.uint8)

    if source.shape != (N, H, W, 3) or source.dtype != np.uint8:

    # source-authentic set plate without either moving marble or its shadow.
    clean = np.median(source[:30].astype(np.float32), axis=0)
    clean = np.clip(np.rint(clean), 0, 255).astype(np.uint8)

    dynamics = simulate_heavy_collision()

        frames[f] = frame
    return frames
