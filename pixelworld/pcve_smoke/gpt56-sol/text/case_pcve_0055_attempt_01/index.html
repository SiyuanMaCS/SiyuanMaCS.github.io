+"""Generate the requested counterfactual PixelWorld video.
+
+The source contains two small stones approaching one another from opposite
+sides of the house.  Their measured image centres in source frame 1 are
+approximately (48, 68) for red and (213, 79) for yellow.  Thus the requested
+blue stone starts at (130.5, 73.5).  The incoming momenta are almost balanced;
+the source collision midpoint settles close to (128, 73.5), so the inserted
+stone is given only that small, smooth displacement as the impacts occur.
+
+Only local RGB compositing is used.  The small stone is rendered with an
+anti-aliased elliptical footprint, a soft contact shadow, shaded blue enamel,
+and a tiny highlight so that it matches the scale and lighting of the two
+source stones without disturbing any unaffected source pixels.
+"""
+
+from pathlib import Path
+
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def _smoothstep(edge0, edge1, value):
+    """Vectorised smoothstep, also useful with descending edges."""
+    x = np.clip((value - edge0) / (edge1 - edge0), 0.0, 1.0)
+    return x * x * (3.0 - 2.0 * x)
+
+
+def _over(dst, colour, alpha):
+    """Alpha composite an RGB colour field over an RGB destination."""
+    return dst * (1.0 - alpha[..., None]) + colour * alpha[..., None]
+
+
+def draw_blue_stone(frame, cx, cy):
+    """Draw one perspective-matched blue stone at a subpixel image position."""
+    # The source stones occupy roughly 17 x 10 pixels.  Work only in a small
+    # local box, preserving the source exactly everywhere else.
+    x0 = max(0, int(np.floor(cx - 12)))
+    x1 = min(frame.shape[1], int(np.ceil(cx + 12)) + 1)
+    y0 = max(0, int(np.floor(cy - 9)))
+    y1 = min(frame.shape[0], int(np.ceil(cy + 10)) + 1)
+    yy, xx = np.mgrid[y0:y1, x0:x1].astype(np.float32)
+    patch = frame[y0:y1, x0:x1].astype(np.float32)
+
+    # Soft cast/contact shadow.  It is strongest immediately under the stone
+    # and has a short lower-right tail, as do the red and yellow source stones.
+    sx = (xx - (cx + 0.85)) / 9.1
+    sy = (yy - (cy + 1.65)) / 5.15
+    sd = np.sqrt(sx * sx + sy * sy)
+    shadow_alpha = 0.29 * _smoothstep(1.12, 0.73, sd)
+    # Keep the shadow neutral/cool so target colours remain visible through it.
+    shadow_colour = np.empty((*sd.shape, 3), dtype=np.float32)
+    shadow_colour[...] = (54.0, 65.0, 78.0)
+    patch = _over(patch, shadow_colour, shadow_alpha)
+
+    # Perspective footprint and a one-pixel anti-aliased silhouette.
+    dx = (xx - cx) / 8.15
+    dy = (yy - (cy - 0.55)) / 4.45
+    radius = np.sqrt(dx * dx + dy * dy)
+    body_alpha = _smoothstep(1.075, 0.89, radius)
+
+    # Rounded enamel shading: a blue body, darker lower rim, and light from the
+    # upper-left.  The low-amplitude texture prevents an unnaturally flat patch.
+    z = np.sqrt(np.clip(1.0 - radius * radius, 0.0, 1.0))
+    light = 0.69 + 0.24 * z - 0.045 * dx - 0.075 * dy
+    rim = 1.0 - 0.16 * _smoothstep(0.70, 1.01, radius)
+    texture = 1.0 + 0.018 * np.sin(xx * 2.17 + yy * 1.31)
+    light = light * rim * texture
+    base = np.array((46.0, 116.0, 202.0), dtype=np.float32)
+    body_colour = np.clip(base[None, None, :] * light[..., None], 0, 255)
+
+    # A restrained glossy highlight on the lit shoulder.
+    gloss = np.exp(-(((xx - (cx - 2.35)) / 2.25) ** 2 +
+                     ((yy - (cy - 2.00)) / 1.20) ** 2))
+    body_colour += gloss[..., None] * np.array((38.0, 42.0, 35.0), np.float32)
+    body_colour = np.clip(body_colour, 0, 255)
+    patch = _over(patch, body_colour, body_alpha)
+
+    # Fine pale glint, softened enough to survive 256 x 144 playback naturally.
+    glint = 0.26 * np.exp(-(((xx - (cx - 2.8)) / 0.95) ** 2 +
+                            ((yy - (cy - 2.15)) / 0.72) ** 2))
+    patch = _over(patch, np.full((*glint.shape, 3), 235.0, np.float32),
+                  glint * body_alpha)
+
+    frame[y0:y1, x0:x1] = np.clip(np.rint(patch), 0, 255).astype(np.uint8)
+
+
+def blue_centre(frame_index):
+    """Image-space response to the nearly symmetric two-sided impact."""
+    # Initially this is the measured midpoint of the red/yellow centres.
+    # Contact begins in the low thirties.  A cosine ease avoids artificial
+    # acceleration while allowing the small net leftward momentum to settle at
+    # the observed post-collision midpoint.
+    start = np.array((130.5, 73.5), dtype=np.float32)
+    end = np.array((128.0, 73.5), dtype=np.float32)
+    if frame_index <= 27:
+        return start
+    if frame_index >= 43:
+        return end
+    u = (frame_index - 27.0) / 16.0
+    u = 0.5 - 0.5 * np.cos(np.pi * u)
+    return start * (1.0 - u) + end * u
+
+
+def generate():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3)
+    assert source.dtype == np.uint8
+    frames = source.copy()
+    for index, frame in enumerate(frames):
+        cx, cy = blue_centre(index)
+        draw_blue_stone(frame, float(cx), float(cy))
+    return frames
+
+
+if __name__ == "__main__":
+    result = generate()
+    artifact_io.save(result, CASE_DIR)
