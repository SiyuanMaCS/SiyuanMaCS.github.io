+"""Generate the counterfactual tennis-ball video.
+
+The source ball is isolated from its locally static background with temporal
+clean plates.  Its initial velocity is multiplied by 0.6, gravity and ground
+friction are kept unchanged, and source-derived ball pixels are composited at
+the resulting counterfactual positions.  Everything outside the ball's small
+support is retained from the corresponding source frame.
+"""
+from pathlib import Path
+import numpy as np
+import artifact_io
+
+CASE_DIR = Path(__file__).resolve().parent
+SPEED_SCALE = 0.6
+
+
+def track_source_ball(frames):
+    """Track the yellow-green ball using a color score near its known path."""
+    n, h, w, _ = frames.shape
+    yy, xx = np.mgrid[:h, :w]
+    centers = []
+    for t in range(n):
+        # Predict the airborne parabola, then the observed decelerating roll.
+        if t <= 28:
+            px = 28.4 + 4.50 * t
+            py = 58.7 - 4.51 * t + 0.2076 * t * t
+        else:
+            px = centers[-1][0] + max(0.0, 2.40 - 0.035 * (t - 28))
+            py = 93.5
+        rgb = frames[t].astype(np.float32)
+        r, g, b = rgb[..., 0], rgb[..., 1], rgb[..., 2]
+        # Tennis-ball chroma, brightness, and a tight spatial prior.
+        score = np.maximum(np.minimum(r, g) - b - 15.0, 0.0)
+        score *= np.maximum((r + g) * 0.5 - 80.0, 0.0)
+        d2 = (xx - px) ** 2 + (yy - py) ** 2
+        score *= np.exp(-d2 / (2.0 * 4.0 ** 2))
+        score[d2 > 36.0] = 0.0
+        total = float(score.sum())
+        if total <= 0:
+            centers.append((px, py))
+        else:
+            centers.append((float((score * xx).sum() / total),
+                            float((score * yy).sum() / total)))
+    return np.asarray(centers, dtype=np.float64)
+
+
+def make_clean_plates(source, centers):
+    """Remove the factual ball with per-pixel temporal median inpainting."""
+    n, h, w, _ = source.shape
+    clean = source.astype(np.float32).copy()
+    times = np.arange(n)
+    radius = 5.15
+    feather_start = 4.15
+
+    for t, (cx, cy) in enumerate(centers):
+        x0, x1 = max(0, int(np.floor(cx - 6))), min(w, int(np.ceil(cx + 6)) + 1)
+        y0, y1 = max(0, int(np.floor(cy - 6))), min(h, int(np.ceil(cy + 6)) + 1)
+        for y in range(y0, y1):
+            for x in range(x0, x1):
+                dist = float(np.hypot(x - cx, y - cy))
+                if dist >= radius:
+                    continue
+                # Nearby-in-time frames in which the source ball does not
+                # cover this location make a clean sample of the background.
+                safe = np.hypot(centers[:, 0] - x, centers[:, 1] - y) > 5.2
+                candidates = times[safe]
+                if len(candidates) == 0:
+                    continue
+                order = np.argsort(np.abs(candidates - t), kind="stable")
+                selected = candidates[order[:min(7, len(order))]]
+                replacement = np.median(source[selected, y, x].astype(np.float32), axis=0)
+                alpha = np.clip((radius - dist) / (radius - feather_start), 0.0, 1.0)
+                # Fully replace the ball core, softly meet the untouched image.
+                clean[t, y, x] = (1.0 - alpha) * clean[t, y, x] + alpha * replacement
+    return clean
+
+
+def counterfactual_path(source_centers):
+    """Apply 0.6 to initial vx/vy, retaining gravity and rolling friction."""
+    n = len(source_centers)
+    t_fit = np.arange(29, dtype=np.float64)
+    vx, x0_fit = np.polyfit(t_fit, source_centers[:29, 0], 1)
+    ay_half, vy, y0_fit = np.polyfit(t_fit, source_centers[:29, 1], 2)
+
+    # Anchor exactly to frame 1's factual position; only its velocity changes.
+    x0, y0 = source_centers[0]
+    ground_y = float(np.median(source_centers[28:80, 1]))
+    roots = np.roots([ay_half, SPEED_SCALE * vy, y0 - ground_y])
+    impact_t = float(max(r.real for r in roots if abs(r.imag) < 1e-7 and r.real > 0))
+    impact_x = x0 + SPEED_SCALE * vx * impact_t
+
+    path = np.empty((n, 2), dtype=np.float64)
+    # Factual ground-roll displacement as a smooth interpolation function.
+    src_roll_t = np.arange(n - 28, dtype=np.float64)
+    src_roll_d = source_centers[28:, 0] - source_centers[28, 0]
+
+    for frame in range(n):
+        tt = float(frame)
+        if tt <= impact_t:
+            path[frame, 0] = x0 + SPEED_SCALE * vx * tt
+            path[frame, 1] = y0 + SPEED_SCALE * vy * tt + ay_half * tt * tt
+        else:
+            elapsed = tt - impact_t
+            # With unchanged deceleration: t_stop scales by s and stopping
+            # distance by s^2.  Time-warping the measured source roll retains
+            # its physically observed, slightly non-ideal friction profile.
+            source_elapsed = elapsed / SPEED_SCALE
+            measured_d = np.interp(source_elapsed, src_roll_t, src_roll_d,
+                                   left=0.0, right=src_roll_d[-1])
+            path[frame, 0] = impact_x + SPEED_SCALE ** 2 * measured_d
+            path[frame, 1] = ground_y
+    path[0] = source_centers[0]
+    return path, impact_t
+
+
+def add_shifted_residual(frame, residual, source_center, target_center):
+    """Translate a small source-derived ball residual with bilinear sampling."""
+    h, w, _ = frame.shape
+    sx, sy = source_center
+    tx, ty = target_center
+    rad = 5.3
+    x0, x1 = max(0, int(np.floor(tx - 6))), min(w, int(np.ceil(tx + 6)) + 1)
+    y0, y1 = max(0, int(np.floor(ty - 6))), min(h, int(np.ceil(ty + 6)) + 1)
+
+    out = frame
+    for y in range(y0, y1):
+        for x in range(x0, x1):
+            # Inverse translation into the factual residual image.
+            ux = x - tx + sx
+            uy = y - ty + sy
+            ix, iy = int(np.floor(ux)), int(np.floor(uy))
+            fx, fy = ux - ix, uy - iy
+            if ix < 0 or iy < 0 or ix + 1 >= w or iy + 1 >= h:
+                continue
+            sample = ((1-fx)*(1-fy)*residual[iy, ix] +
+                      fx*(1-fy)*residual[iy, ix+1] +
+                      (1-fx)*fy*residual[iy+1, ix] +
+                      fx*fy*residual[iy+1, ix+1])
+            # Suppress any tiny clean-plate mismatch outside the ball/shadow.
+            ds = np.hypot(ux - sx, uy - sy)
+            window = np.clip((rad - ds) / 1.15, 0.0, 1.0)
+            out[y, x] += window * sample
+    return out
+
+
+def main():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3) and source.dtype == np.uint8
+    source_centers = track_source_ball(source)
+    clean = make_clean_plates(source, source_centers)
+    target_path, impact_t = counterfactual_path(source_centers)
+
+    # The residual cleanly retains the source ball's lighting, antialiasing,
+    # and (on the ground) its small contact shadow.
+    residuals = source.astype(np.float32) - clean
+    output = clean.copy()
+    for t in range(len(source)):
+        if t == 0:
+            # At intervention time the position/appearance is still factual.
+            output[t] = source[t]
+            continue
+        if t <= impact_t:
+            sprite_frame = min(t, 27)
+        else:
+            # Use a grounded source sprite after counterfactual contact.
+            sprite_frame = min(95, 28 + int(round(t - impact_t)))
+        add_shifted_residual(output[t], residuals[sprite_frame],
+                             source_centers[sprite_frame], target_path[t])
+
+    frames = np.clip(np.rint(output), 0, 255).astype(np.uint8)
+    artifact_io.save(frames, CASE_DIR)
+
+    # Lightweight reproducibility/physics diagnostics.
+    print(f"saved {len(frames)} frames; counterfactual impact={impact_t:.2f}; "
+          f"final ball center=({target_path[-1,0]:.2f}, {target_path[-1,1]:.2f})")
+
+
+if __name__ == "__main__":
+    main()

    ground_y = float(np.median(source_centers[28:80, 1]))
    roots = np.roots([ay_half, SPEED_SCALE * vy, y0 - ground_y])
    impact_t = float(max(r.real for r in roots if abs(r.imag) < 1e-7 and r.real > 0))
    impact_x = x0 + SPEED_SCALE * vx * impact_t

    path = np.empty((n, 2), dtype=np.float64)
    # Factual ground-roll displacement as a smooth interpolation function.
    src_roll_t = np.arange(n - 28, dtype=np.float64)
    src_roll_d = source_centers[28:, 0] - source_centers[28, 0]

    for frame in range(n):
        tt = float(frame)
        if tt <= impact_t:
            path[frame, 0] = x0 + SPEED_SCALE * vx * tt
            path[frame, 1] = y0 + SPEED_SCALE * vy * tt + ay_half * tt * tt
        else:
            elapsed = tt - impact_t
            # With unchanged deceleration: t_stop scales by s and stopping
            # distance by s"'^2.  Time-warping the measured source roll retains
            # its physically observed, slightly non-ideal friction profile.
            source_elapsed = elapsed / SPEED_SCALE
            measured_d = np.interp(source_elapsed, src_roll_t, src_roll_d,
                                   left=0.0, right=src_roll_d[-1])
            path[frame, 0] = impact_x + SPEED_SCALE ** 2 * measured_d
            path[frame, 1] = ground_y
    path[0] = source_centers[0]
    return path, impact_t


def add_shifted_residual(frame, residual, source_center, target_center):
    """Translate a small source-derived ball residual with bilinear sampling."""
    h, w, _ = frame.shape
    sx, sy = source_center
    tx, ty = target_center
    rad = 5.3
    x0, x1 = max(0, int(np.floor(tx - 6))), min(w, int(np.ceil(tx + 6)) + 1)
    y0, y1 = max(0, int(np.floor(ty - 6))), min(h, int(np.ceil(ty + 6)) + 1)

    out = frame
    for y in range(y0, y1):
        for x in range(x0, x1):
            # Inverse translation into the factual residual image.
            ux = x - tx + sx
            uy = y - ty + sy
            ix, iy = int(np.floor(ux)), int(np.floor(uy))
            fx, fy = ux - ix, uy - iy
            if ix < 0 or iy < 0 or ix + 1 >= w or iy + 1 >= h:
                continue
            sample = ((1-fx)*(1-fy)*residual[iy, ix] +
                      fx*(1-fy)*residual[iy, ix+1] +
                      (1-fx)*fy*residual[iy+1, ix] +
                      fx*fy*residual[iy+1, ix+1])
            # Suppress any tiny clean-plate mismatch outside the ball/shadow.
            ds = np.hypot(ux - sx, uy - sy)
            window = np.clip((rad - ds) / 1.15, 0.0, 1.0)
            out[y, x] += window * sample
    return out


def main():
    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
    assert source.shape == (96, 144, 256, 3) and source.dtype == np.uint8
    source_centers = track_source_ball(source)
    clean = make_clean_plates(source, source_centers)
    target_path, impact_t = counterfactual_path(source_centers)

    # The residual cleanly retains the source ball'"'s lighting, antialiasing,
    # and (on the ground) its small contact shadow.
    residuals = source.astype(np.float32) - clean
    output = clean.copy()
    for t in range(len(source)):
        if t == 0:
            # At intervention time the position/appearance is still factual.
            output[t] = source[t]
            continue
        if t <= impact_t:
            sprite_frame = min(t, 27)
        else:
            # Use a grounded source sprite after counterfactual contact.
            sprite_frame = min(95, 28 + int(round(t - impact_t)))
        add_shifted_residual(output[t], residuals[sprite_frame],
                             source_centers[sprite_frame], target_path[t])

    frames = np.clip(np.rint(output), 0, 255).astype(np.uint8)
    artifact_io.save(frames, CASE_DIR)

    # Lightweight reproducibility/physics diagnostics.
    print(f\"saved {len(frames)} frames; counterfactual impact={impact_t:.2f}; \"
          f\"final ball center=({target_path[-1,0]:.2f}, {target_path[-1,1]:.2f})\")


if __name__ == \"__main__\":
    main()
