+"""Generate the heavy-white-mallet counterfactual video.
+
+The input clip is a diagonal chain of elastic impacts.  Frames through the
+red/white contact are retained.  At that contact we solve the 1-D elastic
+collision with m_white = 10*m_red.  The red mallet consequently rebounds,
+transfers that velocity back to the equal-mass blue mallet, and the white
+mallet advances at only 2/11 of the incoming speed, missing the left mallet
+within the clip.  Object appearances are lifted from the source with soft
+mattes; no external renderer or learned model is used.
+"""
+from pathlib import Path
+import cv2
+import numpy as np
+import artifact_io
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def ellipse_mask(shape, center, radii, softness=0.0):
+    """Float ellipse mask, optionally with a soft exterior edge."""
+    h, w = shape
+    yy, xx = np.mgrid[:h, :w]
+    q = np.sqrt(((xx-center[0])/radii[0])**2 + ((yy-center[1])/radii[1])**2)
+    if softness <= 0:
+        return (q <= 1).astype(np.float32)
+    return np.clip((1.0 + softness - q) / softness, 0.0, 1.0).astype(np.float32)
+
+
+def extract_sprite(frame, center, radii, pad=4):
+    """Extract a source object and its subtle shadow into a feathered sprite.
+
+    A local Telea fill estimates the pixels hidden by the object.  Difference
+    from that fill provides a data-derived antialias/shadow alpha while the
+    ellipse only limits support so no neighboring object is picked up.
+    """
+    h, w = frame.shape[:2]
+    support = ellipse_mask((h, w), center, (radii[0]+pad, radii[1]+pad), 0)
+    core = ellipse_mask((h, w), center, (radii[0]+2, radii[1]+2), 0)
+    clean = cv2.inpaint(frame, (core*255).astype(np.uint8), 4, cv2.INPAINT_TELEA)
+    diff = np.max(np.abs(frame.astype(np.int16)-clean.astype(np.int16)), axis=2).astype(np.float32)
+    # Strong object pixels become opaque; low-contrast pixels preserve soft
+    # antialiased boundaries and cast shadows.
+    alpha = np.clip((diff-1.0)/10.0, 0.0, 1.0) * support
+    alpha = cv2.GaussianBlur(alpha, (3, 3), 0.45)
+    alpha *= support
+
+    x0 = max(0, int(np.floor(center[0]-radii[0]-pad)))
+    x1 = min(w, int(np.ceil(center[0]+radii[0]+pad+1)))
+    y0 = max(0, int(np.floor(center[1]-radii[1]-pad)))
+    y1 = min(h, int(np.ceil(center[1]+radii[1]+pad+1)))
+    return {
+        "rgb": frame[y0:y1, x0:x1].copy(),
+        "alpha": alpha[y0:y1, x0:x1].copy(),
+        "origin": (x0, y0),
+        "center": center,
+    }
+
+
+def composite(frame, sprite, center):
+    """Subpixel-translate and alpha-composite a sprite in place."""
+    rgb = sprite["rgb"]
+    a = sprite["alpha"]
+    sh, sw = a.shape
+    refx, refy = sprite["center"]
+    ox, oy = sprite["origin"]
+    left = center[0] - (refx-ox)
+    top = center[1] - (refy-oy)
+    ix, iy = int(np.floor(left)), int(np.floor(top))
+    fx, fy = left-ix, top-iy
+    M = np.float32([[1, 0, fx], [0, 1, fy]])
+    wrgb = cv2.warpAffine(rgb, M, (sw, sh), flags=cv2.INTER_LINEAR,
+                         borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+    wa = cv2.warpAffine(a, M, (sw, sh), flags=cv2.INTER_LINEAR,
+                        borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+    h, w = frame.shape[:2]
+    x0, y0 = max(0, ix), max(0, iy)
+    x1, y1 = min(w, ix+sw), min(h, iy+sh)
+    if x1 <= x0 or y1 <= y0:
+        return
+    sx0, sy0 = x0-ix, y0-iy
+    sx1, sy1 = sx0+(x1-x0), sy0+(y1-y0)
+    aa = wa[sy0:sy1, sx0:sx1, None]
+    dst = frame[y0:y1, x0:x1].astype(np.float32)
+    src = wrgb[sy0:sy1, sx0:sx1].astype(np.float32)
+    frame[y0:y1, x0:x1] = np.clip(dst*(1-aa)+src*aa+0.5, 0, 255).astype(np.uint8)
+
+
+def make_clean_plate(source):
+    """Remove the four mallets from a settled source frame."""
+    plate = source[50].copy()
+    h, w = plate.shape[:2]
+    mask = np.zeros((h, w), np.float32)
+    # Settled blue and red, plus the two overlapping objects at lower left.
+    for center, radii in [((145.9, 65.8), (10.0, 9.0)),
+                          ((112.8, 78.6), (11.5, 10.0)),
+                          ((68.0, 95.0), (15.0, 11.0))]:
+        mask = np.maximum(mask, ellipse_mask((h, w), center, radii, 0))
+    # A little dilation removes antialiased fringe and contact shadows.
+    m = cv2.dilate((mask*255).astype(np.uint8), np.ones((3,3), np.uint8), iterations=1)
+    return cv2.inpaint(plate, m, 5, cv2.INPAINT_TELEA)
+
+
+def generate():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3) and source.dtype == np.uint8
+
+    f0 = source[0]
+    sprites = {
+        "blue": extract_sprite(f0, (167.5, 58.3), (7.7, 7.4), 4),
+        "red": extract_sprite(f0, (143.1, 66.7), (9.5, 7.0), 4),
+        "white": extract_sprite(f0, (109.8, 79.5), (12.0, 11.5), 4),
+        "left": extract_sprite(f0, (68.8, 95.2), (10.5, 7.5), 4),
+    }
+    plate = make_clean_plate(source)
+    out = source.copy()
+
+    # The factual incoming red velocity from frames 24..29 is approximately
+    # (-1.78,+0.70) pixels/frame along the diagonal collision axis.
+    incoming = np.array([-1.78, 0.70], np.float64)
+    white_v = (2.0/11.0) * incoming
+    red_rebound_v = (-9.0/11.0) * incoming
+
+    white0 = np.array([109.8, 79.5], np.float64)
+    red0 = np.array([114.4, 78.0], np.float64)
+    blue0 = np.array([145.9, 65.8], np.float64)
+    left0 = np.array([68.8, 95.2], np.float64)
+
+    # Red reaches the parked blue object near frame 44.  Equal masses exchange
+    # velocity, leaving red at contact and sending blue up/right.
+    second_hit = 44
+    red_at_second = red0 + red_rebound_v * (second_hit-30)
+
+    # The plate/source difference is only renderer-level noise after frame 15.
+    # A broad, softly feathered replacement guarantees all factual moving
+    # objects and shadows are removed without a rectangular edge.
+    h, w = out.shape[1:3]
+    yy, xx = np.mgrid[:h, :w]
+    inner = (xx >= 52) & (xx <= 229) & (yy >= 25) & (yy <= 108)
+    corridor = inner.astype(np.float32)
+    corridor = cv2.GaussianBlur(corridor, (11,11), 2.4)[..., None]
+
+    # Frame 30 is the visible contact frame and is retained.  Divergence starts
+    # on frame index 31 (source frame 32), immediately after the impact.
+    for t in range(31, 96):
+        base = out[t].astype(np.float32)
+        p = plate.astype(np.float32)
+        out[t] = np.clip(base*(1-corridor)+p*corridor+0.5, 0, 255).astype(np.uint8)
+
+        white_pos = white0 + white_v * (t-29.5)
+        if t < second_hit:
+            red_pos = red0 + red_rebound_v * (t-30)
+            blue_pos = blue0
+        else:
+            red_pos = red_at_second
+            blue_pos = blue0 + red_rebound_v * (t-second_hit)
+
+        # Back-to-front order follows the source camera/view direction.
+        composite(out[t], sprites["left"], left0)
+        composite(out[t], sprites["white"], white_pos)
+        composite(out[t], sprites["blue"], blue_pos)
+        composite(out[t], sprites["red"], red_pos)
+
+    return out
+
+
+if __name__ == "__main__":
+    frames = generate()
+    artifact_io.save(frames, CASE_DIR)
+    print("saved", frames.shape, frames.dtype,
+          "range", int(frames.min()), int(frames.max()))

\"\"\"Generate the heavy-white-mallet counterfactual video.

The input clip is a diagonal chain of elastic impacts.  Frames through the
red/white contact are retained.  At that contact we solve the 1-D elastic
collision with m_white = 10*m_red.  The red mallet consequently rebounds,
transfers that velocity back to the equal-mass blue mallet, and the white
mallet advances at only 2/11 of the incoming speed, missing the left mallet
within the clip.  Object appearances are lifted from the source with soft
mattes; no external renderer or learned model is used.

from pathlib import Path
import cv2
import numpy as np
import artifact_io

CASE_DIR = Path(__file__).resolve().parent


def ellipse_mask(shape, center, radii, softness=0.0):
    \"\"\"Float ellipse mask, optionally with a soft exterior edge.\"\"\"
    h, w = shape
    yy, xx = np.mgrid[:h, :w]
    q = np.sqrt(((xx-center[0])/radii[0])**2 + ((yy-center[1])/radii[1])**2)
    if softness <= 0:
        return (q <= 1).astype(np.float32)
    return np.clip((1.0 + softness - q) / softness, 0.0, 1.0).astype(np.float32)


def extract_sprite(frame, center, radii, pad=4):
    \"\"\"Extract a source object and its subtle shadow into a feathered sprite.

    A local Telea fill estimates the pixels hidden by the object.  Difference
    from that fill provides a data-derived antialias/shadow alpha while the
    ellipse only limits support so no neighboring object is picked up.
    \"\"\"
    h, w = frame.shape[:2]
    support = ellipse_mask((h, w), center, (radii[0]+pad, radii[1]+pad), 0)
    core = ellipse_mask((h, w), center, (radii[0]+2, radii[1]+2), 0)
    clean = cv2.inpaint(frame, (core*255).astype(np.uint8), 4, cv2.INPAINT_TELEA)
    diff = np.max(np.abs(frame.astype(np.int16)-clean.astype(np.int16)), axis=2).astype(np.float32)
    # Strong object pixels become opaque; low-contrast pixels preserve soft
    # antialiased boundaries and cast shadows.
    alpha = np.clip((diff-1.0)/10.0, 0.0, 1.0) * support
    alpha = cv2.GaussianBlur(alpha, (3, 3), 0.45)
    alpha *= support

    x0 = max(0, int(np.floor(center[0]-radii[0]-pad)))
    x1 = min(w, int(np.ceil(center[0]+radii[0]+pad+1)))
    y0 = max(0, int(np.floor(center[1]-radii[1]-pad)))
    y1 = min(h, int(np.ceil(center[1]+radii[1]+pad+1)))
    return {
        \"rgb\": frame[y0:y1, x0:x1].copy(),
        \"alpha\": alpha[y0:y1, x0:x1].copy(),
        \"origin\": (x0, y0),
        \"center\": center,
    }


def composite(frame, sprite, center):
    \"\"\"Subpixel-translate and alpha-composite a sprite in place.\"\"\"
    rgb = sprite[\"rgb\"]
    a = sprite[\"alpha\"]
    sh, sw = a.shape
    refx, refy = sprite[\"center\"]
    ox, oy = sprite[\"origin\"]
    left = center[0] - (refx-ox)
    top = center[1] - (refy-oy)
    ix, iy = int(np.floor(left)), int(np.floor(top))
    fx, fy = left-ix, top-iy
    M = np.float32([[1, 0, fx], [0, 1, fy]])
    wrgb = cv2.warpAffine(rgb, M, (sw, sh), flags=cv2.INTER_LINEAR,
                         borderMode=cv2.BORDER_CONSTANT, borderValue=0)
    wa = cv2.warpAffine(a, M, (sw, sh), flags=cv2.INTER_LINEAR,
                        borderMode=cv2.BORDER_CONSTANT, borderValue=0)
    h, w = frame.shape[:2]
    x0, y0 = max(0, ix), max(0, iy)
    x1, y1 = min(w, ix+sw), min(h, iy+sh)
    if x1 <= x0 or y1 <= y0:
        return
    sx0, sy0 = x0-ix, y0-iy
    sx1, sy1 = sx0+(x1-x0), sy0+(y1-y0)
    aa = wa[sy0:sy1, sx0:sx1, None]
    dst = frame[y0:y1, x0:x1].astype(np.float32)
    src = wrgb[sy0:sy1, sx0:sx1].astype(np.float32)
    frame[y0:y1, x0:x1] = np.clip(dst*(1-aa)+src*aa+0.5, 0, 255).astype(np.uint8)


def make_clean_plate(source):
    \"\"\"Remove the four mallets from a settled source frame.\"\"\"
    plate = source[50].copy()
    h, w = plate.shape[:2]
    mask = np.zeros((h, w), np.float32)
    # Settled blue and red, plus the two overlapping objects at lower left.
    for center, radii in [((145.9, 65.8), (10.0, 9.0)),
                          ((112.8, 78.6), (11.5, 10.0)),
                          ((68.0, 95.0), (15.0, 11.0))]:
        mask = np.maximum(mask, ellipse_mask((h, w), center, radii, 0))
    # A little dilation removes antialiased fringe and contact shadows.
    m = cv2.dilate((mask*255).astype(np.uint8), np.ones((3,3), np.uint8), iterations=1)
    return cv2.inpaint(plate, m, 5, cv2.INPAINT_TELEA)

        \"left\": extract_sprite(f0, (68.8, 95.2), (10.5, 7.5), 4),

    white0 = np.array([109.8, 79.5], np.float64)

    # Red reaches the parked blue object near frame 44.  Equal masses exchange
    # velocity, leaving red at contact and sending blue up/right.

    # The plate/source difference is only renderer-level noise after frame 15.
    # A broad, softly feathered replacement guarantees all factual moving
    # objects and shadows are removed without a rectangular edge.
    h, w = out.shape[1:3]
    yy, xx = np.mgrid[:h, :w]
    inner = (xx >= 52) & (xx <= 229) & (yy >= 25) & (yy <= 108)

        p = plate.astype(np.float32)
        out[t] = np.clip(base*(1-corridor)+p*corridor+0.5, 0, 255).astype(np.uint8)

        composite(out[t], sprites[\"white\"], white_pos)
        composite(out[t], sprites[\"blue\"], blue_pos)
        composite(out[t], sprites[\"red\"], red_pos)

    return out
