+"""Generate the counterfactual video with the cue ball absent from frame 1.
+
+The shot is locked off.  The cue ball is the only cause of motion: it crosses the
+felt and strikes the dark object ball, which then moves while the cue ball stops.
+With the cue ball removed at t=0, the object ball remains in its original pose and
+the rest of the scene is static.  We therefore preserve source frame 1 exactly,
+except for a temporally reconstructed patch of felt under the initial cue ball,
+and hold that physically correct state for the complete four-second shot.
+"""
+from pathlib import Path
+import numpy as np
+import artifact_io
+
+
+def dilate(mask: np.ndarray, iterations: int = 1) -> np.ndarray:
+    """Small dependency-free 8-connected binary dilation."""
+    out = mask.copy()
+    for _ in range(iterations):
+        padded = np.pad(out, 1, mode="constant")
+        out = np.logical_or.reduce([
+            padded[dy:dy + out.shape[0], dx:dx + out.shape[1]]
+            for dy in range(3) for dx in range(3)
+        ])
+    return out
+
+
+def make_frames(case_dir: Path) -> np.ndarray:
+    source = np.load(case_dir / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3) and source.dtype == np.uint8
+
+    original = source[0].astype(np.float32)
+
+    # By frame 7 the moving cue ball has completely uncovered its starting
+    # footprint, while the struck dark ball is still in its initial pose.
+    # A temporal median suppresses H.264/downsampling noise and supplies the
+    # real felt texture and illumination that were occluded in frame 1.
+    clean = np.median(source[6:12].astype(np.float32), axis=0)
+
+    # Restrict detection to the known starting footprint.  The threshold finds
+    # the white sphere, antialiased edge, contact shadow, and soft cast shadow;
+    # dilation includes sub-threshold fringe pixels without changing nearby
+    # scene content.
+    yy, xx = np.ogrid[:144, :256]
+    roi = ((xx - 85.5) / 12.0) ** 2 + ((yy - 95.0) / 13.0) ** 2 <= 1.0
+    difference = np.max(np.abs(original - clean), axis=2)
+    core = roi & (difference > 3.0)
+    replace = dilate(core, 1) & roi
+
+    # Match the clean temporal estimate to frame 1 using unaffected pixels in
+    # a narrow ring.  This prevents a codec-level brightness seam.
+    outer = ((xx - 85.5) / 14.0) ** 2 + ((yy - 95.0) / 15.0) ** 2 <= 1.0
+    ring = outer & ~dilate(replace, 2)
+    offset = np.median(original[ring] - clean[ring], axis=0)
+    clean = np.clip(clean + offset, 0, 255)
+
+    # One-pixel soft fringe gives a natural antialiased transition.  Pixels in
+    # the detected footprint are fully replaced; only the new outer fringe is
+    # blended.
+    fringe = dilate(replace, 1) & roi & ~replace
+    edited = original.copy()
+    edited[replace] = clean[replace]
+    edited[fringe] = 0.65 * clean[fringe] + 0.35 * original[fringe]
+    edited = np.rint(np.clip(edited, 0, 255)).astype(np.uint8)
+
+    # In the no-cue counterfactual there is no impact, so the dark object ball
+    # and every other element remain at their frame-1 positions.
+    frames = np.repeat(edited[None, ...], 96, axis=0)
+    return frames
+
