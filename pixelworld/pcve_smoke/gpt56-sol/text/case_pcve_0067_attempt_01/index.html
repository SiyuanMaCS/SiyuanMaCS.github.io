+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+def make_clean_plate(source: np.ndarray) -> np.ndarray:
+    """Remove the leftmost domino and the shadow it casts from frame zero."""
+    initial = source[0]
+    height, width = initial.shape[:2]
+
+    # The first polygon follows the projected domino silhouette.  The second
+    # covers its narrow, tapering ground shadow below the contact edge.
+    object_mask = np.zeros((height, width), np.uint8)
+    cv2.fillConvexPoly(
+        object_mask,
+        np.array([(72, 32), (86, 39), (78, 96), (74, 103),
+                  (57, 103), (57, 97), (66, 40)], np.int32),
+        255,
+    )
+    shadow_mask = np.zeros_like(object_mask)
+    cv2.fillConvexPoly(
+        shadow_mask,
+        np.array([(56, 96), (76, 97), (83, 133), (77, 141),
+                  (65, 140), (54, 103)], np.int32),
+        255,
+    )
+
+    repair_mask = cv2.max(object_mask, shadow_mask)
+    repair_mask = cv2.dilate(repair_mask, np.ones((3, 3), np.uint8))
+
+    # Navier-Stokes interpolation is especially clean here because the scene
+    # behind the domino is softly focused grass with a smooth depth gradient.
+    repaired = cv2.inpaint(initial, repair_mask, 5, cv2.INPAINT_NS)
+
+    # Once the domino has tipped, the upper part of its original silhouette
+    # exposes the true background.  Recover that real texture from the final
+    # source frame and softly merge it into the interpolated lower region.
+    yy = np.arange(height, dtype=np.float32)[:, None]
+    vertical = np.clip((83.0 - yy) / 11.0, 0.0, 1.0)
+    alpha = (object_mask.astype(np.float32) / 255.0) * vertical
+    alpha = cv2.GaussianBlur(alpha, (0, 0), 1.15)
+    alpha = np.clip(alpha, 0.0, 1.0)[..., None]
+    clean = repaired.astype(np.float32) * (1.0 - alpha)
+    clean += source[-1].astype(np.float32) * alpha
+    return np.clip(np.rint(clean), 0, 255).astype(np.uint8)
+
+
+def generate(source: np.ndarray) -> np.ndarray:
+    clean = make_clean_plate(source)
+
+    # Locate every pixel changed by the factual cascade.  Closing consolidates
+    # anti-aliased object edges and moving cast shadows; dilation gives a small
+    # safety margin against residual edge ghosts.  Counterfactually this whole
+    # area retains its initial state because no domino receives an impulse.
+    temporal_range = source.max(axis=0).astype(np.int16) - source.min(axis=0)
+    affected = (temporal_range.max(axis=2) > 3).astype(np.uint8) * 255
+    affected = cv2.morphologyEx(
+        affected, cv2.MORPH_CLOSE, np.ones((5, 5), np.uint8)
+    )
+    affected = cv2.dilate(affected, np.ones((5, 5), np.uint8)) > 0
+
+    # Ensure the complete repaired region is used from the very first frame.
+    # Outside physically affected pixels, retain the source's tiny rendering
+    # fluctuations rather than needlessly replacing the factual environment.
+    frames = source.copy()
+    frames[:, affected] = clean[affected]
+    return np.ascontiguousarray(frames, dtype=np.uint8)
+
+
+def main() -> None:
+    case_dir = Path(__file__).resolve().parent
+    source = np.load(case_dir / "source_pixels.npz")["rgb"]
+    frames = generate(source)
+    assert frames.shape == (96, 144, 256, 3)
+    artifact_io.save(frames, case_dir)
+
+
+if __name__ == "__main__":
+    main()
