+"""Generate the requested counterfactual PixelWorld video.
+
+The factual clip contains two small stones moving almost head-on across a fixed
+painted target.  This script removes the factual stones from a source-derived
+clean plate, halves the red stone's initial velocity, resolves the later
+red/yellow collision, and composites source-derived RGB stone sprites along
+the resulting trajectories.
+"""
+
+from pathlib import Path
+
+import numpy as np
+from PIL import Image, ImageFilter
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+N, H, W = 96, 144, 256
+
+
+def clean_source_plate(src: np.ndarray) -> np.ndarray:
+    """Return source frame 1 with its two initial moving stones removed."""
+    plate = src[0].copy()
+
+    # At frame 21 these two locations are clear, while the fixed scene and its
+    # lighting are unchanged.  Include generous margins for soft shadows.
+    plate[61:75, 39:57] = src[20, 61:75, 39:57]
+    plate[70:89, 204:223] = src[20, 70:89, 204:223]
+    return plate
+
+
+def component_near_center(mask: np.ndarray) -> np.ndarray:
+    """Keep thresholded pixels connected to the central part of a crop."""
+    h, w = mask.shape
+    seen = np.zeros_like(mask, dtype=bool)
+    stack = []
+    cy, cx = (h - 1) / 2, (w - 1) / 2
+    seeds = np.argwhere(mask)
+    if len(seeds) == 0:
+        return seen
+    seed = seeds[np.argmin((seeds[:, 0] - cy) ** 2 + (seeds[:, 1] - cx) ** 2)]
+    stack.append((int(seed[0]), int(seed[1])))
+    seen[seed[0], seed[1]] = True
+    while stack:
+        y, x = stack.pop()
+        for dy in (-1, 0, 1):
+            for dx in (-1, 0, 1):
+                yy, xx = y + dy, x + dx
+                if (0 <= yy < h and 0 <= xx < w and mask[yy, xx]
+                        and not seen[yy, xx]):
+                    seen[yy, xx] = True
+                    stack.append((yy, xx))
+    return seen
+
+
+def extract_sprite(src_frame: np.ndarray, plate: np.ndarray,
+                   center: np.ndarray, radius: int = 8):
+    """Extract a naturally antialiased RGB sprite from an isolated source stone."""
+    cx, cy = np.rint(center).astype(int)
+    x0, y0 = cx - radius, cy - radius
+    rgb = src_frame[y0:y0 + 2 * radius + 1,
+                    x0:x0 + 2 * radius + 1].copy()
+    bg = plate[y0:y0 + 2 * radius + 1,
+               x0:x0 + 2 * radius + 1]
+    difference = np.max(np.abs(rgb.astype(np.int16) - bg.astype(np.int16)), axis=2)
+
+    # Threshold, retain the stone/shadow component, then feather its outside
+    # by half a pixel.  The RGB itself comes only from factual source pixels.
+    core = component_near_center(difference > 7)
+    expanded = np.asarray(
+        Image.fromarray((core * 255).astype(np.uint8)).filter(ImageFilter.MaxFilter(3))
+    ) > 0
+    alpha = np.asarray(
+        Image.fromarray((expanded * 255).astype(np.uint8)).filter(
+            ImageFilter.GaussianBlur(0.45)
+        ), dtype=np.float32
+    ) / 255.0
+    # Suppress feathered pixels that are merely codec noise in the source.
+    alpha *= np.clip((difference.astype(np.float32) - 1.5) / 6.5, 0.0, 1.0)
+    return rgb, alpha
+
+
+def source_centers():
+    """Measured isolated-stone trajectories in source frames 1..29."""
+    k = np.arange(29, dtype=np.float64)
+    red = np.column_stack((47.35 + 1.975 * k, 67.75 + 0.119 * k))
+    yellow = np.column_stack((213.0 - 2.145 * k, 78.30 - 0.139 * k))
+    return red, yellow
+
+
+def build_sprite_banks(src: np.ndarray, plate: np.ndarray):
+    red_centers, yellow_centers = source_centers()
+    red_bank, yellow_bank = [], []
+    for k in range(29):
+        red_bank.append(extract_sprite(src[k], plate, red_centers[k]))
+        yellow_bank.append(extract_sprite(src[k], plate, yellow_centers[k]))
+    return red_bank, yellow_bank
+
+
+def simulate():
+    """Simulate the edited trajectories in pixel-space at 24 fps."""
+    red_p = np.array([47.35, 67.75], dtype=np.float64)
+    yellow_p = np.array([213.0, 78.30], dtype=np.float64)
+
+    # The requested intervention: exactly half the factual red velocity.
+    red_v = 0.5 * np.array([1.975, 0.119], dtype=np.float64)
+    yellow_v = np.array([-2.145, -0.139], dtype=np.float64)
+
+    red_path = np.zeros((N, 2), dtype=np.float64)
+    yellow_path = np.zeros((N, 2), dtype=np.float64)
+    collided = False
+    collision_frame = None
+    contact_distance = 10.7
+    restitution = 0.08  # matches the strongly inelastic factual impact
+
+    for t in range(N):
+        red_path[t] = red_p
+        yellow_path[t] = yellow_p
+        if t == N - 1:
+            break
+
+        delta = yellow_p - red_p
+        distance = float(np.linalg.norm(delta))
+        relative = yellow_v - red_v
+        if not collided and distance <= contact_distance + 1.0 and np.dot(relative, delta) < 0:
+            normal = delta / max(distance, 1e-6)
+            closing = float(np.dot(red_v - yellow_v, normal))
+            # Equal masses: apply an impulse to exchange normal momentum while
+            # preserving each stone's tangential component.
+            impulse = 0.5 * (1.0 + restitution) * closing
+            red_v = red_v - impulse * normal
+            yellow_v = yellow_v + impulse * normal
+            collided = True
+            collision_frame = t
+
+        if collided:
+            # Source stones lose most of their energy at impact and then roll
+            # down smoothly.  Rolling friction is applied without changing
+            # direction, avoiding an unphysical reversal at rest.
+            for velocity in (red_v, yellow_v):
+                speed = float(np.linalg.norm(velocity))
+                decel = 0.012
+                if speed <= decel:
+                    velocity[:] = 0.0
+                else:
+                    velocity *= (speed - decel) / speed
+
+        red_p = red_p + red_v
+        yellow_p = yellow_p + yellow_v
+
+    assert collision_frame is not None and 48 <= collision_frame <= 53, collision_frame
+    return red_path, yellow_path, collision_frame
+
+
+def composite_sprite(frame: np.ndarray, sprite, center: np.ndarray):
+    """Alpha composite a sprite at a floating-point center using RGB math."""
+    rgb, alpha = sprite
+    sh, sw = alpha.shape
+    radius_x, radius_y = sw // 2, sh // 2
+
+    # Bilinear subpixel placement by translating both RGB and alpha onto a
+    # slightly larger local canvas with four weighted integer placements.
+    x = float(center[0]) - radius_x
+    y = float(center[1]) - radius_y
+    ix, iy = int(np.floor(x)), int(np.floor(y))
+    fx, fy = x - ix, y - iy
+    weights = ((0, 0, (1 - fx) * (1 - fy)),
+               (1, 0, fx * (1 - fy)),
+               (0, 1, (1 - fx) * fy),
+               (1, 1, fx * fy))
+
+    accum_rgb = np.zeros((sh + 1, sw + 1, 3), dtype=np.float32)
+    accum_a = np.zeros((sh + 1, sw + 1), dtype=np.float32)
+    premult = rgb.astype(np.float32) * alpha[..., None]
+    for dx, dy, weight in weights:
+        accum_rgb[dy:dy + sh, dx:dx + sw] += premult * weight
+        accum_a[dy:dy + sh, dx:dx + sw] += alpha * weight
+
+    x0, y0 = max(ix, 0), max(iy, 0)
+    x1, y1 = min(ix + sw + 1, W), min(iy + sh + 1, H)
+    sx0, sy0 = x0 - ix, y0 - iy
+    sx1, sy1 = sx0 + (x1 - x0), sy0 + (y1 - y0)
+    a = np.clip(accum_a[sy0:sy1, sx0:sx1, None], 0.0, 1.0)
+    foreground = accum_rgb[sy0:sy1, sx0:sx1]
+    region = frame[y0:y1, x0:x1].astype(np.float32)
+    frame[y0:y1, x0:x1] = np.clip(region * (1.0 - a) + foreground, 0, 255).astype(np.uint8)
+
+
+def generate_frames() -> np.ndarray:
+    src = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert src.shape == (N, H, W, 3) and src.dtype == np.uint8
+    plate = clean_source_plate(src)
+    red_bank, yellow_bank = build_sprite_banks(src, plate)
+    red_path, yellow_path, collision_frame = simulate()
+
+    frames = np.repeat(plate[None], N, axis=0)
+    for t in range(N):
+        # Advance source texture orientation in proportion to distance rolled.
+        if t <= collision_frame:
+            red_phase = int(round(0.5 * t)) % len(red_bank)
+            yellow_phase = t % len(yellow_bank)
+        else:
+            red_travel = np.linalg.norm(red_path[t] - red_path[collision_frame])
+            yellow_travel = np.linalg.norm(yellow_path[t] - yellow_path[collision_frame])
+            red_phase = int(round(0.5 * collision_frame - red_travel / 1.975)) % len(red_bank)
+            yellow_phase = int(round(collision_frame + yellow_travel / 2.145)) % len(yellow_bank)
+
+        # Farther/up-screen stone first in the rare case their feathered edges overlap.
+        order = [
+            (yellow_path[t], yellow_bank[yellow_phase]),
+            (red_path[t], red_bank[red_phase]),
+        ]
+        order.sort(key=lambda item: item[0][1])
+        for center, sprite in order:
+            composite_sprite(frames[t], sprite, center)
+
+    return frames.astype(np.uint8)
+
+
+if __name__ == "__main__":
+    output = generate_frames()
+    # Required output call; re-running this file deterministically regenerates
+    # pixels.npz, frame PNGs, inspection sheets, and the HTML player.
+    artifact_io.save(output, CASE_DIR)
+    print(f"saved {output.shape[0]} RGB frames to {CASE_DIR}")
+
