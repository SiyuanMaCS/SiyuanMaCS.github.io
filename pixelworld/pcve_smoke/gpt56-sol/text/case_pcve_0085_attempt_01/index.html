+"""Generate the counterfactual billiards clip with a five-times-massive cue ball.
+
+Only source_pixels.npz is read.  The table/camera are reconstructed from robust
+source pixels, while the two balls are composited at positions obtained from a
+small deterministic 1-D rigid-body simulation along the source shot axis.
+"""
+from pathlib import Path
+import numpy as np
+from PIL import Image, ImageDraw
+from artifact_io import save
+
+ROOT = Path(__file__).resolve().parent
+N, H, W = 96, 144, 256
+FPS = 24
+
+
+def alpha_disc(radius, feather=0.65):
+    """Antialiased circular alpha mask sampled at pixel centres."""
+    r = int(np.ceil(radius + feather + 1))
+    yy, xx = np.mgrid[-r:r+1, -r:r+1]
+    dist = np.sqrt(xx*xx + yy*yy)
+    return np.clip((radius + feather - dist) / (2*feather), 0, 1).astype(np.float32)
+
+
+def synth_ball(kind, s, phase=0.0):
+    """Render a compact shaded sphere in the source's photorealistic style."""
+    # Perspective: objects are slightly larger toward the lower-left foreground.
+    radius = float(np.clip(5.35 - 0.0105*s, 4.75, 5.9))
+    rbox = int(np.ceil(radius + 2))
+    yy, xx = np.mgrid[-rbox:rbox+1, -rbox:rbox+1].astype(np.float32)
+    rr = np.sqrt(xx*xx + yy*yy)
+    alpha = np.clip(radius + .65 - rr, 0, 1)[..., None]
+    z = np.sqrt(np.clip(1.0 - (rr/radius)**2, 0, 1))
+    # Upper-left softbox, plus subtle ambient green bounce from the cloth.
+    nx, ny = xx/radius, yy/radius
+    light = np.clip(.56 + .31*z - .085*nx - .13*ny, 0, 1)
+    if kind == 'cue':
+        base = np.stack([222*light+17, 223*light+17, 222*light+17], axis=-1)
+        # A broad, stable highlight. Rotation is deliberately not visible on a cue ball.
+        glint = np.exp(-((xx+1.6)**2+(yy+2.25)**2)/2.7)[...,None]
+        rgb = base + 17*glint
+        rgb[...,1] += np.clip((yy/radius),0,1)*2.0
+    else:
+        base = np.stack([9+17*light, 12+20*light, 10+17*light], axis=-1)
+        glint = np.exp(-((xx+1.5)**2+(yy+2.0)**2)/2.1)[...,None]
+        rgb = base + 22*glint
+        # Very subdued moving specular variation suggests rolling without adding a mark.
+        rgb += 1.3*np.sin(phase + xx*.8)[...,None]*np.clip(z[...,None],0,1)
+    # Contact shadow is handled separately; outside the sphere is transparent.
+    return np.clip(rgb,0,255).astype(np.float32), alpha.astype(np.float32), radius
+
+
+def paste_rgba(dst, rgb, alpha, cx, cy):
+    h,w = alpha.shape[:2]
+    x0=int(np.floor(cx-w/2)); y0=int(np.floor(cy-h/2))
+    x1=x0+w; y1=y0+h
+    sx0=max(0,-x0); sy0=max(0,-y0); sx1=w-max(0,x1-W); sy1=h-max(0,y1-H)
+    dx0=max(0,x0); dy0=max(0,y0); dx1=min(W,x1); dy1=min(H,y1)
+    if dx1<=dx0 or dy1<=dy0:return
+    a=alpha[sy0:sy1,sx0:sx1]
+    dst[dy0:dy1,dx0:dx1] = dst[dy0:dy1,dx0:dx1]*(1-a)+rgb[sy0:sy1,sx0:sx1]*a
+
+
+def draw_ball(dst, kind, s, phase):
+    # Source-measured shot axis. p is its screen-perpendicular direction.
+    origin=np.array([126.2,70.9]); d=np.array([0.8618,-0.5072]); p=np.array([0.5072,0.8618])
+    # Both source balls sit very slightly above the fitted axis.
+    center=origin+d*s+p*(-.55 if kind=='cue' else .18)
+    rgb,a,r=synth_ball(kind,s,phase)
+    # Directional contact shadow, soft and low-opacity, matching the source lighting.
+    sh=alpha_disc(r*.90,.95)[...,None]*.18
+    sh_rgb=np.zeros((*sh.shape[:2],3),np.float32)
+    shadow_center=center+np.array([1.35,2.05])
+    paste_rgba(dst,sh_rgb,sh,shadow_center[0],shadow_center[1])
+    paste_rgba(dst,rgb,a,center[0],center[1])
+
+
+def source_kinematics(frames):
+    """Fit the preimpact cue trajectory from RGB values (no visual model/tools)."""
+    origin=np.array([126.2,70.9]); d=np.array([0.8618,-0.5072]); p=np.array([0.5072,0.8618])
+    yy,xx=np.indices((H,W)); S=(xx-origin[0])*d[0]+(yy-origin[1])*d[1]; P=(xx-origin[0])*p[0]+(yy-origin[1])*p[1]
+    measured=[]
+    for t in range(12):
+        expected=-46+4*t
+        roi=(np.abs(S-expected)<9)&(np.abs(P)<8)
+        lum=frames[t].mean(2)
+        weight=np.clip(lum-165,0,None)*roi
+        measured.append(float((weight*S).sum()/max(weight.sum(),1)))
+    # Quadratic captures the mild perspective/deceleration visible before impact.
+    coef=np.polyfit(np.arange(12),measured,2)
+    return np.polyval(coef,np.arange(N)), measured
+
+
+def simulate(pre_s):
+    cue=np.empty(N,float); obj=np.empty(N,float)
+    # Preserve the complete factual approach; contact begins just after source frame 11.
+    cue[:12]=pre_s[:12]
+    obj[:12]=1.35
+    cue[11]=-9.2; obj[11]=1.35
+    # Equal-mass source: a nearly stationary cue after impact gives incident speed.
+    incident=max(2.35, pre_s[11]-pre_s[10])
+    m_cue,m_obj=5.0,1.0
+    restitution=.88
+    v1=(m_cue-restitution*m_obj)/(m_cue+m_obj)*incident
+    v2=m_cue*(1+restitution)/(m_cue+m_obj)*incident
+    x1,x2=-9.2,1.35
+    wall=51.2; wall_e=.82; friction=.035
+    # 50 substeps per output frame avoids tunnelling and makes recollisions stable.
+    for f in range(12,N):
+        for _ in range(50):
+            dt=1/50
+            x1+=v1*dt; x2+=v2*dt
+            v1=np.sign(v1)*max(0,abs(v1)-friction*dt)
+            v2=np.sign(v2)*max(0,abs(v2)-friction*dt)
+            for idx in (1,2):
+                x = x1 if idx==1 else x2; v = v1 if idx==1 else v2
+                if x>wall:
+                    x=2*wall-x; v=-abs(v)*wall_e
+                if idx==1:x1,v1=x,v
+                else:x2,v2=x,v
+            # Perspective-adjusted centre spacing (about 10 px near impact).
+            sep=10.55-.004*(x1+x2)
+            if x2-x1<sep and v2-v1<0:
+                mid=(x1+x2)/2; x1=mid-sep/2; x2=mid+sep/2
+                impulse=-(1+restitution)*(v2-v1)/(1/m_cue+1/m_obj)
+                v1-=impulse/m_cue; v2+=impulse/m_obj
+        cue[f]=x1; obj[f]=x2
+    return cue,obj
+
+
+def main():
+    src=np.load(ROOT/'source_pixels.npz')['rgb']
+    pre_s, measured=source_kinematics(src)
+    cue_s,obj_s=simulate(pre_s)
+    # Median over time erases moving balls while retaining all fixed table details.
+    background=np.median(src.astype(np.float32),axis=0)
+    # Use source frame-specific pixels outside the small dynamic region to preserve
+    # native sensor/compression variation; robust background only repairs the ball path.
+    yy,xx=np.indices((H,W)); dynamic=(xx>76)&(xx<178)&(yy>39)&(yy<106)
+    # Path-shaped repair region, limiting edits to where source objects existed.
+    origin=np.array([126.2,70.9]); d=np.array([.8618,-.5072]); p=np.array([.5072,.8618])
+    S=(xx-origin[0])*d[0]+(yy-origin[1])*d[1]; P=(xx-origin[0])*p[0]+(yy-origin[1])*p[1]
+    repair=dynamic&(np.abs(P)<8.8)&(S>-53)&(S<58)
+    out=np.empty_like(src)
+    for t in range(N):
+        canvas=src[t].astype(np.float32)
+        # Feather the inpaint boundary by mixing a one-pixel halo.
+        canvas[repair]=background[repair]
+        draw_ball(canvas,'cue',cue_s[t],phase=t*.42)
+        draw_ball(canvas,'object',obj_s[t],phase=t*.67)
+        out[t]=np.clip(np.rint(canvas),0,255).astype(np.uint8)
+    # Validate required contract before writing artifacts.
+    assert out.shape==(96,144,256,3) and out.dtype==np.uint8
+    save(out,ROOT)
+    print('Saved',out.shape,'preimpact fit',np.round(measured,2).tolist())
+    print('Final axis positions: cue %.2f, object %.2f'%(cue_s[-1],obj_s[-1]))
+
+if __name__=='__main__':
+    main()

N, H, W = 96, 144, 256
FPS = 24


def alpha_disc(radius, feather=0.65):
    \"\"\"Antialiased circular alpha mask sampled at pixel centres.\"\"\"
    r = int(np.ceil(radius + feather + 1))
    yy, xx = np.mgrid[-r:r+1, -r:r+1]
    dist = np.sqrt(xx*xx + yy*yy)
    return np.clip((radius + feather - dist) / (2*feather), 0, 1).astype(np.float32)


def synth_ball(kind, s, phase=0.0):
    \"\"\"Render a compact shaded sphere in the source's photorealistic style.\"\"\"
    # Perspective: objects are slightly larger toward the lower-left foreground.
    radius = float(np.clip(5.35 - 0.0105*s, 4.75, 5.9))
    rbox = int(np.ceil(radius + 2))
    yy, xx = np.mgrid[-rbox:rbox+1, -rbox:rbox+1].astype(np.float32)
    rr = np.sqrt(xx*xx + yy*yy)
    alpha = np.clip(radius + .65 - rr, 0, 1)[..., None]
    z = np.sqrt(np.clip(1.0 - (rr/radius)**2, 0, 1))
    # Upper-left softbox, plus subtle ambient green bounce from the cloth.
    nx, ny = xx/radius, yy/radius
    light = np.clip(.56 + .31*z - .085*nx - .13*ny, 0, 1)
    if kind == 'cue':
        base = np.stack([222*light+17, 223*light+17, 222*light+17], axis=-1)
        # A broad, stable highlight. Rotation is deliberately not visible on a cue ball.
        glint = np.exp(-((xx+1.6)**2+(yy+2.25)**2)/2.7)[...,None]
        rgb = base + 17*glint
        rgb[...,1] += np.clip((yy/radius),0,1)*2.0
    else:
        base = np.stack([9+17*light, 12+20*light, 10+17*light], axis=-1)
        glint = np.exp(-((xx+1.5)**2+(yy+2.0)**2)/2.1)[...,None]
        rgb = base + 22*glint
        # Very subdued moving specular variation suggests rolling without adding a mark.
        rgb += 1.3*np.sin(phase + xx*.8)[...,None]*np.clip(z[...,None],0,1)
    # Contact shadow is handled separately; outside the sphere is transparent.
    return np.clip(rgb,0,255).astype(np.float32), alpha.astype(np.float32), radius


def paste_rgba(dst, rgb, alpha, cx, cy):
    h,w = alpha.shape[:2]
    x0=int(np.floor(cx-w/2)); y0=int(np.floor(cy-h/2))
    x1=x0+w; y1=y0+h
    sx0=max(0,-x0); sy0=max(0,-y0); sx1=w-max(0,x1-W); sy1=h-max(0,y1-H)
    dx0=max(0,x0); dy0=max(0,y0); dx1=min(W,x1); dy1=min(H,y1)
    if dx1<=dx0 or dy1<=dy0:return
    a=alpha[sy0:sy1,sx0:sx1]
    dst[dy0:dy1,dx0:dx1] = dst[dy0:dy1,dx0:dx1]*(1-a)+rgb[sy0:sy1,sx0:sx1]*a


def draw_ball(dst, kind, s, phase):
    # Source-measured shot axis. p is its screen-perpendicular direction.
    origin=np.array([126.2,70.9]); d=np.array([0.8618,-0.5072]); p=np.array([0.5072,0.8618])
    # Both source balls sit very slightly above the fitted axis.
    center=origin+d*s+p*(-.55 if kind=='cue' else .18)
    rgb,a,r=synth_ball(kind,s,phase)
    # Directional contact shadow, soft and low-opacity, matching the source lighting.
    sh=alpha_disc(r*.90,.95)[...,None]*.18
    sh_rgb=np.zeros((*sh.shape[:2],3),np.float32)
    shadow_center=center+np.array([1.35,2.05])
    paste_rgba(dst,sh_rgb,sh,shadow_center[0],shadow_center[1])
    paste_rgba(dst,rgb,a,center[0],center[1])


def source_kinematics(frames):
    \"\"\"Fit the preimpact cue trajectory from RGB values (no visual model/tools).\"\"\"
    origin=np.array([126.2,70.9]); d=np.array([0.8618,-0.5072]); p=np.array([0.5072,0.8618])
    yy,xx=np.indices((H,W)); S=(xx-origin[0])*d[0]+(yy-origin[1])*d[1]; P=(xx-origin[0])*p[0]+(yy-origin[1])*p[1]
    measured=[]
    for t in range(12):
        expected=-46+4*t
        roi=(np.abs(S-expected)<9)&(np.abs(P)<8)
        lum=frames[t].mean(2)
        weight=np.clip(lum-165,0,None)*roi
        measured.append(float((weight*S).sum()/max(weight.sum(),1)))
    # Quadratic captures the mild perspective/deceleration visible before impact.
    coef=np.polyfit(np.arange(12),measured,2)
    return np.polyval(coef,np.arange(N)), measured


def simulate(pre_s):
    cue=np.empty(N,float); obj=np.empty(N,float)
    # Preserve the complete factual approach; contact begins just after source frame 11.
    cue[:12]=pre_s[:12]
    obj[:12]=1.35
    cue[11]=-9.2; obj[11]=1.35
    # Equal-mass source: a nearly stationary cue after impact gives incident speed.
    incident=max(2.35, pre_s[11]-pre_s[10])
    m_cue,m_obj=5.0,1.0
    restitution=.88
    v1=(m_cue-restitution*m_obj)/(m_cue+m_obj)*incident
    v2=m_cue*(1+restitution)/(m_cue+m_obj)*incident
    x1,x2=-9.2,1.35
    wall=51.2; wall_e=.82; friction=.035
    # 50 substeps per output frame avoids tunnelling and makes recollisions stable.
    for f in range(12,N):
        for _ in range(50):
            dt=1/50
            x1+=v1*dt; x2+=v2*dt
            v1=np.sign(v1)*max(0,abs(v1)-friction*dt)
            v2=np.sign(v2)*max(0,abs(v2)-friction*dt)
            for idx in (1,2):
                x = x1 if idx==1 else x2; v = v1 if idx==1 else v2
                if x>wall:
                    x=2*wall-x; v=-abs(v)*wall_e
                if idx==1:x1,v1=x,v
                else:x2,v2=x,v
            # Perspective-adjusted centre spacing (about 10 px near impact).
            sep=10.55-.004*(x1+x2)
            if x2-x1<sep and v2-v1<0:
                mid=(x1+x2)/2; x1=mid-sep/2; x2=mid+sep/2
                impulse=-(1+restitution)*(v2-v1)/(1/m_cue+1/m_obj)
                v1-=impulse/m_cue; v2+=impulse/m_obj
        cue[f]=x1; obj[f]=x2
    return cue,obj


def main():
    src=np.load(ROOT/'source_pixels.npz')['rgb']
    pre_s, measured=source_kinematics(src)
    cue_s,obj_s=simulate(pre_s)
    # Median over time erases moving balls while retaining all fixed table details.
    background=np.median(src.astype(np.float32),axis=0)
    # Use source frame-specific pixels outside the small dynamic region to preserve
    # native sensor/compression variation; robust background only repairs the ball path.
    yy,xx=np.indices((H,W)); dynamic=(xx>76)&(xx<178)&(yy>39)&(yy<106)
    # Path-shaped repair region, limiting edits to where source objects existed.
    origin=np.array([126.2,70.9]); d=np.array([.8618,-.5072]); p=np.array([.5072,.8618])
    S=(xx-origin[0])*d[0]+(yy-origin[1])*d[1]; P=(xx-origin[0])*p[0]+(yy-origin[1])*p[1]
    repair=dynamic&(np.abs(P)<8.8)&(S>-53)&(S<58)
    out=np.empty_like(src)
    for t in range(N):
        canvas=src[t].astype(np.float32)
        # Feather the inpaint boundary by mixing a one-pixel halo.
        canvas[repair]=background[repair]
        draw_ball(canvas,'cue',cue_s[t],phase=t*.42)
        draw_ball(canvas,'object',obj_s[t],phase=t*.67)
        out[t]=np.clip(np.rint(canvas),0,255).astype(np.uint8)
    # Validate required contract before writing artifacts.
    assert out.shape==(96,144,256,3) and out.dtype==np.uint8
    save(out,ROOT)
    print('Saved',out.shape,'preimpact fit',np.round(measured,2).tolist())
    print('Final axis positions: cue %.2f, object %.2f'%(cue_s[-1],obj_s[-1]))

if __name__=='__main__':
    main()
