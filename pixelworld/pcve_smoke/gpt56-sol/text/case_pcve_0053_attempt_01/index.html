+"""Generate the restitution-edited PixelWorld video.
+
+The source scene is almost perfectly static apart from the two small stones.  We
+keep the source through their first contact, build a clean plate from the
+pre-contact frames, and continue both rigid bodies with an equal-mass collision
+whose normal coefficient of restitution is 0.95.  Stone pixels (including the
+soft antialiased edge and contact shadow) are sampled from the source itself.
+"""
+
+from pathlib import Path
+
+import numpy as np
+from PIL import Image, ImageFilter
+
+import artifact_io
+
+
+HERE = Path(__file__).resolve().parent
+
+
+def bilinear(image, xs, ys):
+    """Sample an HxWxC array at floating point image coordinates."""
+    h, w = image.shape[:2]
+    x0 = np.floor(xs).astype(int)
+    y0 = np.floor(ys).astype(int)
+    x1 = np.clip(x0 + 1, 0, w - 1)
+    y1 = np.clip(y0 + 1, 0, h - 1)
+    x0 = np.clip(x0, 0, w - 1)
+    y0 = np.clip(y0, 0, h - 1)
+    fx = (xs - x0)[..., None]
+    fy = (ys - y0)[..., None]
+    return (
+        image[y0, x0] * (1 - fx) * (1 - fy)
+        + image[y0, x1] * fx * (1 - fy)
+        + image[y1, x0] * (1 - fx) * fy
+        + image[y1, x1] * fx * fy
+    )
+
+
+def make_clean_plate(src):
+    # During frames 0..31 every background point is uncovered for the large
+    # majority of samples.  The median therefore removes both moving stones
+    # without inventing any texture.
+    return np.median(src[:32].astype(np.float32), axis=0)
+
+
+def extract_stone(frame, plate, center, size=(25, 21)):
+    """Recover a straight-alpha foreground sprite from a source frame."""
+    w, h = size
+    ox = np.arange(w, dtype=np.float32) - (w - 1) / 2
+    oy = np.arange(h, dtype=np.float32) - (h - 1) / 2
+    xx, yy = np.meshgrid(center[0] + ox, center[1] + oy)
+    seen = bilinear(frame, xx, yy)
+    back = bilinear(plate, xx, yy)
+    difference = np.max(np.abs(seen - back), axis=2)
+
+    # Difference is >50 in the body and 3..20 along the antialiased edge and
+    # tiny grounding shadow.  A soft ramp retains both while rejecting the
+    # 1-level render noise in the otherwise static source.
+    alpha = np.clip((difference - 1.5) / 13.0, 0.0, 1.0)
+    alpha = np.asarray(
+        Image.fromarray(np.uint8(np.round(alpha * 255)), "L").filter(
+            ImageFilter.GaussianBlur(0.35)
+        ),
+        dtype=np.float32,
+    ) / 255.0
+    alpha[alpha < 0.025] = 0.0
+
+    # Undo the background contribution at translucent edge pixels.  This
+    # avoids a rectangular/grey fringe when the stone is moved elsewhere.
+    safe = np.maximum(alpha[..., None], 0.08)
+    foreground = (seen - (1.0 - alpha[..., None]) * back) / safe
+    foreground = np.clip(foreground, 0, 255)
+    foreground[alpha < 0.08] = seen[alpha < 0.08]
+    return foreground.astype(np.float32), alpha.astype(np.float32)
+
+
+def rotate_sprite(foreground, alpha, degrees):
+    """Rotate in premultiplied-alpha space to keep clean soft edges."""
+    a = Image.fromarray(np.float32(alpha), "F")
+    premul = foreground * alpha[..., None]
+    kwargs = dict(angle=float(degrees), resample=Image.Resampling.BICUBIC,
+                  expand=True, fillcolor=0.0)
+    ar = np.asarray(a.rotate(**kwargs), dtype=np.float32)
+    channels = []
+    for c in range(3):
+        p = Image.fromarray(np.float32(premul[..., c]), "F")
+        channels.append(np.asarray(p.rotate(**kwargs), dtype=np.float32))
+    pr = np.stack(channels, axis=2)
+    return pr, np.clip(ar, 0.0, 1.0)
+
+
+def composite(frame, premul, alpha, center):
+    """Place a premultiplied sprite at a subpixel center in-place."""
+    sh, sw = alpha.shape
+    left = center[0] - (sw - 1) / 2
+    top = center[1] - (sh - 1) / 2
+    x0 = int(np.floor(left))
+    y0 = int(np.floor(top))
+    fx, fy = left - x0, top - y0
+
+    # A tiny affine translation supplies smooth subpixel motion.
+    def translated(arr):
+        mode = "F"
+        im = Image.fromarray(np.float32(arr), mode)
+        return np.asarray(
+            im.transform(
+                im.size,
+                Image.Transform.AFFINE,
+                (1, 0, -fx, 0, 1, -fy),
+                resample=Image.Resampling.BILINEAR,
+                fillcolor=0.0,
+            ),
+            dtype=np.float32,
+        )
+
+    aa = np.clip(translated(alpha), 0, 1)
+    pp = np.stack([translated(premul[..., c]) for c in range(3)], axis=2)
+
+    h, w = frame.shape[:2]
+    xa, ya = max(0, x0), max(0, y0)
+    xb, yb = min(w, x0 + sw), min(h, y0 + sh)
+    if xa >= xb or ya >= yb:
+        return
+    sx, sy = xa - x0, ya - y0
+    aroi = aa[sy:sy + (yb - ya), sx:sx + (xb - xa), None]
+    proi = pp[sy:sy + (yb - ya), sx:sx + (xb - xa)]
+    frame[ya:yb, xa:xb] = (
+        frame[ya:yb, xa:xb] * (1.0 - aroi) + proi
+    )
+
+
+def collision_trajectories():
+    """Centers for the 96 samples after an e=.95 equal-mass impact."""
+    # Least-squares tracks measured over source frames 0..33.  Pixel y grows
+    # downwards; both velocities point almost exactly down the line of centers.
+    p1 = np.array([47.37794, 67.63514], dtype=np.float64)
+    p2 = np.array([212.37850, 78.96250], dtype=np.float64)
+    v1 = np.array([1.969565, 0.137080], dtype=np.float64)
+    v2 = np.array([-2.134927, -0.151689], dtype=np.float64)
+
+    # The visible silhouettes contact when their center separation is 12 px.
+    dp, dv = p2 - p1, v2 - v1
+    radius_sum = 12.0
+    roots = np.roots([np.dot(dv, dv), 2 * np.dot(dp, dv),
+                      np.dot(dp, dp) - radius_sum * radius_sum])
+    contact_t = float(np.min(roots[roots > 0]))
+    c1, c2 = p1 + v1 * contact_t, p2 + v2 * contact_t
+    normal = (c2 - c1) / np.linalg.norm(c2 - c1)
+
+    # Standard impulse for two equal masses.
+    restitution = 0.95
+    closing = np.dot(v1 - v2, normal)
+    impulse = 0.5 * (1.0 + restitution) * closing
+    out1 = v1 - impulse * normal
+    out2 = v2 + impulse * normal
+
+    centers1, centers2 = [], []
+    for t in range(96):
+        if t <= contact_t:
+            centers1.append(p1 + v1 * t)
+            centers2.append(p2 + v2 * t)
+        else:
+            centers1.append(c1 + out1 * (t - contact_t))
+            centers2.append(c2 + out2 * (t - contact_t))
+    return np.asarray(centers1), np.asarray(centers2), contact_t
+
+
+def generate():
+    source = np.load(HERE / "source_pixels.npz")["rgb"].astype(np.float32)
+    plate = make_clean_plate(source)
+    out = source.copy()
+    centers1, centers2, contact_t = collision_trajectories()
+
+    # Frame 34 still has ample clear space between the stones and therefore
+    # yields uncontaminated sprites close in phase to the impact.
+    reference_frame = 34
+    fg1, a1 = extract_stone(source[reference_frame], plate,
+                            centers1[reference_frame])
+    fg2, a2 = extract_stone(source[reference_frame], plate,
+                            centers2[reference_frame])
+
+    # Frame 37 is the last fully separated factual sample.  From frame 38 on,
+    # erase the low-restitution continuation and render the e=.95 bounce.
+    first_changed = int(np.floor(contact_t)) + 1
+    for t in range(first_changed, 96):
+        frame = out[t]
+
+        # Both source stones remain near the middle after their inelastic
+        # collision.  Restore just that compact area from the clean plate.  A
+        # feathered mask makes the replacement invisible despite sub-1-level
+        # temporal rendering noise in the source.
+        mask = np.zeros((144, 256), dtype=np.uint8)
+        mask[62:86, 104:154] = 255
+        mask = np.asarray(
+            Image.fromarray(mask, "L").filter(ImageFilter.GaussianBlur(1.5)),
+            dtype=np.float32,
+        ) / 255.0
+        frame[:] = frame * (1.0 - mask[..., None]) + plate * mask[..., None]
+
+        # Rolling before impact advances the reference texture by roughly
+        # distance/radius = .39 rad/frame.  Collision restitution changes the
+        # translational normal velocity, but (without a friction instruction)
+        # does not reverse either stone's spin.
+        angle1 = -22.0 * (t - reference_frame)
+        angle2 = 24.0 * (t - reference_frame)
+        p1r, a1r = rotate_sprite(fg1, a1, angle1)
+        p2r, a2r = rotate_sprite(fg2, a2, angle2)
+        composite(frame, p1r, a1r, centers1[t])
+        composite(frame, p2r, a2r, centers2[t])
+
+    return np.uint8(np.clip(np.rint(out), 0, 255))
+
+
+if __name__ == "__main__":
+    frames = generate()
+    artifact_io.save(frames, HERE)
+
