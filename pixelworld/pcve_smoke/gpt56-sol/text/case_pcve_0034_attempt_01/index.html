+"""Generate the 25x-mass bowling-pin counterfactual.
+
+The camera is fixed and the intervention does not alter anything until contact.
+After contact, the very heavy rack has sub-pixel displacement at this output
+resolution.  We therefore restore the intact, pre-contact interaction region
+and run the source ball appearance backwards at a reduced speed, which models
+the recoil from the now much larger effective collision mass.  Using the
+source RGB samples for the reconstruction preserves the exact ball, pin,
+shadow, perspective, antialiasing, and lighting appearance.
+"""
+from pathlib import Path
+import numpy as np
+import artifact_io
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def smoothstep(x):
+    x = np.clip(x, 0.0, 1.0)
+    return x * x * (3.0 - 2.0 * x)
+
+
+def make_patch_alpha(height=144, width=256):
+    """Feathered window enclosing every pixel touched by ball/pin collision."""
+    yy, xx = np.mgrid[:height, :width]
+    # All factual collision residuals above a visible threshold are inside
+    # x=95..151, y=63..102.  The broader window also encloses the receding
+    # ball and its lane shadow without changing the rest of the scene.
+    x0, x1, y0, y1, feather = 88, 168, 56, 136, 8.0
+    edge_distance = np.minimum.reduce(
+        [xx - x0, x1 - xx, yy - y0, y1 - yy]
+    ).astype(np.float32)
+    return smoothstep(edge_distance / feather)[..., None]
+
+
+def interpolated_frame(source, position):
+    """Linear temporal interpolation, preserving uint8-looking source detail."""
+    lo = int(np.floor(position))
+    hi = min(lo + 1, len(source) - 1)
+    frac = np.float32(position - lo)
+    return source[lo].astype(np.float32) * (1.0 - frac) + source[hi].astype(np.float32) * frac
+
+
+def generate():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3) and source.dtype == np.uint8
+
+    out = source.copy()
+    alpha = make_patch_alpha()
+
+    # Last factual frame before any visible pin response.  The ball begins
+    # contacting the head pin between frames 36 and 37 (zero-based indices).
+    contact = 36.0
+
+    # A 25x pin mass makes the rack effectively wall-like compared with the
+    # ball.  With a moderately dissipative rigid-body contact, the ball's
+    # recoil is about 0.38 of its incoming speed.  Sampling the factual
+    # approach in reverse automatically retains the perspective-dependent
+    # apparent speed and scale of both sphere and shadow.
+    recoil_ratio = 0.38
+
+    for t in range(37, 96):
+        reverse_time = max(0.0, contact - recoil_ratio * (t - contact))
+        intact_rack_and_recoiling_ball = interpolated_frame(source, reverse_time)
+
+        factual = source[t].astype(np.float32)
+        composed = factual * (1.0 - alpha) + intact_rack_and_recoiling_ball * alpha
+        out[t] = np.clip(np.rint(composed), 0, 255).astype(np.uint8)
+
+    return out
+
+
+if __name__ == "__main__":
+    frames = generate()
+    artifact_io.save(frames, CASE_DIR)
