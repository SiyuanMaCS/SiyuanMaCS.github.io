+"""Generate the complete counterfactual: triple the yellow stone's mass.
+
+The unchanged approach is copied verbatim.  At contact, a 1-D rigid-body impulse
+is applied with the restitution measured from the source collision.  The source
+stones are then removed and clean source-derived stone sprites are placed along
+the counterfactual trajectories.  No external renderer or learned asset is used.
+"""
+from pathlib import Path
+import numpy as np
+from scipy import ndimage
+import artifact_io
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def clean_background(src):
+    # Before contact, a stone occupies any one pixel for much less than half of
+    # these frames, so the temporal median exactly recovers the static track.
+    return np.median(src[:29].astype(np.float32), axis=0).round().astype(np.uint8)
+
+
+def component_layer(frame, bg, expected_x, expected_y):
+    """Return a full-frame difference layer, matte, and its visual anchor."""
+    h, w = frame.shape[:2]
+    x0, x1 = max(0, int(expected_x) - 18), min(w, int(expected_x) + 19)
+    y0, y1 = max(0, int(expected_y) - 14), min(h, int(expected_y) + 15)
+    delta = frame.astype(np.int16) - bg.astype(np.int16)
+    mag = np.max(np.abs(delta), axis=2)
+    local = mag[y0:y1, x0:x1]
+
+    # Seed with definite object pixels, then include connected antialias/shadow
+    # pixels.  Selecting the largest seeded component rejects codec speckle.
+    labels, count = ndimage.label(local > 3)
+    if count == 0:
+        raise RuntimeError("stone extraction failed")
+    sizes = ndimage.sum(local > 3, labels, range(1, count + 1))
+    seed = labels == (int(np.argmax(sizes)) + 1)
+    matte_local = ndimage.binary_propagation(seed, mask=(local > 1))
+    # Include one-value edge pixels only when adjacent to the recovered object.
+    edge = (local > 0) & ndimage.binary_dilation(matte_local, iterations=1)
+    matte_local |= edge
+
+    matte = np.zeros((h, w), np.float32)
+    matte[y0:y1, x0:x1] = matte_local.astype(np.float32)
+    layer = delta.astype(np.float32) * matte[:, :, None]
+
+    weight = mag.astype(np.float64) * matte
+    yy, xx = np.indices((h, w))
+    total = weight.sum()
+    anchor = (float((xx * weight).sum() / total),
+              float((yy * weight).sum() / total))
+    return layer, matte, anchor
+
+
+def build_sprite_bank(src, bg):
+    red, yellow = [], []
+    # Use only clean, well-separated approach frames as appearance references.
+    for p in range(29):
+        red_guess = (47.6 + 1.96 * p, 67.4 + 0.125 * p)
+        yellow_guess = (212.9 - 2.13 * p, 78.2 - 0.145 * p)
+        red.append(component_layer(src[p], bg, *red_guess))
+        yellow.append(component_layer(src[p], bg, *yellow_guess))
+    return red, yellow
+
+
+def shifted_sprite(sprite, target_x, target_y):
+    layer, matte, (anchor_x, anchor_y) = sprite
+    shift = (target_y - anchor_y, target_x - anchor_x)
+    moved_a = ndimage.shift(matte, shift, order=1, mode="constant",
+                            cval=0.0, prefilter=False)
+    moved_d = ndimage.shift(layer, (shift[0], shift[1], 0), order=1,
+                            mode="constant", cval=0.0, prefilter=False)
+    return moved_d, np.clip(moved_a, 0.0, 1.0)
+
+
+def composite(frame, bg, sprite, x, y):
+    delta, alpha = shifted_sprite(sprite, x, y)
+    active = alpha > 1e-5
+    # Bilinear shifting premultiplies both the matte and source difference.
+    # Unpremultiply to get a target color, then alpha-composite.  On untouched
+    # background this reduces exactly to bg + shifted source difference.
+    unpremul = delta / np.maximum(alpha[:, :, None], 1e-5)
+    target = np.clip(bg.astype(np.float32) + unpremul, 0, 255)
+    out = frame.astype(np.float32)
+    out[active] = (out[active] * (1.0 - alpha[active, None]) +
+                   target[active] * alpha[active, None])
+    return np.clip(out, 0, 255).round().astype(np.uint8)
+
+
+def make_frames():
+    src = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert src.shape == (96, 144, 256, 3) and src.dtype == np.uint8
+    bg = clean_background(src)
+    red_bank, yellow_bank = build_sprite_bank(src, bg)
+
+    result = src.copy()
+
+    # Source approach at frame 36 has the last unambiguous contact geometry.
+    # Pre-impact speeds are approximately (+2, -2) px/frame.  Source rebound
+    # speeds (~-0.16, +0.16) imply e ~= 0.08.  With masses 1 and 3:
+    # Vcm=(1*2 + 3*(-2))/4=-1; separating speed=e*4=.32, hence
+    # v_red=-1.24 and v_yellow=-0.92 px/frame.
+    contact_frame = 36.0
+    red_contact = np.array([118.5, 73.34], np.float64)
+    yellow_contact = np.array([136.5, 74.00], np.float64)
+    v_red, v_yellow = -1.24, -0.92
+    path_slope = 0.070  # perspective slope measured from the approach tracks
+
+    for t in range(37, 96):
+        dt = t - contact_frame
+        xr = red_contact[0] + v_red * dt
+        xy = yellow_contact[0] + v_yellow * dt
+        yr = red_contact[1] + path_slope * (xr - red_contact[0])
+        yy = yellow_contact[1] + path_slope * (xy - yellow_contact[0])
+
+        # Retain the exact source outside the factual stone region; reveal the
+        # clean static track where the original post-collision stones existed.
+        frame = src[t].copy()
+        frame[58:92, 96:161] = bg[58:92, 96:161]
+
+        # Appearance phase follows distance rolled. Red reverses phase after
+        # rebounding; yellow keeps rolling in its original (leftward) sense.
+        red_phase = int(np.clip(round(28.0 - (red_contact[0] - xr) / 1.96), 0, 28))
+        yellow_phase = int(round(28.0 + (yellow_contact[0] - xy) / 2.13)) % 29
+
+        # The yellow trajectory is slightly lower in screen space, so it is the
+        # foreground object at the brief near-contact interval.
+        frame = composite(frame, bg, red_bank[red_phase], xr, yr)
+        frame = composite(frame, bg, yellow_bank[yellow_phase], xy, yy)
+        result[t] = frame
+
+    return result
+
+
+if __name__ == "__main__":
+    frames = make_frames()
+    artifact_io.save(frames, CASE_DIR)
+    print("saved", frames.shape, frames.dtype,
+          "to", CASE_DIR / "pixels.npz")

from pathlib import Path
import numpy as np
from scipy import ndimage
import artifact_io

CASE_DIR = Path(__file__).resolve().parent


def clean_background(src):
    # Before contact, a stone occupies any one pixel for much less than half of
    # these frames, so the temporal median exactly recovers the static track.
    return np.median(src[:29].astype(np.float32), axis=0).round().astype(np.uint8)


def component_layer(frame, bg, expected_x, expected_y):
    \"\"\"Return a full-frame difference layer, matte, and its visual anchor.\"\"\"
    h, w = frame.shape[:2]
    x0, x1 = max(0, int(expected_x) - 18), min(w, int(expected_x) + 19)
    y0, y1 = max(0, int(expected_y) - 14), min(h, int(expected_y) + 15)
    delta = frame.astype(np.int16) - bg.astype(np.int16)
    mag = np.max(np.abs(delta), axis=2)
    local = mag[y0:y1, x0:x1]

    # Seed with definite object pixels, then include connected antialias/shadow
    # pixels.  Selecting the largest seeded component rejects codec speckle.
    labels, count = ndimage.label(local > 3)
    if count == 0:
        raise RuntimeError(\"stone extraction failed\")
    sizes = ndimage.sum(local > 3, labels, range(1, count + 1))
    seed = labels == (int(np.argmax(sizes)) + 1)
    matte_local = ndimage.binary_propagation(seed, mask=(local > 1))
    # Include one-value edge pixels only when adjacent to the recovered object.
    edge = (local > 0) & ndimage.binary_dilation(matte_local, iterations=1)
    matte_local |= edge

    matte = np.zeros((h, w), np.float32)
    matte[y0:y1, x0:x1] = matte_local.astype(np.float32)
    layer = delta.astype(np.float32) * matte[:, :, None]

    weight = mag.astype(np.float64) * matte
    yy, xx = np.indices((h, w))
    total = weight.sum()
    anchor = (float((xx * weight).sum() / total),
              float((yy * weight).sum() / total))
    return layer, matte, anchor


def build_sprite_bank(src, bg):
    red, yellow = [], []
    # Use only clean, well-separated approach frames as appearance references.
    for p in range(29):
        red_guess = (47.6 + 1.96 * p, 67.4 + 0.125 * p)
        yellow_guess = (212.9 - 2.13 * p, 78.2 - 0.145 * p)
        red.append(component_layer(src[p], bg, *red_guess))
        yellow.append(component_layer(src[p], bg, *yellow_guess))
    return red, yellow


def shifted_sprite(sprite, target_x, target_y):
    layer, matte, (anchor_x, anchor_y) = sprite
    shift = (target_y - anchor_y, target_x - anchor_x)
    moved_a = ndimage.shift(matte, shift, order=1, mode=\"constant\",
                            cval=0.0, prefilter=False)
    moved_d = ndimage.shift(layer, (shift[0], shift[1], 0), order=1,
                            mode=\"constant\", cval=0.0, prefilter=False)
    return moved_d, np.clip(moved_a, 0.0, 1.0)


def composite(frame, bg, sprite, x, y):
    delta, alpha = shifted_sprite(sprite, x, y)
    active = alpha > 1e-5
    # Bilinear shifting premultiplies both the matte and source difference.
    # Unpremultiply to get a target color, then alpha-composite.  On untouched
    # background this reduces exactly to bg + shifted source difference.
    unpremul = delta / np.maximum(alpha[:, :, None], 1e-5)
    target = np.clip(bg.astype(np.float32) + unpremul, 0, 255)
    out = frame.astype(np.float32)
    out[active] = (out[active] * (1.0 - alpha[active, None]) +
                   target[active] * alpha[active, None])
    return np.clip(out, 0, 255).round().astype(np.uint8)


def make_frames():
    src = np.load(CASE_DIR / \"source_pixels.npz\")[\"rgb\"]
    assert src.shape == (96, 144, 256, 3) and src.dtype == np.uint8
    bg = clean_background(src)
    red_bank, yellow_bank = build_sprite_bank(src, bg)

    result = src.copy()

    # Source approach at frame 36 has the last unambiguous contact geometry.
    # Pre-impact speeds are approximately (+2, -2) px/frame.  Source rebound
    # speeds (~-0.16, +0.16) imply e ~= 0.08.  With masses 1 and 3:
    # Vcm=(1*2 + 3*(-2))/4=-1; separating speed=e*4=.32, hence
    # v_red=-1.24 and v_yellow=-0.92 px/frame.
    contact_frame = 36.0
    red_contact = np.array([118.5, 73.34], np.float64)
    yellow_contact = np.array([136.5, 74.00], np.float64)
    v_red, v_yellow = -1.24, -0.92
    path_slope = 0.070  # perspective slope measured from the approach tracks

    for t in range(37, 96):
        dt = t - contact_frame
        xr = red_contact[0] + v_red * dt
        xy = yellow_contact[0] + v_yellow * dt
        yr = red_contact[1] + path_slope * (xr - red_contact[0])
        yy = yellow_contact[1] + path_slope * (xy - yellow_contact[0])

        # Retain the exact source outside the factual stone region; reveal the
        # clean static track where the original post-collision stones existed.
        frame = src[t].copy()
        frame[58:92, 96:161] = bg[58:92, 96:161]

        # Appearance phase follows distance rolled. Red reverses phase after
        # rebounding; yellow keeps rolling in its original (leftward) sense.
        red_phase = int(np.clip(round(28.0 - (red_contact[0] - xr) / 1.96), 0, 28))
        yellow_phase = int(round(28.0 + (yellow_contact[0] - xy) / 2.13)) % 29

        # The yellow trajectory is slightly lower in screen space, so it is the
        # foreground object at the brief near-contact interval.
        frame = composite(frame, bg, red_bank[red_phase], xr, yr)
        frame = composite(frame, bg, yellow_bank[yellow_phase], xy, yy)
        result[t] = frame

    return result
