+"""Generate the counterfactual video with the red mallet removed.
+
+The source is a small rigid-body collision chain.  In the factual video a
+blue body travels down-left, hits the red mallet, and the mallet subsequently
+hits the first pale body.  With the mallet absent, conservation of momentum
+makes the blue body continue to that next contact.  The downstream pale-body
+motion in the source already begins at that contact time, so it is retained.
+
+Only local RGB operations are used: source-pixel segmentation, OpenCV's
+classical Telea inpainting, subpixel translation, and alpha compositing.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+from scipy import ndimage
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def largest_component(mask: np.ndarray) -> np.ndarray:
+    labels, count = ndimage.label(mask)
+    if count == 0:
+        return np.zeros_like(mask, dtype=bool)
+    sizes = np.bincount(labels.ravel())
+    sizes[0] = 0
+    return labels == sizes.argmax()
+
+
+def blue_centres(frames: np.ndarray) -> np.ndarray:
+    """Measure the freely moving blue body in pre-impact frames 0..14."""
+    centres = []
+    yy, xx = np.indices(frames.shape[1:3])
+    for frame in frames[:15].astype(np.int16):
+        r, g, b = frame.transpose(2, 0, 1)
+        chroma = (b - r > 30) & (b - g > 15) & (b > 120)
+        roi = (xx >= 130) & (xx < 176) & (yy >= 48) & (yy < 78)
+        component = largest_component(chroma & roi)
+        cy, cx = ndimage.center_of_mass(component)
+        centres.append((cx, cy))
+    return np.asarray(centres, dtype=np.float64)
+
+
+def red_centres(frames: np.ndarray) -> np.ndarray:
+    """Track only the red mallet, avoiding unrelated red scenery."""
+    centres = []
+    yy, xx = np.indices(frames.shape[1:3])
+    for frame in frames.astype(np.int16):
+        r, g, b = frame.transpose(2, 0, 1)
+        red = (r - g > 30) & (r - b > 20) & (r > 150)
+        roi = (xx >= 90) & (xx < 158) & (yy >= 56) & (yy < 91)
+        component = largest_component(red & roi)
+        cy, cx = ndimage.center_of_mass(component)
+        centres.append((cx, cy))
+    return np.asarray(centres, dtype=np.float64)
+
+
+def ellipse(mask: np.ndarray, centre, radii) -> None:
+    """Add a filled antialias-safe object footprint to an inpaint mask."""
+    x, y = centre
+    cv2.ellipse(
+        mask,
+        (int(round(x)), int(round(y))),
+        (int(radii[0]), int(radii[1])),
+        0,
+        0,
+        360,
+        255,
+        thickness=-1,
+        lineType=cv2.LINE_8,
+    )
+
+
+def make_blue_sprite(frames: np.ndarray):
+    """Extract a compact blue sprite and its soft coverage from source RGB."""
+    foreground = frames[0].astype(np.float32)
+    # By frame 14 the starting location is clear, making a local temporal
+    # difference a reliable non-learned matte for the ball and its shading.
+    background = frames[14].astype(np.int16)
+    delta = np.max(np.abs(frames[0].astype(np.int16) - background), axis=2)
+    yy, xx = np.indices(delta.shape)
+    local = (((xx - 166.85) / 10.0) ** 2 + ((yy - 58.75) / 10.0) ** 2) < 1.0
+    r, g, b = frames[0].astype(np.int16).transpose(2, 0, 1)
+    blue = (b - r > 22) & (b - g > 10)
+    matte = largest_component(local & ((delta > 5) | blue))
+    matte = ndimage.binary_closing(matte, iterations=1)
+    matte = ndimage.binary_dilation(matte, iterations=1).astype(np.float32)
+    matte = cv2.GaussianBlur(matte, (0, 0), 0.55)
+    matte = np.clip(matte, 0.0, 1.0)
+    premultiplied = foreground * matte[..., None]
+    return premultiplied, matte
+
+
+def translate_layer(layer: np.ndarray, dx: float, dy: float) -> np.ndarray:
+    matrix = np.float32([[1.0, 0.0, dx], [0.0, 1.0, dy]])
+    return cv2.warpAffine(
+        layer,
+        matrix,
+        (layer.shape[1], layer.shape[0]),
+        flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=0,
+    )
+
+
+def generate() -> np.ndarray:
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3) and source.dtype == np.uint8
+
+    measured_blue = blue_centres(source)
+    measured_red = red_centres(source)
+
+    # Robustly continue the pre-impact blue trajectory.  Contact with the
+    # first pale body occurs at frame 29; equal-mass impact then leaves blue
+    # at that contact while the pale body takes over the source's motion.
+    time = np.arange(len(measured_blue), dtype=np.float64)
+    vx, x0 = np.polyfit(time, measured_blue[:, 0], 1)
+    vy, y0 = np.polyfit(time, measured_blue[:, 1], 1)
+    counterfactual_blue = np.empty((96, 2), dtype=np.float64)
+    for index in range(96):
+        travel_time = min(index, 29)
+        counterfactual_blue[index] = (x0 + vx * travel_time,
+                                      y0 + vy * travel_time)
+
+    premult_sprite, sprite_alpha = make_blue_sprite(source)
+    sprite_origin = measured_blue[0]
+    output = np.empty_like(source)
+
+    for index, original in enumerate(source):
+        removal = np.zeros(original.shape[:2], dtype=np.uint8)
+
+        # Remove the mallet, including its antialiased rim and contact shadow.
+        ellipse(removal, measured_red[index], (10, 8))
+
+        # Remove factual blue.  It moves for the first 15 frames, then remains
+        # at the red impact location for the rest of the source video.
+        if index < len(measured_blue):
+            factual_blue = measured_blue[index]
+        else:
+            factual_blue = (145.8, 65.7)
+        ellipse(removal, factual_blue, (10, 9))
+
+        # A one-pixel expansion catches compression/antialias fringes.  Telea
+        # reconstructs the smooth textured tabletop from the immediate RGB
+        # neighbourhood while leaving all unrelated scenery untouched.
+        removal = cv2.dilate(removal, np.ones((3, 3), np.uint8), iterations=1)
+        clean = cv2.inpaint(original, removal, 3.0, cv2.INPAINT_TELEA)
+
+        target = counterfactual_blue[index]
+        dx, dy = target - sprite_origin
+        alpha = translate_layer(sprite_alpha, float(dx), float(dy))
+        premult = translate_layer(premult_sprite, float(dx), float(dy))
+        composed = clean.astype(np.float32) * (1.0 - alpha[..., None]) + premult
+        output[index] = np.clip(np.rint(composed), 0, 255).astype(np.uint8)
+
+    return output
+
+
+if __name__ == "__main__":
+    frames = generate()
+    artifact_io.save(frames, CASE_DIR)
+    print(f"saved {len(frames)} counterfactual frames to {CASE_DIR}")
