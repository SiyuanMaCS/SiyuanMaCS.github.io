+"""Generate the counterfactual video: remove the soda cup from frame 10.
+
+The edit is derived entirely from source_pixels.npz.  The scene contains a red
+object, a light soda cup (with a narrow straw), and a green object arranged as a
+1-D collision chain.  Frames 1--9 are factual.  At frame 10 the cup is removed;
+the red object consequently travels through the vacated space and directly
+contacts the green object one frame later.  Their post-contact translation uses
+the measured damping curve of the source green object, delayed by one frame.
+"""
+from pathlib import Path
+
+import numpy as np
+from PIL import Image, ImageFilter
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def components(mask):
+    """Return 8-connected components as (size, mean_x, mean_y, bbox)."""
+    h, w = mask.shape
+    seen = np.zeros_like(mask, dtype=bool)
+    out = []
+    for y0, x0 in zip(*np.where(mask & ~seen)):
+        if seen[y0, x0]:
+            continue
+        stack = [(int(y0), int(x0))]
+        seen[y0, x0] = True
+        xs, ys = [], []
+        while stack:
+            y, x = stack.pop()
+            xs.append(x); ys.append(y)
+            for yy in range(max(0, y - 1), min(h, y + 2)):
+                for xx in range(max(0, x - 1), min(w, x + 2)):
+                    if mask[yy, xx] and not seen[yy, xx]:
+                        seen[yy, xx] = True
+                        stack.append((yy, xx))
+        out.append((len(xs), float(np.mean(xs)), float(np.mean(ys)),
+                    (min(xs), min(ys), max(xs), max(ys))))
+    return out
+
+
+def largest_center(mask, minimum=4):
+    cc = [c for c in components(mask) if c[0] >= minimum]
+    if not cc:
+        raise RuntimeError("Unable to locate source object")
+    return max(cc, key=lambda q: q[0])[1]
+
+
+def locate_objects(frames):
+    """Track the strongly colored cores of the three source objects."""
+    a = frames.astype(np.int16)
+    yy = np.indices(a.shape[1:3])[0]
+    band = (yy >= 55) & (yy < 86)
+    red, cup, green = [], [], []
+    for f in a:
+        # The cup body is cream/yellow and has two separated bright bands.  The
+        # largest component is its upper band and gives a stable horizontal
+        # center.  Color predicates are deliberately strict to reject scenery.
+        rm = band & (f[..., 0] - f[..., 1] > 32) & (f[..., 0] - f[..., 2] > 35)
+        cm = band & (f[..., 0] - f[..., 2] > 60) & (f[..., 1] - f[..., 2] > 35)
+        gm = band & (f[..., 1] - f[..., 0] > 22) & (f[..., 1] - f[..., 2] > 18)
+        red.append(largest_center(rm))
+        cup.append(largest_center(cm))
+        green.append(largest_center(gm))
+    return np.array(red), np.array(cup), np.array(green)
+
+
+def clean_plate(frames, red_x, cup_x, green_x):
+    """Recover an object-free plate by a masked temporal median.
+
+    The camera is locked.  During the first 36 frames every location along the
+    tabletop is visible in multiple source frames.  Excluding conservative
+    moving rectangles before taking the median therefore reveals the actual
+    background rather than spatially synthesizing it.
+    """
+    n = 36
+    work = frames[:n].astype(np.float32)
+    valid = np.ones((n, frames.shape[1], frames.shape[2]), dtype=bool)
+    for t in range(n):
+        boxes = [
+            (int(round(red_x[t])) - 17, int(round(red_x[t])) + 10, 53, 87),
+            (int(round(cup_x[t])) - 14, int(round(cup_x[t])) + 13, 53, 87),
+            (int(round(green_x[t])) - 10, int(round(green_x[t])) + 12, 53, 87),
+        ]
+        for x0, x1, y0, y1 in boxes:
+            valid[t, y0:y1, max(0, x0):min(frames.shape[2], x1)] = False
+    masked = np.where(valid[..., None], work, np.nan)
+    plate = np.nanmedian(masked, axis=0)
+    # No pixel should be missing, but retain a deterministic fallback.
+    fallback = np.median(work, axis=0)
+    plate = np.where(np.isnan(plate), fallback, plate)
+    return plate.astype(np.float32)
+
+
+def smooth_1d(values, radius=8):
+    pad = np.pad(values, ((radius, radius), (0, 0)), mode="edge")
+    cs = np.concatenate([np.zeros((1, values.shape[1])), np.cumsum(pad, axis=0)], axis=0)
+    return (cs[2 * radius + 1:] - cs[:-(2 * radius + 1)]) / (2 * radius + 1)
+
+
+def matched_plate(plate, frame):
+    """Match the clean plate to each frame's tiny low-frequency color drift."""
+    residual = frame.astype(np.float32) - plate
+    top = np.median(residual[43:52], axis=0)
+    bottom = np.median(residual[89:98], axis=0)
+    top = smooth_1d(top, 8)
+    bottom = smooth_1d(bottom, 8)
+    out = plate.copy()
+    for y in range(52, 90):
+        q = (y - 52) / 37.0
+        out[y] += (1.0 - q) * top + q * bottom
+    return np.clip(out, 0, 255)
+
+
+def feather_box(h, w, x0, y0, x1, y1, blur=1.2):
+    m = np.zeros((h, w), dtype=np.uint8)
+    m[max(0, y0):min(h, y1), max(0, x0):min(w, x1)] = 255
+    if blur:
+        m = np.asarray(Image.fromarray(m).filter(ImageFilter.GaussianBlur(blur)), dtype=np.float32)
+    return m / 255.0
+
+
+def extract_sprite(frame, base, x0, y0, x1, y1, anchor_x):
+    """Extract source RGB and a soft matte from its difference to clean plate."""
+    rgb = frame[y0:y1, x0:x1].astype(np.float32)
+    bg = base[y0:y1, x0:x1].astype(np.float32)
+    diff = np.max(np.abs(rgb - bg), axis=2)
+    # Low-contrast contact shadows remain translucent; object interiors become
+    # fully opaque.  This also preserves the native antialiasing at silhouettes.
+    alpha = np.clip((diff - 1.5) / 10.0, 0.0, 1.0)
+    alpha = np.asarray(
+        Image.fromarray(np.round(alpha * 255).astype(np.uint8)).filter(
+            ImageFilter.GaussianBlur(0.45)), dtype=np.float32) / 255.0
+    alpha[[0, -1], :] = 0
+    alpha[:, [0, -1]] = 0
+    return {"rgb": rgb, "alpha": alpha, "anchor": float(anchor_x - x0), "y0": y0}
+
+
+def paste_sprite(canvas, sprite, center_x):
+    rgb, alpha = sprite["rgb"], sprite["alpha"]
+    hh, ww = alpha.shape
+    x0 = int(round(center_x - sprite["anchor"]))
+    y0 = sprite["y0"]
+    xa, xb = max(0, x0), min(canvas.shape[1], x0 + ww)
+    ya, yb = max(0, y0), min(canvas.shape[0], y0 + hh)
+    if xa >= xb or ya >= yb:
+        return
+    sx0, sy0 = xa - x0, ya - y0
+    sx1, sy1 = sx0 + (xb - xa), sy0 + (yb - ya)
+    a = alpha[sy0:sy1, sx0:sx1, None]
+    canvas[ya:yb, xa:xb] = canvas[ya:yb, xa:xb] * (1.0 - a) + rgb[sy0:sy1, sx0:sx1] * a
+
+
+def generate():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    if source.shape != (96, 144, 256, 3) or source.dtype != np.uint8:
+        raise ValueError((source.shape, source.dtype))
+
+    red_x, cup_x, green_x = locate_objects(source)
+    plate = clean_plate(source, red_x, cup_x, green_x)
+    bases = [matched_plate(plate, f) for f in source]
+
+    # Use isolated, pre-intervention appearances.  Frame 9 (index 8) has the
+    # final factual lighting and crisp object silhouettes.
+    rt = 8
+    rx0 = int(round(red_x[rt])) - 14
+    red_sprite = extract_sprite(source[rt], bases[rt], rx0, 53, rx0 + 24, 87, red_x[rt])
+    gx0 = int(round(green_x[rt])) - 10
+    green_sprite = extract_sprite(source[rt], bases[rt], gx0, 53, gx0 + 22, 87, green_x[rt])
+
+    result = source.copy()
+
+    # Frame 10 (index 9): intervene only on the soda cup.  Keeping the two
+    # neighboring objects pixel-exact makes the factual-to-counterfactual cut
+    # visually seamless.
+    t = 9
+    canvas = result[t].astype(np.float32)
+    x0 = int(round(cup_x[t])) - 9
+    x1 = int(round(cup_x[t])) + 10
+    matte = feather_box(144, 256, x0, 52, x1, 87, 1.0)[..., None]
+    canvas = canvas * (1.0 - matte) + bases[t] * matte
+    result[t] = np.clip(np.rint(canvas), 0, 255).astype(np.uint8)
+
+    # The red object crosses the removed cup's former location.  It reaches the
+    # green object at index 11.  Thereafter both share the measured source green
+    # trajectory, delayed by one frame, with their silhouettes touching.
+    out_red = red_x.copy()
+    out_green = green_x.copy()
+    out_green[9:11] = green_x[9]                 # stationary until direct hit
+    out_red[9] = red_x[9]
+    out_red[10] = red_x[10]
+    spacing = 14.7
+    for t in range(11, len(source)):
+        out_green[t] = green_x[t - 1]
+        out_red[t] = out_green[t] - spacing
+
+    for t in range(10, len(source)):
+        canvas = source[t].astype(np.float32)
+        # Remove all three objects from their factual source positions.  A
+        # conservative connected box includes their soft contact shadows.
+        left = int(np.floor(min(red_x[t] - 17, cup_x[t] - 14, green_x[t] - 10)))
+        right = int(np.ceil(max(red_x[t] + 10, cup_x[t] + 13, green_x[t] + 12)))
+        matte = feather_box(144, 256, left, 51, right, 88, 1.15)[..., None]
+        canvas = canvas * (1.0 - matte) + bases[t] * matte
+        paste_sprite(canvas, red_sprite, out_red[t])
+        paste_sprite(canvas, green_sprite, out_green[t])
+        result[t] = np.clip(np.rint(canvas), 0, 255).astype(np.uint8)
+
+    return result
+
+
+if __name__ == "__main__":
+    frames = generate()
+    artifact_io.save(frames, CASE_DIR)
+    print("saved", frames.shape, frames.dtype, "to", CASE_DIR)

def components(mask):

def smooth_1d(values, radius=8):
    pad = np.pad(values, ((radius, radius), (0, 0)), mode=\"edge\")
    cs = np.concatenate([np.zeros((1, values.shape[1])), np.cumsum(pad, axis=0)], axis=0)
    return (cs[2 * radius + 1:] - cs[:-(2 * radius + 1)]) / (2 * radius + 1)


def matched_plate(plate, frame):

    residual = frame.astype(np.float32) - plate

    bottom = np.median(residual[89:98], axis=0)
    top = smooth_1d(top, 8)
    bottom = smooth_1d(bottom, 8)
