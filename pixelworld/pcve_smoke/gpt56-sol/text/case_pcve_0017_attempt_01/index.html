+"""Generate the counterfactual video for this case.
+
+The star-marked ball is freely moving until its first contact (source frame 12),
+so changing restitution cannot alter the preceding motion.  After contact its
+separation/rebound speed is proportional to restitution.  We therefore keep
+the factual approach exactly, and run the local post-contact interaction at
+1/4 speed.  Only the compact interaction area is retimed; the room, camera,
+lighting and all unaffected pixels stay on the original 24 fps timeline.
+
+Dense bidirectional warping is used for the quarter-frame samples.  This is
+considerably smoother than repeating each source frame four times and avoids
+inventing or redrawing any of the source's detailed object appearance.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+CONTACT_INDEX = 11          # zero based: source frame 12
+RESTITUTION_RATIO = 0.25
+
+
+def feathered_interaction_mask(height: int, width: int) -> np.ndarray:
+    """Soft mask covering the ball rebound and every object it can affect."""
+    mask = np.zeros((height, width), np.float32)
+    # The ball enters the assembly near (82, 70).  Its rebound and the moving
+    # receiving body remain inside this box; the lower edge intentionally runs
+    # off-screen so no seam can cross the falling body.
+    mask[45:height, 42:132] = 1.0
+    mask = cv2.GaussianBlur(mask, (0, 0), sigmaX=4.0, sigmaY=4.0)
+    return np.clip(mask, 0.0, 1.0)[..., None]
+
+
+def flow_pair(a: np.ndarray, b: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
+    """Return dense a->b and b->a optical flow in pixel coordinates."""
+    ga = cv2.cvtColor(a, cv2.COLOR_RGB2GRAY)
+    gb = cv2.cvtColor(b, cv2.COLOR_RGB2GRAY)
+    args = dict(
+        pyr_scale=0.5,
+        levels=4,
+        winsize=21,
+        iterations=4,
+        poly_n=7,
+        poly_sigma=1.5,
+        flags=cv2.OPTFLOW_FARNEBACK_GAUSSIAN,
+    )
+    return cv2.calcOpticalFlowFarneback(ga, gb, None, **args), cv2.calcOpticalFlowFarneback(gb, ga, None, **args)
+
+
+def remap_toward(image: np.ndarray, flow: np.ndarray, amount: float,
+                 grid_x: np.ndarray, grid_y: np.ndarray) -> np.ndarray:
+    """Move an endpoint image `amount` of the way along its endpoint flow."""
+    map_x = (grid_x - amount * flow[..., 0]).astype(np.float32)
+    map_y = (grid_y - amount * flow[..., 1]).astype(np.float32)
+    return cv2.remap(
+        image,
+        map_x,
+        map_y,
+        interpolation=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_REFLECT_101,
+    )
+
+
+def motion_interpolate(a: np.ndarray, b: np.ndarray, alpha: float,
+                       flows: tuple[np.ndarray, np.ndarray],
+                       grid_x: np.ndarray, grid_y: np.ndarray) -> np.ndarray:
+    """Bidirectionally warp and blend two adjacent RGB source frames."""
+    if alpha <= 1e-7:
+        return a
+    if alpha >= 1.0 - 1e-7:
+        return b
+    forward, backward = flows
+    wa = remap_toward(a, forward, alpha, grid_x, grid_y).astype(np.float32)
+    wb = remap_toward(b, backward, 1.0 - alpha, grid_x, grid_y).astype(np.float32)
+    return np.clip(wa * (1.0 - alpha) + wb * alpha + 0.5, 0, 255).astype(np.uint8)
+
+
+def generate() -> np.ndarray:
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3) and source.dtype == np.uint8
+
+    frame_count, height, width, _ = source.shape
+    yy, xx = np.mgrid[0:height, 0:width]
+    grid_x, grid_y = xx.astype(np.float32), yy.astype(np.float32)
+    local_mask = feathered_interaction_mask(height, width)
+
+    # Only 22 adjacent pairs are needed: source times 11.00 through 32.00.
+    max_local_time = CONTACT_INDEX + RESTITUTION_RATIO * (frame_count - 1 - CONTACT_INDEX)
+    last_pair = min(frame_count - 2, int(np.floor(max_local_time)))
+    flows = {i: flow_pair(source[i], source[i + 1]) for i in range(CONTACT_INDEX, last_pair + 1)}
+
+    output = source.copy()
+    for out_index in range(CONTACT_INDEX + 1, frame_count):
+        local_time = CONTACT_INDEX + RESTITUTION_RATIO * (out_index - CONTACT_INDEX)
+        left = int(np.floor(local_time))
+        alpha = float(local_time - left)
+        if left >= frame_count - 1:
+            local = source[-1]
+        else:
+            local = motion_interpolate(
+                source[left], source[left + 1], alpha, flows[left], grid_x, grid_y
+            )
+
+        # Keep the current factual frame everywhere unrelated to the impact.
+        factual = source[out_index].astype(np.float32)
+        mixed = factual * (1.0 - local_mask) + local.astype(np.float32) * local_mask
+        output[out_index] = np.clip(mixed + 0.5, 0, 255).astype(np.uint8)
+
+    # The free-flight history, including the exact contact frame, is inviolate.
+    assert np.array_equal(output[: CONTACT_INDEX + 1], source[: CONTACT_INDEX + 1])
+    return output
+
+
+if __name__ == "__main__":
+    frames = generate()
+    artifact_io.save(frames, CASE_DIR)
+    print(f"saved {frames.shape[0]} RGB frames to {CASE_DIR}")
