+"""Generate the counterfactual: the red ball has 0.4x source restitution.
+
+The source is a fixed-camera photoreal render.  We estimate its clean plate by a
+temporal median, recover the ball's appearance from frame 0, and render the
+same initial motion until the first contact.  At contact only the normal
+(outgoing vertical) speed is changed, by the requested 0.4 factor.  Gravity and
+horizontal motion are retained.  A soft contact shadow is composited on the
+receiving surface.
+"""
+from pathlib import Path
+import numpy as np
+from PIL import Image
+from scipy.ndimage import binary_dilation, distance_transform_edt, map_coordinates
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+src = np.load(ROOT / "source_pixels.npz")["rgb"]
+N, H, W, _ = src.shape
+
+# Robust clean plate: the ball crosses each pixel only briefly.
+plate = np.median(src, axis=0).astype(np.uint8)
+
+# Empirically recovered source trajectory (centres, source pixel coordinates).
+# Frames 0..12 are the unmodified ballistic fall.  The receiving surface is at
+# y~=94, so the radius-12 ball first contacts at centre y~=82 on frame 12.
+source_x = np.array([127,128,129,130,131,132,133,134,135,136,137,137,138], float)
+source_y = np.array([22,23,24,26,30,33,38,43,50,56,63,72,82], float)
+
+# Texture extraction from the unobstructed ball in frame zero.
+cx0, cy0 = source_x[0], source_y[0]
+rad = 12.2
+pad = 14
+x0, y0 = int(cx0)-pad, int(cy0)-pad
+patch = src[0, y0:y0+2*pad+1, x0:x0+2*pad+1].astype(float)
+yy, xx = np.mgrid[-pad:pad+1, -pad:pad+1]
+dist = np.sqrt(xx*xx + yy*yy)
+# Matte follows the source silhouette, with a one-pixel antialiased rim.
+alpha_patch = np.clip(rad + .55 - dist, 0, 1)
+
+# Contact kinematics.  Pixel y points downward. Fit the incoming source fall;
+# outgoing speed is 0.4 of the observed original outgoing speed (frames 13-17).
+t_contact = 12.0
+x_contact = 138.0
+y_contact = 82.0
+vx = np.polyfit(np.arange(4, 13), source_x[4:], 1)[0]
+# Use source upward post-impact displacement: 82 -> 74 -> 67 -> 62 ...
+orig_v_out = -8.65
+new_v_out = 0.4 * orig_v_out
+g = 0.78                         # source gravity, px/frame^2
+floor_y = y_contact
+
+# Lower restitution gives one short hop.  After the next contact, damp the tiny
+# residual bounce and let the ball settle while preserving tangential drift.
+def center_at(t):
+    if t <= t_contact:
+        # Exact source history through impact (linear interpolation permits AA).
+        return np.interp(t, np.arange(13), source_x), np.interp(t, np.arange(13), source_y)
+    u = t - t_contact
+    y = y_contact + new_v_out*u + .5*g*u*u
+    # First root after u=0 is 8.87 frames. Clamp at the surface thereafter;
+    # a restitution-scaled secondary hop would be sub-pixel in this rendering.
+    landing = -2*new_v_out/g
+    if u >= landing:
+        y = floor_y
+    x = x_contact + vx*u
+    return x, y
+
+# Atmospheric/contact shadow sampled from source style: broad, subtle and
+# strongest near the surface. It helps spatial grounding without changing the
+# static set.
+def add_shadow(frame, cx, cy):
+    clearance = max(0.0, floor_y-cy)
+    strength = 0.17*np.exp(-clearance/8.0)
+    if strength < .004:
+        return frame
+    Y, X = np.mgrid[:H, :W]
+    # shadow lies on the visible tabletop immediately behind/below contact
+    sx, sy = cx+2.5, 91.5
+    sigx = 9.0 + .12*clearance
+    sigy = 2.2 + .035*clearance
+    a = strength*np.exp(-.5*((X-sx)/sigx)**2-.5*((Y-sy)/sigy)**2)
+    # Restrict to tabletop, not upper background.
+    a *= (Y >= 68) & (Y <= 98)
+    return frame*(1-a[...,None])
+
+def render_ball(frame, cx, cy):
+    # Subpixel placement via inverse bilinear sampling of the source patch.
+    xmin = max(0, int(np.floor(cx-pad-1))); xmax = min(W, int(np.ceil(cx+pad+2)))
+    ymin = max(0, int(np.floor(cy-pad-1))); ymax = min(H, int(np.ceil(cy+pad+2)))
+    Y, X = np.mgrid[ymin:ymax, xmin:xmax]
+    px = X-cx+pad; py = Y-cy+pad
+    coords = np.array([py, px])
+    tex = np.stack([map_coordinates(patch[...,c], coords, order=1, mode='nearest') for c in range(3)], -1)
+    a = map_coordinates(alpha_patch, coords, order=1, mode='constant', cval=0.0)
+    roi = frame[ymin:ymax, xmin:xmax]
+    frame[ymin:ymax, xmin:xmax] = roi*(1-a[...,None]) + tex*a[...,None]
+    return frame
+
+out = np.empty_like(src)
+for t in range(N):
+    cx, cy = center_at(float(t))
+    canvas = plate.astype(float).copy()
+    canvas = add_shadow(canvas, cx, cy)
+    canvas = render_ball(canvas, cx, cy)
+    out[t] = np.clip(np.rint(canvas), 0, 255).astype(np.uint8)
+
+artifact_io.save(out, ROOT)

same initial motion until the first contact.  At contact only the normal
(outgoing vertical) speed is changed, by the requested 0.4 factor.  Gravity and
horizontal motion are retained.  A soft contact shadow is composited on the
receiving surface.

from pathlib import Path
import numpy as np
from PIL import Image
from scipy.ndimage import binary_dilation, distance_transform_edt, map_coordinates
import artifact_io

ROOT = Path(__file__).resolve().parent
src = np.load(ROOT / \"source_pixels.npz\")[\"rgb\"]
N, H, W, _ = src.shape

# Robust clean plate: the ball crosses each pixel only briefly.
plate = np.median(src, axis=0).astype(np.uint8)

# Empirically recovered source trajectory (centres, source pixel coordinates).
# Frames 0..12 are the unmodified ballistic fall.  The receiving surface is at
# y~=94, so the radius-12 ball first contacts at centre y~=82 on frame 12.
source_x = np.array([127,128,129,130,131,132,133,134,135,136,137,137,138], float)
source_y = np.array([22,23,24,26,30,33,38,43,50,56,63,72,82], float)

# Texture extraction from the unobstructed ball in frame zero.
cx0, cy0 = source_x[0], source_y[0]
rad = 12.2
pad = 14
x0, y0 = int(cx0)-pad, int(cy0)-pad
patch = src[0, y0:y0+2*pad+1, x0:x0+2*pad+1].astype(float)
yy, xx = np.mgrid[-pad:pad+1, -pad:pad+1]
dist = np.sqrt(xx*xx + yy*yy)
# Matte follows the source silhouette, with a one-pixel antialiased rim.
alpha_patch = np.clip(rad + .55 - dist, 0, 1)

# Contact kinematics.  Pixel y points downward. Fit the incoming source fall;
# outgoing speed is 0.4 of the observed original outgoing speed (frames 13-17).
t_contact = 12.0
x_contact = 138.0
y_contact = 82.0
vx = np.polyfit(np.arange(4, 13), source_x[4:], 1)[0]
# Use source upward post-impact displacement: 82 -> 74 -> 67 -> 62 ...
orig_v_out = -8.65
new_v_out = 0.4 * orig_v_out
g = 0.78                         # source gravity, px/frame"'^2
floor_y = y_contact

# Lower restitution gives one short hop.  After the next contact, damp the tiny
# residual bounce and let the ball settle while preserving tangential drift.
def center_at(t):
    if t <= t_contact:
        # Exact source history through impact (linear interpolation permits AA).
        return np.interp(t, np.arange(13), source_x), np.interp(t, np.arange(13), source_y)
    u = t - t_contact
    y = y_contact + new_v_out*u + .5*g*u*u
    # First root after u=0 is 8.87 frames. Clamp at the surface thereafter;
    # a restitution-scaled secondary hop would be sub-pixel in this rendering.
    landing = -2*new_v_out/g
    if u >= landing:
        y = floor_y
    x = x_contact + vx*u
    return x, y

# Atmospheric/contact shadow sampled from source style: broad, subtle and
# strongest near the surface. It helps spatial grounding without changing the
# static set.
def add_shadow(frame, cx, cy):
    clearance = max(0.0, floor_y-cy)
    strength = 0.17*np.exp(-clearance/8.0)
    if strength < .004:
        return frame
    Y, X = np.mgrid[:H, :W]
    # shadow lies on the visible tabletop immediately behind/below contact
    sx, sy = cx+2.5, 91.5
    sigx = 9.0 + .12*clearance
    sigy = 2.2 + .035*clearance
    a = strength*np.exp(-.5*((X-sx)/sigx)**2-.5*((Y-sy)/sigy)**2)
    # Restrict to tabletop, not upper background.
    a *= (Y >= 68) & (Y <= 98)
    return frame*(1-a[...,None])

def render_ball(frame, cx, cy):
    # Subpixel placement via inverse bilinear sampling of the source patch.
    xmin = max(0, int(np.floor(cx-pad-1))); xmax = min(W, int(np.ceil(cx+pad+2)))
    ymin = max(0, int(np.floor(cy-pad-1))); ymax = min(H, int(np.ceil(cy+pad+2)))
    Y, X = np.mgrid[ymin:ymax, xmin:xmax]
    px = X-cx+pad; py = Y-cy+pad
    coords = np.array([py, px])
    tex = np.stack([map_coordinates(patch[...,c], coords, order=1, mode='"'nearest') for c in range(3)], -1)
    a = map_coordinates(alpha_patch, coords, order=1, mode='constant', cval=0.0)
    roi = frame[ymin:ymax, xmin:xmax]
    frame[ymin:ymax, xmin:xmax] = roi*(1-a[...,None]) + tex*a[...,None]
    return frame

out = np.empty_like(src)
for t in range(N):
    cx, cy = center_at(float(t))
    canvas = plate.astype(float).copy()
    canvas = add_shadow(canvas, cx, cy)
    canvas = render_ball(canvas, cx, cy)
    out[t] = np.clip(np.rint(canvas), 0, 255).astype(np.uint8)

artifact_io.save(out, ROOT)
