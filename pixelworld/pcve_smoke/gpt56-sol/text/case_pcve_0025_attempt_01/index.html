+def extract_ball_sprite(frame, bg, x, y, r=14):
+    """Extract the small sphere at (x,y), using the static settled frame as background."""
+    h, w = frame.shape[:2]
+    yy, xx = np.mgrid[:h, :w]
+    circle = ((xx-x)**2 + (yy-y)**2 <= r*r).astype(np.float32)
+    color_delta = np.max(np.abs(frame.astype(np.int16)-bg.astype(np.int16)), axis=2)
+    # Keep strong ball pixels, plus a narrow soft fringe for antialiasing.
+    raw = ((color_delta > 7).astype(np.float32) * circle)
+    raw = cv2.morphologyEx(raw, cv2.MORPH_CLOSE, np.ones((3,3),np.uint8))
+    alpha = np.clip(soft_mask(raw, .7), 0, 1) * circle
+    x0=max(0,int(np.floor(x-r-2))); x1=min(w,int(np.ceil(x+r+3)))
+    y0=max(0,int(np.floor(y-r-2))); y1=min(h,int(np.ceil(y+r+3)))
+    return frame[y0:y1,x0:x1].copy(), alpha[y0:y1,x0:x1].copy(), (x-x0,y-y0)
+
+
+def erase_ball(frame, bg, cx, cy, r):
+    """Restore static scene under the factual ball only, with a feathered edge."""
+    h,w=frame.shape[:2]; yy,xx=np.mgrid[:h,:w]
+    a=np.clip((r+1.0-np.hypot(xx-cx,yy-cy))/1.25,0,1)[...,None]
+    return np.rint(frame*(1-a)+bg*a).astype(np.uint8)
+

+def paste_sprite(dst, sprite, alpha, anchor, cx, cy):
+    h,w=dst.shape[:2]; sh,sw=sprite.shape[:2]
+    x0=int(round(cx-anchor[0])); y0=int(round(cy-anchor[1]))
+    xa=max(0,x0); ya=max(0,y0); xb=min(w,x0+sw); yb=min(h,y0+sh)
+    if xa>=xb or ya>=yb: return

+    a=alpha[sy:sy+yb-ya,sx:sx+xb-xa,None]

    sprite,alpha,anchor=extract_ball_sprite(src[5],bg,sx,sy,14)

    # Reduced friction: kinetic deceleration along the floor is only 30% as large.
    # Infer source deceleration from the visible trajectory and preserve launch velocity.
    # The ball reaches the source stopping pocket at ~frame 18; the counterfactual keeps
    # sliding, then is gently constrained by the visible left face of the target near x=107.
    mu_scale=.3
    ax_cf=ax*mu_scale
    ay_cf=ay*mu_scale
    centers=[]
    for t in range(96):
        x=x0+vx*t+.5*ax_cf*t*t
        y=y0+vy*t+.5*ay_cf*t*t
        # Perspective floor path tends toward the target/base contact line.
        y=max(68.0, y)
        # Low-friction sphere slides beneath/along the obstacle rather than sticking:
        # after reaching it, it glances right with retained tangential speed.
        if x>107.0 and t>8:
            hit_t=8.0
            xhit=x0+vx*hit_t+.5*ax_cf*hit_t*hit_t
            speed=max(0.0, vx+ax_cf*hit_t)
            dt=t-hit_t
            x=xhit + .52*speed*dt - .5*.018*dt*dt
            y=68.0 + 0.012*dt*dt
        centers.append((x,y))

    for t in range(96):
        frame=src[t].copy()
        # Remove only the factual ball. At overlap, keep the target/obstacle from bg.
        if t<19:
            fx,fy=factual[t]
            radius=max(8.5, 14.0-.28*t)
            frame=erase_ball(frame,bg,fx,fy,radius)
        # After factual settling the source equals bg and needs no cleanup.
        cx,cy=centers[t]
        # Scale down with depth using source perspective, and fade naturally offscreen.
        scale=np.clip(.58 + .42*(cy-68)/(122.8-68),.55,1.0)
        if abs(scale-1)>1e-3:
            sw=max(1,round(sprite.shape[1]*scale)); sh=max(1,round(sprite.shape[0]*scale))
            sp=cv2.resize(sprite,(sw,sh),interpolation=cv2.INTER_AREA)
            al=cv2.resize(alpha,(sw,sh),interpolation=cv2.INTER_LINEAR)
            an=(anchor[0]*scale,anchor[1]*scale)
        else: sp,al,an=sprite,alpha,anchor
        paste_sprite(frame,sp,al,an,cx,cy)
        out[t]=frame
    return out


if __name__ == '__main__':
    frames=generate()
    assert frames.shape==(96,144,256,3) and frames.dtype==np.uint8
    artifact_io.save(frames, CASE_DIR)
