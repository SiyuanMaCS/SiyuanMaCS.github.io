+"""Generate the counterfactual video: remove the wooden block from frame 17.
+
+The camera and set are static.  The wooden cuboid occupies x=109..174,
+y=55..110 before the collision.  A clean plate is reconstructed from source
+frames in which each location is unobscured.  The incoming red ball is then
+continued from its pre-intervention velocity, with a small gravity term, since
+there is no block to collide with.
+"""
+from pathlib import Path
+import numpy as np
+from PIL import Image
+from scipy import ndimage as ndi
+import artifact_io
+
+CASE_DIR = Path(__file__).resolve().parent
+
+def build_clean_plate(src):
+    # Pre-collision frame 1 is the best exact observation of the stationary
+    # block and set.  The block's footprint is filled from later observations
+    # of the same static set after it has moved away.  Row-wise donor choices
+    # avoid the ball's late-time path and preserve perspective/texture.
+    plate = src[0].copy()
+    # The now exposed wall/table area. Late source frame 41 has the block far
+    # right; frame 96 is only used at the block's extreme right boundary.
+    plate[52:116, 106:176] = src[40, 52:116, 106:176]
+    # A tiny upper-right strip remains under the block at frame 41; it is fully
+    # exposed in late frames. Blend donor frames robustly against video noise.
+    plate[52:116, 164:177] = np.median(src[36:65, 52:116, 164:177], axis=0).astype(np.uint8)
+    # Keep untouched source background outside the actual removal footprint.
+    return plate
+
+def extract_ball_rgba(src, frame_index, plate):
+    """Segment the incoming red ball and its contact shadow from source."""
+    im = src[frame_index]
+    yy, xx = np.mgrid[:144, :256]
+    # Ball center follows ~7 px/frame before contact. Broad ROI captures its
+    # lower shadow while color/difference tests reject the wooden set.
+    cx = 11.5 + 7.0 * frame_index
+    cy = 62.0 + 1.45 * frame_index
+    roi = ((xx-cx)/33.0)**2 + ((yy-cy)/34.0)**2 < 1.0
+    r,g,b = [im[...,k].astype(int) for k in range(3)]
+    red = (r > 130) & (r-g > 42) & (g-b > 16)
+    changed = np.max(np.abs(im.astype(int)-plate.astype(int)), axis=2) > 9
+    # Main opaque ball; only retain the component connected to the expected ROI.
+    m = roi & (red | changed)
+    m = ndi.binary_closing(m, np.ones((5,5)))
+    m = ndi.binary_fill_holes(m)
+    lab,n = ndi.label(m)
+    if n:
+        hit = int(lab[int(round(cy)), int(round(cx))])
+        if hit == 0:
+            sizes = ndi.sum(m, lab, range(1,n+1)); hit=int(np.argmax(sizes))+1
+        m = lab == hit
+    m = ndi.binary_dilation(m, iterations=1)
+    alpha = ndi.gaussian_filter(m.astype(np.float32), .65)
+    alpha = np.clip(alpha,0,1)
+    ys,xs=np.where(alpha>.01)
+    x0,x1=max(0,xs.min()-2),min(256,xs.max()+3)
+    y0,y1=max(0,ys.min()-2),min(144,ys.max()+3)
+    return im[y0:y1,x0:x1].copy(), alpha[y0:y1,x0:x1], (x0,y0), (cx,cy)
+
+def composite(frame, patch, alpha, x, y):
+    h,w=alpha.shape; H,W=frame.shape[:2]
+    x0=max(0,x); y0=max(0,y); x1=min(W,x+w); y1=min(H,y+h)
+    if x0>=x1 or y0>=y1: return
+    sx=x0-x; sy=y0-y
+    a=alpha[sy:sy+y1-y0, sx:sx+x1-x0,None]
+    dst=frame[y0:y1,x0:x1].astype(np.float32)
+    fg=patch[sy:sy+y1-y0,sx:sx+x1-x0].astype(np.float32)
+    frame[y0:y1,x0:x1]=np.clip(fg*a+dst*(1-a),0,255).astype(np.uint8)
+
+def generate():
+    src=np.load(CASE_DIR/'source_pixels.npz')['rgb']
+    out=src.copy()
+    plate=build_clean_plate(src)
+
+    # Texture source from the last factual frame. Frame 16 is immediately
+    # before intervention and has the ball nearly at the block; frame 13 gives
+    # a clean, complete sphere. Continue its factual 7 px/frame motion.
+    patch,alpha,(px,py),(scx,scy)=extract_ball_rgba(src,12,plate)
+
+    for f in range(16,96):
+        # Begin with the factual frame to retain all unaffected pixels, then
+        # restore every pixel touched by the original block/consequences from
+        # the clean static plate.  This region includes the original block and
+        # its source trajectory to the right edge.
+        fr=src[f].copy()
+        # Object footprint: source block moves right after collision, spanning
+        # roughly x=109+4.5*(f-12) to +66, y=55..122. Remove it everywhere.
+        bx0=max(104, int(round(109 + 4.45*max(0,f-12))) - 4)
+        bx1=min(256, bx0+79)
+        # Also clear its original location, including old cast shadow.
+        fr[52:124,104:178]=plate[52:124,104:178]
+        if bx0<256:
+            fr[52:136,bx0:bx1]=plate[52:136,bx0:bx1]
+        # As the source block exits, its trajectory extends through right edge.
+        if f<37:
+            tx0=max(104,int(109+4.0*(f-12))-5)
+            fr[52:143,tx0:256]=plate[52:143,tx0:256]
+
+        # Counterfactual ball: constant horizontal velocity and downward
+        # acceleration in screen perspective. It quickly travels out of frame.
+        dt=f-12
+        cx=scx+7.0*dt
+        cy=scy+1.45*dt+0.075*dt*dt
+        x=int(round(px+(cx-scx))); y=int(round(py+(cy-scy)))
+        composite(fr,patch,alpha,x,y)
+        out[f]=fr
+    return out.astype(np.uint8)
+
+if __name__=='__main__':
+    frames=generate()
+    artifact_io.save(frames, CASE_DIR)

def build_clean_plate(src):
    # Pre-collision frame 1 is the best exact observation of the stationary
    # block and set.  The block's footprint is filled from later observations
    # of the same static set after it has moved away.  Row-wise donor choices
    # avoid the ball's late-time path and preserve perspective/texture.
    plate = src[0].copy()
    # The now exposed wall/table area. Late source frame 41 has the block far
    # right; frame 96 is only used at the block's extreme right boundary.
    plate[52:116, 106:176] = src[40, 52:116, 106:176]
    # A tiny upper-right strip remains under the block at frame 41; it is fully
    # exposed in late frames. Blend donor frames robustly against video noise.
    plate[52:116, 164:177] = np.median(src[36:65, 52:116, 164:177], axis=0).astype(np.uint8)
    # Keep untouched source background outside the actual removal footprint.
    return plate

def extract_ball_rgba(src, frame_index, plate):
    \"\"\"Segment the incoming red ball and its contact shadow from source.\"\"\"
    im = src[frame_index]
    yy, xx = np.mgrid[:144, :256]
    # Ball center follows ~7 px/frame before contact. Broad ROI captures its
    # lower shadow while color/difference tests reject the wooden set.
    cx = 11.5 + 7.0 * frame_index
    cy = 62.0 + 1.45 * frame_index
    roi = ((xx-cx)/33.0)**2 + ((yy-cy)/34.0)**2 < 1.0
    r,g,b = [im[...,k].astype(int) for k in range(3)]
    red = (r > 130) & (r-g > 42) & (g-b > 16)
    changed = np.max(np.abs(im.astype(int)-plate.astype(int)), axis=2) > 9
    # Main opaque ball; only retain the component connected to the expected ROI.
    m = roi & (red | changed)
    m = ndi.binary_closing(m, np.ones((5,5)))
    m = ndi.binary_fill_holes(m)
    lab,n = ndi.label(m)
    if n:
        hit = int(lab[int(round(cy)), int(round(cx))])
        if hit == 0:
            sizes = ndi.sum(m, lab, range(1,n+1)); hit=int(np.argmax(sizes))+1
        m = lab == hit
    m = ndi.binary_dilation(m, iterations=1)
    alpha = ndi.gaussian_filter(m.astype(np.float32), .65)
    alpha = np.clip(alpha,0,1)
    ys,xs=np.where(alpha>.01)
    x0,x1=max(0,xs.min()-2),min(256,xs.max()+3)
    y0,y1=max(0,ys.min()-2),min(144,ys.max()+3)
    return im[y0:y1,x0:x1].copy(), alpha[y0:y1,x0:x1], (x0,y0), (cx,cy)

def composite(frame, patch, alpha, x, y):
    h,w=alpha.shape; H,W=frame.shape[:2]
    x0=max(0,x); y0=max(0,y); x1=min(W,x+w); y1=min(H,y+h)
    if x0>=x1 or y0>=y1: return
    sx=x0-x; sy=y0-y
    a=alpha[sy:sy+y1-y0, sx:sx+x1-x0,None]
    dst=frame[y0:y1,x0:x1].astype(np.float32)
    fg=patch[sy:sy+y1-y0,sx:sx+x1-x0].astype(np.float32)
    frame[y0:y1,x0:x1]=np.clip(fg*a+dst*(1-a),0,255).astype(np.uint8)

    # before intervention and has the ball nearly at the block; frame 13 gives
    # a clean, complete sphere. Continue its factual 7 px/frame motion.
    patch,alpha,(px,py),(scx,scy)=extract_ball_rgba(src,12,plate)

        bx0=max(104, int(round(109 + 4.45*max(0,f-12))) - 4)

        if bx0<256:

        # As the source block exits, its trajectory extends through right edge.
        if f<37:
