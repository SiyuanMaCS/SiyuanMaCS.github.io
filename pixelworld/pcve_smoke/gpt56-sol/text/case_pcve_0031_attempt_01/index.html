+"""Generate the counterfactual: red ball launched at 6x its original speed.
+
+Only the supplied RGB tensor is used.  The camera is locked, so a clean plate is
+recovered temporally.  The source sphere and its height-dependent cast/contact
+shadow are isolated each frame and translated according to the changed launch
+velocity.  Gravity and vertical collision timing remain factual because the
+original launch is horizontal.
+"""
+from pathlib import Path
+import numpy as np
+from scipy.ndimage import gaussian_filter, label, shift
+import artifact_io
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def track_red_ball(frames):
+    """Subpixel center of the largest strongly red component in every frame."""
+    centers = []
+    for frame in frames:
+        rgb = frame.astype(np.int16)
+        redness = rgb[..., 0] - np.maximum(rgb[..., 1], rgb[..., 2])
+        components, count = label(redness >= 58)
+        candidates = []
+        for component in range(1, count + 1):
+            yy, xx = np.where(components == component)
+            if len(xx) >= 80:
+                candidates.append((len(xx), component, yy, xx))
+        if not candidates:
+            raise RuntimeError("Could not locate the red ball")
+        _, _, yy, xx = max(candidates)
+        weights = np.maximum(redness[yy, xx] - 45, 1)
+        centers.append((np.average(xx, weights=weights),
+                        np.average(yy, weights=weights)))
+    return np.asarray(centers, dtype=np.float32)
+
+
+def make_clean_plate(frames):
+    """Robust bright-background estimate that rejects the darker ball/shadow."""
+    f = frames.astype(np.float32)
+    luminance = f.mean(axis=3)
+    # Median of the twenty brightest observations is stable against codec noise,
+    # yet every pixel on the object's path is selected from uncovered frames.
+    top = np.argpartition(luminance, -20, axis=0)[-20:]
+    gathered = np.take_along_axis(f, top[..., None], axis=0)
+    return np.median(gathered, axis=0).astype(np.float32)
+
+
+def soft_box(xx, yy, x0, x1, y0, y1, feather=3.0):
+    """Soft rectangular support used only to reject unrelated image noise."""
+    return (np.clip((xx - x0) / feather, 0, 1) *
+            np.clip((x1 - xx) / feather, 0, 1) *
+            np.clip((yy - y0) / feather, 0, 1) *
+            np.clip((y1 - yy) / feather, 0, 1))
+
+
+def generate():
+    frames = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    if frames.shape != (96, 144, 256, 3) or frames.dtype != np.uint8:
+        raise ValueError(f"Unexpected source tensor: {frames.shape}, {frames.dtype}")
+
+    centers = track_red_ball(frames)
+    clean = make_clean_plate(frames)
+    yy, xx = np.indices(frames.shape[1:3], dtype=np.float32)
+
+    # Fit only the uninterrupted first flight.  Its horizontal screen velocity
+    # is 0.93 px/frame; the y fit confirms constant gravity and no meaningful
+    # initial vertical launch component.
+    initial_vx = float(np.polyfit(np.arange(13), centers[:13, 0], 1)[0])
+
+    output = np.empty_like(frames)
+    for t, source_u8 in enumerate(frames):
+        source = source_u8.astype(np.float32)
+        cx, cy = centers[t]
+

+        radius = np.hypot(xx - cx, yy - cy)
+        ball_mask = np.clip((15.0 - radius) / 1.25, 0.0, 1.0)
+

+        # ground.  It changes shape with height, so use the same source time and
+        # translate it with the sphere rather than redrawing a generic ellipse.
+        darkness = np.maximum((clean - source).mean(axis=2), 0.0)
+        shadow_signal = gaussian_filter(darkness, 0.9)
+        shadow_mask = np.clip((shadow_signal - 0.65) / 1.65, 0.0, 1.0)
+        shadow_mask *= soft_box(xx, yy, cx - 38, cx + 52, 53, 132, 4.0)
+
+        mask = np.maximum(ball_mask, shadow_mask)[..., None]
+        residual = (source - clean) * mask
+
+        # Remove the factual object but retain the exact source background and
+        # its tiny frame-to-frame variations everywhere else.
+        base = source - residual
+
+        # v_new = 6*v_original.  With unchanged gravity/contact impulses, the
+        # counterfactual differs by an added 5*v_initial*t horizontal displacement.
+        dx = 5.0 * initial_vx * t

+                      mode="constant", cval=0.0, prefilter=False)
+        output[t] = np.clip(np.rint(base + moved), 0, 255).astype(np.uint8)
+

+    output[0] = frames[0]
+    return output

+    result = generate()
+    artifact_io.save(result, CASE_DIR)
+    print("saved", result.shape, result.dtype)
