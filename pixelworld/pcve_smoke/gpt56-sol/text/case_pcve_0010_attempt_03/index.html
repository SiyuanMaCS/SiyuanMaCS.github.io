+"""Generate the 10x-mass counterfactual by retiming the source simulation.
+
+Mass is changed at t=0.  In this force-driven rigid-body scene, the appropriate
+counterfactual is the same early physical trajectory run at one tenth speed
+(a=F/m).  We synthesize sub-frame source times with motion-compensated RGB
+interpolation rather than dissolving adjacent frames, preserving the block's
+texture and sharp silhouette.
+"""
+from pathlib import Path
+import cv2
+import numpy as np
+import artifact_io
+
+FPS = 24
+N = 96
+MASS_FACTOR = 10.0
+

+def optical_interpolate(a, b, u):
+    """Bidirectional, occlusion-aware interpolation of two RGB source frames."""
+    if u <= 1e-8:
+        return a.copy()
+    if u >= 1 - 1e-8:
+        return b.copy()
+    ga = cv2.cvtColor(a, cv2.COLOR_RGB2GRAY)
+    gb = cv2.cvtColor(b, cv2.COLOR_RGB2GRAY)
+    kw = dict(pyr_scale=.5, levels=3, winsize=19, iterations=4,
+              poly_n=5, poly_sigma=1.2, flags=0)
+    fab = cv2.calcOpticalFlowFarneback(ga, gb, None, **kw)
+    fba = cv2.calcOpticalFlowFarneback(gb, ga, None, **kw)
+    h, w = ga.shape
+    x, y = np.meshgrid(np.arange(w, dtype=np.float32),
+                       np.arange(h, dtype=np.float32))
+
+    # Backward sampling approximations for the intermediate instant.
+    wa = cv2.remap(a, x - u * fab[..., 0], y - u * fab[..., 1],
+                   cv2.INTER_LINEAR, borderMode=cv2.BORDER_REFLECT101)
+    wb = cv2.remap(b, x - (1-u) * fba[..., 0], y - (1-u) * fba[..., 1],
+                   cv2.INTER_LINEAR, borderMode=cv2.BORDER_REFLECT101)
+
+    # Forward/backward consistency rejects the opposite side of disocclusions.
+    bx = cv2.remap(fba[..., 0], x + fab[..., 0], y + fab[..., 1],
+                   cv2.INTER_LINEAR, borderMode=cv2.BORDER_REPLICATE)
+    by = cv2.remap(fba[..., 1], x + fab[..., 0], y + fab[..., 1],
+                   cv2.INTER_LINEAR, borderMode=cv2.BORDER_REPLICATE)
+    ex = np.hypot(fab[..., 0] + bx, fab[..., 1] + by)
+    ax = cv2.remap(fab[..., 0], x + fba[..., 0], y + fba[..., 1],
+                   cv2.INTER_LINEAR, borderMode=cv2.BORDER_REPLICATE)
+    ay = cv2.remap(fab[..., 1], x + fba[..., 0], y + fba[..., 1],
+                   cv2.INTER_LINEAR, borderMode=cv2.BORDER_REPLICATE)
+    eb = np.hypot(fba[..., 0] + ax, fba[..., 1] + ay)
+    # Confidence softly favors the source that is not occluded.
+    ca = np.exp(-0.28 * ex)[..., None]
+    cb = np.exp(-0.28 * eb)[..., None]
+    w0 = (1-u) * ca + 1e-4
+    w1 = u * cb + 1e-4
+    out = (wa.astype(np.float32)*w0 + wb.astype(np.float32)*w1)/(w0+w1)
+    return np.clip(np.rint(out), 0, 255).astype(np.uint8)
+
+
+def generate():
+    src = np.load('source_pixels.npz')['rgb']
+    assert src.shape == (N, 144, 256, 3) and src.dtype == np.uint8
+    out = np.empty_like(src)
+    # Delta-v accumulated under a fixed applied force is divided by ten.
+    # For the observed, nearly uniform force-driven motion this maps output
+    # frame k to source frame k/10, while retaining the exact intervention frame.
+    for k in range(N):
+        t = k / MASS_FACTOR
+        lo = int(np.floor(t))
+        u = t - lo
+        out[k] = optical_interpolate(src[lo], src[min(lo+1, N-1)], u)
+    out[0] = src[0]  # preserve the factual image exactly at intervention time
+    return out
+
+
+if __name__ == '__main__':
+    frames = generate()
+    artifact_io.save(frames, Path(__file__).resolve().parent)
+    print('saved', frames.shape, frames.dtype)

\"\"\"Generate the 10x-mass counterfactual by retiming the source simulation.

Mass is changed at t=0.  In this force-driven rigid-body scene, the appropriate
counterfactual is the same early physical trajectory run at one tenth speed
(a=F/m).  We synthesize sub-frame source times with motion-compensated RGB
interpolation rather than dissolving adjacent frames, preserving the block's
texture and sharp silhouette.
\"\"\"
from pathlib import Path
import cv2
import numpy as np
import artifact_io

FPS = 24
N = 96
MASS_FACTOR = 10.0


def optical_interpolate(a, b, u):
    \"\"\"Bidirectional, occlusion-aware interpolation of two RGB source frames.\"\"\"
    if u <= 1e-8:
        return a.copy()
    if u >= 1 - 1e-8:
        return b.copy()
    ga = cv2.cvtColor(a, cv2.COLOR_RGB2GRAY)
    gb = cv2.cvtColor(b, cv2.COLOR_RGB2GRAY)
    kw = dict(pyr_scale=.5, levels=3, winsize=19, iterations=4,
              poly_n=5, poly_sigma=1.2, flags=0)
    fab = cv2.calcOpticalFlowFarneback(ga, gb, None, **kw)
    fba = cv2.calcOpticalFlowFarneback(gb, ga, None, **kw)
    h, w = ga.shape
    x, y = np.meshgrid(np.arange(w, dtype=np.float32),
                       np.arange(h, dtype=np.float32))

    # Backward sampling approximations for the intermediate instant.
    wa = cv2.remap(a, x - u * fab[..., 0], y - u * fab[..., 1],
                   cv2.INTER_LINEAR, borderMode=cv2.BORDER_REFLECT101)
    wb = cv2.remap(b, x - (1-u) * fba[..., 0], y - (1-u) * fba[..., 1],
                   cv2.INTER_LINEAR, borderMode=cv2.BORDER_REFLECT101)

    # Forward/backward consistency rejects the opposite side of disocclusions.
    bx = cv2.remap(fba[..., 0], x + fab[..., 0], y + fab[..., 1],
                   cv2.INTER_LINEAR, borderMode=cv2.BORDER_REPLICATE)
    by = cv2.remap(fba[..., 1], x + fab[..., 0], y + fab[..., 1],
                   cv2.INTER_LINEAR, borderMode=cv2.BORDER_REPLICATE)
    ex = np.hypot(fab[..., 0] + bx, fab[..., 1] + by)
    ax = cv2.remap(fab[..., 0], x + fba[..., 0], y + fba[..., 1],
                   cv2.INTER_LINEAR, borderMode=cv2.BORDER_REPLICATE)
    ay = cv2.remap(fab[..., 1], x + fba[..., 0], y + fba[..., 1],
                   cv2.INTER_LINEAR, borderMode=cv2.BORDER_REPLICATE)
    eb = np.hypot(fba[..., 0] + ax, fba[..., 1] + ay)
    # Confidence softly favors the source that is not occluded.
    ca = np.exp(-0.28 * ex)[..., None]
    cb = np.exp(-0.28 * eb)[..., None]
    w0 = (1-u) * ca + 1e-4
    w1 = u * cb + 1e-4
    out = (wa.astype(np.float32)*w0 + wb.astype(np.float32)*w1)/(w0+w1)
    return np.clip(np.rint(out), 0, 255).astype(np.uint8)


def generate():
    src = np.load('source_pixels.npz')['rgb']
    assert src.shape == (N, 144, 256, 3) and src.dtype == np.uint8
    out = np.empty_like(src)
    # Delta-v accumulated under a fixed applied force is divided by ten.
    # For the observed, nearly uniform force-driven motion this maps output
    # frame k to source frame k/10, while retaining the exact intervention frame.
    for k in range(N):
        t = k / MASS_FACTOR
        lo = int(np.floor(t))
        u = t - lo
        out[k] = optical_interpolate(src[lo], src[min(lo+1, N-1)], u)
    out[0] = src[0]  # preserve the factual image exactly at intervention time
    return out


if __name__ == '__main__':
    frames = generate()
    artifact_io.save(frames, Path(__file__).resolve().parent)
    print('saved', frames.shape, frames.dtype)

\"\"\"Generate the 10x-mass rigid-body counterfactual from source RGB frames.

The wooden block is initially at rest, so the incoming object's free flight is
unchanged.  At contact (source frame 12, zero based), a ten-times-heavier block
acquires about one tenth the original speed, while the incoming body rebounds.
Source pixels are motion-interpolated and composited; no external renderer is
used.

from pathlib import Path
import cv2
import numpy as np
import artifact_io

N, FPS = 96, 24
CONTACT = 12                 # last factual/contact frame, zero based
MASS_FACTOR = 10.0
REBOUND_SOURCE_FRAMES = 0.82 # rebound speed / incident speed for ~10:1 impact


def flow_interpolate(a, b, u):
    \"\"\"Bidirectional motion-compensated RGB interpolation.\"\"\"
    if u <= 1e-7:
        return a.copy()
    if u >= 1-1e-7:
        return b.copy()
    ga = cv2.cvtColor(a, cv2.COLOR_RGB2GRAY)
    gb = cv2.cvtColor(b, cv2.COLOR_RGB2GRAY)
    kw = dict(pyr_scale=.5, levels=3, winsize=19, iterations=4,
              poly_n=5, poly_sigma=1.2, flags=0)
    fab = cv2.calcOpticalFlowFarneback(ga, gb, None, **kw)
    fba = cv2.calcOpticalFlowFarneback(gb, ga, None, **kw)
    h, w = ga.shape
    x, y = np.meshgrid(np.arange(w, dtype=np.float32), np.arange(h, dtype=np.float32))
    wa = cv2.remap(a, x-u*fab[...,0], y-u*fab[...,1], cv2.INTER_LINEAR,
                   borderMode=cv2.BORDER_REFLECT101)
    wb = cv2.remap(b, x-(1-u)*fba[...,0], y-(1-u)*fba[...,1], cv2.INTER_LINEAR,
                   borderMode=cv2.BORDER_REFLECT101)
    return np.clip(np.rint(wa*(1-u) + wb*u), 0, 255).astype(np.uint8)


def sample(src, t):
    t = float(np.clip(t, 0, len(src)-1))
    lo = int(np.floor(t))
    return flow_interpolate(src[lo], src[min(lo+1, len(src)-1)], t-lo)


def foreground_alpha(image, clean, source_t):
    \"\"\"Soft alpha for the incoming rounded body, constrained near its trajectory.\"\"\"
    # Its observed center advances about 6.7 pixels per source frame.
    cx = 18.0 + 6.65*source_t
    yy, xx = np.ogrid[:144, :256]
    roi = ((xx-cx)/34.0)**2 + ((yy-(66+1.45*source_t))/34.0)**2 < 1.0
    difference = np.max(np.abs(image.astype(np.int16)-clean.astype(np.int16)), axis=2)
    raw = ((difference > 6) & roi & (xx < 110)).astype(np.uint8)
    raw = cv2.morphologyEx(raw, cv2.MORPH_CLOSE, np.ones((5,5),np.uint8), iterations=2)
    raw = cv2.morphologyEx(raw, cv2.MORPH_OPEN, np.ones((3,3),np.uint8))
    # Keep only the largest connected foreground component.
    n, lab, stats, _ = cv2.connectedComponentsWithStats(raw, 8)
    if n > 1:
        best = 1 + int(np.argmax(stats[1:, cv2.CC_STAT_AREA]))
        raw = (lab == best).astype(np.uint8)
    # Include antialiased silhouette pixels and make a narrow feathered edge.
    raw = cv2.dilate(raw, np.ones((3,3),np.uint8), iterations=1)
    alpha = cv2.GaussianBlur(raw.astype(np.float32), (5,5), .8)
    return np.clip(alpha, 0, 1)[...,None]


def generate():
    src = np.load('source_pixels.npz')['rgb']
    assert src.shape == (N,144,256,3) and src.dtype == np.uint8
    out = np.empty_like(src)
    out[:CONTACT+1] = src[:CONTACT+1]

    # Once the incoming body has passed, this region exposes the unchanging
    # scene; a temporal median gives a clean plate without hallucinated pixels.
    clean = np.median(src[20:96], axis=0).astype(np.uint8)

    for k in range(CONTACT+1, N):
        elapsed = k-CONTACT
        # The heavy block and anything it later contacts evolve at one tenth
        # the original post-impact rate.
        base = sample(src, CONTACT + elapsed/MASS_FACTOR)

        # Remove the original light-mass incoming-body trajectory on the left.
        # The blend ends just before the wooden block's left edge (x ~= 108).
        xweight = np.ones((144,256,1), np.float32)
        xweight[:,104:110,0] = np.linspace(1,0,6)[None,:]
        xweight[:,110:,0] = 0
        base = np.clip(np.rint(base*(1-xweight)+clean*xweight),0,255).astype(np.uint8)

        # Elastic collision with a target near ten times heavier reverses the
        # incident body's velocity to roughly 9/11 of its approach speed.
        t_rebound = CONTACT - REBOUND_SOURCE_FRAMES*elapsed
        if t_rebound >= 0:
            body = sample(src, t_rebound)
            alpha = foreground_alpha(body, clean, t_rebound)
            base = np.clip(np.rint(base*(1-alpha)+body*alpha),0,255).astype(np.uint8)
        out[k] = base

    out[0] = src[0]
    return out


if __name__ == '__main__':
    frames = generate()
    artifact_io.save(frames, Path(__file__).resolve().parent)
    print('saved revised candidate', frames.shape, frames.dtype)
