+"""Generate the 10x-ball-mass counterfactual for this PixelWorld case.
+
+The camera is fixed.  Mass does not alter the ball's pre-contact ballistic/rolling
+path, so frames through first contact are preserved.  After contact we advance
+the source collision response more quickly (the heavy ball transfers momentum
+through the whole cluster sooner), then continue the ball along the floor with
+substantial retained speed.  All imagery comes from source RGB pixels.
+"""
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+CASE_DIR = Path(__file__).resolve().parent
+N, H, W = 96, 144, 256
+

+def largest_center_component(mask: np.ndarray, cx: float, cy: float) -> np.ndarray:
+    """Return the connected foreground component containing/nearest (cx, cy)."""
+    count, labels, stats, centers = cv2.connectedComponentsWithStats(mask, 8)
+    if count <= 1:
+        return mask
+    choices = []
+    for i in range(1, count):
+        area = stats[i, cv2.CC_STAT_AREA]
+        if area < 25:
+            continue
+        dist = (centers[i, 0] - cx) ** 2 + (centers[i, 1] - cy) ** 2
+        choices.append((dist, -area, i))
+    if not choices:
+        return mask
+    index = min(choices)[2]
+    return (labels == index).astype(np.uint8)
+
+
+def make_ball_sprites(source: np.ndarray, plate: np.ndarray):
+    # Ball centers measured from its unoccluded source motion.  The changing
+    # source textures provide natural rolling orientations without drawing.
+    centers = [(18.0, 64.0), (24.8, 65.0), (31.8, 66.3), (39.0, 67.9),
+               (46.2, 69.5), (53.5, 70.5), (60.5, 72.2), (67.7, 74.5)]
+    sprites = []
+    for t, (cx, cy) in enumerate(centers):
+        diff = np.max(np.abs(source[t].astype(np.int16) - plate.astype(np.int16)), axis=2)
+        raw = (diff > 5).astype(np.uint8)
+        component = largest_center_component(raw, cx, cy)
+        component = cv2.morphologyEx(component, cv2.MORPH_CLOSE,
+                                     np.ones((5, 5), np.uint8))
+        component = cv2.dilate(component, np.ones((3, 3), np.uint8), iterations=1)
+
+        # Keep the moving sphere and its immediately attached contact shadow,
+        # while excluding unrelated distant texture differences.
+        yy, xx = np.ogrid[:H, :W]
+        support = (((xx - cx) / 27.0) ** 2 + ((yy - cy) / 28.0) ** 2 <= 1.0)
+        component = (component.astype(bool) & support).astype(np.uint8)
+        alpha = cv2.GaussianBlur(component.astype(np.float32), (5, 5), 0.8)
+        alpha = np.clip(alpha, 0.0, 1.0)
+
+        x0 = max(0, int(np.floor(cx - 28)))
+        x1 = min(W, int(np.ceil(cx + 29)))
+        y0 = max(0, int(np.floor(cy - 29)))
+        y1 = min(H, int(np.ceil(cy + 30)))
+        sprites.append((source[t, y0:y1, x0:x1].copy(),
+                        alpha[y0:y1, x0:x1].copy(), cx - x0, cy - y0))
+    return sprites
+
+
+def composite_sprite(frame: np.ndarray, sprite, center_x: float, center_y: float) -> np.ndarray:
+    rgb, alpha, local_cx, local_cy = sprite
+    sh, sw = alpha.shape
+    left = int(round(center_x - local_cx))
+    top = int(round(center_y - local_cy))
+    x0, y0 = max(0, left), max(0, top)
+    x1, y1 = min(W, left + sw), min(H, top + sh)
+    if x1 <= x0 or y1 <= y0:
+        return frame
+    sx0, sy0 = x0 - left, y0 - top
+    sx1, sy1 = sx0 + (x1 - x0), sy0 + (y1 - y0)
+    a = alpha[sy0:sy1, sx0:sx1, None]
+    dst = frame[y0:y1, x0:x1].astype(np.float32)
+    src = rgb[sy0:sy1, sx0:sx1].astype(np.float32)
+    frame[y0:y1, x0:x1] = np.clip(src * a + dst * (1.0 - a), 0, 255).astype(np.uint8)
+    return frame
+
+
+def make_clean_late_plate(source: np.ndarray) -> np.ndarray:
+    """Recover scenery behind the final, slowly moving ball from earlier late frames."""
+    final = source[95].copy()
+    # By frame 36 the collision debris outside the ball's path has settled.
+    # Its ball is around x=119, leaving the right portion of the final ball site
+    # visible.  Copy that genuine source texture, then inpaint only the narrow
+    # overlap strip where both old and final ball silhouettes occurred.
+    cx, cy, radius = 157.5, 93.0, 25.0
+    yy, xx = np.ogrid[:H, :W]

+    visible_from_36 = final_ball & (xx >= 142)
+    final[visible_from_36] = source[36][visible_from_36]
+
+    overlap = (final_ball & (xx < 142)).astype(np.uint8) * 255
+    # Only a slim left crescent remains; Telea propagates adjacent genuine floor
+    # texture into it and avoids a blurred circular patch.
+    overlap = cv2.dilate(overlap, np.ones((3, 3), np.uint8), iterations=1)
+    final = cv2.inpaint(cv2.cvtColor(final, cv2.COLOR_RGB2BGR), overlap, 3,
+                        cv2.INPAINT_TELEA)
+    return cv2.cvtColor(final, cv2.COLOR_BGR2RGB)
+

+def generate() -> np.ndarray:
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (N, H, W, 3) and source.dtype == np.uint8
+
+    # A temporal median is an authentic source-derived scenery plate for sprite
+    # segmentation; the ball occupies each pre-impact pixel for only a few frames.
+    plate = np.median(source, axis=0).astype(np.uint8)
+    sprites = make_ball_sprites(source, plate)
+    clean_late = make_clean_late_plate(source)
+
+    result = np.empty_like(source)
+    contact_frame = 12
+    # Up through initial contact the trajectory is independent of mass.
+    result[:contact_frame + 1] = source[:contact_frame + 1]
+
+    # A 10x ball retains much more momentum.  Advance the *complete* observed
+    # response (including all struck objects) at 3.2x after contact.  Nearest
+    # source frames preserve crisp source pixels rather than crossfade ghosts.
+    rate = 3.2
+    last_warped_out = contact_frame
+    for t in range(contact_frame + 1, N):
+        source_t = int(np.floor(contact_frame + rate * (t - contact_frame)))
+        if source_t <= 95:
+            result[t] = source[source_t]
+            last_warped_out = t
+        else:
+            break
+
+    # Continue the heavy ball beyond the factual slow-down.  Its path follows
+    # the measured floor slope.  The affected scene remains in its fully knocked-
+    # through source state; after the ball exits, it naturally stays at rest.
+    start_x, start_y = 157.5, 93.0
+    speed = 3.8  # pixels/frame after traversing the collision cluster
+    for t in range(last_warped_out + 1, N):
+        frame = clean_late.copy()
+        dt = t - last_warped_out
+        x = start_x + speed * dt
+        y = start_y + 0.30 * (x - start_x)
+        if x < W + 30 and y < H + 30:
+            # Cycle authentic source orientations at a rate proportional to travel.
+            sprite_index = int(np.floor((x - start_x) / 7.0)) % len(sprites)
+            frame = composite_sprite(frame, sprites[sprite_index], x, y)
+        result[t] = frame
+
+    assert result.shape == (N, H, W, 3)
+    assert result.dtype == np.uint8
+    return result
+
+
+if __name__ == "__main__":
+    frames = generate()
+    artifact_io.save(frames, CASE_DIR)
+    print(f"saved {len(frames)} RGB frames to {CASE_DIR}")

    # Ball centers measured from its unoccluded source motion.  The changing
    # source textures provide natural rolling orientations without drawing.
    centers = [(18.0, 64.0), (24.8, 65.0), (31.8, 66.3), (39.0, 67.9),
               (46.2, 69.5), (53.5, 70.5), (60.5, 72.2), (67.7, 74.5)]
    sprites = []
    for t, (cx, cy) in enumerate(centers):
        diff = np.max(np.abs(source[t].astype(np.int16) - plate.astype(np.int16)), axis=2)
        raw = (diff > 5).astype(np.uint8)
        component = largest_center_component(raw, cx, cy)
        component = cv2.morphologyEx(component, cv2.MORPH_CLOSE,
                                     np.ones((5, 5), np.uint8))
        component = cv2.dilate(component, np.ones((3, 3), np.uint8), iterations=1)

        # Keep the moving sphere and its immediately attached contact shadow,
        # while excluding unrelated distant texture differences.
        yy, xx = np.ogrid[:H, :W]
        support = (((xx - cx) / 27.0) ** 2 + ((yy - cy) / 28.0) ** 2 <= 1.0)
        component = (component.astype(bool) & support).astype(np.uint8)
        alpha = cv2.GaussianBlur(component.astype(np.float32), (5, 5), 0.8)
        alpha = np.clip(alpha, 0.0, 1.0)

        x0 = max(0, int(np.floor(cx - 28)))
        x1 = min(W, int(np.ceil(cx + 29)))
        y0 = max(0, int(np.floor(cy - 29)))
        y1 = min(H, int(np.ceil(cy + 30)))
        sprites.append((source[t, y0:y1, x0:x1].copy(),
                        alpha[y0:y1, x0:x1].copy(), cx - x0, cy - y0))
    return sprites


def composite_sprite(frame: np.ndarray, sprite, center_x: float, center_y: float) -> np.ndarray:
    rgb, alpha, local_cx, local_cy = sprite
    sh, sw = alpha.shape
    left = int(round(center_x - local_cx))
    top = int(round(center_y - local_cy))
    x0, y0 = max(0, left), max(0, top)
    x1, y1 = min(W, left + sw), min(H, top + sh)
    if x1 <= x0 or y1 <= y0:
        return frame
    sx0, sy0 = x0 - left, y0 - top
    sx1, sy1 = sx0 + (x1 - x0), sy0 + (y1 - y0)
    a = alpha[sy0:sy1, sx0:sx1, None]
    dst = frame[y0:y1, x0:x1].astype(np.float32)
    src = rgb[sy0:sy1, sx0:sx1].astype(np.float32)
    frame[y0:y1, x0:x1] = np.clip(src * a + dst * (1.0 - a), 0, 255).astype(np.uint8)
    return frame


def make_clean_late_plate(source: np.ndarray) -> np.ndarray:
    \"\"\"Recover scenery behind the final, slowly moving ball from earlier late frames.\"\"\"
    final = source[95].copy()
    # By frame 36 the collision debris outside the ball's path has settled.
    # Its ball is around x=119, leaving the right portion of the final ball site
    # visible.  Copy that genuine source texture, then inpaint only the narrow
    # overlap strip where both old and final ball silhouettes occurred.
    cx, cy, radius = 157.5, 93.0, 25.0
    yy, xx = np.ogrid[:H, :W]
    final_ball = ((xx - cx) ** 2 + (yy - cy) ** 2 <= radius ** 2)
    visible_from_36 = final_ball & (xx >= 142)
    final[visible_from_36] = source[36][visible_from_36]

    overlap = (final_ball & (xx < 142)).astype(np.uint8) * 255
    # Only a slim left crescent remains; Telea propagates adjacent genuine floor
    # texture into it and avoids a blurred circular patch.
    overlap = cv2.dilate(overlap, np.ones((3, 3), np.uint8), iterations=1)
    final = cv2.inpaint(cv2.cvtColor(final, cv2.COLOR_RGB2BGR), overlap, 3,
                        cv2.INPAINT_TELEA)
    return cv2.cvtColor(final, cv2.COLOR_BGR2RGB)


def generate() -> np.ndarray:
    source = np.load(CASE_DIR / \"source_pixels.npz\")[\"rgb\"]
    assert source.shape == (N, H, W, 3) and source.dtype == np.uint8

    # A temporal median is an authentic source-derived scenery plate for sprite
    # segmentation; the ball occupies each pre-impact pixel for only a few frames.
    plate = np.median(source, axis=0).astype(np.uint8)
    sprites = make_ball_sprites(source, plate)
    clean_late = make_clean_late_plate(source)

    result = np.empty_like(source)
    contact_frame = 12
    # Up through initial contact the trajectory is independent of mass.
    result[:contact_frame + 1] = source[:contact_frame + 1]

    # A 10x ball retains much more momentum.  Advance the *complete* observed
    # response (including all struck objects) at 3.2x after contact.  Nearest
    # source frames preserve crisp source pixels rather than crossfade ghosts.
    rate = 3.2
    last_warped_out = contact_frame
    for t in range(contact_frame + 1, N):
        source_t = int(np.floor(contact_frame + rate * (t - contact_frame)))
        if source_t <= 95:
            result[t] = source[source_t]
            last_warped_out = t
        else:
            break

    # Continue the heavy ball beyond the factual slow-down.  Its path follows
    # the measured floor slope.  The affected scene remains in its fully knocked-
    # through source state; after the ball exits, it naturally stays at rest.
    start_x, start_y = 157.5, 93.0
    speed = 3.8  # pixels/frame after traversing the collision cluster
    for t in range(last_warped_out + 1, N):
        frame = clean_late.copy()
        dt = t - last_warped_out
        x = start_x + speed * dt
        y = start_y + 0.30 * (x - start_x)
        if x < W + 30 and y < H + 30:
            # Cycle authentic source orientations at a rate proportional to travel.
            sprite_index = int(np.floor((x - start_x) / 7.0)) % len(sprites)
            frame = composite_sprite(frame, sprites[sprite_index], x, y)
        result[t] = frame

    assert result.shape == (N, H, W, 3)
    assert result.dtype == np.uint8
    return result


if __name__ == \"__main__\":
