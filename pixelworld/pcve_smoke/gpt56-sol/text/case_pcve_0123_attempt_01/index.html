+"""Generate the 0.3-mass purple-marble counterfactual from supplied pixels.
+
+The scene contains equal-size orange and purple marbles rolling toward the center
+of a shallow track.  Their approach is mass independent and is retained exactly.
+At the first contact, momentum/restitution are recomputed with m_purple=0.3.
+The altered post-impact sprites are composited over the source's robust static
+background, preserving the original shading, shadows, rails and camera.
+"""
+from pathlib import Path
+import numpy as np
+from scipy import ndimage
+from PIL import Image
+import artifact_io
+
+CASE = Path(__file__).resolve().parent
+SRC = np.load(CASE / "source_pixels.npz")["rgb"]
+N,H,W,_ = SRC.shape
+
+
+def center_tracks():
+    """Color-based subpixel tracking, restricted to the moving lane."""
+    purple=[]; orange=[]; po=228.; oo=27.
+    yy=np.indices((H,W))[0]
+    for frame in SRC:
+        r,g,b=frame.astype(float).transpose(2,0,1)
+        masks=[(r>70)&(b>70)&(b>1.12*g)&(r>1.05*g)&(np.maximum(r,b)-g>18)&(yy<100),
+               (r>125)&(r>g+18)&(g>b+18)&(yy>45)&(yy<90)]
+        vals=[]
+        for mask,old in zip(masks,(po,oo)):
+            lab,num=ndimage.label(mask); candidates=[]
+            for k in range(1,num+1):
+                y,x=np.where(lab==k)
+                if len(x)>=30: candidates.append((abs(x.mean()-old),x.mean(),y.mean()))
+            _,x,y=min(candidates)
+            vals.append((x,y))
+        po,oo=vals[0][0],vals[1][0]
+        purple.append(vals[0]); orange.append(vals[1])
+    return np.array(purple),np.array(orange)
+
+
+def alpha_sprite(frame, cx, cy, background):
+    """Extract a softly antialiased 19x19 sprite (marble plus cast shadow)."""
+    x0=int(round(cx))-9; y0=int(round(cy))-8
+    patch=frame[y0:y0+19,x0:x0+19].astype(float)
+    base=background[y0:y0+19,x0:x0+19].astype(float)
+    diff=np.max(np.abs(patch-base),axis=2)
+    # Local source/background difference is strongest on the body, weaker on shadow.
+    a=np.clip((diff-1.8)/13.0,0,1)
+    body=(np.indices(a.shape)[1]-(cx-x0))**2+(np.indices(a.shape)[0]-(cy-y0))**2 <= 7.3**2
+    a=np.maximum(a, body.astype(float)*np.clip((diff-0.8)/4.0,0,1))
+    a=ndimage.gaussian_filter(a,0.45)
+    return patch,a,(cx-x0,cy-y0)
+
+
+def paste(canvas, sprite, alpha, anchor, cx, cy):
+    """Subpixel translate a premultiplied sprite into canvas."""
+    ph,pw=alpha.shape
+    x0=int(np.floor(cx-anchor[0])); y0=int(np.floor(cy-anchor[1]))
+    fx=(cx-anchor[0])-x0; fy=(cy-anchor[1])-y0
+    # shift arrays rather than resampling the final image: smooth original texture.
+    sp=ndimage.shift(sprite,(fy,fx,0),order=1,mode='nearest',prefilter=False)
+    aa=ndimage.shift(alpha,(fy,fx),order=1,mode='constant',cval=0,prefilter=False)
+    xa=max(0,x0); ya=max(0,y0); xb=min(W,x0+pw); yb=min(H,y0+ph)
+    if xa>=xb or ya>=yb:return
+    sx=xa-x0; sy=ya-y0
+    a=aa[sy:sy+yb-ya,sx:sx+xb-xa,None]
+    canvas[ya:yb,xa:xb]=canvas[ya:yb,xa:xb]*(1-a)+sp[sy:sy+yb-ya,sx:sx+xb-xa]*a
+
+
+def simulate(contact, orange_xy, purple_xy):
+    """Discrete shallow-track dynamics after an unequal-mass oblique-free impact."""
+    # Empirical incoming center velocities (px/frame), robustly fit before contact.
+    vo=np.polyfit(np.arange(contact-5,contact),orange_xy[contact-5:contact,0],1)[0]
+    vp=np.polyfit(np.arange(contact-5,contact),purple_xy[contact-5:contact,0],1)[0]
+    mo,mp=1.0,0.3
+    # Source collision gives e ~= .65 (and this also captures rolling losses at impact).
+    e=.65
+    uo=(mo*vo+mp*vp-mp*e*(vo-vp))/(mo+mp)
+    up=(mo*vo+mp*vp+mo*e*(vo-vp))/(mo+mp)
+    xo=float(orange_xy[contact,0]); xp=float(purple_xy[contact,0])
+    yo=float(orange_xy[contact,1]); yp=float(purple_xy[contact,1])
+    ox=[]; px=[]
+    # Shallow V acceleration seen on factual approach; mild rolling drag.
+    g=.23; drag=.06; trough=127.5; diameter=15.0
+    for f in range(contact,N):
+        ox.append((xo,yo)); px.append((xp,yp))
+        # ten substeps avoid integration/contact jitter
+        for _ in range(10):
+            dt=.1
+            for name in (0,1):
+                x,v=(xo,uo) if name==0 else (xp,up)
+                a=(g if x<trough else -g)-drag*v
+                v+=a*dt; x+=v*dt
+                if name==0: xo,uo=x,v
+                else: xp,up=x,v
+            # Subsequent unequal-mass collisions remain physically consistent.
+            if xp-xo<diameter and up-uo<0:
+                overlap=diameter-(xp-xo)
+                xo-=overlap*mp/(mo+mp); xp+=overlap*mo/(mo+mp)
+                a,b=uo,up
+                uo=(mo*a+mp*b-mp*e*(a-b))/(mo+mp)
+                up=(mo*a+mp*b+mo*e*(a-b))/(mo+mp)
+    return np.array(ox),np.array(px)
+
+
+def generate():
+    purple,orange=center_tracks()
+    # Minimum separation / velocity reversal identifies the first physical contact.
+    contact=int(np.argmin(np.abs((purple[:,0]-orange[:,0])-15.0)[:40]))
+    # Per-pixel temporal median removes moving marbles while retaining the supplied scene.
+    bg=np.median(SRC,axis=0).astype(np.uint8)
+    os,oa,oanch=alpha_sprite(SRC[24],orange[24,0],orange[24,1],bg)
+    ps,pa,panch=alpha_sprite(SRC[24],purple[24,0],purple[24,1],bg)
+    opath,ppath=simulate(contact,orange,purple)
+    out=SRC.copy()
+    for f in range(contact,N):
+        canvas=bg.astype(float)
+        # Preserve tiny source-wide illumination/compression variation, excluding objects.
+        residual=np.median(SRC[f].astype(float)-bg.astype(float),axis=(0,1))
+        canvas=np.clip(canvas+residual,0,255)
+        paste(canvas,os,oa,oanch,*opath[f-contact])
+        paste(canvas,ps,pa,panch,*ppath[f-contact])
+        out[f]=np.clip(np.rint(canvas),0,255).astype(np.uint8)
+    # Before intervention consequences arise, history is byte-identical to source.
+    return out,contact,opath,ppath
+
+
+if __name__ == '__main__':
+    frames,contact,op,pp=generate()
+    artifact_io.save(frames,CASE)
+    print(f'saved {frames.shape}; first collision frame={contact+1}')
+    print('final orange/purple centers:',op[-1].round(2),pp[-1].round(2))

\"\"\"Generate the 0.3-mass purple-marble counterfactual from supplied pixels.

The scene contains equal-size orange and purple marbles rolling toward the center
of a shallow track.  Their approach is mass independent and is retained exactly.
At the first contact, momentum/restitution are recomputed with m_purple=0.3.
The altered post-impact sprites are composited over the source's robust static
background, preserving the original shading, shadows, rails and camera.

from pathlib import Path
import numpy as np
from scipy import ndimage
from PIL import Image
import artifact_io

CASE = Path(__file__).resolve().parent
SRC = np.load(CASE / \"source_pixels.npz\")[\"rgb\"]
N,H,W,_ = SRC.shape


def center_tracks():
    \"\"\"Color-based subpixel tracking, restricted to the moving lane.\"\"\"
    purple=[]; orange=[]; po=228.; oo=27.
    yy=np.indices((H,W))[0]
    for frame in SRC:
        r,g,b=frame.astype(float).transpose(2,0,1)
        masks=[(r>70)&(b>70)&(b>1.12*g)&(r>1.05*g)&(np.maximum(r,b)-g>18)&(yy<100),
               (r>125)&(r>g+18)&(g>b+18)&(yy>45)&(yy<90)]
        vals=[]
        for mask,old in zip(masks,(po,oo)):
            lab,num=ndimage.label(mask); candidates=[]
            for k in range(1,num+1):
                y,x=np.where(lab==k)
                if len(x)>=30: candidates.append((abs(x.mean()-old),x.mean(),y.mean()))
            _,x,y=min(candidates)
            vals.append((x,y))
        po,oo=vals[0][0],vals[1][0]
        purple.append(vals[0]); orange.append(vals[1])
    return np.array(purple),np.array(orange)


def alpha_sprite(frame, cx, cy, background):
    \"\"\"Extract a softly antialiased 19x19 sprite (marble plus cast shadow).\"\"\"
    x0=int(round(cx))-9; y0=int(round(cy))-8
    patch=frame[y0:y0+19,x0:x0+19].astype(float)
    base=background[y0:y0+19,x0:x0+19].astype(float)
    diff=np.max(np.abs(patch-base),axis=2)
    # Local source/background difference is strongest on the body, weaker on shadow.
    a=np.clip((diff-1.8)/13.0,0,1)
    body=(np.indices(a.shape)[1]-(cx-x0))**2+(np.indices(a.shape)[0]-(cy-y0))**2 <= 7.3**2
    a=np.maximum(a, body.astype(float)*np.clip((diff-0.8)/4.0,0,1))
    a=ndimage.gaussian_filter(a,0.45)
    return patch,a,(cx-x0,cy-y0)


def paste(canvas, sprite, alpha, anchor, cx, cy):
    \"\"\"Subpixel translate a premultiplied sprite into canvas.\"\"\"
    ph,pw=alpha.shape
    x0=int(np.floor(cx-anchor[0])); y0=int(np.floor(cy-anchor[1]))
    fx=(cx-anchor[0])-x0; fy=(cy-anchor[1])-y0
    # shift arrays rather than resampling the final image: smooth original texture.
    sp=ndimage.shift(sprite,(fy,fx,0),order=1,mode='nearest',prefilter=False)
    aa=ndimage.shift(alpha,(fy,fx),order=1,mode='constant',cval=0,prefilter=False)
    xa=max(0,x0); ya=max(0,y0); xb=min(W,x0+pw); yb=min(H,y0+ph)
    if xa>=xb or ya>=yb:return
    sx=xa-x0; sy=ya-y0
    a=aa[sy:sy+yb-ya,sx:sx+xb-xa,None]
    canvas[ya:yb,xa:xb]=canvas[ya:yb,xa:xb]*(1-a)+sp[sy:sy+yb-ya,sx:sx+xb-xa]*a


def simulate(contact, orange_xy, purple_xy):
    \"\"\"Discrete shallow-track dynamics after an unequal-mass oblique-free impact.\"\"\"
    # Empirical incoming center velocities (px/frame), robustly fit before contact.
    vo=np.polyfit(np.arange(contact-5,contact),orange_xy[contact-5:contact,0],1)[0]
    vp=np.polyfit(np.arange(contact-5,contact),purple_xy[contact-5:contact,0],1)[0]
    mo,mp=1.0,0.3
    # Source collision gives e ~= .65 (and this also captures rolling losses at impact).
    e=.65
    uo=(mo*vo+mp*vp-mp*e*(vo-vp))/(mo+mp)
    up=(mo*vo+mp*vp+mo*e*(vo-vp))/(mo+mp)
    xo=float(orange_xy[contact,0]); xp=float(purple_xy[contact,0])
    yo=float(orange_xy[contact,1]); yp=float(purple_xy[contact,1])
    ox=[]; px=[]
    # Shallow V acceleration seen on factual approach; mild rolling drag.
    g=.23; drag=.06; trough=127.5; diameter=15.0
    for f in range(contact,N):
        ox.append((xo,yo)); px.append((xp,yp))
        # ten substeps avoid integration/contact jitter
        for _ in range(10):
            dt=.1
            for name in (0,1):
                x,v=(xo,uo) if name==0 else (xp,up)
                a=(g if x<trough else -g)-drag*v
                v+=a*dt; x+=v*dt
                if name==0: xo,uo=x,v
                else: xp,up=x,v
            # Subsequent unequal-mass collisions remain physically consistent.
            if xp-xo<diameter and up-uo<0:

                xo-=overlap*mp/(mo+mp); xp+=overlap*mo/(mo+mp)
                a,b=uo,up
                uo=(mo*a+mp*b-mp*e*(a-b))/(mo+mp)
                up=(mo*a+mp*b+mo*e*(a-b))/(mo+mp)
    return np.array(ox),np.array(px)


def generate():

    # Minimum separation / velocity reversal identifies the first physical contact.
    contact=int(np.argmin(np.abs((purple[:,0]-orange[:,0])-15.0)[:40]))
    # Per-pixel temporal median removes moving marbles while retaining the supplied scene.
    bg=np.median(SRC,axis=0).astype(np.uint8)

    ps,pa,panch=alpha_sprite(SRC[24],purple[24,0],purple[24,1],bg)

    out=SRC.copy()

        canvas=np.clip(canvas+residual,0,255)
        paste(canvas,os,oa,oanch,*opath[f-contact])
        paste(canvas,ps,pa,panch,*ppath[f-contact])
        out[f]=np.clip(np.rint(canvas),0,255).astype(np.uint8)
    # Before intervention consequences arise, history is byte-identical to source.
    return out,contact,opath,ppath
