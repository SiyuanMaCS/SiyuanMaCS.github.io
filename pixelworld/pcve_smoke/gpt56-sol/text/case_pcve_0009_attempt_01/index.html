+"""Generate the complete counterfactual video for this case.
+
+The source contains a three-mallet momentum-transfer sequence.  At frame 1 the
+moving green mallet is relocated to the midpoint of the initially stationary
+red and white mallets.  It keeps its incoming velocity, reaches the white
+mallet first, transfers its motion, and the red mallet consequently stays put.
+Only local source pixels are reused; no external visual assets are needed.
+"""
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def smoothstep(x):
+    x = np.clip(x, 0.0, 1.0)
+    return x * x * (3.0 - 2.0 * x)
+
+
+def extract_sprite(foreground, clean_reference, box, diff_lo=1.5, diff_hi=10.0):
+    """Extract an antialiased source sprite using a clean frame as reference."""
+    x0, y0, x1, y1 = box
+    src = foreground[y0:y1, x0:x1].copy()
+    ref = clean_reference[y0:y1, x0:x1]
+    difference = np.max(np.abs(src.astype(np.float32) - ref.astype(np.float32)), axis=2)
+
+    # Reject isolated codec noise, retain the coherent changed component, and
+    # retain faint antialiased/shadow pixels immediately around it.
+    strong = (difference >= 6.0).astype(np.uint8)
+    count, labels, stats, _ = cv2.connectedComponentsWithStats(strong, 8)
+    if count > 1:
+        keep_id = 1 + int(np.argmax(stats[1:, cv2.CC_STAT_AREA]))
+        coherent = (labels == keep_id).astype(np.uint8)
+    else:
+        coherent = strong
+    support = cv2.dilate(coherent, np.ones((3, 3), np.uint8), iterations=2)
+    raw = smoothstep((difference - diff_lo) / (diff_hi - diff_lo)) * support
+    alpha = cv2.GaussianBlur(raw.astype(np.float32), (3, 3), 0.55)
+    alpha[coherent.astype(bool)] = np.maximum(alpha[coherent.astype(bool)], 0.94)
+    return src, np.clip(alpha, 0.0, 1.0), (x0, y0)
+
+
+def place_sprite(canvas, sprite, alpha, source_origin, source_center, target_center):
+    """Place a sprite by translating its measured center to target_center."""
+    x0, y0 = source_origin
+    dx = int(round(float(target_center[0] - source_center[0])))
+    dy = int(round(float(target_center[1] - source_center[1])))
+    tx, ty = x0 + dx, y0 + dy
+    h, w = alpha.shape
+
+    # Clip safely (not normally needed for this scene).
+    cx0, cy0 = max(0, tx), max(0, ty)
+    cx1, cy1 = min(canvas.shape[1], tx + w), min(canvas.shape[0], ty + h)
+    if cx1 <= cx0 or cy1 <= cy0:
+        return
+    sx0, sy0 = cx0 - tx, cy0 - ty
+    sx1, sy1 = sx0 + (cx1 - cx0), sy0 + (cy1 - cy0)
+    a = alpha[sy0:sy1, sx0:sx1, None]
+    under = canvas[cy0:cy1, cx0:cx1].astype(np.float32)
+    over = sprite[sy0:sy1, sx0:sx1].astype(np.float32)
+    canvas[cy0:cy1, cx0:cx1] = np.clip(under * (1.0 - a) + over * a + 0.5, 0, 255).astype(np.uint8)
+
+
+def erase_initial_objects(frame, green_alpha, green_origin, white_alpha, white_origin, white_center_final, white_center_initial):
+    """Remove the original green and white while preserving the fixed red."""
+    clean = frame.copy()
+
+    # Green's original location has a clean observation late in the source.
+    # A feathered patch transfer preserves the patterned table much better than
+    # synthesizing it.
+    source = SOURCE[48]
+    gx0, gy0 = green_origin
+    gh, gw = green_alpha.shape
+    pad = 3
+    x0, y0 = gx0 - pad, gy0 - pad
+    x1, y1 = gx0 + gw + pad, gy0 + gh + pad
+    patch = source[y0:y1, x0:x1].astype(np.float32)
+    feather = np.zeros((y1-y0, x1-x0), np.float32)
+    feather[pad:pad+gh, pad:pad+gw] = np.maximum(green_alpha, 0.25 * (green_alpha > 0.02))
+    feather = cv2.GaussianBlur(feather, (7, 7), 1.35)[..., None]
+    dst = clean[y0:y1, x0:x1].astype(np.float32)
+    clean[y0:y1, x0:x1] = np.clip(dst * (1.0-feather) + patch * feather + .5, 0, 255).astype(np.uint8)
+
+    # The late clean observation at the white start is partly occupied by the
+    # stopped red mallet, so use the measured white silhouette as an inpaint
+    # mask rather than copying that patch.
+    shift_x = int(round(white_center_initial[0] - white_center_final[0]))
+    shift_y = int(round(white_center_initial[1] - white_center_final[1]))
+    wx0, wy0 = white_origin
+    mask = np.zeros(clean.shape[:2], np.uint8)
+    binary = (white_alpha > 0.035).astype(np.uint8) * 255
+    binary = cv2.dilate(binary, np.ones((3, 3), np.uint8), iterations=1)
+    tx, ty = wx0 + shift_x, wy0 + shift_y
+    h, w = binary.shape
+    mask[ty:ty+h, tx:tx+w] = binary
+    clean_bgr = cv2.cvtColor(clean, cv2.COLOR_RGB2BGR)
+    clean_bgr = cv2.inpaint(clean_bgr, mask, 3.0, cv2.INPAINT_TELEA)
+    return cv2.cvtColor(clean_bgr, cv2.COLOR_BGR2RGB)
+
+
+def ease_linear(a, b, t):
+    return np.asarray(a, np.float32) * (1.0 - t) + np.asarray(b, np.float32) * t
+
+
+def generate():
+    global SOURCE
+    SOURCE = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert SOURCE.shape == (96, 144, 256, 3) and SOURCE.dtype == np.uint8
+
+    # Measured centers in source-frame coordinates.
+    red_center = np.array([143.2, 66.7], np.float32)
+    white_start = np.array([105.2, 81.2], np.float32)
+    white_finish = np.array([68.0, 95.2], np.float32)
+    green_source_center = np.array([167.9, 57.8], np.float32)
+
+    # The exact requested frame-1 placement.
+    green_start = (red_center + white_start) * 0.5
+    # Contact point immediately to the right/up of the white mallet.  It is the
+    # resting point of the striker after equal-mass momentum transfer.
+    green_contact = np.array([113.0, 78.2], np.float32)
+
+    green_sprite, green_alpha, green_origin = extract_sprite(
+        SOURCE[0], SOURCE[48], (156, 48, 181, 72), 1.2, 9.0
+    )
+    # At frame 48 the white mallet is isolated at its final resting location.
+    white_sprite, white_alpha, white_origin = extract_sprite(
+        SOURCE[48], SOURCE[0], (55, 85, 82, 106), 1.2, 8.0
+    )
+
+    background = erase_initial_objects(
+        SOURCE[0], green_alpha, green_origin, white_alpha, white_origin,
+        white_finish, white_start
+    )
+
+    frames = np.empty_like(SOURCE)
+    collision_frame = 7
+    white_stop_frame = 25
+    for index in range(96):
+        frame = background.copy()
+
+        # Green retains the source's incoming direction and stops on contact.
+        if index <= collision_frame:
+            t = index / collision_frame
+            green_position = ease_linear(green_start, green_contact, t)
+        else:
+            green_position = green_contact
+
+        # White is stationary until impact, then follows the same constant
+        # left/down translation seen in the factual video, only earlier.
+        if index <= collision_frame:
+            white_position = white_start
+        elif index <= white_stop_frame:
+            t = (index - collision_frame) / (white_stop_frame - collision_frame)
+            white_position = ease_linear(white_start, white_finish, t)
+        else:
+            white_position = white_finish
+
+        # Depth order at impact: white first, then the incoming green; away
+        # from contact the silhouettes do not overlap.
+        place_sprite(frame, white_sprite, white_alpha, white_origin, white_finish, white_position)
+        place_sprite(frame, green_sprite, green_alpha, green_origin, green_source_center, green_position)
+        frames[index] = frame
+
+    return frames
+
+
+if __name__ == "__main__":
+    result = generate()
+    artifact_io.save(result, CASE_DIR)
+    print("saved", result.shape, result.dtype, "to", CASE_DIR)

The source contains a three-mallet momentum-transfer sequence.  At frame 1 the
moving green mallet is relocated to the midpoint of the initially stationary

from pathlib import Path

import cv2
import numpy as np

import artifact_io

CASE_DIR = Path(__file__).resolve().parent


def smoothstep(x):
    x = np.clip(x, 0.0, 1.0)
    return x * x * (3.0 - 2.0 * x)


def extract_sprite(foreground, clean_reference, box, diff_lo=1.5, diff_hi=10.0):
    \"\"\"Extract an antialiased source sprite using a clean frame as reference.\"\"\"
    x0, y0, x1, y1 = box
    src = foreground[y0:y1, x0:x1].copy()
    ref = clean_reference[y0:y1, x0:x1]
    difference = np.max(np.abs(src.astype(np.float32) - ref.astype(np.float32)), axis=2)

    # Reject isolated codec noise, retain the coherent changed component, and
    # retain faint antialiased/shadow pixels immediately around it.
    strong = (difference >= 6.0).astype(np.uint8)
    count, labels, stats, _ = cv2.connectedComponentsWithStats(strong, 8)
    if count > 1:
        keep_id = 1 + int(np.argmax(stats[1:, cv2.CC_STAT_AREA]))
        coherent = (labels == keep_id).astype(np.uint8)
    else:
        coherent = strong
    support = cv2.dilate(coherent, np.ones((3, 3), np.uint8), iterations=2)
    raw = smoothstep((difference - diff_lo) / (diff_hi - diff_lo)) * support
    alpha = cv2.GaussianBlur(raw.astype(np.float32), (3, 3), 0.55)
    alpha[coherent.astype(bool)] = np.maximum(alpha[coherent.astype(bool)], 0.94)
    return src, np.clip(alpha, 0.0, 1.0), (x0, y0)


def place_sprite(canvas, sprite, alpha, source_origin, source_center, target_center):
    \"\"\"Place a sprite by translating its measured center to target_center.\"\"\"
    x0, y0 = source_origin
    dx = int(round(float(target_center[0] - source_center[0])))
    dy = int(round(float(target_center[1] - source_center[1])))
    tx, ty = x0 + dx, y0 + dy
    h, w = alpha.shape

    # Clip safely (not normally needed for this scene).
    cx0, cy0 = max(0, tx), max(0, ty)
    cx1, cy1 = min(canvas.shape[1], tx + w), min(canvas.shape[0], ty + h)
    if cx1 <= cx0 or cy1 <= cy0:
        return
    sx0, sy0 = cx0 - tx, cy0 - ty
    sx1, sy1 = sx0 + (cx1 - cx0), sy0 + (cy1 - cy0)
    a = alpha[sy0:sy1, sx0:sx1, None]
    under = canvas[cy0:cy1, cx0:cx1].astype(np.float32)
    over = sprite[sy0:sy1, sx0:sx1].astype(np.float32)
    canvas[cy0:cy1, cx0:cx1] = np.clip(under * (1.0 - a) + over * a + 0.5, 0, 255).astype(np.uint8)


def erase_initial_objects(frame, green_alpha, green_origin, white_alpha, white_origin, white_center_final, white_center_initial):
    \"\"\"Remove the original green and white while preserving the fixed red.\"\"\"
    clean = frame.copy()

    # Green's original location has a clean observation late in the source.
    # A feathered patch transfer preserves the patterned table much better than
    # synthesizing it.
    source = SOURCE[48]
    gx0, gy0 = green_origin
    gh, gw = green_alpha.shape
    pad = 3
    x0, y0 = gx0 - pad, gy0 - pad
    x1, y1 = gx0 + gw + pad, gy0 + gh + pad
    patch = source[y0:y1, x0:x1].astype(np.float32)
    feather = np.zeros((y1-y0, x1-x0), np.float32)
    feather[pad:pad+gh, pad:pad+gw] = np.maximum(green_alpha, 0.25 * (green_alpha > 0.02))
    feather = cv2.GaussianBlur(feather, (7, 7), 1.35)[..., None]
    dst = clean[y0:y1, x0:x1].astype(np.float32)
    clean[y0:y1, x0:x1] = np.clip(dst * (1.0-feather) + patch * feather + .5, 0, 255).astype(np.uint8)

    # The late clean observation at the white start is partly occupied by the
    # stopped red mallet, so use the measured white silhouette as an inpaint
    # mask rather than copying that patch.
    shift_x = int(round(white_center_initial[0] - white_center_final[0]))
    shift_y = int(round(white_center_initial[1] - white_center_final[1]))
    wx0, wy0 = white_origin
    mask = np.zeros(clean.shape[:2], np.uint8)
    binary = (white_alpha > 0.035).astype(np.uint8) * 255
    binary = cv2.dilate(binary, np.ones((3, 3), np.uint8), iterations=1)
    tx, ty = wx0 + shift_x, wy0 + shift_y
    h, w = binary.shape
    mask[ty:ty+h, tx:tx+w] = binary
    clean_bgr = cv2.cvtColor(clean, cv2.COLOR_RGB2BGR)
    clean_bgr = cv2.inpaint(clean_bgr, mask, 3.0, cv2.INPAINT_TELEA)
    return cv2.cvtColor(clean_bgr, cv2.COLOR_BGR2RGB)


def ease_linear(a, b, t):
    return np.asarray(a, np.float32) * (1.0 - t) + np.asarray(b, np.float32) * t

    global SOURCE
    SOURCE = np.load(CASE_DIR / \"source_pixels.npz\")[\"rgb\"]
    assert SOURCE.shape == (96, 144, 256, 3) and SOURCE.dtype == np.uint8

    # Measured centers in source-frame coordinates.
    red_center = np.array([143.2, 66.7], np.float32)
    white_start = np.array([105.2, 81.2], np.float32)
    white_finish = np.array([68.0, 95.2], np.float32)
    green_source_center = np.array([167.9, 57.8], np.float32)

    # The exact requested frame-1 placement.
    green_start = (red_center + white_start) * 0.5
    # Contact point immediately to the right/up of the white mallet.  It is the
    # resting point of the striker after equal-mass momentum transfer.
    green_contact = np.array([113.0, 78.2], np.float32)

    green_sprite, green_alpha, green_origin = extract_sprite(
        SOURCE[0], SOURCE[48], (156, 48, 181, 72), 1.2, 9.0
    )
    # At frame 48 the white mallet is isolated at its final resting location.
    white_sprite, white_alpha, white_origin = extract_sprite(
        SOURCE[48], SOURCE[0], (55, 85, 82, 106), 1.2, 8.0
    )

    background = erase_initial_objects(
        SOURCE[0], green_alpha, green_origin, white_alpha, white_origin,
        white_finish, white_start
    )

    frames = np.empty_like(SOURCE)
    collision_frame = 7
    white_stop_frame = 25
    for index in range(96):
        frame = background.copy()

        # Green retains the source's incoming direction and stops on contact.
        if index <= collision_frame:
            t = index / collision_frame
            green_position = ease_linear(green_start, green_contact, t)
        else:
            green_position = green_contact

        # White is stationary until impact, then follows the same constant
        # left/down translation seen in the factual video, only earlier.
        if index <= collision_frame:
            white_position = white_start
        elif index <= white_stop_frame:
            t = (index - collision_frame) / (white_stop_frame - collision_frame)
            white_position = ease_linear(white_start, white_finish, t)
        else:
            white_position = white_finish

        # Depth order at impact: white first, then the incoming green; away
        # from contact the silhouettes do not overlap.
        place_sprite(frame, white_sprite, white_alpha, white_origin, white_finish, white_position)
        place_sprite(frame, green_sprite, green_alpha, green_origin, green_source_center, green_position)
        frames[index] = frame

    return frames
