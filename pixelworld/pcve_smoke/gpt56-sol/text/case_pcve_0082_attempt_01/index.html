+"""Remove the falling apple and its physical effects from the source timeline.
+
+The source camera is locked.  Before the apple enters the top of frame, frames
+0--2 provide an unobscured clean plate.  In the factual video the apple first
+casts a small moving shadow, then falls into the wheeled object at the bottom
+and sets it in motion.  The counterfactual therefore removes (1) that shadow,
+(2) the falling/rolling apple, and (3) the resulting motion of the object,
+restoring the latter to its pre-impact pose.
+"""
+
+from pathlib import Path
+
+import numpy as np
+from scipy.ndimage import binary_dilation, gaussian_filter
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def clean_plate(frames: np.ndarray) -> np.ndarray:
+    """Robust pre-intervention plate, retaining the source's fine detail."""
+    return np.median(frames[:3].astype(np.float32), axis=0).astype(np.uint8)
+
+
+def local_change_mask(frame: np.ndarray, plate: np.ndarray, ellipses) -> np.ndarray:
+    """Select source changes near a known apple/shadow position.
+
+    Gating the temporal difference by geometric support avoids touching the
+    detailed but static scene.  Dilation and a soft edge also cover antialiasing
+    and the faint penumbra of the apple's shadow.
+    """
+    h, w = frame.shape[:2]
+    yy, xx = np.mgrid[:h, :w]
+    support = np.zeros((h, w), bool)
+    for cx, cy, rx, ry in ellipses:
+        support |= ((xx - cx) / rx) ** 2 + ((yy - cy) / ry) ** 2 <= 1.0
+
+    delta = np.max(np.abs(frame.astype(np.int16) - plate.astype(np.int16)), axis=2)
+    changed = support & (delta >= 3)
+    changed = binary_dilation(changed, iterations=2)
+    # Keep dilation local even when two nearby supports approach one another.
+    changed &= binary_dilation(support, iterations=1)
+    return np.clip(gaussian_filter(changed.astype(np.float32), 0.75), 0.0, 1.0)
+
+
+def action_strip_mask(h: int, w: int) -> np.ndarray:
+    """Soft mask covering the apple impact, cart displacement and apple roll."""
+    mask = np.zeros((h, w), np.float32)
+    # All factual motion after impact is confined to x=64..255, y=70..92.
+    # Extra static margins allow the feather to end on unchanged pixels.
+    mask[66:96, 57:w] = 1.0
+    return gaussian_filter(mask, 1.05)
+
+
+def blend(frame: np.ndarray, plate: np.ndarray, alpha: np.ndarray) -> np.ndarray:
+    a = alpha[..., None]
+    out = frame.astype(np.float32) * (1.0 - a) + plate.astype(np.float32) * a
+    return np.clip(np.rint(out), 0, 255).astype(np.uint8)
+
+
+def generate() -> np.ndarray:
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3) and source.dtype == np.uint8
+    plate = clean_plate(source)
+    result = source.copy()
+
+    # Centers measured on the native 256x144 source.  The first track is the
+    # apple-shaped shadow; from frame 9 onward the second is the apple itself.
+    tracks = {
+        3: [(185.5, 30.5, 6.0, 5.0)],
+        4: [(184.0, 32.0, 6.0, 5.0)],
+        5: [(182.5, 35.0, 6.5, 5.5)],
+        6: [(180.5, 37.0, 6.5, 5.5)],
+        7: [(177.0, 40.0, 6.5, 5.5)],
+        8: [(173.5, 44.0, 6.5, 5.5)],
+        9: [(169.5, 49.0, 7.0, 5.5), (135.0, 12.0, 6.5, 6.5)],
+        10: [(164.0, 55.0, 7.0, 5.5), (135.0, 29.5, 6.5, 6.5)],
+        11: [(158.0, 62.5, 7.0, 5.5), (134.5, 46.5, 6.5, 6.5)],
+        12: [(150.0, 71.5, 7.0, 5.5), (134.5, 63.5, 6.5, 6.5)],
+    }
+    for index, ellipses in tracks.items():
+        alpha = local_change_mask(source[index], plate, ellipses)
+        result[index] = blend(source[index], plate, alpha)
+
+    # With no apple there is no impact: restore the struck object to its
+    # pre-impact pose and remove the apple's subsequent rightward roll.
+    strip = action_strip_mask(source.shape[1], source.shape[2])
+    for index in range(13, source.shape[0]):
+        result[index] = blend(source[index], plate, strip)
+
+    return result
+
+
+if __name__ == "__main__":
+    artifact_io.save(generate(), CASE_DIR)
+
