+def connected_centroid(mask, weight, x0=0, y0=0):
+    """Return the weighted centroid of the strongest small 8-connected blob."""
+    h, w = mask.shape
+    seen = np.zeros_like(mask, dtype=bool)
+    best = None
+    for sy, sx in zip(*np.nonzero(mask)):
+        if seen[sy, sx]:
+            continue
+        stack = [(int(sy), int(sx))]
+        seen[sy, sx] = True
+        points = []
+        while stack:
+            y, x = stack.pop()
+            points.append((y, x))
+            for yy in range(max(0, y - 1), min(h, y + 2)):
+                for xx in range(max(0, x - 1), min(w, x + 2)):
+                    if mask[yy, xx] and not seen[yy, xx]:
+                        seen[yy, xx] = True
+                        stack.append((yy, xx))
+        if not (6 <= len(points) <= 80):
+            continue
+        ys = np.array([p[0] for p in points])
+        xs = np.array([p[1] for p in points])
+        ww = weight[ys, xs].astype(np.float64)
+        score = float(ww.sum())
+        if best is None or score > best[0]:
+            best = (score, x0 + float((xs * ww).sum() / ww.sum()),
+                    y0 + float((ys * ww).sum() / ww.sum()))
+    if best is None:
+        raise RuntimeError("Could not locate the tennis ball")
+    return best[1], best[2]
+

+def track_source_ball(source, background):
+    """Track the sole moving component without relying on any supplied mask."""
+    positions = []
+    # The ball's factual path is wholly inside this conservative region.
+    x0, x1, y0, y1 = 20, 216, 25, 101
+    for frame in source:
+        delta = np.max(np.abs(frame.astype(np.int16) - background), axis=2)
+        crop = delta[y0:y1, x0:x1]
+        positions.append(connected_centroid(crop > 8, crop, x0, y0))
+    return np.asarray(positions, dtype=np.float64)

+def clear_original_ball(frame, background, center):
+    """Remove the factual ball with a softly edged, temporally matched clean plate."""
+    out = frame.astype(np.float32).copy()
+    cx, cy = center
+    yy, xx = np.indices(frame.shape[:2])
+    distance = np.hypot(xx - cx, yy - cy)
+
+    # Match the tiny per-frame codec/lighting drift before taking pixels from the
+    # temporal clean plate.  The median is measured away from the moving ball.
+    residual = frame.astype(np.int16) - background.astype(np.int16)
+    annulus = (distance >= 5.0) & (distance <= 9.0)
+    drift = np.median(residual[annulus], axis=0)
+    clean = np.clip(background.astype(np.float32) + drift, 0, 255)
+
+    # All visible ball pixels lie within r=3; the narrow feather avoids a seam.
+    alpha = np.clip((3.8 - distance) / 0.8, 0.0, 1.0)[..., None]
+    out = out * (1.0 - alpha) + clean * alpha
+    return out

+def make_ball_templates(source, background, tracked):
+    """Collect crisp factual ball sprites from its unobstructed flight."""
+    templates = []
+    # Frames 4..22 have clean surroundings and varied sub-pixel registrations.
+    for t in range(4, 23):
+        cx, cy = tracked[t]
+        left, top = int(np.floor(cx)) - 5, int(np.floor(cy)) - 5
+        src = source[t, top:top + 11, left:left + 11].astype(np.float32)
+        bg = background[top:top + 11, left:left + 11].astype(np.float32)
+        strength = np.max(np.abs(src - bg), axis=2)
+        # Antialias coverage inferred from temporal foreground separation.
+        alpha = np.clip((strength - 1.5) / 70.0, 0.0, 1.0)
+        alpha[strength > 62.0] = 1.0
+        alpha[strength < 3.0] = 0.0
+        # Suppress isolated compression differences outside the 3.5 px ball.
+        yy, xx = np.indices(alpha.shape)
+        local_cx, local_cy = cx - left, cy - top
+        alpha *= np.clip((4.0 - np.hypot(xx - local_cx, yy - local_cy)) / 1.0, 0, 1)
+        templates.append({

+            "bg": bg,
+            "alpha": alpha,
+            "cx": local_cx,
+            "cy": local_cy,
+            "frac": np.array([cx % 1.0, cy % 1.0]),
+        })
+    return templates

+def composite_ball(frame, center, templates):
+    """Composite the nearest sub-pixel factual sprite at a new center."""
+    h, w = frame.shape[:2]
+    cx, cy = center
+    target_frac = np.array([cx % 1.0, cy % 1.0])
+    # Circular fractional distance (0.98 is close to 0.02 after integer shift).
+    best = None
+    for templ in templates:
+        d = np.abs(templ["frac"] - target_frac)
+        d = np.minimum(d, 1.0 - d)
+        score = float(np.dot(d, d))
+        if best is None or score < best[0]:
+            best = (score, templ)
+    templ = best[1]
+
+    # An integer translation of a source sprite keeps the five-pixel ball crisp.
+    left = int(round(cx - templ["cx"]))
+    top = int(round(cy - templ["cy"]))
+    x0, y0 = max(0, left), max(0, top)
+    x1, y1 = min(w, left + 11), min(h, top + 11)
+    if x0 >= x1 or y0 >= y1:
+        return frame
+    sx0, sy0 = x0 - left, y0 - top
+    sx1, sy1 = sx0 + x1 - x0, sy0 + y1 - y0
+
+    src = templ["src"][sy0:sy1, sx0:sx1]
+    old_bg = templ["bg"][sy0:sy1, sx0:sx1]
+    alpha = templ["alpha"][sy0:sy1, sx0:sx1, None]
+    destination = frame[y0:y1, x0:x1]
+
+    # Recover an approximate foreground color from the original antialiasing.
+    safe_alpha = np.maximum(alpha, 0.18)
+    foreground = old_bg + (src - old_bg) / safe_alpha
+    foreground = np.clip(foreground, 0, 255)
+    destination[:] = destination * (1.0 - alpha) + foreground * alpha
+    return frame
+

+def simulate_rebound(start_x, ground_y, impact_speed, frame_count):
+    """Counterfactual flight using measured image-space gravity and 18x e."""
+    gravity = 0.4166                  # px / frame^2, fitted before first impact
+    original_restitution = 0.05      # low source value; its bounce is sub-pixel
+    restitution = 18.0 * original_restitution
+    vx = 2.40                        # measured first-contact tangential speed
+    vy = -restitution * impact_speed
+    x, y = float(start_x), float(ground_y)
+    path = []
+
+    # Semi-implicit substeps give stable collision timing and smooth arcs.
+    substeps = 8
+    dt = 1.0 / substeps
+    for _ in range(frame_count):
+        path.append((x, y))
+        for _ in range(substeps):
+            vy += gravity * dt
+            x += vx * dt
+            y += vy * dt
+            if y > ground_y:
+                y = ground_y
+                vy = -abs(vy) * restitution
+                # A brief court contact removes some tangential energy.  Unlike
+                # the factual rolling ball, there is no continuous ground drag
+                # while the counterfactual ball is airborne.
+                vx *= 0.92
+    return np.asarray(path)
+

+def generate():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3) and source.dtype == np.uint8
+
+    # A temporal median is a clean plate because the small ball occupies any
+    # one pixel for far fewer than half of the 96 frames.
+    background = np.median(source, axis=0).astype(np.uint8)
+    tracked = track_source_ball(source, background)
+    templates = make_ball_templates(source, background, tracked)
+
+    # The source first reaches the court in zero-based frame 28.  Restitution
+    # cannot alter the preceding free flight, so those frames remain bit exact.
+    impact_frame = 28
+    impact_x = tracked[impact_frame, 0]
+    ground_y = float(np.median(tracked[28:70, 1]))
+    gravity = 0.4166
+    impact_speed = (-4.5326 + gravity * impact_frame)  # fitted pre-impact vy
+    rebound = simulate_rebound(impact_x, ground_y, impact_speed,
+                               len(source) - impact_frame)
+
+    output = source.astype(np.float32).copy()
+    # Frame 28 itself is the instant of contact and remains factual.  Starting
+    # with frame 29, erase the rolling source ball and draw the rebound.
+    for t in range(impact_frame + 1, len(source)):
+        output[t] = clear_original_ball(output[t], background, tracked[t])
+        new_center = rebound[t - impact_frame]
+        if (-6.0 < new_center[0] < source.shape[2] + 6.0 and
+                -6.0 < new_center[1] < source.shape[1] + 6.0):
+            output[t] = composite_ball(output[t], new_center, templates)
+
+    return np.clip(np.rint(output), 0, 255).astype(np.uint8)
+

+    frames = generate()
+    artifact_io.save(frames, CASE_DIR)

    left = int(round(cx - templ[\"cx\"]))
    top = int(round(cy - templ[\"cy\"]))
    x0, y0 = max(0, left), max(0, top)
    x1, y1 = min(w, left + 11), min(h, top + 11)
    if x0 >= x1 or y0 >= y1:
        return frame
    sx0, sy0 = x0 - left, y0 - top
    sx1, sy1 = sx0 + x1 - x0, sy0 + y1 - y0

    src = templ[\"src\"][sy0:sy1, sx0:sx1]
    old_bg = templ[\"bg\"][sy0:sy1, sx0:sx1]
    alpha = templ[\"alpha\"][sy0:sy1, sx0:sx1, None]
    destination = frame[y0:y1, x0:x1]

    # Recover an approximate foreground color from the original antialiasing.
    safe_alpha = np.maximum(alpha, 0.18)
    foreground = old_bg + (src - old_bg) / safe_alpha
    foreground = np.clip(foreground, 0, 255)
    destination[:] = destination * (1.0 - alpha) + foreground * alpha
    return frame


def simulate_rebound(start_x, ground_y, impact_speed, frame_count):
    \"\"\"Counterfactual flight using measured image-space gravity and 18x e.\"\"\"
    gravity = 0.4166                  # px / frame"'^2, fitted before first impact
    original_restitution = 0.05      # low source value; its bounce is sub-pixel
    restitution = 18.0 * original_restitution
    vx = 2.40                        # measured first-contact tangential speed
    vy = -restitution * impact_speed
    x, y = float(start_x), float(ground_y)
    path = []

    # Semi-implicit substeps give stable collision timing and smooth arcs.
    substeps = 8
    dt = 1.0 / substeps
    for _ in range(frame_count):
        path.append((x, y))
        for _ in range(substeps):
            vy += gravity * dt
            x += vx * dt
            y += vy * dt
            if y > ground_y:
                y = ground_y
                vy = -abs(vy) * restitution
                # A brief court contact removes some tangential energy.  Unlike
                # the factual rolling ball, there is no continuous ground drag
                # while the counterfactual ball is airborne.
                vx *= 0.92
    return np.asarray(path)


def generate():
    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
    assert source.shape == (96, 144, 256, 3) and source.dtype == np.uint8

    # A temporal median is a clean plate because the small ball occupies any
    # one pixel for far fewer than half of the 96 frames.
    background = np.median(source, axis=0).astype(np.uint8)
    tracked = track_source_ball(source, background)
    templates = make_ball_templates(source, background, tracked)

    # The source first reaches the court in zero-based frame 28.  Restitution
    # cannot alter the preceding free flight, so those frames remain bit exact.
    impact_frame = 28
    impact_x = tracked[impact_frame, 0]
    ground_y = float(np.median(tracked[28:70, 1]))
    gravity = 0.4166
    impact_speed = (-4.5326 + gravity * impact_frame)  # fitted pre-impact vy
    rebound = simulate_rebound(impact_x, ground_y, impact_speed,
                               len(source) - impact_frame)

    output = source.astype(np.float32).copy()
    # Frame 28 itself is the instant of contact and remains factual.  Starting
    # with frame 29, erase the rolling source ball and draw the rebound.
    for t in range(impact_frame + 1, len(source)):
        output[t] = clear_original_ball(output[t], background, tracked[t])
        new_center = rebound[t - impact_frame]
        if (-6.0 < new_center[0] < source.shape[2] + 6.0 and
                -6.0 < new_center[1] < source.shape[1] + 6.0):
            output[t] = composite_ball(output[t], new_center, templates)

    return np.clip(np.rint(output), 0, 255).astype(np.uint8)


if __name__ == "__main__":
    frames = generate()
    artifact_io.save(frames, CASE_DIR)
