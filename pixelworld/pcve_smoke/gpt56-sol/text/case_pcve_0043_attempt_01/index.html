+"""Generate the requested 0.95x-initial-speed counterfactual video.
+
+The intervention begins at time zero.  In this shot all meaningful motion is
+the toy-car event in the first second; the remainder is a still scene with
+minor source/compression variation.  We therefore retime the moving
+foreground to source time 0.95*t while retaining the factual background at
+output time t.  Dense motion compensation avoids the double images that a
+plain temporal cross-fade would produce.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+SPEED_SCALE = 0.95
+
+
+def foreground_mask(frame: np.ndarray, clean_plate: np.ndarray) -> np.ndarray:
+    """Return a clean binary mask for the car and the object it interacts with."""
+    delta = np.max(
+        np.abs(frame.astype(np.int16) - clean_plate.astype(np.int16)), axis=2
+    )
+    # The action is entirely in this region.  Restricting it keeps tiny codec
+    # differences elsewhere from being mistaken for moving foreground.
+    seed = ((delta > 10) & (np.indices(delta.shape)[0] >= 55) &
+            (np.indices(delta.shape)[1] < 205)).astype(np.uint8)
+    seed = cv2.morphologyEx(seed, cv2.MORPH_CLOSE, np.ones((3, 3), np.uint8))
+
+    count, labels, stats, _ = cv2.connectedComponentsWithStats(seed, 8)
+    mask = np.zeros_like(seed)
+    for label in range(1, count):
+        if stats[label, cv2.CC_STAT_AREA] >= 7:
+            mask[labels == label] = 1
+    # Include antialiased edge pixels and a narrow halo needed when removing
+    # the foreground from its factual (slightly faster) position.
+    return cv2.dilate(mask, np.ones((3, 3), np.uint8), iterations=1)
+
+
+def farneback(a: np.ndarray, b: np.ndarray) -> np.ndarray:
+    """Dense flow taking pixels in RGB image a to their locations in b."""
+    ga = cv2.cvtColor(a, cv2.COLOR_RGB2GRAY)
+    gb = cv2.cvtColor(b, cv2.COLOR_RGB2GRAY)
+    return cv2.calcOpticalFlowFarneback(
+        ga,
+        gb,
+        None,
+        pyr_scale=0.5,
+        levels=5,
+        winsize=31,
+        iterations=5,
+        poly_n=7,
+        poly_sigma=1.5,
+        flags=cv2.OPTFLOW_FARNEBACK_GAUSSIAN,
+    )
+
+
+def remap_toward(
+    endpoint: np.ndarray,
+    endpoint_mask: np.ndarray,
+    flow: np.ndarray,
+    amount: float,
+    grid_x: np.ndarray,
+    grid_y: np.ndarray,
+) -> tuple[np.ndarray, np.ndarray]:
+    """Move one endpoint ``amount`` of its flow toward the adjacent frame."""
+    map_x = grid_x - np.float32(amount) * flow[..., 0]
+    map_y = grid_y - np.float32(amount) * flow[..., 1]
+    moved = cv2.remap(
+        endpoint,
+        map_x,
+        map_y,
+        cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_REFLECT_101,
+    )
+    moved_mask = cv2.remap(
+        endpoint_mask.astype(np.float32),
+        map_x,
+        map_y,
+        cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=0,
+    )
+    return moved, moved_mask
+
+
+def synthesize_foreground_time(
+    frames: np.ndarray,
+    masks: list[np.ndarray],
+    source_time: float,
+    grid_x: np.ndarray,
+    grid_y: np.ndarray,
+) -> tuple[np.ndarray, np.ndarray]:
+    """Motion-compensated source frame and foreground alpha at fractional time."""
+    lo = int(np.floor(source_time))
+    hi = min(lo + 1, len(frames) - 1)
+    fraction = float(source_time - lo)
+    if hi == lo or fraction < 1e-7:
+        return frames[lo].copy(), masks[lo].astype(np.float32)
+
+    # Warp from the closer source endpoint.  Unlike a cross-fade this keeps
+    # the small, high-contrast toy sharply defined.  Forward flow is used from
+    # lo; reverse flow is used from hi.
+    if fraction < 0.5:
+        flow = farneback(frames[lo], frames[hi])
+        return remap_toward(
+            frames[lo], masks[lo], flow, fraction, grid_x, grid_y
+        )
+    flow = farneback(frames[hi], frames[lo])
+    return remap_toward(
+        frames[hi], masks[hi], flow, 1.0 - fraction, grid_x, grid_y
+    )
+
+
+def generate() -> np.ndarray:
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3) and source.dtype == np.uint8
+
+    # Frames 23--34 are after the car and affected prop have exited, and
+    # precede later encoder refreshes.  Their median is a stable clean plate.
+    clean_plate = np.median(source[23:35], axis=0).astype(np.uint8)
+    masks = [foreground_mask(frame, clean_plate) for frame in source]
+
+    height, width = source.shape[1:3]
+    grid_x, grid_y = np.meshgrid(
+        np.arange(width, dtype=np.float32),
+        np.arange(height, dtype=np.float32),
+    )
+    output = source.copy()
+
+    for output_index in range(len(source)):
+        source_time = SPEED_SCALE * output_index
+        # Once both the factual and slower events are over, preserving the
+        # source frame exactly is preferable to retiming static codec noise.
+        if source_time >= 23.0:
+            continue
+
+        moved, moved_mask = synthesize_foreground_time(
+            source, masks, source_time, grid_x, grid_y
+        )
+
+        # Replace both the old factual location and the desired slower
+        # location.  A softly feathered union removes seams while leaving the
+        # rest of the source image bit-for-bit unchanged.
+        old_mask = masks[output_index].astype(np.float32)
+        union = np.maximum(old_mask, moved_mask)
+        union = cv2.dilate((union > 0.08).astype(np.uint8),
+                           np.ones((3, 3), np.uint8)).astype(np.float32)
+        alpha = cv2.GaussianBlur(union, (5, 5), 0.8)[..., None]
+        composed = (
+            source[output_index].astype(np.float32) * (1.0 - alpha)
+            + moved.astype(np.float32) * alpha
+        )
+        output[output_index] = np.clip(np.rint(composed), 0, 255).astype(np.uint8)
+
+    # The initial position/history at t=0 must be identical; only velocity is
+    # changed.  This also guards against any mask-edge numerical alteration.
+    output[0] = source[0]
+    return output
+
+
+if __name__ == "__main__":
+    frames = generate()
+    artifact_io.save(frames, CASE_DIR)
+    print(
+        f"saved {len(frames)} frames at {frames.shape[2]}x{frames.shape[1]} "
+        f"with speed scale {SPEED_SCALE:.2f}"
+    )
