+"""Generate the requested 0.6x-initial-speed volleyball counterfactual.
+
+The source's only appreciable opening motion is the volleyball travelling away
+from the camera.  It reaches its resting/contact pose at about source index 19.
+Reducing its initial speed to 60% therefore maps source time s to 0.6*t until
+that pose is reached.  After the delayed arrival, source time advances normally
+again so that the intervention is not incorrectly applied to later dynamics.
+
+Fractional source times are rendered with bidirectional dense optical flow.
+This avoids the doubled/transparent ball produced by ordinary frame blending,
+while a feathered change mask leaves the static room completely undisturbed.
+All operations are local numeric operations on the supplied RGB tensor.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+SPEED_SCALE = 0.6
+# Inspection of consecutive RGB differences shows that the volleyball has
+# reached its final pose at source index 19 (source frame 20, one-based).
+SOURCE_ARRIVAL_INDEX = 19.0
+
+
+def source_time(output_index: int) -> float:
+    """Causal source-time map for the reduced launch speed."""
+    delayed_arrival = SOURCE_ARRIVAL_INDEX / SPEED_SCALE
+    if output_index <= delayed_arrival:
+        return SPEED_SCALE * output_index
+    # Once the ball has arrived, let subsequent events run at normal speed,
+    # shifted later by the extra flight time.
+    return SOURCE_ARRIVAL_INDEX + (output_index - delayed_arrival)
+
+
+def make_dis_flow() -> cv2.DISOpticalFlow:
+    """Create a high-quality dense flow estimator suited to small RGB frames."""
+    flow = cv2.DISOpticalFlow_create(cv2.DISOPTICAL_FLOW_PRESET_MEDIUM)
+    flow.setFinestScale(0)
+    flow.setPatchSize(8)
+    flow.setPatchStride(3)
+    flow.setGradientDescentIterations(40)
+    return flow
+
+
+def interpolate_motion_compensated(
+    first: np.ndarray,
+    second: np.ndarray,
+    amount: float,
+    grid_x: np.ndarray,
+    grid_y: np.ndarray,
+) -> np.ndarray:
+    """Interpolate two uint8 RGB frames with bidirectional optical flow."""
+    if amount <= 1e-7:
+        return first.copy()
+    if amount >= 1.0 - 1e-7:
+        return second.copy()
+
+    gray_first = cv2.cvtColor(first, cv2.COLOR_RGB2GRAY)
+    gray_second = cv2.cvtColor(second, cv2.COLOR_RGB2GRAY)
+
+    forward = make_dis_flow().calc(gray_first, gray_second, None)
+    backward = make_dis_flow().calc(gray_second, gray_first, None)
+
+    # Approximate inverse warps from the intermediate frame into each endpoint.
+    warped_first = cv2.remap(
+        first,
+        grid_x - amount * forward[..., 0],
+        grid_y - amount * forward[..., 1],
+        cv2.INTER_CUBIC,
+        borderMode=cv2.BORDER_REFLECT_101,
+    )
+    warped_second = cv2.remap(
+        second,
+        grid_x - (1.0 - amount) * backward[..., 0],
+        grid_y - (1.0 - amount) * backward[..., 1],
+        cv2.INTER_CUBIC,
+        borderMode=cv2.BORDER_REFLECT_101,
+    )
+    flow_result = (
+        (1.0 - amount) * warped_first.astype(np.float32)
+        + amount * warped_second.astype(np.float32)
+    )
+
+    # Ordinary interpolation is ideal in unchanged areas.  Restrict optical
+    # warping to a generously dilated, softly feathered union of changed pixels.
+    delta = np.max(
+        np.abs(second.astype(np.int16) - first.astype(np.int16)), axis=2
+    )
+    changed = (delta > 3).astype(np.uint8)
+    changed = cv2.dilate(changed, np.ones((9, 9), np.uint8), iterations=1)
+    feather = cv2.GaussianBlur(changed.astype(np.float32), (9, 9), 0)
+    feather = np.clip(feather[..., None], 0.0, 1.0)
+
+    linear_result = (
+        (1.0 - amount) * first.astype(np.float32)
+        + amount * second.astype(np.float32)
+    )
+    result = linear_result * (1.0 - feather) + flow_result * feather
+    return np.clip(np.rint(result), 0, 255).astype(np.uint8)
+
+
+def generate() -> np.ndarray:
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3)
+    assert source.dtype == np.uint8
+
+    height, width = source.shape[1:3]
+    grid_x, grid_y = np.meshgrid(
+        np.arange(width, dtype=np.float32),
+        np.arange(height, dtype=np.float32),
+    )
+
+    output = np.empty_like(source)
+    for output_index in range(len(output)):
+        time = float(np.clip(source_time(output_index), 0, len(source) - 1))
+        lower = int(np.floor(time))
+        upper = min(lower + 1, len(source) - 1)
+        fraction = time - lower
+
+        if upper == lower or fraction <= 1e-7:
+            output[output_index] = source[lower]
+            continue
+
+        # The post-arrival source is essentially static.  Linear RGB temporal
+        # interpolation retains its tiny natural lighting variations without
+        # inventing flow in flat areas; flow is reserved for visible movement.
+        pair_delta = np.max(
+            np.abs(source[upper].astype(np.int16) - source[lower].astype(np.int16)),
+            axis=2,
+        )
+        if float(pair_delta.mean()) < 0.20:
+            mixed = (
+                (1.0 - fraction) * source[lower].astype(np.float32)
+                + fraction * source[upper].astype(np.float32)
+            )
+            output[output_index] = np.clip(np.rint(mixed), 0, 255).astype(np.uint8)
+        else:
+            output[output_index] = interpolate_motion_compensated(
+                source[lower], source[upper], fraction, grid_x, grid_y
+            )
+
+    # The intervention starts at frame 1/time zero, so position and appearance
+    # at that instant are factual even though the following velocity is reduced.
+    output[0] = source[0]
+    return output
+
+
+if __name__ == "__main__":
+    frames = generate()
+    assert frames.shape == (96, 144, 256, 3)
+    assert frames.dtype == np.uint8
+    artifact_io.save(frames, CASE_DIR)
