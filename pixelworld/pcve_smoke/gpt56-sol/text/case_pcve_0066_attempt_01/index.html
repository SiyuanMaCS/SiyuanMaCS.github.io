+"""Generate the counterfactual domino video.
+
+The third domino is made ten times heavier from the first frame.  Mass is not
+visible, so the factual pixels remain valid until the second domino reaches
+the third.  In the source this happens at the transition from zero-based
+frame 30 to 31.  From that contact onward, the impacted/coupled motion is
+played at 1/10 speed.  This retains all of the source's material, lighting,
+camera, shadows, and contact detail while depicting the much greater inertia:
+the third domino moves only partway and the fourth does not complete its fall
+during the four-second shot.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+CONTACT_FRAME = 30          # zero based; last fully factual contact pose
+MASS_MULTIPLIER = 10.0
+
+
+def _flow_interpolate(a: np.ndarray, b: np.ndarray, amount: float) -> np.ndarray:
+    """Motion-compensated interpolation of two adjacent RGB source frames."""
+    if amount <= 1e-8:
+        return a.copy()
+    if amount >= 1.0 - 1e-8:
+        return b.copy()
+
+    ga = cv2.cvtColor(a, cv2.COLOR_RGB2GRAY)
+    gb = cv2.cvtColor(b, cv2.COLOR_RGB2GRAY)
+    kwargs = dict(
+        pyr_scale=0.5,
+        levels=4,
+        winsize=19,
+        iterations=4,
+        poly_n=7,
+        poly_sigma=1.5,
+        flags=cv2.OPTFLOW_FARNEBACK_GAUSSIAN,
+    )
+    forward = cv2.calcOpticalFlowFarneback(ga, gb, None, **kwargs)
+    backward = cv2.calcOpticalFlowFarneback(gb, ga, None, **kwargs)
+
+    h, w = ga.shape
+    x, y = np.meshgrid(np.arange(w, dtype=np.float32),
+                       np.arange(h, dtype=np.float32))
+
+    # A point at the intermediate time is approximately alpha of the way
+    # along its forward track.  Warp both endpoints toward that point, then
+    # blend; using reflected borders avoids edge-colored seams.
+    wa = cv2.remap(
+        a, x - amount * forward[..., 0], y - amount * forward[..., 1],
+        cv2.INTER_CUBIC, borderMode=cv2.BORDER_REFLECT_101,
+    )
+    wb = cv2.remap(
+        b, x - (1.0 - amount) * backward[..., 0],
+        y - (1.0 - amount) * backward[..., 1],
+        cv2.INTER_CUBIC, borderMode=cv2.BORDER_REFLECT_101,
+    )
+
+    # The nearer endpoint gets more weight.  This also makes disocclusions
+    # settle cleanly rather than exposing transparent/empty pixels.
+    mixed = wa.astype(np.float32) * (1.0 - amount)
+    mixed += wb.astype(np.float32) * amount
+    return np.clip(np.rint(mixed), 0, 255).astype(np.uint8)
+
+
+def generate() -> np.ndarray:
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3) and source.dtype == np.uint8
+
+    result = np.empty_like(source)
+
+    # Before impact, changing mass alone has no visible consequence.
+    result[: CONTACT_FRAME + 1] = source[: CONTACT_FRAME + 1]
+
+    # Once contacted, tenfold inertia makes the coupled progression one tenth
+    # as fast.  Fractional source times are motion-compensated to avoid a
+    # low-frame-rate freeze/step appearance.
+    for output_index in range(CONTACT_FRAME + 1, len(result)):
+        source_time = CONTACT_FRAME + (
+            (output_index - CONTACT_FRAME) / MASS_MULTIPLIER
+        )
+        lo = int(np.floor(source_time))
+        hi = min(lo + 1, len(source) - 1)
+        alpha = float(source_time - lo)
+        result[output_index] = _flow_interpolate(source[lo], source[hi], alpha)
+
+    return result
+
+
+if __name__ == "__main__":
+    frames = generate()
+    artifact_io.save(frames, CASE_DIR)
+    print(
+        f"saved {len(frames)} frames; exact source prefix 0..{CONTACT_FRAME}; "
+        f"final physical source time {CONTACT_FRAME + (95-CONTACT_FRAME)/MASS_MULTIPLIER:.2f}"
+    )
