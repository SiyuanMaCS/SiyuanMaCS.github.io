+#!/usr/bin/env python3
+"""Generate the 4x-soccer-ball-mass counterfactual video.
+
+The incoming small ball and all scene animation are retained exactly through contact.
+After contact, a 1-D impulse estimate is used: the much heavier soccer ball receives
+about 28% of its factual recoil speed, while the incoming ball rebounds about 1.27x
+faster.  The lower-field objects are re-composited over a robust clean plate so the
+rest of the source remains pixel-faithful.
+"""
+from pathlib import Path
+
+import cv2
+import numpy as np
+from scipy.ndimage import binary_fill_holes, gaussian_filter
+
+import artifact_io
+
+CASE_DIR = Path(__file__).resolve().parent
+H, W = 144, 256
+
+
+def foreground_mask(im):
+    """Segment the pale/blue/dark soccer ball from the green pitch."""
+    r, g, b = im.astype(np.int16).transpose(2, 0, 1)
+    return ((r - g > 22) | (b - g > 18) | (r > 145) |
+            (b > 145) | (g < 65)).astype(np.uint8)
+
+
+def track_soccer(frames):
+    centers = []
+    for im in frames:
+        mask = foreground_mask(im)
+        roi = np.zeros((H, W), np.uint8)
+        roi[66:99, 45:150] = mask[66:99, 45:150]
+        count, labels, stats, centroids = cv2.connectedComponentsWithStats(roi, 8)
+        candidates = [(stats[i, cv2.CC_STAT_AREA], i) for i in range(1, count)
+                      if stats[i, cv2.CC_STAT_AREA] > 100]
+        if not candidates:
+            centers.append(centers[-1])
+        else:
+            _, i = max(candidates)
+            centers.append(tuple(centroids[i]))
+    return np.asarray(centers, np.float32)
+
+
+def track_small_ball(frames):
+    """Track the little warm-colored rebounder after the impact."""
+    result = [None] * len(frames)
+    for t, im in enumerate(frames):
+        r, g, b = im.astype(np.int16).transpose(2, 0, 1)
+        warm = ((r - g > 20) & (r - b > 28) & (r > 115)).astype(np.uint8)
+        roi = np.zeros_like(warm)
+        roi[75:96, 135:W] = warm[75:96, 135:W]
+        n, labels, stats, centroids = cv2.connectedComponentsWithStats(roi, 8)
+        candidates = []
+        for i in range(1, n):
+            area = stats[i, cv2.CC_STAT_AREA]
+            ww, hh = stats[i, cv2.CC_STAT_WIDTH], stats[i, cv2.CC_STAT_HEIGHT]
+            if 8 <= area <= 80 and ww <= 12 and hh <= 12:
+                candidates.append((area, i))
+        if candidates:
+            _, i = max(candidates)
+            result[t] = tuple(centroids[i])
+    return result
+
+
+def clean_plate(frames, soccer_centers):
+    """Temporal clean plate, excluding every factual soccer-ball occupancy."""
+    data = frames.astype(np.float32)
+    yy, xx = np.mgrid[:H, :W]
+    valid = np.ones((len(frames), H, W), bool)
+    for t, (cx, cy) in enumerate(soccer_centers):
+        # Generous exclusion includes antialiasing and the contact shadow.
+        valid[t] &= (((xx - cx) / 17.0) ** 2 + ((yy - cy) / 14.0) ** 2 > 1.0)
+    # Only the lower pitch is ever sampled from this plate. NaN median removes
+    # moving-object samples while retaining the source's fine grass texture.
+    masked = np.where(valid[..., None], data, np.nan)
+    plate = np.nanmedian(masked, axis=0)
+    fallback = np.median(data, axis=0)
+    plate = np.where(np.isfinite(plate), plate, fallback)
+    return np.clip(np.rint(plate), 0, 255).astype(np.uint8)
+
+
+def largest_component(mask):
+    n, labels, stats, _ = cv2.connectedComponentsWithStats(mask.astype(np.uint8), 8)
+    if n <= 1:
+        return mask.astype(bool)
+    i = 1 + np.argmax(stats[1:, cv2.CC_STAT_AREA])
+    return labels == i
+
+
+def soccer_sprite(frames, plate, centers, t, size=(37, 31)):
+    sw, sh = size
+    cx, cy = centers[t]
+    x0, y0 = int(round(cx)) - sw // 2, int(round(cy)) - sh // 2
+    rgb = frames[t, y0:y0 + sh, x0:x0 + sw].copy()
+    sem = foreground_mask(rgb).astype(bool)
+    # Restrict to the central object and select its connected silhouette.
+    yy, xx = np.mgrid[:sh, :sw]
+    sem &= ((xx - sw // 2) / 15.5) ** 2 + ((yy - sh // 2) / 13.0) ** 2 <= 1
+    core = binary_fill_holes(largest_component(sem))
+    # Keep only the one-pixel fringe whose source really differs from clean grass.
+    dil = cv2.dilate(core.astype(np.uint8), np.ones((3, 3), np.uint8), iterations=1).astype(bool)
+    clean = plate[y0:y0 + sh, x0:x0 + sw]
+    difference = np.max(np.abs(rgb.astype(np.int16) - clean.astype(np.int16)), axis=2)
+    mask = core | (dil & (difference > 7))
+    alpha = gaussian_filter(mask.astype(np.float32), 0.38)
+    alpha = np.clip(alpha, 0.0, 1.0)
+    return rgb.astype(np.float32), alpha
+
+
+def small_sprite(frames, plate, center, t=20, size=13):
+    cx, cy = center
+    x0, y0 = int(round(cx)) - size // 2, int(round(cy)) - size // 2
+    rgb = frames[t, y0:y0 + size, x0:x0 + size].copy()
+    r, g, b = rgb.astype(np.int16).transpose(2, 0, 1)
+    core = (r - g > 20) & (r - b > 28) & (r > 115)
+    core = binary_fill_holes(largest_component(core))
+    dil = cv2.dilate(core.astype(np.uint8), np.ones((3, 3), np.uint8), iterations=1).astype(bool)
+    clean = plate[y0:y0 + size, x0:x0 + size]
+    difference = np.max(np.abs(rgb.astype(np.int16) - clean.astype(np.int16)), axis=2)
+    mask = core | (dil & (difference > 7))
+    alpha = np.clip(gaussian_filter(mask.astype(np.float32), 0.35), 0.0, 1.0)
+    return rgb.astype(np.float32), alpha
+
+
+def patch_ellipse(frame, plate, cx, cy, rx, ry):
+    """Feather a clean source plate over a factual object's old location."""
+    x0, x1 = max(0, int(cx - rx * 1.25 - 2)), min(W, int(cx + rx * 1.25 + 3))
+    y0, y1 = max(0, int(cy - ry * 1.25 - 2)), min(H, int(cy + ry * 1.25 + 3))
+    yy, xx = np.mgrid[y0:y1, x0:x1]
+    q = np.sqrt(((xx - cx) / rx) ** 2 + ((yy - cy) / ry) ** 2)
+    a = np.clip((1.16 - q) / 0.16, 0.0, 1.0)[..., None]
+    dst = frame[y0:y1, x0:x1].astype(np.float32)
+    bg = plate[y0:y1, x0:x1].astype(np.float32)
+    frame[y0:y1, x0:x1] = np.clip(np.rint(dst * (1 - a) + bg * a), 0, 255).astype(np.uint8)
+
+
+def blend_sprites(a, b, amount):
+    """Cross-dissolve aligned source sprites in premultiplied-alpha space."""
+    rgb0, al0 = a
+    rgb1, al1 = b
+    s = float(np.clip(amount, 0, 1))
+    al = al0 * (1 - s) + al1 * s
+    pre = rgb0 * al0[..., None] * (1 - s) + rgb1 * al1[..., None] * s
+    rgb = np.divide(pre, np.maximum(al[..., None], 1e-6), out=np.zeros_like(pre),
+                    where=al[..., None] > 1e-6)
+    return rgb, al
+
+
+def composite_sprite(frame, sprite, cx, cy):
+    """Subpixel-place an RGB/alpha sprite using numeric affine warping."""
+    rgb, alpha = sprite
+    sh, sw = alpha.shape
+    x = cx - sw // 2
+    y = cy - sh // 2
+    matrix = np.array([[1, 0, x], [0, 1, y]], np.float32)
+    pre = rgb * alpha[..., None]
+    wp = cv2.warpAffine(pre, matrix, (W, H), flags=cv2.INTER_LINEAR,
+                        borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+    wa = cv2.warpAffine(alpha, matrix, (W, H), flags=cv2.INTER_LINEAR,
+                        borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+    dst = frame.astype(np.float32)
+    out = dst * (1 - wa[..., None]) + wp
+    frame[:] = np.clip(np.rint(out), 0, 255).astype(np.uint8)
+
+
+def main():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, H, W, 3) and source.dtype == np.uint8
+
+    soccer_centers = track_soccer(source)
+    small_centers = track_small_ball(source)
+    plate = clean_plate(source, soccer_centers)
+
+    # Source appearances at 0, ~3.7 and ~6.2 pixels of factual roll.
+    ball_sprites = [soccer_sprite(source, plate, soccer_centers, t) for t in (0, 14, 15)]
+    # A clean, fully separated source view of the small rebounder.
+    small_ref_t = 20
+    small = small_sprite(source, plate, small_centers[small_ref_t], small_ref_t)
+
+    out = source.copy()
+    contact_frame = 13  # zero-based; visual contact itself is unchanged
+    original_cx, original_cy = map(float, soccer_centers[0])
+
+    for t in range(contact_frame + 1, len(out)):
+        frame = out[t]
+
+        # Remove the factual soccer ball and the factual small rebounder.
+        oldx, oldy = map(float, soccer_centers[t])
+        patch_ellipse(frame, plate, oldx, oldy, 16.0, 13.0)
+        if small_centers[t] is not None:
+            sx, sy = small_centers[t]
+            patch_ellipse(frame, plate, sx, sy, 5.7, 5.3)
+
+        n = float(t - contact_frame)
+
+        # Collision estimate. Factual initial recoil is ~2.5 px/frame. From the
+        # two outgoing speeds the original soccer/small mass ratio is about 6.2;
+        # quadrupling the soccer mass scales its recoil to 0.279. Rolling friction
+        # has mass-independent acceleration, so this heavier-ball recoil stops
+        # after only ~13 frames, about 4.5 pixels from its starting point.
+        v_soccer = 0.70
+        rolling_decel = 0.055
+        stop_t = v_soccer / rolling_decel
+        u = min(n, stop_t)
+        displacement = v_soccer * u - 0.5 * rolling_decel * u * u
+        new_ball_x = original_cx - displacement
+        new_ball_y = original_cy + min(displacement / 4.5, 1.0) * 0.25
+
+        # Interpolate genuine source roll appearances according to distance.
+        if displacement <= 3.71:
+            ball = blend_sprites(ball_sprites[0], ball_sprites[1], displacement / 3.71)
+        else:
+            ball = blend_sprites(ball_sprites[1], ball_sprites[2],
+                                 (displacement - 3.71) / (6.20 - 3.71))
+        composite_sprite(frame, ball, new_ball_x, new_ball_y)
+
+        # The lighter striker receives the complementary impulse and rebounds
+        # faster. It settles onto the same pitch level and exits frame naturally.
+        v_small = 8.40
+        small_decel = 0.15
+        new_small_x = 138.0 + v_small * n - 0.5 * small_decel * n * n
+        if n <= 1.0:
+            new_small_y = 77.0 + 8.5 * n
+        else:
+            new_small_y = 85.5
+        if new_small_x < W + 7:
+            composite_sprite(frame, small, new_small_x, new_small_y)
+
+    assert out.shape == (96, 144, 256, 3) and out.dtype == np.uint8
+    artifact_io.save(out, CASE_DIR)
+
+
+if __name__ == "__main__":
+    main()

def foreground_mask(im):

        warm = ((r - g > 20) & (r - b > 28) & (r > 115)).astype(np.uint8)
        roi = np.zeros_like(warm)
        roi[75:96, 135:W] = warm[75:96, 135:W]
        n, labels, stats, centroids = cv2.connectedComponentsWithStats(roi, 8)
        candidates = []
        for i in range(1, n):
            area = stats[i, cv2.CC_STAT_AREA]
            ww, hh = stats[i, cv2.CC_STAT_WIDTH], stats[i, cv2.CC_STAT_HEIGHT]
            if 8 <= area <= 80 and ww <= 12 and hh <= 12:
                candidates.append((area, i))
        if candidates:
            _, i = max(candidates)
            result[t] = tuple(centroids[i])
    return result


def clean_plate(frames, soccer_centers):
    """Temporal clean plate, excluding every factual soccer-ball occupancy."""
    data = frames.astype(np.float32)
    yy, xx = np.mgrid[:H, :W]
    valid = np.ones((len(frames), H, W), bool)
    for t, (cx, cy) in enumerate(soccer_centers):
        # Generous exclusion includes antialiasing and the contact shadow.
        valid[t] &= (((xx - cx) / 17.0) ** 2 + ((yy - cy) / 14.0) ** 2 > 1.0)
    # Only the lower pitch is ever sampled from this plate. NaN median removes
    # moving-object samples while retaining the source'"'s fine grass texture.
    masked = np.where(valid[..., None], data, np.nan)
    plate = np.nanmedian(masked, axis=0)
    fallback = np.median(data, axis=0)
    plate = np.where(np.isfinite(plate), plate, fallback)
    return np.clip(np.rint(plate), 0, 255).astype(np.uint8)


def largest_component(mask):
    n, labels, stats, _ = cv2.connectedComponentsWithStats(mask.astype(np.uint8), 8)
    if n <= 1:
        return mask.astype(bool)
    i = 1 + np.argmax(stats[1:, cv2.CC_STAT_AREA])
    return labels == i


def soccer_sprite(frames, plate, centers, t, size=(37, 31)):
    sw, sh = size
    cx, cy = centers[t]
    x0, y0 = int(round(cx)) - sw // 2, int(round(cy)) - sh // 2
    rgb = frames[t, y0:y0 + sh, x0:x0 + sw].copy()
    sem = foreground_mask(rgb).astype(bool)
    # Restrict to the central object and select its connected silhouette.
    yy, xx = np.mgrid[:sh, :sw]
    sem &= ((xx - sw // 2) / 15.5) ** 2 + ((yy - sh // 2) / 13.0) ** 2 <= 1
    core = binary_fill_holes(largest_component(sem))
    # Keep only the one-pixel fringe whose source really differs from clean grass.
    dil = cv2.dilate(core.astype(np.uint8), np.ones((3, 3), np.uint8), iterations=1).astype(bool)
    clean = plate[y0:y0 + sh, x0:x0 + sw]
    difference = np.max(np.abs(rgb.astype(np.int16) - clean.astype(np.int16)), axis=2)
    mask = core | (dil & (difference > 7))
    alpha = gaussian_filter(mask.astype(np.float32), 0.38)
    alpha = np.clip(alpha, 0.0, 1.0)
    return rgb.astype(np.float32), alpha


def small_sprite(frames, plate, center, t=20, size=13):
    cx, cy = center
    x0, y0 = int(round(cx)) - size // 2, int(round(cy)) - size // 2
    rgb = frames[t, y0:y0 + size, x0:x0 + size].copy()
    r, g, b = rgb.astype(np.int16).transpose(2, 0, 1)
    core = (r - g > 20) & (r - b > 28) & (r > 115)
    core = binary_fill_holes(largest_component(core))
    dil = cv2.dilate(core.astype(np.uint8), np.ones((3, 3), np.uint8), iterations=1).astype(bool)
    clean = plate[y0:y0 + size, x0:x0 + size]
    difference = np.max(np.abs(rgb.astype(np.int16) - clean.astype(np.int16)), axis=2)
    mask = core | (dil & (difference > 7))
    alpha = np.clip(gaussian_filter(mask.astype(np.float32), 0.35), 0.0, 1.0)
    return rgb.astype(np.float32), alpha


def patch_ellipse(frame, plate, cx, cy, rx, ry):
    \"\"\"Feather a clean source plate over a factual object's old location.\"\"\"
    x0, x1 = max(0, int(cx - rx * 1.25 - 2)), min(W, int(cx + rx * 1.25 + 3))
    y0, y1 = max(0, int(cy - ry * 1.25 - 2)), min(H, int(cy + ry * 1.25 + 3))
    yy, xx = np.mgrid[y0:y1, x0:x1]
    q = np.sqrt(((xx - cx) / rx) ** 2 + ((yy - cy) / ry) ** 2)
    a = np.clip((1.16 - q) / 0.16, 0.0, 1.0)[..., None]
    dst = frame[y0:y1, x0:x1].astype(np.float32)
    bg = plate[y0:y1, x0:x1].astype(np.float32)
    frame[y0:y1, x0:x1] = np.clip(np.rint(dst * (1 - a) + bg * a), 0, 255).astype(np.uint8)


def blend_sprites(a, b, amount):
    \"\"\"Cross-dissolve aligned source sprites in premultiplied-alpha space.\"\"\"
    rgb0, al0 = a
    rgb1, al1 = b
    s = float(np.clip(amount, 0, 1))
    al = al0 * (1 - s) + al1 * s
    pre = rgb0 * al0[..., None] * (1 - s) + rgb1 * al1[..., None] * s
    rgb = np.divide(pre, np.maximum(al[..., None], 1e-6), out=np.zeros_like(pre),
                    where=al[..., None] > 1e-6)
    return rgb, al


def composite_sprite(frame, sprite, cx, cy):
    \"\"\"Subpixel-place an RGB/alpha sprite using numeric affine warping.\"\"\"
    rgb, alpha = sprite
    sh, sw = alpha.shape
    x = cx - sw // 2
    y = cy - sh // 2
    matrix = np.array([[1, 0, x], [0, 1, y]], np.float32)
    pre = rgb * alpha[..., None]
    wp = cv2.warpAffine(pre, matrix, (W, H), flags=cv2.INTER_LINEAR,
                        borderMode=cv2.BORDER_CONSTANT, borderValue=0)
    wa = cv2.warpAffine(alpha, matrix, (W, H), flags=cv2.INTER_LINEAR,
                        borderMode=cv2.BORDER_CONSTANT, borderValue=0)
    dst = frame.astype(np.float32)
    out = dst * (1 - wa[..., None]) + wp
    frame[:] = np.clip(np.rint(out), 0, 255).astype(np.uint8)


def main():
    source = np.load(CASE_DIR / \"source_pixels.npz\")[\"rgb\"]
    assert source.shape == (96, H, W, 3) and source.dtype == np.uint8

    soccer_centers = track_soccer(source)
    small_centers = track_small_ball(source)
    plate = clean_plate(source, soccer_centers)

    # Source appearances at 0, ~3.7 and ~6.2 pixels of factual roll.
    ball_sprites = [soccer_sprite(source, plate, soccer_centers, t) for t in (0, 14, 15)]
    # A clean, fully separated source view of the small rebounder.
    small_ref_t = 20
    small = small_sprite(source, plate, small_centers[small_ref_t], small_ref_t)

    out = source.copy()
    contact_frame = 13  # zero-based; visual contact itself is unchanged
    original_cx, original_cy = map(float, soccer_centers[0])

    for t in range(contact_frame + 1, len(out)):
        frame = out[t]

        # Remove the factual soccer ball and the factual small rebounder.
        oldx, oldy = map(float, soccer_centers[t])
        patch_ellipse(frame, plate, oldx, oldy, 16.0, 13.0)
        if small_centers[t] is not None:
            sx, sy = small_centers[t]
            patch_ellipse(frame, plate, sx, sy, 5.7, 5.3)

        n = float(t - contact_frame)

        # Collision estimate. Factual initial recoil is ~2.5 px/frame. From the
        # two outgoing speeds the original soccer/small mass ratio is about 6.2;
        # quadrupling the soccer mass scales its recoil to 0.279. Rolling friction
        # has mass-independent acceleration, so this heavier-ball recoil stops
        # after only ~13 frames, about 4.5 pixels from its starting point.
        v_soccer = 0.70
        rolling_decel = 0.055
        stop_t = v_soccer / rolling_decel
        u = min(n, stop_t)
        displacement = v_soccer * u - 0.5 * rolling_decel * u * u
        new_ball_x = original_cx - displacement
        new_ball_y = original_cy + min(displacement / 4.5, 1.0) * 0.25

        # Interpolate genuine source roll appearances according to distance.
        if displacement <= 3.71:
            ball = blend_sprites(ball_sprites[0], ball_sprites[1], displacement / 3.71)
        else:
            ball = blend_sprites(ball_sprites[1], ball_sprites[2],
                                 (displacement - 3.71) / (6.20 - 3.71))
        composite_sprite(frame, ball, new_ball_x, new_ball_y)

        # The lighter striker receives the complementary impulse and rebounds
        # faster. It settles onto the same pitch level and exits frame naturally.
        v_small = 8.40
        small_decel = 0.15
        new_small_x = 138.0 + v_small * n - 0.5 * small_decel * n * n
        if n <= 1.0:
            new_small_y = 77.0 + 8.5 * n
        else:
            new_small_y = 85.5
        if new_small_x < W + 7:
            composite_sprite(frame, small, new_small_x, new_small_y)

    assert out.shape == (96, 144, 256, 3) and out.dtype == np.uint8
    artifact_io.save(out, CASE_DIR)


if __name__ == \"__main__\":
    main()
