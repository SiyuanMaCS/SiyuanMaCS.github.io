+"""Remove the red mallet beginning at source frame 22 (one-based).
+
+The shot is locked-off.  The mallet is the only sizeable moving object before
+frame 31; it then hits a light block, which moves left.  A robust clean plate
+is therefore made from the long, settled tail of the source.  We composite
+only the compact, measured dynamic envelope, preserving every unaffected
+pixel (and all frames through frame 21) exactly.
+"""
+from pathlib import Path
+import numpy as np
+from scipy.ndimage import binary_dilation, binary_fill_holes, label
+from artifact_io import save
+
+CASE_DIR = Path(__file__).resolve().parent
+START = 21                         # frame 22 in zero-based indexing
+
+
+def largest_component(mask):
+    lab, count = label(mask)
+    if not count:
+        return mask
+    sizes = np.bincount(lab.ravel())
+    sizes[0] = 0
+    return lab == sizes.argmax()
+
+
+def make_clean_plate(src):
+    # Frames 51--96 are essentially static (subpixel render noise only).  The
+    # temporal median removes that noise and reconstructs exposed background.
+    return np.median(src[50:96].astype(np.float32), axis=0).astype(np.uint8)
+
+
+def main():
+    src = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert src.shape == (96, 144, 256, 3) and src.dtype == np.uint8
+    out = src.copy()
+    plate = make_clean_plate(src)
+
+    # At frame 21 the red head occupies x=125..138, y=67..75.  Compared to the
+    # clean plate, color distance isolates head + dark handle/edge.  Restrict
+    # detection to its local region so unrelated red set dressing is untouched.
+    frame = src[20].astype(np.int16)
+    diff = np.max(np.abs(frame - plate.astype(np.int16)), axis=2)
+    red = ((frame[..., 0] - frame[..., 1] > 35) &
+           (frame[..., 0] - frame[..., 2] > 25) &
+           (frame[..., 0] > 110))
+    yy, xx = np.indices(diff.shape)
+    seed = (diff > 9) & ((red) | (diff > 24))
+    seed &= (xx >= 121) & (xx <= 142) & (yy >= 63) & (yy <= 79)
+    seed = largest_component(seed)
+    seed = binary_fill_holes(binary_dilation(seed, iterations=2))
+
+    # Include the slim adjacent handle/antialias fringe detected from temporal
+    # difference, but keep a tight union to avoid changing unaffected content.
+    fringe = (diff > 7) & (xx >= 121) & (xx <= 145) & (yy >= 62) & (yy <= 80)
+    mask = binary_dilation(seed | fringe, iterations=1)
+
+    # The counterfactual history after removal is the static scene at that
+    # instant: no mallet means no later collision, so the struck block stays in
+    # its pre-impact pose.  Freeze only the measured mallet envelope and use the
+    # clean plate there; everything outside remains the original source.
+    out[START:, mask] = plate[mask]
+
+    # Contract invariants and useful deterministic checks.
+    assert np.array_equal(out[:START], src[:START])
+    assert mask.sum() < 600
+    assert out.shape == (96, 144, 256, 3) and out.dtype == np.uint8
+    save(out, CASE_DIR)
+
+
+if __name__ == "__main__":
+    main()

def largest_component(mask):
    lab, count = label(mask)
    if not count:
        return mask
    sizes = np.bincount(lab.ravel())
    sizes[0] = 0
    return lab == sizes.argmax()


def make_clean_plate(src):
    # Frames 51--96 are essentially static (subpixel render noise only).  The
    # temporal median removes that noise and reconstructs exposed background.
    return np.median(src[50:96].astype(np.float32), axis=0).astype(np.uint8)


def main():
    src = np.load(CASE_DIR / \"source_pixels.npz\")[\"rgb\"]
    assert src.shape == (96, 144, 256, 3) and src.dtype == np.uint8
    out = src.copy()
    plate = make_clean_plate(src)

    # At frame 21 the red head occupies x=125..138, y=67..75.  Compared to the
    # clean plate, color distance isolates head + dark handle/edge.  Restrict
    # detection to its local region so unrelated red set dressing is untouched.
    frame = src[20].astype(np.int16)
    diff = np.max(np.abs(frame - plate.astype(np.int16)), axis=2)
    red = ((frame[..., 0] - frame[..., 1] > 35) &
           (frame[..., 0] - frame[..., 2] > 25) &
           (frame[..., 0] > 110))
    yy, xx = np.indices(diff.shape)
    seed = (diff > 9) & ((red) | (diff > 24))
    seed &= (xx >= 121) & (xx <= 142) & (yy >= 63) & (yy <= 79)
    seed = largest_component(seed)
    seed = binary_fill_holes(binary_dilation(seed, iterations=2))

    # difference, but keep a tight union to avoid changing unaffected content.

    # instant: no mallet means no later collision, so the struck block stays in
    # its pre-impact pose.  Freeze only the measured mallet envelope and use the

import numpy as np
from scipy.ndimage import binary_dilation, binary_fill_holes, label
from artifact_io import save

CASE_DIR = Path(__file__).resolve().parent

    if not n:
        return mask
    sizes = np.bincount(lab.ravel()); sizes[0] = 0
    return lab == sizes.argmax()


def main():
    src = np.load(CASE_DIR / \"source_pixels.npz\")[\"rgb\"]
    assert src.shape == (96, 144, 256, 3) and src.dtype == np.uint8
    out = src.copy()
    anchor = src[START - 1]  # last factual frame, immediately before removal
    si = src.astype(np.int16)
    ai = anchor.astype(np.int16)
    yy, xx = np.indices(anchor.shape[:2])

    # Find the complete downstream interaction envelope from the maximum
    # post-intervention departure from the last factual frame.  This connected
    # component contains the moving mallet and the light block it later strikes.
    departure = np.max(np.max(np.abs(si[START:] - ai), axis=3), axis=0)
    roi = (xx >= 52) & (xx <= 143) & (yy >= 58) & (yy <= 106)
    dynamic = largest_component((departure > 10) & roi)

    # antialiasing and its thin dark edge.  A later frame exposes this region.
    exposed = np.max(np.abs(ai - si[30]), axis=2)
    origin = (exposed > 3) & (xx >= 120) & (xx <= 143) & (yy >= 62) & (yy <= 81)
    origin = binary_dilation(origin, iterations=1)

    # No mallet => no collision.  Hold the measured interaction region in its
    # exact pre-intervention state; preserve every other source pixel, including
    # the subtle static render variation outside the affected objects.
