+"""Generate the 5x-purple-mass counterfactual for this PixelWorld case.
+
+The source depicts two equal-size marbles approaching one another in a symmetric
+wooden bowl.  Their first contact is between frames 33/34.  Before contact mass
+does not alter the gravity-driven trajectory, so the source is retained exactly.
+At contact, the observed incoming velocities are used with a 5:1 elastic impulse.
+The resulting trajectories are integrated on the bowl's measured constant-slope
+sections with rolling resistance, while preserving the source's photorealistic
+marble sprites and static scene.
+"""
+from pathlib import Path
+import numpy as np
+from scipy import ndimage
+from artifact_io import save
+
+CASE = Path(__file__).resolve().parent
+N,H,W = 96,144,256
+CENTER = 127.75
+RADIUS = 7.75
+CONTACT_FRAME = 34
+
+
+def detect_centres(src):
+    """Colour-segment the purple and gold marble cores in all source frames."""
+    out = {}
+    for name in ('purple','gold'):
+        xy=[]
+        for a in src.astype(np.int16):
+            r,g,b = a.transpose(2,0,1)
+            if name == 'purple':
+                mask=(b-r>25)&(b-g>45)&(b>100)
+            else:
+                mask=(r-b>65)&(g-b>35)&(r>135)&(g>85)&(np.indices((H,W))[0]<100)
+            lab,n=ndimage.label(mask)
+            comps=[]
+            for k in range(1,n+1):
+                yy,xx=np.where(lab==k)
+                if len(xx)>=25: comps.append((len(xx),float(xx.mean()),float(yy.mean())))
+            # gold floor/wood can form giant components: choose component near y=71
+            comps=[c for c in comps if 50<c[2]<85 and 0<c[1]<W]
+            c=max(comps,key=lambda z:z[0])
+            xy.append((c[1],c[2]))
+        out[name]=np.asarray(xy)
+    return out
+
+
+def simulate(centres):
+    """One-dimensional rigid-body dynamics in screen x coordinates."""
+    px=np.empty(N); gx=np.empty(N)
+    px[:CONTACT_FRAME]=centres['purple'][:CONTACT_FRAME,0]
+    gx[:CONTACT_FRAME]=centres['gold'][:CONTACT_FRAME,0]
+
+    # At the factual first-contact pose.  Measured incoming speeds from the last
+    # smooth source increments, then a restitution matching the source collision.
+    p=float(centres['purple'][CONTACT_FRAME,0])
+    g=float(centres['gold'][CONTACT_FRAME,0])
+    up=float(np.mean(np.diff(centres['purple'][29:33,0])))
+    ug=float(np.mean(np.diff(centres['gold'][29:33,0])))
+    restitution=.63
+    mp,mg=5.0,1.0
+    vp=((mp-restitution*mg)*up+(1+restitution)*mg*ug)/(mp+mg)
+    vg=((mg-restitution*mp)*ug+(1+restitution)*mp*up)/(mp+mg)
+
+    # Source free-flight curvature measures acceleration ~0.235 px/frame^2.
+    # A slightly dissipative bowl model reproduces the observed settling while
+    # allowing the much stronger 5:1 impulse to affect both marbles naturally.
+    gravity=.47
+    rolling=.235
+    dt=.05
+    settled=False
+    for frame in range(CONTACT_FRAME,N):
+        px[frame],gx[frame]=p,g
+        for _ in range(20):
+            # If both have reached the centre in enduring contact, stay stacked
+            # against each other rather than numerical micro-bouncing.
+            if settled:
+                p=CENTER+RADIUS; g=CENTER-RADIUS; vp=vg=0.0
+                continue
+            vals=[[p,vp],[g,vg]]
+            for z in vals:
+                pos,vel=z
+                downhill=1.0 if pos<CENTER else -1.0
+                if abs(vel)>.035:
+                    acc=gravity*downhill-rolling*np.sign(vel)
+                else:
+                    acc=(gravity-rolling)*downhill
+                vel+=acc*dt; pos+=vel*dt
+                z[:]=pos,vel
+            (p,vp),(g,vg)=vals
+            # Purple remains the right-hand marble. Resolve only closing contact.
+            if p-g < 2*RADIUS and vg>vp:
+                overlap=2*RADIUS-(p-g)
+                p += overlap*mg/(mp+mg); g -= overlap*mp/(mp+mg)
+                oldp,oldg=vp,vg
+                vp=((mp-restitution*mg)*oldp+(1+restitution)*mg*oldg)/(mp+mg)
+                vg=((mg-restitution*mp)*oldg+(1+restitution)*mp*oldp)/(mp+mg)
+            if frame>82 and abs(p-(CENTER+RADIUS))<.25 and abs(g-(CENTER-RADIUS))<.25 and abs(vp)<.2 and abs(vg)<.2:
+                settled=True
+    return px,gx
+
+
+def sprite_rgba(frame,cx,cy,kind):
+    """Extract an antialiased 23x23 marble sprite from a source frame."""
+    r=12; x0=int(round(cx))-r; y0=int(round(cy))-r
+    patch=frame[max(0,y0):min(H,y0+2*r+1),max(0,x0):min(W,x0+2*r+1)].copy()
+    yy,xx=np.mgrid[:patch.shape[0],:patch.shape[1]]
+    local_cx=cx-max(0,x0); local_cy=cy-max(0,y0)
+    dist=np.sqrt((xx-local_cx)**2+(yy-local_cy)**2)
+    # The visible sphere has radius about 6; include its soft rim but not track.
+    alpha=np.clip(7.1-dist,0,1)
+    rr,gg,bb=patch.astype(np.int16).transpose(2,0,1)
+    if kind=='purple': chrom=(bb-rr>10)&(bb-gg>15)
+    else: chrom=(rr-bb>25)&(gg-bb>10)
+    chrom=ndimage.binary_dilation(chrom,iterations=1)
+    alpha*=chrom
+    return patch,alpha.astype(np.float32),x0,y0
+
+
+def paste_sprite(dst, patch, alpha, x, y):
+    """Subpixel-aware alpha paste (sprite is shifted via bilinear interpolation)."""
+    ix=int(np.floor(x)); iy=int(np.floor(y)); fx=x-ix; fy=y-iy
+    p=np.empty_like(patch,dtype=np.float32)
+    a=np.empty_like(alpha,dtype=np.float32)
+    for c in range(3): p[...,c]=ndimage.shift(patch[...,c].astype(float),(fy,fx),order=1,mode='nearest')
+    a[:]=ndimage.shift(alpha,(fy,fx),order=1,mode='constant',cval=0)
+    h,w=a.shape; x1=max(0,ix);y1=max(0,iy);x2=min(W,ix+w);y2=min(H,iy+h)
+    if x2<=x1 or y2<=y1:return
+    sx1=x1-ix;sy1=y1-iy;sx2=sx1+x2-x1;sy2=sy1+y2-y1
+    aa=a[sy1:sy2,sx1:sx2,None]
+    dst[y1:y2,x1:x2]=np.clip(dst[y1:y2,x1:x2]*(1-aa)+p[sy1:sy2,sx1:sx2]*aa,0,255)
+
+
+def make_video():
+    src=np.load(CASE/'source_pixels.npz')['rgb']
+    centres=detect_centres(src)
+    px,gx=simulate(centres)
+    out=src.copy()
+
+    # A late source frame has static lighting and no moving marbles at their old
+    # pre-collision locations. Use it as a clean bowl plate after intervention.
+    plate=src[80].copy()
+    # Source sprite frames 32 and 33 show both marbles cleanly near track centre.
+    ps=sprite_rgba(src[32],*centres['purple'][32],'purple')
+    gs=sprite_rgba(src[32],*centres['gold'][32],'gold')
+    p_ref=centres['purple'][32]; g_ref=centres['gold'][32]
+
+    # Keep every pre-contact pixel factual. From first altered collision onward,
+    # composite physically relocated source marbles onto the untouched scene.
+    for t in range(CONTACT_FRAME,N):
+        f=plate.astype(np.float32).copy()
+        # marbles follow the same track height inferred from the source approach
+        py=72.69-.13932*abs(px[t]-CENTER)
+        gy=72.69-.13932*abs(gx[t]-CENTER)
+        # Draw left marble first; contact occlusion is then consistent.
+        patch,a,x0,y0=gs
+        paste_sprite(f,patch,a,x0+(gx[t]-g_ref[0]),y0+(gy-g_ref[1]))
+        patch,a,x0,y0=ps
+        paste_sprite(f,patch,a,x0+(px[t]-p_ref[0]),y0+(py-p_ref[1]))
+        out[t]=np.rint(f).astype(np.uint8)
+    return out
+
+
+if __name__=='__main__':
+    frames=make_video()
+    save(frames,CASE)

def simulate(centres):
    \"\"\"One-dimensional rigid-body dynamics in screen x coordinates.\"\"\"
    px=np.empty(N); gx=np.empty(N)
    px[:CONTACT_FRAME]=centres['purple'][:CONTACT_FRAME,0]
    gx[:CONTACT_FRAME]=centres['gold'][:CONTACT_FRAME,0]

    # smooth source increments, then a restitution matching the source collision.
    p=float(centres['purple'][CONTACT_FRAME,0])
    g=float(centres['gold'][CONTACT_FRAME,0])

def sprite_rgba(frame,cx,cy,kind):
    """Extract an antialiased 23x23 marble sprite from a source frame."""
    r=12; x0=int(round(cx))-r; y0=int(round(cy))-r
    patch=frame[max(0,y0):min(H,y0+2*r+1),max(0,x0):min(W,x0+2*r+1)].copy()
    yy,xx=np.mgrid[:patch.shape[0],:patch.shape[1]]
    local_cx=cx-max(0,x0); local_cy=cy-max(0,y0)
    dist=np.sqrt((xx-local_cx)**2+(yy-local_cy)**2)
    # The visible sphere has radius about 6; include its soft rim but not track.
    alpha=np.clip(7.1-dist,0,1)

    \"\"\"Subpixel-aware alpha paste (sprite is shifted via bilinear interpolation).\"\"\"
    ix=int(np.floor(x)); iy=int(np.floor(y)); fx=x-ix; fy=y-iy
    p=np.empty_like(patch,dtype=np.float32)
    a=np.empty_like(alpha,dtype=np.float32)
    for c in range(3): p[...,c]=ndimage.shift(patch[...,c].astype(float),(fy,fx),order=1,mode='nearest')

    h,w=a.shape; x1=max(0,ix);y1=max(0,iy);x2=min(W,ix+w);y2=min(H,iy+h)
    if x2<=x1 or y2<=y1:return
    sx1=x1-ix;sy1=y1-iy;sx2=sx1+x2-x1;sy2=sy1+y2-y1
    aa=a[sy1:sy2,sx1:sx2,None]
    dst[y1:y2,x1:x2]=np.clip(dst[y1:y2,x1:x2]*(1-aa)+p[sy1:sy2,sx1:sx2]*aa,0,255)
