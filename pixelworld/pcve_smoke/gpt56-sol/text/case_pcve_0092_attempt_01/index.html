+def inpaint_stationary_ball(background):
+    """Replace the cue ball left in the temporal median with local felt texture."""
+    out = background.astype(np.float64).copy()
+    yy, xx = np.mgrid[0:out.shape[0], 0:out.shape[1]]
+    cx, cy = 126.2, 71.1
+    r = np.sqrt(((xx-cx)/1.0)**2 + ((yy-cy)/1.0)**2)
+    sample = (r >= 9.0) & (r <= 15.0)
+    # Reject non-felt outliers, then fit a gently varying quadratic surface.
+    pix = out[sample]
+    felt = (pix[:, 1] > pix[:, 0] + 25) & (pix[:, 1] > pix[:, 2] + 15)
+    sy, sx = np.where(sample)
+    sx, sy, pix = sx[felt], sy[felt], pix[felt]
+    X = np.c_[np.ones(len(sx)), sx-cx, sy-cy,
+              (sx-cx)**2, (sx-cx)*(sy-cy), (sy-cy)**2]
+    target = r < 9.5
+    ty, tx = np.where(target)
+    TX = np.c_[np.ones(len(tx)), tx-cx, ty-cy,
+               (tx-cx)**2, (tx-cx)*(ty-cy), (ty-cy)**2]
+    for ch in range(3):
+        coef = np.linalg.lstsq(X, pix[:, ch], rcond=None)[0]
+        out[ty, tx, ch] = TX @ coef
+    # Add back a tiny deterministic felt grain sampled from the nearby clean ring.
+    residual = pix - X @ np.linalg.lstsq(X, pix, rcond=None)[0]
+    if len(residual):
+        idx = ((tx * 17 + ty * 29) % len(residual)).astype(int)
+        out[ty, tx] += 0.28 * residual[idx]
+    return np.clip(out, 0, 255)

+def detect_target_track(frames, background):
+    """Track the black target from source frame 17 through the bank shot."""
+    tracks = np.zeros((len(frames), 2), np.float64)
+    tracks[:17] = (126.2, 71.2)
+    previous = np.array([134.0, 67.0])
+    velocity = np.array([1.7, -1.0])
+    for t in range(17, len(frames)):
+        difference = np.max(np.abs(frames[t].astype(float)-background), axis=2)
+        yy, xx = np.mgrid[:difference.shape[0], :difference.shape[1]]
+        predicted = previous + velocity
+        search = ((xx-predicted[0])**2 + (yy-predicted[1])**2) < 11.0**2
+        mask = (difference > 8.0) & search & (xx > 118) & (xx < 181) & (yy > 37) & (yy < 82)
+        labels, count = ndimage.label(mask)
+        choices = []
+        for k in range(1, count+1):
+            yk, xk = np.where(labels == k)
+            if len(xk) < 5:
+                continue
+            center = np.array([xk.mean(), yk.mean()])
+            distance = np.linalg.norm(center-predicted)
+            choices.append((len(xk)-1.2*distance, center))
+        if choices:
+            center = max(choices, key=lambda q:q[0])[1]
+        else:
+            center = predicted
+        tracks[t] = center
+        new_velocity = center-previous
+        velocity = 0.55*velocity + 0.45*new_velocity
+        previous = center
+    return tracks

+def feather_clean(image, background, cx, cy, rx, ry):
+    yy, xx = np.mgrid[:image.shape[0], :image.shape[1]]
+    q = np.sqrt(((xx-cx)/rx)**2 + ((yy-cy)/ry)**2)
+    alpha = np.clip((1.08-q)/0.18, 0.0, 1.0)[...,None]

+def add_residual_sprite(image, source, background, source_center, dest_center, radius):
+    """Move a rendered source sprite as a premultiplied RGB residual."""

+    h, w = image.shape[:2]

+    Y, X = np.mgrid[y0:y1, x0:x1]

+    sa = np.clip(1.0-sq, 0, 1)**1.35 * 0.29

+    light = np.array([-0.44,-0.58,0.68]); light /= np.linalg.norm(light)

+    radius = 5.15
+    for iy in range(S):

+            rgb[numeral|foot] = np.array([32.,27.,18.])

+            cov += inside[...,None]
+    cov /= S*S
+    rgb = acc/(S*S)
+    patch = patch*(1-cov) + rgb
+    image[y0:y1,x0:x1] = np.clip(patch,0,255)

    return inpaint_stationary_ball(median)


def cue_centers():
    # Centers of the projected cue ball in source frames 0..13, measured from RGB.
    return np.array([
        [85.3,95.7],[89.0,93.3],[92.4,91.2],[95.7,89.5],
        [99.0,87.6],[102.2,85.7],[105.1,83.8],[107.7,82.3],
        [110.4,80.6],[113.1,78.9],[115.8,77.6],[118.4,75.0],
        [120.8,73.3],[123.3,71.8]
    ], dtype=np.float64)


def detect_target_track(frames, background):
    \"\"\"Track the black target from source frame 17 through the bank shot.\"\"\"
    tracks = np.zeros((len(frames), 2), np.float64)
    tracks[:17] = (126.2, 71.2)
    previous = np.array([134.0, 67.0])
    velocity = np.array([1.7, -1.0])
    for t in range(17, len(frames)):
        difference = np.max(np.abs(frames[t].astype(float)-background), axis=2)
        yy, xx = np.mgrid[:difference.shape[0], :difference.shape[1]]
        predicted = previous + velocity
        search = ((xx-predicted[0])**2 + (yy-predicted[1])**2) < 11.0**2
        mask = (difference > 8.0) & search & (xx > 118) & (xx < 181) & (yy > 37) & (yy < 82)
        labels, count = ndimage.label(mask)
        choices = []
        for k in range(1, count+1):
            yk, xk = np.where(labels == k)
            if len(xk) < 5:
                continue
            center = np.array([xk.mean(), yk.mean()])
            distance = np.linalg.norm(center-predicted)
            choices.append((len(xk)-1.2*distance, center))
        if choices:
            center = max(choices, key=lambda q:q[0])[1]
        else:
            center = predicted
        tracks[t] = center
        new_velocity = center-previous
        velocity = 0.55*velocity + 0.45*new_velocity
        previous = center
    return tracks


def feather_clean(image, background, cx, cy, rx, ry):
    yy, xx = np.mgrid[:image.shape[0], :image.shape[1]]
    q = np.sqrt(((xx-cx)/rx)**2 + ((yy-cy)/ry)**2)
    alpha = np.clip((1.08-q)/0.18, 0.0, 1.0)[...,None]
    image[:] = image*(1-alpha) + background*alpha


def clean_source_balls(image, background, t, target_track, cues):
    # Source cue: moving through frame 13, then stationary at the old target spot.
    cc = cues[t] if t < len(cues) else np.array([125.2, 70.8])
    feather_clean(image, background, cc[0], cc[1], 8.0, 8.2)
    # Source target: stationary/overlapping at first, then follows tracked bank shot.
    tc = np.array([126.2,71.2]) if t < 17 else target_track[t]
    rad = 8.0 if t < 20 else 6.4
    feather_clean(image, background, tc[0], tc[1], rad, rad)
    # Collision frames contain two fused silhouettes; cover their connecting sliver.
    if 14 <= t <= 18:
        feather_clean(image, background, 130.0+(t-15)*1.4, 69.0-(t-15)*0.8, 7.0, 7.0)


def add_residual_sprite(image, source, background, source_center, dest_center, radius):
    \"\"\"Move a rendered source sprite as a premultiplied RGB residual.\"\"\"
    h, w = image.shape[:2]
    yy, xx = np.mgrid[:h, :w]
    radial = np.sqrt((xx-source_center[0])**2 + (yy-source_center[1])**2)
    gate = np.clip((radius+0.7-radial)/1.0, 0.0, 1.0)[...,None]
    residual = (source.astype(float)-background)*gate
    shift = (dest_center[1]-source_center[1], dest_center[0]-source_center[0], 0)
    moved = ndimage.shift(residual, shift=shift, order=1, mode='constant', cval=0.0,
                          prefilter=False)
    image[:] = np.clip(image + moved, 0, 255)


def render_yellow_one(image, center):
    \"\"\"Numerically render an antialiased yellow one-ball and contact shadow.\"\"\"
    h, w = image.shape[:2]
    cx, cy = center
    S = 5  # supersampling
    x0, x1 = max(0,int(np.floor(cx-8))), min(w,int(np.ceil(cx+9)))
    y0, y1 = max(0,int(np.floor(cy-7))), min(h,int(np.ceil(cy+9)))
    patch = image[y0:y1,x0:x1].astype(float)
    acc = np.zeros_like(patch)
    cov = np.zeros(patch.shape[:2]+(1,), float)

    # Soft, compact table-contact shadow, drawn before the sphere.
    Y, X = np.mgrid[y0:y1, x0:x1]
    sq = ((X-(cx+1.25))/6.0)**2 + ((Y-(cy+3.0))/2.5)**2
    sa = np.clip(1.0-sq, 0, 1)**1.35 * 0.29
    patch *= (1.0-sa[...,None])

    light = np.array([-0.44,-0.58,0.68]); light /= np.linalg.norm(light)
    base = np.array([238.0,184.0,18.0])
    radius = 5.15
    for iy in range(S):
        for ix in range(S):
            xs = X + (ix+0.5)/S
            ys = Y + (iy+0.5)/S
            dx, dy = (xs-cx)/radius, (ys-cy)/radius
            rr = dx*dx+dy*dy
            inside = rr <= 1.0
            z = np.sqrt(np.maximum(0.0,1.0-rr))
            diffuse = np.maximum(0.0, dx*light[0]+dy*light[1]+z*light[2])
            shade = 0.43 + 0.57*diffuse
            rgb = base[None,None,:]*shade[...,None]
            # Small studio-light glint on the upper-left hemisphere.
            glint = np.exp(-((dx+0.42)**2+(dy+0.48)**2)/0.035)*0.48
            rgb = rgb*(1-glint[...,None]) + 245.0*glint[...,None]
            # White number medallion facing the camera.
            med = ((xs-(cx+0.45))**2 + (ys-(cy+0.05))**2) <= 1.62**2
            rgb[med] = rgb[med]*0.12 + np.array([239.,237.,220.])*0.88
            # A compact but legible numeral 1.
            numeral = med & (np.abs(xs-(cx+0.50)) < 0.31) & (ys > cy-0.95) & (ys < cy+1.03)
            foot = med & (np.abs(ys-(cy+0.92)) < 0.25) & (xs > cx-0.15) & (xs < cx+1.05)
            rgb[numeral|foot] = np.array([32.,27.,18.])
            acc += rgb*inside[...,None]
            cov += inside[...,None]
    cov /= S*S
    rgb = acc/(S*S)
    patch = patch*(1-cov) + rgb
    image[y0:y1,x0:x1] = np.clip(patch,0,255)


def generate():
    frames = np.load(ROOT/'source_pixels.npz')['rgb']
    background = felt_background(frames)
    tracks = detect_target_track(frames, background)
    cues = cue_centers()
    output = np.empty_like(frames)

    # Frame-1 intervention geometry: 1/4 of the target-to-cue segment.
    cue0 = np.array([85.3,95.7])
    target0 = np.array([126.2,71.2])
    yellow_initial = 0.75*target0 + 0.25*cue0
    yellow_rest = yellow_initial + 1.55*(target0-yellow_initial)/np.linalg.norm(target0-yellow_initial)

    for t in range(len(frames)):
        if t <= 6:
            # Before first contact, source motion is factual; only insert the ball.
            canvas = frames[t].astype(float).copy()
            render_yellow_one(canvas, yellow_initial)
        else:
            canvas = frames[t].astype(float).copy()
            clean_source_balls(canvas, background, t, tracks, cues)

            # The black target receives the impulse through the touching 1-ball.
            if t == 7:
                desired_target = np.array([128.7,69.8])
                add_residual_sprite(canvas, frames[0], background,
                                    np.array([126.2,71.2]), desired_target, 7.0)
            elif t == 8:
                desired_target = np.array([131.5,68.2])
                add_residual_sprite(canvas, frames[0], background,
                                    np.array([126.2,71.2]), desired_target, 7.0)
            else:
                source_t = min(95, t+8)
                source_center = tracks[source_t]
                if t+8 <= 95:
                    desired_target = source_center
                else:
                    velocity = np.mean(np.diff(tracks[88:96], axis=0), axis=0)
                    desired_target = tracks[95] + velocity*(t+8-95)
                add_residual_sprite(canvas, frames[source_t], background,
                                    source_center, desired_target, 5.9)

            # The one-ball crosses the small initial gap, then remains after transfer.
            if t == 7:
                yellow = 0.52*yellow_initial + 0.48*yellow_rest
            else:
                yellow = yellow_rest
            render_yellow_one(canvas, yellow)

            # Cue ball stops at its first contact point (source frame 7 pose).
            cue_stop = cues[7]
            add_residual_sprite(canvas, frames[7], background,
                                cue_stop, cue_stop, 7.8)
        output[t] = np.clip(np.rint(canvas),0,255).astype(np.uint8)
    return output


if __name__ == '__main__':
    result = generate()
    artifact_io.save(result, ROOT)
    print('saved', result.shape, result.dtype, 'to', ROOT)
