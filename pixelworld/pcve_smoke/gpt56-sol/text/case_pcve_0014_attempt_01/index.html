+"""Generate the counterfactual video for case pcve_0014.
+
+The source is a locked-off, rendered scene.  We recover a clean plate from the
+96 RGB observations, recover the wooden block as a soft delta sprite, and then
+re-composite it on a physically modified post-impact trajectory.  The impact
+is at the discontinuity between source frames 12 and 13 (zero based 11/12).
+"""
+from pathlib import Path
+import numpy as np
+from scipy.ndimage import gaussian_filter
+from artifact_io import save
+
+CASE_DIR = Path(__file__).resolve().parent
+N, H, W = 96, 144, 256
+FPS = 24.0
+
+
+def centers_from_colour(src):
+    """Robust screen centers from the block's distinctive orange face."""
+    r, g, b = [src[..., k].astype(np.int16) for k in range(3)]
+    orange = (r > 175) & (g > 55) & (g < 118) & (b > 20) & (b < 90) & ((r-g) > 66)
+    out = np.empty((N, 2), np.float32)
+    for i in range(N):
+        yy, xx = np.where(orange[i])
+        # Medians are insensitive to small highlights and antialiasing fragments.
+        out[i] = (np.median(xx), np.median(yy))
+    return out
+
+
+def clean_plate(src, centers):
+    """Temporal robust plate, excluding a generous ellipse around the block.
+
+    The exclusion also removes its contact shadow.  A median of all valid
+    observations preserves the scene's fine texture while rejecting the one
+    moving rigid body.  Every pixel has dozens of valid observations.
+    """
+    yy, xx = np.mgrid[:H, :W]
+    stack = src.astype(np.float32).copy()
+    valid = np.ones((N, H, W), bool)
+    for i, (cx, cy) in enumerate(centers):
+        # Actual silhouette is about 49x67 at its largest.  The larger ellipse
+        # additionally removes the soft shadow trailing down the ramp.
+        valid[i] = (((xx-cx)/37.0)**2 + ((yy-(cy+11.0))/46.0)**2) > 1.0
+    stack[~valid] = np.nan
+    plate = np.nanmedian(stack, axis=0)
+    # There are no expected holes, but use the all-frame median defensively.
+    fallback = np.median(src.astype(np.float32), axis=0)
+    plate = np.where(np.isfinite(plate), plate, fallback)
+    return plate
+
+
+def tone_plate(plate, frame, center):
+    """Match the recovered plate to this frame's tiny renderer/codec drift."""
+    yy, xx = np.mgrid[:H, :W]
+    cx, cy = center
+    safe = (((xx-cx)/52.0)**2 + ((yy-(cy+10))/60.0)**2) > 1.0
+    # Fit RGB affine transform (cross-channel terms preserve warm lighting).
+    p = plate[safe][::7]
+    q = frame.astype(np.float32)[safe][::7]
+    A = np.c_[p, np.ones(len(p), np.float32)]
+    coef, *_ = np.linalg.lstsq(A, q, rcond=None)
+    fitted = np.c_[plate.reshape(-1,3), np.ones(H*W)] @ coef
+    return fitted.reshape(H,W,3).astype(np.float32)
+
+
+def sprites(src, plate, centers):
+    """Recover soft alpha and RGB for source frames 0..12."""
+    yy, xx = np.mgrid[:H, :W]
+    result = []
+    for i in range(13):
+        cx, cy = centers[i]
+        bg = tone_plate(plate, src[i], centers[i])
+        delta = np.sqrt(np.mean((src[i].astype(np.float32)-bg)**2, axis=2))
+        # Compact block+shadow support.  The asymmetric y support is observed
+        # from source differences and prevents any unrelated ramp edge leaking.
+        support = (np.abs(xx-cx) < 34) & (yy > cy-18) & (yy < cy+48)
+        # A very low threshold keeps dark grain on the wood; the sigmoid and
+        # slight blur produce naturally antialiased edges at subpixel shifts.
+        alpha = np.clip((delta-1.6)/5.0, 0, 1) * support
+        alpha = gaussian_filter(alpha.astype(np.float32), 0.55)
+        alpha = np.clip(alpha, 0, 1)
+        x0 = max(0, int(np.floor(cx-34))); x1 = min(W, int(np.ceil(cx+34)))
+        y0 = max(0, int(np.floor(cy-19))); y1 = min(H, int(np.ceil(cy+49)))
+        result.append((src[i,y0:y1,x0:x1].copy(), alpha[y0:y1,x0:x1].copy(),
+                       np.array([cx-x0, cy-y0], np.float32)))
+    return result
+
+
+def bilinear_sample(arr, x, y):
+    """Sample HxWxC/HxW array at floating source-coordinate grids."""
+    h,w = arr.shape[:2]
+    x0 = np.floor(x).astype(np.int32); y0 = np.floor(y).astype(np.int32)
+    fx = x-x0; fy=y-y0
+    x0=np.clip(x0,0,w-1); y0=np.clip(y0,0,h-1)
+    x1=np.clip(x0+1,0,w-1); y1=np.clip(y0+1,0,h-1)
+    if arr.ndim==3:
+        fx=fx[...,None];fy=fy[...,None]
+    return ((1-fx)*(1-fy)*arr[y0,x0] + fx*(1-fy)*arr[y0,x1] +
+            (1-fx)*fy*arr[y1,x0] + fx*fy*arr[y1,x1])
+
+
+def composite_sprite(canvas, sprite, alpha, anchor, center):
+    """Subpixel translate a sprite onto canvas."""
+    sh,sw=alpha.shape
+    left=center[0]-anchor[0]; top=center[1]-anchor[1]
+    x0=max(0,int(np.floor(left))-1); y0=max(0,int(np.floor(top))-1)
+    x1=min(W,int(np.ceil(left+sw))+1); y1=min(H,int(np.ceil(top+sh))+1)
+    if x0>=x1 or y0>=y1: return
+    gy,gx=np.mgrid[y0:y1,x0:x1]
+    sx=gx-left; sy=gy-top
+    inside=(sx>=0)&(sx<=sw-1)&(sy>=0)&(sy<=sh-1)
+    a=bilinear_sample(alpha,sx,sy)*inside
+    rgb=bilinear_sample(sprite.astype(np.float32),sx,sy)
+    canvas[y0:y1,x0:x1] = canvas[y0:y1,x0:x1]*(1-a[...,None])+rgb*a[...,None]
+
+
+def make_video(src):
+    centers=centers_from_colour(src)
+    plate=clean_plate(src,centers)
+    spr=sprites(src,plate,centers)
+    out=np.empty_like(src)
+
+    # The factual history before the collision is untouched, exactly as required.
+    out[:12]=src[:12]
+    contact=centers[12].astype(np.float64)
+    # Restitution controls the separating normal speed.  The counterfactual
+    # value (1.8x baseline) gives a visibly stronger up-ramp rebound.  Gravity's
+    # projection down the ramp then decelerates it smoothly.
+    v0=-3.05                 # px/frame along ramp immediately after impact
+    accel=0.055              # gravity projection, px/frame^2
+    ramp_dir=np.array([1.0,0.285],np.float64)
+
+    for t in range(12,N):
+        n=float(t-12)
+        travel=v0*n+0.5*accel*n*n
+        center=contact+ramp_dir*travel
+        # Use source appearance at the corresponding ramp position, retaining
+        # true wood texture, changing perspective and source antialiasing.
+        appearance=np.clip((center[0]-centers[0,0]) /
+                           max(1e-6,centers[12,0]-centers[0,0])*12.0, 0, 12)
+        j=int(np.floor(appearance)); k=min(12,j+1); f=appearance-j
+        # Pick the nearer true sprite rather than cross-dissolving textures.
+        si=k if f>=.5 else j
+        rgb,a,anchor=spr[si]
+        # Start from a clean but frame-toned plate, thus removing the factual
+        # low-restitution trajectory and all of its old shadow.
+        base=tone_plate(plate,src[t],centers[t])
+        composite_sprite(base,rgb,a,anchor,center)
+        out[t]=np.clip(np.rint(base),0,255).astype(np.uint8)
+    return out
+
+
+def main():
+    src=np.load(CASE_DIR/'source_pixels.npz')['rgb']
+    assert src.shape==(N,H,W,3) and src.dtype==np.uint8
+    frames=make_video(src)
+    save(frames,CASE_DIR)
+
+if __name__=='__main__':
+    main()

from pathlib import Path
import numpy as np
from scipy.ndimage import gaussian_filter
from artifact_io import save

CASE_DIR = Path(__file__).resolve().parent
N, H, W = 96, 144, 256
FPS = 24.0


def centers_from_colour(src):
    \"\"\"Robust screen centers from the block's distinctive orange face.\"\"\"
    r, g, b = [src[..., k].astype(np.int16) for k in range(3)]
    orange = (r > 175) & (g > 55) & (g < 118) & (b > 20) & (b < 90) & ((r-g) > 66)
    out = np.empty((N, 2), np.float32)
    for i in range(N):
        yy, xx = np.where(orange[i])
        # Medians are insensitive to small highlights and antialiasing fragments.
        out[i] = (np.median(xx), np.median(yy))
    return out


def clean_plate(src, centers):
    \"\"\"Temporal robust plate, excluding a generous ellipse around the block.

    The exclusion also removes its contact shadow.  A median of all valid
    observations preserves the scene's fine texture while rejecting the one
    moving rigid body.  Every pixel has dozens of valid observations.
    \"\"\"
    yy, xx = np.mgrid[:H, :W]
    stack = src.astype(np.float32).copy()
    valid = np.ones((N, H, W), bool)
    for i, (cx, cy) in enumerate(centers):
        # Actual silhouette is about 49x67 at its largest.  The larger ellipse
        # additionally removes the soft shadow trailing down the ramp.
        valid[i] = (((xx-cx)/37.0)**2 + ((yy-(cy+11.0))/46.0)**2) > 1.0
    stack[~valid] = np.nan
    plate = np.nanmedian(stack, axis=0)
    # There are no expected holes, but use the all-frame median defensively.
    fallback = np.median(src.astype(np.float32), axis=0)
    plate = np.where(np.isfinite(plate), plate, fallback)
    return plate


def tone_plate(plate, frame, center):
    \"\"\"Match the recovered plate to this frame's tiny renderer/codec drift.\"\"\"
    yy, xx = np.mgrid[:H, :W]
    cx, cy = center
    safe = (((xx-cx)/52.0)**2 + ((yy-(cy+10))/60.0)**2) > 1.0
    # Fit RGB affine transform (cross-channel terms preserve warm lighting).
    p = plate[safe][::7]
    q = frame.astype(np.float32)[safe][::7]
    A = np.c_[p, np.ones(len(p), np.float32)]
    coef, *_ = np.linalg.lstsq(A, q, rcond=None)
    fitted = np.c_[plate.reshape(-1,3), np.ones(H*W)] @ coef
    return fitted.reshape(H,W,3).astype(np.float32)


def sprites(src, plate, centers):
    \"\"\"Recover soft alpha and RGB for source frames 0..12.\"\"\"
    yy, xx = np.mgrid[:H, :W]
    result = []
    for i in range(13):
        cx, cy = centers[i]
        bg = tone_plate(plate, src[i], centers[i])
        delta = np.sqrt(np.mean((src[i].astype(np.float32)-bg)**2, axis=2))
        # Compact block+shadow support.  The asymmetric y support is observed
        # from source differences and prevents any unrelated ramp edge leaking.
        support = (np.abs(xx-cx) < 34) & (yy > cy-18) & (yy < cy+48)
        # A very low threshold keeps dark grain on the wood; the sigmoid and
        # slight blur produce naturally antialiased edges at subpixel shifts.
        alpha = np.clip((delta-1.6)/5.0, 0, 1) * support
        alpha = gaussian_filter(alpha.astype(np.float32), 0.55)
        alpha = np.clip(alpha, 0, 1)
        x0 = max(0, int(np.floor(cx-34))); x1 = min(W, int(np.ceil(cx+34)))
        y0 = max(0, int(np.floor(cy-19))); y1 = min(H, int(np.ceil(cy+49)))
        result.append((src[i,y0:y1,x0:x1].copy(), alpha[y0:y1,x0:x1].copy(),
                       np.array([cx-x0, cy-y0], np.float32)))
    return result


def bilinear_sample(arr, x, y):
    \"\"\"Sample HxWxC/HxW array at floating source-coordinate grids.\"\"\"
    h,w = arr.shape[:2]
    x0 = np.floor(x).astype(np.int32); y0 = np.floor(y).astype(np.int32)
    fx = x-x0; fy=y-y0
    x0=np.clip(x0,0,w-1); y0=np.clip(y0,0,h-1)
    x1=np.clip(x0+1,0,w-1); y1=np.clip(y0+1,0,h-1)
    if arr.ndim==3:
        fx=fx[...,None];fy=fy[...,None]
    return ((1-fx)*(1-fy)*arr[y0,x0] + fx*(1-fy)*arr[y0,x1] +
            (1-fx)*fy*arr[y1,x0] + fx*fy*arr[y1,x1])


def composite_sprite(canvas, sprite, alpha, anchor, center):
    \"\"\"Subpixel translate a sprite onto canvas.\"\"\"
    sh,sw=alpha.shape
    left=center[0]-anchor[0]; top=center[1]-anchor[1]
    x0=max(0,int(np.floor(left))-1); y0=max(0,int(np.floor(top))-1)
    x1=min(W,int(np.ceil(left+sw))+1); y1=min(H,int(np.ceil(top+sh))+1)
    if x0>=x1 or y0>=y1: return
    gy,gx=np.mgrid[y0:y1,x0:x1]
    sx=gx-left; sy=gy-top
    inside=(sx>=0)&(sx<=sw-1)&(sy>=0)&(sy<=sh-1)
    a=bilinear_sample(alpha,sx,sy)*inside
    rgb=bilinear_sample(sprite.astype(np.float32),sx,sy)
    canvas[y0:y1,x0:x1] = canvas[y0:y1,x0:x1]*(1-a[...,None])+rgb*a[...,None]


def make_video(src):
    centers=centers_from_colour(src)
    plate=clean_plate(src,centers)
    spr=sprites(src,plate,centers)
    out=np.empty_like(src)

    # The factual history before the collision is untouched, exactly as required.
    out[:12]=src[:12]
    contact=centers[12].astype(np.float64)
    # Restitution controls the separating normal speed.  The counterfactual
    # value (1.8x baseline) gives a visibly stronger up-ramp rebound.  Gravity's
    # projection down the ramp then decelerates it smoothly.
    v0=-3.05                 # px/frame along ramp immediately after impact
    accel=0.055              # gravity projection, px/frame"'^2
    ramp_dir=np.array([1.0,0.285],np.float64)

    for t in range(12,N):
        n=float(t-12)
        travel=v0*n+0.5*accel*n*n
        center=contact+ramp_dir*travel
        # Use source appearance at the corresponding ramp position, retaining
        # true wood texture, changing perspective and source antialiasing.
        appearance=np.clip((center[0]-centers[0,0]) /
                           max(1e-6,centers[12,0]-centers[0,0])*12.0, 0, 12)
        j=int(np.floor(appearance)); k=min(12,j+1); f=appearance-j
        # Pick the nearer true sprite rather than cross-dissolving textures.
        si=k if f>=.5 else j
        rgb,a,anchor=spr[si]
        # Start from a clean but frame-toned plate, thus removing the factual
        # low-restitution trajectory and all of its old shadow.
        base=tone_plate(plate,src[t],centers[t])
        composite_sprite(base,rgb,a,anchor,center)
        out[t]=np.clip(np.rint(base),0,255).astype(np.uint8)
    return out


def main():
    src=np.load(CASE_DIR/'"'source_pixels.npz')['rgb']
    assert src.shape==(N,H,W,3) and src.dtype==np.uint8
    frames=make_video(src)
    save(frames,CASE_DIR)

if __name__=='__main__':
    main()
