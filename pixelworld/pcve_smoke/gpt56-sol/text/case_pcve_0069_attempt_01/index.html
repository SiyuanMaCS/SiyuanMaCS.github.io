+"""Remove the second domino and simulate the resulting stopped chain reaction.
+
+The scene is static-camera footage. Domino 1 (leftmost) is already falling at frame 1;
+removing domino 2 means it falls harmlessly into the empty gap. Dominoes 3--5,
+which only move after being struck in the source, therefore remain in their initial poses.
+"""
+from pathlib import Path
+import cv2
+import numpy as np
+import artifact_io
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def remove_domino2(frame0):
+    """Numerically fill the second-domino footprint from neighboring ground texture."""
+    # Its visible silhouette is x=101..114, y=39..101. Include antialiasing,
+    # contact shadow, and a narrow safety margin, but avoid dominoes 1 and 3.
+    mask = np.zeros(frame0.shape[:2], np.uint8)
+    cv2.rectangle(mask, (98, 35), (117, 104), 255, -1)
+    # Telea interpolation follows the scene's smooth perspective illumination and
+    # grass gradients far better than a flat fill or a copied rectangular seam.
+    bgr = cv2.cvtColor(frame0, cv2.COLOR_RGB2BGR)
+    clean = cv2.inpaint(bgr, mask, 7, cv2.INPAINT_TELEA)
+    return cv2.cvtColor(clean, cv2.COLOR_BGR2RGB)
+
+
+def first_domino_mask(frame):
+    """Segment the moving leftmost domino in the region where it cannot overlap #3."""
+    a = frame.astype(np.int16)
+    mean = a.mean(axis=2)
+    spread = a.max(axis=2) - a.min(axis=2)
+    # Domino is dark/near-neutral against green grass. Source shadows and details
+    # are included by taking the hull of the robust dark core.
+    core = ((mean < 108) & (spread < 68)).astype(np.uint8)
+    scope = np.zeros_like(core)
+    scope[31:106, 45:99] = 1
+    core &= scope
+    n, labels, stats, _ = cv2.connectedComponentsWithStats(core, 8)
+    ids = [i for i in range(1, n) if stats[i, cv2.CC_STAT_AREA] > 80]
+    mask = np.zeros_like(core)
+    if ids:
+        # Components can fragment in highlights; collect substantial parts and hull.
+        points = []
+        for i in ids:
+            yy, xx = np.where(labels == i)
+            if xx.mean() < 95:
+                points.extend(zip(xx.tolist(), yy.tolist()))
+        if points:
+            hull = cv2.convexHull(np.asarray(points, np.int32))
+            cv2.fillConvexPoly(mask, hull, 255)
+            mask = cv2.dilate(mask, np.ones((3, 3), np.uint8), iterations=1)
+            mask = cv2.GaussianBlur(mask, (3, 3), 0)
+    return mask
+
+
+def compose():
+    src = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert src.shape == (96, 144, 256, 3) and src.dtype == np.uint8
+
+    # Counterfactual background: source frame 1 has the correct original poses of
+    # dominoes 3--5 and all unaffected scenery, with only domino 2 removed.
+    base = remove_domino2(src[0])
+    out = np.repeat(base[None], 96, axis=0)
+
+    # Preserve the factual free fall of domino 1. It never reaches domino 3 across
+    # the enlarged gap, so after its source motion settles (~frame 19) it stays down.
+    # Copy only its segmented silhouette; this preserves all internal texture/shading.
+    last_layer = None
+    for t in range(96):
+        ref_t = min(t, 18)
+        m = first_domino_mask(src[ref_t])
+        alpha = (m.astype(np.float32) / 255.0)[..., None]
+        layer = src[ref_t]
+        out[t] = np.clip(out[t].astype(np.float32) * (1 - alpha) + layer * alpha,
+                         0, 255).astype(np.uint8)
+        if t == 18:
+            last_layer = out[t].copy()
+        elif t > 18:
+            # Fully stable end state: avoids re-segmenting and keeps exact continuity.
+            out[t] = last_layer
+
+    return out
+
+
+if __name__ == "__main__":
+    frames = compose()
+    artifact_io.save(frames, CASE_DIR)
+    print("saved", frames.shape, frames.dtype)

def first_domino_mask(frame):
    \"\"\"Segment the moving leftmost domino in the region where it cannot overlap #3.\"\"\"
    a = frame.astype(np.int16)
    mean = a.mean(axis=2)
    spread = a.max(axis=2) - a.min(axis=2)
    # Domino is dark/near-neutral against green grass. Source shadows and details
    # are included by taking the hull of the robust dark core.
    core = ((mean < 108) & (spread < 68)).astype(np.uint8)
    scope = np.zeros_like(core)
    scope[31:106, 45:99] = 1
    core &= scope
    n, labels, stats, _ = cv2.connectedComponentsWithStats(core, 8)
    ids = [i for i in range(1, n) if stats[i, cv2.CC_STAT_AREA] > 80]
    mask = np.zeros_like(core)
    if ids:
        # Components can fragment in highlights; collect substantial parts and hull.
        points = []
        for i in ids:
            yy, xx = np.where(labels == i)
            if xx.mean() < 95:
                points.extend(zip(xx.tolist(), yy.tolist()))
        if points:
            hull = cv2.convexHull(np.asarray(points, np.int32))
            cv2.fillConvexPoly(mask, hull, 255)
            mask = cv2.dilate(mask, np.ones((3, 3), np.uint8), iterations=1)
            mask = cv2.GaussianBlur(mask, (3, 3), 0)
    return mask


def compose():
    src = np.load(CASE_DIR / \"source_pixels.npz\")[\"rgb\"]
    assert src.shape == (96, 144, 256, 3) and src.dtype == np.uint8

    # Counterfactual background: source frame 1 has the correct original poses of
    # dominoes 3--5 and all unaffected scenery, with only domino 2 removed.
    base = remove_domino2(src[0])
    out = np.repeat(base[None], 96, axis=0)

    # Preserve the factual free fall of domino 1. It never reaches domino 3 across
    # the enlarged gap, so after its source motion settles (~frame 19) it stays down.
    # Copy only its segmented silhouette; this preserves all internal texture/shading.
    last_layer = None
    for t in range(96):
        ref_t = min(t, 18)
        m = first_domino_mask(src[ref_t])
        alpha = (m.astype(np.float32) / 255.0)[..., None]
        layer = src[ref_t]
        out[t] = np.clip(out[t].astype(np.float32) * (1 - alpha) + layer * alpha,
                         0, 255).astype(np.uint8)
        if t == 18:
            last_layer = out[t].copy()
        elif t > 18:
            # Fully stable end state: avoids re-segmenting and keeps exact continuity.
            out[t] = last_layer

    return out
