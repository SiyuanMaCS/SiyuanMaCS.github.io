+"""Generate the counterfactual video for the high-friction blue mallet.
+
+The source is a locked-off rigid-body shot.  In the factual sequence the blue
+mallet first travels down-left, then starts a three-mallet collision chain.
+With 25x friction it should retain its initial velocity at t=0 but rapidly
+decelerate and stop before the first impact.  The first four source poses are
+all safely before contact, so sampling that short trajectory with a quadratic
+time warp preserves the exact source rendering while imposing constant,
+greatly increased deceleration.  Once stopped, the last clean pre-contact pose
+is held; consequently the two downstream mallets stay in their initial poses.
+"""
+
+from pathlib import Path
+
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+FRAME_COUNT = 96
+STOP_FRAME = 6.0       # 0.25 seconds at 24 fps
+STOP_SOURCE_TIME = 3.0 # ends ~4.5 image pixels from the initial position
+
+
+def lerp_source(source: np.ndarray, sample_time: float) -> np.ndarray:
+    """Linearly sample the RGB source timeline at a fractional frame time."""
+    left = int(np.floor(sample_time))
+    right = min(left + 1, len(source) - 1)
+    weight = sample_time - left
+    if weight <= 1e-12:
+        return source[left].copy()
+    mixed = ((1.0 - weight) * source[left].astype(np.float32)
+             + weight * source[right].astype(np.float32))
+    return np.clip(np.rint(mixed), 0, 255).astype(np.uint8)
+
+
+def make_frames() -> np.ndarray:
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3) and source.dtype == np.uint8
+
+    frames = np.empty_like(source)
+    for frame_index in range(FRAME_COUNT):
+        if frame_index >= STOP_FRAME:
+            # Exact, sharp source pose once static.
+            sample_time = STOP_SOURCE_TIME
+        else:
+            # s(t) = v0*t + .5*a*t^2 with v(STOP_FRAME)=0.
+            # Choosing v0=1 source-frame/output-frame gives a stop distance of
+            # STOP_FRAME/2 = 3 source frames, exactly STOP_SOURCE_TIME.
+            t = float(frame_index)
+            sample_time = t - (t * t) / (2.0 * STOP_FRAME)
+        frames[frame_index] = lerp_source(source, sample_time)
+
+    # Frame 1 is the source's initial state, and all post-stop images are
+    # identical, ensuring there is no residual sliding or collision.
+    assert np.array_equal(frames[0], source[0])
+    assert np.array_equal(frames[6], source[3])
+    assert np.all(frames[6:] == source[3][None, ...])
+    return frames
+
+
+if __name__ == "__main__":
+    artifact_io.save(make_frames(), CASE_DIR)
+
