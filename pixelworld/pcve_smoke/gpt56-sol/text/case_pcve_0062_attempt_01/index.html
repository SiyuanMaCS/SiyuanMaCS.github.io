+"""Remove the soda cup from every source frame using numeric RGB inpainting."""
+from pathlib import Path
+import numpy as np
+import cv2
+import artifact_io
+
+CASE_DIR = Path(__file__).resolve().parent
+
+def cup_mask(frame, frame_index):
+    """Return a tight support for the soda cup tracked along its belt path."""
+    # The cup's center trajectory was measured directly from source RGB
+    # components. It moves smoothly right, then remains parked at x=214.
+    # Tracking by this trajectory avoids absorbing the red car/furniture
+    # surfaces which touch the cup chromatically around frames 20--25.
+    key_t = np.array([0, 1, 2, 3, 4, 8, 12, 16, 20, 24, 28, 32, 36, 95], float)
+    key_x = np.array([29,48,67,85,96,132,161,178,192,203,210,213,214,214], float)
+    center = float(np.interp(frame_index, key_t, key_x))
+    mask = np.zeros(frame.shape[:2], np.uint8)
+    # Complete cup: slightly flared 14px rim, tapered red body, and bottom.
+    poly = np.array([[round(center-7), 64], [round(center+7), 64],
+                     [round(center+6), 80], [round(center+5), 83],
+                     [round(center-5), 83], [round(center-6), 80]], np.int32)
+    cv2.fillPoly(mask, [poly], 255)
+    return mask
+
+def reconstruct_background(frame, mask):
+    """Fill cup region from same-frame surroundings with structure-aware rows."""
+    out = frame.copy()
+    ys, xs = np.where(mask > 0)
+    x0, x1 = xs.min(), xs.max()
+    # The scene behind the cup consists of a nearly uniform gray wall above
+    # the belt and continuous tan horizontal surfaces below it. For each row,
+    # interpolate local pixels from both sides. This preserves the scene's
+    # native illumination/noise and its strong horizontal edges.
+    for y in range(ys.min(), ys.max()+1):
+        row_x = np.where(mask[y] > 0)[0]
+        if len(row_x) == 0:
+            continue
+        a, z = row_x.min(), row_x.max()
+        left = frame[y, max(0, a-5):a]
+        right = frame[y, z+1:min(frame.shape[1], z+6)]
+        # Robust side colors suppress adjacent moving-object edge pixels.
+        lc = np.median(left, axis=0) if len(left) else np.median(right, axis=0)
+        rc = np.median(right, axis=0) if len(right) else lc
+        n = z-a+1
+        t = np.linspace(0, 1, n+2)[1:-1, None]
+        vals = lc[None]*(1-t) + rc[None]*t
+        out[y, a:z+1] = np.clip(np.rint(vals), 0, 255).astype(np.uint8)
+    # Feather just the silhouette boundary, without blurring scene edges.
+    inner = cv2.erode(mask, np.ones((3,3),np.uint8), iterations=1)
+    boundary = (mask > 0) & (inner == 0)
+    soft = cv2.GaussianBlur(out, (3,3), 0.55)
+    out[boundary] = soft[boundary]
+    return out
+
+def generate():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3) and source.dtype == np.uint8
+    result = np.empty_like(source)
+    for i, frame in enumerate(source):
+        mask = cup_mask(frame, i)
+        result[i] = reconstruct_background(frame, mask)
+    return result

\"\"\"Remove the soda cup from every source frame using numeric RGB inpainting.\"\"\"
from pathlib import Path
import numpy as np
import cv2
import artifact_io

CASE_DIR = Path(__file__).resolve().parent

def cup_mask(frame):
    \"\"\"Segment the moving red soda cup (not the red decorative furniture).\"\"\"
    im = frame.astype(np.int16)
    r, g, b = im[..., 0], im[..., 1], im[..., 2]
    red = ((r - g > 19) & (r - b > 22) & (r > 120)).astype(np.uint8)
    # Cup is the compact red component in the conveyor/object band.
    roi = np.zeros(red.shape, np.uint8)
    roi[57:85] = red[57:85]
    count, labels, stats, centroids = cv2.connectedComponentsWithStats(roi, 8)
    selected = np.zeros(red.shape, np.uint8)
    candidates = []
    for i in range(1, count):
        x, y, w, h, area = stats[i]
        cx, cy = centroids[i]
        # The cup body is 10-14 px wide, ~18 px high and sits on the belt.
        if 8 <= w <= 16 and 14 <= h <= 21 and area >= 105 and 64 <= y <= 66:
            candidates.append((area, i))
    if not candidates:
        raise RuntimeError(\"Could not locate soda cup\")
    _, idx = max(candidates)
    selected[labels == idx] = 255

    # Recover its full silhouette, including anti-aliased pale rim/edge pixels.
    # A compact trapezoidal support is much safer than color alone because the
    # cup crosses similarly colored foreground upholstery.
    x, y, w, h, _ = stats[idx]
    support = np.zeros_like(selected)
    center = centroids[idx][0]
    top = max(63, y - 1)
    bottom = min(84, y + h + 2)
    poly = np.array([[round(center-7), top], [round(center+7), top],
                     [round(center+7), bottom-2], [round(center+5), bottom],
                     [round(center-5), bottom], [round(center-7), bottom-2]], np.int32)
    cv2.fillPoly(support, [poly], 255)
    return support

def reconstruct_background(frame, mask):
    \"\"\"Fill cup region from same-frame surroundings with structure-aware rows.\"\"\"
    out = frame.copy()
    ys, xs = np.where(mask > 0)
    x0, x1 = xs.min(), xs.max()
    # The scene behind the cup consists of a nearly uniform gray wall above
    # the belt and continuous tan horizontal surfaces below it. For each row,
    # interpolate local pixels from both sides. This preserves the scene's
    # native illumination/noise and its strong horizontal edges.
    for y in range(ys.min(), ys.max()+1):
        row_x = np.where(mask[y] > 0)[0]
        if len(row_x) == 0:
            continue
        a, z = row_x.min(), row_x.max()
        left = frame[y, max(0, a-5):a]
        right = frame[y, z+1:min(frame.shape[1], z+6)]
        # Robust side colors suppress adjacent moving-object edge pixels.
        lc = np.median(left, axis=0) if len(left) else np.median(right, axis=0)
        rc = np.median(right, axis=0) if len(right) else lc
        n = z-a+1
        t = np.linspace(0, 1, n+2)[1:-1, None]
        vals = lc[None]*(1-t) + rc[None]*t
        out[y, a:z+1] = np.clip(np.rint(vals), 0, 255).astype(np.uint8)
    # Feather just the silhouette boundary, without blurring scene edges.
    inner = cv2.erode(mask, np.ones((3,3),np.uint8), iterations=1)
    boundary = (mask > 0) & (inner == 0)
    soft = cv2.GaussianBlur(out, (3,3), 0.55)
    out[boundary] = soft[boundary]
    return out

    result = np.empty_like(source)
    for i, frame in enumerate(source):
        mask = cup_mask(frame)
        result[i] = reconstruct_background(frame, mask)
    return result

    artifact_io.save(frames, CASE_DIR)
