+def largest_component(mask: np.ndarray) -> np.ndarray:
+    labels, count = ndimage.label(mask)
+    if count == 0:
+        raise RuntimeError("Could not isolate the toy car")
+    sizes = ndimage.sum(mask, labels, range(1, count + 1))
+    return labels == (int(np.argmax(sizes)) + 1)

+def paste_rgba(canvas: np.ndarray, rgb: np.ndarray, alpha: np.ndarray,
+               x: int, y: int) -> np.ndarray:
+    """Paste a small floating-point RGB/alpha sprite with clipping."""
+    out = canvas.astype(np.float32, copy=True)
+    sh, sw = alpha.shape
+    x0, y0 = max(0, x), max(0, y)
+    x1, y1 = min(W, x + sw), min(H, y + sh)
+    if x1 <= x0 or y1 <= y0:
+        return out
+    sx0, sy0 = x0 - x, y0 - y
+    sx1, sy1 = sx0 + (x1 - x0), sy0 + (y1 - y0)
+    a = alpha[sy0:sy1, sx0:sx1, None]
+    out[y0:y1, x0:x1] = (
+        rgb[sy0:sy1, sx0:sx1] * a + out[y0:y1, x0:x1] * (1.0 - a)
+    )
+    return out
+

+def build_frames() -> np.ndarray:
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    if source.shape != (N, H, W, 3) or source.dtype != np.uint8:

+

+    # untouched target at x=6..46, remains from the pre-collision source state.
+    base = sprite_frame * (1.0 - soft_mask[..., None]) + clear_frame * soft_mask[..., None]
+

+    # Arc-length fit gives about 0.945 px/frame^2.  Fourfold friction therefore
+    # gives a=3.78.  The initial fitted speed is 15.56 px/frame.  The modified

+    # of the source's ~128 px approach to the target.

+        u = (distance - path_s[j]) / (path_s[j + 1] - path_s[j])
+        return path_xy[j] * (1.0 - u) + path_xy[j + 1] * u
