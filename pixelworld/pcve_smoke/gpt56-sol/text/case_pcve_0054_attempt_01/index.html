+#!/usr/bin/env python3
+"""Generate the complete counterfactual video using only source RGB pixels.
+
+The source scene is a symmetric top-down rigid-body setup.  The blue stone is
+reconstructed from the visible blue annulus (the source contains it centred at
+(127.5, 79.5), underneath the red stone), while the small stones are composited
+from their source appearances.  Dynamics are one-dimensional equal-mass rigid
+collisions with mild inelasticity, matching the source trajectory scale.
+"""
+from pathlib import Path
+import numpy as np
+import cv2
+import artifact_io
+
+CASE_DIR = Path(__file__).resolve().parent
+N, H, W = 96, 144, 256
+
+
+def bilinear(img, x, y):
+    """Bilinearly sample an RGB image at floating coordinate arrays."""
+    h, w = img.shape[:2]
+    x = np.clip(x, 0, w - 1)
+    y = np.clip(y, 0, h - 1)
+    x0 = np.floor(x).astype(np.int32); y0 = np.floor(y).astype(np.int32)
+    x1 = np.minimum(x0 + 1, w - 1); y1 = np.minimum(y0 + 1, h - 1)
+    fx = (x - x0)[..., None]; fy = (y - y0)[..., None]
+    a = img[y0, x0].astype(np.float32) * (1 - fx) + img[y0, x1].astype(np.float32) * fx
+    b = img[y1, x0].astype(np.float32) * (1 - fx) + img[y1, x1].astype(np.float32) * fx
+    return a * (1 - fy) + b * fy
+
+
+def reconstruct_background(src):
+    """Remove the three source stones but retain the textured board/platform."""
+    # Median over time removes the moving red/yellow pucks.  The stationary blue
+    # stone is removed by radial interpolation between clean rings just outside it.
+    bg = np.median(src, axis=0).astype(np.float32)
+    yy, xx = np.indices((H, W), dtype=np.float32)
+    cx, cy = 127.5, 79.5
+    dx, dy = xx - cx, yy - cy
+    r = np.hypot(dx, dy)
+    ux = dx / np.maximum(r, 1e-4); uy = dy / np.maximum(r, 1e-4)
+
+    # The visible blue stone has a 61x28 projected ellipse.  Its underlying board
+    # is reconstructed from corresponding samples at the clean outer ring.
+    rho = np.sqrt((dx / 61.8) ** 2 + (dy / 29.0) ** 2)
+    mask = rho < 1.035
+    edge_x = cx + ux * 65.0
+    edge_y = cy + uy * 33.0
+    clean = bilinear(bg, edge_x, edge_y)
+    # Preserve local board illumination by blending two clean radial samples.
+    clean2 = bilinear(bg, cx + ux * 70.0, cy + uy * 36.0)
+    clean = 0.68 * clean + 0.32 * clean2
+    feather = np.clip((1.06 - rho) / 0.055, 0.0, 1.0)[..., None]
+    bg = bg * (1 - feather) + clean * feather
+
+    # Small residuals of source pucks lie outside the large ellipse. Restore those
+    # thin areas from their immediate vertical surroundings (the board varies slowly).
+    for x0, x1, y0, y1 in [(35, 61, 61, 75), (198, 225, 71, 88)]:
+        patch = bg[y0:y1, x0:x1]
+        top = bg[y0-3:y0, x0:x1].mean(0)
+        bot = bg[y1:y1+3, x0:x1].mean(0)
+        for j in range(y1-y0):
+            u = (j + 0.5) / (y1-y0)
+            patch[j] = top * (1-u) + bot * u
+    return np.clip(bg, 0, 255).astype(np.uint8)
+
+
+def sprite_from_frame(frame, center, rx, ry, pad=3):
+    """Cut a soft elliptical RGBA sprite out of a source RGB frame."""
+    cx, cy = center
+    x0 = int(np.floor(cx-rx-pad)); x1 = int(np.ceil(cx+rx+pad))+1
+    y0 = int(np.floor(cy-ry-pad)); y1 = int(np.ceil(cy+ry+pad))+1
+    crop = frame[y0:y1, x0:x1].copy()
+    yy, xx = np.indices(crop.shape[:2], dtype=np.float32)
+    ex = (xx + x0 - cx) / rx; ey = (yy + y0 - cy) / ry
+    rho = np.sqrt(ex*ex + ey*ey)
+    alpha = np.clip((1.10-rho)/0.14, 0, 1)
+    return crop, alpha.astype(np.float32), (cx-x0, cy-y0)
+
+
+def paste_sprite(dst, sprite, alpha, anchor, center):
+    """Subpixel bilinear sprite compositing."""
+    sh, sw = alpha.shape
+    cx, cy = center; ax, ay = anchor
+    x0 = int(np.floor(cx-ax)); y0 = int(np.floor(cy-ay))
+    fx = (cx-ax)-x0; fy = (cy-ay)-y0
+    M = np.float32([[1, 0, fx], [0, 1, fy]])
+    s = cv2.warpAffine(sprite, M, (sw, sh), flags=cv2.INTER_LINEAR,
+                       borderMode=cv2.BORDER_REPLICATE)
+    a = cv2.warpAffine(alpha, M, (sw, sh), flags=cv2.INTER_LINEAR,
+                       borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+    xa=max(0,x0); ya=max(0,y0); xb=min(W,x0+sw); yb=min(H,y0+sh)
+    if xb<=xa or yb<=ya: return
+    sx0=xa-x0; sy0=ya-y0; sx1=sx0+xb-xa; sy1=sy0+yb-ya
+    aa=a[sy0:sy1,sx0:sx1,None]
+    dst[ya:yb,xa:xb] = dst[ya:yb,xa:xb]*(1-aa)+s[sy0:sy1,sx0:sx1]*aa
+
+
+def simulate():
+    """Equal-mass, mildly inelastic one-dimensional rigid puck dynamics."""
+    # Source initial red/yellow centres, with blue placed exactly 1/4 red->yellow.
+    p = np.array([47.5, 47.5 + 0.25*(213.8-47.5), 213.8], np.float64)
+    v = np.array([1.98, 0.0, -2.10], np.float64)  # source pixels/frame
+    radii = np.array([5.5, 5.5, 5.5])
+    pos = np.empty((N,3), np.float64)
+    restitution = 0.08
+    for t in range(N):
+        pos[t] = p
+        for _ in range(10):
+            p += v/10.0
+            for i in range(2):
+                contact = radii[i]+radii[i+1]
+                gap = p[i+1]-p[i]
+                if gap < contact and v[i] > v[i+1]:
+                    overlap = contact-gap
+                    p[i] -= overlap*.5; p[i+1] += overlap*.5
+                    a,b = v[i],v[i+1]
+                    v[i]   = .5*((1-restitution)*a+(1+restitution)*b)
+                    v[i+1] = .5*((1+restitution)*a+(1-restitution)*b)
+            v *= 0.99995
+    return pos
+
+
+def make_video():
+    src = np.load(CASE_DIR/'source_pixels.npz')['rgb']
+    bg = reconstruct_background(src)
+
+    # Use unobstructed source appearances at frame 0. Blue texture is the visible
+    # large blue object itself; alpha isolates its exact projected silhouette.
+    blue, ba, banc = sprite_from_frame(src[0], (127.5,79.5), 62.0,29.0, 3)
+    red,  ra, ranc = sprite_from_frame(src[0], (47.5,67.5), 5.4,5.0, 3)
+    yellow,ya,yanc = sprite_from_frame(src[0], (213.8,78.3), 5.5,5.4, 3)
+
+    # Refine alpha by colour confidence, but retain anti-aliased edge pixels.
+    br,bgch,bb = [blue[...,i].astype(np.float32) for i in range(3)]
+    conf = np.clip((bb-br-16)/35,0,1) * np.clip((bb-bgch-4)/22,0,1)
+    ba *= np.clip(conf*1.35,0,1)
+    # The source's small central red puck occluded a tiny part of the blue stone.
+    # Fill that hole by left/right symmetric texture, appropriate for this stone.
+    sh,sw=ba.shape; yy,xx=np.indices((sh,sw)); gx=xx+(127.5-banc[0]); gy=yy+(79.5-banc[1])
+    hole=((gx-127.5)/22)**2+((gy-75.0)/11)**2<1.25
+    mirror=blue[:,::-1].copy()
+    # Local blue surface around the hole is nearly uniform; use its radial mean.
+    ring=(((gx-127.5)/27)**2+((gy-75.0)/15)**2<1.0)&(~hole)&(ba>.45)
+    fill=np.median(blue[ring],axis=0) if np.any(ring) else np.array([106,172,209])
+    blue[hole]=fill
+    ba[hole]=np.maximum(ba[hole], .98)
+
+    pos = simulate()
+    frames=np.empty((N,H,W,3),np.uint8)
+    # Preserve the source's slight vertical perspective paths; after collision all
+    # pucks stay on the line defined by their initial centres.
+    p0=pos[0]
+    red_y0, yellow_y0 = 67.5, 78.3
+    for t in range(N):
+        canvas=bg.astype(np.float32).copy()
+        xr,xb,xy=pos[t]
+        # Counterfactual blue remains on the instantaneous red-to-yellow line.
+        # Simulated x controls collision response; y follows the same source lane.
+        yr = red_y0 + (xr-p0[0])*(72.0-red_y0)/(108.0-p0[0])
+        yyel = yellow_y0 + (p0[2]-xy)*(72.0-yellow_y0)/(p0[2]-136.0)
+        # Blue follows the lane interpolation between its neighbours.
+        yb = yr + (xb-xr)/max(xyel:=xy-xr,1e-5)*(yyel-yr)
+
+        # Large blue stone is scaled to the small physical puck footprint.  Its
+        # source texture is downsampled, preserving native material/highlighting.
+        scale=.105
+        bs=cv2.resize(blue,None,fx=scale,fy=scale,interpolation=cv2.INTER_AREA)
+        bal=cv2.resize(ba,None,fx=scale,fy=scale,interpolation=cv2.INTER_AREA)
+        banc2=(banc[0]*scale,banc[1]*scale)
+
+        # Layer order by y coordinate gives stable physically plausible occlusion.
+        items=[(yr,red,ra,ranc,(xr,yr)),(yb,bs,bal,banc2,(xb,yb)),
+               (yyel,yellow,ya,yanc,(xy,yyel))]
+        for _,sp,al,an,ce in sorted(items,key=lambda q:q[0]):
+            paste_sprite(canvas,sp,al,an,ce)
+        frames[t]=np.clip(np.rint(canvas),0,255).astype(np.uint8)
+    return frames
+
+
+if __name__ == '__main__':
+    frames=make_video()
+    assert frames.shape==(96,144,256,3) and frames.dtype==np.uint8
+    artifact_io.save(frames, CASE_DIR)
+    print('saved', frames.shape, frames.dtype)

#"'!/usr/bin/env python3
"""Generate the complete counterfactual video using only source RGB pixels.

The source scene is a symmetric top-down rigid-body setup.  The blue stone is
reconstructed from the visible blue annulus (the source contains it centred at
(127.5, 79.5), underneath the red stone), while the small stones are composited
from their source appearances.  Dynamics are one-dimensional equal-mass rigid
collisions with mild inelasticity, matching the source trajectory scale.
"""
from pathlib import Path
import numpy as np
import cv2
import artifact_io

CASE_DIR = Path(__file__).resolve().parent
N, H, W = 96, 144, 256


def bilinear(img, x, y):
    """Bilinearly sample an RGB image at floating coordinate arrays."""
    h, w = img.shape[:2]
    x = np.clip(x, 0, w - 1)
    y = np.clip(y, 0, h - 1)
    x0 = np.floor(x).astype(np.int32); y0 = np.floor(y).astype(np.int32)
    x1 = np.minimum(x0 + 1, w - 1); y1 = np.minimum(y0 + 1, h - 1)
    fx = (x - x0)[..., None]; fy = (y - y0)[..., None]
    a = img[y0, x0].astype(np.float32) * (1 - fx) + img[y0, x1].astype(np.float32) * fx
    b = img[y1, x0].astype(np.float32) * (1 - fx) + img[y1, x1].astype(np.float32) * fx
    return a * (1 - fy) + b * fy


def reconstruct_background(src):
    """Remove the three source stones but retain the textured board/platform."""
    # Median over time removes the moving red/yellow pucks.  The stationary blue
    # stone is removed by radial interpolation between clean rings just outside it.
    bg = np.median(src, axis=0).astype(np.float32)
    yy, xx = np.indices((H, W), dtype=np.float32)
    cx, cy = 127.5, 79.5
    dx, dy = xx - cx, yy - cy
    r = np.hypot(dx, dy)
    ux = dx / np.maximum(r, 1e-4); uy = dy / np.maximum(r, 1e-4)

    # The visible blue stone has a 61x28 projected ellipse.  Its underlying board
    # is reconstructed from corresponding samples at the clean outer ring.
    rho = np.sqrt((dx / 61.8) ** 2 + (dy / 29.0) ** 2)
    mask = rho < 1.035
    edge_x = cx + ux * 65.0
    edge_y = cy + uy * 33.0
    clean = bilinear(bg, edge_x, edge_y)
    # Preserve local board illumination by blending two clean radial samples.
    clean2 = bilinear(bg, cx + ux * 70.0, cy + uy * 36.0)
    clean = 0.68 * clean + 0.32 * clean2
    feather = np.clip((1.06 - rho) / 0.055, 0.0, 1.0)[..., None]
    bg = bg * (1 - feather) + clean * feather

    # Small residuals of source pucks lie outside the large ellipse. Restore those
    # thin areas from their immediate vertical surroundings (the board varies slowly).
    for x0, x1, y0, y1 in [(35, 61, 61, 75), (198, 225, 71, 88)]:
        patch = bg[y0:y1, x0:x1]
        top = bg[y0-3:y0, x0:x1].mean(0)
        bot = bg[y1:y1+3, x0:x1].mean(0)
        for j in range(y1-y0):
            u = (j + 0.5) / (y1-y0)
            patch[j] = top * (1-u) + bot * u
    return np.clip(bg, 0, 255).astype(np.uint8)


def sprite_from_frame(frame, center, rx, ry, pad=3):
    """Cut a soft elliptical RGBA sprite out of a source RGB frame."""
    cx, cy = center
    x0 = int(np.floor(cx-rx-pad)); x1 = int(np.ceil(cx+rx+pad))+1
    y0 = int(np.floor(cy-ry-pad)); y1 = int(np.ceil(cy+ry+pad))+1
    crop = frame[y0:y1, x0:x1].copy()
    yy, xx = np.indices(crop.shape[:2], dtype=np.float32)
    ex = (xx + x0 - cx) / rx; ey = (yy + y0 - cy) / ry
    rho = np.sqrt(ex*ex + ey*ey)
    alpha = np.clip((1.10-rho)/0.14, 0, 1)
    return crop, alpha.astype(np.float32), (cx-x0, cy-y0)


def paste_sprite(dst, sprite, alpha, anchor, center):
    """Subpixel bilinear sprite compositing."""
    sh, sw = alpha.shape
    cx, cy = center; ax, ay = anchor
    x0 = int(np.floor(cx-ax)); y0 = int(np.floor(cy-ay))
    fx = (cx-ax)-x0; fy = (cy-ay)-y0
    M = np.float32([[1, 0, fx], [0, 1, fy]])
    s = cv2.warpAffine(sprite, M, (sw, sh), flags=cv2.INTER_LINEAR,
                       borderMode=cv2.BORDER_REPLICATE)
    a = cv2.warpAffine(alpha, M, (sw, sh), flags=cv2.INTER_LINEAR,
                       borderMode=cv2.BORDER_CONSTANT, borderValue=0)
    xa=max(0,x0); ya=max(0,y0); xb=min(W,x0+sw); yb=min(H,y0+sh)
    if xb<=xa or yb<=ya: return
    sx0=xa-x0; sy0=ya-y0; sx1=sx0+xb-xa; sy1=sy0+yb-ya
    aa=a[sy0:sy1,sx0:sx1,None]
    dst[ya:yb,xa:xb] = dst[ya:yb,xa:xb]*(1-aa)+s[sy0:sy1,sx0:sx1]*aa


def simulate():
    """Equal-mass, mildly inelastic one-dimensional rigid puck dynamics."""
    # Source initial red/yellow centres, with blue placed exactly 1/4 red->yellow.
    p = np.array([47.5, 47.5 + 0.25*(213.8-47.5), 213.8], np.float64)
    v = np.array([1.98, 0.0, -2.10], np.float64)  # source pixels/frame
    radii = np.array([5.5, 5.5, 5.5])
    pos = np.empty((N,3), np.float64)
    restitution = 0.08
    for t in range(N):
        pos[t] = p
        for _ in range(10):
            p += v/10.0
            for i in range(2):
                contact = radii[i]+radii[i+1]
                gap = p[i+1]-p[i]
                if gap < contact and v[i] > v[i+1]:
                    overlap = contact-gap
                    p[i] -= overlap*.5; p[i+1] += overlap*.5
                    a,b = v[i],v[i+1]
                    v[i]   = .5*((1-restitution)*a+(1+restitution)*b)
                    v[i+1] = .5*((1+restitution)*a+(1-restitution)*b)
            v *= 0.99995
    return pos


def make_video():
    src = np.load(CASE_DIR/'"'source_pixels.npz')['rgb']
    bg = reconstruct_background(src)

    # Use unobstructed source appearances at frame 0. Blue texture is the visible
    # large blue object itself; alpha isolates its exact projected silhouette.
    blue, ba, banc = sprite_from_frame(src[0], (127.5,79.5), 62.0,29.0, 3)
    red,  ra, ranc = sprite_from_frame(src[0], (47.5,67.5), 5.4,5.0, 3)
    yellow,ya,yanc = sprite_from_frame(src[0], (213.8,78.3), 5.5,5.4, 3)
