+"""Generate the counterfactual video: remove the purple marble from frame 1.
+
+The source contains two colored pendulum marbles.  The yellow marble swings from
+left, collides with the purple one near the middle, and then rebounds.  With the
+purple marble absent the yellow one keeps swinging through the collision point.
+This script builds a robust clean plate from the source, extracts the intact
+pre-collision yellow marble (including its contact shadow), and continues its
+measured pendulum trajectory across the empty centre.
+"""
+from pathlib import Path
+import numpy as np
+from scipy.ndimage import binary_dilation, gaussian_filter, map_coordinates
+import artifact_io
+
+CASE_DIR = Path(__file__).resolve().parent
+N, H, W = 96, 144, 256
+
+
+def sample_bilinear(image, sx, sy):
+    """Bilinearly sample RGB image at floating-point source coordinates."""
+    out = np.empty((*sx.shape, 3), np.float32)
+    for c in range(3):
+        out[..., c] = map_coordinates(
+            image[..., c].astype(np.float32), [sy, sx], order=1,
+            mode="nearest", prefilter=False
+        )
+    return out
+
+
+def make_clean_plate(src):
+    # At any fixed pixel the small marbles/shadows are present for only a
+    # minority of frames, so the temporal median recovers the untouched set.
+    return np.median(src.astype(np.float32), axis=0)
+
+
+def estimate_yellow_track(src):
+    """Track the warm yellow core while it is physically factual."""
+    a = src.astype(np.int16)
+    r, g, b = [a[..., i] for i in range(3)]
+    chroma = (r > 140) & (r - g > 20) & (g - b > 35)
+    yy, xx = np.indices((H, W))
+    centers = []
+    previous = 27.0
+    # Stop before the center interaction/overlap.
+    for t in range(34):
+        keep = chroma[t] & (yy >= 45) & (yy < 86) & (np.abs(xx - previous) < 16)
+        y, x = np.where(keep)
+        if len(x) < 20:
+            raise RuntimeError(f"yellow marble tracking failed at frame {t}")
+        previous = float(x.mean())
+        centers.append((float(x.mean()), float(y.mean())))
+    return np.asarray(centers, np.float32)
+
+
+def fitted_trajectory(track):
+    """Fit a released pendulum to the factual, pre-impact yellow motion."""
+    x0 = float(track[:6, 0].mean())
+    # Symmetry of the two equal pendulums gives the rest-point midpoint.
+    # Purple's measured initial core centre is 228.30 px.
+    center_x = 0.5 * (x0 + 228.30)
+    release = 5
+    # A small deterministic grid fit is stable and keeps generate.py runnable.
+    times = np.arange(len(track), dtype=np.float32)
+    active = np.maximum(times - release, 0.0)
+    frequencies = np.linspace(0.045, 0.058, 2601)
+    predictions = x0 + (center_x - x0) * (
+        1.0 - np.cos(frequencies[:, None] * active[None, :])
+    )
+    errors = np.mean((predictions - track[None, :, 0]) ** 2, axis=1)
+    omega = float(frequencies[np.argmin(errors)])
+
+    t = np.arange(N, dtype=np.float32)
+    theta0 = np.arcsin(np.clip((x0 - center_x) / (center_x - x0), -1, 1))
+    # Equivalent to x0 + amplitude*(1-cos), written explicitly as pendulum arc.
+    theta = theta0 * np.cos(omega * np.maximum(t - release, 0.0))
+    x = center_x + (center_x - x0) * np.sin(theta)
+
+    # Perspective/arc sag measured from factual tracking. A quadratic in
+    # horizontal displacement robustly captures the sub-pixel vertical path.
+    q = (track[:, 0] - center_x) ** 2
+    A = np.c_[np.ones(len(q)), q]
+    c0, c2 = np.linalg.lstsq(A, track[:, 1], rcond=None)[0]
+    y = c0 + c2 * (x - center_x) ** 2
+    return x.astype(np.float32), y.astype(np.float32), omega
+
+
+def build_sprite(source, plate, center):
+    """Extract yellow marble + its coherent low shadow from a factual frame."""
+    cx, cy = center
+    yy, xx = np.indices((H, W), dtype=np.float32)
+    # Visible marble and the soft lower-left cast/contact shadow.
+    marble = ((xx - cx) / 7.7) ** 2 + ((yy - cy) / 7.3) ** 2 <= 1.0
+    shadow = ((xx - (cx - 4.2)) / 14.0) ** 2 + ((yy - (cy + 6.8)) / 4.7) ** 2 <= 1.0
+    support = binary_dilation(marble | shadow, iterations=3)
+
+    difference = np.max(np.abs(source.astype(np.float32) - plate), axis=2)
+    # Difference-derived alpha preserves antialiasing while geometry prevents
+    # unrelated parts of the set entering the sprite.
+    alpha = np.clip((difference - 0.8) / 4.5, 0.0, 1.0) * support
+    # Keep the opaque physical body; soften only the shadow and outer edge.
+    core = ((xx - cx) / 6.6) ** 2 + ((yy - cy) / 6.3) ** 2 <= 1.0
+    alpha[core] = np.maximum(alpha[core], 0.985)
+    alpha = gaussian_filter(alpha.astype(np.float32), 0.42)
+    alpha = np.clip(alpha, 0.0, 1.0)
+    return source.astype(np.float32), alpha.astype(np.float32), (float(cx), float(cy))
+
+
+def render():
+    data = np.load(CASE_DIR / "source_pixels.npz")
+    src = data["rgb"]
+    assert src.shape == (N, H, W, 3) and src.dtype == np.uint8
+
+    plate = make_clean_plate(src)
+    track = estimate_yellow_track(src)
+    path_x, path_y, omega = fitted_trajectory(track)
+
+    # Frame 24 is well into the factual swing, isolated from both center and
+    # boundary, and supplies a crisp motion-consistent marble/shadow sprite.
+    ref_t = 24
+    sprite, sprite_alpha, (ref_x, ref_y) = build_sprite(
+        src[ref_t], plate, track[ref_t]
+    )
+
+    yy, xx = np.indices((H, W), dtype=np.float32)
+    frames = np.empty((N, H, W, 3), np.uint8)
+    for t in range(N):
+        # Translation of the factual sprite. The tiny y shift follows the
+        # fitted pendulum arc; texture translation naturally rotates neither
+        # the sphere nor set, which is correct for a visually symmetric marble.
+        sx = xx - (path_x[t] - ref_x)
+        sy = yy - (path_y[t] - ref_y)
+        fg = sample_bilinear(sprite, sx, sy)
+        a = map_coordinates(
+            sprite_alpha, [sy, sx], order=1, mode="constant", cval=0.0,
+            prefilter=False
+        )[..., None]
+        frame = plate * (1.0 - a) + fg * a
+        frames[t] = np.clip(np.rint(frame), 0, 255).astype(np.uint8)
+
+    print(
+        f"yellow trajectory: x={path_x.min():.2f}..{path_x.max():.2f}, "
+        f"y={path_y.min():.2f}..{path_y.max():.2f}, omega={omega:.5f}"
+    )
+    return frames
+
+
+if __name__ == "__main__":
+    result = render()
+    artifact_io.save(result, CASE_DIR)
+    print("saved", result.shape, result.dtype, "to", CASE_DIR)

\"\"\"Generate the counterfactual video: remove the purple marble from frame 1.

The source contains two colored pendulum marbles.  The yellow marble swings from
left, collides with the purple one near the middle, and then rebounds.  With the
purple marble absent the yellow one keeps swinging through the collision point.
This script builds a robust clean plate from the source, extracts the intact
pre-collision yellow marble (including its contact shadow), and continues its
measured pendulum trajectory across the empty centre.
\"\"\"
from pathlib import Path
import numpy as np
from scipy.ndimage import binary_dilation, gaussian_filter, map_coordinates
import artifact_io

CASE_DIR = Path(__file__).resolve().parent
N, H, W = 96, 144, 256


def sample_bilinear(image, sx, sy):
    \"\"\"Bilinearly sample RGB image at floating-point source coordinates.\"\"\"
    out = np.empty((*sx.shape, 3), np.float32)
    for c in range(3):
        out[..., c] = map_coordinates(
            image[..., c].astype(np.float32), [sy, sx], order=1,
            mode=\"nearest\", prefilter=False
        )
    return out


def make_clean_plate(src):
    # At any fixed pixel the small marbles/shadows are present for only a
    # minority of frames, so the temporal median recovers the untouched set.
    return np.median(src.astype(np.float32), axis=0)


def estimate_yellow_track(src):
    \"\"\"Track the warm yellow core while it is physically factual.\"\"\"
    a = src.astype(np.int16)
    r, g, b = [a[..., i] for i in range(3)]
    chroma = (r > 140) & (r - g > 20) & (g - b > 35)
    yy, xx = np.indices((H, W))
    centers = []
    previous = 27.0
    # Stop before the center interaction/overlap.
    for t in range(34):
        keep = chroma[t] & (yy >= 45) & (yy < 86) & (np.abs(xx - previous) < 16)
        y, x = np.where(keep)
        if len(x) < 20:
            raise RuntimeError(f\"yellow marble tracking failed at frame {t}\")
        previous = float(x.mean())
        centers.append((float(x.mean()), float(y.mean())))
    return np.asarray(centers, np.float32)


def fitted_trajectory(track):
    \"\"\"Fit a released pendulum to the factual, pre-impact yellow motion.\"\"\"
    x0 = float(track[:6, 0].mean())
    # Symmetry of the two equal pendulums gives the rest-point midpoint.
    # Purple's measured initial core centre is 228.30 px.
    center_x = 0.5 * (x0 + 228.30)
    release = 5
    # A small deterministic grid fit is stable and keeps generate.py runnable.
    times = np.arange(len(track), dtype=np.float32)
    active = np.maximum(times - release, 0.0)
    frequencies = np.linspace(0.045, 0.058, 2601)
    predictions = x0 + (center_x - x0) * (
        1.0 - np.cos(frequencies[:, None] * active[None, :])
    )
    errors = np.mean((predictions - track[None, :, 0]) ** 2, axis=1)
    omega = float(frequencies[np.argmin(errors)])

    t = np.arange(N, dtype=np.float32)
    theta0 = np.arcsin(np.clip((x0 - center_x) / (center_x - x0), -1, 1))
    # Equivalent to x0 + amplitude*(1-cos), written explicitly as pendulum arc.
    theta = theta0 * np.cos(omega * np.maximum(t - release, 0.0))
    x = center_x + (center_x - x0) * np.sin(theta)

    # Perspective/arc sag measured from factual tracking. A quadratic in
    # horizontal displacement robustly captures the sub-pixel vertical path.
    q = (track[:, 0] - center_x) ** 2
    A = np.c_[np.ones(len(q)), q]
    c0, c2 = np.linalg.lstsq(A, track[:, 1], rcond=None)[0]
    y = c0 + c2 * (x - center_x) ** 2
    return x.astype(np.float32), y.astype(np.float32), omega


def build_sprite(source, plate, center):
    \"\"\"Extract yellow marble + its coherent low shadow from a factual frame.\"\"\"
    cx, cy = center
    yy, xx = np.indices((H, W), dtype=np.float32)
    # Visible marble and the soft lower-left cast/contact shadow.
    marble = ((xx - cx) / 7.7) ** 2 + ((yy - cy) / 7.3) ** 2 <= 1.0
    shadow = ((xx - (cx - 4.2)) / 14.0) ** 2 + ((yy - (cy + 6.8)) / 4.7) ** 2 <= 1.0
    support = binary_dilation(marble | shadow, iterations=3)

    difference = np.max(np.abs(source.astype(np.float32) - plate), axis=2)
    # Difference-derived alpha preserves antialiasing while geometry prevents
    # unrelated parts of the set entering the sprite.
    alpha = np.clip((difference - 0.8) / 4.5, 0.0, 1.0) * support
    # Keep the opaque physical body; soften only the shadow and outer edge.
    core = ((xx - cx) / 6.6) ** 2 + ((yy - cy) / 6.3) ** 2 <= 1.0
    alpha[core] = np.maximum(alpha[core], 0.985)
    alpha = gaussian_filter(alpha.astype(np.float32), 0.42)
    alpha = np.clip(alpha, 0.0, 1.0)
    return source.astype(np.float32), alpha.astype(np.float32), (float(cx), float(cy))


def render():
    data = np.load(CASE_DIR / \"source_pixels.npz\")
    src = data[\"rgb\"]
    assert src.shape == (N, H, W, 3) and src.dtype == np.uint8

    plate = make_clean_plate(src)
    track = estimate_yellow_track(src)
    path_x, path_y, omega = fitted_trajectory(track)

    # Frame 24 is well into the factual swing, isolated from both center and
    # boundary, and supplies a crisp motion-consistent marble/shadow sprite.
    ref_t = 24
    sprite, sprite_alpha, (ref_x, ref_y) = build_sprite(
        src[ref_t], plate, track[ref_t]
    )

    yy, xx = np.indices((H, W), dtype=np.float32)
    frames = np.empty((N, H, W, 3), np.uint8)
    for t in range(N):
        # Translation of the factual sprite. The tiny y shift follows the
        # fitted pendulum arc; texture translation naturally rotates neither
        # the sphere nor set, which is correct for a visually symmetric marble.
        sx = xx - (path_x[t] - ref_x)
        sy = yy - (path_y[t] - ref_y)
        fg = sample_bilinear(sprite, sx, sy)
        a = map_coordinates(
            sprite_alpha, [sy, sx], order=1, mode=\"constant\", cval=0.0,
            prefilter=False
        )[..., None]
        frame = plate * (1.0 - a) + fg * a
        frames[t] = np.clip(np.rint(frame), 0, 255).astype(np.uint8)

    print(
        f\"yellow trajectory: x={path_x.min():.2f}..{path_x.max():.2f}, \"
        f\"y={path_y.min():.2f}..{path_y.max():.2f}, omega={omega:.5f}\"
    )
    return frames


if __name__ == \"__main__\":
    result = render()
    artifact_io.save(result, CASE_DIR)
    print(\"saved\", result.shape, result.dtype, \"to\", CASE_DIR)
