+"""Remove the white die from the complete source timeline.
+
+The camera and scenery in this clip are static.  A clean plate can therefore
+be assembled from two moments at which the relevant part of the set is not
+covered: frame 40 for the upper part of the image, and frame 0 for the floor.
+The white die is segmented against that plate on every frame.  Its dark pips
+are filled as holes in the bright connected component, and a small local
+difference halo removes its antialiased boundary and moving shadow.  The red
+die is independently segmented and composited back wherever the two local
+halos meet, preserving its original (unaffected) bounce.
+"""
+
+from pathlib import Path
+
+import numpy as np
+from scipy import ndimage
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def clean_plate(source: np.ndarray) -> np.ndarray:
+    """Construct an object-free plate for the narrow edited part of the set."""
+    upper = source[40].astype(np.float32)
+    lower = source[0].astype(np.float32)
+    plate = upper.copy()
+
+    # The dice have left the upper scene by frame 40, while the floor below
+    # row 68 is unoccupied in frame 0.  Feather between the two equivalent
+    # static views to avoid introducing even a one-pixel compression seam.
+    for y in range(source.shape[1]):
+        a = np.clip((y - 58.0) / 12.0, 0.0, 1.0)
+        plate[y] = upper[y] * (1.0 - a) + lower[y] * a
+    return np.rint(plate).astype(np.uint8)
+
+
+def component_near(mask: np.ndarray, target_x: float) -> np.ndarray:
+    """Return the useful connected component nearest a known horizontal lane."""
+    labels, count = ndimage.label(mask)
+    best_label = 0
+    best_score = -1e9
+    for label_id in range(1, count + 1):
+        yy, xx = np.where(labels == label_id)
+        area = len(xx)
+        if area < 8:
+            continue
+        # Dice components are compact and remain in a fixed horizontal lane.
+        score = area - 3.0 * abs(float(xx.mean()) - target_x)
+        if score > best_score:
+            best_score = score
+            best_label = label_id
+    if best_label == 0:
+        raise RuntimeError("could not locate die component")
+    return labels == best_label
+
+
+def filled_die_core(frame: np.ndarray, plate: np.ndarray, color: str) -> np.ndarray:
+    """Segment a die face, fill its pips, and include its antialiased outline."""
+    rgb = frame.astype(np.int16)
+    delta = np.max(np.abs(rgb - plate.astype(np.int16)), axis=2)
+    r, g, b = rgb[..., 0], rgb[..., 1], rgb[..., 2]
+    yy, xx = np.indices(frame.shape[:2])
+
+    lane = (xx >= 112) & (xx <= 143) & (yy <= 100)
+    if color == "white":
+        chroma = np.maximum.reduce([r, g, b]) - np.minimum.reduce([r, g, b])
+        candidates = lane & (np.minimum.reduce([r, g, b]) > 155) & (chroma < 58) & (delta > 8)
+        core = component_near(candidates, 133.0)
+    elif color == "red":
+        candidates = lane & (r > 125) & (r - g > 16) & (r - b > 18) & (delta > 8)
+        core = component_near(candidates, 123.0)
+    else:
+        raise ValueError(color)
+
+    # The colored/white face surrounds the dark pips, so hole filling recovers
+    # them without confusing them with unrelated dark scenery.
+    core = ndimage.binary_closing(core, iterations=1)
+    core = ndimage.binary_fill_holes(core)
+    return ndimage.binary_dilation(core, iterations=1)
+
+
+def edit(source: np.ndarray) -> np.ndarray:
+    plate = clean_plate(source)
+    result = source.astype(np.float32).copy()
+    plate_f = plate.astype(np.float32)
+
+    white_areas = []
+    for t, frame in enumerate(source):
+        white = filled_die_core(frame, plate, "white")
+        red = filled_die_core(frame, plate, "red")
+        white_areas.append(int(white.sum()))
+
+        delta = np.max(
+            np.abs(frame.astype(np.int16) - plate.astype(np.int16)), axis=2
+        )
+
+        # Include a compact halo of changed pixels around the die.  This takes
+        # out both edge antialiasing and the die's local moving contact shadow,
+        # while leaving the detailed static floor untouched.
+        near_white = ndimage.binary_dilation(white, iterations=5)
+        remove = white | (near_white & (delta > 4))
+        remove = ndimage.binary_closing(remove, iterations=1)
+        remove_alpha = ndimage.gaussian_filter(remove.astype(np.float32), 0.65)
+        remove_alpha = np.clip(remove_alpha, 0.0, 1.0)
+
+        edited = (
+            frame.astype(np.float32) * (1.0 - remove_alpha[..., None])
+            + plate_f * remove_alpha[..., None]
+        )
+
+        # The two dice pass very close to each other.  Restore the red die and
+        # its immediate visible edge/shadow after removing the white die.
+        near_red = ndimage.binary_dilation(red, iterations=4)
+        red_visible = red | (near_red & (delta > 4) & ~white)
+        red_alpha = ndimage.gaussian_filter(red_visible.astype(np.float32), 0.45)
+        red_alpha = np.clip(red_alpha, 0.0, 1.0)
+        edited = (
+            edited * (1.0 - red_alpha[..., None])
+            + frame.astype(np.float32) * red_alpha[..., None]
+        )
+
+        result[t] = edited
+
+    result = np.clip(np.rint(result), 0, 255).astype(np.uint8)
+    print(
+        "white die masks:",
+        f"min={min(white_areas)} max={max(white_areas)} mean={np.mean(white_areas):.1f}",
+    )
+    return result
+
+
+def main() -> None:
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3) and source.dtype == np.uint8
+    frames = edit(source)
+    artifact_io.save(frames, CASE_DIR)
+    print("saved", frames.shape, frames.dtype, "to", CASE_DIR)
+
+
+if __name__ == "__main__":
+    main()
