+"""Generate the counterfactual video with the red die absent from frame 1.
+
+The shot is locked off.  A clean plate is assembled from source instants at
+which each half of the dice's path is empty.  The two factual dice are removed
+with a source-vs-plate matte, and the remaining (pale) die is restored.  Before
+the original interaction its measured source trajectory is retained; after
+that point it follows a damped rigid-body bounce on the tabletop.
+"""
+
+from pathlib import Path
+
+import numpy as np
+from PIL import Image
+from scipy import ndimage
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def clean_plate(src):
+    """Build a static, object-free plate from unoccluded source samples."""
+    # Frame 31 has the stable illumination and is clean along the upper path.
+    plate = src[30].astype(np.float32).copy()
+
+    # After frame 21 both dice are below y=65, so this median is clean above.
+    upper = np.median(src[20:31].astype(np.float32), axis=0)
+    # During the first four frames the lower tabletop is clear in at least
+    # three samples at every pixel (the small die does not reach it yet).
+    lower = np.median(src[:4].astype(np.float32), axis=0)
+
+    yy = np.arange(src.shape[1], dtype=np.float32)[:, None]
+    # A short blend avoids creating a horizontal seam in the clean plate.
+    blend = np.clip((yy - 42.0) / 10.0, 0.0, 1.0)[..., None]
+    local = upper * (1.0 - blend) + lower * blend
+    plate[:, 106:150] = local[:, 106:150]
+    return plate
+
+
+def largest_filled_component(mask):
+    lab, count = ndimage.label(mask)
+    if count == 0:
+        return mask
+    sizes = ndimage.sum(mask, lab, np.arange(1, count + 1))
+    out = lab == (1 + int(np.argmax(sizes)))
+    out = ndimage.binary_closing(out, iterations=1)
+    return ndimage.binary_fill_holes(out)
+
+
+def extract_pale_die(src_frame, plate, bounds):
+    """Return a premultiplied-free RGBA crop of the non-red die."""
+    x0, y0, x1, y1 = bounds
+    h, w = src_frame.shape[:2]
+    cx0, cy0, cx1, cy1 = max(0, x0), max(0, y0), min(w, x1), min(h, y1)
+    rgb = src_frame[cy0:cy1, cx0:cx1].copy()
+    ref = plate[cy0:cy1, cx0:cx1]
+    delta = np.max(np.abs(rgb.astype(np.float32) - ref), axis=2)
+    core = largest_filled_component(delta > 10.0)
+    # Keep antialiased silhouette pixels but avoid making the crop's unrelated
+    # background opaque.  The filled core also retains dark pips on the die.
+    support = ndimage.binary_dilation(core, iterations=1)
+    alpha = np.clip((delta - 3.0) / 9.0, 0.0, 1.0) * support
+    alpha = np.maximum(alpha, ndimage.gaussian_filter(core.astype(float), 0.55))
+    alpha = np.clip(alpha, 0.0, 1.0)
+    return rgb, alpha
+
+
+def paste_rgba(frame, rgb, alpha, center, angle=0.0):
+    """Rotate an extracted source sprite and alpha-composite at center."""
+    # Pad first so rotation never clips a corner of the rigid die.
+    pad = 7
+    rgba = np.zeros((rgb.shape[0] + 2 * pad, rgb.shape[1] + 2 * pad, 4), np.uint8)
+    rgba[pad : pad + rgb.shape[0], pad : pad + rgb.shape[1], :3] = rgb
+    rgba[pad : pad + rgb.shape[0], pad : pad + rgb.shape[1], 3] = np.rint(alpha * 255).astype(np.uint8)
+    im = Image.fromarray(rgba, "RGBA")
+    if abs(angle) > 0.01:
+        im = im.rotate(angle, resample=Image.Resampling.BICUBIC, expand=False)
+    arr = np.asarray(im).astype(np.float32)
+    a = arr[:, :, 3:4] / 255.0
+    hh, ww = arr.shape[:2]
+    x0 = int(round(center[0] - ww / 2.0))
+    y0 = int(round(center[1] - hh / 2.0))
+    x1, y1 = x0 + ww, y0 + hh
+    fx0, fy0, fx1, fy1 = max(0, x0), max(0, y0), min(frame.shape[1], x1), min(frame.shape[0], y1)
+    if fx0 >= fx1 or fy0 >= fy1:
+        return
+    sx0, sy0 = fx0 - x0, fy0 - y0
+    sx1, sy1 = sx0 + fx1 - fx0, sy0 + fy1 - fy0
+    aa = a[sy0:sy1, sx0:sx1]
+    frame[fy0:fy1, fx0:fx1] = (
+        frame[fy0:fy1, fx0:fx1] * (1.0 - aa)
+        + arr[sy0:sy1, sx0:sx1, :3] * aa
+    )
+
+
+def die_center_y(t):
+    """Ballistic vertical motion in output-frame units, with damped bounces."""
+    # Least-squares continuation of the untouched centers in source frames
+    # 1--7: y = .75*t^2 + 5.107*t + 7.
+    impact_t = 7.397
+    floor_y = 85.8
+    if t <= impact_t:
+        return 0.75 * t * t + 5.107142857 * t + 7.0
+
+    # First bounce.  Acceleration is 1.5 px/frame^2, matching the fall.
+    s = t - impact_t
+    v1 = 8.35
+    flight1 = 2.0 * v1 / 1.5
+    if s <= flight1:
+        return floor_y - v1 * s + 0.75 * s * s
+
+    # A small second bounce prevents an implausibly abrupt dead stop.
+    s -= flight1
+    v2 = 2.35
+    flight2 = 2.0 * v2 / 1.5
+    if s <= flight2:
+        return floor_y - v2 * s + 0.75 * s * s
+    return floor_y
+
+
+def add_contact_shadow(frame, center_y):
+    """Add the pale die's subtle tabletop contact shadow."""
+    h, w = frame.shape[:2]
+    yy, xx = np.mgrid[:h, :w]
+    clearance = max(0.0, 85.8 - center_y)
+    strength = 0.16 * np.exp(-clearance / 11.0)
+    # The light comes from upper-left, so the small shadow falls rightward.
+    shadow = np.exp(-0.5 * (((xx - 134.0) / 6.4) ** 2 + ((yy - 94.0) / 1.55) ** 2))
+    factor = 1.0 - strength * shadow[..., None]
+    frame *= factor
+
+
+def generate():
+    src = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert src.shape == (96, 144, 256, 3) and src.dtype == np.uint8
+    plate = clean_plate(src)
+
+    # Bounds are exclusive.  They trace the pale die before it first touches
+    # the red die.  Source frame 7 supplies the clean canonical post-contact
+    # sprite (there is still a gap beneath it at this instant).
+    pale_bounds = [
+        (122, 0, 142, 16),
+        (122, 4, 142, 23),
+        (122, 11, 142, 30),
+        (122, 20, 142, 39),
+        (122, 30, 142, 49),
+        (122, 42, 142, 61),
+        (122, 55, 142, 75),
+    ]
+    sprites = [extract_pale_die(src[i], plate, pale_bounds[i]) for i in range(7)]
+    canonical = sprites[6]
+
+    out = np.empty_like(src)
+    region = np.zeros((144, 256), bool)
+    region[:106, 106:150] = True
+
+    for i in range(96):
+        frame = src[i].astype(np.float32).copy()
+
+        # Remove both factual moving layers locally.  Dilation includes their
+        # antialiasing, pips, and moving cast shadows; the soft edge makes the
+        # restoration imperceptible against source compression noise.
+        delta = np.max(np.abs(src[i].astype(np.float32) - plate), axis=2)
+        matte = (delta > 9.0) & region
+        matte = ndimage.binary_dilation(matte, iterations=2)
+        matte = ndimage.binary_closing(matte, iterations=1)
+        soft = ndimage.gaussian_filter(matte.astype(np.float32), 0.7)[..., None]
+        frame = frame * (1.0 - soft) + plate * soft
+
+        cy = die_center_y(float(i))
+        add_contact_shadow(frame, cy)
+
+        if i < 7:
+            sprite = sprites[i]
+            # These poses are factual and precede interaction with the removed
+            # die, so preserve their exact source orientation.
+            angle = 0.0
+        else:
+            sprite = canonical
+            # Continue the measured clockwise tumble, damping it after rest.
+            moving_end = 22.0
+            spin_t = min(float(i - 6), moving_end - 6.0)
+            angle = -10.5 * spin_t
+        paste_rgba(frame, sprite[0], sprite[1], (132.0, cy), angle)
+        out[i] = np.clip(np.rint(frame), 0, 255).astype(np.uint8)
+
+    return out
+
+
+if __name__ == "__main__":
+    frames = generate()
+    artifact_io.save(frames, CASE_DIR)
+    print("saved", frames.shape, frames.dtype, "to", CASE_DIR)
