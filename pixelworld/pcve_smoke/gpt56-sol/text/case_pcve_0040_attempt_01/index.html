+"""Generate the counterfactual video for case_pcve_0040_attempt_01.
+
+The car is airborne at the start of the shot and reaches the lower surface around
+source frame 8 (zero based).  Friction has no effect while it is airborne.  The
+source horizontal coast has about -0.44 px/frame^2 acceleration; multiplying the
+surface friction by seven brings the car to rest about 8 px after touchdown.
+The first landing frame is therefore retained, followed by two small translated
+steps and then a stationary car.
+"""
+from pathlib import Path
+
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def dilate(mask: np.ndarray, iterations: int = 1) -> np.ndarray:
+    """Small dependency-free 8-connected binary dilation."""
+    out = mask.copy()
+    for _ in range(iterations):
+        p = np.pad(out, 1, mode="constant")
+        out = np.logical_or.reduce(
+            [p[dy : dy + out.shape[0], dx : dx + out.shape[1]]
+             for dy in range(3) for dx in range(3)]
+        )
+    return out
+
+
+def translated_landing_still(source: np.ndarray, dx: int) -> np.ndarray:
+    """Move the landed car in source frame 8 by an integer horizontal offset.
+
+    This uses a clean temporal background plate and translates only the measured
+    RGB residual of the car.  Moving the residual (rather than cutting a box)
+    retains the original antialiasing, contact shadow, and reflected light.
+    """
+    car_frame = source[8].astype(np.int16)
+
+    # Frames 0..4 show this part of the lower surface before the car arrives.
+    background = np.median(source[:5], axis=0).astype(np.int16)
+    residual = car_frame - background
+    difference = np.max(np.abs(residual), axis=2)
+
+    # Source frame 8 car and its contact shadow are confined to this small ROI.
+    roi = np.zeros(difference.shape, dtype=bool)
+    roi[38:58, 130:161] = True
+    core = (difference > 4) & roi
+    support = dilate(core, 2) & roi
+
+    # Suppress tiny codec noise at the edge while retaining partially blended
+    # silhouette pixels.  The residual already contains premultiplied edge color.
+    strength = np.clip((difference.astype(np.float32) - 1.0) / 3.0, 0.0, 1.0)
+    residual_f = residual.astype(np.float32) * (support * strength)[..., None]
+
+    shifted = np.zeros_like(residual_f)
+    if dx >= 0:
+        shifted[:, dx:] = residual_f[:, : residual_f.shape[1] - dx]
+    else:
+        shifted[:, :dx] = residual_f[:, -dx:]
+
+    # At dx=0 this construction reproduces source frame 8 exactly apart from
+    # imperceptible codec-noise suppression.  Positive dx exposes the clean floor.
+    edited = car_frame.astype(np.float32) - residual_f + shifted
+    return np.clip(np.rint(edited), 0, 255).astype(np.uint8)
+
+
+def build_frames() -> np.ndarray:
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3) and source.dtype == np.uint8
+
+    output = np.empty_like(source)
+
+    # Exact factual appearance through flight and first surface contact.
+    output[:9] = source[:9]
+
+    # Sevenfold surface friction: only one pixel of travel on each of the next
+    # two frames, versus the source's ~6 px/frame, then no further movement.
+    output[9] = translated_landing_still(source, 1)
+    stopped = translated_landing_still(source, 2)
+    output[10:] = stopped
+    return output
+
+
+if __name__ == "__main__":
+    frames = build_frames()
+    artifact_io.save(frames, CASE_DIR)
+    print("saved", frames.shape, frames.dtype)

from pathlib import Path

import numpy as np

import artifact_io


CASE_DIR = Path(__file__).resolve().parent


def dilate(mask: np.ndarray, iterations: int = 1) -> np.ndarray:
    """Small dependency-free 8-connected binary dilation."""
    out = mask.copy()
    for _ in range(iterations):
        p = np.pad(out, 1, mode="constant")
        out = np.logical_or.reduce(
            [p[dy : dy + out.shape[0], dx : dx + out.shape[1]]
             for dy in range(3) for dx in range(3)]
        )
    return out


def translated_landing_still(source: np.ndarray, dx: int) -> np.ndarray:
    """Move the landed car in source frame 8 by an integer horizontal offset.

    This uses a clean temporal background plate and translates only the measured
    RGB residual of the car.  Moving the residual (rather than cutting a box)
    retains the original antialiasing, contact shadow, and reflected light.
    """
    car_frame = source[8].astype(np.int16)

    # Frames 0..4 show this part of the lower surface before the car arrives.
    background = np.median(source[:5], axis=0).astype(np.int16)
    residual = car_frame - background
    difference = np.max(np.abs(residual), axis=2)

    # Source frame 8 car and its contact shadow are confined to this small ROI.
    roi = np.zeros(difference.shape, dtype=bool)
    roi[38:58, 130:161] = True
    core = (difference > 4) & roi
    support = dilate(core, 2) & roi

    # Suppress tiny codec noise at the edge while retaining partially blended
    # silhouette pixels.  The residual already contains premultiplied edge color.
    strength = np.clip((difference.astype(np.float32) - 1.0) / 3.0, 0.0, 1.0)
    residual_f = residual.astype(np.float32) * (support * strength)[..., None]

    shifted = np.zeros_like(residual_f)
    if dx >= 0:
        shifted[:, dx:] = residual_f[:, : residual_f.shape[1] - dx]
    else:
        shifted[:, :dx] = residual_f[:, -dx:]

    # At dx=0 this construction reproduces source frame 8 exactly apart from
    # imperceptible codec-noise suppression.  Positive dx exposes the clean floor.
    edited = car_frame.astype(np.float32) - residual_f + shifted
    return np.clip(np.rint(edited), 0, 255).astype(np.uint8)

def shift_rgb(array: np.ndarray, dx: int, dy: int = 0) -> np.ndarray:
    \"\"\"Translate an HWC float array without wraparound.\"\"\"
    out = np.zeros_like(array)
    h, w = array.shape[:2]
    sx0, sx1 = max(0, -dx), min(w, w - dx)
    sy0, sy1 = max(0, -dy), min(h, h - dy)
    if sx1 > sx0 and sy1 > sy0:
        out[sy0 + dy:sy1 + dy, sx0 + dx:sx1 + dx] = array[sy0:sy1, sx0:sx1]
    return out


def car_residual(frame: np.ndarray, clean: np.ndarray, cx: float, cy: float) -> np.ndarray:
    \"\"\"Extract the car, its antialiased edge, and its immediate contact shadow.\"\"\"
    delta = frame.astype(np.int16) - clean.astype(np.int16)
    difference = np.max(np.abs(delta), axis=2)
    yy, xx = np.ogrid[:difference.shape[0], :difference.shape[1]]
    roi = (np.abs(xx - cx) <= 18) & (np.abs(yy - cy) <= 13)
    core = (difference > 4) & roi

    alpha = np.clip((difference.astype(np.float32) - 1.0) / 3.0, 0.0, 1.0)
