+from artifact_io import save

+def circular_sprite(frame, cx, cy, radius, pad=2):
+    """Extract an RGB sprite and a soft circular alpha from source pixels."""
+    half = int(np.ceil(radius + pad))
+    x0, y0 = int(np.floor(cx)) - half, int(np.floor(cy)) - half
+    x1, y1 = int(np.floor(cx)) + half + 1, int(np.floor(cy)) + half + 1
+    rgb = frame[y0:y1, x0:x1].copy()
+    yy, xx = np.mgrid[y0:y1, x0:x1]
+    # A one-pixel antialiased edge retains the ball's original contact shadow.
+    dist = np.hypot(xx - cx, yy - cy)
+    alpha = np.clip(radius + 0.7 - dist, 0.0, 1.0).astype(np.float32)
+    return rgb, alpha
+
+
+def fit_clean_felt(frame):
+    """Remove the two touching balls near (130,69) with local felt shading."""
+    result = frame.astype(np.float32).copy()
+    yy, xx = np.mgrid[0:H, 0:W]
+    # Union encloses the stationary cue ball, returning dark ball, and shadows.
+    hole = (((xx - 126.1) / 7.2) ** 2 + ((yy - 71.1) / 7.2) ** 2 < 1.0)
+    hole |= (((xx - 134.0) / 6.0) ** 2 + ((yy - 66.8) / 6.0) ** 2 < 1.0)
+    outer = (((xx - 129.5) / 16.0) ** 2 + ((yy - 69.0) / 14.0) ** 2 < 1.0)
+    samples = outer & ~hole
+    sx = (xx[samples] - 129.5) / 16.0
+    sy = (yy[samples] - 69.0) / 14.0
+    # Quadratic illumination model follows the smooth, diagonally lit cloth.
+    design = np.column_stack([np.ones_like(sx), sx, sy, sx*sx, sx*sy, sy*sy])
+    hx = (xx[hole] - 129.5) / 16.0
+    hy = (yy[hole] - 69.0) / 14.0
+    query = np.column_stack([np.ones_like(hx), hx, hy, hx*hx, hx*hy, hy*hy])
+    for ch in range(3):
+        coef = np.linalg.lstsq(design, frame[:, :, ch][samples], rcond=None)[0]
+        result[:, :, ch][hole] = query @ coef
+    return np.clip(result, 0, 255).astype(np.uint8)
+
+
+def resized_sprite(rgb, alpha, scale):
+    nh = max(2, int(round(rgb.shape[0] * scale)))
+    nw = max(2, int(round(rgb.shape[1] * scale)))
+    im = Image.fromarray(rgb).resize((nw, nh), Image.Resampling.LANCZOS)
+    am = Image.fromarray(np.uint8(np.clip(alpha, 0, 1) * 255), mode="L").resize(
+        (nw, nh), Image.Resampling.LANCZOS
+    )
+    return np.asarray(im), np.asarray(am).astype(np.float32) / 255.0
+
+
+def composite(frame, rgb, alpha, cx, cy, scale=1.0):
+    """Alpha composite a sprite at a subpixel center."""
+    spr, a = resized_sprite(rgb, alpha, scale)
+    sh, sw = a.shape
+    x0 = int(round(cx - (sw - 1) / 2))
+    y0 = int(round(cy - (sh - 1) / 2))
+    x1, y1 = x0 + sw, y0 + sh
+    dx0, dy0 = max(0, x0), max(0, y0)
+    dx1, dy1 = min(W, x1), min(H, y1)
+    if dx0 >= dx1 or dy0 >= dy1:

+    src = spr[sy0:sy1, sx0:sx1].astype(np.float32)
+    frame[dy0:dy1, dx0:dx1] = np.clip(dst*(1-aa) + src*aa, 0, 255).astype(np.uint8)

+            speed *= restitution
