+"""Generate the 10x-blue-mallet counterfactual for this PixelWorld case.
+
+The source is a nearly fixed camera shot.  The small blue mallet travels down-left,
+hits a row of equal-size bodies at about frame 15, and in the factual clip gives up
+essentially all of its speed.  For a one-dimensional elastic impact, changing its
+mass from m to 10m changes the two post-impact velocity factors from (0, 1) to
+(9/11, 20/11).  We therefore speed up the visible downstream response by 20/11
+and composite the now-continuing blue mallet at 9/11 of its incoming velocity.
+All imagery is sampled from the supplied RGB tensor; no external assets are used.
+"""
+from pathlib import Path
+import cv2
+import numpy as np
+from artifact_io import save
+
+CASE_DIR = Path(__file__).resolve().parent
+N, H, W = 96, 144, 256
+IMPACT = 14.0                 # last essentially pre-impact frame (zero based)
+DOWNSTREAM_RATE = 20.0 / 11.0
+BLUE_RATE = 9.0 / 11.0
+
+
+def lerp_frame(frames, time):
+    """Linear temporal sampling, avoiding duplicate/jerky accelerated frames."""
+    time = float(np.clip(time, 0.0, len(frames) - 1.0))
+    lo = int(np.floor(time))
+    hi = min(lo + 1, len(frames) - 1)
+    f = time - lo
+    if hi == lo:
+        return frames[lo].copy()
+    out = frames[lo].astype(np.float32) * (1.0 - f) + frames[hi].astype(np.float32) * f
+    return np.clip(out + 0.5, 0, 255).astype(np.uint8)
+
+
+def remove_factual_stopped_blue(image):
+    """Remove only the blue head that remains at the first impact point.
+
+    A chroma mask keeps the red body and lavender surroundings intact.  Telea
+    inpainting is confined to a small local window, so the rest of the source is
+    bit-for-bit unchanged.
+    """
+    out = image.copy()
+    x0, y0, x1, y1 = 136, 56, 157, 78
+    p = out[y0:y1, x0:x1]
+    r, g, b = [p[:, :, k].astype(np.int16) for k in range(3)]
+    mask = ((b - r > 28) & (b - g > 16) & (b > 105)).astype(np.uint8) * 255
+    # Include anti-aliased blue perimeter, but remain local to the mallet head.
+    mask = cv2.dilate(mask, np.ones((3, 3), np.uint8), iterations=1)
+    if np.any(mask):
+        cleaned = cv2.inpaint(p, mask, 3.0, cv2.INPAINT_TELEA)
+        out[y0:y1, x0:x1] = cleaned
+    return out
+
+
+def extract_blue_sprite(source):
+    """Extract the original blue mallet head and a soft source-derived alpha."""
+    # At frame 2 the blue head is isolated near (165,59).  Frame 95 supplies the
+    # same static background at that location.
+    t = 2
+    x0, y0, x1, y1 = 156, 50, 176, 69
+    fg = source[t, y0:y1, x0:x1].copy()
+    bg = source[95, y0:y1, x0:x1]
+    delta = np.max(np.abs(fg.astype(np.int16) - bg.astype(np.int16)), axis=2)
+    seed = (delta > 9).astype(np.uint8)
+    n, labels, stats, _ = cv2.connectedComponentsWithStats(seed, 8)
+    if n > 1:
+        # Component nearest patch center; this rejects codec speckle.
+        scores = []
+        cy, cx = (y1-y0-1)/2, (x1-x0-1)/2
+        for i in range(1, n):
+            ys, xs = np.where(labels == i)
+            scores.append((stats[i, cv2.CC_STAT_AREA] / (1 + (xs.mean()-cx)**2 + (ys.mean()-cy)**2), i))
+        core = (labels == max(scores)[1]).astype(np.uint8)
+    else:
+        core = seed
+    core = cv2.dilate(core, np.ones((3, 3), np.uint8), iterations=1)
+    # Difference strength retains translucent antialiasing around the silhouette.
+    alpha = np.clip((delta.astype(np.float32) - 3.0) / 18.0, 0.0, 1.0) * core
+    alpha = cv2.GaussianBlur(alpha, (3, 3), 0.55)
+    alpha[core == 0] = 0
+    return fg, alpha, (165.0, 59.0), (x0, y0)
+
+
+def composite_sprite(canvas, rgb, alpha, center, reference_center, patch_origin):
+    """Subpixel translate and alpha composite the source sprite."""
+    ph, pw = alpha.shape
+    # Desired top-left follows from the source patch and source head center.
+    left = center[0] + (patch_origin[0] - reference_center[0])
+    top = center[1] + (patch_origin[1] - reference_center[1])
+    ix, iy = int(np.floor(left)), int(np.floor(top))
+    fx, fy = left - ix, top - iy
+    M = np.float32([[1, 0, fx], [0, 1, fy]])
+    wrgb = cv2.warpAffine(rgb, M, (pw, ph), flags=cv2.INTER_LINEAR,
+                          borderMode=cv2.BORDER_REPLICATE)
+    wa = cv2.warpAffine(alpha, M, (pw, ph), flags=cv2.INTER_LINEAR,
+                        borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+    xa, ya = max(0, ix), max(0, iy)
+    xb, yb = min(W, ix + pw), min(H, iy + ph)
+    if xa >= xb or ya >= yb:
+        return
+    sx0, sy0 = xa - ix, ya - iy
+    a = wa[sy0:sy0 + yb-ya, sx0:sx0 + xb-xa, None].astype(np.float32)
+    fg = wrgb[sy0:sy0 + yb-ya, sx0:sx0 + xb-xa].astype(np.float32)
+    bg = canvas[ya:yb, xa:xb].astype(np.float32)
+    canvas[ya:yb, xa:xb] = np.clip(fg*a + bg*(1-a) + 0.5, 0, 255).astype(np.uint8)
+
+
+def build():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (N, H, W, 3) and source.dtype == np.uint8
+    sprite, alpha, ref_center, patch_origin = extract_blue_sprite(source)
+    output = np.empty_like(source)
+
+    # Fitted incoming center velocity from the isolated source head over frames
+    # 0..14.  Mass does not alter this free motion, only the impact response.
+    contact_center = np.array([146.8, 65.2], dtype=np.float32)
+    incoming_v = np.array([-1.40, 0.47], dtype=np.float32)
+    outgoing_v = incoming_v * BLUE_RATE
+
+    for t in range(N):
+        if t <= int(IMPACT):
+            # Exact factual history through the onset of impact.
+            output[t] = source[t]
+            continue
+
+        # Earlier occurrence of every source-observed downstream collision.
+        source_time = IMPACT + (t - IMPACT) * DOWNSTREAM_RATE
+        frame = lerp_frame(source, source_time)
+        frame = remove_factual_stopped_blue(frame)
+
+        # A short smooth handoff avoids a one-frame positional discontinuity at
+        # contact while quickly converging to the 10m elastic trajectory.
+        dt = t - IMPACT
+        center = contact_center + outgoing_v * dt
+        composite_sprite(frame, sprite, alpha, center, ref_center, patch_origin)
+        output[t] = frame
+
+    return output
+
+
+if __name__ == "__main__":
+    frames = build()
+    save(frames, CASE_DIR)
+    print(f"saved {len(frames)} counterfactual frames to {CASE_DIR}")

        cy, cx = (y1-y0-1)/2, (x1-x0-1)/2
        for i in range(1, n):
            ys, xs = np.where(labels == i)
            scores.append((stats[i, cv2.CC_STAT_AREA] / (1 + (xs.mean()-cx)**2 + (ys.mean()-cy)**2), i))
        core = (labels == max(scores)[1]).astype(np.uint8)
    else:
        core = seed
    core = cv2.dilate(core, np.ones((3, 3), np.uint8), iterations=1)
    # Difference strength retains translucent antialiasing around the silhouette.
    alpha = np.clip((delta.astype(np.float32) - 3.0) / 18.0, 0.0, 1.0) * core
    alpha = cv2.GaussianBlur(alpha, (3, 3), 0.55)
    alpha[core == 0] = 0
    return fg, alpha, (165.0, 59.0), (x0, y0)


def composite_sprite(canvas, rgb, alpha, center, reference_center, patch_origin):
    \"\"\"Subpixel translate and alpha composite the source sprite.\"\"\"
    ph, pw = alpha.shape
    # Desired top-left follows from the source patch and source head center.
    left = center[0] + (patch_origin[0] - reference_center[0])
    top = center[1] + (patch_origin[1] - reference_center[1])
    ix, iy = int(np.floor(left)), int(np.floor(top))
    fx, fy = left - ix, top - iy
    M = np.float32([[1, 0, fx], [0, 1, fy]])
    wrgb = cv2.warpAffine(rgb, M, (pw, ph), flags=cv2.INTER_LINEAR,
                          borderMode=cv2.BORDER_REPLICATE)
    wa = cv2.warpAffine(alpha, M, (pw, ph), flags=cv2.INTER_LINEAR,
                        borderMode=cv2.BORDER_CONSTANT, borderValue=0)
    xa, ya = max(0, ix), max(0, iy)
    xb, yb = min(W, ix + pw), min(H, iy + ph)
    if xa >= xb or ya >= yb:
        return
    sx0, sy0 = xa - ix, ya - iy
    a = wa[sy0:sy0 + yb-ya, sx0:sx0 + xb-xa, None].astype(np.float32)
    fg = wrgb[sy0:sy0 + yb-ya, sx0:sx0 + xb-xa].astype(np.float32)
    bg = canvas[ya:yb, xa:xb].astype(np.float32)
    canvas[ya:yb, xa:xb] = np.clip(fg*a + bg*(1-a) + 0.5, 0, 255).astype(np.uint8)


def build():
    source = np.load(CASE_DIR / \"source_pixels.npz\")[\"rgb\"]
    assert source.shape == (N, H, W, 3) and source.dtype == np.uint8
    sprite, alpha, ref_center, patch_origin = extract_blue_sprite(source)
    output = np.empty_like(source)

    # Fitted incoming center velocity from the isolated source head over frames
    # 0..14.  Mass does not alter this free motion, only the impact response.
    contact_center = np.array([146.8, 65.2], dtype=np.float32)
    incoming_v = np.array([-1.40, 0.47], dtype=np.float32)
    outgoing_v = incoming_v * BLUE_RATE

    for t in range(N):
        if t <= int(IMPACT):
            # Exact factual history through the onset of impact.
            output[t] = source[t]
            continue

        # Earlier occurrence of every source-observed downstream collision.
        source_time = IMPACT + (t - IMPACT) * DOWNSTREAM_RATE
        frame = lerp_frame(source, source_time)
        frame = remove_factual_stopped_blue(frame)

        # A short smooth handoff avoids a one-frame positional discontinuity at
        # contact while quickly converging to the 10m elastic trajectory.
        dt = t - IMPACT
        center = contact_center + outgoing_v * dt
        composite_sprite(frame, sprite, alpha, center, ref_center, patch_origin)
        output[t] = frame

    return output


if __name__ == \"__main__\":
