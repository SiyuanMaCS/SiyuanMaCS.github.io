+"""Generate the 10x-football-mass counterfactual.
+
+The ball is already in free flight at the beginning of the source.  Changing
+mass does not change that ballistic motion; its first visible consequence is
+at the impact near source frame 10.  The source collision supplies the exact
+appearance, occlusions, and direction of every moving part.  We increase the
+post-impact rate along that motion smoothly, modeling the larger impulse from
+the heavier ball without changing its size or any scene texture.
+"""
+
+from pathlib import Path
+
+import numpy as np
+
+import artifact_io
+
+
+N_FRAMES = 96
+CONTACT_FRAME = 10.0
+
+
+def post_impact_source_time(output_frame: int) -> float:
+    """Map an output frame to the accelerated factual collision timeline.
+
+    Before and at contact the mapping is exactly the identity.  Afterwards
+    its derivative eases from 1 to 2.4 over a few frames, avoiding a cut at
+    impact while representing the much larger transferred momentum.  Once
+    the source has settled, holding its final frame is physically appropriate.
+    """
+
+    t = float(output_frame)
+    if t <= CONTACT_FRAME:
+        return t
+
+    dt = t - CONTACT_FRAME
+    speed_gain = 2.4
+    ease_frames = 2.0
+    # Integral of a speed that rises smoothly from 1 to speed_gain.
+    accelerated_dt = dt + (speed_gain - 1.0) * (
+        dt - ease_frames * (1.0 - np.exp(-dt / ease_frames))
+    )
+    return min(CONTACT_FRAME + accelerated_dt, N_FRAMES - 1.0)
+
+
+def sample_linear(source: np.ndarray, time: float) -> np.ndarray:
+    """Sample source RGB frames at fractional time with temporal antialiasing."""
+
+    lo = int(np.floor(time))
+    hi = min(lo + 1, len(source) - 1)
+    alpha = np.float32(time - lo)
+    if hi == lo or alpha == 0:
+        return source[lo].copy()
+    mixed = source[lo].astype(np.float32) * (1.0 - alpha)
+    mixed += source[hi].astype(np.float32) * alpha
+    return np.clip(np.rint(mixed), 0, 255).astype(np.uint8)
+
+
+def generate(case_dir: Path) -> np.ndarray:
+    source = np.load(case_dir / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3)
+    assert source.dtype == np.uint8
+
+    frames = np.empty_like(source)
+    for frame_index in range(N_FRAMES):
+        source_time = post_impact_source_time(frame_index)
+        frames[frame_index] = sample_linear(source, source_time)
+
+    # The intervention cannot affect the factual history through contact.
+    assert np.array_equal(frames[: CONTACT_FRAME.__int__() + 1], source[:11])
+    return frames
+
+
+if __name__ == "__main__":
+    root = Path(__file__).resolve().parent
+    output = generate(root)
+    artifact_io.save(output, root)
+    print("saved", output.shape, output.dtype)
