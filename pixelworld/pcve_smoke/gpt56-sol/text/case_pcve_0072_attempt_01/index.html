+The source has its first floor impact between frames 6 and 7 (zero based).
+Frames through that impact approach are preserved exactly.  Afterwards, the

+def feathered_rect(h, w, x0, y0, x1, y1, feather=3.0):
+    """Soft rectangular alpha, one in the interior and feathered at its edge."""
+    m = np.zeros((h, w), bool)
+    m[y0:y1, x0:x1] = True
+    d = distance_transform_edt(m)
+    return np.clip(d / feather, 0.0, 1.0)

+def die_alpha(frame, expected_y):
+    """Recover the complete red die silhouette, including its dark pips.
+
+    Red face pixels seed the mask. Closing and hole filling include dark pips,
+    while component filtering prevents unrelated red scenery from entering it.
+    """
+    im = frame.astype(np.int16)
+    r, g, b = im[..., 0], im[..., 1], im[..., 2]
+    seed = (r > 82) & (r - g > 23) & (r - b > 16) & (r * 100 > g * 120)
+
+    roi = np.zeros(seed.shape, bool)
+    y0 = max(0, expected_y - 5)
+    y1 = min(seed.shape[0], expected_y + 31)
+    roi[y0:y1, 112:144] = True
+    seed &= roi
+
+    # Link the shaded faces, seal the black pips, then retain central die blobs.
+    mask = binary_closing(seed, iterations=2)
+    mask = binary_fill_holes(mask)
+    labs, count = label(mask)
+    keep = np.zeros_like(mask)
+    for k in range(1, count + 1):
+        yy, xx = np.where(labs == k)
+        if len(xx) >= 12 and xx.min() <= 132 and xx.max() >= 120:
+            keep[labs == k] = True
+    keep = binary_dilation(keep, iterations=1)
+
+    # A softly antialiased boundary avoids cut-out edges while leaving the
+    # actual source pixels (including highlights and pips) untouched inside.
+    return np.clip(gaussian_filter(keep.astype(np.float32), 0.72), 0.0, 1.0)
+
+
+def composite(base, layer, alpha):
+    a = alpha[..., None]
+    return base * (1.0 - a) + layer * a
+

    return np.clip(d / feather, 0.0, 1.0)


def die_alpha(frame, expected_y):
    \"\"\"Recover the complete red die silhouette, including its dark pips.

    Red face pixels seed the mask. Closing and hole filling include dark pips,
    while component filtering prevents unrelated red scenery from entering it.
    \"\"\"
    im = frame.astype(np.int16)
    r, g, b = im[..., 0], im[..., 1], im[..., 2]
    seed = (r > 82) & (r - g > 23) & (r - b > 16) & (r * 100 > g * 120)

    roi = np.zeros(seed.shape, bool)
    y0 = max(0, expected_y - 5)
    y1 = min(seed.shape[0], expected_y + 31)
    roi[y0:y1, 112:144] = True
    seed &= roi

    # Link the shaded faces, seal the black pips, then retain central die blobs.
    mask = binary_closing(seed, iterations=2)
    mask = binary_fill_holes(mask)
    labs, count = label(mask)
    keep = np.zeros_like(mask)
    for k in range(1, count + 1):
        yy, xx = np.where(labs == k)
        if len(xx) >= 12 and xx.min() <= 132 and xx.max() >= 120:
            keep[labs == k] = True
    keep = binary_dilation(keep, iterations=1)

    # A softly antialiased boundary avoids cut-out edges while leaving the
    # actual source pixels (including highlights and pips) untouched inside.
    return np.clip(gaussian_filter(keep.astype(np.float32), 0.72), 0.0, 1.0)


def composite(base, layer, alpha):
    a = alpha[..., None]
    return base * (1.0 - a) + layer * a


def main():
    src = np.load(ROOT / \"source_pixels.npz\")[\"rgb\"]
    assert src.shape == (96, 144, 256, 3) and src.dtype == np.uint8
    h, w = src.shape[1:3]

    # Before the first lighting-state transition, a temporal median produces a
    # clean plate because the falling/bouncing die occupies each height briefly.
    clean_pre = np.median(src[:18].astype(np.float32), axis=0)

    # The factual die occupies only this compact central region. Replacing that
    # region leaves every unaffected source pixel and all scene animation intact.
    erase_alpha = feathered_rect(h, w, 107, 43, 148, 99, feather=3.0)

    out = src.astype(np.float32).copy()

    # Original frames selected at successive floor-contact / settling poses.
    # Their vertical extents remain at the floor, eliminating the factual high
    # rebound while retaining realistic rigid-die rotation and face appearance.
    settle_sources = [16, 21, 22, 23, 24, 25, 26, 27, 28, 29, 29]
    expected_tops = [68, 68, 70, 69, 69, 70, 70, 70, 70, 70, 70]

    # Frame 0 through frame 6 are copied exactly: restitution cannot alter the
    # trajectory until the collision response. Frames 7--17 depict the changed
    # response and rapid non-bouncing settle.
    for t, (s, top) in enumerate(zip(settle_sources, expected_tops), start=7):
        base = composite(out[t], clean_pre, erase_alpha)
        alpha = die_alpha(src[s], top)
        out[t] = composite(base, src[s].astype(np.float32), alpha)

    # At frame 18 the source changes illumination state. The die is already at
    # rest in the counterfactual. A softly feathered settled source patch both
    # removes the still-bouncing factual die and preserves the exact final pose.
    settled_patch_alpha = feathered_rect(h, w, 108, 48, 146, 98, feather=3.0)
    for t in range(18, 29):
        out[t] = composite(out[t], src[29].astype(np.float32), settled_patch_alpha)

    frames = np.clip(np.rint(out), 0, 255).astype(np.uint8)

    # Contract and intervention checks.
    assert frames.shape == (96, 144, 256, 3)
    assert frames.dtype == np.uint8
    assert np.array_equal(frames[:7], src[:7])
    assert np.array_equal(frames[29:], src[29:])

    artifact_io.save(frames, ROOT)
