+"""Generate the 5x-mass counterfactual for this PixelWorld case.
+
+The source is a locked-off 256x144 shot.  A newly introduced tennis ball rolls
+left across an elevated surface, falls, and strikes a stationary tennis ball
+at floor level between source frames 21 and 22.  Mass does not change the
+pre-impact ballistic path, so those frames are retained exactly.  After the
+impact, source-derived sprites are moved according to the unequal-mass
+collision: the heavy incoming ball continues left and the light target ball
+is kicked much faster to the left.
+"""
+
+from pathlib import Path
+
+import numpy as np
+from scipy.ndimage import binary_dilation, gaussian_filter
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def clean_floor_plate(src):
+    """Recover the static floor behind both source balls.
+
+    A temporal median removes every moving traversal.  The two balls remain
+    at their final source positions for more than half the clip, so those two
+    small areas are replaced by an early-frame median, when they were empty.
+    """
+    plate = np.median(src, axis=0).astype(np.uint8)
+    early = np.median(src[:14], axis=0).astype(np.uint8)
+    # Final resting place of the struck ball (left).
+    plate[81:108, 0:46] = early[81:108, 0:46]
+    # Final resting place of the incoming ball (right).
+    plate[81:108, 94:132] = early[81:108, 94:132]
+    return plate
+
+
+def weighted_center(frame, box, kind):
+    """Color-weighted tennis-ball center within an expected tracking box."""
+    x0, y0, x1, y1 = box
+    p = frame[y0:y1, x0:x1].astype(np.float32)
+    r, g, b = p[..., 0], p[..., 1], p[..., 2]
+    if kind == "incoming":
+        # The new ball is a bright yellow-green felt.
+        w = np.maximum(0.0, g - b - 42.0) * np.maximum(0.0, g - r + 5.0)
+    else:
+        # The stationary ball is a slightly duller, less yellow green.
+        w = (np.maximum(0.0, g - r - 7.0)
+             * np.maximum(0.0, g - b - 14.0)
+             * np.clip((b - 42.0) / 55.0, 0.0, 1.0))
+    yy, xx = np.mgrid[y0:y1, x0:x1]
+    total = float(w.sum())
+    if total < 1.0:
+        return np.array([(x0 + x1 - 1) / 2, (y0 + y1 - 1) / 2], np.float32)
+    return np.array([(xx * w).sum() / total, (yy * w).sum() / total], np.float32)
+
+
+def source_tracks(src):
+    """Track both source balls from the first separated post-impact frame."""
+    n = len(src)
+    incoming = np.zeros((n, 2), np.float32)
+    target = np.zeros((n, 2), np.float32)
+    incoming[21] = (74.7, 90.9)
+    target[21] = (66.5, 94.3)
+
+    # Broad expected boxes are sufficient because the two felt colors differ
+    # and, from frame 22 onward, the objects move in opposite directions.
+    for t in range(22, n):
+        incoming[t] = weighted_center(src[t], (70, 82, 125, 104), "incoming")
+        target[t] = weighted_center(src[t], (12, 86, 70, 104), "target")
+    return incoming, target
+
+
+def counterfactual_tracks(n):
+    """Unequal-mass post-impact trajectories in image coordinates.
+
+    For a 5:1 collision, the impulse delivered to the light target is 5/3 of
+    the equal-mass transfer, while the incoming ball receives only one third
+    of its former velocity change.  Thus the target leaves frame left and the
+    incoming ball retains a modest leftward velocity.  Rolling resistance is
+    applied after contact; floor-normal motion is critically damped over the
+    small bounces visible in this scene.
+    """
+    heavy = np.full((n, 2), np.nan, np.float32)
+    light = np.full((n, 2), np.nan, np.float32)
+    heavy[21] = (74.7, 90.9)
+    light[21] = (66.5, 94.3)
+
+    # Post-impact horizontal state, pixels per frame.  The acceleration is
+    # rolling resistance and opposes the current direction.
+    hx, hv = 74.7, -2.55
+    lx, lv = 66.5, -14.2
+    # Handled as a short, damped floor bounce.  Values are ball-center y.
+    bounce_y = [91.6, 89.8, 89.5, 90.7, 92.8, 94.0,
+                92.5, 92.3, 93.2, 94.0, 93.5, 94.0]
+
+    for t in range(22, n):
+        hx += hv
+        if hv < 0:
+            hv = min(0.0, hv + 0.42)
+        # The light ball has the larger impulse.  It remains floor-constrained
+        # and leaves the image after a few frames.
+        lx += lv
+        if lv < 0:
+            lv = min(0.0, lv + 0.95)
+        k = t - 22
+        hy = bounce_y[k] if k < len(bounce_y) else 94.0
+        heavy[t] = (hx, hy)
+        light[t] = (lx, 94.3)
+    return heavy, light
+
+
+def sprite_rgba(frame, plate, center, kind):
+    """Extract a compact source sprite, preserving felt detail and shadow."""
+    cx, cy = np.rint(center).astype(int)
+    radius = 11
+    x0, x1 = cx - radius, cx + radius + 1
+    y0, y1 = cy - radius, cy + radius + 1
+    # Tracks stay far enough from vertical borders and all source sprites are
+    # horizontally on-screen in the frames from which we extract them.
+    raw = frame[y0:y1, x0:x1].copy()
+    bg = plate[y0:y1, x0:x1]
+    p = raw.astype(np.int16)
+    diff = np.max(np.abs(p - bg.astype(np.int16)), axis=2)
+    r, g, b = p[..., 0], p[..., 1], p[..., 2]
+    if kind == "incoming":
+        felt = (g > b + 18) & (g > r - 2)
+    else:
+        felt = (g > b + 12) & (g > r + 5)
+    yy, xx = np.mgrid[-radius:radius + 1, -radius:radius + 1]
+    support = (xx / 9.5) ** 2 + (yy / 9.0) ** 2 < 1.0
+    seed = felt & support
+    # Include antialiased perimeter and the small contact shadow, but not the
+    # source background carried by the crop.
+    mask = binary_dilation(seed, iterations=3) & (diff > 3)
+    mask |= seed
+    alpha = gaussian_filter(mask.astype(np.float32), 0.65)
+    alpha = np.clip(alpha, 0.0, 1.0)
+    return raw, alpha
+
+
+def restore_source_ball(frame, plate, center):
+    """Feather a clean plate over one original source-ball footprint."""
+    cx, cy = np.rint(center).astype(int)
+    r = 12
+    x0, x1 = max(0, cx - r), min(frame.shape[1], cx + r + 1)
+    y0, y1 = max(0, cy - r), min(frame.shape[0], cy + r + 1)
+    yy, xx = np.mgrid[y0:y1, x0:x1]
+    # Slight right/down extension includes the floor contact shadow.
+    d = ((xx - (cx + 0.7)) / 10.2) ** 2 + ((yy - (cy + 0.5)) / 9.5) ** 2
+    hard = d < 1.0
+    alpha = gaussian_filter(hard.astype(np.float32), 0.9)[..., None]
+    a = alpha
+    out = frame[y0:y1, x0:x1].astype(np.float32)
+    bg = plate[y0:y1, x0:x1].astype(np.float32)
+    frame[y0:y1, x0:x1] = np.clip(out * (1 - a) + bg * a, 0, 255).astype(np.uint8)
+
+
+def paste_sprite(frame, sprite, alpha, center):
+    """Alpha-composite a sprite at a possibly edge-clipped center."""
+    h, w = sprite.shape[:2]
+    cx, cy = np.rint(center).astype(int)
+    x0, y0 = cx - w // 2, cy - h // 2
+    dx0, dy0 = max(0, x0), max(0, y0)
+    dx1, dy1 = min(frame.shape[1], x0 + w), min(frame.shape[0], y0 + h)
+    if dx0 >= dx1 or dy0 >= dy1:
+        return
+    sx0, sy0 = dx0 - x0, dy0 - y0
+    sx1, sy1 = sx0 + (dx1 - dx0), sy0 + (dy1 - dy0)
+    a = alpha[sy0:sy1, sx0:sx1, None]
+    fg = sprite[sy0:sy1, sx0:sx1].astype(np.float32)
+    bg = frame[dy0:dy1, dx0:dx1].astype(np.float32)
+    frame[dy0:dy1, dx0:dx1] = np.clip(fg * a + bg * (1 - a), 0, 255).astype(np.uint8)
+
+
+def generate():
+    src = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert src.shape == (96, 144, 256, 3) and src.dtype == np.uint8
+    plate = clean_floor_plate(src)
+    src_heavy, src_light = source_tracks(src)
+    cf_heavy, cf_light = counterfactual_tracks(len(src))
+    out = src.copy()
+
+    # Frame 21 is the instant of first contact and remains the common causal
+    # state.  The visibly different response starts on frame 22.
+    for t in range(22, len(src)):
+        heavy_sprite, heavy_alpha = sprite_rgba(src[t], plate, src_heavy[t], "incoming")
+        light_sprite, light_alpha = sprite_rgba(src[t], plate, src_light[t], "target")
+
+        frame = out[t]
+        restore_source_ball(frame, plate, src_heavy[t])
+        restore_source_ball(frame, plate, src_light[t])
+
+        # Draw the fast light ball first and the retained heavy ball second.
+        # They separate immediately, but this order is also physically correct
+        # for the brief near-contact interval in this camera view.
+        paste_sprite(frame, light_sprite, light_alpha, cf_light[t])
+        paste_sprite(frame, heavy_sprite, heavy_alpha, cf_heavy[t])
+
+    return out
+
+
+if __name__ == "__main__":
+    frames = generate()
+    assert frames.shape == (96, 144, 256, 3)
+    assert frames.dtype == np.uint8
+    # Required benchmark serialization (pixels, PNG sequence, inspections,
+    # and the local HTML player).
+    artifact_io.save(frames, CASE_DIR)
+    print("saved", frames.shape, frames.dtype,
+          "changed_pixels", int(np.any(frames != np.load(CASE_DIR / "source_pixels.npz")["rgb"], axis=3).sum()))
