+"""Generate the requested 0.75x-initial-speed counterfactual video.
+
+The source is a deterministic rendered rigid-body scene.  Slowing the launch
+from the first frame is equivalent to retiming the complete source state by
+0.75 while keeping the requested 4 s output clock.  This preserves all scene
+appearance and correctly delays the car's ballistic/contact trajectory and
+its settling.  Linear RGB interpolation provides smooth 24 fps subframes.
+"""
+from pathlib import Path
+import numpy as np
+import artifact_io
+
+
+def generate(case_dir: Path) -> np.ndarray:
+    source = np.load(case_dir / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3) and source.dtype == np.uint8
+
+    # Frame 1 is t=0.  For v' = 0.75 v, evaluate the source trajectory at
+    # source-time 0.75*t.  Keep frame 1 exact and interpolate fractional times.
+    output_times = np.arange(96, dtype=np.float64) * 0.75
+    lo = np.floor(output_times).astype(np.int32)
+    hi = np.minimum(lo + 1, source.shape[0] - 1)
+    alpha = (output_times - lo).astype(np.float32)[:, None, None, None]
+    frames = np.rint(
+        source[lo].astype(np.float32) * (1.0 - alpha)
+        + source[hi].astype(np.float32) * alpha
+    ).clip(0, 255).astype(np.uint8)
+
+    assert np.array_equal(frames[0], source[0])
+    return frames
+
+
+if __name__ == "__main__":
+    case_dir = Path(__file__).resolve().parent
+    frames = generate(case_dir)
+    artifact_io.save(frames, case_dir)
+    print(f"saved {len(frames)} frames, shape={frames.shape}, dtype={frames.dtype}")

import numpy as np
import artifact_io


def generate(case_dir: Path) -> np.ndarray:
