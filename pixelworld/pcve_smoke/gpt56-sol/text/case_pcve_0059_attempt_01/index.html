+"""Generate the 5x-mass milk-carton counterfactual.
+
+The factual clip is a one-dimensional chain impact.  A red can travels right,
+contacts the milk carton at frame 3/4, and in the source the resulting motion
+continues into the green bottle.  With carton mass 5m (can mass m), the
+one-dimensional elastic impact coefficients are
+
+    v_can'    = (m-5m)/(m+5m) * u = -2u/3
+    v_carton' = 2m/(m+5m) * u     =  u/3.
+
+We retain the exact source through contact and then use those directions with
+surface friction.  The heavier carton moves only modestly, so it never reaches
+the downstream bottle.  Pixels for the objects, their antialiased edges, and
+their contact shadows are taken directly from the source; only translation is
+synthesized.
+"""
+from pathlib import Path
+import cv2
+import numpy as np
+
+import artifact_io
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def object_mask(frame0, clean_reference, roi, threshold=5):
+    """Return a source-derived soft mask for the object/shadow in *roi*."""
+    x0, y0, x1, y1 = roi
+    src = frame0[y0:y1, x0:x1]
+    ref = clean_reference[y0:y1, x0:x1]
+    delta = np.max(np.abs(src.astype(np.int16) - ref.astype(np.int16)), axis=2)
+
+    # Strong differences define the object. Include connected antialiased edge
+    # and shadow pixels, while rejecting isolated renderer noise.
+    raw = (delta > threshold).astype(np.uint8)
+    n, labels, stats, _ = cv2.connectedComponentsWithStats(raw, 8)
+    keep = np.zeros_like(raw)
+    for i in range(1, n):
+        if stats[i, cv2.CC_STAT_AREA] >= 2:
+            keep[labels == i] = 1
+    keep = cv2.morphologyEx(keep, cv2.MORPH_CLOSE, np.ones((3, 3), np.uint8))
+
+    # A one-pixel feather retains the source antialiasing without a cutout edge.
+    soft = cv2.GaussianBlur(keep.astype(np.float32), (3, 3), 0.45)
+    soft[keep > 0] = 1.0
+    return src.copy(), soft
+
+
+def composite_shifted(dst, sprite, alpha, roi, dx, dy=0.0):
+    """Translate and composite a small source sprite at subpixel precision."""
+    x0, y0, x1, y1 = roi
+    h, w = sprite.shape[:2]
+    # Premultiply before warping so transparent edge colors cannot bleed.
+    premul = sprite.astype(np.float32) * alpha[..., None]
+    M = np.float32([[1, 0, x0 + dx], [0, 1, y0 + dy]])
+    warped_rgb = cv2.warpAffine(
+        premul, M, (dst.shape[1], dst.shape[0]), flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT, borderValue=0,
+    )
+    warped_a = cv2.warpAffine(
+        alpha, M, (dst.shape[1], dst.shape[0]), flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT, borderValue=0,
+    )
+    a = np.clip(warped_a, 0.0, 1.0)[..., None]
+    out = dst.astype(np.float32) * (1.0 - a) + warped_rgb
+    return np.clip(np.rint(out), 0, 255).astype(np.uint8)
+
+
+def friction_displacement(t, velocity, deceleration):
+    """Signed displacement after t frames under Coulomb-like friction."""
+    if t <= 0:
+        return 0.0
+    sign = 1.0 if velocity >= 0 else -1.0
+    speed = abs(velocity)
+    stop_t = speed / deceleration
+    tau = min(float(t), stop_t)
+    return sign * (speed * tau - 0.5 * deceleration * tau * tau)
+
+
+def generate():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3) and source.dtype == np.uint8
+
+    # Frame 34 is the first fully settled source frame. At the initial can and
+    # carton positions it therefore supplies the clean, previously hidden set.
+    frame0 = source[0]
+    settled = source[34]
+
+    # Broad enough to retain each object's little floor/contact shadow.
+    can_roi = (14, 59, 42, 88)
+    carton_roi = (84, 52, 117, 90)
+    can_sprite, can_alpha = object_mask(frame0, settled, can_roi, threshold=5)
+    carton_sprite, carton_alpha = object_mask(frame0, settled, carton_roi, threshold=5)
+
+    # Static factual scene with the initial can and carton removed. The green
+    # bottle and every unrelated scene pixel remain exactly as at time zero.
+    base = frame0.copy().astype(np.float32)
+    for roi, alpha in ((can_roi, can_alpha), (carton_roi, carton_alpha)):
+        x0, y0, x1, y1 = roi
+        clean = settled[y0:y1, x0:x1].astype(np.float32)
+        a = alpha[..., None]
+        base[y0:y1, x0:x1] = base[y0:y1, x0:x1] * (1.0 - a) + clean * a
+    base = np.clip(np.rint(base), 0, 255).astype(np.uint8)
+
+    result = np.empty_like(source)
+    # Exact factual approach and first-contact image. Mass has no visible effect
+    # until the collision impulse has been applied between frames 3 and 4.
+    result[:4] = source[:4]
+
+    # Frame 0 anchors: the source can advances ~19 px/frame and reaches contact
+    # at dx=57 in frame 3. The trajectories below begin at that contact state.
+    can_contact_dx = 57.0
+    carton_contact_dx = 0.0
+
+    for f in range(4, 96):
+        t = f - 3
+        # Post-impact velocities follow the 1:5 collision response, then decay
+        # under table friction. The can rebounds and the carton settles around
+        # x=127, far short of the bottle at x=166.
+        can_dx = can_contact_dx + friction_displacement(t, -10.0, 1.25)
+        carton_dx = carton_contact_dx + friction_displacement(t, 6.2, 0.55)
+
+        frame = base.copy()
+        # Back-to-front order preserves contact shadows naturally.
+        frame = composite_shifted(frame, carton_sprite, carton_alpha, carton_roi, carton_dx)
+        frame = composite_shifted(frame, can_sprite, can_alpha, can_roi, can_dx)
+        result[f] = frame
+
+    return result
+
+
+if __name__ == "__main__":
+    frames = generate()
+    artifact_io.save(frames, CASE_DIR)
+    print("saved", frames.shape, frames.dtype, "to", CASE_DIR)

    out = dst.astype(np.float32) * (1.0 - a) + warped_rgb
    return np.clip(np.rint(out), 0, 255).astype(np.uint8)


def friction_displacement(t, velocity, deceleration):
    \"\"\"Signed displacement after t frames under Coulomb-like friction.\"\"\"
    if t <= 0:
        return 0.0
    sign = 1.0 if velocity >= 0 else -1.0
    speed = abs(velocity)
    stop_t = speed / deceleration
    tau = min(float(t), stop_t)
    return sign * (speed * tau - 0.5 * deceleration * tau * tau)


def generate():
    source = np.load(CASE_DIR / \"source_pixels.npz\")[\"rgb\"]
    assert source.shape == (96, 144, 256, 3) and source.dtype == np.uint8

    # Frame 34 is the first fully settled source frame. At the initial can and
    # carton positions it therefore supplies the clean, previously hidden set.
    frame0 = source[0]
    settled = source[34]

    # Broad enough to retain each object's little floor/contact shadow.
    can_roi = (14, 59, 42, 88)
    carton_roi = (84, 52, 117, 90)
    can_sprite, can_alpha = object_mask(frame0, settled, can_roi, threshold=5)
    carton_sprite, carton_alpha = object_mask(frame0, settled, carton_roi, threshold=5)

    # Static factual scene with the initial can and carton removed. The green
    # bottle and every unrelated scene pixel remain exactly as at time zero.
    base = frame0.copy().astype(np.float32)
    for roi, alpha in ((can_roi, can_alpha), (carton_roi, carton_alpha)):
        x0, y0, x1, y1 = roi
        clean = settled[y0:y1, x0:x1].astype(np.float32)
        a = alpha[..., None]
        base[y0:y1, x0:x1] = base[y0:y1, x0:x1] * (1.0 - a) + clean * a
    base = np.clip(np.rint(base), 0, 255).astype(np.uint8)

    result = np.empty_like(source)
    # Exact factual approach and first-contact image. Mass has no visible effect
    # until the collision impulse has been applied between frames 3 and 4.
    result[:4] = source[:4]

    # Frame 0 anchors: the source can advances ~19 px/frame and reaches contact
    # at dx=57 in frame 3. The trajectories below begin at that contact state.
    can_contact_dx = 57.0
    carton_contact_dx = 0.0

    for f in range(4, 96):
        t = f - 3
        # Post-impact velocities follow the 1:5 collision response, then decay
        # under table friction. The can rebounds and the carton settles around
        # x=127, far short of the bottle at x=166.
        can_dx = can_contact_dx + friction_displacement(t, -10.0, 1.25)
        carton_dx = carton_contact_dx + friction_displacement(t, 6.2, 0.55)

        frame = base.copy()
        # Back-to-front order preserves contact shadows naturally.
        frame = composite_shifted(frame, carton_sprite, carton_alpha, carton_roi, carton_dx)
        frame = composite_shifted(frame, can_sprite, can_alpha, can_roi, can_dx)
        result[f] = frame

    return result


if __name__ == \"__main__\":
    frames = generate()
    artifact_io.save(frames, CASE_DIR)
