+"""Generate the block-removed counterfactual video using only source RGB pixels."""
+from pathlib import Path
+import numpy as np
+from scipy import ndimage
+import artifact_io
+
+CASE_DIR = Path(__file__).resolve().parent
+

+def make_clean_background(src):
+    """Assemble views of the stationary table revealed at different times.
+
+    The ball and the pushed block travel left-to-right.  These time/column
+    selections use only portions known to be behind both objects.  Two tiny
+    never-clean column bands are continued from adjacent table texture.
+    """
+    bg = np.empty_like(src[0])
+    bg[:, :136] = src[95, :, :136]       # block and ball have both cleared
+    bg[:, 136:146] = src[95, :, 126:136] # continue underlying wood texture
+    bg[:, 146:160] = src[24, :, 146:160] # gap behind ball, ahead of block
+    bg[:, 160:178] = src[32, :, 160:178] # newly exposed gap
+    bg[:, 178:183] = bg[:, 173:178]       # narrow occluded seam
+    bg[:, 183:] = src[0, :, 183:]         # ahead of the original block
+
+    # Clear any nearly invisible vertical joins while retaining the detailed
+    # source texture. Only one column on either side is mixed.
+    for seam in (136, 146, 160, 178, 183):
+        if 1 <= seam < 255:
+            left = bg[:, seam - 2].astype(np.uint16)
+            right = bg[:, seam + 1].astype(np.uint16)
+            bg[:, seam - 1] = ((2 * left + right) // 3).astype(np.uint8)
+            bg[:, seam] = ((left + 2 * right) // 3).astype(np.uint8)
+    return bg
+
+
+def ball_mask(frame, bg, cx, cy):
+    """Extract the ball and its contact shadow from an unobstructed frame."""
+    yy, xx = np.mgrid[:frame.shape[0], :frame.shape[1]]
+    # The rendered sphere/shadow is about 46 px wide, with a very soft fringe.
+    region = ((xx - cx) / 25.0) ** 2 + ((yy - cy) / 24.0) ** 2 <= 1.0
+    delta = np.max(np.abs(frame.astype(np.int16) - bg.astype(np.int16)), axis=2)
+    core = region & (delta >= 5)
+
+    # Reject isolated table-noise pixels but retain the connected soft shadow.
+    labels, count = ndimage.label(core)
+    keep = np.zeros_like(core)
+    for index in range(1, count + 1):
+        component = labels == index
+        size = int(component.sum())
+        if size >= 8:
+            ys, xs = np.where(component)
+            if np.min((xs - cx) ** 2 + (ys - cy) ** 2) < 22 ** 2:
+                keep |= component
+    keep = ndimage.binary_fill_holes(keep)
+
+    # Include antialiased neighbors only when they really differ from the
+    # recovered table.  This avoids both a hard cutout and a background halo.
+    fringe = ndimage.binary_dilation(keep, iterations=1) & region & (delta >= 3)
+    return keep | fringe
+
+
+def paste_translated(dst, sprite, mask, dx, dy):
+    """Translate an RGB sprite/mask with exact clipping at the image edge."""
+    h, w = dst.shape[:2]
+    sy0, sy1 = max(0, -dy), min(h, h - dy)
+    sx0, sx1 = max(0, -dx), min(w, w - dx)
+    if sy0 >= sy1 or sx0 >= sx1:
+        return
+    submask = mask[sy0:sy1, sx0:sx1]
+    target = dst[sy0 + dy:sy1 + dy, sx0 + dx:sx1 + dx]
+    source = sprite[sy0:sy1, sx0:sx1]
+    target[submask] = source[submask]
+
+
+def generate():
+    src = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert src.shape == (96, 144, 256, 3) and src.dtype == np.uint8
+    bg = make_clean_background(src)
+    frames = np.repeat(bg[None], 96, axis=0)
+
+    # Before contact the sphere's center follows this constant screen-space
+    # velocity. With the wooden block absent it simply keeps that velocity and
+    # leaves the right edge, rather than slowing and pushing/toppling the block.
+    center = lambda t: (22 + 7 * t, 61 + (3 * t) // 2)
+
+    # Exact source appearances for the fully unobstructed approach.
+    masks = {}
+    for t in range(0, 9):
+        cx, cy = center(t)
+        masks[t] = ball_mask(src[t], bg, cx, cy)
+        paste_translated(frames[t], src[t], masks[t], 0, 0)
+
+    # Frame 8 is the last generously separated, complete view of the sphere.
+    # Reusing its rendered appearance preserves source lighting and shadow;
+    # translation supplies the counterfactual continuation.
+    sprite_t = 8
+    sx, sy = center(sprite_t)
+    sprite_mask = masks[sprite_t]
+    for t in range(9, 96):
+        cx, cy = center(t)
+        if cx - 26 > 255 or cy - 25 > 143:
+            continue
+        paste_translated(frames[t], src[sprite_t], sprite_mask,
+                         int(cx - sx), int(cy - sy))
+    return frames
+
+
+if __name__ == "__main__":

+    artifact_io.save(result, CASE_DIR)
+    print("saved", result.shape, result.dtype)
