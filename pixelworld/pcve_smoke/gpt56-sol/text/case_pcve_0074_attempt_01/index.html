+"""Generate the requested counterfactual PixelWorld video.
+
+The camera and environment are static in this case.  We recover a clean plate from
+frames in which the die is elsewhere, extract source-rendered die silhouettes, and
+move those silhouettes along a rigid-body trajectory whose *linear* initial speed
+is three times the measured source value.  Gravity and bounce restitution are
+measured from the original path.  Source shadow deltas are selected by height so
+that the edited die remains grounded in the original lighting.
+"""
+from pathlib import Path
+import math
+import numpy as np
+from PIL import Image, ImageDraw, ImageFilter
+import artifact_io
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def convex_hull(points):
+    """Andrew monotone chain; points are (x,y)."""
+    pts = sorted(set((int(x), int(y)) for x, y in points))
+    if len(pts) <= 1:
+        return pts
+    def cross(o, a, b):
+        return (a[0]-o[0])*(b[1]-o[1]) - (a[1]-o[1])*(b[0]-o[0])
+    lo = []
+    for p in pts:
+        while len(lo) >= 2 and cross(lo[-2], lo[-1], p) <= 0:
+            lo.pop()
+        lo.append(p)
+    hi = []
+    for p in reversed(pts):
+        while len(hi) >= 2 and cross(hi[-2], hi[-1], p) <= 0:
+            hi.pop()
+        hi.append(p)
+    return lo[:-1] + hi[:-1]
+
+
+def physics_path(v0, frames=96, y0=13.3614, floor=82.5,
+                 gravity=1.5357, restitution=0.43):
+    """Semi-continuous vertical rigid-body path, in pixels per source frame."""
+    y, v = float(y0), float(v0)
+    out = [y]
+    impacts = []
+    # Fine steps make impacts occur between video samples instead of one frame late.
+    steps = 200
+    dt = 1.0 / steps
+    resting = False
+    for frame in range(1, frames):
+        for j in range(steps):
+            if resting:
+                y, v = floor, 0.0
+                continue
+            v += gravity * dt
+            y += v * dt
+            if y >= floor and v > 0:
+                y = floor
+                impacts.append((frame - 1 + (j + 1) * dt, v))
+                v = -restitution * v
+                # Sub-pixel hops are visually a stationary die.
+                if abs(v) < 1.45:
+                    resting = True
+                    v = 0.0
+        out.append(y)
+    return np.asarray(out, np.float32), impacts
+
+
+def source_phase_index(t, impact_times):
+    """Select source orientation from the corresponding collision phase."""
+    # Approximate source contact events measured from the die's vertical extent.
+    src_events = (6.18, 16.25, 21.15, 25.2, 29.0)
+    dst_events = tuple(q[0] for q in impact_times[:4]) + (29.0,)
+    if t <= dst_events[0]:
+        # Preserve angular time before the altered first impact.
+        return min(6, int(round(t)))
+    for k in range(4):
+        a = dst_events[k]
+        b = dst_events[k+1]
+        if a <= t <= b:
+            u = (t-a) / max(b-a, 1e-6)
+            s = src_events[k] + u * (src_events[k+1]-src_events[k])
+            return int(np.clip(round(s), 0, 30))
+    return 30
+
+
+def make_clean_plate(src):
+    # Frame 1 is clear below the falling die.  A late frame is clear at the top.
+    plate = src[0].copy()
+    plate[:38, 104:152] = src[40, :38, 104:152]
+    return plate
+
+
+def die_mask(frame, plate):
+    """Soft convex silhouette around the opaque die, excluding its cast shadow."""
+    delta = np.max(np.abs(frame.astype(np.int16)-plate.astype(np.int16)), axis=2)
+    yy, xx = np.where((delta > 7) &
+                      (np.indices(delta.shape)[1] >= 114) &
+                      (np.indices(delta.shape)[1] <= 144) &
+                      (np.indices(delta.shape)[0] <= 95))
+    if len(xx) < 12:
+        raise RuntimeError("Could not recover die silhouette")
+    hull = convex_hull(zip(xx, yy))
+    m = Image.new("L", (256, 144), 0)
+    ImageDraw.Draw(m).polygon(hull, fill=255)
+    # A small feather keeps original anti-aliased edges without a hard cutout halo.
+    return np.asarray(m.filter(ImageFilter.GaussianBlur(0.65)), dtype=np.float32) / 255.0
+
+
+def shifted_composite(base, sprite, alpha, dy):
+    """Integer translate with alpha compositing and tabletop-front occlusion."""
+    shift = int(round(float(dy)))
+    out = base.astype(np.float32)
+    ys0, ys1 = max(0, -shift), min(144, 144-shift)
+    yd0, yd1 = ys0+shift, ys1+shift
+    if ys1 <= ys0:
+        return base
+    a = alpha[ys0:ys1]
+    pix = sprite[ys0:ys1].astype(np.float32)
+    # The die may meet but cannot penetrate the tabletop contact line.
+    if yd1 > 96:
+        keep = max(0, 96-yd0)
+        a = a[:keep]
+        pix = pix[:keep]
+        yd1 = yd0 + keep
+    if yd1 > yd0:
+        aa = a[..., None]
+        out[yd0:yd1] = pix*aa + out[yd0:yd1]*(1.0-aa)
+    return np.clip(out, 0, 255).astype(np.uint8)
+
+
+def add_shadow(base, src, plate, source_index, target_y, source_centers, settled):
+    """Transfer only the source cast-shadow delta onto the clean table plate."""
+    # Shadow appearance depends mainly on elevation. For a settled die use its
+    # stable source shadow; otherwise choose the original sample at equal height.
+    if settled:
+        si = 30
+    else:
+        candidates = np.arange(0, 31)
+        # Prefer the descent/first bounce samples and penalize phase mismatch lightly.
+        score = np.abs(source_centers[candidates] - target_y)
+        si = int(candidates[np.argmin(score)])
+    # The cast shadow lies on the table behind/in front of the contact line.
+    y0, y1, x0, x1 = 92, 123, 88, 168
+    delta = src[si, y0:y1, x0:x1].astype(np.int16) - plate[y0:y1, x0:x1].astype(np.int16)
+    # Exclude bright/object pixels and retain neutral darkening only.  Taper the
+    # upper rows so the shadow does not overwrite an airborne die.
+    lum = delta.mean(axis=2)
+    weight = np.clip((-lum - 1.0) / 9.0, 0.0, 1.0)
+    neutral = (np.max(delta, axis=2)-np.min(delta, axis=2) < 13)
+    weight *= neutral
+    delta = np.minimum(delta, 0)
+    reg = base[y0:y1, x0:x1].astype(np.float32)
+    reg += delta.astype(np.float32) * weight[..., None]
+    out = base.copy()
+    out[y0:y1, x0:x1] = np.clip(reg, 0, 255).astype(np.uint8)
+    return out
+
+
+def main():
+    src = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert src.shape == (96, 144, 256, 3) and src.dtype == np.uint8
+    plate = make_clean_plate(src)
+
+    # Least-squares fit to source descent centers: y=13.3614+6.4093*t+.7679*t^2.
+    gravity = 1.5357
+    original_v0 = 6.4093
+    source_centers, source_impacts = physics_path(original_v0, gravity=gravity)
+    target_centers, target_impacts = physics_path(3.0*original_v0, gravity=gravity)
+
+    masks = [die_mask(src[i], plate) for i in range(31)]
+    out = np.empty_like(src)
+    out[0] = src[0]  # Velocity intervention does not change the t=0 pose.
+
+    for t in range(1, 96):
+        # Preserve the factual timeline everywhere except the die/shadow corridor.
+        frame = src[t].copy()
+        frame[0:97, 104:152] = plate[0:97, 104:152]
+        frame[92:123, 88:168] = plate[92:123, 88:168]
+
+        settled = t >= 29 or abs(target_centers[t]-82.5) < 0.05
+        si = source_phase_index(float(t), target_impacts)
+        frame = add_shadow(frame, src, plate, si, float(target_centers[t]),
+                           source_centers, settled)
+        dy = float(target_centers[t] - source_centers[si])
+        frame = shifted_composite(frame, src[si], masks[si], dy)
+        out[t] = frame
+
+    artifact_io.save(out, CASE_DIR)
+    print("saved", out.shape, out.dtype)
+    print("source impacts:", [(round(a,2), round(b,2)) for a,b in source_impacts[:4]])
+    print("target impacts:", [(round(a,2), round(b,2)) for a,b in target_impacts[:4]])
+    print("target centers 0..30:", np.round(target_centers[:31], 1).tolist())
+
+
+if __name__ == "__main__":
+    main()

The camera and environment are static in this case.  We recover a clean plate from
frames in which the die is elsewhere, extract source-rendered die silhouettes, and
move those silhouettes along a rigid-body trajectory whose *linear* initial speed
is three times the measured source value.  Gravity and bounce restitution are
measured from the original path.  Source shadow deltas are selected by height so
that the edited die remains grounded in the original lighting.
\"\"\"
from pathlib import Path
import math
import numpy as np
from PIL import Image, ImageDraw, ImageFilter
import artifact_io

CASE_DIR = Path(__file__).resolve().parent


def convex_hull(points):
    \"\"\"Andrew monotone chain; points are (x,y).\"\"\"
    pts = sorted(set((int(x), int(y)) for x, y in points))
    if len(pts) <= 1:
        return pts
    def cross(o, a, b):
        return (a[0]-o[0])*(b[1]-o[1]) - (a[1]-o[1])*(b[0]-o[0])
    lo = []
    for p in pts:
        while len(lo) >= 2 and cross(lo[-2], lo[-1], p) <= 0:
            lo.pop()
        lo.append(p)
    hi = []
    for p in reversed(pts):
        while len(hi) >= 2 and cross(hi[-2], hi[-1], p) <= 0:
            hi.pop()
        hi.append(p)
    return lo[:-1] + hi[:-1]


def physics_path(v0, frames=96, y0=13.3614, floor=82.5,
                 gravity=1.5357, restitution=0.43):
    \"\"\"Semi-continuous vertical rigid-body path, in pixels per source frame.\"\"\"
    y, v = float(y0), float(v0)
    out = [y]
    impacts = []
    # Fine steps make impacts occur between video samples instead of one frame late.
    steps = 200
    dt = 1.0 / steps
    resting = False
    for frame in range(1, frames):
        for j in range(steps):
            if resting:
                y, v = floor, 0.0
                continue
            v += gravity * dt
            y += v * dt
            if y >= floor and v > 0:
                y = floor
                impacts.append((frame - 1 + (j + 1) * dt, v))
                v = -restitution * v
                # Sub-pixel hops are visually a stationary die.
                if abs(v) < 1.45:
                    resting = True
                    v = 0.0
        out.append(y)
    return np.asarray(out, np.float32), impacts


def source_phase_index(t, impact_times):
    \"\"\"Select source orientation from the corresponding collision phase.\"\"\"
    # Approximate source contact events measured from the die's vertical extent.
    src_events = (6.18, 16.25, 21.15, 25.2, 29.0)
    dst_events = tuple(q[0] for q in impact_times[:4]) + (29.0,)
    if t <= dst_events[0]:
        # Preserve angular time before the altered first impact.
        return min(6, int(round(t)))
    for k in range(4):
        a = dst_events[k]
        b = dst_events[k+1]
        if a <= t <= b:
            u = (t-a) / max(b-a, 1e-6)
            s = src_events[k] + u * (src_events[k+1]-src_events[k])
            return int(np.clip(round(s), 0, 30))
    return 30


def make_clean_plate(src):
    # Frame 1 is clear below the falling die.  A late frame is clear at the top.
    plate = src[0].copy()
    plate[:38, 104:152] = src[40, :38, 104:152]
    return plate


def die_mask(frame, plate):
    \"\"\"Soft convex silhouette around the opaque die, excluding its cast shadow.\"\"\"
    delta = np.max(np.abs(frame.astype(np.int16)-plate.astype(np.int16)), axis=2)
    yy, xx = np.where((delta > 7) &
                      (np.indices(delta.shape)[1] >= 114) &
                      (np.indices(delta.shape)[1] <= 144) &
                      (np.indices(delta.shape)[0] <= 95))
    if len(xx) < 12:
        raise RuntimeError(\"Could not recover die silhouette\")
    hull = convex_hull(zip(xx, yy))
    m = Image.new(\"L\", (256, 144), 0)
    ImageDraw.Draw(m).polygon(hull, fill=255)
    # A small feather keeps original anti-aliased edges without a hard cutout halo.
    return np.asarray(m.filter(ImageFilter.GaussianBlur(0.65)), dtype=np.float32) / 255.0


def shifted_composite(base, sprite, alpha, dy):
    \"\"\"Integer translate with alpha compositing and tabletop-front occlusion.\"\"\"
    shift = int(round(float(dy)))
    out = base.astype(np.float32)
    ys0, ys1 = max(0, -shift), min(144, 144-shift)
    yd0, yd1 = ys0+shift, ys1+shift
    if ys1 <= ys0:
        return base
    a = alpha[ys0:ys1]
    pix = sprite[ys0:ys1].astype(np.float32)
    # The die may meet but cannot penetrate the tabletop contact line.
    if yd1 > 96:
        keep = max(0, 96-yd0)
        a = a[:keep]
        pix = pix[:keep]
        yd1 = yd0 + keep
    if yd1 > yd0:
        aa = a[..., None]
        out[yd0:yd1] = pix*aa + out[yd0:yd1]*(1.0-aa)
    return np.clip(out, 0, 255).astype(np.uint8)


def add_shadow(base, src, plate, source_index, target_y, source_centers, settled):
    \"\"\"Transfer only the source cast-shadow delta onto the clean table plate.\"\"\"
    # Shadow appearance depends mainly on elevation. For a settled die use its
    # stable source shadow; otherwise choose the original sample at equal height.
    if settled:
        si = 30
    else:
        candidates = np.arange(0, 31)
        # Prefer the descent/first bounce samples and penalize phase mismatch lightly.
        score = np.abs(source_centers[candidates] - target_y)
        si = int(candidates[np.argmin(score)])
    # The cast shadow lies on the table behind/in front of the contact line.
    y0, y1, x0, x1 = 92, 123, 88, 168
    delta = src[si, y0:y1, x0:x1].astype(np.int16) - plate[y0:y1, x0:x1].astype(np.int16)
    # Exclude bright/object pixels and retain neutral darkening only.  Taper the
    # upper rows so the shadow does not overwrite an airborne die.
    lum = delta.mean(axis=2)
    weight = np.clip((-lum - 1.0) / 9.0, 0.0, 1.0)
    neutral = (np.max(delta, axis=2)-np.min(delta, axis=2) < 13)
    weight *= neutral
    delta = np.minimum(delta, 0)
    reg = base[y0:y1, x0:x1].astype(np.float32)
    reg += delta.astype(np.float32) * weight[..., None]
    out = base.copy()
    out[y0:y1, x0:x1] = np.clip(reg, 0, 255).astype(np.uint8)
    return out


def main():
    src = np.load(CASE_DIR / \"source_pixels.npz\")[\"rgb\"]
    assert src.shape == (96, 144, 256, 3) and src.dtype == np.uint8
    plate = make_clean_plate(src)

    # Least-squares fit to source descent centers: y=13.3614+6.4093*t+.7679*t"'^2.
    gravity = 1.5357
    original_v0 = 6.4093
    source_centers, source_impacts = physics_path(original_v0, gravity=gravity)
    target_centers, target_impacts = physics_path(3.0*original_v0, gravity=gravity)

    masks = [die_mask(src[i], plate) for i in range(31)]
    out = np.empty_like(src)
    out[0] = src[0]  # Velocity intervention does not change the t=0 pose.

    for t in range(1, 96):
        # Preserve the factual timeline everywhere except the die/shadow corridor.
        frame = src[t].copy()
        frame[0:97, 104:152] = plate[0:97, 104:152]
        frame[92:123, 88:168] = plate[92:123, 88:168]

        settled = t >= 29 or abs(target_centers[t]-82.5) < 0.05
        si = source_phase_index(float(t), target_impacts)
        frame = add_shadow(frame, src, plate, si, float(target_centers[t]),
                           source_centers, settled)
        dy = float(target_centers[t] - source_centers[si])
        frame = shifted_composite(frame, src[si], masks[si], dy)
        out[t] = frame

    artifact_io.save(out, CASE_DIR)
    print("saved", out.shape, out.dtype)
    print("source impacts:", [(round(a,2), round(b,2)) for a,b in source_impacts[:4]])
    print("target impacts:", [(round(a,2), round(b,2)) for a,b in target_impacts[:4]])
    print("target centers 0..30:", np.round(target_centers[:31], 1).tolist())


if __name__ == "__main__":
    main()
