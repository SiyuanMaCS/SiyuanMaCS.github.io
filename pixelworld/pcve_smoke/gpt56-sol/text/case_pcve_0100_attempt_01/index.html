+from pathlib import Path
+import numpy as np
+from scipy.ndimage import label, binary_closing, binary_dilation, gaussian_filter
+import artifact_io
+
+CASE_DIR = Path(__file__).resolve().parent
+

+def green_components(frame):
+    """Return the two tennis-ball green-core components, largest first."""
+    p = frame.astype(np.int16)
+    r, g, b = p[..., 0], p[..., 1], p[..., 2]
+    mask = (g > r + 10) & (g > b + 10) & (np.indices(g.shape)[0] >= 15)
+    labs, count = label(mask)
+    result = []
+    for k in range(1, count + 1):
+        yy, xx = np.where(labs == k)
+        if len(xx) >= 20:
+            result.append((len(xx), float(xx.mean()), float(yy.mean())))
+    return sorted(result, reverse=True)
+
+
+def track_balls(src):
+    """Track original floor ball (left) and the newly arriving ball (right)."""
+    n = len(src)
+    floor_ball = np.zeros((n, 2), np.float64)
+    new_ball = np.zeros((n, 2), np.float64)
+    for t, frame in enumerate(src):
+        comps = green_components(frame)
+        centers = np.array([[c[1], c[2]] for c in comps[:4]], dtype=float)
+        if t <= 20:
+            # The old ball is the low component around (66,94); the new ball approaches it.
+            i_old = np.argmin(np.sum((centers - [66.2, 94.1]) ** 2, axis=1))
+            floor_ball[t] = centers[i_old]
+            candidates = np.delete(centers, i_old, axis=0)
+            i_new = np.argmin(np.sum((candidates - ([78, 78] if t == 20 else [225-7.5*t, 24])) ** 2, axis=1))
+            new_ball[t] = candidates[i_new]
+        else:
+            # After impact the old floor ball travels left and the new ball rebounds right.
+            by_x = centers[np.argsort(centers[:, 0])]
+            floor_ball[t] = by_x[0]
+            new_ball[t] = by_x[-1]
+    return floor_ball, new_ball
+
+
+def raised_cosine_ellipse(shape, center, rx, ry, feather=2.0):
+    yy, xx = np.indices(shape)
+    d = np.sqrt(((xx-center[0])/rx)**2 + ((yy-center[1])/ry)**2)
+    # 1 through the body, smooth over the outer feather fraction.
+    inner = max(0.0, 1.0 - feather / max(rx, ry))
+    a = np.clip((1.0-d) / max(1e-6, 1.0-inner), 0.0, 1.0)
+    return a
+
+
+def best_clean_reference(src, old_c, new_c, point, candidates=(0, 10, 40, 95)):
+    """Select a source frame whose two balls are farthest from point."""
+    best, best_score = candidates[0], -1.0
+    for r in candidates:
+        # Y is weighted because both balls are small and the floor is nearly horizontal.
+        d0 = np.hypot(old_c[r,0]-point[0], 1.25*(old_c[r,1]-point[1]))
+        d1 = np.hypot(new_c[r,0]-point[0], 1.25*(new_c[r,1]-point[1]))
+        score = min(d0, d1)
+        if score > best_score:
+            best, best_score = r, score
+    return src[best]
+
+
+def erase_ball(canvas, src, old_c, new_c, center, rx=13.0, ry=11.0):
+    ref = best_clean_reference(src, old_c, new_c, center)
+    alpha = raised_cosine_ellipse(canvas.shape[:2], center, rx, ry, feather=3.0)[..., None]
+    return np.clip(canvas*(1-alpha) + ref*alpha, 0, 255)
+
+
+def make_sprite(src, old_c, new_c, frame_index, which):
+    """Extract an authentic anti-aliased ball sprite using a clean temporal plate."""
+    center = old_c[frame_index] if which == 'old' else new_c[frame_index]
+    clean = best_clean_reference(src, old_c, new_c, center)
+    frame = src[frame_index]
+    h, w = frame.shape[:2]
+    x0 = max(0, int(np.floor(center[0]-14))); x1 = min(w, int(np.ceil(center[0]+15)))
+    y0 = max(0, int(np.floor(center[1]-13))); y1 = min(h, int(np.ceil(center[1]+13)))
+    patch = frame[y0:y1, x0:x1].copy()
+    plate = clean[y0:y1, x0:x1]
+    cy, cx = center[1]-y0, center[0]-x0
+    yy, xx = np.indices(patch.shape[:2])
+    # Temporal difference isolates body, anti-aliasing, and the physically rendered shadow.
+    diff = np.max(np.abs(patch.astype(np.int16)-plate.astype(np.int16)), axis=2)
+    local = (((xx-cx)/10.5)**2 + ((yy-cy)/10.0)**2) <= 1.0
+    seed = (diff > 7) & local
+    seed |= ((((xx-cx)/7.2)**2 + ((yy-cy)/7.2)**2) <= 1.0) & (diff > 3)
+    seed = binary_closing(seed, iterations=1)
+    seed = binary_dilation(seed, iterations=1)
+    alpha = gaussian_filter(seed.astype(np.float32), 0.65)
+    alpha *= ((((xx-cx)/11.0)**2 + ((yy-cy)/10.5)**2) < 1.0)
+    alpha = np.clip(alpha, 0, 1)
+    return patch, alpha, (cx, cy)
+
+
+def paste_sprite(canvas, sprite, center):
+    patch, alpha, anchor = sprite
+    x0 = int(round(center[0] - anchor[0])); y0 = int(round(center[1] - anchor[1]))
+    x1, y1 = x0 + patch.shape[1], y0 + patch.shape[0]
+    H, W = canvas.shape[:2]
+    dx0, dy0 = max(0, x0), max(0, y0)
+    dx1, dy1 = min(W, x1), min(H, y1)
+    if dx0 >= dx1 or dy0 >= dy1:
+        return canvas
+    sx0, sy0 = dx0-x0, dy0-y0
+    sx1, sy1 = sx0+(dx1-dx0), sy0+(dy1-dy0)
+    a = alpha[sy0:sy1, sx0:sx1, None]
+    p = patch[sy0:sy1, sx0:sx1]
+    canvas[dy0:dy1, dx0:dx1] = canvas[dy0:dy1, dx0:dx1]*(1-a) + p*a
+    return canvas
+
+
+def simulate_lighter_ball():
+    """Counterfactual post-impact centers for the quarter-mass incoming ball."""
+    centers = {21: np.array([76.16, 90.67], dtype=float)}
+    p = centers[21].copy()
+    # Reduced mass changes collision impulse: J'=0.4J for an equal-mass original,
+    # while acceleration of the quarter-mass ball from that impulse is 1.6x.
+    v = np.array([9.35, -11.10], dtype=float)
+    g = 2.0
+    settled = False
+    for t in range(22, 96):
+        if not settled:
+            v[1] += g
+            p += v
+            v[0] *= 0.997
+            if p[1] >= 94.0:
+                p[1] = 94.0
+                if v[1] > 2.1:
+                    v[1] = -0.46*v[1]
+                    v[0] *= 0.82
+                else:
+                    v[1] = 0.0
+                    settled = True
+        else:
+            p[1] = 94.0
+            if v[0] > 0:
+                v[0] = max(0.0, v[0]-0.43)
+            elif v[0] < 0:
+                v[0] = min(0.0, v[0]+0.43)
+            p[0] += v[0]
+        centers[t] = p.copy()
+    return centers
+
+
+def simulate_struck_ball(start):
+    """The heavy floor ball gets 40% of the equal-mass collision impulse."""
+    centers = {21: np.array(start, dtype=float)}
+    p = centers[21].copy()
+    vx = -2.85
+    for t in range(22, 96):
+        p[0] += vx
+        # Rolling/sliding deceleration is mass independent.
+        vx = min(0.0, vx + 0.52)
+        p[1] = 94.1
+        centers[t] = p.copy()
+    return centers
+
+
+def choose_air_sprite(src, old_c, new_c, desired_y, phase):
+    # Frames 0..20 provide unobstructed authentic views throughout the fall.
+    valid = np.arange(0, 21)
+    # Avoid the contact frame and select by apparent elevation; phase breaks ties for rotation.
+    cost = np.abs(new_c[valid,1] - desired_y) + 0.025*np.abs(valid-(20-phase%20))
+    k = int(valid[np.argmin(cost)])
+    return make_sprite(src, old_c, new_c, k, 'new')
+
+
+def generate():
+    src = np.load(CASE_DIR / 'source_pixels.npz')['rgb']
+    assert src.shape == (96, 144, 256, 3) and src.dtype == np.uint8
+    old_c, new_c = track_balls(src)
+    output = src.copy().astype(np.float32)
+    cf_new = simulate_lighter_ball()
+    cf_old = simulate_struck_ball(old_c[21])
+
+    # Frames 0..20 are factual history: mass has no visual effect before impact.
+    # Frame 21 is the contact image and anchors a seamless branch in the timeline.
+    for t in range(22, 96):
+        canvas = output[t]
+        # Remove both factual post-impact trajectories using clean source plates.
+        canvas = erase_ball(canvas, src, old_c, new_c, old_c[t], 13.5, 11.0)
+        canvas = erase_ball(canvas, src, old_c, new_c, new_c[t], 13.5, 11.0)
+
+        # The struck floor ball rolls only a short distance and stops near x=56.
+        old_src_t = min(35, 23 + (t-22))
+        old_sprite = make_sprite(src, old_c, new_c, old_src_t, 'old')
+        canvas = paste_sprite(canvas, old_sprite, cf_old[t])
+
+        # The lighter new ball rebounds high to the right. Use airborne source sprites
+        # away from the floor and rendered post-impact sprites for contact/bounce frames.
+        q = cf_new[t]
+        if q[1] < 86.5:
+            new_sprite = choose_air_sprite(src, old_c, new_c, q[1], t-21)
+        else:
+            new_src_t = 23 + ((t-22) % 8)
+            new_sprite = make_sprite(src, old_c, new_c, new_src_t, 'new')
+        canvas = paste_sprite(canvas, new_sprite, q)
+        output[t] = canvas
+
+    return np.clip(np.rint(output), 0, 255).astype(np.uint8)
+
+
+if __name__ == '__main__':
+    frames = generate()
+    artifact_io.save(frames, CASE_DIR)

from pathlib import Path
import numpy as np
from scipy.ndimage import label, binary_closing, binary_dilation, gaussian_filter
import artifact_io

CASE_DIR = Path(__file__).resolve().parent


def green_components(frame):
    \"\"\"Return the two tennis-ball green-core components, largest first.\"\"\"
    p = frame.astype(np.int16)
    r, g, b = p[..., 0], p[..., 1], p[..., 2]
    mask = (g > r + 10) & (g > b + 10) & (np.indices(g.shape)[0] >= 15)
    labs, count = label(mask)
    result = []
    for k in range(1, count + 1):
        yy, xx = np.where(labs == k)
        if len(xx) >= 20:
            result.append((len(xx), float(xx.mean()), float(yy.mean())))
    return sorted(result, reverse=True)


def track_balls(src):
    \"\"\"Track original floor ball (left) and the newly arriving ball (right).\"\"\"
    n = len(src)
    floor_ball = np.zeros((n, 2), np.float64)
    new_ball = np.zeros((n, 2), np.float64)
    for t, frame in enumerate(src):
        comps = green_components(frame)
        centers = np.array([[c[1], c[2]] for c in comps[:4]], dtype=float)
        if t <= 20:
            # The old ball is the low component around (66,94); the new ball approaches it.
            i_old = np.argmin(np.sum((centers - [66.2, 94.1]) ** 2, axis=1))
            floor_ball[t] = centers[i_old]
            candidates = np.delete(centers, i_old, axis=0)
            i_new = np.argmin(np.sum((candidates - ([78, 78] if t == 20 else [225-7.5*t, 24])) ** 2, axis=1))
            new_ball[t] = candidates[i_new]
        else:
            # After impact the old floor ball travels left and the new ball rebounds right.
            by_x = centers[np.argsort(centers[:, 0])]
            floor_ball[t] = by_x[0]
            new_ball[t] = by_x[-1]
    return floor_ball, new_ball


def raised_cosine_ellipse(shape, center, rx, ry, feather=2.0):
    yy, xx = np.indices(shape)
    d = np.sqrt(((xx-center[0])/rx)**2 + ((yy-center[1])/ry)**2)
    # 1 through the body, smooth over the outer feather fraction.
    inner = max(0.0, 1.0 - feather / max(rx, ry))
    a = np.clip((1.0-d) / max(1e-6, 1.0-inner), 0.0, 1.0)
    return a


def best_clean_reference(src, old_c, new_c, point, candidates=(0, 10, 40, 95)):
    \"\"\"Select a source frame whose two balls are farthest from point.\"\"\"
    best, best_score = candidates[0], -1.0
    for r in candidates:
        # Y is weighted because both balls are small and the floor is nearly horizontal.
        d0 = np.hypot(old_c[r,0]-point[0], 1.25*(old_c[r,1]-point[1]))
        d1 = np.hypot(new_c[r,0]-point[0], 1.25*(new_c[r,1]-point[1]))
        score = min(d0, d1)
        if score > best_score:
            best, best_score = r, score
    return src[best]


def erase_ball(canvas, src, old_c, new_c, center, rx=13.0, ry=11.0):
    ref = best_clean_reference(src, old_c, new_c, center)
    alpha = raised_cosine_ellipse(canvas.shape[:2], center, rx, ry, feather=3.0)[..., None]
    return np.clip(canvas*(1-alpha) + ref*alpha, 0, 255)


def make_sprite(src, old_c, new_c, frame_index, which):
    \"\"\"Extract an authentic anti-aliased ball sprite using a clean temporal plate.\"\"\"
    center = old_c[frame_index] if which == 'old' else new_c[frame_index]
    clean = best_clean_reference(src, old_c, new_c, center)
    frame = src[frame_index]
    h, w = frame.shape[:2]
    x0 = max(0, int(np.floor(center[0]-14))); x1 = min(w, int(np.ceil(center[0]+15)))
    y0 = max(0, int(np.floor(center[1]-13))); y1 = min(h, int(np.ceil(center[1]+13)))
    patch = frame[y0:y1, x0:x1].copy()
    plate = clean[y0:y1, x0:x1]
    cy, cx = center[1]-y0, center[0]-x0
    yy, xx = np.indices(patch.shape[:2])
    # Temporal difference isolates body, anti-aliasing, and the physically rendered shadow.
    diff = np.max(np.abs(patch.astype(np.int16)-plate.astype(np.int16)), axis=2)
    local = (((xx-cx)/10.5)**2 + ((yy-cy)/10.0)**2) <= 1.0
    seed = (diff > 7) & local
    seed |= ((((xx-cx)/7.2)**2 + ((yy-cy)/7.2)**2) <= 1.0) & (diff > 3)
    seed = binary_closing(seed, iterations=1)
    seed = binary_dilation(seed, iterations=1)
    alpha = gaussian_filter(seed.astype(np.float32), 0.65)
    alpha *= ((((xx-cx)/11.0)**2 + ((yy-cy)/10.5)**2) < 1.0)
    alpha = np.clip(alpha, 0, 1)
    return patch, alpha, (cx, cy)


def paste_sprite(canvas, sprite, center):
    patch, alpha, anchor = sprite
    x0 = int(round(center[0] - anchor[0])); y0 = int(round(center[1] - anchor[1]))
    x1, y1 = x0 + patch.shape[1], y0 + patch.shape[0]
    H, W = canvas.shape[:2]
    dx0, dy0 = max(0, x0), max(0, y0)
    dx1, dy1 = min(W, x1), min(H, y1)
    if dx0 >= dx1 or dy0 >= dy1:
        return canvas
    sx0, sy0 = dx0-x0, dy0-y0
    sx1, sy1 = sx0+(dx1-dx0), sy0+(dy1-dy0)
    a = alpha[sy0:sy1, sx0:sx1, None]
    p = patch[sy0:sy1, sx0:sx1]
    canvas[dy0:dy1, dx0:dx1] = canvas[dy0:dy1, dx0:dx1]*(1-a) + p*a
    return canvas


def simulate_lighter_ball():
    \"\"\"Counterfactual post-impact centers for the quarter-mass incoming ball.\"\"\"
    centers = {21: np.array([76.16, 90.67], dtype=float)}
    p = centers[21].copy()
    # Reduced mass changes collision impulse: J'=0.4J for an equal-mass original,
    # while acceleration of the quarter-mass ball from that impulse is 1.6x.
    v = np.array([9.35, -11.10], dtype=float)
    g = 2.0
    settled = False
    for t in range(22, 96):
        if not settled:
            v[1] += g
            p += v
            v[0] *= 0.997
            if p[1] >= 94.0:
                p[1] = 94.0
                if v[1] > 2.1:
                    v[1] = -0.46*v[1]
                    v[0] *= 0.82
                else:
                    v[1] = 0.0
                    settled = True
        else:
            p[1] = 94.0
            if v[0] > 0:
                v[0] = max(0.0, v[0]-0.43)
            elif v[0] < 0:
                v[0] = min(0.0, v[0]+0.43)
            p[0] += v[0]
        centers[t] = p.copy()
    return centers


def simulate_struck_ball(start):
    \"\"\"The heavy floor ball gets 40% of the equal-mass collision impulse.\"\"\"
    centers = {21: np.array(start, dtype=float)}
    p = centers[21].copy()
    vx = -2.85
    for t in range(22, 96):
        p[0] += vx
        # Rolling/sliding deceleration is mass independent.
        vx = min(0.0, vx + 0.52)
        p[1] = 94.1
        centers[t] = p.copy()
    return centers


def choose_air_sprite(src, old_c, new_c, desired_y, phase):
    # Frames 0..20 provide unobstructed authentic views throughout the fall.
    valid = np.arange(0, 21)
    # Avoid the contact frame and select by apparent elevation; phase breaks ties for rotation.
    cost = np.abs(new_c[valid,1] - desired_y) + 0.025*np.abs(valid-(20-phase%20))
    k = int(valid[np.argmin(cost)])
    return make_sprite(src, old_c, new_c, k, 'new')


def generate():
    src = np.load(CASE_DIR / 'source_pixels.npz')['rgb']
    assert src.shape == (96, 144, 256, 3) and src.dtype == np.uint8
    old_c, new_c = track_balls(src)
    output = src.copy().astype(np.float32)
    cf_new = simulate_lighter_ball()
    cf_old = simulate_struck_ball(old_c[21])

    # Frames 0..20 are factual history: mass has no visual effect before impact.
    # Frame 21 is the contact image and anchors a seamless branch in the timeline.
    for t in range(22, 96):
        canvas = output[t]
        # Remove both factual post-impact trajectories using clean source plates.
        canvas = erase_ball(canvas, src, old_c, new_c, old_c[t], 13.5, 11.0)
        canvas = erase_ball(canvas, src, old_c, new_c, new_c[t], 13.5, 11.0)

        # The struck floor ball rolls only a short distance and stops near x=56.
        old_src_t = min(35, 23 + (t-22))
        old_sprite = make_sprite(src, old_c, new_c, old_src_t, 'old')
        canvas = paste_sprite(canvas, old_sprite, cf_old[t])

        # The lighter new ball rebounds high to the right. Use airborne source sprites
        # away from the floor and rendered post-impact sprites for contact/bounce frames.
        q = cf_new[t]
        if q[1] < 86.5:
            new_sprite = choose_air_sprite(src, old_c, new_c, q[1], t-21)
        else:
            new_src_t = 23 + ((t-22) % 8)
            new_sprite = make_sprite(src, old_c, new_c, new_src_t, 'new')
        canvas = paste_sprite(canvas, new_sprite, q)
        output[t] = canvas

    return np.clip(np.rint(output), 0, 255).astype(np.uint8)


if __name__ == '__main__':
    frames = generate()
    artifact_io.save(frames, CASE_DIR)
