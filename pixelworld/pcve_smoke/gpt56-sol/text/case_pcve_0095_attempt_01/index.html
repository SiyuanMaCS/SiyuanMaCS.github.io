+"""Remove the blue marble from frame 1 and simulate the red marble without impact.
+
+The source is a nearly static, real-camera shot.  We retain every source frame, use
+per-frame OpenCV inpainting only for the two marble footprints, and translate an
+RGBA-like cutout of the red marble along the least-squares continuation of its
+pre-contact ballistic path.  All geometry is in the 256x144 source pixel grid.
+"""
+from pathlib import Path
+import cv2
+import numpy as np
+from scipy import ndimage
+import artifact_io
+
+CASE_DIR = Path(__file__).resolve().parent
+
+# Stable blue-marble centre measured from the blue-dominant connected component.
+BLUE_CENTER = (24.5, 87.7)
+BLUE_RADIUS = 10.2
+
+
+def disk_mask(height, width, cx, cy, radius, feather=0.0):
+    yy, xx = np.mgrid[:height, :width]
+    distance = np.hypot(xx - cx, yy - cy)
+    if feather <= 0:
+        return distance <= radius
+    return np.clip((radius + feather - distance) / feather, 0.0, 1.0)
+
+
+def detect_red_marble(frame):
+    """Return red marble centre from its distinctive chroma component."""
+    rgb = frame.astype(np.int16)
+    r, g, b = rgb[..., 0], rgb[..., 1], rgb[..., 2]
+    red = (r - g > 55) & (r - b > 45) & (r > 80)
+    labels, count = ndimage.label(red)
+    candidates = []
+    for label in range(1, count + 1):
+        yy, xx = np.where(labels == label)
+        if len(xx) >= 80 and yy.mean() < 110:
+            candidates.append((len(xx), xx.mean(), yy.mean()))
+    if not candidates:
+        raise RuntimeError("red marble was not detected")
+    _, cx, cy = max(candidates)
+    return float(cx), float(cy)
+
+
+def inpaint_rgb(frame, mask, radius=3):
+    # OpenCV expects BGR, though the operation is channel-independent.
+    bgr = cv2.cvtColor(frame, cv2.COLOR_RGB2BGR)
+    repaired = cv2.inpaint(bgr, mask.astype(np.uint8) * 255, radius,
+                           cv2.INPAINT_NS)
+    return cv2.cvtColor(repaired, cv2.COLOR_BGR2RGB)
+
+
+def main():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    if source.shape != (96, 144, 256, 3) or source.dtype != np.uint8:
+        raise ValueError((source.shape, source.dtype))
+    n, h, w, _ = source.shape
+    frames = source.copy()
+
+    # The blue marble is stationary until impact and then exits left.  Remove its
+    # complete silhouette (including antialiasing and its tight contact shadow)
+    # only while that footprint is present, frames 0..26.
+    blue_mask = disk_mask(h, w, *BLUE_CENTER, BLUE_RADIUS).astype(bool)
+    for t in range(27):
+        frames[t] = inpaint_rgb(frames[t], blue_mask, radius=3)
+
+    # Measure the incoming red ball through the last clean pre-contact frame.
+    # Fit constant acceleration; this is exceptionally accurate for the source
+    # trajectory and gives the no-collision continuation off the left boundary.
+    observed_t = np.arange(21, dtype=np.float64)  # source frames 0..20
+    observed = np.array([detect_red_marble(source[t]) for t in range(21)])
+    fit_x = np.polyfit(observed_t, observed[:, 0], 2)
+    fit_y = np.polyfit(observed_t, observed[:, 1], 2)
+
+    # Use frame 20 as a sharp, unobstructed cutout.  Its 19x21 patch contains the
+    # whole marble.  Estimate alpha by color distance from frame 30's clean plate
+    # at the same position; this retains highlight, shading, and edge antialiasing.
+    template_t = 20
+    tcx, tcy = observed[template_t]
+    half = 11
+    x0, x1 = int(round(tcx)) - half, int(round(tcx)) + half + 1
+    y0, y1 = int(round(tcy)) - half, int(round(tcy)) + half + 1
+    foreground = source[template_t, y0:y1, x0:x1].astype(np.float32)
+    clean_plate = source[30, y0:y1, x0:x1].astype(np.float32)
+    difference = np.max(np.abs(foreground - clean_plate), axis=2)
+    alpha = np.clip((difference - 3.0) / 18.0, 0.0, 1.0)
+    alpha = cv2.GaussianBlur(alpha, (3, 3), 0.48)
+    # Constrain the cutout geometrically to avoid carrying unrelated texture.
+    ph, pw = alpha.shape
+    alpha *= disk_mask(ph, pw, (pw - 1) / 2, (ph - 1) / 2,
+                       9.8, feather=1.3)
+
+    # Source frame 21 already contains the correct incoming marble.  Replace the
+    # post-contact segment beginning at 22.  By frame 25 the ballistic marble is
+    # wholly outside the left edge, so later source frames need no red edit.
+    for t in range(22, 26):
+        # Remove all factual collision remnants: red marble at its deflected/stopped
+        # location, blue marble (already addressed), and local contact shadows.
+        # One compact mask spans their overlapping silhouettes in these 4 frames.
+        original_red = detect_red_marble(source[t])
+        collision_mask = disk_mask(h, w, *original_red, 11.0).astype(bool)
+        collision_mask |= blue_mask
+        canvas = inpaint_rgb(frames[t], collision_mask, radius=3).astype(np.float32)
+
+        cx = float(np.polyval(fit_x, t))
+        cy = float(np.polyval(fit_y, t))
+        left = int(round(cx)) - half
+        top = int(round(cy)) - half
+        right, bottom = left + pw, top + ph
+        dx0, dy0 = max(0, left), max(0, top)
+        dx1, dy1 = min(w, right), min(h, bottom)
+        if dx1 > dx0 and dy1 > dy0:
+            sx0, sy0 = dx0 - left, dy0 - top
+            sx1, sy1 = sx0 + (dx1 - dx0), sy0 + (dy1 - dy0)
+            a = alpha[sy0:sy1, sx0:sx1, None]
+            fg = foreground[sy0:sy1, sx0:sx1]
+            canvas[dy0:dy1, dx0:dx1] = (
+                fg * a + canvas[dy0:dy1, dx0:dx1] * (1.0 - a)
+            )
+        frames[t] = np.clip(np.rint(canvas), 0, 255).astype(np.uint8)
+
+    artifact_io.save(frames, CASE_DIR)
+
+
+if __name__ == "__main__":
+    main()

\"\"\"Remove the blue marble from frame 1 and simulate the red marble without impact.

The source is a nearly static, real-camera shot.  We retain every source frame, use
per-frame OpenCV inpainting only for the two marble footprints, and translate an
RGBA-like cutout of the red marble along the least-squares continuation of its
pre-contact ballistic path.  All geometry is in the 256x144 source pixel grid.
\"\"\"
from pathlib import Path
import cv2
import numpy as np
from scipy import ndimage
import artifact_io

CASE_DIR = Path(__file__).resolve().parent

# Stable blue-marble centre measured from the blue-dominant connected component.
BLUE_CENTER = (24.5, 87.7)
BLUE_RADIUS = 10.2


def disk_mask(height, width, cx, cy, radius, feather=0.0):
    yy, xx = np.mgrid[:height, :width]
    distance = np.hypot(xx - cx, yy - cy)
    if feather <= 0:
        return distance <= radius
    return np.clip((radius + feather - distance) / feather, 0.0, 1.0)


def detect_red_marble(frame):
    \"\"\"Return red marble centre from its distinctive chroma component.\"\"\"
    rgb = frame.astype(np.int16)
    r, g, b = rgb[..., 0], rgb[..., 1], rgb[..., 2]
    red = (r - g > 55) & (r - b > 45) & (r > 80)
    labels, count = ndimage.label(red)
    candidates = []
    for label in range(1, count + 1):
        yy, xx = np.where(labels == label)
        if len(xx) >= 80 and yy.mean() < 110:
            candidates.append((len(xx), xx.mean(), yy.mean()))
    if not candidates:
        raise RuntimeError(\"red marble was not detected\")
    _, cx, cy = max(candidates)
    return float(cx), float(cy)


def inpaint_rgb(frame, mask, radius=3):
    # OpenCV expects BGR, though the operation is channel-independent.
    bgr = cv2.cvtColor(frame, cv2.COLOR_RGB2BGR)
    repaired = cv2.inpaint(bgr, mask.astype(np.uint8) * 255, radius,
                           cv2.INPAINT_NS)
    return cv2.cvtColor(repaired, cv2.COLOR_BGR2RGB)


def main():
    source = np.load(CASE_DIR / \"source_pixels.npz\")[\"rgb\"]
    if source.shape "'!= (96, 144, 256, 3) or source.dtype != np.uint8:
        raise ValueError((source.shape, source.dtype))
    n, h, w, _ = source.shape
    frames = source.copy()

    # The blue marble is stationary until impact and then exits left.  Remove its
    # complete silhouette (including antialiasing and its tight contact shadow)
    # only while that footprint is present, frames 0..26.
    blue_mask = disk_mask(h, w, *BLUE_CENTER, BLUE_RADIUS).astype(bool)
    for t in range(27):
        frames[t] = inpaint_rgb(frames[t], blue_mask, radius=3)

    # Measure the incoming red ball through the last clean pre-contact frame.
    # Fit constant acceleration; this is exceptionally accurate for the source
    # trajectory and gives the no-collision continuation off the left boundary.
    observed_t = np.arange(21, dtype=np.float64)  # source frames 0..20
    observed = np.array([detect_red_marble(source[t]) for t in range(21)])
    fit_x = np.polyfit(observed_t, observed[:, 0], 2)
    fit_y = np.polyfit(observed_t, observed[:, 1], 2)

    # Use frame 20 as a sharp, unobstructed cutout.  Its 19x21 patch contains the
    # whole marble.  Estimate alpha by color distance from frame 30'"'s clean plate
    # at the same position; this retains highlight, shading, and edge antialiasing.
    template_t = 20
    tcx, tcy = observed[template_t]
    half = 11
    x0, x1 = int(round(tcx)) - half, int(round(tcx)) + half + 1
    y0, y1 = int(round(tcy)) - half, int(round(tcy)) + half + 1
    foreground = source[template_t, y0:y1, x0:x1].astype(np.float32)
    clean_plate = source[30, y0:y1, x0:x1].astype(np.float32)
    difference = np.max(np.abs(foreground - clean_plate), axis=2)
    alpha = np.clip((difference - 3.0) / 18.0, 0.0, 1.0)
    alpha = cv2.GaussianBlur(alpha, (3, 3), 0.48)
    # Constrain the cutout geometrically to avoid carrying unrelated texture.
    ph, pw = alpha.shape
    alpha *= disk_mask(ph, pw, (pw - 1) / 2, (ph - 1) / 2,
                       9.8, feather=1.3)

    # Source frame 21 already contains the correct incoming marble.  Replace the
    # post-contact segment beginning at 22.  By frame 25 the ballistic marble is
    # wholly outside the left edge, so later source frames need no red edit.
    for t in range(22, 26):
        # Remove all factual collision remnants: red marble at its deflected/stopped
        # location, blue marble (already addressed), and local contact shadows.
        # One compact mask spans their overlapping silhouettes in these 4 frames.
        original_red = detect_red_marble(source[t])
        collision_mask = disk_mask(h, w, *original_red, 11.0).astype(bool)
        collision_mask |= blue_mask
        canvas = inpaint_rgb(frames[t], collision_mask, radius=3).astype(np.float32)

        cx = float(np.polyval(fit_x, t))
        cy = float(np.polyval(fit_y, t))
        left = int(round(cx)) - half
        top = int(round(cy)) - half
        right, bottom = left + pw, top + ph
        dx0, dy0 = max(0, left), max(0, top)
        dx1, dy1 = min(w, right), min(h, bottom)
        if dx1 > dx0 and dy1 > dy0:
            sx0, sy0 = dx0 - left, dy0 - top
            sx1, sy1 = sx0 + (dx1 - dx0), sy0 + (dy1 - dy0)
            a = alpha[sy0:sy1, sx0:sx1, None]
            fg = foreground[sy0:sy1, sx0:sx1]
            canvas[dy0:dy1, dx0:dx1] = (
                fg * a + canvas[dy0:dy1, dx0:dx1] * (1.0 - a)
            )
        frames[t] = np.clip(np.rint(canvas), 0, 255).astype(np.uint8)

    artifact_io.save(frames, CASE_DIR)


if __name__ == \"__main__\":
    main()
