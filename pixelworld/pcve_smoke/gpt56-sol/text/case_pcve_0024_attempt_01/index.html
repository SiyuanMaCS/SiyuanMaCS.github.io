+"""Generate the friction-edited PixelWorld video.
+
+The source ball is sliding/rolling to rest under approximately constant dry
+friction.  For its original stopping time T, displacement along the observed
+path is
+
+    s(t) = v0*t - (v0/(2*T))*t**2.
+
+Tripling friction triples the deceleration.  It therefore stops at T/3 after
+travelling one third as far.  Before that stop, each edited state is obtained
+by finding the time on the original trajectory having the same displacement.
+This preserves the rendered ball, its rotation, shadows, and occlusions rather
+than redrawing it.  Adjacent source samples are blended only while it is in
+motion; once stopped, an actual (sharp) source frame is held.
+"""
+
+from pathlib import Path
+
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+FRAME_COUNT = 96
+
+# The last visible motion is between source indices 17 and 18.  A stopping
+# time of 18 source-frame intervals is consequently the best sampled estimate.
+ORIGINAL_STOP_FRAME = 18.0
+FRICTION_MULTIPLIER = 3.0
+
+
+def equivalent_source_time(t: float) -> float:
+    """Map edited frame time t to equal displacement on the source motion."""
+    T = ORIGINAL_STOP_FRAME
+    edited_stop = T / FRICTION_MULTIPLIER
+    if t >= edited_stop:
+        # The exact equal-displacement source time is about 3.30.  Holding
+        # source frame 3 avoids a permanently blended/double edge, with less
+        # than two pixels of positional quantisation at this resolution.
+        return 3.0
+
+    # Source displacement / v0 = u - u^2/(2T).
+    # Edited displacement / v0 = t - 3t^2/(2T).
+    displacement = t - FRICTION_MULTIPLIER * t * t / (2.0 * T)
+    discriminant = max(0.0, T * T - 2.0 * T * displacement)
+    return T - np.sqrt(discriminant)
+
+
+def temporal_sample(source: np.ndarray, source_time: float) -> np.ndarray:
+    """Linearly sample uint8 RGB source frames at a fractional frame time."""
+    lo = int(np.floor(source_time))
+    hi = min(lo + 1, len(source) - 1)
+    fraction = source_time - lo
+    if fraction < 1e-8 or lo == hi:
+        return source[lo].copy()
+    mixed = ((1.0 - fraction) * source[lo].astype(np.float32)
+             + fraction * source[hi].astype(np.float32))
+    return np.clip(np.rint(mixed), 0, 255).astype(np.uint8)
+
+
+def generate() -> np.ndarray:
+    with np.load(CASE_DIR / "source_pixels.npz") as archive:
+        source = archive["rgb"]
+
+    if source.shape != (FRAME_COUNT, 144, 256, 3):
+        raise ValueError(f"Unexpected source shape: {source.shape}")
+    if source.dtype != np.uint8:
+        raise ValueError(f"Unexpected source dtype: {source.dtype}")
+
+    frames = np.empty_like(source)
+    for frame_index in range(FRAME_COUNT):
+        source_time = equivalent_source_time(float(frame_index))
+        frames[frame_index] = temporal_sample(source, source_time)
+
+    # The intervention changes a coefficient, not the state at its exact
+    # starting instant, so frame 1 (zero-based index 0) remains factual.
+    frames[0] = source[0]
+    return frames
+
+
+if __name__ == "__main__":
+    output = generate()
+    artifact_io.save(output, CASE_DIR)
+    print("saved", output.shape, output.dtype)
+    print("source times:", [round(equivalent_source_time(float(i)), 3)
+                            for i in range(8)])
