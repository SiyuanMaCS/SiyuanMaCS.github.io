+"""Generate the counterfactual video with domino 3 removed from frame 1.
+
+The source is a fixed-camera domino shot.  Dominoes 1 and 2 are preserved from
+the source up to the instant at which domino 2 reaches domino 3.  Thereafter we
+retain only the two upstream dominoes over a repaired copy of the original
+scene.  Advancing their late source poses slightly models the unimpeded fall of
+domino 2 into the newly empty gap, while domino 4 and everything downstream
+remain in their initial, upright state.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def soft_polygon(shape, points, supersample=4, blur=0.65):
+    """Return an antialiased float mask for a convex polygon."""
+    h, w = shape
+    large = np.zeros((h * supersample, w * supersample), np.uint8)
+    pts = np.rint(np.asarray(points) * supersample).astype(np.int32)
+    cv2.fillConvexPoly(large, pts, 255, lineType=cv2.LINE_AA)
+    mask = cv2.resize(large, (w, h), interpolation=cv2.INTER_AREA) / 255.0
+    if blur:
+        mask = cv2.GaussianBlur(mask, (0, 0), blur)
+    return np.clip(mask, 0.0, 1.0)
+
+
+def composite(dst, src, alpha):
+    a = alpha[..., None].astype(np.float32)
+    return np.rint(dst.astype(np.float32) * (1.0 - a) +
+                    src.astype(np.float32) * a).astype(np.uint8)
+
+
+def remove_third_domino(frame):
+    """Repair the narrow image region occupied by domino 3.
+
+    The plate occupies x=142..152, y=39..103.  The somewhat larger mask also
+    removes its antialiasing, top highlight, and small ground-contact shadow.
+    OpenCV's deterministic numerical inpainting is particularly effective here
+    because the occluded backdrop is a smooth grass/lighting gradient.
+    """
+    h, w = frame.shape[:2]
+    mask = np.zeros((h, w), np.uint8)
+
+    # Slightly tapered silhouette, with room for antialiasing and contact shade.
+    body = np.array([
+        [140, 34], [154, 34], [155, 100], [157, 106],
+        [154, 110], [139, 110], [138, 104], [140, 98],
+    ], np.int32)
+    cv2.fillConvexPoly(mask, body, 255, lineType=cv2.LINE_8)
+
+    repaired = cv2.inpaint(frame, mask, 6.0, cv2.INPAINT_TELEA)
+
+    # A soft edge hides any one-pixel seam but keeps the repair spatially local.
+    alpha = cv2.GaussianBlur(mask.astype(np.float32) / 255.0,
+                             (0, 0), 0.85)
+    alpha[mask > 0] = 1.0
+    return composite(frame, repaired, np.clip(alpha, 0, 1))
+
+
+def domino2_mask(frame_number, shape):
+    """Projected image footprint of domino 2 in a source frame.
+
+    Angles are measured from the upright image axis.  They follow the measured
+    source poses; perspective makes a physically horizontal domino appear near
+    74 degrees in the image.
+    """
+    measured = np.array([
+        [0, 0], [20, 0], [21, 1], [22, 3], [23, 5], [24, 8],
+        [25, 10], [26, 13], [27, 16], [28, 20], [29, 24],
+        [30, 29], [31, 32], [32, 34], [33, 37], [34, 40],
+        [35, 43], [36, 46], [37, 49], [38, 52], [39, 54],
+        [40, 56], [41, 58], [42, 60], [43, 62], [44, 64],
+        [45, 66], [46, 69], [47, 71], [48, 73], [50, 74],
+        [95, 74],
+    ], np.float32)
+    angle = float(np.interp(frame_number, measured[:, 0], measured[:, 1]))
+    theta = np.deg2rad(angle)
+
+    # Image-space geometry of the second domino.
+    pivot = np.array([108.0, 102.5])
+    axis = np.array([np.sin(theta), -np.cos(theta)])
+    across = np.array([np.cos(theta), np.sin(theta)])
+    top = pivot + 65.0 * axis
+    half_width = 8.2
+    # Extend the mask a little beyond both caps to include bevel and shadowed rim.
+    low = pivot - 2.0 * axis
+    high = top + 2.0 * axis
+    points = [low - half_width * across, low + half_width * across,
+              high + half_width * across, high - half_width * across]
+    return soft_polygon(shape, points, supersample=4, blur=0.55)
+
+
+def generate():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3)
+    assert source.dtype == np.uint8
+    n, h, w, _ = source.shape
+
+    # The repaired frame is also the counterfactual static background.  It keeps
+    # every downstream domino and all scenery exactly where they began.
+    background = remove_third_domino(source[0])
+    result = np.empty_like(source)
+
+    # Soft vertical gate for source motion on the upstream side.  It ends before
+    # the original left edge of domino 3, preventing any part of that domino from
+    # leaking into late frames.
+    xs = np.arange(w, dtype=np.float32)
+    left_gate_1d = np.clip((141.0 - xs) / 3.0, 0.0, 1.0)
+    left_gate = np.repeat(left_gate_1d[None, :], h, axis=0)
+
+    for t in range(n):
+        if t < 30:
+            # Until contact with domino 3, the upstream physics are factual.
+            # Domino 3 itself is absent from the very first frame onward.
+            result[t] = remove_third_domino(source[t])
+            continue
+
+        # With no third domino to resist it, domino 2 completes its fall sooner.
+        # Pose 50 is the stable source pose of the first two fallen dominoes.
+        source_t = min(50, int(round(30 + 1.45 * (t - 30))))
+
+        out = background.copy()
+        # Source pixels on the left contain dominoes 1 and 2 plus the correctly
+        # exposed ground behind their original upright positions.
+        out = composite(out, source[source_t], left_gate)
+
+        # Preserve the portion of falling domino 2 that crosses into the empty
+        # gap.  Restrict this mask on the left where the broad gate already gives
+        # exact coverage, avoiding a double-edge.
+        moving = domino2_mask(source_t, (h, w))
+        extension_gate = np.clip((xs - 133.0) / 5.0, 0.0, 1.0)
+        moving *= extension_gate[None, :]
+        out = composite(out, source[source_t], moving)
+        result[t] = out
+
+    return result
+
+
+if __name__ == "__main__":
+    frames = generate()
+    artifact_io.save(frames, CASE_DIR)
+    print("saved", frames.shape, frames.dtype,
+          "to", CASE_DIR / "pixels.npz")
