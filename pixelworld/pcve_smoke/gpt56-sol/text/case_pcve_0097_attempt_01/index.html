+"""Generate the friction counterfactual for this PixelWorld case.
+
+The source is a fixed-camera rigid-body scene.  Prior to the intervention the
+red ball accelerates down the left-hand rail and then knocks the blue ball out
+of the scene.  Three times the contact friction makes the red ball shed its
+existing speed on the rail: it comes to rest just short of the blue ball, so
+that the blue ball is never struck.
+
+Only source RGB pixels and numerical masks/warps are used.  Keeping the
+pre-intervention frames byte-identical and transferring the source-rendered
+ball as a difference sprite retains the original lighting and anti-aliasing.
+"""
+from pathlib import Path
+import numpy as np
+from scipy import ndimage
+import artifact_io
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def red_component(frame, cx, cy):
+    """Return the red-ball component nearest the supplied source centre."""
+    a = frame.astype(np.int16)
+    red = ((a[..., 0] > 95) &
+           (a[..., 0] > 1.28 * a[..., 1] + 18) &
+           (a[..., 0] > 1.28 * a[..., 2] + 18))
+    labels, count = ndimage.label(red)
+    yy, xx = np.ogrid[:frame.shape[0], :frame.shape[1]]
+    best = None
+    best_d = 1e9
+    for label in range(1, count + 1):
+        ys, xs = np.where(labels == label)
+        if len(xs) < 70:
+            continue
+        mx, my = xs.mean(), ys.mean()
+        d = (mx - cx) ** 2 + (my - cy) ** 2
+        if d < best_d:
+            best_d = d
+            best = labels == label
+    if best is None:
+        raise RuntimeError("red ball was not found")
+    # Fill the dark specular/texture details enclosed by the red shell.
+    return ndimage.binary_fill_holes(best)
+
+
+def make_clean_base(source, source_index, centres, donor_index):
+    """Remove the source red ball while retaining the stationary scene."""
+    base = source[source_index].astype(np.float32)
+    donor = source[donor_index].astype(np.float32)
+    cx, cy = centres[source_index]
+    yy, xx = np.mgrid[:base.shape[0], :base.shape[1]]
+    # The larger lower lobe also removes the ball's small contact shadow.
+    r = np.sqrt(((xx - cx) / 12.0) ** 2 + ((yy - (cy + 1.0)) / 12.5) ** 2)
+    erase = np.clip((1.08 - r) / 0.18, 0.0, 1.0)[..., None]
+    return base * (1.0 - erase) + donor * erase
+
+
+def difference_sprite(source, source_index, centres):
+    """Extract ball, rim, highlight, and contact shadow as a delta sprite."""
+    cx, cy = centres[source_index]
+    # Pick a pre-collision donor whose ball is far from this location.  The
+    # blue ball is still present in every donor considered here.
+    candidates = [i for i in range(22) if abs(centres[i][0] - cx) > 45]
+    donor_index = max(candidates, key=lambda i: abs(centres[i][0] - cx))
+    image = source[source_index].astype(np.float32)
+    clean = source[donor_index].astype(np.float32)
+    delta = image - clean
+
+    core = red_component(source[source_index], cx, cy)
+    core = ndimage.binary_dilation(core, iterations=1)
+    diff = np.max(np.abs(delta), axis=2)
+    evidence = np.clip((diff - 2.5) / 15.0, 0.0, 1.0)
+    yy, xx = np.mgrid[:image.shape[0], :image.shape[1]]
+    vicinity = (((xx - cx) / 11.5) ** 2 + ((yy - (cy + 1.0)) / 12.0) ** 2) < 1.0
+    alpha = np.maximum(evidence * vicinity, core.astype(np.float32))
+    alpha = ndimage.gaussian_filter(alpha, 0.45)
+    alpha *= vicinity
+    return delta * alpha[..., None], alpha
+
+
+def generate():

+        (59.01,70.24),(47.22,77.77),

+
+    out = source.copy()
+

+    # teleport the ball, so this frame is unchanged.  Thereafter model the
+    # tangential motion with the incoming velocity and a constant opposing
+    # acceleration.  Triple friction is strong enough for static friction to
+    # hold the ball once speed reaches zero.
+    x0, y0 = centres[17]
+    incoming_speed = 10.25       # px/frame, leftward
+    friction_decel = 1.32        # px/frame^2, opposing motion
+    stop_after = incoming_speed / friction_decel
+

+    rail_x = centres[17:22, 0][::-1]
+    rail_y = centres[17:22, 1][::-1]

+                    for i in range(17, 22)}
+

+        x = x0 - incoming_speed * elapsed + 0.5 * friction_decel * elapsed**2
+        y = float(np.interp(x, rail_x, rail_y))
+

+        # displacement is only a few pixels, minimizing background mismatch.
+        distances = np.sum((centres[17:22] - (x, y)) ** 2, axis=1)
+        pose = 17 + int(np.argmin(distances))
+
+        # Until factual frame 21, retain each frame's own subtle renderer
+        # variation.  Afterwards use the final pre-impact state, so the blue
+        # ball and every unaffected prop remain in place.
+        base_index = min(frame_index, 20)
+        bx, by = centres[base_index]
+        donor_candidates = [i for i in range(22)
+                            if abs(centres[i][0] - bx) > 50]
+        donor = max(donor_candidates,
+                    key=lambda i: abs(centres[i][0] - bx))
+        canvas = make_clean_base(source, base_index, centres, donor)
+
+        delta, alpha = sprite_cache[pose]
+        dx = x - centres[pose, 0]
+        dy = y - centres[pose, 1]
+        moved_delta = ndimage.shift(delta, (dy, dx, 0), order=1,
+                                    mode="constant", cval=0.0,
+                                    prefilter=False)
+        moved_alpha = ndimage.shift(alpha, (dy, dx), order=1,
+                                    mode="constant", cval=0.0,
+                                    prefilter=False)
+
+        # Add the source-rendered foreground difference.  Alpha gates tiny
+        # donor/render differences away from the sphere and its shadow.
+        canvas += moved_delta * moved_alpha[..., None]
+        out[frame_index] = np.clip(np.rint(canvas), 0, 255).astype(np.uint8)
+
+    return out

+    artifact_io.save(frames, CASE_DIR)
+    print("saved", frames.shape, frames.dtype)
