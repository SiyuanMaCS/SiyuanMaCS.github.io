+"""Generate the counterfactual video for the local PixelWorld case.
+
+The red toy car is isolated from the source RGB frames.  Its measured flat-
+surface acceleration is scaled to 0.3 (the requested friction multiplier),
+then the original photographed car is composited back along that trajectory.
+No external assets, models, masks, or renderers are used.
+"""
+from pathlib import Path
+import numpy as np
+from scipy.ndimage import (
+    binary_closing, binary_dilation, binary_fill_holes,
+    gaussian_filter, label, shift,
+)
+import artifact_io
+
+CASE_DIR = Path(__file__).resolve().parent
+
+# Measured centers of the moving car in source frames 0..23.  They are used
+# both to exclude the car from the clean plate and to preserve its path down
+# the short ramp.  Thereafter the factual car is stationary.
+SRC_X = np.array([
+    67.01, 77.75, 88.60, 99.25, 109.54, 119.23, 130.68, 137.98,
+    144.51, 150.80, 156.60, 162.01, 167.05, 171.57, 175.18, 178.87,
+    182.03, 184.74, 186.86, 188.66, 190.11, 191.19, 191.89, 192.10,
+], dtype=np.float32)
+SRC_Y = np.array([
+    29.63, 29.84, 29.61, 29.95, 32.58, 36.86, 45.79, 48.36,
+    47.89, 47.36, 47.07, 46.78, 46.69, 46.26, 46.05, 45.95,
+    45.75, 45.58, 45.59, 45.53, 45.47, 45.52, 45.45, 45.46,
+], dtype=np.float32)
+
+
+def source_centers(n):
+    x = np.full(n, SRC_X[-1], dtype=np.float32)
+    y = np.full(n, SRC_Y[-1], dtype=np.float32)
+    k = min(n, len(SRC_X))
+    x[:k], y[:k] = SRC_X[:k], SRC_Y[:k]
+    return x, y
+
+
+def build_clean_plate(frames, sx, sy):
+    """Robust temporal plate, excluding the car and its contact shadow."""
+    n, h, w, _ = frames.shape
+    work = frames.astype(np.float32)
+    valid = np.ones((n, h, w), dtype=bool)
+    # Exclusion is deliberately wider than the visible car (about 23x12 px)
+    # so antialiasing, the wheels, and the small contact shadow are excluded.
+    for t in range(n):
+        cx, cy = int(round(sx[t])), int(round(sy[t]))
+        x0, x1 = max(0, cx - 17), min(w, cx + 18)
+        y0, y1 = max(0, cy - 14), min(h, cy + 15)
+        valid[t, y0:y1, x0:x1] = False
+    masked = np.where(valid[..., None], work, np.nan)
+    plate = np.nanmedian(masked, axis=0)
+    # Every pixel has many valid observations, but retain a deterministic
+    # fallback for completeness.
+    fallback = np.median(work, axis=0)
+    plate = np.where(np.isnan(plate), fallback, plate)
+    return np.clip(np.rint(plate), 0, 255).astype(np.uint8)
+
+
+def isolate_car_masks(frames, plate, sx, sy):
+    """Find a compact connected difference component at each known center."""
+    n, h, w, _ = frames.shape
+    masks = np.zeros((n, h, w), dtype=np.float32)
+    p = plate.astype(np.int16)
+    yy, xx = np.ogrid[:h, :w]
+    for t in range(n):
+        cx, cy = sx[t], sy[t]
+        local = (np.abs(xx - cx) <= 18) & (np.abs(yy - cy) <= 14)
+        delta = np.max(np.abs(frames[t].astype(np.int16) - p), axis=2)
+        binary = (delta >= 7) & local
+        components, count = label(binary)
+        if count:
+            best_id, best_score = 0, -1.0
+            for idx in range(1, count + 1):
+                ys, xs = np.where(components == idx)
+                if len(xs) == 0:
+                    continue
+                # Prefer a substantial component near the measured center.
+                dist = (xs.mean() - cx) ** 2 + (ys.mean() - cy) ** 2
+                score = len(xs) - 0.55 * dist
+                if score > best_score:
+                    best_id, best_score = idx, score
+            core = components == best_id
+        else:
+            core = local & (delta >= 4)
+        core = binary_fill_holes(binary_closing(core, iterations=1))
+        # Include edge antialiasing while keeping the alpha compact.
+        edge = binary_dilation(core, iterations=1).astype(np.float32)
+        alpha = gaussian_filter(edge, 0.55)
+        alpha *= np.clip((delta.astype(np.float32) - 2.0) / 10.0, 0.0, 1.0)
+        # The interior must remain opaque even where a car highlight happens
+        # to resemble the underlying scene.
+        alpha[core] = 1.0
+        masks[t] = np.clip(alpha, 0.0, 1.0)
+    return masks
+
+
+def counterfactual_centers(n):
+    """Trajectory with kinetic-friction acceleration reduced to 30 percent."""
+    tx = np.empty(n, dtype=np.float32)
+    ty = np.empty(n, dtype=np.float32)
+
+    # On the approach/ramp, lower friction produces a small, smoothly growing
+    # lead.  Sampling y by path position preserves the photographed geometry.
+    ramp_end = min(7, n)
+    for t in range(ramp_end):
+        tx[t] = SRC_X[t] + 0.040 * t * t
+        tx[t] = min(tx[t], 132.2)
+        ty[t] = np.interp(tx[t], SRC_X[:8], SRC_Y[:8])
+
+    # Quadratic fit to source frames 7..23 gives acceleration -0.4348
+    # px/frame^2.  Scale that by exactly 0.3.  The exit velocity is measured
+    # from the same fit, with the small ramp-friction gain included.
+    x_exit = float(tx[6]) if n > 6 else float(tx[-1])
+    v_exit = 7.95
+    a_new = -0.43476883385 * 0.3
+    stop_n = -v_exit / a_new
+    stop_x = x_exit + v_exit * stop_n + 0.5 * a_new * stop_n * stop_n
+    for t in range(7, n):
+        dt = t - 6
+        if dt < stop_n:
+            tx[t] = x_exit + v_exit * dt + 0.5 * a_new * dt * dt
+        else:
+            tx[t] = stop_x
+        # Preserve the brief suspension settling after the ramp; friction has
+        # no material effect on this vertical motion.
+        if t < 9:
+            ty[t] = SRC_Y[t]
+        else:
+            ty[t] = 45.45 + 2.45 * np.exp(-0.34 * (t - 8))
+    return tx, ty
+
+
+def translate_layer(rgb, alpha, dx, dy):
+    """Subpixel translate an RGB/alpha layer using local numeric resampling."""
+    moved_rgb = shift(
+        rgb.astype(np.float32), (dy, dx, 0), order=1,
+        mode="nearest", prefilter=False,
+    )
+    moved_a = shift(
+        alpha.astype(np.float32), (dy, dx), order=1,
+        mode="constant", cval=0.0, prefilter=False,
+    )
+    return moved_rgb, np.clip(moved_a, 0.0, 1.0)
+
+
+def render():
+    frames = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert frames.shape == (96, 144, 256, 3) and frames.dtype == np.uint8
+    n, h, w, _ = frames.shape
+    sx, sy = source_centers(n)
+    plate = build_clean_plate(frames, sx, sy)
+    car_alpha = isolate_car_masks(frames, plate, sx, sy)
+    tx, ty = counterfactual_centers(n)
+
+    out = frames.astype(np.float32).copy()
+    for t in range(1, n):
+        # Erase the factual car and shadow, feathering only this tiny region
+        # into the clean plate while leaving the rest of the source untouched.
+        erase = gaussian_filter(
+            binary_dilation(car_alpha[t] > 0.08, iterations=2).astype(np.float32),
+            0.65,
+        )
+        erase = np.clip(erase, 0.0, 1.0)[..., None]
+        out[t] = out[t] * (1.0 - erase) + plate.astype(np.float32) * erase
+
+        # Once entirely out of frame there is nothing to composite.
+        if tx[t] - 15.0 >= w:
+            continue
+
+        # Before the flat, use the nearest source path pose.  On and after the
+        # flat, use late photographed poses; this preserves perspective,
+        # wheels, highlights, and the original contact shadow.
+        donor = int(np.argmin(np.abs(SRC_X - min(tx[t], SRC_X[-1]))))
+        donor = min(donor, 23)
+        dx, dy = float(tx[t] - sx[donor]), float(ty[t] - sy[donor])
+        sprite, alpha = translate_layer(frames[donor], car_alpha[donor], dx, dy)
+        a = alpha[..., None]
+        out[t] = sprite * a + out[t] * (1.0 - a)
+
+    result = np.clip(np.rint(out), 0, 255).astype(np.uint8)
+    # At intervention time the coefficient changes but position and appearance
+    # are continuous, so frame 1 (index 0) is exactly the factual initial state.
+    result[0] = frames[0]
+    return result
+
+
+if __name__ == "__main__":
+    artifact_io.save(render(), CASE_DIR)

\"\"\"Generate the counterfactual video for the local PixelWorld case.

The red toy car is isolated from the source RGB frames.  Its measured flat-
surface acceleration is scaled to 0.3 (the requested friction multiplier),
then the original photographed car is composited back along that trajectory.
No external assets, models, masks, or renderers are used.

from pathlib import Path
import numpy as np
from scipy.ndimage import (
    binary_closing, binary_dilation, binary_fill_holes,
    gaussian_filter, label, shift,
)
import artifact_io

CASE_DIR = Path(__file__).resolve().parent

# Measured centers of the moving car in source frames 0..23.  They are used
# both to exclude the car from the clean plate and to preserve its path down
# the short ramp.  Thereafter the factual car is stationary.
SRC_X = np.array([
    67.01, 77.75, 88.60, 99.25, 109.54, 119.23, 130.68, 137.98,
    144.51, 150.80, 156.60, 162.01, 167.05, 171.57, 175.18, 178.87,
    182.03, 184.74, 186.86, 188.66, 190.11, 191.19, 191.89, 192.10,
], dtype=np.float32)
SRC_Y = np.array([
    29.63, 29.84, 29.61, 29.95, 32.58, 36.86, 45.79, 48.36,
    47.89, 47.36, 47.07, 46.78, 46.69, 46.26, 46.05, 45.95,
    45.75, 45.58, 45.59, 45.53, 45.47, 45.52, 45.45, 45.46,
], dtype=np.float32)


def source_centers(n):
    x = np.full(n, SRC_X[-1], dtype=np.float32)
    y = np.full(n, SRC_Y[-1], dtype=np.float32)
    k = min(n, len(SRC_X))
    x[:k], y[:k] = SRC_X[:k], SRC_Y[:k]
    return x, y


def build_clean_plate(frames, sx, sy):
    \"\"\"Robust temporal plate, excluding the car and its contact shadow.\"\"\"
    n, h, w, _ = frames.shape
    work = frames.astype(np.float32)
    valid = np.ones((n, h, w), dtype=bool)
    # Exclusion is deliberately wider than the visible car (about 23x12 px)
    # so antialiasing, the wheels, and the small contact shadow are excluded.
    for t in range(n):
        cx, cy = int(round(sx[t])), int(round(sy[t]))
        x0, x1 = max(0, cx - 17), min(w, cx + 18)
        y0, y1 = max(0, cy - 14), min(h, cy + 15)
        valid[t, y0:y1, x0:x1] = False
    masked = np.where(valid[..., None], work, np.nan)
    plate = np.nanmedian(masked, axis=0)
    # Every pixel has many valid observations, but retain a deterministic
    # fallback for completeness.
    fallback = np.median(work, axis=0)
    plate = np.where(np.isnan(plate), fallback, plate)
    return np.clip(np.rint(plate), 0, 255).astype(np.uint8)


def isolate_car_masks(frames, plate, sx, sy):
    \"\"\"Find a compact connected difference component at each known center.\"\"\"
    n, h, w, _ = frames.shape
    masks = np.zeros((n, h, w), dtype=np.float32)
    p = plate.astype(np.int16)
    yy, xx = np.ogrid[:h, :w]
    for t in range(n):
        cx, cy = sx[t], sy[t]
        local = (np.abs(xx - cx) <= 18) & (np.abs(yy - cy) <= 14)
        delta = np.max(np.abs(frames[t].astype(np.int16) - p), axis=2)
        binary = (delta >= 7) & local
        components, count = label(binary)
        if count:
            best_id, best_score = 0, -1.0
            for idx in range(1, count + 1):
                ys, xs = np.where(components == idx)
                if len(xs) == 0:
                    continue
                # Prefer a substantial component near the measured center.
                dist = (xs.mean() - cx) ** 2 + (ys.mean() - cy) ** 2
                score = len(xs) - 0.55 * dist
                if score > best_score:
                    best_id, best_score = idx, score
            core = components == best_id
        else:
            core = local & (delta >= 4)
        core = binary_fill_holes(binary_closing(core, iterations=1))
        # Include edge antialiasing while keeping the alpha compact.
        edge = binary_dilation(core, iterations=1).astype(np.float32)
        alpha = gaussian_filter(edge, 0.55)
        alpha *= np.clip((delta.astype(np.float32) - 2.0) / 10.0, 0.0, 1.0)
        # The interior must remain opaque even where a car highlight happens
        # to resemble the underlying scene.
        alpha[core] = 1.0
        masks[t] = np.clip(alpha, 0.0, 1.0)
    return masks


def counterfactual_centers(n):
    \"\"\"Trajectory with kinetic-friction acceleration reduced to 30 percent.\"\"\"
    tx = np.empty(n, dtype=np.float32)
    ty = np.empty(n, dtype=np.float32)

    # On the approach/ramp, lower friction produces a small, smoothly growing
    # lead.  Sampling y by path position preserves the photographed geometry.
    ramp_end = min(7, n)
    for t in range(ramp_end):
        tx[t] = SRC_X[t] + 0.040 * t * t
        tx[t] = min(tx[t], 132.2)
        ty[t] = np.interp(tx[t], SRC_X[:8], SRC_Y[:8])

    # Quadratic fit to source frames 7..23 gives acceleration -0.4348
    # px/frame"'^2.  Scale that by exactly 0.3.  The exit velocity is measured
    # from the same fit, with the small ramp-friction gain included.
    x_exit = float(tx[6]) if n > 6 else float(tx[-1])
    v_exit = 7.95
    a_new = -0.43476883385 * 0.3
    stop_n = -v_exit / a_new
    stop_x = x_exit + v_exit * stop_n + 0.5 * a_new * stop_n * stop_n
    for t in range(7, n):
        dt = t - 6
        if dt < stop_n:
            tx[t] = x_exit + v_exit * dt + 0.5 * a_new * dt * dt
        else:
            tx[t] = stop_x
        # Preserve the brief suspension settling after the ramp; friction has
        # no material effect on this vertical motion.
        if t < 9:
            ty[t] = SRC_Y[t]
        else:
            ty[t] = 45.45 + 2.45 * np.exp(-0.34 * (t - 8))
    return tx, ty


def translate_layer(rgb, alpha, dx, dy):
    """Subpixel translate an RGB/alpha layer using local numeric resampling."""
    moved_rgb = shift(
        rgb.astype(np.float32), (dy, dx, 0), order=1,
        mode="nearest", prefilter=False,
    )
    moved_a = shift(
        alpha.astype(np.float32), (dy, dx), order=1,
        mode="constant", cval=0.0, prefilter=False,
    )
    return moved_rgb, np.clip(moved_a, 0.0, 1.0)


def render():
    frames = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
    assert frames.shape == (96, 144, 256, 3) and frames.dtype == np.uint8
    n, h, w, _ = frames.shape
    sx, sy = source_centers(n)
    plate = build_clean_plate(frames, sx, sy)
    car_alpha = isolate_car_masks(frames, plate, sx, sy)
    tx, ty = counterfactual_centers(n)

    out = frames.astype(np.float32).copy()
    for t in range(1, n):
        # Erase the factual car and shadow, feathering only this tiny region
        # into the clean plate while leaving the rest of the source untouched.
        erase = gaussian_filter(
            binary_dilation(car_alpha[t] > 0.08, iterations=2).astype(np.float32),
            0.65,
        )
        erase = np.clip(erase, 0.0, 1.0)[..., None]
        out[t] = out[t] * (1.0 - erase) + plate.astype(np.float32) * erase

        # Once entirely out of frame there is nothing to composite.
        if tx[t] - 15.0 >= w:
            continue

        # Before the flat, use the nearest source path pose.  On and after the
        # flat, use late photographed poses; this preserves perspective,
        # wheels, highlights, and the original contact shadow.
        donor = int(np.argmin(np.abs(SRC_X - min(tx[t], SRC_X[-1]))))
        donor = min(donor, 23)
        dx, dy = float(tx[t] - sx[donor]), float(ty[t] - sy[donor])
        sprite, alpha = translate_layer(frames[donor], car_alpha[donor], dx, dy)
        a = alpha[..., None]
        out[t] = sprite * a + out[t] * (1.0 - a)

    result = np.clip(np.rint(out), 0, 255).astype(np.uint8)
    # At intervention time the coefficient changes but position and appearance
    # are continuous, so frame 1 (index 0) is exactly the factual initial state.
    result[0] = frames[0]
    return result


if __name__ == "__main__":
    artifact_io.save(render(), CASE_DIR)
