+"""Remove the first (leftmost) domino from source frame 24 onward.
+
+The camera is locked.  A clean plate is recovered independently at each pixel
+from the moment in the source sequence where that pixel looks most like the
+green ground.  This retains the real grass texture instead of painting a flat
+patch.  The first domino's projected quadrilateral is tracked through a small
+set of measured keyframes.  The second domino is protected at their overlap:
+it had already been set in motion before the requested intervention and must
+therefore remain in the counterfactual history.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+EXECUTION_INDEX = 23  # prompt uses one-based frame 24
+
+
+def interpolate_quad(frame_number, keys):
+    """Linearly interpolate a four-corner image-space rigid-body footprint."""
+    times = np.asarray(sorted(keys), dtype=np.float32)
+    if frame_number <= times[0]:
+        return np.asarray(keys[int(times[0])], dtype=np.float32)
+    if frame_number >= times[-1]:
+        return np.asarray(keys[int(times[-1])], dtype=np.float32)
+    hi = int(np.searchsorted(times, frame_number))
+    lo = hi - 1
+    t0, t1 = times[lo], times[hi]
+    u = (frame_number - t0) / (t1 - t0)
+    q0 = np.asarray(keys[int(t0)], dtype=np.float32)
+    q1 = np.asarray(keys[int(t1)], dtype=np.float32)
+    return q0 * (1.0 - u) + q1 * u
+
+
+# Corners are ordered around each visible domino silhouette.  The observations
+# include the narrow top/side face, so replacing this polygon also removes the
+# anti-aliased outline rather than leaving a dark halo.
+FIRST_KEYS = {
+    24: [(96, 34), (109, 42), (70, 104), (58, 96)],
+    26: [(107, 34), (119, 42), (70, 104), (60, 95)],
+    31: [(118, 57), (124, 66), (68, 104), (61, 94)],
+    36: [(132, 60), (136, 71), (68, 103), (62, 93)],
+    42: [(136, 70), (139, 80), (69, 102), (63, 92)],
+    48: [(128, 70), (132, 80), (70, 103), (64, 93)],
+    56: [(128, 70), (132, 80), (70, 103), (64, 93)],
+    96: [(128, 70), (132, 80), (70, 103), (64, 93)],
+}
+
+# The second domino is retained where the two projected silhouettes cross.
+# Its small lean at frame 24 is factual history (the contact occurs earlier).
+SECOND_KEYS = {
+    24: [(110, 33), (120, 35), (119, 96), (111, 103)],
+    26: [(112, 34), (124, 36), (124, 91), (113, 103)],
+    31: [(128, 36), (139, 43), (113, 103), (104, 96)],
+    36: [(147, 42), (157, 51), (115, 103), (106, 95)],
+    42: [(174, 53), (183, 63), (151, 104), (142, 94)],
+    48: [(170, 69), (175, 79), (114, 103), (108, 94)],
+    56: [(170, 69), (175, 79), (114, 103), (108, 94)],
+    96: [(170, 69), (175, 79), (114, 103), (108, 94)],
+}
+
+
+def polygon_mask(shape, quad, grow=0):
+    mask = np.zeros(shape, dtype=np.uint8)
+    cv2.fillConvexPoly(mask, np.rint(quad).astype(np.int32), 255, lineType=cv2.LINE_AA)
+    if grow:
+        k = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (2 * grow + 1,) * 2)
+        mask = cv2.dilate(mask, k)
+    return mask
+
+
+def make_clean_plate(source):
+    """Recover static ground using the most grass-like temporal observations."""
+    rgb = source.astype(np.float32)
+    # Green excess separates grass from the nearly neutral charcoal dominoes,
+    # their white pips, and their contact shadows.
+    grass_score = rgb[..., 1] - 0.65 * rgb[..., 0] - 0.35 * rgb[..., 2]
+    best = np.argpartition(grass_score, -10, axis=0)[-10:]
+    yy, xx = np.indices(source.shape[1:3])
+    samples = source[best, yy[None, ...], xx[None, ...]]
+    plate = np.median(samples, axis=0).astype(np.uint8)
+
+    # A few pixels at the first domino's ground contact are occluded throughout
+    # the source.  Fill only non-grass remnants inside its swept footprint.
+    swept = np.zeros(source.shape[1:3], np.uint8)
+    for frame_number in range(24, 97):
+        swept = cv2.max(swept, polygon_mask(swept.shape,
+                                             interpolate_quad(frame_number, FIRST_KEYS),
+                                             grow=3))
+    p = plate.astype(np.float32)
+    plate_score = p[..., 1] - 0.65 * p[..., 0] - 0.35 * p[..., 2]
+    remnants = ((swept > 0) & (plate_score < 18.0)).astype(np.uint8) * 255
+    remnants = cv2.morphologyEx(remnants, cv2.MORPH_CLOSE, np.ones((3, 3), np.uint8))
+    return cv2.inpaint(plate, remnants, 4, cv2.INPAINT_TELEA)
+
+
+def generate():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3) and source.dtype == np.uint8
+    result = source.copy()
+    plate = make_clean_plate(source)
+
+    for index in range(EXECUTION_INDEX, len(result)):
+        frame_number = index + 1
+        first = polygon_mask(source.shape[1:3],
+                             interpolate_quad(frame_number, FIRST_KEYS), grow=2)
+        second = polygon_mask(source.shape[1:3],
+                              interpolate_quad(frame_number, SECOND_KEYS), grow=0)
+
+        # Preserve the already-moving second domino in overlap regions.  Erode
+        # by one pixel so the first domino's antialiased rim cannot survive.
+        second_core = cv2.erode(second, np.ones((3, 3), np.uint8))
+        removal = first.copy()
+        removal[second_core > 0] = 0
+
+        # Sub-pixel feathering matches the native downsampled source edges.
+        alpha = cv2.GaussianBlur(removal, (0, 0), 0.7).astype(np.float32) / 255.0
+        alpha = alpha[..., None]
+        composed = source[index].astype(np.float32) * (1.0 - alpha)
+        composed += plate.astype(np.float32) * alpha
+        result[index] = np.clip(np.rint(composed), 0, 255).astype(np.uint8)
+
+    # Required output contract.  All frames, including the untouched factual
+    # prefix, are saved as one complete 4-second RGB video artifact.
+    artifact_io.save(result, CASE_DIR)
+    return result
+
+
+if __name__ == "__main__":
+    generate()
