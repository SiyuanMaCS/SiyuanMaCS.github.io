+"""Generate the counterfactual 1.5x-speed volleyball video.
+
+The source shot has a single period of meaningful motion: the volleyball moves
+from frame 1 and reaches its resting configuration at approximately source
+frame 21 (zero-based index 20).  Increasing its initial speed by 1.5 maps an
+output time t to source motion time 1.5*t.  This preserves the exact source
+trajectory and all of its interactions, but makes them happen sooner.  After
+the accelerated motion has completed, the original settled frames are used so
+that the long static tail retains its subtle source-image variation.
+
+Half-frame source times are reconstructed with bidirectional dense optical
+flow rather than a dissolve.  All operations are local numeric RGB operations.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+SOURCE = CASE_DIR / "source_pixels.npz"
+SPEED_MULTIPLIER = 1.5
+LAST_MOVING_SOURCE_INDEX = 20
+
+
+def _remap_rgb(image: np.ndarray, map_x: np.ndarray, map_y: np.ndarray) -> np.ndarray:
+    """Subpixel-remap an RGB image, reflecting only at the outer image edge."""
+    return cv2.remap(
+        image,
+        map_x.astype(np.float32),
+        map_y.astype(np.float32),
+        interpolation=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_REFLECT_101,
+    )
+
+
+def motion_midpoint(frame_a: np.ndarray, frame_b: np.ndarray) -> np.ndarray:
+    """Synthesize the temporal midpoint of two adjacent RGB source frames."""
+    gray_a = cv2.cvtColor(frame_a, cv2.COLOR_RGB2GRAY)
+    gray_b = cv2.cvtColor(frame_b, cv2.COLOR_RGB2GRAY)
+
+    # Farneback is stable on this small, textured shot and preserves the
+    # volleyball's sharp colored panels substantially better than crossfading.
+    params = dict(
+        pyr_scale=0.5,
+        levels=4,
+        winsize=15,
+        iterations=4,
+        poly_n=5,
+        poly_sigma=1.1,
+        flags=0,
+    )
+    flow_ab = cv2.calcOpticalFlowFarneback(gray_a, gray_b, None, **params)
+    flow_ba = cv2.calcOpticalFlowFarneback(gray_b, gray_a, None, **params)
+
+    height, width = gray_a.shape
+    grid_x, grid_y = np.meshgrid(
+        np.arange(width, dtype=np.float32), np.arange(height, dtype=np.float32)
+    )
+
+    # For constant local velocity, inverse-warp each endpoint to the midpoint.
+    warped_a = _remap_rgb(
+        frame_a,
+        grid_x - 0.5 * flow_ab[..., 0],
+        grid_y - 0.5 * flow_ab[..., 1],
+    )
+    warped_b = _remap_rgb(
+        frame_b,
+        grid_x - 0.5 * flow_ba[..., 0],
+        grid_y - 0.5 * flow_ba[..., 1],
+    )
+
+    # Occlusion-aware weighting.  A warped endpoint that disagrees strongly
+    # with the other endpoint gets less weight, preventing double edges at the
+    # leading and trailing silhouettes of the moving ball.
+    difference = np.mean(
+        np.abs(warped_a.astype(np.float32) - warped_b.astype(np.float32)), axis=2
+    )
+    reliability = np.exp(-difference / 22.0)
+    # Confidence is intentionally symmetric in ordinary regions.  Near a true
+    # disocclusion, use local sharpness to favor the less-blurred reconstruction.
+    lap_a = np.abs(cv2.Laplacian(cv2.cvtColor(warped_a, cv2.COLOR_RGB2GRAY), cv2.CV_32F))
+    lap_b = np.abs(cv2.Laplacian(cv2.cvtColor(warped_b, cv2.COLOR_RGB2GRAY), cv2.CV_32F))
+    preference = np.tanh((lap_a - lap_b) / 18.0)
+    confidence_delta = (1.0 - reliability) * 0.16 * preference
+    weight_a = np.clip(0.5 + confidence_delta, 0.34, 0.66)[..., None]
+
+    midpoint = warped_a.astype(np.float32) * weight_a + warped_b.astype(np.float32) * (1.0 - weight_a)
+    return np.clip(np.rint(midpoint), 0, 255).astype(np.uint8)
+
+
+def generate() -> np.ndarray:
+    source = np.load(SOURCE)["rgb"]
+    if source.shape != (96, 144, 256, 3) or source.dtype != np.uint8:
+        raise ValueError(f"Unexpected source tensor: {source.shape}, {source.dtype}")
+
+    result = np.empty_like(source)
+    midpoint_cache: dict[int, np.ndarray] = {}
+
+    for output_index in range(len(result)):
+        accelerated_time = SPEED_MULTIPLIER * output_index
+
+        if accelerated_time <= LAST_MOVING_SOURCE_INDEX:
+            lower = int(np.floor(accelerated_time))
+            fraction = accelerated_time - lower
+            if fraction < 1e-8:
+                result[output_index] = source[lower]
+            else:
+                # With a 1.5 multiplier all nonintegral samples are midpoints.
+                if lower not in midpoint_cache:
+                    midpoint_cache[lower] = motion_midpoint(source[lower], source[lower + 1])
+                result[output_index] = midpoint_cache[lower]
+        elif output_index < LAST_MOVING_SOURCE_INDEX:
+            # The accelerated volleyball has already settled, while the source
+            # timeline at these few frames had not.  Hold the settled source
+            # configuration until the factual timeline catches up.
+            result[output_index] = source[LAST_MOVING_SOURCE_INDEX]
+        else:
+            # No meaningful object motion remains.  Preserve the corresponding
+            # source frame rather than freezing the complete 4-second shot.
+            result[output_index] = source[output_index]
+
+    # The intervention changes velocity, not initial state: frame 1 is exact.
+    assert np.array_equal(result[0], source[0])
+    return result
+
+
+if __name__ == "__main__":
+    frames = generate()
+    artifact_io.save(frames, CASE_DIR)
+    print(
+        f"Saved {len(frames)} RGB frames at {frames.shape[2]}x{frames.shape[1]} "
+        f"to {CASE_DIR}"
+    )
