+"""Generate the 0.7x-initial-speed counterfactual video.
+
+The intervention occurs at source frame 1 (t=0).  In this rigid, static-camera
+scene the visible event is initiated by one moving ball.  Scaling that initial
+velocity and the resulting collision velocities by 0.7 is represented by a
+0.7 dilation of the source physical timeline.  Bidirectional dense flow gives
+continuous positions at the fractional source times without crossfade ghosts.
+"""
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+SPEED_SCALE = 0.7
+
+
+def dense_flow(a: np.ndarray, b: np.ndarray) -> np.ndarray:
+    """Dense displacement field from RGB image a to RGB image b."""
+    ga = cv2.cvtColor(a, cv2.COLOR_RGB2GRAY)
+    gb = cv2.cvtColor(b, cv2.COLOR_RGB2GRAY)
+    # The source ball moves about seven pixels per source frame.  A five-level
+    # pyramid comfortably spans this while preserving its fine boundary.
+    return cv2.calcOpticalFlowFarneback(
+        ga, gb, None,
+        pyr_scale=0.5, levels=5, winsize=21, iterations=5,
+        poly_n=7, poly_sigma=1.5, flags=cv2.OPTFLOW_FARNEBACK_GAUSSIAN,
+    )
+
+
+def remap(image: np.ndarray, flow: np.ndarray, amount: float) -> np.ndarray:
+    """Sample image after moving it by ``amount * flow``."""
+    h, w = image.shape[:2]
+    yy, xx = np.mgrid[0:h, 0:w].astype(np.float32)
+    map_x = xx - np.float32(amount) * flow[..., 0]
+    map_y = yy - np.float32(amount) * flow[..., 1]
+    return cv2.remap(
+        image, map_x, map_y, interpolation=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_REPLICATE,
+    )
+
+
+def interpolate(a: np.ndarray, b: np.ndarray, alpha: float,
+                forward: np.ndarray, backward: np.ndarray) -> np.ndarray:
+    """Occlusion-aware bidirectional interpolation between adjacent frames."""
+    if alpha <= 1e-7:
+        return a.copy()
+    if alpha >= 1.0 - 1e-7:
+        return b.copy()
+
+    wa = remap(a, forward, alpha).astype(np.float32)
+    wb = remap(b, backward, 1.0 - alpha).astype(np.float32)
+
+    # Forward/backward disagreement rises at disocclusions.  Prefer the nearer
+    # endpoint there; elsewhere use the smooth temporal blend.
+    h, w = a.shape[:2]
+    yy, xx = np.mgrid[0:h, 0:w].astype(np.float32)
+    bx = np.clip(xx + forward[..., 0], 0, w - 1)
+    by = np.clip(yy + forward[..., 1], 0, h - 1)
+    back_at_b = cv2.remap(backward, bx, by, cv2.INTER_LINEAR,
+                          borderMode=cv2.BORDER_REPLICATE)
+    consistency = np.linalg.norm(forward + back_at_b, axis=2)
+    reliable = np.exp(-(consistency / 1.5) ** 2)[..., None].astype(np.float32)
+
+    regular = (1.0 - alpha) * wa + alpha * wb
+    nearer = wa if alpha < 0.5 else wb
+    result = reliable * regular + (1.0 - reliable) * nearer
+    return np.clip(np.rint(result), 0, 255).astype(np.uint8)
+
+
+def generate() -> np.ndarray:

+    assert source.shape == (96, 144, 256, 3) and source.dtype == np.uint8

+    # Cache only adjacent pairs actually needed by the slowed timeline.
+    flow_cache = {}

+            continue

+            flow_cache[key] = (
+                dense_flow(source[lo], source[lo + 1]),
+                dense_flow(source[lo + 1], source[lo]),
+            )
+        output[out_index] = interpolate(
+            source[lo], source[lo + 1], alpha, *flow_cache[key]
+        )
+
+    # Frame 1 is the intervention instant and must be exactly factual.
+    output[0] = source[0]
+    return output
