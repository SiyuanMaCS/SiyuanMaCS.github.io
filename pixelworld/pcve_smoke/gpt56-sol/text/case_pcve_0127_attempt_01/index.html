+
+The intervention is at frame 1.  The source experiment has a yellow marble
+travelling right and a purple marble travelling left; without the yellow one,
+the purple marble never collides and therefore continues on its pre-impact
+ballistic path.  Frames are reconstructed from source pixels only: a temporal
+background plate removes both factual moving marbles, then the purple marble's
+own pre-collision sprite is translated along the extrapolated trajectory.

+import numpy as np
+import cv2
+from artifact_io import save
+
+ROOT = Path(__file__).resolve().parent
+N, H, W = 96, 144, 256
+

+def detect_centers(src):
+    """Color-based centroids; used to estimate the factual pre-impact motion."""
+    out = []
+    for im in src:
+        a = im.astype(np.int16)
+        mask = ((a[..., 2] > 125) & (a[..., 2] - a[..., 0] > 25) &
+                (a[..., 2] - a[..., 1] > 45))
+        mask[:45] = False
+        mask[92:] = False
+        count, labels, stats, centers = cv2.connectedComponentsWithStats(
+            mask.astype(np.uint8), 8)
+        choices = [(stats[k, 4], centers[k]) for k in range(1, count)
+                   if 40 <= stats[k, 4] <= 130]
+        out.append(max(choices, key=lambda z: z[0])[1])
+    return np.asarray(out, np.float64)
+
+
+def temporal_background(src):
+    # Marbles occupy a pixel briefly; the temporal median restores the static
+    # tabletop/track, including the unobstructed background and soft shading.
+    bg = np.median(src, axis=0).astype(np.uint8)
+    return bg
+
+
+def bilinear_sample_rgb(image, xs, ys):
+    """Bilinear RGB sampler for a small floating-point translated patch."""
+    x0 = np.floor(xs).astype(np.int32)
+    y0 = np.floor(ys).astype(np.int32)
+    fx = (xs - x0)[..., None]
+    fy = (ys - y0)[..., None]
+    x0 = np.clip(x0, 0, W - 2)
+    y0 = np.clip(y0, 0, H - 2)
+    p00 = image[y0, x0].astype(np.float32)
+    p10 = image[y0, x0 + 1].astype(np.float32)
+    p01 = image[y0 + 1, x0].astype(np.float32)
+    p11 = image[y0 + 1, x0 + 1].astype(np.float32)
+    return ((1-fx)*(1-fy)*p00 + fx*(1-fy)*p10 +
+            (1-fx)*fy*p01 + fx*fy*p11)
+
+
+def render_counterfactual(src):
+    bg = temporal_background(src)
+    centers = detect_centers(src)
+
+    # The marbles are released after five stationary frames.  Fit horizontal
+    # displacement to elapsed-motion squared (constant acceleration) over the
+    # clean pre-contact interval.  Vertical drift is much smaller but follows
+    # the same deterministic camera-perspective trajectory, so fit it likewise.
+    pre = np.arange(5, 32)
+    tau = (pre - 5).astype(np.float64)
+    A = np.stack([np.ones_like(tau), tau, tau*tau], axis=1)
+    coef_x = np.linalg.lstsq(A, centers[pre, 0], rcond=None)[0]
+    coef_y = np.linalg.lstsq(A, centers[pre, 1], rcond=None)[0]
+
+    # Use a clean stationary source sprite.  Alpha comes from difference to the
+    # temporal background and includes the marble's antialiased rim and shadow.
+    anchor_frame = 0
+    anchor = centers[anchor_frame]
+    radius = 10
+    x0 = int(round(anchor[0])) - radius
+    y0 = int(round(anchor[1])) - radius
+    yy, xx = np.mgrid[y0:y0+2*radius+1, x0:x0+2*radius+1]
+    sprite = src[anchor_frame, yy, xx].astype(np.float32)
+    back_patch = bg[yy, xx].astype(np.float32)
+    color_delta = np.max(np.abs(sprite - back_patch), axis=2)
+    # Strong difference in the marble body, feathered support for antialiasing
+    # and the cast/contact shadow below it.
+    alpha = np.clip((color_delta - 2.0) / 20.0, 0.0, 1.0)
+    core = (((sprite[..., 2] - sprite[..., 0]) > 18) &
+            ((sprite[..., 2] - sprite[..., 1]) > 28))
+    alpha = np.maximum(alpha, core.astype(np.float32))
+    alpha = cv2.GaussianBlur(alpha, (3, 3), 0.45)
+    # Confine support to a soft ellipse around marble plus its lower shadow.
+    dx = xx - anchor[0]
+    dy = yy - anchor[1]
+    support = np.clip((8.5 - np.sqrt((dx/1.0)**2 + (dy/1.12)**2)) / 1.3, 0, 1)
+    alpha *= support.astype(np.float32)
+
+    result = np.repeat(bg[None], N, axis=0)
+    for t in range(N):
+        q = max(0.0, float(t - 5))
+        cx = coef_x[0] + coef_x[1]*q + coef_x[2]*q*q
+        cy = coef_y[0] + coef_y[1]*q + coef_y[2]*q*q
+        # Before release preserve exact initial location.  The fitted path then
+        # continues naturally through and beyond the absent collision.
+        if t <= 5:
+            cx, cy = anchor
+        if cx < -12 or cx > W + 12 or cy < -12 or cy > H + 12:
+            continue
+        left = int(np.floor(cx - radius))
+        top = int(np.floor(cy - radius))
+        X0, X1 = max(0, left), min(W, left + 2*radius + 1)
+        Y0, Y1 = max(0, top), min(H, top + 2*radius + 1)
+        if X0 >= X1 or Y0 >= Y1:
+            continue
+        # Sampling grid in original sprite coordinates, retaining fractional
+        # subpixel motion rather than visibly snapping to whole pixels.
+        gx, gy = np.meshgrid(np.arange(X0, X1), np.arange(Y0, Y1))
+        sx = gx - cx + anchor[0]
+        sy = gy - cy + anchor[1]
+        sampled = bilinear_sample_rgb(src[anchor_frame], sx, sy)
+        sampled_bg = bilinear_sample_rgb(bg, sx, sy)
+        # Sample alpha from the local alpha tile with cv2 interpolation.
+        ax = sx - x0
+        ay = sy - y0
+        amap = cv2.remap(alpha, ax.astype(np.float32), ay.astype(np.float32),
+                         cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT,
+                         borderValue=0)
+        # Recompose the translated foreground residual over the local static
+        # plate.  This preserves fine marble texture and avoids a rectangular
+        # copied-background halo.
+        local = result[t, Y0:Y1, X0:X1].astype(np.float32)
+        residual = sampled - sampled_bg
+        composed = local + residual * amap[..., None]
+        result[t, Y0:Y1, X0:X1] = np.clip(composed, 0, 255).round().astype(np.uint8)
+    return result
+
+
+def main():
+    src = np.load(ROOT / 'source_pixels.npz')['rgb']
+    assert src.shape == (N, H, W, 3) and src.dtype == np.uint8
+    frames = render_counterfactual(src)
+    save(frames, ROOT)
+
+
+if __name__ == '__main__':
+    main()

        # continues naturally through and beyond the absent collision.
        if t <= 5:
            cx, cy = anchor
        if cx < -12 or cx > W + 12 or cy < -12 or cy > H + 12:
            continue
        left = int(np.floor(cx - radius))
        top = int(np.floor(cy - radius))
        X0, X1 = max(0, left), min(W, left + 2*radius + 1)
        Y0, Y1 = max(0, top), min(H, top + 2*radius + 1)
        if X0 >= X1 or Y0 >= Y1:
            continue
        # Sampling grid in original sprite coordinates, retaining fractional
        # subpixel motion rather than visibly snapping to whole pixels.
        gx, gy = np.meshgrid(np.arange(X0, X1), np.arange(Y0, Y1))
        sx = gx - cx + anchor[0]
        sy = gy - cy + anchor[1]
        sampled = bilinear_sample_rgb(src[anchor_frame], sx, sy)
        sampled_bg = bilinear_sample_rgb(bg, sx, sy)
        # Sample alpha from the local alpha tile with cv2 interpolation.
        ax = sx - x0
        ay = sy - y0
        amap = cv2.remap(alpha, ax.astype(np.float32), ay.astype(np.float32),
                         cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT,
                         borderValue=0)
        # Recompose the translated foreground residual over the local static
        # plate.  This preserves fine marble texture and avoids a rectangular
        # copied-background halo.
        local = result[t, Y0:Y1, X0:X1].astype(np.float32)
        residual = sampled - sampled_bg
        composed = local + residual * amap[..., None]
        result[t, Y0:Y1, X0:X1] = np.clip(composed, 0, 255).round().astype(np.uint8)
    return result

    frames = render_counterfactual(src)
