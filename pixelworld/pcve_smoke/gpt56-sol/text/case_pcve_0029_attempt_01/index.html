+"""Generate the requested counterfactual video using only the supplied RGB tensor.
+
+The red sphere's observed ballistic motion is reconstructed in image-plane units.
+Its original normal restitution is 0.80; from the first frame onward it is set to
+1.2 * 0.80 = 0.96.  Frames preceding the first impact are copied losslessly.
+"""
+from pathlib import Path
+import numpy as np
+from scipy import ndimage
+import artifact_io
+
+CASE_DIR = Path(__file__).resolve().parent
+N, H, W = 96, 144, 256
+
+
+def sphere_centers(frames):
+    """Track the isolated saturated-red component, ignoring warm scenery."""
+    centers = []
+    for image in frames.astype(np.int16):
+        red = ((image[..., 0] - image[..., 1] > 48) &
+               (image[..., 0] - image[..., 2] > 48) &
+               (image[..., 0] > 60))
+        labels, count = ndimage.label(red)
+        choices = []
+        for k in range(1, count + 1):
+            yy, xx = np.nonzero(labels == k)
+            if len(xx) > 100 and 100 < xx.mean() < 190 and yy.mean() < 105:
+                choices.append((len(xx), xx.mean(), yy.mean()))
+        if not choices:
+            raise RuntimeError("Unable to locate the red sphere")
+        centers.append(max(choices)[1:])
+    return np.asarray(centers, dtype=np.float64)
+
+
+def circle_coverage(cx, cy, radius=12.35):
+    """Analytic-style one-pixel antialiasing for a projected sphere disk."""
+    yy, xx = np.ogrid[:H, :W]
+    distance = np.hypot(xx - cx, yy - cy)
+    return np.clip(radius + 0.55 - distance, 0.0, 1.0)
+
+
+def clean_plate(source, centers):
+    """Make a clean static plate from frame 0, replacing only its sphere."""
+    plate = source[0].astype(np.float64).copy()
+    # Frame 24 has clean wall at the initial sphere position.  A feathered patch
+    # also removes the tiny antialiased fringe while retaining all nearby detail.
+    yy, xx = np.ogrid[:H, :W]
+    cx, cy = centers[0]
+    d = np.hypot(xx - cx, yy - cy)
+    patch_alpha = np.clip(16.0 - d, 0.0, 1.0)[..., None]
+    plate = plate * (1.0 - patch_alpha) + source[24] * patch_alpha
+    return plate
+
+
+def altered_trajectory(centers):
+    """Event-based ballistic integration in frame units."""
+    gravity = 0.738                 # px / frame^2, fitted across source arcs
+    ground_y = 81.65               # sphere center at contact
+    restitution = 1.2 * 0.80       # requested intervention
+
+    result = centers.copy()
+    # Frame 12 is the last supplied sample before the first impact.
+    time = 12.0
+    x, y = map(float, centers[12])
+    vx = 0.9094
+    vy = 0.5262 + gravity * 12.0
+    bounce_count = 0
+
+    for frame in range(13, N):
+        target = float(frame)
+        while time < target - 1e-10:
+            remaining = target - time
+            # Positive root of y + vy*t + g*t^2/2 == ground_y.
+            disc = vy * vy + 2.0 * gravity * (ground_y - y)
+            hit = (-vy + np.sqrt(max(0.0, disc))) / gravity
+            if hit > 1e-8 and hit <= remaining + 1e-10:
+                x += vx * hit
+                y = ground_y
+                vy += gravity * hit
+                time += hit
+                vy = -restitution * vy
+                bounce_count += 1
+                # Match the observed frictional transfer at impacts. The first
+                # impact brings the sphere close to rolling; later contacts cause
+                # only the smaller decrement measured in the source trajectory.
+                if bounce_count == 1:
+                    vx = 0.3181
+                else:
+                    vx *= 0.85
+            else:
+                x += vx * remaining
+                y += vy * remaining + 0.5 * gravity * remaining * remaining
+                vy += gravity * remaining
+                time = target
+        result[frame] = (x, y)
+    return result
+
+
+def add_shadow(image, cx, cy):
+    """Render the scene's soft, warm contact shadow on the horizontal surface."""
+    yy, xx = np.indices((H, W), dtype=np.float64)
+    height = max(0.0, 81.65 - cy)
+    strength = np.exp(-height / 20.0)
+    # The broad penumbra grows and fades as the sphere rises.  The image-plane
+    # floor recedes upward, hence the slight downward projection of the shadow.
+    sigma_x = 9.0 + 0.10 * height
+    sigma_y = 4.3 + 0.055 * height
+    sx = cx + 0.035 * height
+    sy = 97.0 + 0.035 * height
+    core = np.exp(-0.5 * (((xx - sx) / sigma_x) ** 2 +
+                          ((yy - sy) / sigma_y) ** 2))
+    halo = np.exp(-0.5 * (((xx - sx) / (sigma_x * 1.65)) ** 2 +
+                          ((yy - (sy + 3.0)) / (sigma_y * 2.0)) ** 2))
+    floor_gate = 1.0 / (1.0 + np.exp(-(yy - 86.0) * 1.25))
+    density = strength * floor_gate * (0.72 * core + 0.28 * halo)
+    # Warm scene lighting makes the occlusion strongest in blue.
+    attenuation = np.array([8.0, 16.0, 27.0])[None, None, :]
+    return np.clip(image - density[..., None] * attenuation, 0, 255)
+
+
+def render():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (N, H, W, 3) and source.dtype == np.uint8
+    centers = sphere_centers(source)
+    trajectory = altered_trajectory(centers)
+    plate = clean_plate(source, centers)
+    output = np.empty_like(source)
+
+    # Restitution has no observable effect until contact. Keep factual history
+    # through frame 12 exactly, including source antialiasing and tiny shadows.
+    output[:13] = source[:13]
+
+    for frame in range(13, N):
+        old_x, old_y = centers[frame]
+        new_x, new_y = trajectory[frame]
+        canvas = add_shadow(plate.copy(), new_x, new_y)
+
+        # Recover the sphere foreground from this source frame.  Reusing each
+        # frame's actual texture retains its natural rolling/lighting animation.
+        old_alpha = circle_coverage(old_x, old_y)
+        old_bg = plate
+        observed = source[frame].astype(np.float64)
+        safe = np.maximum(old_alpha[..., None], 0.10)
+        foreground = (observed - (1.0 - old_alpha[..., None]) * old_bg) / safe
+        foreground = np.clip(foreground, 0, 255)
+
+        # Translate foreground texture and source alpha to the altered position.
+        shift = (new_y - old_y, new_x - old_x)
+        moved_fg = np.empty_like(foreground)
+        for channel in range(3):
+            moved_fg[..., channel] = ndimage.shift(
+                foreground[..., channel], shift, order=1, mode="nearest",
+                prefilter=False)
+        moved_alpha = ndimage.shift(old_alpha, shift, order=1, mode="constant",
+                                    cval=0.0, prefilter=False)
+        # Geometric target coverage suppresses any interpolation residue.
+        moved_alpha = np.minimum(moved_alpha, circle_coverage(new_x, new_y))
+        a = moved_alpha[..., None]
+        canvas = canvas * (1.0 - a) + moved_fg * a
+        output[frame] = np.clip(np.rint(canvas), 0, 255).astype(np.uint8)
+
+    return output
+
+
+if __name__ == "__main__":
+    frames = render()
+    artifact_io.save(frames, CASE_DIR)
+    print("saved", frames.shape, frames.dtype)

\"\"\"Generate the requested counterfactual video using only the supplied RGB tensor.

The red sphere's observed ballistic motion is reconstructed in image-plane units.
Its original normal restitution is 0.80; from the first frame onward it is set to
1.2 * 0.80 = 0.96.  Frames preceding the first impact are copied losslessly.

from pathlib import Path
import numpy as np
from scipy import ndimage
import artifact_io

CASE_DIR = Path(__file__).resolve().parent
N, H, W = 96, 144, 256


def sphere_centers(frames):
    \"\"\"Track the isolated saturated-red component, ignoring warm scenery.\"\"\"
    centers = []
    for image in frames.astype(np.int16):
        red = ((image[..., 0] - image[..., 1] > 48) &
               (image[..., 0] - image[..., 2] > 48) &
               (image[..., 0] > 60))
        labels, count = ndimage.label(red)
        choices = []
        for k in range(1, count + 1):
            yy, xx = np.nonzero(labels == k)
            if len(xx) > 100 and 100 < xx.mean() < 190 and yy.mean() < 105:
                choices.append((len(xx), xx.mean(), yy.mean()))
        if not choices:
            raise RuntimeError(\"Unable to locate the red sphere\")
        centers.append(max(choices)[1:])
    return np.asarray(centers, dtype=np.float64)


def circle_coverage(cx, cy, radius=12.35):
    \"\"\"Analytic-style one-pixel antialiasing for a projected sphere disk.\"\"\"
    yy, xx = np.ogrid[:H, :W]
    distance = np.hypot(xx - cx, yy - cy)
    return np.clip(radius + 0.55 - distance, 0.0, 1.0)


def clean_plate(source, centers):
    \"\"\"Make a clean static plate from frame 0, replacing only its sphere.\"\"\"
    plate = source[0].astype(np.float64).copy()
    # Frame 24 has clean wall at the initial sphere position.  A feathered patch
    # also removes the tiny antialiased fringe while retaining all nearby detail.
    yy, xx = np.ogrid[:H, :W]
    cx, cy = centers[0]
    d = np.hypot(xx - cx, yy - cy)
    patch_alpha = np.clip(16.0 - d, 0.0, 1.0)[..., None]
    plate = plate * (1.0 - patch_alpha) + source[24] * patch_alpha
    return plate


def altered_trajectory(centers):
    \"\"\"Event-based ballistic integration in frame units.\"\"\"
    gravity = 0.738                 # px / frame"'^2, fitted across source arcs
    ground_y = 81.65               # sphere center at contact
    restitution = 1.2 * 0.80       # requested intervention

    result = centers.copy()
    # Frame 12 is the last supplied sample before the first impact.
    time = 12.0
    x, y = map(float, centers[12])
    vx = 0.9094
    vy = 0.5262 + gravity * 12.0
    bounce_count = 0

    for frame in range(13, N):
        target = float(frame)
        while time < target - 1e-10:
            remaining = target - time
            # Positive root of y + vy*t + g*t''^2/2 == ground_y.
            disc = vy * vy + 2.0 * gravity * (ground_y - y)
            hit = (-vy + np.sqrt(max(0.0, disc))) / gravity
            if hit > 1e-8 and hit <= remaining + 1e-10:
                x += vx * hit
                y = ground_y
                vy += gravity * hit
                time += hit
                vy = -restitution * vy
                bounce_count += 1
                # Match the observed frictional transfer at impacts. The first
                # impact brings the sphere close to rolling; later contacts cause
                # only the smaller decrement measured in the source trajectory.
                if bounce_count == 1:
                    vx = 0.3181
                else:
                    vx *= 0.85
            else:
                x += vx * remaining
                y += vy * remaining + 0.5 * gravity * remaining * remaining
                vy += gravity * remaining
                time = target
        result[frame] = (x, y)
    return result


def add_shadow(image, cx, cy):
    """Render the scene'"'s soft, warm contact shadow on the horizontal surface.\"\"\"
    yy, xx = np.indices((H, W), dtype=np.float64)
    height = max(0.0, 81.65 - cy)
    strength = np.exp(-height / 20.0)
    # The broad penumbra grows and fades as the sphere rises.  The image-plane
    # floor recedes upward, hence the slight downward projection of the shadow.
    sigma_x = 9.0 + 0.10 * height
    sigma_y = 4.3 + 0.055 * height
    sx = cx + 0.035 * height
    sy = 97.0 + 0.035 * height
    core = np.exp(-0.5 * (((xx - sx) / sigma_x) ** 2 +
                          ((yy - sy) / sigma_y) ** 2))
    halo = np.exp(-0.5 * (((xx - sx) / (sigma_x * 1.65)) ** 2 +
                          ((yy - (sy + 3.0)) / (sigma_y * 2.0)) ** 2))
    floor_gate = 1.0 / (1.0 + np.exp(-(yy - 86.0) * 1.25))
    density = strength * floor_gate * (0.72 * core + 0.28 * halo)
    # Warm scene lighting makes the occlusion strongest in blue.
    attenuation = np.array([8.0, 16.0, 27.0])[None, None, :]
    return np.clip(image - density[..., None] * attenuation, 0, 255)


def render():
    source = np.load(CASE_DIR / \"source_pixels.npz\")[\"rgb\"]
    assert source.shape == (N, H, W, 3) and source.dtype == np.uint8
    centers = sphere_centers(source)
    trajectory = altered_trajectory(centers)
    plate = clean_plate(source, centers)
    output = np.empty_like(source)

    # Restitution has no observable effect until contact. Keep factual history
    # through frame 12 exactly, including source antialiasing and tiny shadows.
    output[:13] = source[:13]

    for frame in range(13, N):
        old_x, old_y = centers[frame]
        new_x, new_y = trajectory[frame]
        canvas = add_shadow(plate.copy(), new_x, new_y)

        # Recover the sphere foreground from this source frame.  Reusing each
        # frame's actual texture retains its natural rolling/lighting animation.
        old_alpha = circle_coverage(old_x, old_y)
        old_bg = plate
        observed = source[frame].astype(np.float64)
        safe = np.maximum(old_alpha[..., None], 0.10)
        foreground = (observed - (1.0 - old_alpha[..., None]) * old_bg) / safe
        foreground = np.clip(foreground, 0, 255)

        # Translate foreground texture and source alpha to the altered position.
        shift = (new_y - old_y, new_x - old_x)
        moved_fg = np.empty_like(foreground)
        for channel in range(3):
            moved_fg[..., channel] = ndimage.shift(
                foreground[..., channel], shift, order=1, mode=\"nearest\",
                prefilter=False)
        moved_alpha = ndimage.shift(old_alpha, shift, order=1, mode=\"constant\",
                                    cval=0.0, prefilter=False)
        # Geometric target coverage suppresses any interpolation residue.
        moved_alpha = np.minimum(moved_alpha, circle_coverage(new_x, new_y))
        a = moved_alpha[..., None]
        canvas = canvas * (1.0 - a) + moved_fg * a
        output[frame] = np.clip(np.rint(canvas), 0, 255).astype(np.uint8)

    return output


if __name__ == \"__main__\":
    frames = render()
    artifact_io.save(frames, CASE_DIR)
    print(\"saved\", frames.shape, frames.dtype)
