+"""Generate the requested high-friction counterfactual air-hockey video.
+
+The intervention begins at source frame 1 (t=0), so there is no factual
+pre-intervention motion to retain.  In the factual clip the blue mallet moves
+about 1.4 pixels per frame and starts a collision chain (blue -> red -> white).
+Multiplying its friction by 25 makes the stopping time shorter than a video
+frame at this scale.  We therefore retain the small first-frame inertial step
+visible between source frames 1 and 2, then hold the resulting state.  This
+also correctly prevents the red mallet and white puck from ever being struck.
+
+Using an actual source state rather than repainting the objects preserves all
+lighting, perspective, shadows, antialiasing, and table reflections exactly.
+"""
+
+from pathlib import Path
+
+import numpy as np
+
+import artifact_io
+
+
+def generate(case_dir: Path) -> np.ndarray:
+    source = np.load(case_dir / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3)
+    assert source.dtype == np.uint8
+
+    # t=0 retains the initial state.  One very small inertial displacement is
+    # allowed in the first 1/24 s; thereafter the 25x surface friction has
+    # arrested the blue mallet.  Since contact never occurs, every other body
+    # remains in its unperturbed pose as present in source frame 2.
+    frames = np.empty_like(source)
+    frames[0] = source[0]
+    frames[1:] = source[1]
+    return frames
+
+
+if __name__ == "__main__":
+    case_dir = Path(__file__).resolve().parent
+    frames = generate(case_dir)
+    artifact_io.save(frames, case_dir)
+    print(f"saved {len(frames)} frames: {frames.shape}, {frames.dtype}")
