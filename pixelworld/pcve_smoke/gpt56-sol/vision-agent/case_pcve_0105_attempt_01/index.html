+"""Generate the 5x-small-marble-mass counterfactual.
+
+The factual shot is retained through first contact.  Thereafter the source
+marbles are removed from each frame and two source-derived marble sprites are
+placed along a damped, momentum-consistent counterfactual trajectory.
+"""
+
+from pathlib import Path
+
+import numpy as np
+from PIL import Image, ImageDraw, ImageFilter
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+N, H, W = 96, 144, 256
+
+
+def ellipse_mask(cx, cy, rx, ry):
+    yy, xx = np.mgrid[:H, :W]
+    return ((xx - cx) / rx) ** 2 + ((yy - cy) / ry) ** 2 <= 1.0
+
+
+def find_cyan_centers(frames):
+    """The cyan marble is the only strongly cyan component on the tabletop."""
+    centers = []
+    yy, xx = np.mgrid[:H, :W]
+    for frame in frames:
+        mask = ((frame[..., 2].astype(int) - frame[..., 0].astype(int) > 45) &
+                (frame[..., 1].astype(int) > 70) & (frame[..., 0] < 95) &
+                (yy >= 62) & (yy <= 88))
+        y, x = np.where(mask)
+        if len(x):
+            centers.append((float(x.mean()), float(y.mean())))
+        else:
+            centers.append(centers[-1] if centers else (131.0, 72.4))
+    return centers
+
+
+def source_yellow_centers():
+    # Measured visual centers in the source.  The ball moves approximately
+    # six pixels/frame before impact, then rolls to rest near x=97.
+    x = np.array([
+        225, 219, 211, 204, 198, 192, 186, 179, 173, 167, 161, 156, 150, 145,
+        142, 136, 132, 128, 124, 121, 117, 114, 111, 108, 105, 103, 102, 101,
+        100, 99, 98, 97.5, 97.3, 97.2, 97.1
+    ], dtype=float)
+    x = np.r_[x, np.full(N - len(x), 97.1)]
+    # Geometric center follows the receding top-plane perspective.
+    y = 64.5 + 0.050 * (225.0 - x)
+    return list(zip(x, y))
+
+
+def build_clean_plate(frames, cyan_centers, yellow_centers):
+    """Temporal median using only samples not occupied by either marble.
+
+    Masking includes the soft vertically stretched tabletop reflections.
+    Every location has many clean observations because both balls traverse it.
+    """
+    valid = np.ones((N, H, W), dtype=bool)
+    for i, ((cx, cy), (yx, yy)) in enumerate(zip(cyan_centers, yellow_centers)):
+        cm = ellipse_mask(cx, cy, 6.5, 6.0)
+        cm |= ellipse_mask(cx, 83.0, 7.0, 11.0)
+        ym = ellipse_mask(yx, yy, 11.5, 11.0)
+        ym |= ellipse_mask(yx, 82.0, 13.0, 12.0)
+        valid[i, cm | ym] = False
+
+    plate = np.empty((H, W, 3), dtype=np.uint8)
+    for channel in range(3):
+        data = frames[..., channel].astype(np.float32)
+        data[~valid] = np.nan
+        med = np.nanmedian(data, axis=0)
+        # This should not be needed, but gives a deterministic fallback for
+        # any pixel covered in all 96 masks.
+        fallback = np.median(frames[..., channel], axis=0)
+        med = np.where(np.isnan(med), fallback, med)
+        plate[..., channel] = np.clip(np.rint(med), 0, 255).astype(np.uint8)
+    return plate
+
+
+def source_removal_mask(cx, cy, yx, yy):
+    """Soft replacement mask for source balls, shadows, and reflections."""
+    raw = np.zeros((H, W), dtype=np.uint8)
+    raw[ellipse_mask(cx, cy, 6.6, 6.2)] = 255
+    raw[ellipse_mask(cx, 82.5, 7.5, 11.5)] = 235
+    raw[ellipse_mask(yx, yy, 11.8, 11.5)] = 255
+    raw[ellipse_mask(yx, 82.0, 13.5, 12.5)] = 235
+    return np.asarray(Image.fromarray(raw).filter(ImageFilter.GaussianBlur(1.0)), dtype=np.float32) / 255.0
+
+
+def extract_sprite(frame, plate, cx, cy, radius, half_size):
+    """Take a source-rendered marble with a feathered circular matte."""
+    x0, y0 = int(round(cx)) - half_size, int(round(cy)) - half_size
+    size = 2 * half_size + 1
+    rgb = frame[y0:y0 + size, x0:x0 + size].copy()
+    # At the anti-aliased fringe, use the source itself; beyond it the alpha
+    # is zero.  A little of the local source background is harmless and makes
+    # the sphere edge match the original renderer.
+    yy, xx = np.mgrid[:size, :size]
+    d = np.sqrt((xx - half_size) ** 2 + (yy - half_size) ** 2)
+    alpha = np.clip(radius + 0.75 - d, 0.0, 1.0)
+    rgba = np.dstack([rgb, np.rint(alpha * 255).astype(np.uint8)])
+    return Image.fromarray(rgba, "RGBA")
+
+
+def add_contact_and_reflection(frame, x, center_y, kind):
+    """Add subdued contact shadow and the source shot's glossy reflection."""
+    base = Image.fromarray(frame, "RGB").convert("RGBA")
+    layer = Image.new("RGBA", (W, H), (0, 0, 0, 0))
+    draw = ImageDraw.Draw(layer, "RGBA")
+    if kind == "yellow":
+        ground = center_y + 8.0
+        draw.ellipse((x - 8, ground - 1.0, x + 8, ground + 2.0), fill=(0, 0, 0, 70))
+        draw.ellipse((x - 6, ground + 1.0, x + 6, ground + 12.5), fill=(191, 133, 34, 29))
+        draw.ellipse((x - 3, ground + 1.0, x + 3, ground + 10.0), fill=(224, 187, 89, 20))
+        blur = 1.6
+    else:
+        ground = center_y + 4.2
+        draw.ellipse((x - 4.5, ground - 0.7, x + 4.5, ground + 1.2), fill=(0, 0, 0, 60))
+        draw.ellipse((x - 3.5, ground + 0.5, x + 3.5, ground + 10.0), fill=(34, 179, 190, 26))
+        draw.ellipse((x - 1.5, ground + 1.0, x + 1.5, ground + 8.0), fill=(127, 221, 221, 17))
+        blur = 1.2
+    layer = layer.filter(ImageFilter.GaussianBlur(blur))
+    return np.asarray(Image.alpha_composite(base, layer).convert("RGB"), dtype=np.uint8)
+
+
+def paste_sprite(frame, sprite, cx, cy):
+    im = Image.fromarray(frame, "RGB").convert("RGBA")
+    x = int(round(cx - sprite.width / 2))
+    y = int(round(cy - sprite.height / 2))
+    im.alpha_composite(sprite, (x, y))
+    return np.asarray(im.convert("RGB"), dtype=np.uint8)
+
+
+def damped_displacement(v0, q, dt):
+    """Distance after dt frames under per-frame multiplicative rolling drag."""
+    if dt <= 0:
+        return 0.0
+    return v0 * (1.0 - q ** dt) / (1.0 - q)
+
+
+def generate():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (N, H, W, 3) and source.dtype == np.uint8
+
+    cyan_src = find_cyan_centers(source)
+    yellow_src = source_yellow_centers()
+    plate = build_clean_plate(source, cyan_src, yellow_src)
+
+    # Source-rendered appearances from frame zero, where the marbles are
+    # separate and unobscured.
+    yellow_sprite = extract_sprite(source[0], plate, 225.0, 64.5, 8.45, 10)
+    cyan_sprite = extract_sprite(source[0], plate, 131.0, 72.4, 4.15, 5)
+
+    output = source.copy()
+    contact_frame = 14
+    contact_cyan_x = 128.2
+    contact_yellow_x = 141.7
+
+    # Estimated from the factual collision: original m_small/m_large ~= .215
+    # and restitution ~= .75.  The intervention multiplies that ratio by 5.
+    incoming = -5.70
+    mass_ratio = 0.215 * 5.0
+    restitution = 0.75
+    v_yellow = ((1.0 - restitution * mass_ratio) / (1.0 + mass_ratio)) * incoming
+    v_cyan = ((1.0 + restitution) / (1.0 + mass_ratio)) * incoming
+
+    # Frame 14 is first contact and remains visually identical.  The changed
+    # impulse becomes visible on the following frame.
+    for i in range(contact_frame + 1, N):
+        cx, cy = cyan_src[i]
+        yx, yy = yellow_src[i]
+        alpha = source_removal_mask(cx, cy, yx, yy)[..., None]
+        clean = np.rint(source[i] * (1.0 - alpha) + plate * alpha).astype(np.uint8)
+
+        dt = i - contact_frame
+        new_cx = contact_cyan_x + damped_displacement(v_cyan, 0.930, dt)
+        new_yx = contact_yellow_x + damped_displacement(v_yellow, 0.895, dt)
+        new_cy = 72.45 + 0.025 * (contact_cyan_x - new_cx)
+        new_yy = 68.65 + 0.050 * (contact_yellow_x - new_yx)
+
+        # Reflections lie below the spheres; then spheres occlude the table and
+        # the candlestick stems exactly as foreground objects should.
+        clean = add_contact_and_reflection(clean, new_yx, new_yy, "yellow")
+        clean = add_contact_and_reflection(clean, new_cx, new_cy, "cyan")
+        clean = paste_sprite(clean, yellow_sprite, new_yx, new_yy)
+        clean = paste_sprite(clean, cyan_sprite, new_cx, new_cy)
+        output[i] = clean
+
+    return output
+
+
+if __name__ == "__main__":
+    frames = generate()
+    artifact_io.save(frames, CASE_DIR)
+    print("saved", frames.shape, frames.dtype)
+
