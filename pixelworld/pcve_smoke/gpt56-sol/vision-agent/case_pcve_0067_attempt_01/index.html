+"""Generate the requested counterfactual domino video.
+
+The leftmost domino is the initiator in the factual clip.  Removing it at
+frame 1 means that no collision happens, so the other three dominoes retain
+their frame-1 poses for the full four seconds.  We therefore edit the first
+source frame and hold that physically stable state for all 96 output frames.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def make_clean_plate(first_frame: np.ndarray) -> np.ndarray:
+    """Remove the first domino and its attached shadow from frame 1.
+
+    The mask follows the domino with a small antialiasing margin and widens
+    below its foot to include its soft cast shadow.  The second domino starts
+    at x=102, safely outside the mask.  OpenCV's Navier-Stokes inpainting
+    continues the very smooth tabletop shading from all sides of the hole.
+    """
+    mask = np.zeros(first_frame.shape[:2], dtype=np.uint8)
+    removal_region = np.array(
+        [
+            [64, 32], [85, 32],  # top and antialiased rim
+            [83, 80], [81, 91],  # right side of the leaning domino
+            [89, 99], [97, 108], [101, 122], [98, 138],  # cast shadow
+            [53, 138], [51, 119], [52, 102], [57, 91],  # shadow/left side
+        ],
+        dtype=np.int32,
+    )
+    cv2.fillPoly(mask, [removal_region], 255)
+
+    # Prevent the pale wall above y=29 from influencing reconstruction at the
+    # mask's top.  These temporary pixels are outside the output mask and are
+    # used only as green boundary conditions for the PDE inpaint.
+    work = first_frame.copy()
+    work[:33, 45:99] = first_frame[33, 45:99]
+    reconstructed = cv2.inpaint(work, mask, 5.0, cv2.INPAINT_NS)
+
+    clean = first_frame.copy()
+    clean[mask != 0] = reconstructed[mask != 0]
+    return clean
+
+
+def generate() -> np.ndarray:
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3)
+    assert source.dtype == np.uint8
+
+    clean_first = make_clean_plate(source[0])
+
+    # No initiating domino remains, hence no forces or subsequent motion.
+    frames = np.repeat(clean_first[None, ...], 96, axis=0)
+    return np.ascontiguousarray(frames, dtype=np.uint8)
+
+
+if __name__ == "__main__":
+    output_frames = generate()
+    artifact_io.save(output_frames, CASE_DIR)
+    print(
+        f"saved {len(output_frames)} RGB frames at "
+        f"{output_frames.shape[2]}x{output_frames.shape[1]}"
+    )
