+"""Generate the 5x-friction counterfactual video for this case.
+
+The camera and room are static.  The patterned ("star") ball is stationary until
+it is hit by the pink ball between source frames 23 and 24.  Source motion then
+settles in about 22 frames.  Coulomb rolling/sliding friction gives a constant
+retarding acceleration: multiplying friction by five divides stopping time by
+five and stopping distance by five.  We therefore map the counterfactual
+post-contact displacement onto the corresponding early part of the real
+post-contact trajectory.  Motion-compensated interpolation keeps both balls,
+their shadows, occlusion, and the collision visually faithful.
+"""
+from pathlib import Path
+import cv2
+import numpy as np
+import artifact_io
+
+CASE_DIR = Path(__file__).resolve().parent
+N, H, W = 96, 144, 256
+CONTACT = 22.0       # zero-based frame/time of the last unaffected pose
+BASE_STOP = 22.0     # measured source settling duration, in frames
+FRICTION_MULT = 5.0
+
+
+def dense_flow(a, b):
+    """Dense A->B optical flow, tuned for the small, clean source motion."""
+    ga = cv2.cvtColor(a, cv2.COLOR_RGB2GRAY)
+    gb = cv2.cvtColor(b, cv2.COLOR_RGB2GRAY)
+    dis = cv2.DISOpticalFlow_create(cv2.DISOPTICAL_FLOW_PRESET_MEDIUM)
+    dis.setUseSpatialPropagation(True)
+    return dis.calc(ga, gb, None)
+
+
+def remap_with_displacement(image, displacement):
+    """Inverse-sample image after applying a forward displacement field."""
+    yy, xx = np.mgrid[0:H, 0:W].astype(np.float32)
+    map_x = (xx - displacement[..., 0]).astype(np.float32)
+    map_y = (yy - displacement[..., 1]).astype(np.float32)
+    return cv2.remap(image, map_x, map_y, cv2.INTER_CUBIC,
+                     borderMode=cv2.BORDER_REFLECT_101)
+
+
+def interpolate(a, b, alpha, flow_ab, flow_ba):
+    """Bidirectional motion-compensated subframe interpolation."""
+    if alpha <= 1e-6:
+        return a.copy()
+    if alpha >= 1.0 - 1e-6:
+        return b.copy()
+    wa = remap_with_displacement(a, alpha * flow_ab)
+    wb = remap_with_displacement(b, (1.0 - alpha) * flow_ba)
+    # The two warped images align moving ball edges.  A smooth crossfade also
+    # resolves newly uncovered floor pixels without holes.
+    out = wa.astype(np.float32) * (1.0 - alpha) + wb.astype(np.float32) * alpha
+    return np.clip(np.rint(out), 0, 255).astype(np.uint8)
+
+
+def source_time_for_output(frame_index):
+    """Map output time to source time under 5x constant friction.
+
+    Baseline normalized displacement is g(u)=2u-u^2 for 0<=u<=1.
+    Counterfactual displacement in baseline-distance units is
+    2q-M*q^2, where q=t/T and M=5.  Inverting g gives the source pose
+    having exactly that displacement.  At t=T/M we hold the stopped pose.
+    """
+    t = max(0.0, float(frame_index) - CONTACT)
+    stop_t = BASE_STOP / FRICTION_MULT
+    t = min(t, stop_t)
+    q = t / BASE_STOP
+    displacement = 2.0 * q - FRICTION_MULT * q * q
+    displacement = float(np.clip(displacement, 0.0, 1.0 / FRICTION_MULT))
+    source_fraction = 1.0 - np.sqrt(max(0.0, 1.0 - displacement))
+    return CONTACT + BASE_STOP * source_fraction
+
+
+def generate():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (N, H, W, 3) and source.dtype == np.uint8
+
+    # Only source intervals 22->23, 23->24, and 24->25 are needed: after
+    # one-fifth the original travel the high-friction ball has stopped.
+    flows = {}
+    for k in range(22, 25):
+        flows[(k, k + 1)] = dense_flow(source[k], source[k + 1])
+        flows[(k + 1, k)] = dense_flow(source[k + 1], source[k])
+
+    frames = np.empty_like(source)
+    frames[:23] = source[:23]  # no consequence before physical contact
+    for i in range(23, N):
+        tau = source_time_for_output(i)
+        lo = int(np.floor(tau))
+        hi = min(lo + 1, N - 1)
+        alpha = tau - lo
+        if hi == lo or alpha < 1e-6:
+            frames[i] = source[lo]
+        else:
+            frames[i] = interpolate(source[lo], source[hi], alpha,
+                                    flows[(lo, hi)], flows[(hi, lo)])
+
+    assert frames.shape == (N, H, W, 3) and frames.dtype == np.uint8
+    artifact_io.save(frames, CASE_DIR)
+    return frames
+
+
+if __name__ == "__main__":
+    generate()
