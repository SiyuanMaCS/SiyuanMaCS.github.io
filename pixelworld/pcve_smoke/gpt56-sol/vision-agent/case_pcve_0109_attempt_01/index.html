+import numpy as np
+
+import artifact_io

+CASE_DIR = Path(__file__).resolve().parent

+def smooth_inside(value, inner, outer):
+    """One inside `inner`, zero outside `outer`, linear in between."""
+    return np.clip((outer - value) / (outer - inner), 0.0, 1.0)
+

+def tracks(frames):
+    """Recover the cyan track and a smooth track for the gold marble."""
+    cyan = []
+    for im in frames:
+        # The saturated turquoise is unique in the scene.
+        m = ((im[:, :, 1] > 80) & (im[:, :, 2] > 80) &
+             (im[:, :, 0] < .65 * im[:, :, 1]) &
+             (im[:, :, 0] < .65 * im[:, :, 2]))
+        y, x = np.where(m)
+        cyan.append((float(x.mean()), float(y.mean())))
+    cyan = np.asarray(cyan)
+
+    # Before contact the large marble translates at almost constant speed.
+    # After contact its measured motion is a uniformly decelerating slide.
+    gold_x = np.empty(96, dtype=float)
+    measured_pre = [
+        225.0, 219.4, 214.1, 205.6, 199.3, 193.1, 186.8,
+        180.9, 174.9, 169.0, 163.1, 157.4, 150.6, 146.1,
+        141.5, 137.1, 133.2,
+    ]
+    gold_x[:17] = measured_pre
+    for i in range(17, 96):
+        n = i - 16
+        gold_x[i] = max(98.0, 133.2 - 4.35 * n + .125 * n * n)
+    # The tabletop rises gently to the right, so resting centre height varies.
+    gold_y = 74.32 - .0428 * gold_x
+    gold = np.c_[gold_x, gold_y]
+    return cyan, gold
+
+
+def clean_plate(frames, cyan, gold):
+    """Temporal median after excluding both moving bodies and reflections."""
+    n, h, w, _ = frames.shape
+    valid = np.ones((n, h, w), dtype=bool)
+    for i in range(n):
+        cx, cy = cyan[i]
+        valid[i, max(0, int(cy - 9)):min(h, int(cy + 29)),
+                 max(0, int(cx - 11)):min(w, int(cx + 12))] = False
+        gx, gy = gold[i]
+        valid[i, max(0, int(gy - 13)):min(h, int(gy + 31)),
+                 max(0, int(gx - 15)):min(w, int(gx + 16))] = False
+
+    plate = np.empty((h, w, 3), dtype=np.uint8)
+    # Chunking keeps peak memory modest.
+    for y0 in range(0, h, 16):
+        block = frames[:, y0:y0 + 16].astype(np.float32)
+        block[~valid[:, y0:y0 + 16]] = np.nan
+        plate[y0:y0 + 16] = np.nanmedian(block, axis=0).round().clip(0, 255)
+    return plate
+
+
+def object_alpha(h, w, center, kind, reflection=True):
+    yy, xx = np.ogrid[:h, :w]
+    cx, cy = center
+    if kind == "cyan":
+        r = np.sqrt((xx - cx) ** 2 + (yy - cy) ** 2)
+        main = smooth_inside(r, 6.1, 8.0)
+        rx, ry, off = 7.2, 13.5, 14.0
+    else:
+        r = np.sqrt((xx - cx) ** 2 + (yy - cy) ** 2)
+        main = smooth_inside(r, 9.0, 11.1)
+        rx, ry, off = 10.2, 14.0, 15.0
+    if not reflection:
+        return main
+    er = np.sqrt(((xx - cx) / rx) ** 2 + ((yy - (cy + off)) / ry) ** 2)
+    refl = smooth_inside(er, .72, 1.0)
+    # The reflection begins below the ball/table contact, not through the ball.
+    refl *= (yy >= cy + (5.0 if kind == "cyan" else 8.0))
+    return np.maximum(main, refl)
+
+
+def erase_original(image, plate, center, kind):
+    a = object_alpha(*image.shape[:2], center, kind, reflection=True)
+    # A strong clean replacement prevents colored antialiasing remnants.
+    a = np.clip(a * 1.12, 0, 1)[..., None]
+    return image * (1 - a) + plate * a
+
+
+def sprite_layers(source, plate, center, kind):
+    """Return reflection and body alpha layers for an observed marble."""
+    h, w = source.shape[:2]
+    main = object_alpha(h, w, center, kind, reflection=False)
+    full = object_alpha(h, w, center, kind, reflection=True)
+    refl_geom = np.clip(full - main, 0, 1)
+    # Reflections are already composited into source pixels. Select only where
+    # they measurably differ from the clean table, avoiding a visible cutout.
+    diff = np.sqrt(np.mean((source.astype(float) - plate.astype(float)) ** 2,
+                           axis=2))
+    refl = refl_geom * np.clip((diff - 0.8) / 4.0, 0, 1)
+    return refl, main
+
+
+def translate_layer(rgb, alpha, dx, dy):
+    """Translate without wrapping; source imagery is naturally antialiased."""
+    h, w = alpha.shape
+    out_rgb = np.zeros_like(rgb)
+    out_a = np.zeros_like(alpha)
+    sx0, sx1 = max(0, -dx), min(w, w - dx)
+    sy0, sy1 = max(0, -dy), min(h, h - dy)
+    tx0, tx1 = sx0 + dx, sx1 + dx
+    ty0, ty1 = sy0 + dy, sy1 + dy
+    if sx1 > sx0 and sy1 > sy0:
+        out_rgb[ty0:ty1, tx0:tx1] = rgb[sy0:sy1, sx0:sx1]
+        out_a[ty0:ty1, tx0:tx1] = alpha[sy0:sy1, sx0:sx1]
+    return out_rgb, out_a
+
+
+def composite(image, rgb, alpha):
+    a = alpha[..., None]
+    return image * (1 - a) + rgb * a
+
+
+def paste_observed(image, frames, plate, observed_track, target, kind,
+                   layer):
+    # Select the observation nearest the requested horizontal location. This
+    # preserves the exact source shading and naturally changing table height.
+    j = int(np.argmin(np.abs(observed_track[:, 0] - target[0])))
+    src_center = observed_track[j]
+    refl, main = sprite_layers(frames[j], plate, src_center, kind)
+    alpha = refl if layer == "reflection" else main
+    dx = int(round(target[0] - src_center[0]))
+    dy = int(round(target[1] - src_center[1]))
+    rgb, alpha = translate_layer(frames[j], alpha, dx, dy)
+    return composite(image, rgb, alpha)
+
+
+def main():
+    frames = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    cyan, gold = tracks(frames)
+    plate = clean_plate(frames, cyan, gold)
+
+    # Collision is first resolved between source frames 14 and 15 (zero based).
+    # With the original e ~= .65 and a mass ratio near 4.5, reducing the small
+    # marble's restitution to 5% gives post-impact speeds of approximately
+    # -5.0 and -4.8 px/frame. Their tiny separation speed is immediately
+    # overcome by rolling friction, so they travel in sustained contact.
+    cyan_cf = cyan.copy()
+    gold_cf = gold.copy()
+    contact_frame = 14
+    c0 = np.array([128.3, 72.45])
+    g0 = np.array([141.5, 68.26])
+    separation = g0[0] - c0[0]
+    v0 = -4.90
+    decel = .260
+    stop_n = -v0 / decel
+    for i in range(contact_frame, 96):
+        n = min(float(i - contact_frame), stop_n)
+        x = c0[0] + v0 * n + .5 * decel * n * n
+        cyan_cf[i] = (x, 75.60 - .0250 * x)
+        gold_cf[i] = (x + separation, 74.32 - .0428 * (x + separation))
+
+    out = frames.copy()
+    # Up through the contact configuration, the factual and counterfactual
+    # images coincide exactly. Consequences appear on the following frame.
+    for i in range(contact_frame + 1, 96):
+        base = frames[i].astype(float)
+        base = erase_original(base, plate, cyan[i], "cyan")
+        base = erase_original(base, plate, gold[i], "gold")
+
+        # Both reflections belong behind both solid bodies.
+        base = paste_observed(base, frames, plate, cyan, cyan_cf[i],
+                              "cyan", "reflection")
+        base = paste_observed(base, frames, plate, gold, gold_cf[i],
+                              "gold", "reflection")
+        base = paste_observed(base, frames, plate, cyan, cyan_cf[i],
+                              "cyan", "body")
+        base = paste_observed(base, frames, plate, gold, gold_cf[i],
+                              "gold", "body")
+        out[i] = np.rint(base).clip(0, 255).astype(np.uint8)
+
+    artifact_io.save(out, CASE_DIR)
+
+
+if __name__ == "__main__":
+    main()
