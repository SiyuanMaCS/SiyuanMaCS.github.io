+"""Remove the soda cup and reconstruct the resulting rigid-body motion.
+
+The camera and room are static.  A temporal median of the early source frames
+therefore provides the real, source-derived table/background hidden by the
+three small moving props.  The red and green prop appearances are lifted from
+frame 1; after the cup is removed they undergo a direct, nearly-inelastic
+collision and slide out of the right side of the shot.
+"""
+from pathlib import Path
+
+import numpy as np
+from PIL import Image, ImageFilter
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def component_center(mask, preferred_x=None):
+    """Center of the useful colored pixels, optionally near a prior x."""
+    yy, xx = np.where(mask)
+    if len(xx) == 0:
+        return preferred_x
+    if preferred_x is not None:
+        keep = np.abs(xx - preferred_x) < 22
+        if keep.any():
+            xx, yy = xx[keep], yy[keep]
+    return float(np.median(xx))
+
+
+def centers(frames):
+    """Track each source object from its distinctive label color."""
+    tracks = {"red": [], "cup": [], "green": []}
+    prior = {"red": 29.0, "cup": 99.0, "green": 170.0}
+    yy = np.arange(frames.shape[1])[:, None]
+    band = (yy >= 61) & (yy <= 83)
+    for frame in frames:
+        p = frame.astype(np.float32)
+        r, g, b = p[..., 0], p[..., 1], p[..., 2]
+        masks = {
+            "red": band & (r > 105) & (r > 1.24 * g) & (r > 1.24 * b),
+            "cup": band & (r > 135) & (g > 103) & (b < 112)
+                   & (r > 1.30 * b) & (g > 1.20 * b),
+            "green": band & (g > 46) & (g > 1.07 * r) & (g > 1.10 * b),
+        }
+        for name in tracks:
+            x = component_center(masks[name], prior[name])
+            tracks[name].append(x)
+            prior[name] = x
+    return tracks
+
+
+def clean_patch(frame, background, x0, y0, x1, y1):
+    """Insert background with a soft one-pixel perimeter to hide compression."""
+    h, w = frame.shape[:2]
+    x0, x1 = max(0, x0), min(w, x1)
+    y0, y1 = max(0, y0), min(h, y1)
+    if x1 <= x0 or y1 <= y0:
+        return
+    # A wide inner patch removes the prop and its contact shadow.  Blend only
+    # the outer two pixels; the underlying scene is static and aligned.
+    ph, pw = y1 - y0, x1 - x0
+    alpha = Image.new("L", (pw, ph), 255)
+    a = np.asarray(alpha).copy()
+    a[[0, -1], :] = 72
+    a[:, [0, -1]] = 72
+    if ph > 3 and pw > 3:
+        a[[1, -2], 1:-1] = 190
+        a[1:-1, [1, -2]] = 190
+    alpha = Image.fromarray(a)
+    dst = Image.fromarray(frame[y0:y1, x0:x1])
+    bg = Image.fromarray(background[y0:y1, x0:x1])
+    frame[y0:y1, x0:x1] = np.asarray(Image.composite(bg, dst, alpha))
+
+
+def make_sprite(source, background, box):
+    """Return an RGBA sprite using temporal-background difference as alpha."""
+    x0, y0, x1, y1 = box
+    crop = source[y0:y1, x0:x1]
+    bg = background[y0:y1, x0:x1]
+    diff = np.max(np.abs(crop.astype(np.int16) - bg.astype(np.int16)), axis=2)
+    # Keep antialiasing and the subtle contact shadow while rejecting codec
+    # noise.  MaxFilter closes tiny gaps in labels and along pale boundaries.
+    alpha = np.where(diff >= 5, 255, np.where(diff >= 3, 120, 0)).astype(np.uint8)
+    alpha = np.asarray(Image.fromarray(alpha).filter(ImageFilter.MaxFilter(3)))
+    alpha = np.asarray(Image.fromarray(alpha).filter(ImageFilter.GaussianBlur(0.42)))
+    rgba = np.dstack([crop, alpha]).astype(np.uint8)
+    return Image.fromarray(rgba, "RGBA"), (x0, y0)
+
+
+def paste_centered(frame, sprite, source_origin, source_center, new_center):
+    """Paste a source-aligned sprite at a new horizontal center."""
+    im = Image.fromarray(frame)
+    dx = int(round(new_center - source_center))
+    im.alpha_composite(sprite, (source_origin[0] + dx, source_origin[1])) \
+        if im.mode == "RGBA" else im.paste(sprite, (source_origin[0] + dx, source_origin[1]), sprite)
+    return np.asarray(im.convert("RGB"))
+
+
+def generate():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    # Across the first 36 frames every small prop traverses enough table area
+    # that the median is a clean plate, while all unaffected scene content is
+    # retained exactly from the source.
+    background = np.median(source[:36], axis=0).astype(np.uint8)
+    tr = centers(source)
+
+    red_center0 = tr["red"][0]
+    green_center0 = tr["green"][0]
+    red_sprite, red_origin = make_sprite(source[0], background, (20, 58, 38, 87))
+    green_sprite, green_origin = make_sprite(source[0], background, (162, 58, 178, 87))
+
+    output = source.copy()
+    for i in range(len(output)):
+        frame = output[i]
+        # Remove the complete factual three-object chain.  The cup rectangle
+        # includes its white lid and straw, not just its yellow label.
+        xr, xc, xg = tr["red"][i], tr["cup"][i], tr["green"][i]
+        clean_patch(frame, background, int(xr) - 10, 57, int(xr) + 12, 87)
+        clean_patch(frame, background, int(xc) - 9, 53, int(xc) + 11, 87)
+        clean_patch(frame, background, int(xg) - 9, 57, int(xg) + 11, 87)
+
+        # With no cup, the red carton travels freely until frame 8, where it
+        # contacts the stationary green carton.  Their low-restitution direct
+        # collision halves its speed.  Source-observed table friction is about
+        # 0.28 px/frame^2; the pair consequently slides out of view.
+        if i <= 7:
+            red_x = 28.6 + 19.35 * i - 0.145 * i * i
+            green_x = 169.7
+        else:
+            s = i - 7
+            displacement = 9.2 * s - 0.14 * s * s
+            displacement = max(0.0, displacement)
+            red_x = 158.0 + displacement
+            green_x = 169.7 + displacement
+
+        if red_x < 267:
+            frame = paste_centered(frame, red_sprite, red_origin, red_center0, red_x)
+        if green_x < 266:
+            frame = paste_centered(frame, green_sprite, green_origin, green_center0, green_x)
+        output[i] = frame
+
+    return output.astype(np.uint8)
+
+
+if __name__ == "__main__":
+    artifact_io.save(generate(), CASE_DIR)
