+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def interp_track(t):
+    """Observed centre/radius of the soccer (star-patterned) ball."""
+    key_t = np.array([0, 8, 12, 16, 20, 24, 28, 32, 36, 40, 44, 52, 95], np.float32)
+    key_x = np.array([100, 100, 100, 99, 98, 98, 100, 102, 103, 103, 104, 104, 104], np.float32)
+    key_y = np.array([92, 93, 94, 96, 99, 103, 106, 108, 110, 112, 114, 115, 115], np.float32)
+    key_r = np.array([13.0, 13.0, 13.2, 14.0, 15.0, 16.2, 17.2,
+                      18.0, 18.8, 19.3, 19.7, 20.0, 20.0], np.float32)
+    return (float(np.interp(t, key_t, key_x)),
+            float(np.interp(t, key_t, key_y)),
+            float(np.interp(t, key_t, key_r)))
+
+
+def soft_object_alpha(h, w, cx, cy, radius):
+    """Feathered ball silhouette plus its soft floor contact shadow."""
+    yy, xx = np.mgrid[:h, :w].astype(np.float32)
+    distance = np.sqrt((xx - cx) ** 2 + (yy - cy) ** 2)
+    # A narrow antialiased edge keeps the photographed contour, without a halo.
+    ball = np.clip(radius + 1.0 - distance, 0.0, 1.0)
+
+    # The shadow is part of the source pixels too.  Its low alpha softly carries
+    # the real source shadow while avoiding a hard, pasted rectangular patch.
+    sx, sy = cx - 3.0, cy + radius * 0.72
+    q = ((xx - sx) / (radius * 1.20)) ** 2 + ((yy - sy) / (radius * 0.38)) ** 2
+    shadow = np.clip((1.22 - q) / 0.34, 0.0, 1.0) * 0.58
+    return np.maximum(ball, shadow)[..., None]
+
+
+def removal_mask(h, w, cx, cy, radius):
+    """Mask the factual ball and shadow before placing its slower version."""
+    yy, xx = np.mgrid[:h, :w].astype(np.float32)
+    ball = (xx - cx) ** 2 + (yy - cy) ** 2 <= (radius + 1.5) ** 2
+    sx, sy = cx - 3.0, cy + radius * 0.74
+    shadow = ((xx - sx) / (radius * 1.25)) ** 2 + ((yy - sy) / (radius * 0.43)) ** 2 <= 1.0
+    return ((ball | shadow).astype(np.uint8) * 255)
+
+
+def source_at(frames, t):
+    """Linear source sampling gives smooth 18-fps-equivalent motion at 24 fps."""
+    lo = int(np.floor(t))
+    hi = min(lo + 1, len(frames) - 1)
+    a = np.float32(t - lo)
+    return frames[lo].astype(np.float32) * (1.0 - a) + frames[hi].astype(np.float32) * a
+
+
+def generate():
+    frames = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    n, h, w, _ = frames.shape
+    output = frames.copy()
+
+    # The intervention is active from source frame 1 (array index 0).  Position
+    # at the instant of intervention is unchanged; elapsed trajectory time is
+    # subsequently 0.75 of factual elapsed time.
+    for i in range(1, n):
+        slower_t = 0.75 * i
+
+        # Once both factual and retimed trajectories are at rest, retaining the
+        # source frame exactly is the cleanest physically equivalent result.
+        if slower_t >= 52.0:
+            output[i] = frames[i]
+            continue
+
+        old_x, old_y, old_r = interp_track(float(i))
+        mask = removal_mask(h, w, old_x, old_y, old_r)
+
+        # Telea reconstruction is only exposed in the small strip between the
+        # old and new silhouettes; the slower ball covers nearly all of it.
+        bgr = cv2.cvtColor(frames[i], cv2.COLOR_RGB2BGR)
+        clean = cv2.inpaint(bgr, mask, 3.0, cv2.INPAINT_TELEA)
+        clean = cv2.cvtColor(clean, cv2.COLOR_BGR2RGB).astype(np.float32)
+
+        new_x, new_y, new_r = interp_track(slower_t)
+        photographed = source_at(frames, slower_t)
+        alpha = soft_object_alpha(h, w, new_x, new_y, new_r)
+        composed = clean * (1.0 - alpha) + photographed * alpha
+        output[i] = np.clip(np.rint(composed), 0, 255).astype(np.uint8)
+
+    # Ensure intervention-frame continuity is exact.
+    output[0] = frames[0]
+    return output
+
+
+if __name__ == "__main__":
+    result = generate()
+    artifact_io.save(result, CASE_DIR)
+    print(f"saved {result.shape[0]} frames at {result.shape[2]}x{result.shape[1]}")
