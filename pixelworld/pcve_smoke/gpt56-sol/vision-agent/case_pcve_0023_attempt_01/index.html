+"""Remove the small football from source frame 32 onward.
+
+The shot is locked off.  We build a clean floor plate from the first native-
+resolution frame, use it only under the football, and restore the part of the
+pink ball which the football used to occlude.  Frames 1..31 are byte-for-byte
+the supplied sampled source frames.

+

+
+import cv2
+import numpy as np

+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+SCALE = 5  # native source is 1280x720; requested output is 256x144

+def read_native_video(path: Path) -> list[np.ndarray]:
+    cap = cv2.VideoCapture(str(path))
+    frames = []
+    while True:
+        ok, bgr = cap.read()
+        if not ok:
+            break
+        frames.append(cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB))
+    cap.release()
+    if len(frames) != 96:
+        raise RuntimeError(f"Expected 96 native frames, got {len(frames)}")
+    return frames
+

+def soft_ellipse(shape, center, axes, feather=1.4):
+    """Anti-aliased ellipse matte in output-pixel coordinates."""
+    h, w = shape
+    yy, xx = np.mgrid[:h, :w].astype(np.float32)
+    cx, cy = center
+    ax, ay = axes
+    d = np.sqrt(((xx - cx) / ax) ** 2 + ((yy - cy) / ay) ** 2)
+    # Convert radial distance to approximately pixel-sized feathering.
+    edge = feather / max(ax, ay)
+    return np.clip((1.0 + edge - d) / (2.0 * edge), 0.0, 1.0)
+
+
+def football_track(i: int):
+    """Measured football circle on the 256x144 source timeline."""
+    # During the last part of its roll it advances toward camera, then settles.
+    if i <= 39:
+        t = (i - 31) / 8.0
+        cx = 103.5 + 0.5 * t
+        cy = 104.5 + 8.5 * t
+        r = 16.4 + 1.1 * t
+    else:
+        cx, cy, r = 104.0, 112.5, 17.8
+    return cx, cy, r
+
+
+def make_clean_floor(native0: np.ndarray) -> np.ndarray:
+    """Inpaint the football in frame 1 at native resolution, then downsample."""
+    h, w = native0.shape[:2]
+    mask = np.zeros((h, w), np.uint8)
+    # Ball plus its soft contact/cast shadow.  This location is floor-only.
+    cv2.circle(mask, (501, 460), 83, 255, -1, lineType=cv2.LINE_AA)
+    cv2.ellipse(mask, (493, 510), (92, 34), 0, 0, 360, 255, -1,
+                lineType=cv2.LINE_AA)
+    # TELEA is particularly clean here because the floor grain supplies strong
+    # directional context on all sides of the small masked area.
+    bgr = cv2.cvtColor(native0, cv2.COLOR_RGB2BGR)
+    clean = cv2.inpaint(bgr, mask, 25, cv2.INPAINT_TELEA)
+    clean = cv2.cvtColor(clean, cv2.COLOR_BGR2RGB)
+    return cv2.resize(clean, (256, 144), interpolation=cv2.INTER_AREA)
+
+
+def make_pink_template(native: np.ndarray):
+    """Extract the unobstructed pink ball from source frame 14."""
+    # Measured circle in native frame 14. It is fully visible immediately
+    # before contact with the football.
+    cx, cy, radius = 433.5, 328.0, 52.0
+    pad = 58
+    x0, y0 = int(round(cx)) - pad, int(round(cy)) - pad
+    patch = native[y0:y0 + 2 * pad + 1, x0:x0 + 2 * pad + 1].copy()
+    yy, xx = np.mgrid[:patch.shape[0], :patch.shape[1]].astype(np.float32)
+    pcx, pcy = cx - x0, cy - y0
+    dist = np.sqrt((xx - pcx) ** 2 + (yy - pcy) ** 2)
+    alpha = np.clip((radius + 1.4 - dist) / 2.8, 0.0, 1.0)
+    return patch, alpha
+
+
+def resize_template(patch, alpha, target_radius=17.3):
+    # The native template radius is 52 px; produce a compact output-resolution
+    # sprite while retaining the highlight and subtle mottling of the real ball.
+    scale = target_radius / 52.0
+    size = int(round(patch.shape[0] * scale))
+    if size % 2 == 0:
+        size += 1
+    rgb = cv2.resize(patch, (size, size), interpolation=cv2.INTER_AREA)
+    a = cv2.resize(alpha, (size, size), interpolation=cv2.INTER_AREA)
+    return rgb.astype(np.float32), a.astype(np.float32)
+
+
+def paste_missing_pink(frame, sprite, alpha, football_matte):
+    """Restore only pink pixels that were hidden by the removed football."""
+    # The pink ball has already stopped at this measured source position.
+    cx, cy = 105.6, 83.7
+    sh, sw = alpha.shape
+    x0 = int(round(cx - (sw - 1) / 2))
+    y0 = int(round(cy - (sh - 1) / 2))
+    x1, y1 = x0 + sw, y0 + sh
+
+    # The visible upper half remains untouched source imagery.  Restrict the
+    # sprite to exactly the area opened by football removal, with a tiny seam
+    # blend into the already-visible pink boundary.
+    fm = football_matte[y0:y1, x0:x1]
+    use = alpha * np.clip(fm * 1.25, 0.0, 1.0)
+    roi = frame[y0:y1, x0:x1].astype(np.float32)
+    roi = roi * (1.0 - use[..., None]) + sprite * use[..., None]
+    frame[y0:y1, x0:x1] = np.clip(roi + 0.5, 0, 255).astype(np.uint8)
+
+
+def generate() -> np.ndarray:
+    source = np.load(ROOT / "source_pixels.npz")["rgb"]
+    if source.shape != (96, 144, 256, 3) or source.dtype != np.uint8:
+        raise RuntimeError(f"Unexpected source tensor: {source.shape} {source.dtype}")
+    native = read_native_video(ROOT / "source_video.mp4")
+
+    clean_floor = make_clean_floor(native[0]).astype(np.float32)
+    pink_patch, pink_alpha = make_pink_template(native[13])
+    pink_sprite, pink_matte = resize_template(pink_patch, pink_alpha)
+
+    out = source.copy()
+    for i in range(31, 96):  # source frame 32 is zero-based array index 31
+        cx, cy, r = football_track(i)
+
+        # Ball body. Expansion by ~1.3 px removes its antialiased rim.
+        body = soft_ellipse((144, 256), (cx, cy), (r + 1.35, r + 1.35), 1.15)
+        # Floor shadow remains visible below/left even when the lower ball clips
+        # the image. It is deliberately a lighter matte so floor texture survives.
+        shadow = soft_ellipse((144, 256), (cx - 1.8, cy + r * 0.72),
+                              (r * 1.18, max(3.5, r * 0.34)), 1.6) * 0.70
+        removal = np.maximum(body, shadow)
+
+        base = out[i].astype(np.float32)
+        base = base * (1.0 - removal[..., None]) + clean_floor * removal[..., None]
+        out[i] = np.clip(base + 0.5, 0, 255).astype(np.uint8)
+
+        # Fill the football-occluded lower part of the stationary pink ball.
+        paste_missing_pink(out[i], pink_sprite, pink_matte, body)
+
+    return out
+
+
+if __name__ == "__main__":
+    frames = generate()
+    artifact_io.save(frames, ROOT)
+    print("Saved", frames.shape, frames.dtype, "to", ROOT)
