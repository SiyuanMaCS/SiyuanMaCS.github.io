+"""Generate the counterfactual tennis video.
+
+The scene is static apart from the tennis ball.  We estimate the original
+ballistic launch from the supplied RGB frames, remove the factual ball from
+each frame with a temporal clean plate, and render a ball whose initial
+velocity vector is 1.5 times as large.  Gravity is deliberately left at its
+observed value: the requested intervention changes speed, not the world.
+"""
+
+from pathlib import Path
+
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def ball_centres(frames: np.ndarray, clean: np.ndarray) -> np.ndarray:
+    """Locate the small yellow-green ball in every factual source frame."""
+    centres = []
+    for frame in frames:
+        rgb = frame.astype(np.int16)
+        delta = np.max(np.abs(rgb - clean.astype(np.int16)), axis=2)
+        r, g, b = rgb[..., 0], rgb[..., 1], rgb[..., 2]
+        # Chroma rejects compression flicker and the temporal difference
+        # rejects the similarly coloured strip of grass.
+        mask = (delta > 8) & (g > r + 8) & (r > b + 25) & (g > 85)
+        yy, xx = np.nonzero(mask)
+        if len(xx) == 0:
+            raise RuntimeError("Could not recover the factual ball position")
+        centres.append((float(xx.mean()), float(yy.mean())))
+    return np.asarray(centres, dtype=np.float64)
+
+
+def erase_factual_ball(frame: np.ndarray, clean: np.ndarray,
+                       centre: np.ndarray) -> np.ndarray:
+    """Restore a feathered, ball-sized area from the temporal clean plate."""
+    result = frame.astype(np.float64).copy()
+    cx, cy = centre
+    x0, x1 = max(0, int(cx) - 8), min(frame.shape[1], int(cx) + 9)
+    y0, y1 = max(0, int(cy) - 8), min(frame.shape[0], int(cy) + 9)
+    yy, xx = np.mgrid[y0:y1, x0:x1]
+    distance = np.hypot(xx - cx, yy - cy)
+    # The visible ball radius is about 3 px.  The wider feather also removes
+    # its tiny codec/antialias halo without making a visible patch.
+    amount = np.clip((6.4 - distance) / 1.5, 0.0, 1.0)[..., None]
+    result[y0:y1, x0:x1] = (
+        result[y0:y1, x0:x1] * (1.0 - amount)
+        + clean[y0:y1, x0:x1].astype(np.float64) * amount
+    )
+    return result
+
+
+def render_ball(frame: np.ndarray, cx: float, cy: float) -> None:
+    """Antialiased shaded tennis ball, matched to the source's 5 px sprite."""
+    h, w = frame.shape[:2]
+    radius = 2.55
+    x0, x1 = max(0, int(np.floor(cx - 4))), min(w, int(np.ceil(cx + 4)) + 1)
+    y0, y1 = max(0, int(np.floor(cy - 4))), min(h, int(np.ceil(cy + 4)) + 1)
+    if x0 >= x1 or y0 >= y1:
+        return
+
+    # Supersampling preserves the soft source-video edge at fractional pixel
+    # positions.  Lighting is from the upper left, as on the factual ball.
+    ss = 8
+    light = np.array([-0.47, -0.54, 0.70], dtype=np.float64)
+    light /= np.linalg.norm(light)
+    accum_colour = np.zeros((y1 - y0, x1 - x0, 3), dtype=np.float64)
+    accum_alpha = np.zeros((y1 - y0, x1 - x0), dtype=np.float64)
+    base = np.array([142.0, 169.0, 23.0])
+
+    for sy in range(ss):
+        py = np.arange(y0, y1, dtype=np.float64)[:, None] + (sy + 0.5) / ss
+        for sx in range(ss):
+            px = np.arange(x0, x1, dtype=np.float64)[None, :] + (sx + 0.5) / ss
+            nx = (px - cx) / radius
+            ny = (py - cy) / radius
+            rr = nx * nx + ny * ny
+            inside = rr < 1.0
+            nz = np.sqrt(np.maximum(0.0, 1.0 - rr))
+            diffuse = np.maximum(0.0, nx * light[0] + ny * light[1] + nz * light[2])
+            # A matte felt sphere: gentle directional light and a small broad
+            # highlight, with darker grazing angles.
+            rim = 0.76 + 0.24 * nz
+            intensity = rim * (0.78 + 0.31 * diffuse)
+            highlight = 29.0 * np.power(diffuse, 6.0)
+            colour = base[None, None, :] * intensity[..., None]
+            colour += highlight[..., None] * np.array([1.0, 1.0, 0.55])
+            weight = inside.astype(np.float64) / (ss * ss)
+            accum_colour += colour * weight[..., None]
+            accum_alpha += weight
+
+    patch = frame[y0:y1, x0:x1]
+    alpha = accum_alpha[..., None]
+    # accum_colour is already coverage-weighted.
+    patch[:] = patch * (1.0 - alpha) + accum_colour
+
+
+def generate() -> np.ndarray:
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    clean = np.median(source, axis=0).astype(np.uint8)
+    factual = ball_centres(source, clean)
+
+    # Fit only the airborne portion (frames 1..28, zero-based 0..27).  Frame
+    # 28 is the first ground-contact image and later frames are rolling/sliding.
+    n = np.arange(28, dtype=np.float64)
+    x0, y0 = factual[0]
+    vx = np.linalg.lstsq(n[:, None], factual[:28, 0] - x0, rcond=None)[0][0]
+    vy, half_g = np.linalg.lstsq(
+        np.column_stack((n, n * n)), factual[:28, 1] - y0, rcond=None
+    )[0]
+
+    frames = np.empty_like(source)
+    speed_scale = 1.5
+    for index in range(len(source)):
+        if index == 0:
+            # Position at the instant of intervention is unchanged, and this
+            # retains the exact factual initial ball appearance.
+            frames[index] = source[index]
+            continue
+
+        canvas = erase_factual_ball(source[index], clean, factual[index])
+        t = float(index)
+        cx = x0 + speed_scale * vx * t
+        cy = y0 + speed_scale * vy * t + half_g * t * t
+        render_ball(canvas, cx, cy)
+        frames[index] = np.clip(np.rint(canvas), 0, 255).astype(np.uint8)
+
+    return frames
+
+
+if __name__ == "__main__":
+    output = generate()
+    artifact_io.save(output, CASE_DIR)
+    print(f"saved {len(output)} frames to {CASE_DIR}")
+
