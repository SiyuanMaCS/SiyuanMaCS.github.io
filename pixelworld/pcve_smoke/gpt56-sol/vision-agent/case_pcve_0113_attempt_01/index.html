+"""Generate the requested 18x-restitution counterfactual tennis video.
+
+The source is used verbatim through the first court contact.  After contact,
+the low-restitution source ball is removed from each source frame and a ball
+is composited on the counterfactual ballistic trajectory.  The static camera,
+court, net, lighting, and all unrelated image motion therefore remain sourced
+directly from the factual video.
+"""
+
+from pathlib import Path
+
+import numpy as np
+from PIL import Image, ImageDraw
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+CONTACT_FRAME = 28          # zero-based: source frame 29, first court contact
+CONTACT_X = 153.125
+CONTACT_Y = 94.0
+GRAVITY_Y = 0.42335         # pixels / frame^2, fit from frames 1..27
+INCOMING_VY = 7.25          # pixels / frame at impact
+TANGENTIAL_VX = 4.52        # pixels / frame immediately before impact
+ORIGINAL_RESTITUTION = 0.05
+NEW_RESTITUTION = 18.0 * ORIGINAL_RESTITUTION  # 0.90
+
+
+def yellow_ball_center(frame: np.ndarray) -> tuple[float, float]:
+    """Locate the compact yellow-green ball component in a source frame."""
+    p = frame.astype(np.int16)
+    r, g, b = p[..., 0], p[..., 1], p[..., 2]
+    mask = (
+        (g > 75)
+        & (g * 100 > r * 105)
+        & (r > 45)
+        & (b < 90)
+        & (r + g - 2 * b > 80)
+    )
+
+    seen = np.zeros(mask.shape, dtype=bool)
+    components = []
+    for y, x in np.argwhere(mask):
+        if seen[y, x]:
+            continue
+        stack = [(int(y), int(x))]
+        seen[y, x] = True
+        points = []
+        while stack:
+            yy, xx = stack.pop()
+            points.append((yy, xx))
+            for dy in (-1, 0, 1):
+                for dx in (-1, 0, 1):
+                    ny, nx = yy + dy, xx + dx
+                    if (
+                        0 <= ny < mask.shape[0]
+                        and 0 <= nx < mask.shape[1]
+                        and mask[ny, nx]
+                        and not seen[ny, nx]
+                    ):
+                        seen[ny, nx] = True
+                        stack.append((ny, nx))
+        if len(points) >= 3:
+            a = np.asarray(points)
+            if np.ptp(a[:, 0]) <= 10 and np.ptp(a[:, 1]) <= 10:
+                components.append((len(points), a[:, 1].mean(), a[:, 0].mean()))
+
+    if not components:
+        raise RuntimeError("Could not locate source tennis ball")
+    _, x, y = max(components)
+    return float(x), float(y)
+
+
+def remove_source_ball(frame: np.ndarray, cx: float, cy: float) -> np.ndarray:
+    """Remove the 7px source ball with row-wise court-aware interpolation."""
+    result = frame.copy()
+    x = int(round(cx))
+    y = int(round(cy))
+    x0, x1 = max(1, x - 5), min(frame.shape[1] - 2, x + 5)
+    y0, y1 = max(0, y - 5), min(frame.shape[0] - 1, y + 5)
+
+    # Interpolating independently on every row preserves the horizontal court
+    # stripe crossing behind the rolling source ball at y ~= 95.
+    for yy in range(y0, y1 + 1):
+        left = frame[yy, x0 - 1].astype(np.float32)
+        right = frame[yy, x1 + 1].astype(np.float32)
+        count = x1 - x0 + 1
+        alpha = np.linspace(1.0 / (count + 1), count / (count + 1), count)
+        row = left[None, :] * (1.0 - alpha[:, None]) + right[None, :] * alpha[:, None]
+        result[yy, x0 : x1 + 1] = np.clip(row + 0.5, 0, 255).astype(np.uint8)
+    return result
+
+
+def composite_ball(frame: np.ndarray, cx: float, cy: float) -> np.ndarray:
+    """Draw a small anti-aliased tennis ball matching the source rendering."""
+    h, w = frame.shape[:2]
+    scale = 4
+    overlay = Image.new("RGBA", (w * scale, h * scale), (0, 0, 0, 0))
+    draw = ImageDraw.Draw(overlay, "RGBA")
+    x, y = cx * scale, cy * scale
+
+    def ellipse(radius, color, dx=0.0, dy=0.0):
+        rr = radius * scale
+        ox, oy = dx * scale, dy * scale
+        draw.ellipse((x + ox - rr, y + oy - rr, x + ox + rr, y + oy + rr), fill=color)
+
+    # Olive antialiased rim, shaded lower-right body, then lime face/highlight.
+    ellipse(3.45, (86, 102, 31, 205), 0.10, 0.16)
+    ellipse(2.95, (116, 145, 24, 255), 0.25, 0.28)
+    ellipse(2.60, (147, 180, 27, 255), -0.18, -0.22)
+    ellipse(1.20, (183, 207, 82, 210), -0.82, -0.88)
+    ellipse(0.60, (201, 217, 115, 180), -1.03, -1.08)
+
+    overlay = overlay.resize((w, h), Image.Resampling.LANCZOS)
+    base = Image.fromarray(frame, "RGB").convert("RGBA")
+    return np.asarray(Image.alpha_composite(base, overlay).convert("RGB"), dtype=np.uint8)
+
+
+def generate() -> np.ndarray:
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3) and source.dtype == np.uint8
+    output = source.copy()
+
+    outgoing_vy = -NEW_RESTITUTION * INCOMING_VY
+    for index in range(CONTACT_FRAME + 1, len(output)):
+        # First erase the factual low-restitution ball (including its edge
+        # antialiasing); retain the independently evolving source background.
+        old_x, old_y = yellow_ball_center(source[index])
+        frame = remove_source_ball(source[index], old_x, old_y)
+
+        t = float(index - CONTACT_FRAME)
+        new_x = CONTACT_X + TANGENTIAL_VX * t
+        new_y = CONTACT_Y + outgoing_vy * t + 0.5 * GRAVITY_Y * t * t
+        if -5.0 < new_x < 261.0 and -5.0 < new_y < 149.0:
+            frame = composite_ball(frame, new_x, new_y)
+        output[index] = frame
+
+    return output
+
+
+if __name__ == "__main__":
+    frames = generate()
+    artifact_io.save(frames, CASE_DIR)
+    print(
+        f"saved {len(frames)} frames; restitution {ORIGINAL_RESTITUTION:.2f} -> "
+        f"{NEW_RESTITUTION:.2f}; contact frame {CONTACT_FRAME + 1}"
+    )
