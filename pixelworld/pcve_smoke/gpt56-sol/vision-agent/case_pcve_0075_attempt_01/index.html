+#!/usr/bin/env python3
+"""Generate the 4x-initial-speed counterfactual for the white die.
+
+The scene is static, so a robust temporal background is used.  Native die
+appearances are extracted from unobstructed source frames and composited over a
+small deterministic rigid-body simulation.  The faster white die immediately
+catches the offset red die, transferring horizontal momentum to both.
+"""
+from pathlib import Path
+import math
+import numpy as np
+from PIL import Image, ImageDraw, ImageFilter
+from scipy.ndimage import label, binary_dilation, binary_fill_holes
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+N, H, W = 96, 144, 256
+
+
+def clean_background(src):
+    # During the first twenty frames neither die occupies any pixel for half of
+    # the samples.  The median therefore exactly recovers the static artwork.
+    return np.median(src[:20].astype(np.float32), axis=0).astype(np.uint8)
+
+
+def extract_sprite(src, bg, frame, center, half=(11, 13)):
+    """Extract a soft-edged RGBA sprite around one isolated source die."""
+    cx, cy = center
+    hw, hh = half
+    x0, y0 = int(round(cx - hw)), int(round(cy - hh))
+    x1, y1 = int(round(cx + hw + 1)), int(round(cy + hh + 1))
+    rgb = src[frame, y0:y1, x0:x1].copy()
+    ref = bg[y0:y1, x0:x1]
+    delta = np.max(np.abs(rgb.astype(np.int16) - ref.astype(np.int16)), axis=2)
+    hard = delta > 12
+    lab, count = label(hard)
+    # Select all substantial changed components close to crop center. This
+    # retains disconnected pips/edge antialiasing while rejecting scene marks.
+    yy, xx = np.mgrid[:hard.shape[0], :hard.shape[1]]
+    support = np.zeros_like(hard)
+    for k in range(1, count + 1):
+        m = lab == k
+        area = int(m.sum())
+        if not area:
+            continue
+        mx, my = float(xx[m].mean()), float(yy[m].mean())
+        if area >= 3 and (mx-hw)**2 + (my-hh)**2 < 14**2:
+            support |= m
+    support = binary_fill_holes(binary_dilation(support, iterations=1))
+    alpha = np.clip((delta.astype(np.float32) - 2.0) / 13.0, 0, 1)
+    alpha *= support
+    # Exclude detached soft shadows; those are rendered consistently below.
+    alpha[(yy > hh + 8) & (alpha < .92)] = 0
+    rgba = np.dstack([rgb, np.rint(alpha * 255).astype(np.uint8)])
+    return Image.fromarray(rgba, 'RGBA')
+
+
+def make_sprite_sets(src, bg):
+    # Unobstructed native appearances sampled as the dice tumble.
+    white_specs = [(2, (132, 20)), (3, (132, 29)), (4, (132, 39)),
+                   (5, (132, 51)), (6, (132, 64))]
+    red_specs = [(0, (123, 21)), (1, (123, 29)), (2, (123, 40)),
+                 (3, (123, 52)), (4, (123, 65))]
+    whites = [extract_sprite(src, bg, f, c) for f, c in white_specs]
+    reds = [extract_sprite(src, bg, f, c) for f, c in red_specs]
+    return whites, reds
+
+
+def simulate():
+    """Two equal-mass dice in screen-space perspective, sampled at 24 fps."""
+    # Source initial screen positions and velocities. The white source launch is
+    # about 5.5 px/frame downward; only it is multiplied by four.
+    p = np.array([[132.0, 7.0], [123.0, 21.0]], np.float64)  # white, red
+    v = np.array([[0.0, 22.0], [0.0, 8.0]], np.float64)
+    radii = np.array([8.0, 7.5])
+    floor = np.array([86.0, 87.0])
+    gravity, restitution = 1.45, 0.55
+    history_p, history_v = [], []
+    for frame in range(N):
+        history_p.append(p.copy())
+        history_v.append(v.copy())
+        # Substeps make the high-speed first impact stable and avoid tunneling.
+        for _ in range(24):
+            dt = 1.0 / 24.0
+            v[:, 1] += gravity * dt
+            p += v * dt
+
+            # Dice-dice impact. The initial horizontal offset naturally splits
+            # the boosted vertical impulse into left/right motion.
+            d = p[1] - p[0]
+            dist = float(np.linalg.norm(d))
+            contact = float(radii.sum())
+            if 1e-5 < dist < contact:
+                normal = d / dist
+                closing = float(np.dot(v[1] - v[0], normal))
+                if closing < 0:
+                    impulse = -(1.0 + restitution) * closing / 2.0
+                    v[0] -= impulse * normal
+                    v[1] += impulse * normal
+                correction = (contact - dist) / 2.0 + 0.01
+                p[0] -= correction * normal
+                p[1] += correction * normal
+
+            # Felt surface and the raised left/right table walls.
+            for i in range(2):
+                if p[i, 1] > floor[i]:
+                    p[i, 1] = floor[i]
+                    if v[i, 1] > 2.2:
+                        v[i, 1] *= -0.42
+                    else:
+                        v[i, 1] = 0.0
+                    v[i, 0] *= 0.84
+                lo, hi = 66.0 + radii[i], 198.0 - radii[i]
+                if p[i, 0] < lo:
+                    p[i, 0] = lo
+                    v[i, 0] = abs(v[i, 0]) * 0.45
+                elif p[i, 0] > hi:
+                    p[i, 0] = hi
+                    v[i, 0] = -abs(v[i, 0]) * 0.45
+        for i in range(2):
+            if p[i, 1] >= floor[i] - 0.01:
+                v[i, 0] *= 0.82
+                if v[i, 1] == 0 and abs(v[i, 0]) < 0.08:
+                    v[i, 0] = 0.0
+    return np.asarray(history_p), np.asarray(history_v), floor
+
+
+def add_shadow(frame, x, ground_y, height):
+    # Source shadows are small cool-gray ellipses that sharpen near contact.
+    layer = Image.new('RGBA', (W, H), (0, 0, 0, 0))
+    spread = 3.0 + min(height, 50.0) * 0.035
+    alpha = int(45 * math.exp(-height / 55.0) + 12)
+    d = ImageDraw.Draw(layer)
+    d.ellipse((x - spread, ground_y + 3 - 1.4,
+               x + spread, ground_y + 3 + 1.4), fill=(38, 54, 49, alpha))
+    layer = layer.filter(ImageFilter.GaussianBlur(1.35 + min(height, 50) * .012))
+    frame.alpha_composite(layer)
+
+
+def paste_center(canvas, sprite, x, y, opacity=1.0):
+    if opacity < 0.999:
+        sprite = sprite.copy()
+        a = np.asarray(sprite.getchannel('A'), dtype=np.float32)
+        sprite.putalpha(Image.fromarray(np.rint(a * opacity).astype(np.uint8)))
+    canvas.alpha_composite(sprite, (int(round(x - sprite.width/2)),
+                                    int(round(y - sprite.height/2))))
+
+
+def render(src):
+    bg = clean_background(src)
+    whites, reds = make_sprite_sets(src, bg)
+    positions, velocities, floors = simulate()
+    output = []
+    for t in range(N):
+        canvas = Image.fromarray(bg.copy(), 'RGB').convert('RGBA')
+        # Shadows remain on the felt while the cubes move vertically above them.
+        for i in range(2):
+            x, y = positions[t, i]
+            add_shadow(canvas, x, floors[i], max(0.0, floors[i] - y))
+
+        # A subtle exposure trail only during very fast motion communicates the
+        # intervention without obscuring the native pixel-art silhouettes.
+        if t > 0:
+            for i, sprites in ((0, whites), (1, reds)):
+                speed = float(np.linalg.norm(velocities[t, i]))
+                if speed > 14:
+                    prev = positions[t-1, i]
+                    cur = positions[t, i]
+                    sprite = sprites[(t * 3 + i) % len(sprites)]
+                    paste_center(canvas, sprite,
+                                 prev[0] * .65 + cur[0] * .35,
+                                 prev[1] * .65 + cur[1] * .35, .13)
+
+        # Back-to-front compositing handles crossings correctly.
+        order = sorted(range(2), key=lambda i: positions[t, i, 1])
+        for i in order:
+            p = positions[t, i]
+            moving = np.linalg.norm(velocities[t, i]) > .25
+            sprites = whites if i == 0 else reds
+            # Accelerated tumbling cycles native source orientations rapidly;
+            # once settled, the face remains fixed.
+            idx = ((t * (4 if i == 0 else 2)) % len(sprites)) if moving else (2 if i == 0 else 3)
+            paste_center(canvas, sprites[idx], p[0], p[1])
+        output.append(np.asarray(canvas.convert('RGB'), dtype=np.uint8))
+    return np.stack(output)
+
+
+def main():
+    src = np.load(ROOT / 'source_pixels.npz')['rgb']
+    assert src.shape == (N, H, W, 3) and src.dtype == np.uint8
+    frames = render(src)
+    assert frames.shape == (N, H, W, 3) and frames.dtype == np.uint8
+    artifact_io.save(frames, ROOT)
+    print('saved', frames.shape, frames.dtype, 'to', ROOT)
+
+
+if __name__ == '__main__':
+    main()

appearances are extracted from unobstructed source frames and composited over a
small deterministic rigid-body simulation.  The faster white die immediately
catches the offset red die, transferring horizontal momentum to both.
"""

import numpy as np

import artifact_io

N, H, W = 96, 144, 256

    # the samples.  The median therefore exactly recovers the static artwork.
    return np.median(src[:20].astype(np.float32), axis=0).astype(np.uint8)


def extract_sprite(src, bg, frame, center, half=(11, 13)):
    """Extract a soft-edged RGBA sprite around one isolated source die."""
    cx, cy = center
    hw, hh = half
    x0, y0 = int(round(cx - hw)), int(round(cy - hh))
    x1, y1 = int(round(cx + hw + 1)), int(round(cy + hh + 1))

    # retains disconnected pips/edge antialiasing while rejecting scene marks.

    support = np.zeros_like(hard)
    for k in range(1, count + 1):
        m = lab == k
        area = int(m.sum())
        if not area:
            continue
        mx, my = float(xx[m].mean()), float(yy[m].mean())
        if area >= 3 and (mx-hw)**2 + (my-hh)**2 < 14**2:
            support |= m
    support = binary_fill_holes(binary_dilation(support, iterations=1))
    alpha = np.clip((delta.astype(np.float32) - 2.0) / 13.0, 0, 1)
    alpha *= support
    # Exclude detached soft shadows; those are rendered consistently below.
    alpha[(yy > hh + 8) & (alpha < .92)] = 0
    rgba = np.dstack([rgb, np.rint(alpha * 255).astype(np.uint8)])
    return Image.fromarray(rgba, '"'RGBA')


def make_sprite_sets(src, bg):
    # Unobstructed native appearances sampled as the dice tumble.
    white_specs = [(2, (132, 20)), (3, (132, 29)), (4, (132, 39)),
                   (5, (132, 51)), (6, (132, 64))]
    red_specs = [(0, (123, 21)), (1, (123, 29)), (2, (123, 40)),
                 (3, (123, 52)), (4, (123, 65))]
    whites = [extract_sprite(src, bg, f, c) for f, c in white_specs]
    reds = [extract_sprite(src, bg, f, c) for f, c in red_specs]
    return whites, reds


def simulate():
    \"\"\"Two equal-mass dice in screen-space perspective, sampled at 24 fps.\"\"\"
    # Source initial screen positions and velocities. The white source launch is
    # about 5.5 px/frame downward; only it is multiplied by four.
    p = np.array([[132.0, 7.0], [123.0, 21.0]], np.float64)  # white, red
    v = np.array([[0.0, 22.0], [0.0, 8.0]], np.float64)
    radii = np.array([8.0, 7.5])
    floor = np.array([86.0, 87.0])
    gravity, restitution = 1.45, 0.55
    history_p, history_v = [], []
    for frame in range(N):
        history_p.append(p.copy())
        history_v.append(v.copy())
        # Substeps make the high-speed first impact stable and avoid tunneling.
        for _ in range(24):
            dt = 1.0 / 24.0
            v[:, 1] += gravity * dt
            p += v * dt

            # Dice-dice impact. The initial horizontal offset naturally splits
            # the boosted vertical impulse into left/right motion.
            d = p[1] - p[0]
            dist = float(np.linalg.norm(d))
            contact = float(radii.sum())
            if 1e-5 < dist < contact:
                normal = d / dist
                closing = float(np.dot(v[1] - v[0], normal))
                if closing < 0:
                    impulse = -(1.0 + restitution) * closing / 2.0
                    v[0] -= impulse * normal
                    v[1] += impulse * normal
                correction = (contact - dist) / 2.0 + 0.01
                p[0] -= correction * normal
                p[1] += correction * normal

            # Felt surface and the raised left/right table walls.
            for i in range(2):
                if p[i, 1] > floor[i]:
                    p[i, 1] = floor[i]
                    if v[i, 1] > 2.2:
                        v[i, 1] *= -0.42
                    else:
                        v[i, 1] = 0.0
                    v[i, 0] *= 0.84
                lo, hi = 66.0 + radii[i], 198.0 - radii[i]
                if p[i, 0] < lo:
                    p[i, 0] = lo
                    v[i, 0] = abs(v[i, 0]) * 0.45
                elif p[i, 0] > hi:
                    p[i, 0] = hi
                    v[i, 0] = -abs(v[i, 0]) * 0.45
        for i in range(2):
            if p[i, 1] >= floor[i] - 0.01:
                v[i, 0] *= 0.82
                if v[i, 1] == 0 and abs(v[i, 0]) < 0.08:
                    v[i, 0] = 0.0
    return np.asarray(history_p), np.asarray(history_v), floor


def add_shadow(frame, x, ground_y, height):
    # Source shadows are small cool-gray ellipses that sharpen near contact.
    layer = Image.new('RGBA', (W, H), (0, 0, 0, 0))
    spread = 3.0 + min(height, 50.0) * 0.035
    alpha = int(45 * math.exp(-height / 55.0) + 12)
    d = ImageDraw.Draw(layer)
    d.ellipse((x - spread, ground_y + 3 - 1.4,
               x + spread, ground_y + 3 + 1.4), fill=(38, 54, 49, alpha))
    layer = layer.filter(ImageFilter.GaussianBlur(1.35 + min(height, 50) * .012))
    frame.alpha_composite(layer)


def paste_center(canvas, sprite, x, y, opacity=1.0):
    if opacity < 0.999:
        sprite = sprite.copy()
        a = np.asarray(sprite.getchannel('A'), dtype=np.float32)
        sprite.putalpha(Image.fromarray(np.rint(a * opacity).astype(np.uint8)))
    canvas.alpha_composite(sprite, (int(round(x - sprite.width/2)),
                                    int(round(y - sprite.height/2))))


def render(src):
    bg = clean_background(src)
    whites, reds = make_sprite_sets(src, bg)
    positions, velocities, floors = simulate()
    output = []
    for t in range(N):
        canvas = Image.fromarray(bg.copy(), 'RGB').convert('RGBA')
        # Shadows remain on the felt while the cubes move vertically above them.
        for i in range(2):
            x, y = positions[t, i]
            add_shadow(canvas, x, floors[i], max(0.0, floors[i] - y))

        # A subtle exposure trail only during very fast motion communicates the
        # intervention without obscuring the native pixel-art silhouettes.
        if t > 0:
            for i, sprites in ((0, whites), (1, reds)):
                speed = float(np.linalg.norm(velocities[t, i]))
                if speed > 14:
                    prev = positions[t-1, i]
                    cur = positions[t, i]
                    sprite = sprites[(t * 3 + i) % len(sprites)]
                    paste_center(canvas, sprite,
                                 prev[0] * .65 + cur[0] * .35,
                                 prev[1] * .65 + cur[1] * .35, .13)

        # Back-to-front compositing handles crossings correctly.
        order = sorted(range(2), key=lambda i: positions[t, i, 1])
        for i in order:
            p = positions[t, i]
            moving = np.linalg.norm(velocities[t, i]) > .25
            sprites = whites if i == 0 else reds
            # Accelerated tumbling cycles native source orientations rapidly;
            # once settled, the face remains fixed.
            idx = ((t * (4 if i == 0 else 2)) % len(sprites)) if moving else (2 if i == 0 else 3)
            paste_center(canvas, sprites[idx], p[0], p[1])
        output.append(np.asarray(canvas.convert('RGB'), dtype=np.uint8))
    return np.stack(output)


def main():
    src = np.load(ROOT / 'source_pixels.npz')['rgb']
    assert src.shape == (N, H, W, 3) and src.dtype == np.uint8
    frames = render(src)
    assert frames.shape == (N, H, W, 3) and frames.dtype == np.uint8
    artifact_io.save(frames, ROOT)
    print('saved', frames.shape, frames.dtype, 'to', ROOT)


if __name__ == '__main__':
    main()
