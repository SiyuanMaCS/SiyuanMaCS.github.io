+"""Generate the requested counterfactual curling-stone video.
+
+The first-frame intervention inserts a blue stone one quarter of the way from
+the yellow stone to the red stone.  The stones are treated as equal-mass,
+slightly inelastic pucks constrained to the source line of travel.  Visuals
+are sampled from the source itself so that material, blur, and lighting match.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+N, H, W = 96, 144, 256
+
+
+def source_centers(frame_index):
+    """Approximate the two factual source trajectories (top-ellipse centers)."""
+    f = float(frame_index)
+    # Before their factual collision the stones have nearly constant speeds.
+    if f <= 39.0:
+        red_x = 47.0 + 1.985 * f
+        yellow_x = 214.0 - 2.085 * f
+    else:
+        # The low-restitution source collision leaves a slow outward drift.
+        red_x = 124.4 - 0.190 * (f - 39.0)
+        yellow_x = 132.7 + 0.155 * (f - 39.0)
+    red_y = 66.2 + (red_x - 47.0) * (5.45 / 167.0)
+    yellow_y = 76.5 + (yellow_x - 214.0) * (5.45 / 167.0)
+    return (red_x, red_y), (yellow_x, yellow_y)
+
+
+def erase_source_stones(frames):
+    """Remove both factual pucks while retaining the source scene per frame."""
+    cleaned = np.empty_like(frames)
+    yy, xx = np.mgrid[:H, :W]
+    for f, image in enumerate(frames):
+        mask = np.zeros((H, W), np.uint8)
+        for cx, cy in source_centers(f):
+            # Includes cap, cylindrical side, contact shadow, and soft fringe.
+            shape = (((xx - cx) / 10.8) ** 2 + ((yy - (cy + 3.5)) / 10.0) ** 2) <= 1.0
+            mask[shape] = 255
+        # TELEA restores both the narrow red guide line and house colors well
+        # from the immediately surrounding, rotationally smooth surface.
+        cleaned[f] = cv2.inpaint(image, mask, 3.0, cv2.INPAINT_TELEA)
+    return cleaned
+
+
+def extract_sprite(image, clean, center):
+    """Extract a soft-alpha source puck against its locally repaired scene."""
+    cx, cy = center
+    x0, x1 = int(round(cx)) - 14, int(round(cx)) + 15
+    y0, y1 = int(round(cy)) - 11, int(round(cy)) + 17
+    rgb = image[y0:y1, x0:x1].astype(np.float32)
+    bg = clean[y0:y1, x0:x1].astype(np.float32)
+    difference = np.max(np.abs(rgb - bg), axis=2)
+
+    # Difference isolates the puck without baking the guide line/background
+    # into the sprite.  Geometry suppresses occasional distant inpaint noise.
+    sy, sx = np.mgrid[: rgb.shape[0], : rgb.shape[1]]
+    geom = (((sx - 14.0) / 11.0) ** 2 + ((sy - 14.5) / 11.5) ** 2) <= 1.0
+    alpha = np.clip((difference - 1.0) / 15.0, 0.0, 1.0) * geom
+    alpha = cv2.GaussianBlur(alpha.astype(np.float32), (3, 3), 0.45)
+    alpha[alpha < 0.025] = 0.0
+    return rgb, alpha
+
+
+def recolor_cap(sprite, hue):
+    """Hue-shift the red top only, retaining source highlights and shading."""
+    out = np.clip(sprite, 0, 255).astype(np.uint8)
+    hsv = cv2.cvtColor(out, cv2.COLOR_RGB2HSV)
+    sy, sx = np.mgrid[: out.shape[0], : out.shape[1]]
+    # Red cap lies above the grey body; include its antialiased colored rim.
+    cap = (sy <= 13) & (sx >= 6) & (sx <= 22) & (hsv[..., 1] > 35)
+    hsv[..., 0][cap] = hue
+    hsv[..., 1][cap] = np.maximum(hsv[..., 1][cap], 145)
+    return cv2.cvtColor(hsv, cv2.COLOR_HSV2RGB).astype(np.float32)
+
+
+def simulate():
+    """Damped 1-D rigid-puck dynamics, sampled before each video step."""
+    # Coordinate is horizontal image x.  The initial blue location is exactly
+    # yellow + 1/4 * (red-yellow), as requested.
+    pos = np.array([47.0, 214.0, 214.0 + 0.25 * (47.0 - 214.0)], np.float64)
+    vel = np.array([1.985, -2.085, 0.0], np.float64)
+    history = np.empty((N, 3), np.float64)
+    diameter = 14.2
+    restitution = 0.15
+    substeps = 32
+
+    for frame in range(N):
+        history[frame] = pos
+        for _ in range(substeps):
+            pos += vel / substeps
+            order = np.argsort(pos)
+            for left, right in zip(order[:-1], order[1:]):
+                gap = pos[right] - pos[left]
+                if gap < diameter:
+                    correction = 0.5 * (diameter - gap)
+                    pos[left] -= correction
+                    pos[right] += correction
+                    if vel[right] < vel[left]:
+                        vl, vr = vel[left], vel[right]
+                        vel[left] = 0.5 * ((1.0 - restitution) * vl + (1.0 + restitution) * vr)
+                        vel[right] = 0.5 * ((1.0 + restitution) * vl + (1.0 - restitution) * vr)
+        # Very light ice drag, consistent with the almost uniform source glide.
+        vel *= 0.998
+    return history
+
+
+def paste_sprite(canvas, sprite, alpha, center):
+    """Subpixel translate and alpha composite one source sprite."""
+    cx, cy = center
+    h, w = alpha.shape
+    x0 = int(np.floor(cx)) - 14
+    y0 = int(np.floor(cy)) - 11
+    frac_x, frac_y = cx - np.floor(cx), cy - np.floor(cy)
+    matrix = np.float32([[1, 0, frac_x], [0, 1, frac_y]])
+    shifted_rgb = cv2.warpAffine(sprite, matrix, (w, h), flags=cv2.INTER_LINEAR,
+                                 borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+    shifted_a = cv2.warpAffine(alpha, matrix, (w, h), flags=cv2.INTER_LINEAR,
+                               borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+
+    xa, ya = max(0, x0), max(0, y0)
+    xb, yb = min(W, x0 + w), min(H, y0 + h)
+    if xa >= xb or ya >= yb:
+        return
+    sx0, sy0 = xa - x0, ya - y0
+    sx1, sy1 = sx0 + (xb - xa), sy0 + (yb - ya)
+    a = shifted_a[sy0:sy1, sx0:sx1, None]
+    dst = canvas[ya:yb, xa:xb].astype(np.float32)
+    src = shifted_rgb[sy0:sy1, sx0:sx1]
+    canvas[ya:yb, xa:xb] = np.clip(src * a + dst * (1.0 - a), 0, 255).astype(np.uint8)
+
+
+def main():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    clean = erase_source_stones(source)
+
+    red_rgb, red_alpha = extract_sprite(source[0], clean[0], source_centers(0)[0])
+    yellow_rgb, yellow_alpha = extract_sprite(source[0], clean[0], source_centers(0)[1])
+    blue_rgb = recolor_cap(red_rgb, 112)  # OpenCV hue: saturated royal blue
+
+    x_positions = simulate()
+    output = np.empty_like(source)
+    # Source travel line slopes gently down toward camera-right.
+    line_y = lambda x: 66.2 + (x - 47.0) * (10.3 / 167.0)
+
+    for f in range(N):
+        frame = clean[f].copy()
+        items = [
+            (x_positions[f, 0], line_y(x_positions[f, 0]), red_rgb, red_alpha),
+            (x_positions[f, 1], line_y(x_positions[f, 1]), yellow_rgb, yellow_alpha),
+            (x_positions[f, 2], line_y(x_positions[f, 2]), blue_rgb, red_alpha),
+        ]
+        # Back-to-front ordering is important during contact/overlap.
+        for x, y, rgb, alpha in sorted(items, key=lambda item: item[1]):
+            paste_sprite(frame, rgb, alpha, (x, y))
+        output[f] = frame
+
+    artifact_io.save(output.astype(np.uint8), CASE_DIR)
+
+
+if __name__ == "__main__":
+    main()
