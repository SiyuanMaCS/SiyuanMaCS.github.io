+"""Generate the 10x-target-mass counterfactual billiards video.
+
+The black ball is the target/object ball.  The original clip is a nearly
+head-on elastic collision between equal masses: the white ball stops and the
+black ball takes its velocity.  At a 1:10 cue/target mass ratio, conservation
+of momentum and kinetic energy instead gives -9/11 of the incoming velocity
+to the cue ball and 2/11 to the target ball.
+
+Only the supplied RGB frames are used.  The static plate and small ball
+sprites are recovered from frame zero, and the alternate paths are rendered
+with the same source shading and projective scale.
+"""
+
+from pathlib import Path
+
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def harmonic_inpaint(image, mask, iterations=360):
+    """Fill tiny masked regions by discrete harmonic interpolation."""
+    work = image.astype(np.float32).copy()
+    # Initialize from the mean of a narrow ring around the holes.
+    ring = np.zeros_like(mask)
+    for dy in (-1, 0, 1):
+        for dx in (-1, 0, 1):
+            ring |= np.roll(np.roll(mask, dy, axis=0), dx, axis=1)
+    ring &= ~mask
+    seed = work[ring].mean(axis=0)
+    work[mask] = seed
+    for _ in range(iterations):
+        avg = (
+            np.roll(work, 1, 0) + np.roll(work, -1, 0)
+            + np.roll(work, 1, 1) + np.roll(work, -1, 1)
+            + np.roll(np.roll(work, 1, 0), 1, 1)
+            + np.roll(np.roll(work, 1, 0), -1, 1)
+            + np.roll(np.roll(work, -1, 0), 1, 1)
+            + np.roll(np.roll(work, -1, 0), -1, 1)
+        ) / 8.0
+        work[mask] = avg[mask]
+    return np.clip(work, 0, 255).astype(np.uint8)
+
+
+def homography(src, dst):
+    """Return the projective transform mapping four src points to dst."""
+    rows, rhs = [], []
+    for (x, y), (u, v) in zip(src, dst):
+        rows.extend([
+            [x, y, 1, 0, 0, 0, -u * x, -u * y],
+            [0, 0, 0, x, y, 1, -v * x, -v * y],
+        ])
+        rhs.extend([u, v])
+    h = np.r_[np.linalg.solve(np.asarray(rows), np.asarray(rhs)), 1.0]
+    return h.reshape(3, 3)
+
+
+def project(point, matrix):
+    q = matrix @ np.asarray([point[0], point[1], 1.0])
+    return q[:2] / q[2]
+
+
+def local_scale(point, world_to_screen):
+    """Perspective area scale at a world-space point."""
+    p = np.asarray(point, dtype=np.float64)
+    eps = 1.0e-3
+    c = project(p, world_to_screen)
+    j = np.column_stack((
+        (project(p + [eps, 0], world_to_screen) - c) / eps,
+        (project(p + [0, eps], world_to_screen) - c) / eps,
+    ))
+    return np.sqrt(abs(np.linalg.det(j)))
+
+
+def extract_sprite(source, background, center, half=9):
+    """Represent the observed sprite as alpha plus color offset from felt."""
+    cx, cy = center
+    x0, y0 = int(round(cx)) - half, int(round(cy)) - half
+    src = source[y0:y0 + 2 * half + 1, x0:x0 + 2 * half + 1].astype(np.float32)
+    bg = background[y0:y0 + 2 * half + 1, x0:x0 + 2 * half + 1].astype(np.float32)
+    delta = src - bg
+    strength = np.max(np.abs(delta), axis=2)
+    # Core pixels are opaque; soft antialiasing and the contact shadow retain
+    # fractional alpha. Tiny source quantization differences are discarded.
+    alpha = np.clip((strength - 0.7) / 18.0, 0.0, 1.0)
+    alpha[strength < 1.5] = 0.0
+    return {"delta": delta, "alpha": alpha, "half": half}
+
+
+def resize_float(array, size):
+    """Bicubic resize for a 2-D float array using Pillow's F mode."""
+    im = Image.fromarray(array.astype(np.float32), mode="F")
+    return np.asarray(im.resize(size, Image.Resampling.BICUBIC), dtype=np.float32)
+
+
+def composite_sprite(canvas, static_plate, sprite, center, scale):
+    """Composite a projectively scaled source sprite at a floating center."""
+    old_h, old_w = sprite["alpha"].shape
+    new_w = max(5, int(round(old_w * scale)))
+    new_h = max(5, int(round(old_h * scale)))
+    alpha = np.clip(resize_float(sprite["alpha"], (new_w, new_h)), 0.0, 1.0)
+    delta = np.stack(
+        [resize_float(sprite["delta"][:, :, c], (new_w, new_h)) for c in range(3)],
+        axis=2,
+    )
+
+    # Recover the foreground color relative to the local felt. This makes the
+    # second ball correctly occlude the first during the separating frames.
+    rel_foreground = delta / np.maximum(alpha[:, :, None], 0.055)
+    rel_foreground = np.clip(rel_foreground, -255.0, 255.0)
+
+    x0 = int(round(center[0] - (new_w - 1) / 2.0))
+    y0 = int(round(center[1] - (new_h - 1) / 2.0))
+    x1, y1 = x0 + new_w, y0 + new_h
+    sx0, sy0 = max(0, -x0), max(0, -y0)
+    sx1, sy1 = new_w - max(0, x1 - canvas.shape[1]), new_h - max(0, y1 - canvas.shape[0])
+    x0c, y0c = max(0, x0), max(0, y0)
+    x1c, y1c = min(canvas.shape[1], x1), min(canvas.shape[0], y1)
+    if x0c >= x1c or y0c >= y1c:
+        return
+
+    a = alpha[sy0:sy1, sx0:sx1, None]
+    base = static_plate[y0c:y1c, x0c:x1c].astype(np.float32)
+    foreground = np.clip(base + rel_foreground[sy0:sy1, sx0:sx1], 0, 255)
+    current = canvas[y0c:y1c, x0c:x1c].astype(np.float32)
+    canvas[y0c:y1c, x0c:x1c] = np.clip(
+        current * (1.0 - a) + foreground * a, 0, 255
+    ).astype(np.uint8)
+
+
+def build_frames():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3) and source.dtype == np.uint8
+
+    # Recover a clean static plate from frame zero. These masks cover each
+    # sphere, its antialiased edge, and its compact contact shadow.
+    yy, xx = np.ogrid[:144, :256]
+    clean_mask = (((xx - 85.4) / 8.2) ** 2 + ((yy - 96.0) / 8.8) ** 2 <= 1.0)
+    clean_mask |= (((xx - 126.4) / 7.4) ** 2 + ((yy - 71.2) / 7.8) ** 2 <= 1.0)
+    plate = harmonic_inpaint(source[0], clean_mask)
+
+    white_sprite = extract_sprite(source[0], plate, (85.4, 95.4))
+    black_sprite = extract_sprite(source[0], plate, (126.4, 71.0))
+
+    # Table corners in image and normalized physical table coordinates.  This
+    # lets ball size and paths follow the source perspective rather than move
+    # in screen-space straight lines.
+    screen_corners = np.asarray([[138, 39], [205, 47], [111, 148], [-3, 106]], float)
+    world_corners = np.asarray([[0, 0], [2, 0], [2, 1], [0, 1]], float)
+    screen_to_world = homography(screen_corners, world_corners)
+    world_to_screen = np.linalg.inv(screen_to_world)
+
+    white_ref_world = project((85.4, 95.4), screen_to_world)
+    black_ref_world = project((126.4, 71.0), screen_to_world)
+    white_ref_scale = local_scale(white_ref_world, world_to_screen)
+    black_ref_scale = local_scale(black_ref_world, world_to_screen)
+
+    # Measured undisturbed incoming cue-ball velocity, in normalized table
+    # coordinates per source frame. Collision is at frame 13.
+    incoming = np.asarray([0.00415, -0.01860], dtype=np.float64)
+    target_at_contact = black_ref_world.copy()
+    direction = incoming / np.linalg.norm(incoming)
+    # Physical center separation at contact, calibrated from the source.
+    cue_at_contact = target_at_contact - direction * 0.0225
+
+    cue_pos = cue_at_contact.copy()
+    target_pos = target_at_contact.copy()
+    cue_vel = (-9.0 / 11.0) * incoming
+    target_vel = (2.0 / 11.0) * incoming
+
+    frames = source.copy()
+    drag = 0.990
+    cushion_restitution = 0.90
+    # Center limits leave each sphere against, not inside, the cushion.
+    x_min, x_max = 0.055, 1.945
+    y_min, y_max = 0.057, 0.943
+
+    for frame_index in range(14, 96):
+        cue_pos += cue_vel
+        target_pos += target_vel
+
+        for pos, vel in ((cue_pos, cue_vel), (target_pos, target_vel)):
+            if pos[0] < x_min:
+                pos[0] = 2 * x_min - pos[0]
+                vel[0] = abs(vel[0]) * cushion_restitution
+            elif pos[0] > x_max:
+                pos[0] = 2 * x_max - pos[0]
+                vel[0] = -abs(vel[0]) * cushion_restitution
+            if pos[1] < y_min:
+                pos[1] = 2 * y_min - pos[1]
+                vel[1] = abs(vel[1]) * cushion_restitution
+            elif pos[1] > y_max:
+                pos[1] = 2 * y_max - pos[1]
+                vel[1] = -abs(vel[1]) * cushion_restitution
+
+        cue_vel *= drag
+        target_vel *= drag
+
+        canvas = plate.copy()
+        cue_screen = project(cue_pos, world_to_screen)
+        target_screen = project(target_pos, world_to_screen)
+        cue_scale = local_scale(cue_pos, world_to_screen) / white_ref_scale
+        target_scale = local_scale(target_pos, world_to_screen) / black_ref_scale
+
+        # Farther object first; the nearer sphere then has correct overlap.
+        objects = [
+            (target_pos[1], black_sprite, target_screen, target_scale),
+            (cue_pos[1], white_sprite, cue_screen, cue_scale),
+        ]
+        for _, sprite, center, scale in sorted(objects, key=lambda item: item[0]):
+            composite_sprite(canvas, plate, sprite, center, scale)
+        frames[frame_index] = canvas
+
+    return frames
+
+
+if __name__ == "__main__":
+    output = build_frames()
+    artifact_io.save(output, CASE_DIR)
+    print(f"saved {output.shape[0]} frames to {CASE_DIR}")
