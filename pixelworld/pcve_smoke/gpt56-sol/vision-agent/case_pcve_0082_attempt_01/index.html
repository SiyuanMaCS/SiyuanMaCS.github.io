+"""Remove the apple and undo its collision with the soccer ball.
+
+The shot is locked off.  A clean static reference is assembled from the first
+source frame (which contains the ball in its pre-impact pose) and a late frame
+(which reveals the grass hidden by the apple's initial cast shadow).  Only
+small, feathered regions covering the apple, its shadow, and the displaced ball
+are replaced; all unrelated source pixels retain their original temporal
+variation.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def ellipse_alpha(height, width, cx, cy, rx, ry, feather=1.35):
+    """Return a softly feathered elliptical alpha matte."""
+    yy, xx = np.ogrid[:height, :width]
+    # A slightly oversized hard core followed by a short Gaussian feather
+    # prevents both object remnants and visible cut-out boundaries.
+    mask = ((((xx - cx) / rx) ** 2 + ((yy - cy) / ry) ** 2) <= 1.0).astype(
+        np.float32
+    )
+    if feather:
+        mask = cv2.GaussianBlur(mask, (0, 0), feather)
+        mask = np.clip(mask * 1.12, 0.0, 1.0)
+    return mask
+
+
+def composite(dst, reference, alpha):
+    a = alpha[..., None]
+    return dst * (1.0 - a) + reference * a
+
+
+def ball_center(frame):
+    """Locate the soccer ball from its low-saturation white panels."""
+    hsv = cv2.cvtColor(frame, cv2.COLOR_RGB2HSV)
+    mask = (
+        (hsv[:, :, 2] > 130)
+        & (hsv[:, :, 1] < 100)
+        & (np.indices(hsv.shape[:2])[0] >= 65)
+        & (np.indices(hsv.shape[:2])[0] < 106)
+        & (np.indices(hsv.shape[:2])[1] >= 50)
+        & (np.indices(hsv.shape[:2])[1] < 155)
+    ).astype(np.uint8)
+    count, labels, stats, centers = cv2.connectedComponentsWithStats(mask)
+    candidates = []
+    for label in range(1, count):
+        area = int(stats[label, cv2.CC_STAT_AREA])
+        w = int(stats[label, cv2.CC_STAT_WIDTH])
+        h = int(stats[label, cv2.CC_STAT_HEIGHT])
+        if area >= 100 and 14 <= w <= 27 and 14 <= h <= 27:
+            candidates.append((area, centers[label]))
+    if not candidates:
+        raise RuntimeError("Could not locate soccer ball")
+    return max(candidates, key=lambda item: item[0])[1]
+
+
+def build_frames():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3) and source.dtype == np.uint8
+    n, h, w, _ = source.shape
+
+    # First frame supplies the untouched scene and pre-impact ball.  Frame 61
+    # supplies clean grass at the apple shadow's frame-1 location.
+    reference = source[0].astype(np.float32).copy()
+    clean_late = source[60].astype(np.float32)
+    initial_shadow = ellipse_alpha(h, w, 184.0, 31.5, 11.0, 7.0, 1.5)
+    reference = composite(reference, clean_late, initial_shadow)
+
+    initial_cx, initial_cy = ball_center(source[0])
+
+    # Measured image-space path of the apple's cast shadow before impact.
+    shadow_centers = np.array(
+        [
+            [187.5, 30.5], [187.0, 30.8], [186.3, 31.6], [185.4, 32.5],
+            [184.5, 33.5], [183.0, 35.5], [180.5, 37.5], [178.0, 40.5],
+            [173.6, 45.1], [170.0, 50.0], [164.9, 55.9], [158.4, 63.2],
+            [150.4, 72.0], [146.6, 80.1],
+        ],
+        dtype=np.float32,
+    )
+
+    # Once the apple hits the ground/ball, it rolls rapidly to screen-right.
+    # The measured centers through frame 21 anchor a smooth continuation.
+    rolling_x = np.array(
+        [147.7, 153.8, 160.4, 166.7, 173.1, 179.0, 185.3,
+         190.4, 195.6, 200.3, 205.3, 210.4, 215.2, 220.0,
+         223.8, 227.9, 231.5, 235.0, 238.9, 242.6, 245.5,
+         248.5, 251.3, 253.8, 256.5],
+        dtype=np.float32,
+    )
+    rolling_y = np.array(
+        [86.1, 87.1, 86.9, 86.7, 86.3, 86.3, 86.4,
+         86.6, 86.3, 86.4, 86.1, 85.9, 85.7, 85.9,
+         86.0, 86.3, 86.1, 85.9, 85.2, 85.6, 85.9,
+         85.9, 85.9, 86.0, 86.0],
+        dtype=np.float32,
+    )
+
+    output = np.empty_like(source)
+    for index in range(n):
+        frame = source[index].astype(np.float32)
+        alpha = np.zeros((h, w), dtype=np.float32)
+
+        # Remove the moving shadow from the opening through contact.
+        if index < len(shadow_centers):
+            sx, sy = shadow_centers[index]
+            alpha = np.maximum(
+                alpha, ellipse_alpha(h, w, sx, sy, 8.5, 5.6, 1.15)
+            )
+
+        # The apple itself enters at the top of frame 10 and drops vertically.
+        if 9 <= index <= 13:
+            ax = 136.0 - 0.25 * (index - 9)
+            ay = 13.3 + 16.95 * (index - 9)
+            alpha = np.maximum(
+                alpha, ellipse_alpha(h, w, ax, ay, 7.0, 7.0, 1.15)
+            )
+
+        # Remove the rebounding/rolling apple after contact.
+        rolling_index = index - 14
+        if 0 <= rolling_index < len(rolling_x):
+            alpha = np.maximum(
+                alpha,
+                ellipse_alpha(
+                    h, w, rolling_x[rolling_index], rolling_y[rolling_index],
+                    8.0, 7.0, 1.2,
+                ),
+            )
+

+        # then restore the ball at the exact pre-impact location and rotation.
+        current_cx, current_cy = ball_center(source[index])
+        alpha = np.maximum(
+            alpha,
+            ellipse_alpha(h, w, current_cx + 4.5, current_cy + 1.5,
+                          22.5, 16.0, 1.45),
+        )
+        alpha = np.maximum(
+            alpha,
+            ellipse_alpha(h, w, initial_cx + 4.5, initial_cy + 1.5,
+                          22.5, 16.0, 1.45),
+        )
+
+        edited = composite(frame, reference, alpha)
+        output[index] = np.clip(np.rint(edited), 0, 255).astype(np.uint8)
+
+    return output
+

+if __name__ == "__main__":
+    frames = build_frames()
+    artifact_io.save(frames, CASE_DIR)
+    print(f"saved {len(frames)} edited frames to {CASE_DIR}")
