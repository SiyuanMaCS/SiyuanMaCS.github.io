+def line_y(x):
+    """Screen-space center line followed by both stones."""
+    return 63.72 + 0.0549 * x
+

+def soft_ellipse(shape, cx, cy, rx, ry, blur=1.3):
+    h, w = shape
+    yy, xx = np.mgrid[:h, :w]
+    inside = (((xx-cx)/rx)**2 + ((yy-cy)/ry)**2 <= 1).astype(np.float32)
+    return cv2.GaussianBlur(inside, (0, 0), blur)
+
+
+def factual_centers(t):
+    # Fits to the source trajectories; these are only used to remove the
+    # original stones before drawing the counterfactual motion.
+    if t <= 37.0:
+        xr = 46.95 + 1.986*t
+        xy = 213.90 - 2.116*t
+    else:
+        xr = 122.9 - 0.125*(t-38.0)
+        xy = 132.5 + 0.160*(t-38.0)
+    return xr, xy
+
+
+def extract_sprite(frame, plate, anchor_x, anchor_y, half_w=13, top=10, bottom=16):
+    """Extract one stone and its soft contact shadow as an RGBA sprite."""
+    x = int(round(anchor_x)); y = int(round(anchor_y))
+    src = frame[y-top:y+bottom+1, x-half_w:x+half_w+1].astype(np.float32)
+    bg = plate[y-top:y+bottom+1, x-half_w:x+half_w+1].astype(np.float32)
+    ph, pw = src.shape[:2]
+    yy, xx = np.mgrid[:ph, :pw]
+    dx = xx-half_w
+    dy = yy-top
+
+    # Pixel difference gives excellent antialiasing around the rendered mesh.
+    diff = np.max(np.abs(src-bg), axis=2)
+    alpha = np.clip((diff-2.0)/18.0, 0.0, 1.0)
+    # Exclude unrelated background while retaining the low-opacity ground shadow.
+    support = np.maximum(
+        (((dx/10.0)**2 + ((dy-2.5)/10.0)**2) <= 1).astype(np.float32),
+        0.62*(((dx-1.5)/11.0)**2 + ((dy-9.0)/6.0)**2 <= 1).astype(np.float32),
+    )
+    alpha *= support
+    alpha = cv2.GaussianBlur(alpha, (0, 0), 0.45)
+    alpha[alpha < 0.025] = 0
+    return src, alpha
+
+
+def composite_sprite(canvas, sprite, alpha, cx, cy, top=10):
+    """Subpixel translate and alpha-composite a sprite at a cap-center anchor."""
+    rgb = sprite
+    h, w = rgb.shape[:2]
+    # Integer placement plus fractional warp avoids positional judder.
+    x0f = cx - (w//2)
+    y0f = cy - top
+    x0 = int(np.floor(x0f)); y0 = int(np.floor(y0f))
+    fx = x0f-x0; fy = y0f-y0
+    M = np.float32([[1, 0, fx], [0, 1, fy]])
+    wrgb = cv2.warpAffine(rgb, M, (w, h), flags=cv2.INTER_LINEAR,
+                          borderMode=cv2.BORDER_REPLICATE)
+    wa = cv2.warpAffine(alpha, M, (w, h), flags=cv2.INTER_LINEAR,
+                         borderMode=cv2.BORDER_CONSTANT)
+    xa=max(0,x0); ya=max(0,y0); xb=min(canvas.shape[1],x0+w); yb=min(canvas.shape[0],y0+h)
+    if xa>=xb or ya>=yb:
+        return
+    sx0=xa-x0; sy0=ya-y0; sx1=sx0+(xb-xa); sy1=sy0+(yb-ya)
+    a=wa[sy0:sy1,sx0:sx1,None]
+    canvas[ya:yb,xa:xb] = wrgb[sy0:sy1,sx0:sx1]*a + canvas[ya:yb,xa:xb]*(1-a)
+

+def generate():
+    source = np.load(CASE_DIR/'source_pixels.npz')['rgb']
+    n, h, w, _ = source.shape
+
+    # A robust clean plate. During these first 30 frames each moving stone only
+    # occupies any one pixel briefly, so the median preserves the original line,
+    # rings, lighting, and antialiasing without either stone.
+    plate = np.median(source[:30].astype(np.float32), axis=0).astype(np.uint8)
+
+    red_rgb, red_a = extract_sprite(source[0], plate, 47, line_y(47))
+    yel_rgb, yel_a = extract_sprite(source[0], plate, 214, line_y(214))
+
+    # Measured incoming source velocities (px/frame), with only red doubled.
+    ur = 2.0 * 1.986
+    uy = -2.116
+    x0r, x0y = 46.95, 213.90
+    contact_sep = 15.5
+    hit_t = (x0y-x0r-contact_sep)/(ur-uy)
+    hit_r = x0r + ur*hit_t
+    hit_y = x0y + uy*hit_t
+
+    # The source collision is nearly inelastic (small rebound). Applying its
+    # measured restitution to equal stones conserves horizontal momentum.
+    restitution = 0.070
+    vr = 0.5*((ur+uy)-restitution*(ur-uy))
+    vy = 0.5*((ur+uy)+restitution*(ur-uy))
+
+    out = np.empty_like(source)
+    for i in range(n):
+        if i == 0:
+            # Velocity intervention starts here, but position/appearance at t=0
+            # is unchanged; retain the exact factual first frame.
+            out[i] = source[i]
+            continue
+
+        canvas = source[i].astype(np.float32)
+
+        # Remove both stones at their factual locations. A broad feather also
+        # removes their contact shadows while preserving the source's ambient
+        # illumination everywhere else.
+        frx, fyx = factual_centers(float(i))
+        remove = soft_ellipse((h,w), frx, line_y(frx)+4.0, 12.0, 15.0)
+        remove = np.maximum(remove, soft_ellipse((h,w), fyx, line_y(fyx)+4.0, 12.0, 15.0))
+        remove = np.clip(remove*1.08, 0, 1)[...,None]
+        canvas = canvas*(1-remove) + plate.astype(np.float32)*remove
+
+        t = float(i)
+        if t <= hit_t:
+            xr = x0r + ur*t
+            xy = x0y + uy*t
+        else:
+            dt = t-hit_t
+            xr = hit_r + vr*dt
+            xy = hit_y + vy*dt
+
+        # Red is behind after impact; drawing it first gives correct occlusion at contact.
+        composite_sprite(canvas, red_rgb, red_a, xr, line_y(xr))
+        composite_sprite(canvas, yel_rgb, yel_a, xy, line_y(xy))
+        out[i] = np.clip(np.rint(canvas), 0, 255).astype(np.uint8)
+
+    return out
+

+    frames = generate()
+    artifact_io.save(frames, CASE_DIR)

        return
    sx0=xa-x0; sy0=ya-y0; sx1=sx0+(xb-xa); sy1=sy0+(yb-ya)
    a=wa[sy0:sy1,sx0:sx1,None]
    canvas[ya:yb,xa:xb] = wrgb[sy0:sy1,sx0:sx1]*a + canvas[ya:yb,xa:xb]*(1-a)


def generate():
    source = np.load(CASE_DIR/'source_pixels.npz')['rgb']
    n, h, w, _ = source.shape

    # A robust clean plate. During these first 30 frames each moving stone only
    # occupies any one pixel briefly, so the median preserves the original line,
    # rings, lighting, and antialiasing without either stone.
    plate = np.median(source[:30].astype(np.float32), axis=0).astype(np.uint8)

    red_rgb, red_a = extract_sprite(source[0], plate, 47, line_y(47))
    yel_rgb, yel_a = extract_sprite(source[0], plate, 214, line_y(214))

    # Measured incoming source velocities (px/frame), with only red doubled.
    ur = 2.0 * 1.986
    uy = -2.116
    x0r, x0y = 46.95, 213.90
    contact_sep = 15.5
    hit_t = (x0y-x0r-contact_sep)/(ur-uy)
    hit_r = x0r + ur*hit_t
    hit_y = x0y + uy*hit_t

    # The source collision is nearly inelastic (small rebound). Applying its
    # measured restitution to equal stones conserves horizontal momentum.
    restitution = 0.070
    vr = 0.5*((ur+uy)-restitution*(ur-uy))
    vy = 0.5*((ur+uy)+restitution*(ur-uy))

    out = np.empty_like(source)
    for i in range(n):
        if i == 0:
            # Velocity intervention starts here, but position/appearance at t=0
            # is unchanged; retain the exact factual first frame.
            out[i] = source[i]
            continue

        canvas = source[i].astype(np.float32)

        # Remove both stones at their factual locations. A broad feather also
        # removes their contact shadows while preserving the source's ambient
        # illumination everywhere else.
        frx, fyx = factual_centers(float(i))
        remove = soft_ellipse((h,w), frx, line_y(frx)+4.0, 12.0, 15.0)
        remove = np.maximum(remove, soft_ellipse((h,w), fyx, line_y(fyx)+4.0, 12.0, 15.0))
        remove = np.clip(remove*1.08, 0, 1)[...,None]
        canvas = canvas*(1-remove) + plate.astype(np.float32)*remove

        t = float(i)
        if t <= hit_t:
            xr = x0r + ur*t
            xy = x0y + uy*t
        else:
            dt = t-hit_t
            xr = hit_r + vr*dt
            xy = hit_y + vy*dt

        # Red is behind after impact; drawing it first gives correct occlusion at contact.
        composite_sprite(canvas, red_rgb, red_a, xr, line_y(xr))
        composite_sprite(canvas, yel_rgb, yel_a, xy, line_y(xy))
        out[i] = np.clip(np.rint(canvas), 0, 255).astype(np.uint8)

    return out


if __name__ == '__main__':
    frames = generate()
    artifact_io.save(frames, CASE_DIR)
