+"""Generate the counterfactual video for this case.
+
+The source car is tracked during its unobstructed climb.  Its horizontal
+motion is well fit by x(t) = x0 + v0*t + c*t^2, where c is the component of
+gravity along the ramp.  The edit retains c but changes v0 to 0.6*v0.  This
+makes the car reverse before the top, return down the ramp, and continue off
+the right edge of the road.  Car pixels are taken only from source_pixels.npz;
+the clean plate is the temporal median of the pre-landing source frames.
+"""
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+N, H, W = 96, 144, 256
+
+
+def clean_plate(source):
+    # In frames 0..18 the car occupies a different small region in every
+    # image, so a temporal median removes it while retaining the exact scene.
+    return np.median(source[:19], axis=0).astype(np.uint8)
+
+
+def soft_object_mask(frame, plate, seed_xy, radius=34):
+    """Extract the changed connected component around a known car center."""
+    delta = np.max(np.abs(frame.astype(np.int16) - plate.astype(np.int16)), axis=2)
+    changed = (delta > 6).astype(np.uint8)
+    changed = cv2.morphologyEx(changed, cv2.MORPH_CLOSE,
+                               np.ones((3, 3), np.uint8))
+    count, labels, stats, _ = cv2.connectedComponentsWithStats(changed, 8)
+    sx, sy = int(round(seed_xy[0])), int(round(seed_xy[1]))
+    candidates = []
+    for k in range(1, count):
+        x, y, w, h, area = stats[k]
+        if x - radius <= sx <= x + w + radius and y - radius <= sy <= y + h + radius:
+            cx = x + w / 2
+            cy = y + h / 2
+            candidates.append(((cx-sx)**2 + (cy-sy)**2, -area, k))
+    if not candidates:
+        raise RuntimeError("car component was not found")
+    label = min(candidates)[2]
+    core = (labels == label).astype(np.uint8)
+    # Include antialiased boundary pixels, but keep a compact car-shaped matte.
+    support = cv2.dilate(core, np.ones((3, 3), np.uint8), iterations=1)
+    alpha = np.clip((delta.astype(np.float32) - 1.5) / 10.0, 0.0, 1.0)
+    alpha *= support
+    alpha = cv2.GaussianBlur(alpha, (3, 3), 0.55)
+    alpha *= cv2.dilate(support, np.ones((3, 3), np.uint8), iterations=1)
+    return np.clip(alpha, 0.0, 1.0)
+
+
+def make_sprite(frame, plate, center, margin=5):
+    alpha = soft_object_mask(frame, plate, center)
+    yy, xx = np.where(alpha > 0.01)
+    x0 = max(0, int(xx.min()) - margin)
+    x1 = min(W, int(xx.max()) + margin + 1)
+    y0 = max(0, int(yy.min()) - margin)
+    y1 = min(H, int(yy.max()) + margin + 1)
+    rgba = np.dstack((frame[y0:y1, x0:x1], alpha[y0:y1, x0:x1]))
+    anchor = (center[0] - x0, center[1] - y0)
+    return rgba.astype(np.float32), anchor
+
+
+def transformed_sprite(sprite, anchor, angle_deg=0.0):
+    """Rotate a premultiplied sprite about its measured car center."""
+    h, w = sprite.shape[:2]
+    pad = 18
+    canvas = np.zeros((h + 2*pad, w + 2*pad, 4), np.float32)
+    canvas[pad:pad+h, pad:pad+w] = sprite
+    ax, ay = anchor[0] + pad, anchor[1] + pad
+    # Warp premultiplied color to avoid dark fringes at alpha boundaries.
+    premul = canvas.copy()
+    premul[:, :, :3] *= premul[:, :, 3:4]
+    matrix = cv2.getRotationMatrix2D((ax, ay), angle_deg, 1.0)
+    warped = cv2.warpAffine(premul, matrix, (canvas.shape[1], canvas.shape[0]),
+                            flags=cv2.INTER_CUBIC,
+                            borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+    a = np.clip(warped[:, :, 3], 0.0, 1.0)
+    rgb = np.zeros_like(warped[:, :, :3])
+    valid = a > 1e-4
+    rgb[valid] = warped[:, :, :3][valid] / a[valid, None]
+    return np.dstack((np.clip(rgb, 0, 255), a)), (ax, ay)
+
+
+def paste_rgba(background, rgba, anchor, center):
+    """Subpixel composite of rgba such that anchor lands at center."""
+    h, w = rgba.shape[:2]
+    left = center[0] - anchor[0]
+    top = center[1] - anchor[1]
+    # A full-frame affine warp gives stable subpixel motion and clean clipping.
+    matrix = np.array([[1.0, 0.0, left], [0.0, 1.0, top]], np.float32)
+    premul = rgba.copy()
+    premul[:, :, :3] *= premul[:, :, 3:4]
+    warped = cv2.warpAffine(premul, matrix, (W, H), flags=cv2.INTER_CUBIC,
+                            borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+    alpha = np.clip(warped[:, :, 3:4], 0.0, 1.0)
+    color = np.zeros_like(warped[:, :, :3])
+    valid = alpha[:, :, 0] > 1e-4
+    color[valid] = warped[:, :, :3][valid] / alpha[valid]
+    out = background.astype(np.float32) * (1.0 - alpha) + color * alpha
+    return np.clip(np.rint(out), 0, 255).astype(np.uint8)
+
+
+def generate():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (N, H, W, 3) and source.dtype == np.uint8
+    plate = clean_plate(source)
+
+    # Centroids measured from changed components in source frames 0..11.
+    x_track = np.array([173.0950, 158.8527, 145.6091, 133.2157,
+                        121.4803, 110.6925, 100.9451, 92.0681,
+                        84.1268, 76.9032, 70.4248, 64.8228])
+    t_track = np.arange(len(x_track), dtype=np.float64)
+    c, v0, x0 = np.polyfit(t_track, x_track, 2)
+    v_new = 0.6 * v0
+
+    # Ramp centerline and the point where the wheels meet the level road.
+    ramp_slope = 0.36224
+    ramp_intercept = 46.8384
+    ramp_end_x = 196.0
+    ramp_end_y = ramp_slope * ramp_end_x + ramp_intercept
+
+    ramp_center0 = (x0, ramp_slope * x0 + ramp_intercept)
+    ramp_sprite, ramp_anchor = make_sprite(source[0], plate, ramp_center0)
+    # The settled car gives an unoccluded horizontal view plus its contact shadow.
+    ground_center = (27.78, 120.72)
+    ground_sprite, ground_anchor = make_sprite(source[30], plate, ground_center)
+
+    # Solve when the returning car reaches the end of the incline.
+    roots = np.roots([c, v_new, x0 - ramp_end_x])
+    t_end = float(max(r.real for r in roots if abs(r.imag) < 1e-7))
+    speed_end = v_new + 2.0*c*t_end
+
+    frames = np.repeat(plate[None], N, axis=0)
+    for frame_index in range(N):
+        t = float(frame_index)
+        if t <= t_end:
+            x = x0 + v_new*t + c*t*t
+            y = ramp_slope*x + ramp_intercept
+            # As the rear wheels reach the pavement, let the body settle naturally
+            # from the ramp's 20-degree pitch to horizontal.
+            settle = np.clip((x - (ramp_end_x - 13.0)) / 13.0, 0.0, 1.0)
+            if settle > 0:
+                angle = -20.0 * settle
+                sprite, anchor = transformed_sprite(ramp_sprite, ramp_anchor, angle)
+                y = (1.0-settle)*y + settle*120.72
+            else:
+                sprite, anchor = ramp_sprite, ramp_anchor
+            frames[frame_index] = paste_rgba(plate, sprite, anchor, (x, y))
+        else:
+            # Once level, preserve the speed gained down the ramp.  It is moving
+            # backwards (the model still faces left), so it exits camera-right.
+            x = ramp_end_x + speed_end*(t - t_end)
+            if x - ground_anchor[0] < W + 8:
+                frames[frame_index] = paste_rgba(
+                    plate, ground_sprite, ground_anchor, (x, 120.72))
+
+    # At the intervention instant only speed changes; pose is exactly factual.
+    frames[0] = source[0]
+    return frames
+
+
+if __name__ == "__main__":
+    result = generate()
+    artifact_io.save(result, CASE_DIR)
+    print("saved", result.shape, result.dtype)
+
