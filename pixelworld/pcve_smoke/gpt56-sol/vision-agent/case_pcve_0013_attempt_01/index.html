+"""Generate the 4 s counterfactual: the block has 5x floor friction from frame 1.
+
+The source gives us the exact photorealistic appearance.  Up to first contact the
+friction coefficient cannot affect either rigid body, so those frames are retained.
+At impact the high-friction block receives only a very short displacement and then
+sticks; the ball's post-impact rebound/rolling is taken from the source ball track,
+composited over the stationary block.  This preserves lighting, spin and perspective.
+"""
+from pathlib import Path
+import numpy as np
+import cv2
+from artifact_io import save
+
+CASE = Path(__file__).resolve().parent
+SRC = np.load(CASE / "source_pixels.npz")["rgb"]
+H, W = SRC.shape[1:3]
+
+
+def seamless_clone_patch(dst, src, mask, center):
+    """Poisson clone with alpha-feather fallback (RGB in/out)."""
+    x, y, w, h = cv2.boundingRect(mask)
+    if w == 0 or h == 0:
+        return dst
+    pad = 3
+    x0=max(0,x-pad); y0=max(0,y-pad); x1=min(W,x+w+pad); y1=min(H,y+h+pad)
+    patch=src[y0:y1,x0:x1]
+    pm=mask[y0:y1,x0:x1]
+    cx=int(center[0]-x0); cy=int(center[1]-y0)
+    # Clamp clone center so the translated patch stays in frame.
+    tx0=int(center[0]-(x1-x0)/2); ty0=int(center[1]-(y1-y0)/2)
+    if tx0<0 or ty0<0 or tx0+patch.shape[1]>W or ty0+patch.shape[0]>H:
+        return dst
+    return cv2.cvtColor(cv2.seamlessClone(cv2.cvtColor(patch,cv2.COLOR_RGB2BGR),
+        cv2.cvtColor(dst,cv2.COLOR_RGB2BGR),pm.astype(np.uint8),
+        (int(center[0]),int(center[1])),cv2.NORMAL_CLONE),cv2.COLOR_BGR2RGB)
+
+
+def block_mask(frame):
+    """Polygon + color segmentation for the original wooden crate and its shadow."""
+    yy,xx=np.indices((H,W)); a=frame
+    # Dark wood pixels, constrained to the known initial crate envelope.
+    seed=(xx>=106)&(xx<=172)&(yy>=53)&(yy<=123)&(a[:,:,0]<125)&(a[:,:,1]<88)&(a[:,:,2]<68)
+    m=cv2.dilate(seed.astype(np.uint8),np.ones((5,5),np.uint8),iterations=1)
+    # Fill the three projected faces while retaining an antialiased perimeter.
+    poly=np.array([[108,61],[145,55],[164,61],[164,96],[146,104],[109,97]],np.int32)
+    cv2.fillConvexPoly(m,poly,255)
+    # Contact shadow belongs with the block and should remain beneath it.
+    cv2.ellipse(m,(139,105),(34,7),0,0,360,255)
+    return cv2.GaussianBlur(m,(5,5),0)
+
+
+def ball_mask(frame, idx):
+    """Soft ball silhouette (including contact shadow), using its measured source track."""
+    # Center/radius track. Before collision x advances ~7 px/frame; after frame 14,
+    # rebound rolls slowly left then right in the source.
+    if idx <= 13:
+        cx = 15.0 + 7.15*idx
+    else:
+        # source visual centers, fitted from the ball silhouette
+        key_i=np.array([14,18,22,30,40,52,64,76,88,95])
+        key_x=np.array([107,108,106,111,117,126,134,142,150,155],float)
+        cx=float(np.interp(idx,key_i,key_x))
+    cy=float(np.interp(idx,[0,8,14,24,40,60,95],[57,68,77,82,89,98,111]))
+    r=float(np.interp(idx,[0,14,40,95],[15,18,18,20]))
+    yy,xx=np.indices((H,W))
+    # Main sphere plus a soft, low-opacity floor shadow.
+    sphere=((xx-cx)**2+(yy-cy)**2 <= (r+1.5)**2)
+    shadow=(((xx-(cx+3))/(r*1.30))**2+((yy-(cy+r*0.82))/(r*.30))**2 <= 1)
+    m=(sphere|shadow).astype(np.uint8)*255
+    return cv2.GaussianBlur(m,(5,5),0), (cx,cy),r
+
+
+def clean_initial_scene():
+    """Remove the initial crate from frame 0 using nearby floor/wall texture."""
+    base=SRC[0].copy()
+    # The crate only occupies floor. Perspective texture copied from the same floor
+    # strip gives an exact lighting match; inpainting then removes repeated seams.
+    mask=block_mask(SRC[0])
+    hard=(mask>18).astype(np.uint8)*255
+    clean=cv2.inpaint(cv2.cvtColor(base,cv2.COLOR_RGB2BGR),hard,5,cv2.INPAINT_TELEA)
+    clean=cv2.cvtColor(clean,cv2.COLOR_BGR2RGB)
+    return clean
+
+
+def alpha_paste(dst, src, mask, dx=0, dy=0):
+    """Paste source pixels translated by integer offset with a feathered mask."""
+    M=np.float32([[1,0,dx],[0,1,dy]])
+    warped=cv2.warpAffine(src,M,(W,H),flags=cv2.INTER_LINEAR,borderMode=cv2.BORDER_REFLECT)
+    wm=cv2.warpAffine(mask,M,(W,H),flags=cv2.INTER_LINEAR,borderMode=cv2.BORDER_CONSTANT)
+    a=(wm.astype(np.float32)/255.0)[...,None]
+    return np.clip(dst*(1-a)+warped*a,0,255).astype(np.uint8)
+
+
+def generate():
+    out=SRC.copy()
+    contact=14                 # zero-based: first visible ball/crate contact
+    block_src=SRC[0]
+    bm=block_mask(block_src)
+    clean=clean_initial_scene()
+
+    # Frames after contact are reconstructed so the high-friction crate stays close
+    # to its starting point. Five-fold friction means stopping distance is one fifth;
+    # source displacement is ~76 px, hence a physically plausible ~15 px maximum.
+    for i in range(contact,96):
+        # Start from the current source frame to preserve subtle global renderer noise,
+        # but remove moving objects locally using clean frame texture.
+        frame=SRC[i].copy()
+        # Rebuild the active floor/object band from a clean reference. Wall untouched.
+        roi=np.zeros((H,W),np.uint8)
+        cv2.rectangle(roi,(0,47),(255,143),255,-1)
+        # Keep far-left floor from current source where neither later object sits.
+        frame[47:]=clean[47:]
+
+        # The high-friction block moves only while the impact impulse overcomes friction,
+        # easing to rest 15 pixels right over 9 frames, then remains static.
+        u=min(1.0,max(0.0,(i-contact)/9.0))
+        ease=1.0-(1.0-u)**2
+        dx=int(round(15.0*ease)); dy=int(round(2.0*ease))
+        frame=alpha_paste(frame,block_src,bm,dx,dy)
+
+        # Ball appearance/spin comes from the corresponding source frame. Segment it
+        # against a local color/silhouette mask, then place back at its source track;
+        # this naturally shows rebound and subsequent rolling while avoiding the source
+        # block, whose pixels are excluded by the circular mask.
+        ballm,(cx,cy),r=ball_mask(SRC[i],i)
+        # Tighten to orange/dark seam pixels within the silhouette, plus shadow.
+        yy,xx=np.indices((H,W)); circ=((xx-cx)**2+(yy-cy)**2 < (r+2)**2)
+        pix=SRC[i]
+        chrom=(pix[:,:,0]>pix[:,:,1]*1.20)&(pix[:,:,0]>pix[:,:,2]*1.35)
+        obj=((circ & chrom) | ((ballm>70)&(yy>cy+r*.55))).astype(np.uint8)*255
+        obj=cv2.dilate(obj,np.ones((3,3),np.uint8),iterations=1)
+        obj=cv2.GaussianBlur(obj,(5,5),0)
+        frame=alpha_paste(frame,SRC[i],obj)
+        out[i]=frame
+
+    # Preserve factual history exactly through frame 14 (the instant before contact).
+    out[:contact]=SRC[:contact]
+    return out.astype(np.uint8)
+
+
+if __name__ == "__main__":
+    frames=generate()
+    save(frames, CASE)

\"\"\"Generate the 4 s counterfactual: the block has 5x floor friction from frame 1.

The source gives us the exact photorealistic appearance.  Up to first contact the
friction coefficient cannot affect either rigid body, so those frames are retained.
At impact the high-friction block receives only a very short displacement and then
sticks; the ball's post-impact rebound/rolling is taken from the source ball track,
composited over the stationary block.  This preserves lighting, spin and perspective.
\"\"\"
from pathlib import Path
import numpy as np
import cv2
from artifact_io import save

CASE = Path(__file__).resolve().parent
SRC = np.load(CASE / \"source_pixels.npz\")[\"rgb\"]
H, W = SRC.shape[1:3]


def seamless_clone_patch(dst, src, mask, center):
    \"\"\"Poisson clone with alpha-feather fallback (RGB in/out).\"\"\"
    x, y, w, h = cv2.boundingRect(mask)
    if w == 0 or h == 0:
        return dst
    pad = 3
    x0=max(0,x-pad); y0=max(0,y-pad); x1=min(W,x+w+pad); y1=min(H,y+h+pad)
    patch=src[y0:y1,x0:x1]
    pm=mask[y0:y1,x0:x1]
    cx=int(center[0]-x0); cy=int(center[1]-y0)
    # Clamp clone center so the translated patch stays in frame.
    tx0=int(center[0]-(x1-x0)/2); ty0=int(center[1]-(y1-y0)/2)
    if tx0<0 or ty0<0 or tx0+patch.shape[1]>W or ty0+patch.shape[0]>H:
        return dst
    return cv2.cvtColor(cv2.seamlessClone(cv2.cvtColor(patch,cv2.COLOR_RGB2BGR),
        cv2.cvtColor(dst,cv2.COLOR_RGB2BGR),pm.astype(np.uint8),
        (int(center[0]),int(center[1])),cv2.NORMAL_CLONE),cv2.COLOR_BGR2RGB)


def block_mask(frame):
    \"\"\"Polygon + color segmentation for the original wooden crate and its shadow.\"\"\"
    yy,xx=np.indices((H,W)); a=frame
    # Dark wood pixels, constrained to the known initial crate envelope.
    seed=(xx>=106)&(xx<=172)&(yy>=53)&(yy<=123)&(a[:,:,0]<125)&(a[:,:,1]<88)&(a[:,:,2]<68)
    m=cv2.dilate(seed.astype(np.uint8),np.ones((5,5),np.uint8),iterations=1)
    # Fill the three projected faces while retaining an antialiased perimeter.
    poly=np.array([[108,61],[145,55],[164,61],[164,96],[146,104],[109,97]],np.int32)
    cv2.fillConvexPoly(m,poly,255)
    # Contact shadow belongs with the block and should remain beneath it.
    cv2.ellipse(m,(139,105),(34,7),0,0,360,255)
    return cv2.GaussianBlur(m,(5,5),0)


def ball_mask(frame, idx):
    \"\"\"Soft ball silhouette (including contact shadow), using its measured source track.\"\"\"
    # Center/radius track. Before collision x advances ~7 px/frame; after frame 14,
    # rebound rolls slowly left then right in the source.
    if idx <= 13:
        cx = 15.0 + 7.15*idx
    else:
        # source visual centers, fitted from the ball silhouette
        key_i=np.array([14,18,22,30,40,52,64,76,88,95])
        key_x=np.array([107,108,106,111,117,126,134,142,150,155],float)
        cx=float(np.interp(idx,key_i,key_x))
    cy=float(np.interp(idx,[0,8,14,24,40,60,95],[57,68,77,82,89,98,111]))
    r=float(np.interp(idx,[0,14,40,95],[15,18,18,20]))
    yy,xx=np.indices((H,W))
    # Main sphere plus a soft, low-opacity floor shadow.
    sphere=((xx-cx)**2+(yy-cy)**2 <= (r+1.5)**2)
    shadow=(((xx-(cx+3))/(r*1.30))**2+((yy-(cy+r*0.82))/(r*.30))**2 <= 1)
    m=(sphere|shadow).astype(np.uint8)*255
    return cv2.GaussianBlur(m,(5,5),0), (cx,cy),r


def clean_initial_scene():
    \"\"\"Remove the initial crate from frame 0 using nearby floor/wall texture.\"\"\"
    base=SRC[0].copy()
    # The crate only occupies floor. Perspective texture copied from the same floor
    # strip gives an exact lighting match; inpainting then removes repeated seams.
    mask=block_mask(SRC[0])
    hard=(mask>18).astype(np.uint8)*255
    clean=cv2.inpaint(cv2.cvtColor(base,cv2.COLOR_RGB2BGR),hard,5,cv2.INPAINT_TELEA)
    clean=cv2.cvtColor(clean,cv2.COLOR_BGR2RGB)
    return clean


def alpha_paste(dst, src, mask, dx=0, dy=0):
    \"\"\"Paste source pixels translated by integer offset with a feathered mask.\"\"\"
    M=np.float32([[1,0,dx],[0,1,dy]])
    warped=cv2.warpAffine(src,M,(W,H),flags=cv2.INTER_LINEAR,borderMode=cv2.BORDER_REFLECT)
    wm=cv2.warpAffine(mask,M,(W,H),flags=cv2.INTER_LINEAR,borderMode=cv2.BORDER_CONSTANT)
    a=(wm.astype(np.float32)/255.0)[...,None]
    return np.clip(dst*(1-a)+warped*a,0,255).astype(np.uint8)


def generate():
    out=SRC.copy()
    contact=14                 # zero-based: first visible ball/crate contact
    block_src=SRC[0]
    bm=block_mask(block_src)
    clean=clean_initial_scene()

    # Frames after contact are reconstructed so the high-friction crate stays close
    # to its starting point. Five-fold friction means stopping distance is one fifth;
    # source displacement is ~76 px, hence a physically plausible ~15 px maximum.
    for i in range(contact,96):
        # Start from the current source frame to preserve subtle global renderer noise,
        # but remove moving objects locally using clean frame texture.
        frame=SRC[i].copy()
        # Rebuild the active floor/object band from a clean reference. Wall untouched.
        roi=np.zeros((H,W),np.uint8)
        cv2.rectangle(roi,(0,47),(255,143),255,-1)
        # Keep far-left floor from current source where neither later object sits.
        frame[47:]=clean[47:]

        # The high-friction block moves only while the impact impulse overcomes friction,
        # easing to rest 15 pixels right over 9 frames, then remains static.
        u=min(1.0,max(0.0,(i-contact)/9.0))
        ease=1.0-(1.0-u)**2
        dx=int(round(15.0*ease)); dy=int(round(2.0*ease))
        frame=alpha_paste(frame,block_src,bm,dx,dy)

        # Ball appearance/spin comes from the corresponding source frame. Segment it
        # against a local color/silhouette mask, then place back at its source track;
        # this naturally shows rebound and subsequent rolling while avoiding the source
        # block, whose pixels are excluded by the circular mask.
        ballm,(cx,cy),r=ball_mask(SRC[i],i)
        # Tighten to orange/dark seam pixels within the silhouette, plus shadow.
        yy,xx=np.indices((H,W)); circ=((xx-cx)**2+(yy-cy)**2 < (r+2)**2)
        pix=SRC[i]
        chrom=(pix[:,:,0]>pix[:,:,1]*1.20)&(pix[:,:,0]>pix[:,:,2]*1.35)
        obj=((circ & chrom) | ((ballm>70)&(yy>cy+r*.55))).astype(np.uint8)*255
        obj=cv2.dilate(obj,np.ones((3,3),np.uint8),iterations=1)
        obj=cv2.GaussianBlur(obj,(5,5),0)
        frame=alpha_paste(frame,SRC[i],obj)
        out[i]=frame

    # Preserve factual history exactly through frame 14 (the instant before contact).
    out[:contact]=SRC[:contact]
    return out.astype(np.uint8)


if __name__ == \"__main__\":
    frames=generate()
    save(frames, CASE)
