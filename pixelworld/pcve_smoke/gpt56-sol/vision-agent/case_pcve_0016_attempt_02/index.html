+"""Remove the wooden block from frame 17 onward and preserve ball motion.
+
+The source camera is static.  We recover the clean plate behind the block by
+projective texture synthesis from adjacent floorboards/wall, then extract the
+basketball (with its natural shadow/antialiasing) from each original frame and
+translate it along its unobstructed constant-velocity trajectory.
+"""
+from pathlib import Path
+import numpy as np
+import cv2
+from scipy.ndimage import gaussian_filter
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+SRC = np.load(ROOT / 'source_pixels.npz')['rgb']
+H, W = 144, 256
+
+# Ball centres measured from frames 1..12, before contact (pixels).
+# Linear/quadratic fit gives unobstructed image-plane motion.
+centers = np.array([[16,57],[22,58],[29,59],[36,60],[43,62],[50,63],
+                    [57,65],[65,66],[72,68],[80,69],[88,71],[95,73]], float)
+t = np.arange(12, dtype=float)
+px = np.polyfit(t, centers[:,0], 1)
+py = np.polyfit(t, centers[:,1], 2)
+
+def clean_plate(frame):
+    """Fill the object's fixed footprint using nearby scene texture.
+
+    The wall strip is copied horizontally (nearly textureless). The floor fill
+    is a perspective-aware affine remap from the unobstructed floor to the left:
+    rows are preserved so floorboard perspective, illumination and grain agree.
+    A second donor offset is softly mixed to avoid recognizable repetitions.
+    """
+    im = frame.copy()
+    yy, xx = np.mgrid[:H, :W]
+    # Generous silhouette: crate, contact shadow, and surrounding AO.
+    # polygon follows top and perspective-expanded bottom footprint.
+    poly = np.array([[106,55],[172,54],[178,66],[178,109],[169,116],
+                     [104,106],[102,67]], np.int32)
+    hard = np.zeros((H,W), np.uint8)
+    cv2.fillPoly(hard, [poly], 255)
+    # Use same scanline at x-72; this maintains floorboard slope/perspective.
+    mapx1 = (xx.astype(np.float32)-72.0)
+    mapx2 = (xx.astype(np.float32)-91.0)
+    mapy = yy.astype(np.float32)
+    d1 = cv2.remap(frame, mapx1, mapy, cv2.INTER_LINEAR,
+                   borderMode=cv2.BORDER_REFLECT_101)
+    d2 = cv2.remap(frame, mapx2, mapy, cv2.INTER_LINEAR,
+                   borderMode=cv2.BORDER_REFLECT_101)
+    # Low-frequency mix varies gradually; retains sharp wood texture.
+    mix = (0.68*d1.astype(np.float32)+0.32*d2.astype(np.float32)).astype(np.uint8)
+    # Wall/baseboard needs no floor texture; horizontal donor is exact geometry.
+    wall = yy < 67
+    mix[wall] = d1[wall]
+    # Feather only exterior 3 px, leaving all original unaffected pixels intact.
+    alpha = gaussian_filter((hard/255.).astype(np.float32), 1.15)[...,None]
+    return np.clip(im*(1-alpha)+mix*alpha,0,255).astype(np.uint8)
+
+def ball_layer(i):
+    """Extract ball and its small cast/contact shadow via clean-plate delta."""
+    # Source source-ball location from a robust colour/trajectory tracker.
+    # Before impact these measured/fitted centers are exact enough for masking.
+    if i < 12:
+        cx,cy = centers[i]
+    else:
+        # Original ball stays beside the block; estimate center by orange score.
+        score = SRC[i,:,:,0].astype(float)-1.50*SRC[i,:,:,1]+.12*SRC[i,:,:,2]
+        ker=np.zeros((37,37),np.float32); gy,gx=np.mgrid[-18:19,-18:19]
+        ker[gx*gx+gy*gy <= 15*15]=1
+        sm=cv2.filter2D(score.astype(np.float32),-1,ker)
+        sm[:35]=-1e9; sm[120:]=-1e9
+        cy,cx=np.unravel_index(np.argmax(sm),sm.shape)
+    # Elliptical extraction includes antialiased ball edge and floor shadow.
+    rx,ry=20.5,23.0
+    y,x=np.mgrid[:H,:W]
+    a=np.clip(1-((x-cx)/rx)**2-((y-(cy+2))/ry)**2,0,1)
+    a=np.clip(a*4.0,0,1)
+    a=gaussian_filter(a.astype(np.float32),.65)
+    return cx,cy,a[...,None]
+
+def shift_layer(rgb, alpha, dx, dy):
+    M=np.array([[1,0,dx],[0,1,dy]],np.float32)
+    c=cv2.warpAffine(rgb,M,(W,H),flags=cv2.INTER_LINEAR,
+                     borderMode=cv2.BORDER_CONSTANT,borderValue=0)
+    a=cv2.warpAffine(alpha[:,:,0],M,(W,H),flags=cv2.INTER_LINEAR,
+                     borderMode=cv2.BORDER_CONSTANT,borderValue=0)[...,None]
+    return c,a
+
+def generate():
+    out=SRC.copy()
+    for i in range(16,96): # frame 17 in one-based source numbering
+        bg=clean_plate(SRC[i])
+        # Remove original ball as well; it was affected by the deleted collision.
+        ocx,ocy,a=ball_layer(i)
+        # Local clean floor donor under original ball; scanline-preserving shift.
+        donor=cv2.warpAffine(bg,np.array([[1,0,-48],[0,1,0]],np.float32),(W,H),
+                            flags=cv2.INTER_LINEAR,borderMode=cv2.BORDER_REFLECT_101)
+        bg=np.clip(bg*(1-a)+donor*a,0,255).astype(np.uint8)
+        # Continue pre-contact velocity. Slight quadratic vertical path follows
+        # the observed approach; after leaving view the scene remains empty.
+        ncx=np.polyval(px,i)
+        ncy=np.polyval(py,i)
+        # Use the latest pre-contact ball appearance to avoid collision deformation.
+        ref=min(i,11)
+        rcx,rcy,ra=ball_layer(ref)
+        layer=SRC[ref]
+        c,aa=shift_layer(layer,ra,ncx-rcx,ncy-rcy)
+        out[i]=np.clip(bg*(1-aa)+c*aa,0,255).astype(np.uint8)
+    return out
+
+if __name__ == '__main__':
+    frames=generate()
+    artifact_io.save(frames, ROOT)
