        if t>13:
            u=t-13
            x=(centers[10,0]+1.90*13-.006*169) + 1.20*u - .004*u*u
            y=(centers[10,1]-1.15*13+.004*169) + .43*u + .001*u*u
        p=np.array([x,y],np.float32)
    pos.append(p)

    # Since intervention starts at frame 1, every visible ball image is rebuilt.
    out=remove_source_ball(SRC[i].copy(),i)
    x,y=pos[i]
    # perspective size diminishes toward back of room, then grows slightly as
    # redirected motion returns toward camera.
    if i<=10:r=radii[i]
    elif i<=23:r=max(6.9,8.7-.055*(i-10))
    else:r=7.98 + .013*(i-23)
    add_shadow(out,x,y,r,.30 if i>7 else .22)

    # late stable sprite so surface markings continue to spin.
    si=min(i,18)
    rgb,a,_,sr=sprites[si]
    angle=0 if i<18 else -((i-18)*17.0)
    paste_rgba(out,rgb,a,(x,y),rotation=angle,scale=r/sr)
    frames.append(out)
frames=np.stack(frames).astype(np.uint8)
save(frames,CASE)
