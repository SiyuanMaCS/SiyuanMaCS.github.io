+"""Generate the low-restitution counterfactual for this PixelWorld case.
+
+The scene is locked-off.  The yellow (new) tennis ball first touches the
+green ball/the floor around source frame 22 (index 21).  Before that contact,
+restitution cannot affect its motion, so those source frames are retained
+exactly.  At contact the source has a strongly separating collision: the
+green ball shoots left while the yellow ball reverses to the right.  Reducing
+the yellow ball's restitution to one tenth instead makes the two near-equal
+balls leave contact with similar leftward velocities.  Ground friction then
+brings both to rest beside one another.
+"""
+
+from pathlib import Path
+
+import numpy as np
+from PIL import Image, ImageFilter, ImageDraw
+
+import artifact_io
+
+
+HERE = Path(__file__).resolve().parent
+
+
+def clean_floor(source):
+    """Return a ball-free copy of the small floor strip used after impact.
+
+    Different temporal ranges are used because the green ball occupies the
+    left-middle early in the clip and the two source balls occupy the far
+    left/right late in the clip.  The lighting and camera are static.
+    """
+    floor = np.empty((26, 120, 3), dtype=np.uint8)
+    # x=8..44 and x=100..127 are empty before impact.
+    early = np.median(source[:18, 82:108], axis=0).astype(np.uint8)
+    # x=45..99 is empty after the source balls have settled.
+    late = np.median(source[42:88, 82:108], axis=0).astype(np.uint8)
+    floor[:, :37] = early[:, 8:45]
+    floor[:, 37:92] = late[:, 45:100]
+    floor[:, 92:] = early[:, 100:128]
+    return floor
+
+
+def ball_sprite(frame, box, center, radius):
+    """Extract a softly antialiased ball sprite, excluding its cast shadow."""
+    x0, y0, x1, y1 = box
+    rgb = frame[y0:y1, x0:x1].copy()
+    yy, xx = np.mgrid[y0:y1, x0:x1]
+    distance = np.sqrt((xx - center[0]) ** 2 + (yy - center[1]) ** 2)
+    # The rendered source already contains edge antialiasing.  This narrow
+    # geometric feather keeps only the sphere and prevents a rectangular halo.
+    alpha = np.clip((radius + 0.65 - distance) / 1.15, 0.0, 1.0)
+    return rgb, alpha
+
+
+def add_shadow(frame, center, strength=0.27):
+    """Composite the soft, right-cast floor shadow used by the source render."""
+    # Work at 4x resolution so the tiny ellipse remains smooth.
+    scale = 4
+    layer = Image.new("L", (frame.shape[1] * scale, frame.shape[0] * scale), 0)
+    draw = ImageDraw.Draw(layer)
+    cx = (center[0] + 5.0) * scale
+    cy = 99.2 * scale
+    rx, ry = 8.0 * scale, 2.25 * scale
+    draw.ellipse((cx - rx, cy - ry, cx + rx, cy + ry), fill=255)
+    layer = layer.filter(ImageFilter.GaussianBlur(2.5 * scale))
+    alpha = np.asarray(layer.resize((frame.shape[1], frame.shape[0]), Image.Resampling.LANCZOS), dtype=np.float32) / 255.0
+    darken = 1.0 - strength * alpha[..., None]
+    frame[:] = np.clip(frame.astype(np.float32) * darken, 0, 255).astype(np.uint8)
+
+
+def paste_ball(frame, sprite, alpha, box, source_center, target_center):
+    """Paste one extracted sphere at an integer-rounded target center."""
+    x0, y0, x1, y1 = box
+    dx = int(round(target_center[0] - source_center[0]))
+    dy = int(round(target_center[1] - source_center[1]))
+    tx0, ty0 = x0 + dx, y0 + dy
+    tx1, ty1 = x1 + dx, y1 + dy
+    # All trajectories in this case stay safely inside the frame.
+    region = frame[ty0:ty1, tx0:tx1].astype(np.float32)
+    a = alpha[..., None].astype(np.float32)
+    frame[ty0:ty1, tx0:tx1] = np.clip(sprite.astype(np.float32) * a + region * (1.0 - a), 0, 255).astype(np.uint8)
+
+
+def position_after_impact(frame_index):
+    """Analytic low-restitution trajectories from source contact frame 21.
+
+    Source image velocities immediately after impact are roughly +4 px/frame
+    (yellow) and -6 px/frame (green).  One-tenth restitution makes their
+    separating speed about one tenth as large, while preserving their common
+    leftward momentum.  The quadratic terms model the source floor friction.
+    """
+    k = float(frame_index - 21)
+
+    # Green ball: v=-3.3 px/frame, friction=+0.45 px/frame^2.
+    kg = min(max(k, 0.0), 3.3 / 0.45)
+    green_x = 65.3 - 3.3 * kg + 0.5 * 0.45 * kg * kg
+
+    # Yellow ball: v=-2.7 px/frame, friction=+0.30 px/frame^2.
+    ky = min(max(k, 0.0), 2.7 / 0.30)
+    yellow_x = 76.4 - 2.7 * ky + 0.5 * 0.30 * ky * ky
+
+    # Its greatly reduced floor restitution suppresses the visible rebound.
+    yellow_y = 93.4 - 3.2 * np.exp(-1.65 * max(k, 0.0))
+    return (green_x, 94.7), (yellow_x, yellow_y)
+
+
+def generate():
+    source = np.load(HERE / "source_pixels.npz")["rgb"]
+    out = source.copy()
+    floor = clean_floor(source)
+
+    # Stable source appearances after the original simulation has settled.
+    green_box = (15, 88, 30, 103)
+    green_center = (22.1, 94.9)
+    green_sprite, green_alpha = ball_sprite(source[50], green_box, green_center, 6.15)
+
+    yellow_box = (103, 86, 118, 101)
+    yellow_center = (109.9, 93.4)
+    yellow_sprite, yellow_alpha = ball_sprite(source[50], yellow_box, yellow_center, 5.75)
+
+    # Frame 21 (index 20) is still pre-contact.  Frame 22 captures contact and
+    # is retained as the shared event state.  Consequences begin one frame on.
+    for i in range(22, 96):
+        out[i, 82:108, 8:128] = floor
+        gpos, ypos = position_after_impact(i)
+        add_shadow(out[i], gpos, 0.30)
+        add_shadow(out[i], ypos, 0.25)
+        paste_ball(out[i], green_sprite, green_alpha, green_box, green_center, gpos)
+        paste_ball(out[i], yellow_sprite, yellow_alpha, yellow_box, yellow_center, ypos)
+
+    return out
+
+
+if __name__ == "__main__":
+    frames = generate()
+    artifact_io.save(frames, HERE)
+
