+
+The supplied footage is used as the renderer: source frames 0..22 contain the
+high-quality car sprite at every needed pose, while frame 35 supplies a clean
+ball.  Geometry and alpha masks are inferred only from source_pixels.npz.
+"""

+import numpy as np
+from PIL import Image
+from scipy.ndimage import binary_dilation, gaussian_filter, shift
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+SRC = np.load(ROOT / "source_pixels.npz")["rgb"]
+N,H,W,_ = SRC.shape
+
+# Static set and ball: by f35 the shelf area vacated by the moving car is a
+# clean plate; the ball is still fully visible at the left edge.
+plate = SRC[35].copy()
+ball_ref = SRC[0].astype(np.int16)
+ball_bg = SRC[35].astype(np.int16)
+Y0,Y1,X0,X1 = 18,40,123,147
+ball_patch = ball_ref[Y0:Y1,X0:X1].copy()
+diff = np.max(np.abs(ball_patch-ball_bg[Y0:Y1,X0:X1]), axis=2)
+ball_mask = np.clip(gaussian_filter((diff > 9).astype(float), .55)*1.2, 0, 1)
+
+# Extract the car from each clean pre-collision source frame.  Comparing with
+# plate is robust because plant/shelf/camera are fixed. Limit to its corridor.
+car_sprites=[]
+for i in range(23):
+    d=np.max(np.abs(SRC[i].astype(np.int16)-plate.astype(np.int16)),axis=2)
+    m=np.zeros((H,W),bool); m[21:38,125:224] = d[21:38,125:224] > 7
+    m=binary_dilation(m,iterations=1)
+    a=np.clip(gaussian_filter(m.astype(float),.45)*1.25,0,1)
+    car_sprites.append((SRC[i].copy(),a))
+
+# 1-D collision. The source car approaches at about -3 px/frame initially.
+# With m_ball = 20*m_car and a moderately elastic toy collision (e=.58), the
+# car rebounds at ~1.51 times its incident speed while the massive ball gets
+# only a very small leftward velocity. Contact occurs near frame 21.
+contact=21
+x_ball0=135.0
+v_ball=-3.0*(1+.58)/(1+20)       # -0.226 px/frame
+v_car=3.0*(20*.58-1)/(1+20)      # +1.514 px/frame
+
+# The car centre in source frame 21 is 151.5. Use source sprite 21 at contact.
+car_c0=151.5
+frames=[]
+for t in range(N):
+    canvas=plate.astype(float)
+    if t <= contact:
+        # exact source positions up to the collision; ball stays at x=135.
+        si=t
+        cx=None
+    else:
+        # car is pushed back to the right by the 20x ball and then coasts.
+        cx=car_c0 + v_car*(t-contact)
+        # nearest clean source pose gives natural wheel rotation / texture.
+        centers=np.array([202.5,199.5,196.5,193.5,190.5,187.5,185.5,183.,180.5,
+                          178.,176.,174.,172.,170.,168.,166.5,164.5,163.,161.5,
+                          160.,158.5,157.5,156.5])
+        si=int(np.argmin(np.abs(centers-cx)))
+    spr,a=car_sprites[si]
+    if cx is not None:
+        centers_si=[202.5,199.5,196.5,193.5,190.5,187.5,185.5,183.,180.5,
+                    178.,176.,174.,172.,170.,168.,166.5,164.5,163.,161.5,
+                    160.,158.5,157.5,156.5][si]
+        dx=cx-centers_si
+        aa=shift(a,(0,dx),order=1,mode='constant',cval=0,prefilter=False)
+        ss=np.stack([shift(spr[...,c].astype(float),(0,dx),order=1,
+                           mode='nearest',prefilter=False) for c in range(3)],axis=2)
+    else:
+        aa=a; ss=spr.astype(float)
+    canvas=ss*aa[...,None]+canvas*(1-aa[...,None])
+
+    # Massive ball barely recoils, remaining close to the shelf edge instead
+    # of being knocked off. It is composited last at contact for correct
+    # foreground overlap with the car bumper.
+    bx=x_ball0 if t<=contact else x_ball0+v_ball*(t-contact)
+    dx=bx-x_ball0
+    ba=shift(ball_mask,(0,dx),order=1,mode='constant',cval=0,prefilter=False)
+    bp=np.stack([shift(ball_patch[...,c].astype(float),(0,dx),order=1,
+                       mode='nearest',prefilter=False) for c in range(3)],axis=2)
+    fulla=np.zeros((H,W)); fullp=np.zeros((H,W,3))
+    fulla[Y0:Y1,X0:X1]=ba; fullp[Y0:Y1,X0:X1]=bp
+    canvas=fullp*fulla[...,None]+canvas*(1-fulla[...,None])
+    frames.append(np.uint8(np.clip(np.rint(canvas),0,255)))
+
+frames=np.stack(frames)
+artifact_io.save(frames, ROOT)
