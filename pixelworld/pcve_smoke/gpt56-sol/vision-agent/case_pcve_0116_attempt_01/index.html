+"""Generate the requested counterfactual PixelWorld video.
+
+The toy car's initial horizontal speed is reduced to 0.6 of the measured
+source value while retaining the source's measured (approximately constant)
+frictional deceleration.  It therefore comes to rest before touching the ball.
+Only pre-collision source states are sampled, preserving all source appearance
+and leaving the ball undisturbed.
+"""
+from pathlib import Path
+
+import numpy as np
+from scipy.ndimage import label
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+SPEED_SCALE = 0.6
+
+
+def track_car_centers(frames: np.ndarray, count: int = 15) -> np.ndarray:
+    """Measure the large red car body component in early source frames."""
+    centers = []
+    yy, xx = np.indices(frames.shape[1:3])
+    for frame in frames[:count]:
+        rgb = frame.astype(np.int16)
+        red = (
+            (rgb[..., 0] > 150)
+            & (rgb[..., 0] > 1.35 * rgb[..., 1])
+            & (rgb[..., 0] > 1.25 * rgb[..., 2])
+            & (yy >= 25)
+            & (yy <= 36)
+            & (xx > 145)
+        )
+        groups, number = label(red)
+        candidates = []
+        for component in range(1, number + 1):
+            cy, cx = np.where(groups == component)
+            if cx.size >= 60:
+                candidates.append((cx.size, float(cx.mean())))
+        if not candidates:
+            raise RuntimeError("Could not locate the toy car in an early source frame")
+        centers.append(max(candidates)[1])
+    return np.asarray(centers, dtype=np.float64)
+
+
+def measure_motion(frames: np.ndarray) -> tuple[float, float]:
+    """Fit source displacement d(t) = v*t - 0.5*a*t^2."""
+    centers = track_car_centers(frames)
+    t = np.arange(centers.size, dtype=np.float64)
+    displacement = centers[0] - centers
+    design = np.column_stack((t, -0.5 * t * t))
+    velocity, deceleration = np.linalg.lstsq(design, displacement, rcond=None)[0]
+    if velocity <= 0 or deceleration <= 0:
+        raise RuntimeError((velocity, deceleration))
+    return float(velocity), float(deceleration)
+
+
+def source_time_for_displacement(distance: float, velocity: float, deceleration: float) -> float:
+    """Invert the fitted source trajectory on its forward-moving branch."""
+    discriminant = max(0.0, velocity * velocity - 2.0 * deceleration * distance)
+    return (velocity - np.sqrt(discriminant)) / deceleration
+
+
+def render() -> np.ndarray:
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    if source.shape != (96, 144, 256, 3) or source.dtype != np.uint8:
+        raise ValueError((source.shape, source.dtype))
+
+    source_velocity, deceleration = measure_motion(source)
+    new_velocity = SPEED_SCALE * source_velocity
+    stop_time = new_velocity / deceleration
+    stop_distance = new_velocity * new_velocity / (2.0 * deceleration)
+
+    output = np.empty_like(source)
+    for frame_index in range(source.shape[0]):
+        t = min(float(frame_index), stop_time)
+        distance = new_velocity * t - 0.5 * deceleration * t * t
+        distance = min(max(distance, 0.0), stop_distance)
+
+        # The corresponding early source state has exactly this car position.
+        # These source times precede collision, so the ball remains untouched.
+        source_time = source_time_for_displacement(distance, source_velocity, deceleration)
+        lo = int(np.floor(source_time))
+        hi = min(lo + 1, source.shape[0] - 1)
+        fraction = source_time - lo
+        if fraction < 1e-9:
+            output[frame_index] = source[lo]
+        else:
+            blended = (1.0 - fraction) * source[lo].astype(np.float32)
+            blended += fraction * source[hi].astype(np.float32)
+            output[frame_index] = np.clip(np.rint(blended), 0, 255).astype(np.uint8)
+
+    # Preserve the exact factual appearance at the intervention instant.
+    output[0] = source[0]
+    return output
+
+
+if __name__ == "__main__":
+    frames = render()
+    artifact_io.save(frames, CASE_DIR)

    \"\"\"Measure the large red car body component in early source frames.\"\"\"

    \"\"\"Fit source displacement d(t) = v*t - 0.5*a*t"'^2."""
    centers = track_car_centers(frames)

def source_time_for_displacement(distance: float, velocity: float, deceleration: float) -> float:

    discriminant = max(0.0, velocity * velocity - 2.0 * deceleration * distance)
    return (velocity - np.sqrt(discriminant)) / deceleration


def render() -> np.ndarray:

    if source.shape != (96, 144, 256, 3) or source.dtype != np.uint8:
