+import cv2
+import numpy as np
+from PIL import Image, ImageDraw, ImageFilter
+
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+N, H, W = 96, 144, 256
+

+def colored_components(frame):
+    """Return saturated green/yellow component centroids in the action area."""
+    hsv = cv2.cvtColor(frame, cv2.COLOR_RGB2HSV)
+    mask = ((hsv[:, :, 0] >= 25) & (hsv[:, :, 0] <= 58) &
+            (hsv[:, :, 1] >= 65) & (hsv[:, :, 2] >= 65)).astype(np.uint8)
+    count, labels, stats, centers = cv2.connectedComponentsWithStats(mask, 8)
+    result = []
+    for k in range(1, count):
+        x, y, w, h, area = stats[k]
+        if area >= 18 and y < 110:
+            result.append((int(area), float(centers[k, 0]), float(centers[k, 1])))
+    return result
+
+
+def track_centers(src):
+    yellow = []
+    green = []
+    for i, frame in enumerate(src):
+        comps = colored_components(frame)
+        # The launched yellow ball is the rightmost relevant component throughout.
+        yc = max(comps, key=lambda q: q[1])
+        yellow.append((yc[1], yc[2]))
+        if i < 22:
+            green.append((66.9, 95.2))
+        else:
+            left = [q for q in comps if q[1] < 72]
+            if left:
+                # Split highlights can occur; area-weighted centroid gives a stable center.
+                a = sum(q[0] for q in left)
+                green.append((sum(q[0] * q[1] for q in left) / a,
+                              sum(q[0] * q[2] for q in left) / a))
+            else:
+                green.append(green[-1])
+    # Use the stable terminal centers after all motion ceases.
+    for i in range(42, N):
+        yellow[i] = yellow[41]
+        green[i] = green[41]
+    return np.asarray(yellow), np.asarray(green)
+
+
+def clean_background(src):
+    """Temporal plate with the two long-lived terminal balls removed."""
+    plate = np.median(src, axis=0).astype(np.uint8)
+    # Those balls occupy their final positions for most frames and survive the median.
+    # Frame 1 has clean floor at both final positions, so it is an exact scene patch.
+    plate[82:108, 96:124] = src[0, 82:108, 96:124]
+    plate[84:109, 8:38] = src[0, 84:109, 8:38]
+    # Fill the frame-1 green ball using clean neighboring floor with seamless inpainting.
+    m = np.zeros((H, W), np.uint8)
+    cv2.ellipse(m, (70, 97), (12, 10), 0, 0, 360, 255, -1)
+    plate = cv2.inpaint(plate, m, 5, cv2.INPAINT_TELEA)
+    # Very mild denoising only on the floor removes temporal-median pinpricks.
+    smooth = cv2.medianBlur(plate, 3)
+    floor = np.zeros((H, W), np.float32)
+    floor[82:] = 1.0
+    floor = cv2.GaussianBlur(floor, (0, 0), 2)[:, :, None]
+    plate = np.clip(plate * (1 - .18 * floor) + smooth * (.18 * floor), 0, 255).astype(np.uint8)
+    return plate
+
+
+def interpolate(track, t):
+    t = float(np.clip(t, 0, len(track) - 1))
+    lo = int(np.floor(t))
+    hi = min(lo + 1, len(track) - 1)
+    a = t - lo
+    return track[lo] * (1 - a) + track[hi] * a
+
+
+def ball_sprite(src_frame, plate, center, radius):
+    """Extract the rendered ball texture and soft antialiased silhouette."""
+    cx, cy = center
+    pad = int(np.ceil(radius + 4))
+    x0, y0 = max(0, int(cx) - pad), max(0, int(cy) - pad)
+    x1, y1 = min(W, int(cx) + pad + 2), min(H, int(cy) + pad + 2)
+    patch = src_frame[y0:y1, x0:x1].copy()
+    ref = plate[y0:y1, x0:x1]
+    yy, xx = np.mgrid[y0:y1, x0:x1]
+    dist = np.sqrt((xx - cx) ** 2 + (yy - cy) ** 2)
+    diff = np.max(np.abs(patch.astype(np.int16) - ref.astype(np.int16)), axis=2)
+    hsv = cv2.cvtColor(patch, cv2.COLOR_RGB2HSV)
+    colored = (hsv[:, :, 1] > 48) & (hsv[:, :, 0] >= 22) & (hsv[:, :, 0] <= 62)
+    core = ((diff > 5) | colored) & (dist <= radius)
+    core = cv2.morphologyEx(core.astype(np.uint8), cv2.MORPH_CLOSE,
+                            np.ones((3, 3), np.uint8))
+    # Keep seams/highlights enclosed by the colored shell while retaining soft edges.
+    core = cv2.dilate(core, np.ones((2, 2), np.uint8), iterations=1)
+    alpha = cv2.GaussianBlur(core.astype(np.float32), (0, 0), .55)
+    alpha *= np.clip((radius + .65 - dist) / 1.1, 0, 1)
+    return patch, np.clip(alpha, 0, 1), (cx - x0, cy - y0)
+
+
+def paste_sprite(canvas, sprite, target):
+    patch, alpha, anchor = sprite
+    x0 = int(round(target[0] - anchor[0]))
+    y0 = int(round(target[1] - anchor[1]))
+    ph, pw = patch.shape[:2]
+    xa, ya = max(0, x0), max(0, y0)
+    xb, yb = min(W, x0 + pw), min(H, y0 + ph)
+    if xb <= xa or yb <= ya:
+        return
+    sx, sy = xa - x0, ya - y0
+    a = alpha[sy:sy + yb - ya, sx:sx + xb - xa, None]
+    p = patch[sy:sy + yb - ya, sx:sx + xb - xa].astype(np.float32)
+    c = canvas[ya:yb, xa:xb].astype(np.float32)
+    canvas[ya:yb, xa:xb] = np.clip(c * (1 - a) + p * a, 0, 255).astype(np.uint8)
+
+
+def shadow_layer(size, center, radii, opacity, blur):
+    layer = Image.new("L", (W, H), 0)
+    draw = ImageDraw.Draw(layer)
+    cx, cy = center
+    rx, ry = radii
+    draw.ellipse((cx - rx, cy - ry, cx + rx, cy + ry), fill=int(opacity))
+    return layer.filter(ImageFilter.GaussianBlur(blur))
+
+
+def add_shadow(frame, center, airborne=False):
+    im = Image.fromarray(frame)
+    if airborne:
+        cx, cy = center
+        # Fixed upper-left light: the higher the ball, the farther right its floor shadow.
+        height = max(0.0, 93.0 - cy)
+        sc = (cx + .82 * height + 4, 102.5)
+        alpha = shadow_layer((W, H), sc, (6.2 + .025 * height, 1.9), 36, 2.6)
+    else:
+        cx, cy = center
+        alpha = shadow_layer((W, H), (cx + 5.0, cy + 4.7), (6.0, 2.0), 43, 1.7)
+    shade = Image.new("RGB", (W, H), (76, 77, 78))
+    im.paste(shade, (0, 0), alpha)
+    frame[:] = np.asarray(im)
+
+
+def make_video():
+    src = np.load(ROOT / "source_pixels.npz")["rgb"]
+    plate = clean_background(src)
+    yellow, green = track_centers(src)
+    out = np.repeat(plate[None], N, axis=0)
+
+    # Original edge departure occurs at source index 15. At 0.6 speed it occurs at
+    # output index 25. Gravity is unchanged, so the fall itself is not time-stretched.
+    edge_out = 25
+    source_edge = 15
+    source_ground = 21
+    collision_out = 35
+
+    for i in range(N):
+        frame = out[i]
+
+        # Counterfactual yellow-ball position and the source texture frame to use.
+        if i <= edge_out:
+            st = .6 * i
+            y_pos = interpolate(yellow, st)
+            texture_i = int(round(st))
+            airborne = bool(y_pos[1] > 26.0)
+        elif i <= collision_out:
+            sj = source_edge + (i - edge_out)
+            sj = min(sj, 25)
+            if sj <= source_ground:
+                # Same gravity/vertical path; horizontal displacement is 60 percent.
+                y_pos = np.array([yellow[source_edge, 0] +
+                                  .6 * (yellow[sj, 0] - yellow[source_edge, 0]),
+                                  yellow[sj, 1]])
+            else:
+                # The original ball reversed on contact. Here it initially misses, so
+                # mirror that short, damped floor-bounce displacement to continue left.
+                anchor = np.array([yellow[source_edge, 0] +
+                                   .6 * (yellow[source_ground, 0] - yellow[source_edge, 0]),
+                                   yellow[source_ground, 1]])
+                y_pos = np.array([anchor[0] - (yellow[sj, 0] - yellow[source_ground, 0]),
+                                  yellow[sj, 1]])
+            texture_i = sj
+            airborne = i < 33
+        else:
+            # Equal-mass impact transfers most horizontal speed to the green ball.
+            sj = min(25 + (i - collision_out), 41)
+            impact_x = yellow[source_edge, 0] + .6 * (yellow[source_ground, 0] -
+                       yellow[source_edge, 0]) - (yellow[25, 0] - yellow[source_ground, 0])
+            y_pos = np.array([impact_x, yellow[sj, 1]])
+            texture_i = sj
+            airborne = False
+
+        # Green ball remains unaffected until the delayed weaker collision.
+        if i <= collision_out:
+            g_pos = np.array([66.9, 95.2])
+            g_tex = 0
+        else:
+            gj = min(22 + (i - collision_out - 1), 41)
+            # Reduced transferred speed gives 60 percent of the source displacement.
+            g_pos = np.array([66.9 + .6 * (green[gj, 0] - 66.9), green[gj, 1]])
+            g_tex = gj
+
+        # Shadows go beneath both original rendered ball cutouts.
+        add_shadow(frame, tuple(g_pos), airborne=False)
+        add_shadow(frame, tuple(y_pos), airborne=airborne)
+
+        yr = 7.8 if y_pos[1] < 82 else 6.4
+        y_sprite = ball_sprite(src[texture_i], plate, yellow[texture_i], yr)
+        g_sprite = ball_sprite(src[g_tex], plate, green[g_tex], 6.2)
+        paste_sprite(frame, g_sprite, g_pos)
+        paste_sprite(frame, y_sprite, y_pos)
+
+    return out.astype(np.uint8)
+
+
+if __name__ == "__main__":
+    frames = make_video()
+    artifact_io.save(frames, ROOT)
+    print("saved", frames.shape, frames.dtype, "to", ROOT)
