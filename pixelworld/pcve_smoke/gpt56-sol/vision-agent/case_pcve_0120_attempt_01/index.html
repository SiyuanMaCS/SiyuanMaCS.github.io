+The clean plate is the final source frame: by then the ball has left the shelf and
+all ball-related wall/shelf occlusions are exposed.  The car is reconstructed on

+def smooth_mask(mask, radius=1.15):
+    im = Image.fromarray((mask * 255).astype(np.uint8), "L")

+def shift_rgba(rgb, alpha, dx, dy):
+    """Translate a sprite using bilinear sampling and return canvas RGB/alpha."""
+    h, w = rgb.shape[:2]
+    rgba = np.dstack([rgb, np.clip(alpha * 255, 0, 255).astype(np.uint8)])
+    layer = Image.fromarray(rgba, "RGBA")
+    # PIL affine maps destination to source, hence the negative displacement.
+    layer = layer.transform((w, h), Image.Transform.AFFINE,
+                            (1, 0, -dx, 0, 1, -dy),
+                            resample=Image.Resampling.BILINEAR)
+    a = np.asarray(layer, dtype=np.uint8)
+    return a[:, :, :3], a[:, :, 3].astype(np.float32) / 255.0

+def generate():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3) and source.dtype == np.uint8

+    # A soft foreground mask based on local background differences preserves the
+    # antialiased silhouette, wheels, and highlights while excluding wall/shelf.
+    yy, xx = np.mgrid[:144, :256]
+    car_box = (xx >= 135) & (xx <= 171) & (yy >= 18) & (yy <= 37)
+    car_rgb = source[-1].copy()
+
+    # Estimate the smooth wall behind the car from clean columns around it.
+    wall_bg = np.zeros_like(car_rgb, dtype=np.float32)
+    for y in range(144):

+        left = source[-1, y, 126].astype(np.float32)
+        right = source[-1, y, 181].astype(np.float32)
+        q = np.clip((np.arange(256) - 126) / 55.0, 0, 1)[:, None]
+        wall_bg[y] = left * (1-q) + right * q

+    # Explicit box plus color/contrast discriminant captures low-contrast wheel AA.
+    chroma = car_rgb.max(2).astype(int) - car_rgb.min(2).astype(int)
+    dark = car_rgb.mean(2) < 155

+    core = Image.fromarray((alpha * 255).astype(np.uint8), "L").filter(ImageFilter.MaxFilter(3))
+    alpha = (np.asarray(core) > 0) & car_box & (diff > 4)
+    alpha = smooth_mask(alpha, 0.72)
+    alpha *= car_box.astype(np.float32)
+
+    # Remove the stationary car from the plate to reveal its no-object background.
+    # Wall (through y=33) is a horizontal interpolation; shelf (y>=34) is copied
+    # from the adjacent clean shelf strip, which has the same horizontal shading.
+    clean = plate.copy().astype(np.float32)
+    for y in range(18, 34):

+        left = plate[y, x0].astype(np.float32)

+        clean[y, x0:x1+1] = left*(1-q)+right*q

+    # retain the source's exact layered shelf edges.

+        output[i] = np.clip(frame + 0.5, 0, 255).astype(np.uint8)
