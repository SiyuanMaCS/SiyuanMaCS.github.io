+"""Generate the requested rigid-body counterfactual.
+
+The source shot is locked off.  We reconstruct the small portions of table hidden
+by the original blue/green and white mallets, retain the red mallet verbatim,
+then composite source-derived mallet sprites at their counterfactual positions.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def soft_alpha_from_difference(foreground, background, low=3.0, high=32.0):
+    """Estimate a soft object matte from two otherwise matching source images."""
+    delta = np.max(np.abs(foreground.astype(np.float32) -
+                          background.astype(np.float32)), axis=2)
+    alpha = np.clip((delta - low) / (high - low), 0.0, 1.0)
+    return cv2.GaussianBlur(alpha, (3, 3), 0.55)
+
+
+def over(dst, rgb, alpha):
+    """Alpha composite one full-frame RGB layer in place."""
+    a = np.clip(alpha[..., None], 0.0, 1.0)
+    dst[:] = dst * (1.0 - a) + rgb.astype(np.float32) * a
+
+
+def place_sprite(dst, sprite_rgb, sprite_alpha, source_center, target_center,
+                 scale=1.0):
+    """Place a crop by a subpixel affine transform, preserving its soft shadow."""
+    h, w = dst.shape[:2]
+    sh, sw = sprite_alpha.shape
+    sx, sy = source_center
+    tx, ty = target_center
+
+    # The affine acts on crop coordinates and emits into full-frame coordinates.
+    matrix = np.array([
+        [scale, 0.0, tx - scale * sx],
+        [0.0, scale, ty - scale * sy],
+    ], dtype=np.float32)
+    rgb_layer = cv2.warpAffine(
+        sprite_rgb, matrix, (w, h), flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT, borderValue=(0, 0, 0))
+    alpha_layer = cv2.warpAffine(
+        sprite_alpha, matrix, (w, h), flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+    over(dst, rgb_layer, alpha_layer)
+
+
+def build_clean_plate(source):
+    """Make the static table plate while retaining the red mallet in frame 1."""
+    frame0 = source[0].astype(np.float32)
+    later = source[40].astype(np.float32)
+    plate = frame0.copy()
+
+    # The green/blue mallet has left its starting patch by source frame 41.
+    # Difference matting restores the goal/table detail at that old location.
+    box = (153, 41, 183, 78)
+    x0, y0, x1, y1 = box
+    old = source[0, y0:y1, x0:x1]
+    clean = source[40, y0:y1, x0:x1]
+    alpha = soft_alpha_from_difference(old, clean, 2.5, 30.0)
+    # Restrict the matte to the old mallet footprint; this avoids admitting the
+    # distant edge of the mallet at its later resting position.
+    yy, xx = np.mgrid[y0:y1, x0:x1]
+    support = np.exp(-(((xx - 167.0) / 15.0) ** 4 +
+                       ((yy - 59.5) / 18.0) ** 4))
+    alpha *= support
+    patch = plate[y0:y1, x0:x1]
+    patch[:] = patch * (1.0 - alpha[..., None]) + clean * alpha[..., None]
+
+    # The white mallet must be free to move immediately.  Its initial patch is
+    # plain lavender table, so a compact source-aware inpaint is clean here.
+    mask = np.zeros(source.shape[1:3], np.uint8)
+    cv2.ellipse(mask, (105, 82), (13, 12), -8, 0, 360, 255, -1)
+    # Include the soft contact shadow below/right without touching goal lines.
+    cv2.ellipse(mask, (107, 86), (13, 7), 0, 0, 360, 255, -1)
+    plate_u8 = np.clip(plate, 0, 255).astype(np.uint8)
+    plate_u8 = cv2.inpaint(plate_u8, mask, 5, cv2.INPAINT_TELEA)
+    return plate_u8.astype(np.float32)
+
+
+def extract_sprites(source, plate):
+    """Extract anti-aliased RGB sprites, including their contact shadows."""
+    # Green/blue mallet at frame 1 versus its clean patch at frame 41.
+    gx0, gy0, gx1, gy1 = 153, 41, 183, 78
+    green_rgb = source[0, gy0:gy1, gx0:gx1]
+    green_bg = source[40, gy0:gy1, gx0:gx1]
+    green_alpha = soft_alpha_from_difference(green_rgb, green_bg, 2.5, 31.0)
+    gyy, gxx = np.mgrid[gy0:gy1, gx0:gx1]
+    green_alpha *= np.exp(-(((gxx - 167.2) / 14.5) ** 4 +
+                            ((gyy - 59.2) / 17.0) ** 4))
+
+    # White mallet against the just-reconstructed table.  Keep the crop-local
+    # source coordinates because place_sprite accepts absolute source centres.
+    wx0, wy0, wx1, wy1 = 91, 68, 121, 99
+    white_rgb = source[0, wy0:wy1, wx0:wx1]
+    white_bg = np.clip(plate[wy0:wy1, wx0:wx1], 0, 255).astype(np.uint8)
+    white_alpha = soft_alpha_from_difference(white_rgb, white_bg, 2.0, 34.0)
+    wyy, wxx = np.mgrid[wy0:wy1, wx0:wx1]
+    white_alpha *= np.exp(-(((wxx - 105.5) / 14.0) ** 4 +
+                            ((wyy - 82.0) / 13.0) ** 4))
+
+    return ((green_rgb, green_alpha, (167.2, 59.2)),
+            (white_rgb, white_alpha, (105.5, 81.8)))
+
+
+def white_trajectory(frame_index):
+    """Transferred rigid-body motion followed by a short frictional settle."""
+    start = np.array([105.5, 81.8], np.float32)
+    end = np.array([70.4, 94.9], np.float32)
+
+    # It travels at nearly constant velocity after the immediate frame-1 impact,
+    # with a smooth four-frame frictional settle at the end.
+    t = float(frame_index)
+    if t <= 17.0:
+        progress = t / 21.0
+    elif t < 25.0:
+        p0 = 17.0 / 21.0
+        u = (t - 17.0) / 8.0
+        ease = 1.0 - (1.0 - u) ** 3
+        progress = p0 + (1.0 - p0) * ease
+    else:
+        progress = 1.0
+    return start + (end - start) * progress
+
+
+def main():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3) and source.dtype == np.uint8
+
+    plate = build_clean_plate(source)
+    green, white = extract_sprites(source, plate)
+
+    red_center = np.array([140.2, 67.5], np.float32)
+    white_initial = np.array([105.5, 81.8], np.float32)
+    midpoint = (red_center + white_initial) * 0.5
+
+    rendered = []
+    for index in range(96):
+        frame = plate.copy()
+
+        # The intervention is at source frame 1.  At the midpoint the incoming
+        # green mallet is already in contact and transfers its momentum to white;
+        # it remains at the contact point while red is untouched.
+        place_sprite(frame, green[0], green[1], green[2], midpoint, scale=1.0)
+
+        white_center = white_trajectory(index)
+        depth_scale = 1.0 + 0.10 * np.clip(
+            (white_center[1] - white_initial[1]) / (94.9 - white_initial[1]),
+            0.0, 1.0)
+        # White is lower/nearer in the image, so it is composited after green at
+        # their initial contact.  This also gives the shadows correct depth order.
+        place_sprite(frame, white[0], white[1], white[2],
+                     white_center, scale=float(depth_scale))
+
+        rendered.append(np.clip(np.rint(frame), 0, 255).astype(np.uint8))
+
+    frames = np.stack(rendered, axis=0)
+    artifact_io.save(frames, CASE_DIR)
+
+
+if __name__ == "__main__":
+    main()
