+"""Generate the counterfactual dice video.
+
+The source is a locked-off, stylized render.  The white die first reaches the
+table on source frame 8.  Restitution does not affect its incoming fall, so
+frames 1--8 are retained exactly.  Thereafter, each vertical rebound is
+retimed and its launch velocity is multiplied by 1.3.  Under constant gravity
+this gives 1.3**2 times the original bounce height and 1.3 times its airtime.
+
+Rather than redraw the die, this script extracts its original rendered poses,
+removes the factual pose against a clean temporal plate, and composites those
+poses onto the counterfactual ballistic path.  This preserves the source's
+pixel-art lighting, changing facets, and pips while leaving the red die and all
+unaffected scene elements intact.
+"""
+
+from pathlib import Path
+
+import numpy as np
+from PIL import Image, ImageDraw, ImageFilter
+from scipy.ndimage import binary_dilation
+from scipy.spatial import ConvexHull
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+# Measured centers of the white die in the source frames.  The first 28 values
+# follow its visible rebounds; it is at rest from frame 29 onward.
+SRC_Y = np.array([
+    7.5, 12.6, 19.6, 28.3, 38.9, 50.8, 64.1, 77.9,
+    71.3, 65.7, 62.3, 60.0, 59.7, 60.9, 64.3, 68.6,
+    75.4, 75.6, 73.2, 72.4, 73.1, 75.3, 77.4, 76.3,
+    76.3, 77.6, 77.4, 77.5,
+] + [77.8] * (96 - 28), dtype=np.float32)
+
+SRC_X = np.array([
+    132.4, 132.5, 132.6, 132.6, 132.3, 132.4, 132.5, 133.1,
+    133.0, 132.8, 132.3, 132.4, 132.3, 132.5, 132.1, 132.6,
+    131.9, 131.9, 132.4, 132.3, 132.3, 132.4, 132.5, 131.7,
+    132.4, 132.0, 131.8, 132.2,
+] + [132.3] * (96 - 28), dtype=np.float32)

+def hull_mask(frame: np.ndarray, cx: float, cy: float) -> np.ndarray:
+    """Return a soft alpha mask around the rendered white die.
+
+    Bright, nearly neutral pixels identify its faces.  Their convex hull also
+    includes the pips and shaded facets.  Red pixels are explicitly protected
+    where the dice overlap so the red die is never accidentally transported.
+    """
+    h, w = frame.shape[:2]
+    x0, x1 = max(0, int(cx) - 15), min(w, int(cx) + 16)
+    y0, y1 = max(0, int(cy) - 16), min(h, int(cy) + 17)
+    patch = frame[y0:y1, x0:x1].astype(np.int16)
+
+    hi = patch.max(axis=2)
+    lo = patch.min(axis=2)
+    seed = (lo > 145) & ((hi - lo) < 66) & (patch.mean(axis=2) > 164)
+    yy, xx = np.where(seed)
+    mask_local = np.zeros(seed.shape, dtype=np.uint8)
+    if len(xx) >= 4:
+        pts = np.column_stack([xx, yy])
+        try:
+            hull = ConvexHull(pts)
+            poly = [tuple(map(int, p)) for p in pts[hull.vertices]]
+            pil = Image.new("L", (x1 - x0, y1 - y0), 0)
+            ImageDraw.Draw(pil).polygon(poly, fill=255)
+            mask_local = np.asarray(pil).copy()
+        except Exception:
+            mask_local[yy, xx] = 255
+    else:
+        mask_local[yy, xx] = 255
+
+    # One pixel of edge coverage catches the antialiased outline.  Preserve
+    # strongly red pixels in overlap frames (they belong to the other die).
+    grown = binary_dilation(mask_local > 0, iterations=1)
+    red = (patch[:, :, 0] > patch[:, :, 1] * 1.18) & \
+          (patch[:, :, 0] > patch[:, :, 2] * 1.14) & \
+          (patch[:, :, 0] > 115)
+    grown &= ~red
+    local_alpha = Image.fromarray((grown.astype(np.uint8) * 255), "L")
+    local_alpha = local_alpha.filter(ImageFilter.GaussianBlur(0.42))
+
+    alpha = np.zeros((h, w), dtype=np.uint8)
+    alpha[y0:y1, x0:x1] = np.asarray(local_alpha)
+    return alpha
+

+def clean_plate(source: np.ndarray) -> np.ndarray:
+    """Build an object-free plate from two unobstructed source moments."""
+    # At frame 1 the lower table is empty; at frame 96 the upper drop region is
+    # empty.  Selecting by row avoids the resting dice in the late frame and
+    # the entering dice in the early frame.
+    plate = source[0].copy()
+    plate[:65] = source[-1, :65]
+    return plate
+
+
+def adjusted_plate(plate: np.ndarray, current: np.ndarray, alpha: np.ndarray) -> np.ndarray:
+    """Match the plate to tiny frame-to-frame codec/lighting fluctuations."""
+    active = alpha > 16
+    ys, xs = np.where(active)
+    if not len(xs):
+        return plate
+    x0, x1 = max(0, xs.min() - 4), min(plate.shape[1], xs.max() + 5)
+    y0, y1 = max(0, ys.min() - 4), min(plate.shape[0], ys.max() + 5)
+    ring = np.ones((y1 - y0, x1 - x0), bool)
+    ring &= ~active[y0:y1, x0:x1]
+    # Exclude large object/color changes from the brightness estimate.
+    delta = current[y0:y1, x0:x1].astype(np.int16) - plate[y0:y1, x0:x1].astype(np.int16)
+    stable = ring & (np.max(np.abs(delta), axis=2) < 18)
+    if stable.sum() > 8:
+        offset = np.median(delta[stable], axis=0)
+    else:
+        offset = np.zeros(3)
+    return np.clip(plate.astype(np.float32) + offset, 0, 255).astype(np.uint8)
+
+
+def alpha_over(base: np.ndarray, layer: np.ndarray, alpha: np.ndarray) -> np.ndarray:
+    a = alpha.astype(np.float32)[..., None] / 255.0
+    return np.clip(base.astype(np.float32) * (1.0 - a) + layer.astype(np.float32) * a,
+                   0, 255).astype(np.uint8)
+
+
+def shift_image_and_alpha(frame: np.ndarray, alpha: np.ndarray,
+                          dx: float, dy: float) -> tuple[np.ndarray, np.ndarray]:
+    """Translate a sprite with subpixel filtering and transparent boundaries."""
+    # PIL's affine transform samples output at (x-dx, y-dy).
+    matrix = (1, 0, -dx, 0, 1, -dy)
+    rgb = Image.fromarray(frame).transform(
+        (frame.shape[1], frame.shape[0]), Image.Transform.AFFINE, matrix,
+        resample=Image.Resampling.BICUBIC, fillcolor=(0, 0, 0))
+    a = Image.fromarray(alpha, "L").transform(
+        (frame.shape[1], frame.shape[0]), Image.Transform.AFFINE, matrix,
+        resample=Image.Resampling.BILINEAR, fillcolor=0)
+    return np.asarray(rgb), np.asarray(a)
+
+
+def parabola(frame_index: int, start: int, end: int, height: float) -> float:
+    """Screen y offset above the tabletop for a ballistic bounce."""
+    u = (frame_index - start) / float(end - start)
+    return height * 4.0 * u * (1.0 - u)
+
+
+def trajectory_and_pose(i: int) -> tuple[float, int]:
+    """Counterfactual y center and nearest source-pose frame."""
+    ground = 77.8
+    if i <= 7:
+        return float(SRC_Y[i]), i
+    if i <= 20:
+        # Original main rebound: ~18 px high and 10 frames long.  1.3x launch
+        # speed yields ~30.4 px high and 13 frames long.
+        y = ground - parabola(i, 7, 20, 18.0 * 1.3 * 1.3)
+        pose = int(round(7 + (i - 7) * (10.0 / 13.0)))
+        return y, min(pose, 17)
+    if i <= 27:
+        # The source's second hop is about 5.4 px high and 5.4 frames long.
+        y = ground - parabola(i, 20, 27, 5.4 * 1.3 * 1.3)
+        pose = int(round(17 + (i - 20) * (6.0 / 7.0)))
+        return y, min(pose, 23)
+    if i <= 31:
+        # A final, visibly small settling hop.
+        y = ground - parabola(i, 27, 31, 1.45 * 1.3 * 1.3)
+        pose = int(round(23 + (i - 27) * (4.0 / 4.0)))
+        return y, min(pose, 27)
+    return float(SRC_Y[i]), i
+
+
+def generate() -> np.ndarray:
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3) and source.dtype == np.uint8
+    output = source.copy()
+    plate = clean_plate(source)
+
+    # Restitution cannot alter the pre-impact fall.  The edit begins just after
+    # the first table contact at frame 8 (zero-based index 7).
+    for i in range(8, 32):
+        target_y, pose_i = trajectory_and_pose(i)
+
+        # Remove only the visible white die from its factual location.  Its
+        # grounded shadow remains at the same x coordinate, as a real shadow
+        # should while the die rebounds vertically above it.
+        old_alpha = hull_mask(source[i], SRC_X[i], SRC_Y[i])
+        fill = adjusted_plate(plate, source[i], old_alpha)
+        frame = alpha_over(source[i], fill, old_alpha)
+
+        # Extract an original rendered orientation and place it on the edited
+        # vertical trajectory.  Horizontal momentum is unchanged.
+        sprite_alpha = hull_mask(source[pose_i], SRC_X[pose_i], SRC_Y[pose_i])
+        dx = 132.3 - float(SRC_X[pose_i])
+        dy = target_y - float(SRC_Y[pose_i])
+        sprite_rgb, moved_alpha = shift_image_and_alpha(
+            source[pose_i], sprite_alpha, dx, dy)
+        frame = alpha_over(frame, sprite_rgb, moved_alpha)
+        output[i] = frame
+
+    return output
+
+
+if __name__ == "__main__":
+    frames = generate()
+    artifact_io.save(frames, CASE_DIR)
+    print("Saved", frames.shape, frames.dtype, "to", CASE_DIR)
