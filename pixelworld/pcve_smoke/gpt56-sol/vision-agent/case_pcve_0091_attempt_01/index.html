+"""Generate the counterfactual pool-table video.
+
+The factual clip is a straight, glancing-free shot.  A yellow one-ball is
+inserted at the exact image-plane midpoint at frame 1.  Equal-ball impulse
+transfer makes the cue stop at the first contact, the one-ball travel to the
+target, and the target run along its original path a few frames early.
+"""
+from pathlib import Path
+
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def smoothstep(x):
+    x = np.clip(x, 0.0, 1.0)
+    return x * x * (3.0 - 2.0 * x)
+
+
+def make_clean_plate(src):
+    """Median plate, with the long-stationary factual cue ball inpainted."""
+    plate = np.median(src, axis=0).astype(np.float32)
+    h, w = plate.shape[:2]
+    yy, xx = np.mgrid[:h, :w]
+    cx, cy = 125.6, 70.4
+    rr = np.hypot(xx - cx, yy - cy)
+    ring = (rr >= 8.0) & (rr <= 15.0)
+    # Only regress from green cloth, not either ball or a dark compression edge.
+    green = (plate[..., 1] > plate[..., 0] + 24) & (plate[..., 1] > plate[..., 2] + 18)
+    use = ring & green
+    A = np.stack([np.ones(use.sum()), xx[use] - cx, yy[use] - cy], axis=1)
+    fill = rr < 8.5
+    B = np.stack([np.ones(fill.sum()), xx[fill] - cx, yy[fill] - cy], axis=1)
+    for ch in range(3):
+        coef = np.linalg.lstsq(A, plate[..., ch][use], rcond=None)[0]
+        plate[..., ch][fill] = B @ coef
+    return plate
+
+
+def source_cue_position(frame):
+    # Bright-component centroids from the first 15 source frames.  After the
+    # factual hit, the source cue is stationary at the last location.
+    x = np.array([85.43, 88.98, 92.50, 95.74, 99.00, 102.14, 105.07,
+                  108.00, 110.68, 113.42, 116.00, 118.29, 120.88,
+                  123.00, 124.39], dtype=np.float32)
+    y = np.array([93.69, 91.53, 89.50, 87.48, 85.86, 83.93, 82.19,
+                  80.50, 78.85, 77.34, 75.74, 74.50, 72.88,
+                  71.74, 70.67], dtype=np.float32)
+    if frame < len(x):
+        return np.array([x[frame], y[frame]], np.float32)
+    return np.array([125.6, 70.35], np.float32)
+
+
+P0 = np.array([127.0, 70.0], dtype=np.float32)
+SHOT_DIR = np.array([0.855, -0.519], dtype=np.float32)
+SHOT_DIR /= np.linalg.norm(SHOT_DIR)
+
+
+def source_target_position(t):
+    """Continuous fit to the source target, including its cushion rebound."""
+    # It stays put through frame 13, runs to the top cushion, dwells during
+    # impact, then returns on the same line more slowly.
+    if t <= 13.0:
+        s = 0.0
+    elif t <= 42.5:
+        s = min(48.0, 1.68 * (t - 13.0))
+    elif t <= 46.5:
+        s = 48.0
+    else:
+        s = max(0.0, 48.0 - 0.84 * (t - 46.5))
+    return P0 + SHOT_DIR * s
+
+
+def remove_source_object(frame, plate, center, radius):
+    """Feather a static clean plate over one factual ball and its shadow."""
+    h, w = frame.shape[:2]
+    yy, xx = np.mgrid[:h, :w]
+    dist = np.hypot(xx - center[0], yy - center[1])
+    alpha = 1.0 - smoothstep((dist - radius + 1.6) / 2.0)
+    alpha = alpha[..., None]
+
+    # Match the tiny source exposure changes from compression/GI, using cloth
+    # in a surrounding annulus.  Reject non-cloth pixels in both images.
+    ring = (dist > radius + 1.5) & (dist < radius + 5.5)
+    cloth = ring & (frame[..., 1] > frame[..., 0] + 20) & (frame[..., 1] > frame[..., 2] + 13)
+    cloth &= (plate[..., 1] > plate[..., 0] + 20) & (plate[..., 1] > plate[..., 2] + 13)
+    if cloth.sum() > 12:
+        offset = np.median(frame[cloth] - plate[cloth], axis=0)
+        offset = np.clip(offset, -5.0, 5.0)
+    else:
+        offset = np.zeros(3)
+    patch = np.clip(plate + offset, 0, 255)
+    frame[:] = frame * (1.0 - alpha) + patch * alpha
+
+
+def composite_ellipse(frame, cx, cy, rx, ry, color, opacity, softness=0.65):
+    """Small anti-aliased ellipse compositor used for shadows/markings."""
+    h, w = frame.shape[:2]
+    x0 = max(0, int(np.floor(cx - rx - 2)))
+    x1 = min(w, int(np.ceil(cx + rx + 2)))
+    y0 = max(0, int(np.floor(cy - ry - 2)))
+    y1 = min(h, int(np.ceil(cy + ry + 2)))
+    yy, xx = np.mgrid[y0:y1, x0:x1]
+    q = np.sqrt(((xx - cx) / rx) ** 2 + ((yy - cy) / ry) ** 2)
+    a = opacity * np.clip((1.0 + softness - q) / softness, 0.0, 1.0)
+    a = a[..., None]
+    frame[y0:y1, x0:x1] = frame[y0:y1, x0:x1] * (1 - a) + np.asarray(color) * a
+
+
+def draw_ball(frame, center, kind):
+    """Render a shaded, antialiased ball at source-video scale."""
+    x, y = map(float, center)
+    # Perspective in this fixed camera makes balls subtly smaller up-table.
+    r = 3.30 + 0.009 * (y - 55.0)
+    if kind == "cue":
+        base = np.array([224.0, 226.0, 212.0])
+    elif kind == "one":
+        base = np.array([235.0, 184.0, 18.0])
+    else:
+        base = np.array([25.0, 29.0, 23.0])
+
+    # Soft contact shadow, consistently down/right from the overhead light.
+    composite_ellipse(frame, x + 0.75, y + r * 0.73, r * 1.12, r * 0.54,
+                      (15, 28, 22), 0.34, softness=0.82)
+
+    h, w = frame.shape[:2]
+    x0, x1 = max(0, int(x-r-2)), min(w, int(x+r+3))
+    y0, y1 = max(0, int(y-r-2)), min(h, int(y+r+3))
+    yy, xx = np.mgrid[y0:y1, x0:x1]
+    dx, dy = (xx - x) / r, (yy - y) / r
+    rho2 = dx*dx + dy*dy
+    alpha = np.clip((1.08 - np.sqrt(rho2)) / 0.16, 0.0, 1.0)
+    z = np.sqrt(np.clip(1.0 - rho2, 0.0, 1.0))
+    light = np.array([-0.44, -0.62, 0.65])
+    light /= np.linalg.norm(light)
+    ndotl = dx*light[0] + dy*light[1] + z*light[2]
+    shade = 0.57 + 0.45 * np.clip(ndotl, 0.0, 1.0)
+    rgb = base[None, None, :] * shade[..., None]
+    # A compact broad highlight matches the soft source render.
+    spec = np.exp(-((dx + .35)**2 + (dy + .43)**2) / .085) * (50 if kind != "target" else 33)
+    rgb += spec[..., None]
+    rgb = np.clip(rgb, 0, 245)
+    a = alpha[..., None]
+    frame[y0:y1, x0:x1] = frame[y0:y1, x0:x1] * (1-a) + rgb*a
+
+    if kind == "one":
+        # The small white number medallion and dark '1' remain legible at 256p.
+        composite_ellipse(frame, x + .18, y + .12, 1.42, 1.28,
+                          (230, 226, 190), 0.96, softness=.32)
+        composite_ellipse(frame, x + .20, y + .13, .30, .72,
+                          (38, 35, 24), 0.92, softness=.34)
+
+
+def counterfactual_positions(t):
+    cue_start = np.array([85.43, 93.69], dtype=np.float32)
+    target_start = P0.copy()
+    one_start = (cue_start + target_start) * 0.5
+
+    diameter = 6.9
+    cue_stop = one_start - SHOT_DIR * diameter
+    one_stop = target_start - SHOT_DIR * diameter
+
+    # Source cue interpolation until it reaches first contact (about frame 4.6).
+    tc1 = 4.57
+    tc2 = 9.45
+    if t < tc1:
+        lo = int(np.floor(t))
+        hi = min(lo + 1, 14)
+        f = t - lo
+        cue = source_cue_position(lo) * (1-f) + source_cue_position(hi) * f
+    else:
+        cue = cue_stop
+
+    if t <= tc1:
+        one = one_start
+    elif t < tc2:
+        u = (t - tc1) / (tc2 - tc1)
+        # Very mild rolling friction, while still ending at exact contact.
+        u = 1.035*u - 0.035*u*u
+        one = one_start * (1-u) + one_stop * u
+    else:
+        one = one_stop
+
+    # The source target begins moving near 13; align that event to the second
+    # collision.  Its source path includes the physically affected rebound.
+    if t < tc2:
+        target = target_start
+    else:
+        target = source_target_position(t + (13.0 - tc2))
+    return cue, one, target
+
+
+def generate():
+    src = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    plate = make_clean_plate(src)
+    out = src.astype(np.float32).copy()
+
+    for i in range(len(out)):
+        # Remove the two factual balls (including their contact shadows).
+        remove_source_object(out[i], plate, source_cue_position(i), 6.8)
+        remove_source_object(out[i], plate, source_target_position(float(i)), 6.5)
+
+        cue, one, target = counterfactual_positions(float(i))
+        # Painter's order follows depth on the table; shadows and silhouettes
+        # naturally occlude at the two collision instants.
+        balls = [(target, "target"), (one, "one"), (cue, "cue")]
+        for pos, kind in sorted(balls, key=lambda item: item[0][1]):
+            draw_ball(out[i], pos, kind)
+
+    return np.clip(np.rint(out), 0, 255).astype(np.uint8)
+
+
+if __name__ == "__main__":
+    frames = generate()
+    artifact_io.save(frames, CASE_DIR)
+    print("saved", frames.shape, frames.dtype)
