+"""Generate the counterfactual video for the friction intervention.
+
+The source volleyball has an approximately constant-deceleration roll and comes
+to rest after 18 frames.  With three times the frictional acceleration, the
+same initial velocity reaches rest in one third of that time and covers one
+third of the original stopping distance.  The room is reconstructed from the
+static temporal observations; only the volleyball and its soft contact shadow
+are re-timed.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+# Image-space centers measured from the source roll.  The small perspective
+# curvature and apparent-size change are thereby retained rather than replaced
+# by a straight 2-D animation.
+SOURCE_CENTERS = np.array(
+    [
+        (79.0, 122.5),
+        (82.5, 116.0),
+        (86.0, 110.0),
+        (89.5, 105.5),
+        (93.0, 99.5),
+        (96.5, 94.0),
+        (99.5, 89.0),
+        (103.0, 84.5),
+        (105.0, 80.5),
+        (107.0, 77.0),
+        (109.0, 74.0),
+        (110.5, 71.5),
+        (112.0, 70.0),
+        (113.0, 68.5),
+        (113.5, 68.0),
+        (114.0, 67.5),
+        (115.0, 66.5),
+        (115.5, 66.5),
+        (115.5, 66.0),
+        (115.5, 66.0),
+    ],
+    dtype=np.float32,
+)
+
+
+def source_center(frame_index: int) -> np.ndarray:
+    if frame_index < len(SOURCE_CENTERS):
+        return SOURCE_CENTERS[frame_index]
+    return SOURCE_CENTERS[-1]
+
+
+def source_radius(frame_index: int) -> float:
+    """Radius covering the ball, antialiasing and close contact shadow."""
+    # Perspective makes the ball shrink from about 14 px to 8 px.
+    u = min(frame_index, 18) / 18.0
+    return 17.0 * (1.0 - u) + 11.5 * u
+
+
+def ball_exclusion_masks(height: int, width: int, count: int) -> np.ndarray:
+    yy, xx = np.mgrid[:height, :width]
+    masks = np.zeros((count, height, width), dtype=bool)
+    for i in range(count):
+        cx, cy = source_center(i)
+        r = source_radius(i)
+        # Slightly wider/lower ellipse includes the soft floor shadow.
+        masks[i] = ((xx - cx) / r) ** 2 + ((yy - (cy + 0.7)) / (r * 0.92)) ** 2 <= 1.0
+    return masks
+
+
+def make_clean_plate(source: np.ndarray, exclusions: np.ndarray) -> np.ndarray:
+    """Robust static-room image assembled only from ball-free samples."""
+    count, height, width, _ = source.shape
+    plate = np.empty((height, width, 3), dtype=np.float32)
+    # There are at least 12 clean observations even beneath the final resting
+    # location, and many more elsewhere.  Channel-wise masked medians preserve
+    # rug and grout detail under the moving object.
+    for y in range(height):
+        values = source[:, y].astype(np.float32)
+        valid = ~exclusions[:, y]
+        for x in range(width):
+            samples = values[valid[:, x], x]
+            plate[y, x] = np.median(samples, axis=0)
+    return plate
+
+
+def clean_background_for_frame(
+    frame: np.ndarray, plate: np.ndarray, old_mask: np.ndarray
+) -> np.ndarray:
+    """Erase the factual ball while retaining per-frame lighting variation."""
+    image = frame.astype(np.float32)
+    valid = (~old_mask).astype(np.float32)
+    delta = (image - plate) * valid[..., None]
+
+    # A normalized low-frequency correction reproduces the source renderer's
+    # very subtle temporal exposure changes without leaking ball colors in.
+    weight = cv2.GaussianBlur(valid, (0, 0), 10.0)
+    smooth_delta = np.empty_like(delta)
+    for c in range(3):
+        numer = cv2.GaussianBlur(delta[..., c], (0, 0), 10.0)
+        smooth_delta[..., c] = numer / np.maximum(weight, 0.05)
+    replacement = plate + smooth_delta
+
+    # Feather only the outside two pixels.  The broad interior is a clean,
+    # detailed temporal plate rather than a spatially blurred inpaint.
+    soft = cv2.GaussianBlur(old_mask.astype(np.float32), (0, 0), 0.8)
+    soft = np.clip(soft * 1.35, 0.0, 1.0)[..., None]
+    return image * (1.0 - soft) + replacement * soft
+
+
+def foreground_layer(
+    source_frame: np.ndarray,
+    plate: np.ndarray,
+    center: np.ndarray,
+    radius: float,
+) -> tuple[np.ndarray, np.ndarray]:
+    """Extract an additive RGBA-like ball layer from a source frame."""
+    src = source_frame.astype(np.float32)
+    difference = src - plate
+    magnitude = np.max(np.abs(difference), axis=2)
+    height, width = magnitude.shape
+    yy, xx = np.mgrid[:height, :width]
+    cx, cy = center
+    support = (((xx - cx) / radius) ** 2 + ((yy - (cy + 0.7)) / (radius * 0.92)) ** 2) <= 1.0
+
+    # Select genuine changing pixels, bridge antialiased gaps, and feather the
+    # silhouette.  The threshold is above the static renderer variation while
+    # retaining the soft gray contact shadow.
+    core = ((magnitude > 6.0) & support).astype(np.uint8)
+    core = cv2.morphologyEx(core, cv2.MORPH_CLOSE, np.ones((3, 3), np.uint8))
+    core = cv2.dilate(core, np.ones((3, 3), np.uint8), iterations=1)
+    alpha = cv2.GaussianBlur(core.astype(np.float32), (0, 0), 0.65)
+    alpha *= support.astype(np.float32)
+    alpha = np.clip(alpha, 0.0, 1.0)
+    difference *= alpha[..., None]
+    return difference, alpha
+
+
+def warp_layer(
+    difference: np.ndarray, alpha: np.ndarray, shift: np.ndarray
+) -> tuple[np.ndarray, np.ndarray]:
+    height, width = alpha.shape
+    matrix = np.array([[1.0, 0.0, shift[0]], [0.0, 1.0, shift[1]]], dtype=np.float32)
+    moved_difference = cv2.warpAffine(
+        difference, matrix, (width, height), flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT, borderValue=0,
+    )
+    moved_alpha = cv2.warpAffine(
+        alpha, matrix, (width, height), flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT, borderValue=0,
+    )
+    return moved_difference, moved_alpha
+
+
+def render() -> np.ndarray:
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    count, height, width, channels = source.shape
+    assert (count, height, width, channels) == (96, 144, 256, 3)
+
+    exclusions = ball_exclusion_masks(height, width, count)
+    plate = make_clean_plate(source, exclusions)
+
+    # Only the early source appearances are needed: 3x friction stops the ball
+    # at original path progress 1/3, corresponding to source time ~3.3 frames.
+    layers = []
+    for i in range(5):
+        layers.append(
+            foreground_layer(source[i], plate, source_center(i), source_radius(i))
+        )
+
+    output = np.empty_like(source)
+    factual_stop_frames = 18.0
+    counterfactual_stop = factual_stop_frames / 3.0
+
+    for frame_index in range(count):
+        background = clean_background_for_frame(
+            source[frame_index], plate, exclusions[frame_index]
+        )
+
+        if frame_index == 0:
+            # At the instant the coefficient changes, position and appearance
+            # are continuous and exactly equal to the factual first frame.
+            output[frame_index] = source[frame_index]
+            continue
+
+        # q is traveled distance as a fraction of the factual total stopping
+        # distance.  a' = 3a gives q=2u-3u^2 until rest, then q=1/3.
+        if frame_index < counterfactual_stop:
+            u = frame_index / factual_stop_frames
+            q = 2.0 * u - 3.0 * u * u
+        else:
+            q = 1.0 / 3.0
+
+        # Invert the factual constant-deceleration distance relation
+        # q=2s-s^2 to select the matching perspective, rotation, and shadow.
+        factual_u = 1.0 - np.sqrt(max(0.0, 1.0 - q))
+        appearance_time = factual_u * factual_stop_frames
+        lo = min(int(np.floor(appearance_time)), 3)
+        hi = lo + 1
+        mix = float(appearance_time - lo)
+        target_center = (1.0 - mix) * source_center(lo) + mix * source_center(hi)
+
+        d0, a0 = warp_layer(layers[lo][0], layers[lo][1], target_center - source_center(lo))
+        d1, a1 = warp_layer(layers[hi][0], layers[hi][1], target_center - source_center(hi))
+        difference = (1.0 - mix) * d0 + mix * d1
+        alpha = np.clip((1.0 - mix) * a0 + mix * a1, 0.0, 1.0)
+
+        # Additive foreground transfer naturally adapts the source ball edge to
+        # the current clean floor/rug plate while preserving its exact colors.
+        composed = background + difference
+        # Suppress any minute extraction noise outside the extracted support.
+        composed = background * (1.0 - alpha[..., None]) + composed * alpha[..., None]
+        output[frame_index] = np.clip(np.rint(composed), 0, 255).astype(np.uint8)
+
+    return output
+
+
+if __name__ == "__main__":
+    frames = render()
+    artifact_io.save(frames, CASE_DIR)
+    print("saved", frames.shape, frames.dtype, "to", CASE_DIR)
