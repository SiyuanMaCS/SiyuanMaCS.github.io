+from pathlib import Path
+
+import cv2
+import numpy as np
+from PIL import Image, ImageDraw, ImageFilter
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+N, H, W = 96, 144, 256
+
+
+def clean_background(src):
+    """Recover the fixed set by taking a temporal median while both dice move."""
+    # During these frames the dice visit each location only briefly.  A temporal
+    # median therefore removes them (and their moving contact shadows) while
+    # retaining the fine, antialiased table artwork.
+    return np.median(src[:18].astype(np.float32), axis=0).round().astype(np.uint8)
+
+
+def sprite(src, bg, frame_no, center, kind, half=18):
+    """Extract one die as an antialiased RGBA sprite from a source frame."""
+    cx, cy = center
+    x0, y0 = int(cx - half), int(cy - half)
+    x1, y1 = int(cx + half + 1), int(cy + half + 1)
+    patch = src[frame_no, max(0, y0):min(H, y1), max(0, x0):min(W, x1)].copy()
+    back = bg[max(0, y0):min(H, y1), max(0, x0):min(W, x1)]
+    r, g, b = [patch[..., k].astype(np.int16) for k in range(3)]
+    delta = np.max(np.abs(patch.astype(np.int16) - back.astype(np.int16)), axis=2)
+
+    yy, xx = np.indices(r.shape)
+    local_cx, local_cy = cx - max(0, x0), cy - max(0, y0)
+    near = (xx - local_cx) ** 2 + (yy - local_cy) ** 2 < 15 ** 2
+    if kind == "red":
+        key = (r > 105) & (r > g * 1.16) & (r > b * 1.06) & (delta > 14) & near
+    else:
+        key = (r > 135) & (g > 130) & (b > 118) & (delta > 16) & near
+
+    # Use the convex silhouette of the colored faces.  This includes dark pips
+    # and face outlines without picking up similarly colored table markings.
+    points = np.column_stack(np.nonzero(key))[:, ::-1].astype(np.int32)
+    matte = np.zeros(r.shape, np.uint8)
+    if len(points) < 3:
+        raise RuntimeError(f"could not extract {kind} sprite at frame {frame_no}")
+    hull = cv2.convexHull(points)
+    cv2.fillConvexPoly(matte, hull, 255)
+    matte = cv2.dilate(matte, np.ones((3, 3), np.uint8), iterations=1)
+    matte = cv2.GaussianBlur(matte, (3, 3), 0.55)
+
+    rgba = np.dstack([patch, matte])
+    # Anchor coordinates within this (possibly edge-clipped) patch.
+    anchor = (float(local_cx), float(local_cy))
+    return Image.fromarray(rgba, "RGBA"), anchor
+
+
+def paste_sprite(canvas, item, center):
+    image, anchor = item
+    x = int(round(center[0] - anchor[0]))
+    y = int(round(center[1] - anchor[1]))
+    canvas.alpha_composite(image, (x, y))
+
+
+def interp(keys, i):
+    """Piecewise-linear interpolation through (frame, x, y) keyframes."""
+    if i <= keys[0][0]:
+        return float(keys[0][1]), float(keys[0][2])
+    if i >= keys[-1][0]:
+        return float(keys[-1][1]), float(keys[-1][2])
+    for a, b in zip(keys[:-1], keys[1:]):
+        if a[0] <= i <= b[0]:
+            t = (i - a[0]) / (b[0] - a[0])
+            return (a[1] * (1 - t) + b[1] * t,
+                    a[2] * (1 - t) + b[2] * t)
+    raise AssertionError
+
+
+def add_shadow(canvas, center, floor_y, opacity_scale=1.0):
+    """Soft contact/projection shadow that strengthens near the mat."""
+    x, y = center
+    distance = max(0.0, floor_y - y)
+    strength = max(0.0, 1.0 - distance / 31.0) * opacity_scale
+    if strength <= 0:
+        return
+    layer = Image.new("RGBA", (W, H), (0, 0, 0, 0))
+    draw = ImageDraw.Draw(layer)
+    rx = 7.8 + min(distance, 24) * 0.10
+    ry = 2.5 + min(distance, 24) * 0.035
+    sy = floor_y + 6
+    draw.ellipse((x - rx, sy - ry, x + rx, sy + ry), fill=(26, 43, 39, int(55 * strength)))
+    layer = layer.filter(ImageFilter.GaussianBlur(2.0))
+    canvas.alpha_composite(layer)
+
+
+def render():
+    src = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    bg = clean_background(src)
+
+    # Clean, non-overlapping source poses preserve the original game's exact die
+    # design and lighting.  Cycling them supplies the existing tumbling motion.
+    red_centers = [(123, 23), (123, 32), (123, 43), (122, 55), (123, 67)]
+    white_centers = [(132, 13), (134, 22), (138, 34), (139, 46)]
+    red = [sprite(src, bg, k, red_centers[k], "red") for k in range(5)]
+    white = [sprite(src, bg, k, white_centers[k], "white") for k in range(4)]
+
+    # The red die starts in the same place but with 3x its measured downward
+    # speed.  It hits at frame 3 instead of frame 6, rebounds, and strikes the
+    # still-descending white die obliquely around frame 4.  Equal-mass impact
+    # redirects the red die left and launches the white die up and right.
+    red_keys = [
+        (0, 123, 23), (1, 123, 48), (2, 123, 75), (3, 123, 86),
+        (4, 121, 69), (5, 119, 55), (6, 117, 58), (7, 115, 70),
+        (8, 114, 82), (9, 114, 86), (10, 113, 78), (12, 112, 72),
+        (14, 112, 76), (16, 112, 86), (18, 111, 82), (20, 111, 86),
+        (23, 111, 84), (26, 111, 86), (95, 111, 86),
+    ]
+    white_keys = [
+        (0, 132, 13), (1, 135, 22), (2, 138, 33), (3, 140, 45),
+        (4, 142, 56), (5, 145, 40), (7, 150, 16), (9, 155, -3),
+        (12, 162, -20), (14, 166, -24), (16, 169, -22),
+        (19, 174, -6), (22, 178, 23), (24, 180, 49),
+        (26, 182, 79), (27, 183, 84), (29, 185, 63),
+        (32, 187, 47), (34, 188, 44), (36, 189, 50),
+        (39, 190, 73), (40, 191, 84), (42, 192, 71),
+        (45, 193, 66), (49, 194, 84), (52, 194, 76),
+        (55, 194, 84), (58, 194, 81), (61, 194, 84),
+        (64, 194, 83), (68, 194, 84), (95, 194, 84),
+    ]
+
+    frames = []
+    for i in range(N):
+        if i == 0:
+            # At t=0 the intervention changes velocity, not position or pose.
+            frames.append(src[0].copy())
+            continue
+
+        # Carry across the source's very subtle exposure drift while keeping the
+        # recovered set free of old dice trails.
+        sample = src[min(i, 17)].astype(np.int16)
+        drift_area = np.ones((H, W), bool)
+        drift_area[:112, 86:172] = False
+        drift = np.median(sample[drift_area] - src[0].astype(np.int16)[drift_area], axis=0)
+        backdrop = np.clip(bg.astype(np.int16) + drift[None, None, :], 0, 255).astype(np.uint8)
+        canvas = Image.fromarray(backdrop, "RGB").convert("RGBA")
+
+        rp = interp(red_keys, i)
+        wp = interp(white_keys, i)
+        add_shadow(canvas, rp, 86, 1.0)
+        add_shadow(canvas, wp, 84, 0.9)
+
+        # Paint the upper object first.  Near collision this naturally places the
+        # lower/front red die over the rear white die, matching source occlusion.
+        w_index = min(i, 47) % len(white)
+        r_index = min(i, 24) % len(red)
+        paste_sprite(canvas, white[w_index], wp)
+        paste_sprite(canvas, red[r_index], rp)
+        frames.append(np.asarray(canvas.convert("RGB"), dtype=np.uint8))
+
+    return np.stack(frames, axis=0)
+
+
+if __name__ == "__main__":
+    output = render()
+    assert output.shape == (N, H, W, 3) and output.dtype == np.uint8
+    artifact_io.save(output, CASE_DIR)
