+"""Generate the requested counterfactual PixelWorld video.
+
+The source contains three small rigid objects on the table: a red cola can,
+a yellow cup, and a green carton.  The can's source trajectory was used to
+fit a simple 1-D rigid-body model (Coulomb-like constant deceleration plus
+pairwise impacts).  Only the fitted initial velocity is halved here.  This
+keeps the source friction, masses, restitution, and collision spacing, and
+therefore naturally changes the downstream interactions: the weakened can
+moves the cup, but the cup stops short of the green carton.
+
+Rendering is done entirely from source pixels.  A clean background is
+estimated temporally, source objects are removed from each original-time
+background frame, and soft source-derived sprites are composited at the
+counterfactual positions.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def connected_centroid(mask: np.ndarray, previous_x: float, min_area: int = 8) -> float:
+    """Centroid of the component spatially nearest the previous detection."""
+    count, _, stats, centers = cv2.connectedComponentsWithStats(
+        mask.astype(np.uint8), 8
+    )
+    candidates = []
+    for label in range(1, count):
+        if stats[label, cv2.CC_STAT_AREA] >= min_area:
+            x = float(centers[label, 0])
+            candidates.append((abs(x - previous_x), -int(stats[label, 4]), x))
+    return min(candidates)[2] if candidates else previous_x
+
+
+def track_objects(source: np.ndarray) -> np.ndarray:
+    """Track x centers of can, cup, and carton using their source colors."""
+    n, h, w, _ = source.shape
+    y = np.indices((h, w))[0]
+    tracks = np.zeros((n, 3), dtype=np.float32)
+    previous = [28.5, 99.0, 169.5]
+
+    for frame_index, frame_u8 in enumerate(source):
+        frame = frame_u8.astype(np.int16)
+        r, g, b = frame.transpose(2, 0, 1)
+        masks = [
+            # Red cola can.
+            (r > 100)
+            & (r > g * 1.18)
+            & (r > b * 1.15)
+            & (g < 150)
+            & (y > 60)
+            & (y < 86),
+            # Yellow band and body of the cup.
+            (r > 120)
+            & (g > 100)
+            & (b < 105)
+            & (r > 1.25 * b)
+            & (g > 1.20 * b)
+            & (y > 60)
+            & (y < 86),
+            # Green carton.
+            (g > 50)
+            & (g > r * 1.10)
+            & (g > b * 1.05)
+            & (r < 150)
+            & (y > 60)
+            & (y < 86),
+        ]
+        for object_index, mask in enumerate(masks):
+            previous[object_index] = connected_centroid(
+                mask, previous[object_index]
+            )
+            tracks[frame_index, object_index] = previous[object_index]
+    return tracks
+
+
+def estimate_clean_background(source: np.ndarray, tracks: np.ndarray) -> np.ndarray:
+    """Compute an object-free background from unaffected pixels over time."""
+    n, h, w, _ = source.shape
+    valid = np.ones((n, h, w), dtype=bool)
+    vertical_ranges = [(60, 86), (51, 86), (59, 86)]
+    radii = [9, 10, 8]
+
+    for frame_index in range(n):
+        for center_x, (y1, y2), radius in zip(
+            tracks[frame_index], vertical_ranges, radii
+        ):
+            x1 = max(0, int(center_x - radius))
+            x2 = min(w, int(center_x + radius + 1))
+            valid[frame_index, y1:y2, x1:x2] = False
+
+    samples = source.astype(np.float32)
+    samples[~valid] = np.nan
+    with np.errstate(all="ignore"):
+        background = np.nanmedian(samples, axis=0)
+    fallback = np.median(source, axis=0)
+    background = np.where(np.isnan(background), fallback, background)
+    return np.clip(background, 0, 255).astype(np.uint8)
+
+
+def make_sprites(
+    source: np.ndarray, background: np.ndarray, tracks: np.ndarray
+):
+    """Extract soft-edged, source-faithful sprites from source frame 1."""
+    # Crops include each object's contact shadow and, for the cup, its straw.
+    specifications = [
+        (20, 59, 38, 87),
+        (89, 50, 110, 87),
+        (161, 58, 179, 87),
+    ]
+    sprites = []
+    for object_index, (x1, y1, x2, y2) in enumerate(specifications):
+        rgb = source[0, y1:y2, x1:x2].copy()
+        clean = background[y1:y2, x1:x2]
+        difference = np.max(
+            np.abs(rgb.astype(np.int16) - clean.astype(np.int16)), axis=2
+        ).astype(np.float32)
+
+        # Difference matting preserves source antialiasing.  A closed core
+        # prevents small highlight regions inside an object becoming holes.
+        alpha = np.clip((difference - 3.0) / 14.0, 0.0, 1.0)
+        core = (difference > 9).astype(np.uint8)
+        core = cv2.morphologyEx(core, cv2.MORPH_CLOSE, np.ones((3, 3), np.uint8))
+        soft_core = cv2.GaussianBlur(core.astype(np.float32), (3, 3), 0.6)
+        alpha = np.maximum(alpha, soft_core * 0.90)
+        alpha = np.clip(alpha, 0.0, 1.0)
+        sprites.append(
+            {
+                "rgb": rgb,
+                "alpha": alpha,
+                "origin": (x1, y1),
+                "anchor_x": float(tracks[0, object_index]),
+            }
+        )
+    return sprites
+
+
+def simulate_counterfactual(initial_positions: np.ndarray, frame_count: int) -> np.ndarray:
+    """Simulate the source-fitted 1-D system with half can initial speed."""
+    # Parameters obtained by fitting frames 1--36 of this source.  Units are
+    # source pixels and source-frame intervals.  The source initial speed is
+    # 19.2414372 px/frame; this task changes only that value to 50 percent.
+    velocity = np.array([19.2414372 * 0.5, 0.0, 0.0], dtype=np.float64)
+    deceleration = np.array([0.228528261, 0.217984397, 0.215979001])
+    mass = np.array([1.0, 0.890144889, 1.00739310])
+    restitution = [0.0308511015, 0.0]
+    contact_distance = [10.0224984, 9.80537234]
+
+    positions = initial_positions.astype(np.float64).copy()
+    trajectory = np.empty((frame_count, 3), dtype=np.float32)
+    trajectory[0] = positions
+
+    substeps = 100
+    dt = 1.0 / substeps
+    for frame_index in range(1, frame_count):
+        for _ in range(substeps):
+            velocity = np.maximum(0.0, velocity - deceleration * dt)
+            positions += velocity * dt
+
+            for pair_index in range(2):
+                left = pair_index
+                right = pair_index + 1
+                gap = positions[right] - positions[left]
+                if (
+                    gap <= contact_distance[pair_index]
+                    and velocity[left] > velocity[right]
+                ):
+                    # Resolve tiny overlap according to inverse masses.
+                    overlap = contact_distance[pair_index] - gap
+                    total_mass = mass[left] + mass[right]
+                    positions[left] -= overlap * mass[right] / total_mass
+                    positions[right] += overlap * mass[left] / total_mass
+
+                    relative_speed = velocity[left] - velocity[right]
+                    impulse = (
+                        (1.0 + restitution[pair_index])
+                        * relative_speed
+                        / (1.0 / mass[left] + 1.0 / mass[right])
+                    )
+                    velocity[left] -= impulse / mass[left]
+                    velocity[right] += impulse / mass[right]
+        trajectory[frame_index] = positions
+    return trajectory
+
+
+def erase_source_objects(
+    frame: np.ndarray,
+    background: np.ndarray,
+    source_positions: np.ndarray,
+) -> np.ndarray:
+    """Remove factual objects while retaining the source-time background."""
+    h, w = frame.shape[:2]
+    mask = np.zeros((h, w), dtype=np.uint8)
+    vertical_ranges = [(58, 88), (49, 88), (57, 88)]
+    radii = [11, 12, 10]
+    for center_x, (y1, y2), radius in zip(
+        source_positions, vertical_ranges, radii
+    ):
+        cv2.rectangle(
+            mask,
+            (max(0, int(center_x - radius)), y1),
+            (min(w - 1, int(center_x + radius)), y2),
+            255,
+            -1,
+        )
+    opacity = cv2.GaussianBlur(mask, (5, 5), 1.0).astype(np.float32) / 255.0
+    result = (
+        frame.astype(np.float32) * (1.0 - opacity[..., None])
+        + background.astype(np.float32) * opacity[..., None]
+    )
+    return np.clip(result, 0, 255).astype(np.uint8)
+
+
+def composite_sprite(
+    frame: np.ndarray, sprite: dict, target_center_x: float
+) -> None:
+    """Alpha composite one sprite in place, clipping safely to the frame."""
+    rgb = sprite["rgb"]
+    alpha = sprite["alpha"]
+    x0, y0 = sprite["origin"]
+    x0 += int(round(target_center_x - sprite["anchor_x"]))
+
+    frame_h, frame_w = frame.shape[:2]
+    sprite_h, sprite_w = rgb.shape[:2]
+    sx1, sy1 = max(0, -x0), max(0, -y0)
+    sx2, sy2 = min(sprite_w, frame_w - x0), min(sprite_h, frame_h - y0)
+    if sx2 <= sx1 or sy2 <= sy1:
+        return
+
+    target = frame[y0 + sy1 : y0 + sy2, x0 + sx1 : x0 + sx2]
+    color = rgb[sy1:sy2, sx1:sx2].astype(np.float32)
+    opacity = alpha[sy1:sy2, sx1:sx2, None]
+    target[:] = np.clip(
+        color * opacity + target.astype(np.float32) * (1.0 - opacity), 0, 255
+    ).astype(np.uint8)
+
+
+def generate() -> np.ndarray:
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    if source.shape != (96, 144, 256, 3) or source.dtype != np.uint8:
+        raise ValueError(f"Unexpected source tensor: {source.shape}, {source.dtype}")
+
+    source_tracks = track_objects(source)
+    background = estimate_clean_background(source, source_tracks)
+    sprites = make_sprites(source, background, source_tracks)
+    counterfactual = simulate_counterfactual(source_tracks[0], len(source))
+
+    output = np.empty_like(source)
+    for frame_index in range(len(source)):
+        # Use the untouched source frame at the same absolute time so camera,
+        # lighting, and all unaffected scene details follow the factual video.
+        frame = erase_source_objects(
+            source[frame_index], background, source_tracks[frame_index]
+        )
+        # Left-to-right order is also the correct local occlusion order.
+        for object_index, sprite in enumerate(sprites):
+            composite_sprite(frame, sprite, counterfactual[frame_index, object_index])
+        output[frame_index] = frame
+
+    # Required output API call.  This writes pixels.npz, frame PNGs,
+    # inspection sheets, and the HTML player into the case directory.
+    artifact_io.save(output, CASE_DIR)
+
+    print("source initial can speed: 19.2414 px/frame")
+    print("edited initial can speed:  9.6207 px/frame (0.500x)")
+    print("counterfactual final centers:", counterfactual[-1].round(3).tolist())
+    return output
+
+
+if __name__ == "__main__":
+    generate()
