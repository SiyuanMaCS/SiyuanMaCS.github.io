+"""Generate the 4x-cola-mass counterfactual video.
+
+Only benchmark-provided pixels are used.  The rigid objects are extracted from
+frame 1, the source's equal-mass motion is fitted with a small 1-D impulse
+model, and that model is replayed with cola mass=4.  Layered, antialiased
+cutouts preserve the rendered appearance while the clean plate is reconstructed
+from source frames in which each table location is unobstructed.
+"""
+from pathlib import Path
+import cv2
+import numpy as np
+from artifact_io import save
+
+CASE_DIR = Path(__file__).resolve().parent
+N, H, W = 96, 144, 256
+
+
+def soft_cutout(image, box, kind):
+    """Return premultiplied-looking RGB patch and a soft foreground alpha."""
+    x0, y0, x1, y1 = box
+    patch = image[y0:y1, x0:x1].copy()
+    yy, xx = np.indices(patch.shape[:2])
+    rgb = patch.astype(np.float32)
+    # Broad object cues plus conservative geometric support.  Geometry keeps
+    # white packaging; chroma cues retain antialiased colored edge pixels.
+    if kind == "cola":
+        geom = ((xx >= 3) & (xx <= 13) & (yy >= 5) & (yy <= 25))
+        geom |= ((xx >= 4) & (xx <= 12) & (yy >= 3) & (yy <= 26))
+        chroma = (rgb[..., 0] > rgb[..., 1] * 1.18) & (rgb[..., 0] > rgb[..., 2] * 1.12)
+    elif kind == "cup":
+        # Include lid and straw; taper the cup toward its base.
+        center = 9.5
+        half = np.where(yy < 7, 4.5, np.maximum(5.2, 8.0 - .095 * np.maximum(yy-7, 0)))
+        geom = (yy >= 1) & (yy <= 29) & (np.abs(xx-center) <= half)
+        geom |= (yy <= 9) & (xx >= 7) & (xx <= 13)
+        chroma = (rgb[..., 0] > rgb[..., 2] * 1.15) & (rgb[..., 1] > rgb[..., 2] * 1.10)
+    else:
+        # Carton has a small green cap, tapered shoulders, and rectangular body.
+        geom = ((yy >= 7) & (yy <= 28) & (xx >= 3) & (xx <= 13))
+        geom |= ((yy >= 3) & (yy <= 9) & (xx >= 5) & (xx <= 12))
+        geom |= ((yy >= 1) & (yy <= 5) & (xx >= 7) & (xx <= 12))
+        chroma = (rgb[..., 1] > rgb[..., 0] * 1.05) & (rgb[..., 1] > rgb[..., 2] * 1.01)
+    mask = geom.astype(np.uint8) * 255
+    # Preserve colored antialiasing just outside conservative geometry.
+    mask[np.logical_and(chroma, cv2.dilate(mask, np.ones((3,3),np.uint8)) > 0)] = 255
+    alpha = cv2.GaussianBlur(mask.astype(np.float32) / 255.0, (0, 0), .42)
+    alpha[alpha < .025] = 0
+    return patch, alpha
+
+
+def build_clean_plate(src):
+    """Static scene plate: source is static except for the three objects."""
+    # The final frame is clean on the left/centre, while frame 1 is clean on the
+    # far right.  A feathered seam lies in an always-unoccupied gap.
+    a, b = src[-1].astype(np.float32), src[0].astype(np.float32)
+    blend = np.clip((np.arange(W, dtype=np.float32) - 184.0) / 10.0, 0, 1)[None, :, None]
+    plate = a * (1-blend) + b * blend
+    return np.clip(plate + .5, 0, 255).astype(np.uint8)
+
+
+def simulate_heavy():
+    """1-D rigid-body replay, source-calibrated, with the cola at mass 4."""
+    # Frame-1 visual centres (cola, cup, carton), collision half-widths, and
+    # source-calibrated initial velocity/friction (pixels per frame).
+    x = np.array([28.53, 98.90, 169.50], np.float64)
+    v = np.array([20.0, 0.0, 0.0], np.float64)
+    mass = np.array([4.0, 1.0, 1.0], np.float64)
+    radius = np.array([5.5, 7.5, 6.0], np.float64)
+    out = np.empty((N, 3), np.float64)
+    out[0] = x
+    substeps = 100
+    for frame in range(1, N):
+        for _ in range(substeps):
+            # Coulomb table friction, fitted to factual trajectories.
+            dv = 0.22 / substeps
+            v -= np.minimum(np.abs(v), dv) * np.sign(v)
+            x += v / substeps
+            # Sequential perfectly inelastic impulses.  Four-times larger cola
+            # momentum makes both lighter packages leave the right table edge.
+            for j in (0, 1):
+                overlap = radius[j] + radius[j+1] - (x[j+1] - x[j])
+                if overlap > 0:
+                    invsum = 1/mass[j] + 1/mass[j+1]
+                    x[j] -= overlap * mass[j+1] / (mass[j] + mass[j+1])
+                    x[j+1] += overlap * mass[j] / (mass[j] + mass[j+1])
+                    rel = v[j] - v[j+1]
+                    if rel > 0:
+                        impulse = rel / invsum
+                        v[j] -= impulse / mass[j]
+                        v[j+1] += impulse / mass[j+1]
+        out[frame] = x
+    return out
+
+
+def composite(dst, patch, alpha, x, y, angle=0.0):
+    """Subpixel paste, clipped at image boundaries."""
+    h, w = alpha.shape
+    # Affine map the cutout at subpixel location.  Rotation remains zero while
+    # supported on the tabletop; retained for deterministic extensibility.
+    canvas_rgb = np.zeros((H, W, 3), np.float32)
+    canvas_a = np.zeros((H, W), np.float32)
+    ix, iy = int(np.floor(x)), int(np.floor(y))
+    fx, fy = x-ix, y-iy
+    M = np.array([[1, 0, ix+fx], [0, 1, iy+fy]], np.float32)
+    canvas_rgb = cv2.warpAffine(patch.astype(np.float32), M, (W,H), flags=cv2.INTER_LINEAR,
+                                borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+    canvas_a = cv2.warpAffine(alpha, M, (W,H), flags=cv2.INTER_LINEAR,
+                              borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+    a = canvas_a[...,None]
+    dst[:] = np.clip(canvas_rgb*a + dst.astype(np.float32)*(1-a), 0, 255).astype(np.uint8)
+
+
+def main():
+    src = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert src.shape == (N,H,W,3) and src.dtype == np.uint8
+    plate = build_clean_plate(src)
+    # Tight crops in frame 1. Anchors are crop offsets from measured centres.
+    specs = [
+        ("cola",   (20,59,38,87), (28.53-20, 72.54-59)),
+        ("cup",    (89,55,110,87),(98.90-89, 73.0-55)),
+        ("carton", (161,57,179,87),(169.50-161, 73.0-57)),
+    ]
+    sprites=[]
+    for kind, box, anchor in specs:
+        patch, alpha = soft_cutout(src[0], box, kind)
+        sprites.append((patch,alpha,anchor,box))
+    centers=simulate_heavy()
+    frames=np.empty_like(src)
+    for t in range(N):
+        frame=plate.copy()
+        # Painter order left to right matches collision occlusion in source.
+        for j,(patch,alpha,anchor,box) in enumerate(sprites):
+            cx=centers[t,j]
+            x=cx-anchor[0]
+            y=box[1]  # all three remain upright on the tabletop until offscreen
+            if x < W+patch.shape[1] and x+patch.shape[1] > 0:
+                composite(frame,patch,alpha,x,y)
+        frames[t]=frame
+    # Preserve exact factual first frame: mass changes dynamics, not appearance,
+    # and at t=0 no causal displacement has occurred.
+    frames[0]=src[0]
+    save(frames, CASE_DIR)
+
+if __name__ == "__main__":
+    main()

Only benchmark-provided pixels are used.  The rigid objects are extracted from
frame 1, the source's equal-mass motion is fitted with a small 1-D impulse

from pathlib import Path
import cv2
import numpy as np
from artifact_io import save

CASE_DIR = Path(__file__).resolve().parent
N, H, W = 96, 144, 256


def soft_cutout(image, box, kind):
    \"\"\"Return premultiplied-looking RGB patch and a soft foreground alpha.\"\"\"
    x0, y0, x1, y1 = box
    patch = image[y0:y1, x0:x1].copy()
    yy, xx = np.indices(patch.shape[:2])
    rgb = patch.astype(np.float32)
    # Broad object cues plus conservative geometric support.  Geometry keeps
    # white packaging; chroma cues retain antialiased colored edge pixels.
    if kind == \"cola\":
        geom = ((xx >= 3) & (xx <= 13) & (yy >= 5) & (yy <= 25))
        geom |= ((xx >= 4) & (xx <= 12) & (yy >= 3) & (yy <= 26))
        chroma = (rgb[..., 0] > rgb[..., 1] * 1.18) & (rgb[..., 0] > rgb[..., 2] * 1.12)
    elif kind == \"cup\":
        # Include lid and straw; taper the cup toward its base.
        center = 9.5
        half = np.where(yy < 7, 4.5, np.maximum(5.2, 8.0 - .095 * np.maximum(yy-7, 0)))
        geom = (yy >= 1) & (yy <= 29) & (np.abs(xx-center) <= half)
        geom |= (yy <= 9) & (xx >= 7) & (xx <= 13)
        chroma = (rgb[..., 0] > rgb[..., 2] * 1.15) & (rgb[..., 1] > rgb[..., 2] * 1.10)
    else:
        # Carton has a small green cap, tapered shoulders, and rectangular body.
        geom = ((yy >= 7) & (yy <= 28) & (xx >= 3) & (xx <= 13))
        geom |= ((yy >= 3) & (yy <= 9) & (xx >= 5) & (xx <= 12))
        geom |= ((yy >= 1) & (yy <= 5) & (xx >= 7) & (xx <= 12))
        chroma = (rgb[..., 1] > rgb[..., 0] * 1.05) & (rgb[..., 1] > rgb[..., 2] * 1.01)
    mask = geom.astype(np.uint8) * 255
    # Preserve colored antialiasing just outside conservative geometry.
    mask[np.logical_and(chroma, cv2.dilate(mask, np.ones((3,3),np.uint8)) > 0)] = 255
    alpha = cv2.GaussianBlur(mask.astype(np.float32) / 255.0, (0, 0), .42)
    alpha[alpha < .025] = 0
    return patch, alpha


def build_clean_plate(src):
    \"\"\"Static scene plate: source is static except for the three objects.\"\"\"
    # The final frame is clean on the left/centre, while frame 1 is clean on the
    # far right.  A feathered seam lies in an always-unoccupied gap.
    a, b = src[-1].astype(np.float32), src[0].astype(np.float32)
    blend = np.clip((np.arange(W, dtype=np.float32) - 184.0) / 10.0, 0, 1)[None, :, None]
    plate = a * (1-blend) + b * blend
    return np.clip(plate + .5, 0, 255).astype(np.uint8)


def simulate_heavy():
    \"\"\"1-D rigid-body replay, source-calibrated, with the cola at mass 4.\"\"\"
    # Frame-1 visual centres (cola, cup, carton), collision half-widths, and
    # source-calibrated initial velocity/friction (pixels per frame).
    x = np.array([28.53, 98.90, 169.50], np.float64)
    v = np.array([20.0, 0.0, 0.0], np.float64)
    mass = np.array([4.0, 1.0, 1.0], np.float64)
    radius = np.array([5.5, 7.5, 6.0], np.float64)
    out = np.empty((N, 3), np.float64)
    out[0] = x
    substeps = 100
    for frame in range(1, N):
        for _ in range(substeps):
            # Coulomb table friction, fitted to factual trajectories.
            dv = 0.22 / substeps
            v -= np.minimum(np.abs(v), dv) * np.sign(v)
            x += v / substeps
            # Sequential perfectly inelastic impulses.  Four-times larger cola
            # momentum makes both lighter packages leave the right table edge.
            for j in (0, 1):
                overlap = radius[j] + radius[j+1] - (x[j+1] - x[j])
                if overlap > 0:
                    invsum = 1/mass[j] + 1/mass[j+1]
                    x[j] -= overlap * mass[j+1] / (mass[j] + mass[j+1])
                    x[j+1] += overlap * mass[j] / (mass[j] + mass[j+1])
                    rel = v[j] - v[j+1]
                    if rel > 0:
                        impulse = rel / invsum
                        v[j] -= impulse / mass[j]
                        v[j+1] += impulse / mass[j+1]
        out[frame] = x
    return out


def composite(dst, patch, alpha, x, y, angle=0.0):
    \"\"\"Subpixel paste, clipped at image boundaries.\"\"\"
    h, w = alpha.shape
    # Affine map the cutout at subpixel location.  Rotation remains zero while
    # supported on the tabletop; retained for deterministic extensibility.
    canvas_rgb = np.zeros((H, W, 3), np.float32)
    canvas_a = np.zeros((H, W), np.float32)
    ix, iy = int(np.floor(x)), int(np.floor(y))
    fx, fy = x-ix, y-iy
    M = np.array([[1, 0, ix+fx], [0, 1, iy+fy]], np.float32)
    canvas_rgb = cv2.warpAffine(patch.astype(np.float32), M, (W,H), flags=cv2.INTER_LINEAR,
                                borderMode=cv2.BORDER_CONSTANT, borderValue=0)
    canvas_a = cv2.warpAffine(alpha, M, (W,H), flags=cv2.INTER_LINEAR,
                              borderMode=cv2.BORDER_CONSTANT, borderValue=0)
    a = canvas_a[...,None]
    dst[:] = np.clip(canvas_rgb*a + dst.astype(np.float32)*(1-a), 0, 255).astype(np.uint8)


def main():
    src = np.load(CASE_DIR / \"source_pixels.npz\")[\"rgb\"]
    assert src.shape == (N,H,W,3) and src.dtype == np.uint8
    plate = build_clean_plate(src)
    # Tight crops in frame 1. Anchors are crop offsets from measured centres.
    specs = [
        (\"cola\",   (20,59,38,87), (28.53-20, 72.54-59)),
        (\"cup\",    (89,55,110,87),(98.90-89, 73.0-55)),
        (\"carton\", (161,57,179,87),(169.50-161, 73.0-57)),
    ]
    sprites=[]
    for kind, box, anchor in specs:
        patch, alpha = soft_cutout(src[0], box, kind)
        sprites.append((patch,alpha,anchor,box))
    centers=simulate_heavy()
    frames=np.empty_like(src)
    for t in range(N):
        frame=plate.copy()
        # Painter order left to right matches collision occlusion in source.
        for j,(patch,alpha,anchor,box) in enumerate(sprites):
            cx=centers[t,j]
            x=cx-anchor[0]
            y=box[1]  # all three remain upright on the tabletop until offscreen
            if x < W+patch.shape[1] and x+patch.shape[1] > 0:
                composite(frame,patch,alpha,x,y)
        frames[t]=frame
    # Preserve exact factual first frame: mass changes dynamics, not appearance,
    # and at t=0 no causal displacement has occurred.
    frames[0]=src[0]
    save(frames, CASE_DIR)

if __name__ == \"__main__\":
