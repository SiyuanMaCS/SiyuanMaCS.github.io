+"""Generate the restitution counterfactual for this case.
+
+The red die follows the factual fall through its first floor contact.  Its
+large source rebound is then replaced by a very small, rapidly damped hop,
+consistent with multiplying its restitution by 0.05.  All rendering is made
+from the supplied pixels; no external assets are used.
+"""
+
+from pathlib import Path
+
+import numpy as np
+from scipy import ndimage
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def red_silhouette(frame: np.ndarray) -> np.ndarray:
+    """Segment the pink die, including its dark pips, from a source frame."""
+    a = frame.astype(np.int16)
+    r, g, b = a[..., 0], a[..., 1], a[..., 2]
+    yy, xx = np.indices(r.shape)
+    pink = (
+        (r > 104)
+        & (r > 1.15 * g)
+        & (r > 1.07 * b)
+        & (g > 43)
+        & (xx > 106)
+        & (xx < 140)
+        & (yy < 108)
+    )
+    labels, count = ndimage.label(pink)
+    if count == 0:
+        raise RuntimeError("red die could not be segmented")
+    # The die is the sizeable component nearest the center drop line.
+    choices = []
+    for label in range(1, count + 1):
+        ys, xs = np.where(labels == label)
+        if len(xs) >= 20:
+            score = len(xs) - 2.0 * abs(float(xs.mean()) - 123.0)
+            choices.append((score, label))
+    label = max(choices)[1]
+    mask = labels == label
+    mask = ndimage.binary_closing(mask, structure=np.ones((3, 3)), iterations=1)
+    mask = ndimage.binary_fill_holes(mask)
+    # A one-pixel skirt retains the antialiased cube boundary.
+    return ndimage.binary_dilation(mask, iterations=1)
+
+
+def clean_patch(frame: np.ndarray, clean_plate: np.ndarray, mask: np.ndarray) -> np.ndarray:
+    """Remove the factual red die with a softly edged clean-table patch."""
+    # Include the narrow contact shadow beneath the die.  Restrict it to the
+    # object's local envelope so nearby white-die pixels remain untouched.
+    ys, xs = np.where(mask)
+    x0, x1 = max(0, xs.min() - 2), min(frame.shape[1], xs.max() + 3)
+    y0, y1 = max(0, ys.min() - 2), min(frame.shape[0], ys.max() + 5)
+    local = np.zeros(mask.shape, bool)
+    local[y0:y1, x0:x1] = True
+    dark = (frame.mean(axis=2) < 105) & local & (np.indices(mask.shape)[0] >= ys.max() - 1)
+    remove = ndimage.binary_dilation(mask | dark, iterations=2)
+
+    # Feather only the outermost two pixels.  The clean plate is source frame
+    # 1, where this part of the tabletop is unobstructed.
+    distance = ndimage.distance_transform_edt(remove)
+    alpha = np.clip(distance / 2.0, 0.0, 1.0)[..., None]
+    out = frame.astype(np.float32) * (1.0 - alpha) + clean_plate.astype(np.float32) * alpha
+    return np.clip(np.rint(out), 0, 255).astype(np.uint8)
+
+
+def composite_sprite(
+    frame: np.ndarray,
+    sprite_rgb: np.ndarray,
+    sprite_alpha: np.ndarray,
+    shift_x: int,
+    shift_y: int,
+) -> np.ndarray:
+    """Alpha-composite a translated RGBA sprite without wraparound."""
+    rgb = ndimage.shift(
+        sprite_rgb.astype(np.float32),
+        (shift_y, shift_x, 0),
+        order=1,
+        mode="constant",
+        cval=0.0,
+        prefilter=False,
+    )
+    alpha = ndimage.shift(
+        sprite_alpha.astype(np.float32),
+        (shift_y, shift_x),
+        order=1,
+        mode="constant",
+        cval=0.0,
+        prefilter=False,
+    )[..., None]
+    out = frame.astype(np.float32) * (1.0 - alpha) + rgb * alpha
+    return np.clip(np.rint(out), 0, 255).astype(np.uint8)
+
+
+def make_frames() -> np.ndarray:
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3) and source.dtype == np.uint8
+    result = source.copy()
+
+    # At frame 1 the eventual resting area (x~123, y~87) is clean.  Preserve
+    # each source frame globally and consult this plate only beneath the red
+    # die that must be removed.
+    clean_plate = source[0]
+
+    # A settled source pose provides the exact native shading, pips, edges and
+    # contact shadow.  It is isolated from frame 31.
+    settled_index = 30
+    settled_mask = red_silhouette(source[settled_index])
+    ys, xs = np.where(settled_mask)
+    # Add only the compact shadow directly below the cube.
+    shadow = np.zeros(settled_mask.shape, bool)
+    shadow[max(0, ys.max() - 1) : min(144, ys.max() + 4), max(0, xs.min() - 1) : min(256, xs.max() + 2)] = True
+    shadow &= source[settled_index].mean(axis=2) < 125
+    sprite_core = settled_mask | shadow
+    sprite_alpha = ndimage.gaussian_filter(sprite_core.astype(np.float32), sigma=0.42)
+    sprite_alpha = np.clip(sprite_alpha, 0.0, 1.0)
+    sprite_rgb = source[settled_index]
+
+    # Source frame 7 is the last factual contact frame.  The source then
+    # rebounds about ten pixels.  With e'=0.05e the corresponding hop is tiny:
+    # hold four pixels above the eventual pose for two frames, then settle
+    # monotonically over four more.  Integer positions fit the pixel-art look.
+    vertical_shift = {
+        7: -4,  # output frame 8
+        8: -4,
+        9: -3,
+        10: -2,
+        11: -1,
+    }
+
+    for index in range(7, 96):
+        factual_mask = red_silhouette(source[index])
+        base = clean_patch(source[index], clean_plate, factual_mask)
+        dy = vertical_shift.get(index, 0)
+        result[index] = composite_sprite(base, sprite_rgb, sprite_alpha, 0, dy)
+
+    return result
+
+
+def main() -> None:
+    frames = make_frames()
+    artifact_io.save(frames, CASE_DIR)
+    print(f"saved {frames.shape[0]} frames to {CASE_DIR}")
+
+
+if __name__ == "__main__":
+    main()
