+"""Generate the counterfactual video with the purple marble absent.
+
+The source depicts two marbles released from opposite sides of a shallow
+trough.  They first interact between source frames 34 and 35.  Up to that
+point the orange marble is copied verbatim.  Afterwards its measured inbound
+motion is reflected through the centre of the trough: it crosses the centre,
+climbs the right side, turns, and returns.  Object removal uses clean pixels
+from other source instants, selected per patch to preserve the source's tiny
+lighting/rendering variations.
+"""
+
+from pathlib import Path
+
+import numpy as np
+from scipy import ndimage
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def color_components(frame, kind):
+    """Return plausible colored-marble components as (area, x, y)."""
+    p = frame.astype(np.int16)
+    r, g, b = p[..., 0], p[..., 1], p[..., 2]
+    yy = np.indices(r.shape)[0]
+    if kind == "purple":
+        mask = ((b > 75) & (b > r + 20) & (b > g + 14) &
+                (yy >= 48) & (yy <= 80))
+        lo, hi = 35, 125
+    else:
+        # This excludes the ochre wood and the larger orange plant-pot region.
+        mask = ((r > 140) & (g > 68) & (b < 140) &
+                (r > g * 1.07) & (g > b * 1.22) &
+                (yy >= 48) & (yy <= 82))
+        lo, hi = 35, 135
+    labels, count = ndimage.label(mask)
+    items = []
+    for label in range(1, count + 1):
+        ys, xs = np.where(labels == label)
+        area = len(xs)
+        if lo <= area <= hi and (xs.max() - xs.min()) <= 14:
+            items.append((area, float(xs.mean()), float(ys.mean())))
+    return items
+
+
+def track_marble(frames, kind, start):
+    """Track a marble by color, continuity, and compact component size."""
+    result = []
+    previous = np.asarray(start, dtype=np.float64)
+    velocity = np.zeros(2, dtype=np.float64)
+    for frame in frames:
+        candidates = color_components(frame, kind)
+        if candidates:
+            prediction = previous + velocity
+            ranked = sorted(candidates,
+                            key=lambda q: (q[1] - prediction[0]) ** 2 +
+                                          1.4 * (q[2] - prediction[1]) ** 2)
+            _, x, y = ranked[0]
+            current = np.array([x, y])
+            # Suppress only detector subpixel noise; retain physical motion.
+            velocity = 0.55 * velocity + 0.45 * (current - previous)
+            previous = current
+        result.append(previous.copy())
+    return np.asarray(result)
+
+
+def ellipse_weight(shape, cx, cy, rx, ry, feather=1.6):
+    """A feathered ellipse, one in the interior and zero outside."""
+    h, w = shape
+    y, x = np.indices((h, w), dtype=np.float32)
+    d = np.sqrt(((x - cx) / rx) ** 2 + ((y - cy) / ry) ** 2)
+    edge = feather / max(rx, ry)
+    return np.clip((1.0 + edge - d) / edge, 0.0, 1.0)
+
+
+def removal_weight(shape, cx, cy):
+    # Ball plus its low, horizontally spread contact shadow.
+    ball = ellipse_weight(shape, cx, cy, 7.2, 7.1, 1.8)
+    shadow = ellipse_weight(shape, cx, cy + 5.5, 9.2, 3.5, 1.8)
+    return np.maximum(ball, shadow)
+
+
+def choose_donor(frames, index, cx, cy, orange_track, purple_track):
+    """Select an object-free donor with the closest surrounding appearance."""
+    h, w = frames.shape[1:3]
+    x0, x1 = max(0, int(cx) - 13), min(w, int(cx) + 14)
+    y0, y1 = max(0, int(cy) - 11), min(h, int(cy) + 13)
+    yy, xx = np.indices((y1 - y0, x1 - x0))
+    gx, gy = xx + x0, yy + y0
+    dist = np.sqrt((gx - cx) ** 2 + ((gy - cy) * 1.1) ** 2)
+    annulus = (dist >= 9.5) & (dist <= 13.0)
+    target = frames[index, y0:y1, x0:x1].astype(np.float32)
+    best, best_score = None, np.inf
+    for j in range(len(frames)):
+        # Do not borrow pixels containing either factual marble or its shadow.
+        if np.linalg.norm(orange_track[j] - [cx, cy]) < 22.0:
+            continue
+        if np.linalg.norm(purple_track[j] - [cx, cy]) < 22.0:
+            continue
+        candidate = frames[j, y0:y1, x0:x1].astype(np.float32)
+        score = float(np.mean((candidate[annulus] - target[annulus]) ** 2))
+        # A very small temporal preference breaks nearly equal matches.
+        score += 0.0008 * abs(j - index)
+        if score < best_score:
+            best, best_score = j, score
+    if best is None:
+        # This cannot occur for these trajectories, but keeps regeneration safe.
+        best = 0 if cx > 55 else len(frames) - 1
+    return frames[best]
+
+
+def remove_at(frame, donor, cx, cy):
+    weight = removal_weight(frame.shape[:2], cx, cy)[..., None]
+    mixed = frame.astype(np.float32) * (1.0 - weight) + donor * weight
+    return np.clip(np.rint(mixed), 0, 255).astype(np.uint8)
+
+
+def canonical_orange_sprite(frames, orange_track, purple_track):
+    """Recover a small foreground sprite from the crisp initial orange ball."""
+    source_index = 0
+    cx, cy = orange_track[source_index]
+    donor = choose_donor(frames, source_index, cx, cy,
+                         orange_track, purple_track)
+    radius = 8
+    ix, iy = int(round(cx)), int(round(cy))
+    src = frames[source_index, iy-radius:iy+radius+1,
+                 ix-radius:ix+radius+1].astype(np.float32)
+    bg = donor[iy-radius:iy+radius+1,
+               ix-radius:ix+radius+1].astype(np.float32)
+    hh, ww = src.shape[:2]
+    y, x = np.indices((hh, ww), dtype=np.float32)
+    local_cx = radius + (cx - ix)
+    local_cy = radius + (cy - iy)
+
+    # Supersampled-looking circle coverage.  Recover intrinsic edge color from
+    # source over its known clean background, avoiding a brown fringe when the
+    # ball later moves over the white fabric.
+    d = np.sqrt((x - local_cx) ** 2 + (y - local_cy) ** 2)
+    alpha = np.clip(5.65 - d, 0.0, 1.0)
+    foreground = np.zeros_like(src)
+    safe = alpha > 0.08
+    foreground[safe] = ((src[safe] -
+                         (1.0 - alpha[safe, None]) * bg[safe]) /
+                        alpha[safe, None])
+    foreground = np.clip(foreground, 0, 255)
+
+    # At the extreme antialiased rim, nearest interior shading is more stable
+    # than division by a tiny coverage value.
+    interior = alpha >= 0.99
+    _, nearest = ndimage.distance_transform_edt(~interior,
+                                                return_indices=True)
+    rim = (alpha > 0) & (alpha < 0.28)
+    foreground[rim] = foreground[nearest[0][rim], nearest[1][rim]]
+    return foreground, alpha, local_cx, local_cy
+
+
+def sample_sprite(foreground, alpha, local_cx, local_cy, cx, cy, shape):
+    """Place an RGBA sprite at a subpixel position using bilinear sampling."""
+    h, w = shape
+    sh, sw = alpha.shape
+    x0 = max(0, int(np.floor(cx - local_cx)) - 1)
+    y0 = max(0, int(np.floor(cy - local_cy)) - 1)
+    x1 = min(w, int(np.ceil(cx + (sw - 1 - local_cx))) + 2)
+    y1 = min(h, int(np.ceil(cy + (sh - 1 - local_cy))) + 2)
+    yy, xx = np.indices((y1 - y0, x1 - x0), dtype=np.float32)
+    sx = xx + x0 - cx + local_cx
+    sy = yy + y0 - cy + local_cy
+    coords = np.array([sy, sx])
+    a = ndimage.map_coordinates(alpha, coords, order=1,
+                                mode="constant", cval=0.0)
+    fg = np.stack([ndimage.map_coordinates(foreground[..., channel], coords,
+                                            order=1, mode="nearest")
+                   for channel in range(3)], axis=-1)
+    return (x0, y0, x1, y1), fg, a
+
+
+def draw_orange(frame, cx, cy, sprite):
+    """Draw the orange sphere and a surface-following contact shadow."""
+    output = frame.astype(np.float32)
+    h, w = frame.shape[:2]
+
+    # Shadow first; multiplying the local surface naturally works on both the
+    # white fabric and brown end ramp.
+    shadow = ellipse_weight((h, w), cx + 0.3, cy + 5.5,
+                            7.4, 2.45, 1.5)[..., None]
+    shadow_strength = 0.34
+    warm_dark = np.array([34.0, 24.0, 15.0], dtype=np.float32)
+    output = output * (1.0 - shadow * shadow_strength) + \
+             warm_dark * (shadow * shadow_strength)
+
+    fg0, a0, lx, ly = sprite
+    bounds, fg, a = sample_sprite(fg0, a0, lx, ly, cx, cy, (h, w))
+    x0, y0, x1, y1 = bounds
+    patch = output[y0:y1, x0:x1]
+    output[y0:y1, x0:x1] = patch * (1.0 - a[..., None]) + fg * a[..., None]
+    return np.clip(np.rint(output), 0, 255).astype(np.uint8)
+
+
+def counterfactual_path(orange_track):
+    """Continue measured inbound motion periodically without the collision."""
+    # Track centers are chroma centroids, consistently about 0.4 px above the
+    # geometric center.  We preserve that convention for seamless compositing.
+    source_t = np.arange(1, 35, dtype=np.float64)
+    source_x = orange_track[:34, 0]
+    # The collision starts next frame.  Extrapolate one and a half frames to the
+    # visual centre using the measured final approach speed.
+    center = 128.0
+    turn_t = 5.5
+    center_t = 35.5
+    path_t = np.r_[source_t, center_t]
+    path_x = np.r_[source_x, center]
+
+    def inbound(t):
+        return np.interp(t, path_t, path_x)
+
+    result = np.empty((96, 2), dtype=np.float64)
+    half = center_t - turn_t             # 30 frames
+    period = 4.0 * half                  # 120 frames
+    for i, t in enumerate(np.arange(1, 97, dtype=np.float64)):
+        u = (t - turn_t) % period
+        if u < half:                     # left rim -> centre
+            x = inbound(turn_t + u)
+        elif u < 2 * half:               # centre -> right rim
+            x = 2 * center - inbound(turn_t + 2 * half - u)
+        elif u < 3 * half:               # right rim -> centre
+            x = 2 * center - inbound(turn_t + u - 2 * half)
+        else:                            # centre -> left rim
+            x = inbound(turn_t + 4 * half - u)
+        # Empirical trough profile from both factual inbound trajectories.
+        y = 71.65 - 12.75 * (abs(x - center) / 101.0) ** 1.28
+        result[i] = (x, y)
+    return result
+
+
+def main():
+    frames = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert frames.shape == (96, 144, 256, 3) and frames.dtype == np.uint8
+
+    orange = track_marble(frames, "orange", (27.0, 59.0))
+    purple = track_marble(frames, "purple", (228.5, 59.0))
+    cf_orange = counterfactual_path(orange)
+    sprite = canonical_orange_sprite(frames, orange, purple)
+
+    output = frames.copy()
+    for i in range(96):
+        # Purple and its contact shadow are absent from frame 1 onward.
+        px, py = purple[i]
+        donor = choose_donor(frames, i, px, py, orange, purple)
+        output[i] = remove_at(output[i], donor, px, py)
+
+        # The orange marble is factually unchanged through frame 34.  Starting
+        # at frame 35, erase the collision-affected factual marble and draw its
+        # unobstructed continuation.
+        if i >= 34:
+            ox, oy = orange[i]
+            donor = choose_donor(frames, i, ox, oy, orange, purple)
+            output[i] = remove_at(output[i], donor, ox, oy)
+            cx, cy = cf_orange[i]
+            output[i] = draw_orange(output[i], cx, cy, sprite)
+
+    artifact_io.save(output, CASE_DIR)
+
+
+if __name__ == "__main__":
+    main()
