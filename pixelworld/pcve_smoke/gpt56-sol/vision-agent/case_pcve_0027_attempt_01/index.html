+"""Generate the 1.5x-initial-speed counterfactual.
+
+The source contains one moving rigid body and a static room.  The volleyball
+travels until frame 20 (1-based), where it comes to rest against the same
+obstacle for the remainder of the clip.  Increasing its initial speed by 1.5
+therefore advances that motion to the contact point in 2/3 of the time.  Dense
+bidirectional optical-flow interpolation supplies the half-frame samples so
+that the faster motion remains smooth without redrawing (and thereby changing)
+the source ball's texture, lighting, or perspective scaling.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+SPEED_SCALE = 1.5
+# Zero-based frame at which the source ball has completed its motion.
+CONTACT_FRAME = 19
+
+
+def flow_between(a: np.ndarray, b: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
+    """Return dense forward and backward flow for two adjacent RGB frames."""
+    ga = cv2.cvtColor(a, cv2.COLOR_RGB2GRAY)
+    gb = cv2.cvtColor(b, cv2.COLOR_RGB2GRAY)
+    kwargs = dict(
+        pyr_scale=0.5,
+        levels=4,
+        winsize=19,
+        iterations=5,
+        poly_n=7,
+        poly_sigma=1.5,
+        flags=cv2.OPTFLOW_FARNEBACK_GAUSSIAN,
+    )
+    return (
+        cv2.calcOpticalFlowFarneback(ga, gb, None, **kwargs),
+        cv2.calcOpticalFlowFarneback(gb, ga, None, **kwargs),
+    )
+
+
+def inbetween(a: np.ndarray, b: np.ndarray, amount: float) -> np.ndarray:
+    """Motion-compensated interpolation from ``a`` to ``b``."""
+    if amount <= 0.0:
+        return a.copy()
+    if amount >= 1.0:
+        return b.copy()
+
+    forward, backward = flow_between(a, b)
+    h, w = a.shape[:2]
+    xx, yy = np.meshgrid(np.arange(w, dtype=np.float32),
+                         np.arange(h, dtype=np.float32))
+
+    # Pull each endpoint toward the requested intermediate instant.  Using
+    # both directions is especially useful on the newly revealed floor behind
+    # the moving ball.
+    wa = cv2.remap(
+        a, xx - amount * forward[..., 0], yy - amount * forward[..., 1],
+        cv2.INTER_LINEAR, borderMode=cv2.BORDER_REFLECT_101,
+    )
+    wb = cv2.remap(
+        b, xx - (1.0 - amount) * backward[..., 0],
+        yy - (1.0 - amount) * backward[..., 1],
+        cv2.INTER_LINEAR, borderMode=cv2.BORDER_REFLECT_101,
+    )
+    mixed = wa.astype(np.float32) * (1.0 - amount) + wb.astype(np.float32) * amount
+    return np.clip(np.rint(mixed), 0, 255).astype(np.uint8)
+
+
+def generate() -> np.ndarray:
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    output = np.empty_like(source)
+
+    for out_index in range(len(source)):
+        source_time = min(SPEED_SCALE * out_index, float(CONTACT_FRAME))
+        lo = int(np.floor(source_time))
+        fraction = source_time - lo
+        if fraction == 0.0 or lo == CONTACT_FRAME:
+            output[out_index] = source[lo]
+        else:
+            output[out_index] = inbetween(source[lo], source[lo + 1], fraction)
+
+    # The intervention begins at frame 1, so its appearance and position are
+    # exactly preserved while its subsequent displacement is increased.
+    output[0] = source[0]
+    return output
+
+
+if __name__ == "__main__":
+    frames = generate()
+    artifact_io.save(frames, CASE_DIR)
+    print(f"saved {frames.shape[0]} RGB frames at {frames.shape[2]}x{frames.shape[1]}")
