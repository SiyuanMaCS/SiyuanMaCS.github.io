+import numpy as np
+import cv2

+def cap_centers(frames):
+    """Track the colored top of each curling stone."""
+    reds, yellows = [], []
+    for im in frames:
+        hsv = cv2.cvtColor(im, cv2.COLOR_RGB2HSV)
+        specs = [(((hsv[..., 0] < 12) | (hsv[..., 0] > 170)) &
+                  (hsv[..., 1] > 70) & (im[..., 0] > 80)),
+                 ((hsv[..., 0] >= 18) & (hsv[..., 0] <= 45) &
+                  (hsv[..., 1] > 50) & (hsv[..., 2] > 60))]
+        found = []
+        for mask in specs:
+            n, lab, stats, cents = cv2.connectedComponentsWithStats(mask.astype(np.uint8))
+            candidates = []
+            for k in range(1, n):
+                x, y, w, h, area = stats[k]
+                if area >= 7 and 40 < y < 95 and w < 16 and h < 12:
+                    candidates.append((area, cents[k]))
+            found.append(max(candidates, key=lambda q: q[0])[1] if candidates else np.array([np.nan, np.nan]))
+        reds.append(found[0]); yellows.append(found[1])
+    return np.asarray(reds), np.asarray(yellows)
+
+
+def clean_plate(frames):
+    # During the first 32 frames both small stones sweep rapidly across the ice.
+    # A temporal median therefore removes them while reproducing the completely
+    # static camera, target, line, reflections and lighting.
+    return np.median(frames[:32].astype(np.float32), axis=0).round().astype(np.uint8)
+
+
+def sprite_from_frame(frame, plate, center, recolor_red=False):
+    """Return a tightly cropped, softly masked stone sprite and its anchor."""
+    cx, cy = center
+    x0 = max(0, int(np.floor(cx)) - 11)
+    x1 = min(frame.shape[1], int(np.floor(cx)) + 13)
+    y0 = max(0, int(np.floor(cy)) - 7)
+    y1 = min(frame.shape[0], int(np.floor(cy)) + 13)
+    rgb = frame[y0:y1, x0:x1].copy()
+    bg = plate[y0:y1, x0:x1]
+    delta = np.max(np.abs(rgb.astype(np.int16) - bg.astype(np.int16)), axis=2)
+
+    yy, xx = np.mgrid[y0:y1, x0:x1]
+    # Restrict differences to the projected puck and its soft lower shadow.
+    support = (((xx - cx) / 11.0) ** 2 + ((yy - (cy + 3.0)) / 10.0) ** 2) < 1.0
+    core = ((delta >= 3) & support).astype(np.uint8)
+    core = cv2.morphologyEx(core, cv2.MORPH_CLOSE, np.ones((3, 3), np.uint8))
+    core = cv2.dilate(core, np.ones((3, 3), np.uint8), iterations=1)
+    alpha = cv2.GaussianBlur(core.astype(np.float32), (3, 3), 0.55)
+    alpha = np.clip(alpha, 0, 1)
+
+    if recolor_red:
+        hsv = cv2.cvtColor(rgb, cv2.COLOR_RGB2HSV)
+        yellow = ((hsv[..., 0] >= 17) & (hsv[..., 0] <= 48) &
+                  (hsv[..., 1] >= 45) & (hsv[..., 2] >= 70) & (alpha > .05))
+        hsv[..., 0][yellow] = 0
+        # The factual red handle is a little more saturated than the yellow one.
+        hsv[..., 1][yellow] = np.maximum(hsv[..., 1][yellow], 145)
+        rgb = cv2.cvtColor(hsv, cv2.COLOR_HSV2RGB)
+    return rgb, alpha, (cx - x0, cy - y0)
+
+
+def composite(canvas, sprite, alpha, anchor, center):
+    h, w = sprite.shape[:2]
+    # Fractional affine placement avoids one-pixel jumps along the trajectory.
+    x = center[0] - anchor[0]
+    y = center[1] - anchor[1]
+    ix, iy = int(np.floor(x)), int(np.floor(y))
+    fx, fy = x - ix, y - iy
+    M = np.float32([[1, 0, fx], [0, 1, fy]])
+    spr = cv2.warpAffine(sprite, M, (w + 1, h + 1), flags=cv2.INTER_LINEAR,
+                         borderMode=cv2.BORDER_REPLICATE)
+    alp = cv2.warpAffine(alpha, M, (w + 1, h + 1), flags=cv2.INTER_LINEAR,
+                         borderMode=cv2.BORDER_CONSTANT)[..., None]
+    x0, y0 = ix, iy
+    x1, y1 = x0 + w + 1, y0 + h + 1
+    sx0, sy0 = max(0, -x0), max(0, -y0)
+    sx1, sy1 = w + 1 - max(0, x1 - canvas.shape[1]), h + 1 - max(0, y1 - canvas.shape[0])
+    dx0, dy0 = max(0, x0), max(0, y0)
+    dx1, dy1 = dx0 + (sx1 - sx0), dy0 + (sy1 - sy0)
+    a = alp[sy0:sy1, sx0:sx1]
+    dst = canvas[dy0:dy1, dx0:dx1].astype(np.float32)
+    src = spr[sy0:sy1, sx0:sx1].astype(np.float32)
+    canvas[dy0:dy1, dx0:dx1] = np.clip(dst * (1-a) + src * a, 0, 255).round().astype(np.uint8)
+
+
+def make_video():
+    source = np.load(CASE_DIR / 'source_pixels.npz')['rgb']
+    plate = clean_plate(source)
+    red, yellow = cap_centers(source)
+
+    # Robust incoming red trajectory from the untouched, well-separated interval.
+    tt = np.arange(0, 31, dtype=np.float64)
+    px = np.polyfit(tt, red[:31, 0], 1)
+    py = np.polyfit(tt, red[:31, 1], 1)
+    yellow_rest = yellow[0].copy()
+
+    # Projected contact occurs when the cap centers are about ten pixels apart.
+    contact = int(round((yellow_rest[0] - 10.0 - px[1]) / px[0]))
+    contact = max(1, min(94, contact))
+    incoming = np.column_stack((np.polyval(px, np.arange(96)),
+                                np.polyval(py, np.arange(96))))
+    red_path = incoming.copy()
+    contact_pos = incoming[contact].copy()
+    contact_pos[0] = yellow_rest[0] - 10.0
+    contact_pos[1] = yellow_rest[1] + 0.2
+    # The factual impact is strongly inelastic: after impact each stone only
+    # creeps outward at about 0.15 px/frame. Against the immobilized yellow
+    # stone, the red stone recoils at that observed rate while yellow stays put.
+    for t in range(contact, 96):
+        dt = t - contact
+        red_path[t] = contact_pos + np.array([-0.155 * dt, -0.008 * dt])
+
+    # Exact stationary yellow sprite from frame 1 (index 0).
+    ys, ya, yanchor = sprite_from_frame(source[0], plate, yellow_rest, False)
+
+    frames = []
+    for t in range(96):
+        out = plate.copy()
+        if t <= 38 and np.all(np.isfinite(red[t])):
+            # Before the source collision, the factual red puck is precisely the
+            # counterfactual puck. Preserve its native rendering.
+            rs, ra, ranchor = sprite_from_frame(source[t], plate, red[t], False)
+        else:
+            # The original yellow puck traversed the right half of this exact
+            # line and target. Reuse the nearest-position rendering (same puck
+            # geometry, shadow and local illumination), changing only its cap.
+            valid = np.arange(0, 39)
+            score = ((yellow[valid, 0] - red_path[t, 0]) ** 2 +
+                     2.0 * (yellow[valid, 1] - red_path[t, 1]) ** 2)
+            s = int(valid[np.argmin(score)])
+            rs, ra, ranchor = sprite_from_frame(source[s], plate, yellow[s], True)
+        composite(out, rs, ra, ranchor, red_path[t])
+        composite(out, ys, ya, yanchor, yellow_rest)
+        frames.append(out)
+
+    result = np.stack(frames).astype(np.uint8)
+    # At intervention time the source already has the requested state. Preserve
+    # frame 1 bit-for-bit, eliminating any needless transition discrepancy.
+    result[0] = source[0]
+    return result
+
+
+if __name__ == '__main__':
+    frames = make_video()
+    artifact_io.save(frames, CASE_DIR)

            found.append(max(candidates, key=lambda q: q[0])[1] if candidates else np.array([np.nan, np.nan]))
        reds.append(found[0]); yellows.append(found[1])
    return np.asarray(reds), np.asarray(yellows)


def clean_plate(frames):
    # During the first 32 frames both small stones sweep rapidly across the ice.
    # A temporal median therefore removes them while reproducing the completely
    # static camera, target, line, reflections and lighting.
    return np.median(frames[:32].astype(np.float32), axis=0).round().astype(np.uint8)


def sprite_from_frame(frame, plate, center, recolor_red=False):
    \"\"\"Return a tightly cropped, softly masked stone sprite and its anchor.\"\"\"
    cx, cy = center
    x0 = max(0, int(np.floor(cx)) - 11)
    x1 = min(frame.shape[1], int(np.floor(cx)) + 13)
    y0 = max(0, int(np.floor(cy)) - 7)
    y1 = min(frame.shape[0], int(np.floor(cy)) + 13)
    rgb = frame[y0:y1, x0:x1].copy()
    bg = plate[y0:y1, x0:x1]
    delta = np.max(np.abs(rgb.astype(np.int16) - bg.astype(np.int16)), axis=2)

    yy, xx = np.mgrid[y0:y1, x0:x1]
    # Restrict differences to the projected puck and its soft lower shadow.
    support = (((xx - cx) / 11.0) ** 2 + ((yy - (cy + 3.0)) / 10.0) ** 2) < 1.0
    core = ((delta >= 3) & support).astype(np.uint8)
    core = cv2.morphologyEx(core, cv2.MORPH_CLOSE, np.ones((3, 3), np.uint8))
    core = cv2.dilate(core, np.ones((3, 3), np.uint8), iterations=1)
    alpha = cv2.GaussianBlur(core.astype(np.float32), (3, 3), 0.55)
    alpha = np.clip(alpha, 0, 1)

    if recolor_red:
        hsv = cv2.cvtColor(rgb, cv2.COLOR_RGB2HSV)
        yellow = ((hsv[..., 0] >= 17) & (hsv[..., 0] <= 48) &
                  (hsv[..., 1] >= 45) & (hsv[..., 2] >= 70) & (alpha > .05))
        hsv[..., 0][yellow] = 0
        # The factual red handle is a little more saturated than the yellow one.
        hsv[..., 1][yellow] = np.maximum(hsv[..., 1][yellow], 145)
        rgb = cv2.cvtColor(hsv, cv2.COLOR_HSV2RGB)
    return rgb, alpha, (cx - x0, cy - y0)


def composite(canvas, sprite, alpha, anchor, center):
    h, w = sprite.shape[:2]
    # Fractional affine placement avoids one-pixel jumps along the trajectory.
    x = center[0] - anchor[0]
    y = center[1] - anchor[1]
    ix, iy = int(np.floor(x)), int(np.floor(y))
    fx, fy = x - ix, y - iy
    M = np.float32([[1, 0, fx], [0, 1, fy]])
    spr = cv2.warpAffine(sprite, M, (w + 1, h + 1), flags=cv2.INTER_LINEAR,
                         borderMode=cv2.BORDER_REPLICATE)
    alp = cv2.warpAffine(alpha, M, (w + 1, h + 1), flags=cv2.INTER_LINEAR,
                         borderMode=cv2.BORDER_CONSTANT)[..., None]
    x0, y0 = ix, iy
    x1, y1 = x0 + w + 1, y0 + h + 1
    sx0, sy0 = max(0, -x0), max(0, -y0)
    sx1, sy1 = w + 1 - max(0, x1 - canvas.shape[1]), h + 1 - max(0, y1 - canvas.shape[0])
    dx0, dy0 = max(0, x0), max(0, y0)
    dx1, dy1 = dx0 + (sx1 - sx0), dy0 + (sy1 - sy0)
    a = alp[sy0:sy1, sx0:sx1]
    dst = canvas[dy0:dy1, dx0:dx1].astype(np.float32)
    src = spr[sy0:sy1, sx0:sx1].astype(np.float32)
    canvas[dy0:dy1, dx0:dx1] = np.clip(dst * (1-a) + src * a, 0, 255).round().astype(np.uint8)


def make_video():
    source = np.load(CASE_DIR / 'source_pixels.npz')['rgb']
    plate = clean_plate(source)
    red, yellow = cap_centers(source)

    # Robust incoming red trajectory from the untouched, well-separated interval.
    tt = np.arange(0, 31, dtype=np.float64)
    px = np.polyfit(tt, red[:31, 0], 1)
    py = np.polyfit(tt, red[:31, 1], 1)
    yellow_rest = yellow[0].copy()

    # Projected contact occurs when the cap centers are about ten pixels apart.
    contact = int(round((yellow_rest[0] - 10.0 - px[1]) / px[0]))
    contact = max(1, min(94, contact))
    incoming = np.column_stack((np.polyval(px, np.arange(96)),
                                np.polyval(py, np.arange(96))))
    red_path = incoming.copy()
    contact_pos = incoming[contact].copy()
    contact_pos[0] = yellow_rest[0] - 10.0
    contact_pos[1] = yellow_rest[1] + 0.2
    # The factual impact is strongly inelastic: after impact each stone only
    # creeps outward at about 0.15 px/frame. Against the immobilized yellow
    # stone, the red stone recoils at that observed rate while yellow stays put.
    for t in range(contact, 96):
        dt = t - contact
        red_path[t] = contact_pos + np.array([-0.155 * dt, -0.008 * dt])

    # Exact stationary yellow sprite from frame 1 (index 0).
    ys, ya, yanchor = sprite_from_frame(source[0], plate, yellow_rest, False)

    frames = []
    for t in range(96):
        out = plate.copy()
        if t <= 38 and np.all(np.isfinite(red[t])):
            # Before the source collision, the factual red puck is precisely the
            # counterfactual puck. Preserve its native rendering.
            rs, ra, ranchor = sprite_from_frame(source[t], plate, red[t], False)
        else:
            # The original yellow puck traversed the right half of this exact
            # line and target. Reuse the nearest-position rendering (same puck
            # geometry, shadow and local illumination), changing only its cap.
            valid = np.arange(0, 39)
            score = ((yellow[valid, 0] - red_path[t, 0]) ** 2 +
                     2.0 * (yellow[valid, 1] - red_path[t, 1]) ** 2)
            s = int(valid[np.argmin(score)])
            rs, ra, ranchor = sprite_from_frame(source[s], plate, yellow[s], True)
        composite(out, rs, ra, ranchor, red_path[t])
        composite(out, ys, ya, yanchor, yellow_rest)
        frames.append(out)

    result = np.stack(frames).astype(np.uint8)
    # At intervention time the source already has the requested state. Preserve
    # frame 1 bit-for-bit, eliminating any needless transition discrepancy.
    result[0] = source[0]
    return result


if __name__ == '__main__':
    frames = make_video()
    artifact_io.save(frames, CASE_DIR)
