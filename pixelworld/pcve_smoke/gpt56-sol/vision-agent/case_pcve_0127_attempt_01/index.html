+"""Remove the yellow marble and render the resulting collision-free motion."""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+N, H, W = 96, 144, 256
+
+
+def purple_centers(frames):
+    """Measure the purple marble centers in the factual source frames."""
+    centers = []
+    lower = np.array((120, 65, 45), np.uint8)
+    upper = np.array((175, 255, 255), np.uint8)
+    for frame in frames:
+        hsv = cv2.cvtColor(frame, cv2.COLOR_RGB2HSV)
+        mask = cv2.inRange(hsv, lower, upper)
+        count, _, stats, centroids = cv2.connectedComponentsWithStats(mask)
+        choices = [
+            j for j in range(1, count)
+            if stats[j, cv2.CC_STAT_AREA] >= 20
+            and stats[j, cv2.CC_STAT_TOP] < 85
+            and stats[j, cv2.CC_STAT_LEFT] > 100
+        ]
+        j = max(choices, key=lambda k: stats[k, cv2.CC_STAT_AREA])
+        centers.append(centroids[j])
+    return np.asarray(centers, np.float32)
+
+
+def feather_patch(dst, src, x0, y0, x1, y1, feather=3):
+    """Copy a clean source patch with a soft boundary."""
+    yy, xx = np.mgrid[y0:y1, x0:x1]
+    edge = np.minimum.reduce((xx - x0, x1 - 1 - xx, yy - y0, y1 - 1 - yy))
+    alpha = np.clip((edge + 1) / float(feather), 0.0, 1.0)[..., None]
+    region = dst[y0:y1, x0:x1].astype(np.float32)
+    clean = src[y0:y1, x0:x1].astype(np.float32)
+    dst[y0:y1, x0:x1] = np.clip(region * (1 - alpha) + clean * alpha, 0, 255).astype(np.uint8)
+
+
+def clean_background(frames):
+    """Construct a source-faithful static plate with both initial marbles absent."""
+    plate = frames[0].copy()
+    # By the last source frame both initial locations expose their true background.
+    # The larger vertical extent also removes each soft contact shadow.
+    feather_patch(plate, frames[-1], 16, 47, 41, 79, feather=4)
+    feather_patch(plate, frames[-1], 215, 47, 242, 79, feather=4)
+    return plate
+
+
+def track_height(x, measured):
+    """Marble-center height on the symmetric trough, inferred from its approach."""
+    center_x = 127.5
+    incoming = measured[6:34]
+    distance = incoming[:, 0] - center_x
+    order = np.argsort(distance)
+    d = np.r_[0.0, distance[order]]
+    y = np.r_[71.08, incoming[order, 1]]
+    return float(np.interp(abs(x - center_x), d, y))
+
+
+def counterfactual_center(index, measured):
+    """Continue the measured purple motion through the trough without collision."""
+    if index <= 33:
+        return tuple(float(v) for v in measured[index])
+
+    # Least-squares kinematics of source frames 6..33:
+    # x = x_start - 1/2*a*(frame-t_start)^2.  Once the marble passes
+    # the center, the trough slope reverses its acceleration.
+    center_x = 127.5
+    acceleration = 0.23298
+    center_time = 34.3976
+    start_time = 4.9734
+    speed_at_center = acceleration * (center_time - start_time)
+    turn_time = center_time + speed_at_center / acceleration
+    return_center_time = 2.0 * turn_time - center_time
+    t = float(index)
+
+    if t <= return_center_time:
+        s = t - center_time
+        left_displacement = speed_at_center * s - 0.5 * acceleration * s * s
+        x = center_x - left_displacement
+    else:
+        # Having returned from the left, it starts climbing the right side.
+        s = t - return_center_time
+        x = center_x + speed_at_center * s - 0.5 * acceleration * s * s
+    return x, track_height(x, measured)
+
+
+def source_sprite(frames, plate, center, flip=False):
+    """Extract an antialiased 15px purple sphere from a factual frame."""
+    cx, cy = center
+    size = (15, 15)
+    rgb = cv2.getRectSubPix(frames, size, (float(cx), float(cy))).astype(np.float32)
+    bg = cv2.getRectSubPix(plate, size, (float(cx), float(cy))).astype(np.float32)
+
+    yy, xx = np.mgrid[-7:8, -7:8].astype(np.float32)
+    radius = np.sqrt(xx * xx + yy * yy)
+    radial = np.clip(6.15 - radius, 0.0, 1.0)
+
+    hsv = cv2.cvtColor(np.clip(rgb, 0, 255).astype(np.uint8), cv2.COLOR_RGB2HSV)
+    chroma = np.clip((hsv[..., 1].astype(np.float32) - 15.0) / 38.0, 0.0, 1.0)
+    hue = hsv[..., 0]
+    purple = ((hue >= 115) & (hue <= 178)).astype(np.float32)
+    # Chroma finds the colored antialias fringe; radial support keeps unrelated
+    # trough pixels out of the sprite.
+    alpha = radial * np.maximum(purple, chroma * purple)
+    alpha = cv2.GaussianBlur(alpha, (3, 3), 0.42)
+    alpha *= radial
+
+    # Preserve the exact source shading while replacing any edge background by
+    # the clean plate beneath the sphere.
+    sprite = bg + (rgb - bg)
+    if flip:
+        sprite = sprite[:, ::-1]
+        alpha = alpha[:, ::-1]
+    return sprite, np.clip(alpha, 0.0, 1.0)
+
+
+def place_patch(canvas, patch, alpha, center):
+    """Subpixel-place a small RGB/alpha patch."""
+    ph, pw = alpha.shape
+    cx, cy = center
+    matrix = np.array([[1.0, 0.0, cx - (pw // 2)], [0.0, 1.0, cy - (ph // 2)]], np.float32)
+    warped_rgb = cv2.warpAffine(
+        patch, matrix, (W, H), flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT, borderValue=0,
+    )
+    warped_a = cv2.warpAffine(
+        alpha, matrix, (W, H), flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT, borderValue=0,
+    )[..., None]
+    canvas[:] = canvas * (1.0 - warped_a) + warped_rgb * warped_a
+
+
+def add_shadow(canvas, center):
+    """Add the small, soft contact shadow visible beneath the source marble."""
+    cx, cy = center
+    mask = np.zeros((H, W), np.float32)
+    cv2.ellipse(
+        mask, (int(round(cx)), int(round(cy + 6.2))), (7, 2), 0, 0, 360, 0.42, -1,
+        lineType=cv2.LINE_AA,
+    )
+    mask = cv2.GaussianBlur(mask, (7, 5), 1.8, sigmaY=1.0)[..., None]
+    # A neutral multiplicative shadow retains the white paper/wood texture.
+    canvas[:] *= 1.0 - 0.42 * mask
+
+
+def generate():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (N, H, W, 3) and source.dtype == np.uint8
+    measured = purple_centers(source)
+    plate = clean_background(source)
+    output = np.empty_like(source)
+
+    incoming_indices = np.arange(6, 34)
+    incoming_distances = measured[incoming_indices, 0] - 127.5
+
+    for i in range(N):
+        center = counterfactual_center(i, measured)
+        if i <= 33:
+            sprite_frame = i
+        else:
+            wanted = abs(center[0] - 127.5)
+            sprite_frame = int(incoming_indices[np.argmin(abs(incoming_distances - wanted))])
+        flip = center[0] < 127.5
+        sprite, alpha = source_sprite(source[sprite_frame], plate, measured[sprite_frame], flip)
+
+        canvas = plate.astype(np.float32)
+        add_shadow(canvas, center)
+        place_patch(canvas, sprite, alpha, center)
+        output[i] = np.clip(np.rint(canvas), 0, 255).astype(np.uint8)
+
+    return output
+
+
+if __name__ == "__main__":
+    frames = generate()
+    artifact_io.save(frames, CASE_DIR)
+    print(f"saved {len(frames)} frames to {CASE_DIR}")
