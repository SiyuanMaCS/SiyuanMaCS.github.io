+"""Generate the 5x-purple-mass counterfactual for this PixelWorld case.
+
+The source is retained exactly until the marbles touch. Mass has no visual effect and,
+in this scene, no effect on gravity-driven free rolling before impact. At impact a 1-D
+rigid-body impulse is applied with m_purple=5*m_orange; the edited paths then settle in
+the shallow center of the track. Source marble sprites are reused with soft alpha and
+subpixel placement so material, illumination, and camera remain consistent.
+"""
+from pathlib import Path
+import cv2
+import numpy as np
+from artifact_io import save
+
+CASE = Path(__file__).resolve().parent
+N, H, W = 96, 144, 256
+
+
+def detect_centers(frames):
+    """Color-segment the two rendered balls (centers in x,y)."""
+    orange = np.zeros((N, 2), np.float32)
+    purple = np.zeros((N, 2), np.float32)
+    yy = np.indices((H, W))[0]
+    for i, rgb in enumerate(frames):
+        hsv = cv2.cvtColor(rgb, cv2.COLOR_RGB2HSV)
+        masks = [
+            ((hsv[..., 0] >= 10) & (hsv[..., 0] <= 30) &
+             (hsv[..., 1] > 100) & (hsv[..., 2] > 100) &
+             (yy > 46) & (yy < 79)),
+            ((hsv[..., 0] >= 120) & (hsv[..., 0] <= 165) &
+             (hsv[..., 1] > 70) & (yy > 46) & (yy < 79)),
+        ]
+        for kind, mask in enumerate(masks):
+            count, labels, stats, cents = cv2.connectedComponentsWithStats(mask.astype(np.uint8))
+            candidates = []
+            for j in range(1, count):
+                x, y, w, h, area = stats[j]
+                cx, cy = cents[j]
+                if kind == 0:
+                    # Reject the ochre track and background plant pot.
+                    ok = 15 <= area <= 180 and w <= 20 and h <= 20 and 15 < cx < 180
+                else:
+                    ok = 20 <= area <= 150 and w <= 14 and h <= 14 and 20 < cx < 240
+                if ok:
+                    candidates.append((area, cx, cy))
+            if candidates:
+                _, cx, cy = max(candidates)
+                (orange if kind == 0 else purple)[i] = (cx, cy)
+    # Fill rare segmentation gaps smoothly.
+    for arr in (orange, purple):
+        good = arr[:, 0] > 0
+        for c in range(2):
+            arr[:, c] = np.interp(np.arange(N), np.flatnonzero(good), arr[good, c])
+    return orange, purple
+
+
+def source_sprite(frame, cx, cy, radius=7):
+    """Extract an RGBA marble sprite; alpha is radial to avoid rectangular seams."""
+    size = radius * 2 + 1
+    patch = cv2.getRectSubPix(frame, (size, size), (float(cx), float(cy)))
+    y, x = np.mgrid[-radius:radius + 1, -radius:radius + 1]
+    r = np.sqrt(x*x + y*y)
+    # Opaque through the ball, 2-pixel antialiased falloff at its silhouette.
+    alpha = np.clip(radius - r + 0.6, 0, 1)
+    # Include the ball's compact contact shadow only along the lower rim.
+    alpha = np.maximum(alpha, np.exp(-((x/5.0)**2 + ((y-radius+0.3)/1.0)**2)) * 0.23)
+    return patch.astype(np.float32), alpha.astype(np.float32)
+
+
+def erase_ball(dst, clean, cx, cy, radius=8):
+    """Restore the static track beneath a source ball with a feathered mask."""
+    x0, x1 = max(0, int(cx)-radius), min(W, int(cx)+radius+1)
+    y0, y1 = max(0, int(cy)-radius), min(H, int(cy)+radius+1)
+    y, x = np.mgrid[y0:y1, x0:x1]
+    d = np.sqrt((x-cx)**2 + (y-cy)**2)
+    a = np.clip((radius + 0.5 - d) / 1.25, 0, 1)[..., None]
+    dst[y0:y1, x0:x1] = dst[y0:y1, x0:x1]*(1-a) + clean[y0:y1, x0:x1]*a
+
+
+def paste_sprite(dst, sprite, alpha, cx, cy):
+    """Subpixel translate and alpha composite a sprite."""
+    sh, sw = alpha.shape
+    r = sw // 2
+    ix, iy = int(np.floor(cx)), int(np.floor(cy))
+    fx, fy = cx-ix, cy-iy
+    mat = np.float32([[1, 0, fx], [0, 1, fy]])
+    sp = cv2.warpAffine(sprite, mat, (sw, sh), flags=cv2.INTER_LINEAR,
+                        borderMode=cv2.BORDER_REPLICATE)
+    aa = cv2.warpAffine(alpha, mat, (sw, sh), flags=cv2.INTER_LINEAR,
+                        borderMode=cv2.BORDER_CONSTANT)[..., None]
+    x0, y0 = ix-r, iy-r
+    xa, ya = max(0, x0), max(0, y0)
+    xb, yb = min(W, x0+sw), min(H, y0+sh)
+    if xa < xb and ya < yb:
+        sx, sy = xa-x0, ya-y0
+        a = aa[sy:sy+yb-ya, sx:sx+xb-xa]
+        dst[ya:yb, xa:xb] = dst[ya:yb, xa:xb]*(1-a) + sp[sy:sy+yb-ya, sx:sx+xb-xa]*a
+
+
+def main():
+    src = np.load(CASE / 'source_pixels.npz')['rgb']
+    assert src.shape == (N, H, W, 3) and src.dtype == np.uint8
+    orange, purple = detect_centers(src)
+
+    # Source collision happens between frames 33 and 34 (0-based), when center gap
+    # reaches one diameter. The heavier purple receives only 1/5 of the acceleration
+    # from the orange impact. Using the observed equal-mass post-impact velocities
+    # with momentum conservation gives the counterfactual post-impact velocities:
+    # orange ~= -1.8 px/frame, purple ~= -0.54 px/frame.
+    hit = 34
+    ox, px = np.zeros(N, np.float32), np.zeros(N, np.float32)
+    ox[:hit], px[:hit] = orange[:hit, 0], purple[:hit, 0]
+    ox[hit], px[hit] = orange[hit, 0], purple[hit, 0]
+    vo, vp = -1.80, -0.54
+    # Shallow-bowl restoring acceleration plus rolling drag. Purple has the same
+    # gravitational acceleration regardless of mass; mass differs only in collision.
+    for i in range(hit+1, N):
+        vo = 0.965*vo - 0.0055*(ox[i-1]-128.0)
+        vp = 0.965*vp - 0.0055*(px[i-1]-128.0)
+        ox[i] = ox[i-1] + vo
+        px[i] = px[i-1] + vp
+        if px[i] - ox[i] < 10.2:  # nonpenetration, in case of a later recollision
+            mid = (ox[i] + px[i]) * 0.5
+            ox[i], px[i] = mid-5.1, mid+5.1
+            # Restitution impulse for m_orange=1, m_purple=5.
+            rel, e = vo-vp, 0.55
+            if rel > 0:
+                j = (1+e)*rel/(1.0 + 1.0/5.0)
+                vo -= j
+                vp += j/5.0
+
+    # Vertical center follows the visible track profile measured from source motion.
+    # Around the center it is nearly flat; this analytic profile avoids borrowing a
+    # source frame with a mismatched object configuration.
+    def track_y(x):
+        return 71.15 - 0.00105*(x-128.0)**2
+    oy = np.array([track_y(x) for x in ox], np.float32)
+    py = np.array([track_y(x) for x in px], np.float32)
+
+    out = src.copy().astype(np.float32)
+    # A clean center-track plate from frame 90; the camera and environment are static.
+    clean = src[90].astype(np.float32)
+    # Use pristine source material samples close to center, including original shading.
+    o_sprite, o_alpha = source_sprite(src[32], orange[32,0], orange[32,1], 7)
+    p_sprite, p_alpha = source_sprite(src[34], purple[34,0], purple[34,1], 7)
+
+    for i in range(hit, N):
+        frame = src[i].astype(np.float32).copy()
+        erase_ball(frame, clean, orange[i,0], orange[i,1], 8)
+        erase_ball(frame, clean, purple[i,0], purple[i,1], 8)
+        # Back-to-front ordering in the extremely unlikely event of silhouette overlap.
+        paste_sprite(frame, o_sprite, o_alpha, float(ox[i]), float(oy[i]))
+        paste_sprite(frame, p_sprite, p_alpha, float(px[i]), float(py[i]))
+        out[i] = frame
+
+    out = np.clip(np.rint(out), 0, 255).astype(np.uint8)
+    save(out, CASE)
+
+
+if __name__ == '__main__':
+    main()

\"\"\"Generate the 5x-purple-mass counterfactual for this PixelWorld case.

The source is retained exactly until the marbles touch. Mass has no visual effect and,
in this scene, no effect on gravity-driven free rolling before impact. At impact a 1-D
rigid-body impulse is applied with m_purple=5*m_orange; the edited paths then settle in
the shallow center of the track. Source marble sprites are reused with soft alpha and
subpixel placement so material, illumination, and camera remain consistent.
\"\"\"
from pathlib import Path
import cv2
import numpy as np
from artifact_io import save

CASE = Path(__file__).resolve().parent
N, H, W = 96, 144, 256


def detect_centers(frames):
    \"\"\"Color-segment the two rendered balls (centers in x,y).\"\"\"
    orange = np.zeros((N, 2), np.float32)
    purple = np.zeros((N, 2), np.float32)
    yy = np.indices((H, W))[0]
    for i, rgb in enumerate(frames):
        hsv = cv2.cvtColor(rgb, cv2.COLOR_RGB2HSV)
        masks = [
            ((hsv[..., 0] >= 10) & (hsv[..., 0] <= 30) &
             (hsv[..., 1] > 100) & (hsv[..., 2] > 100) &
             (yy > 46) & (yy < 79)),
            ((hsv[..., 0] >= 120) & (hsv[..., 0] <= 165) &
             (hsv[..., 1] > 70) & (yy > 46) & (yy < 79)),
        ]
        for kind, mask in enumerate(masks):
            count, labels, stats, cents = cv2.connectedComponentsWithStats(mask.astype(np.uint8))
            candidates = []
            for j in range(1, count):
                x, y, w, h, area = stats[j]
                cx, cy = cents[j]
                if kind == 0:
                    # Reject the ochre track and background plant pot.
                    ok = 15 <= area <= 180 and w <= 20 and h <= 20 and 15 < cx < 180
                else:
                    ok = 20 <= area <= 150 and w <= 14 and h <= 14 and 20 < cx < 240
                if ok:
                    candidates.append((area, cx, cy))
            if candidates:
                _, cx, cy = max(candidates)
                (orange if kind == 0 else purple)[i] = (cx, cy)
    # Fill rare segmentation gaps smoothly.
    for arr in (orange, purple):
        good = arr[:, 0] > 0
        for c in range(2):
            arr[:, c] = np.interp(np.arange(N), np.flatnonzero(good), arr[good, c])
    return orange, purple


def source_sprite(frame, cx, cy, radius=7):
    \"\"\"Extract an RGBA marble sprite; alpha is radial to avoid rectangular seams.\"\"\"
    size = radius * 2 + 1
    patch = cv2.getRectSubPix(frame, (size, size), (float(cx), float(cy)))
    y, x = np.mgrid[-radius:radius + 1, -radius:radius + 1]
    r = np.sqrt(x*x + y*y)
    # Opaque through the ball, 2-pixel antialiased falloff at its silhouette.
    alpha = np.clip(radius - r + 0.6, 0, 1)
    # Include the ball's compact contact shadow only along the lower rim.
    alpha = np.maximum(alpha, np.exp(-((x/5.0)**2 + ((y-radius+0.3)/1.0)**2)) * 0.23)
    return patch.astype(np.float32), alpha.astype(np.float32)


def erase_ball(dst, clean, cx, cy, radius=8):
    \"\"\"Restore the static track beneath a source ball with a feathered mask.\"\"\"
    x0, x1 = max(0, int(cx)-radius), min(W, int(cx)+radius+1)
    y0, y1 = max(0, int(cy)-radius), min(H, int(cy)+radius+1)
    y, x = np.mgrid[y0:y1, x0:x1]
    d = np.sqrt((x-cx)**2 + (y-cy)**2)
    a = np.clip((radius + 0.5 - d) / 1.25, 0, 1)[..., None]
    dst[y0:y1, x0:x1] = dst[y0:y1, x0:x1]*(1-a) + clean[y0:y1, x0:x1]*a


def paste_sprite(dst, sprite, alpha, cx, cy):
    \"\"\"Subpixel translate and alpha composite a sprite.\"\"\"
    sh, sw = alpha.shape
    r = sw // 2
    ix, iy = int(np.floor(cx)), int(np.floor(cy))
    fx, fy = cx-ix, cy-iy
    mat = np.float32([[1, 0, fx], [0, 1, fy]])
    sp = cv2.warpAffine(sprite, mat, (sw, sh), flags=cv2.INTER_LINEAR,
                        borderMode=cv2.BORDER_REPLICATE)
    aa = cv2.warpAffine(alpha, mat, (sw, sh), flags=cv2.INTER_LINEAR,
                        borderMode=cv2.BORDER_CONSTANT)[..., None]
    x0, y0 = ix-r, iy-r
    xa, ya = max(0, x0), max(0, y0)
    xb, yb = min(W, x0+sw), min(H, y0+sh)
    if xa < xb and ya < yb:
        sx, sy = xa-x0, ya-y0
        a = aa[sy:sy+yb-ya, sx:sx+xb-xa]
        dst[ya:yb, xa:xb] = dst[ya:yb, xa:xb]*(1-a) + sp[sy:sy+yb-ya, sx:sx+xb-xa]*a


def main():
    src = np.load(CASE / 'source_pixels.npz')['rgb']
    assert src.shape == (N, H, W, 3) and src.dtype == np.uint8
    orange, purple = detect_centers(src)

    # Source collision happens between frames 33 and 34 (0-based), when center gap
    # reaches one diameter. The heavier purple receives only 1/5 of the acceleration
    # from the orange impact. Using the observed equal-mass post-impact velocities
    # with momentum conservation gives the counterfactual post-impact velocities:
    # orange ~= -1.8 px/frame, purple ~= -0.54 px/frame.
    hit = 34
    ox, px = np.zeros(N, np.float32), np.zeros(N, np.float32)
    ox[:hit], px[:hit] = orange[:hit, 0], purple[:hit, 0]
    ox[hit], px[hit] = orange[hit, 0], purple[hit, 0]
    vo, vp = -1.80, -0.54
    # Shallow-bowl restoring acceleration plus rolling drag. Purple has the same
    # gravitational acceleration regardless of mass; mass differs only in collision.
    for i in range(hit+1, N):
        vo = 0.965*vo - 0.0055*(ox[i-1]-128.0)
        vp = 0.965*vp - 0.0055*(px[i-1]-128.0)
        ox[i] = ox[i-1] + vo
        px[i] = px[i-1] + vp
        if px[i] - ox[i] < 10.2:  # nonpenetration, in case of a later recollision
            mid = (ox[i] + px[i]) * 0.5
            ox[i], px[i] = mid-5.1, mid+5.1
            # Restitution impulse for m_orange=1, m_purple=5.
            rel, e = vo-vp, 0.55
            if rel > 0:
                j = (1+e)*rel/(1.0 + 1.0/5.0)
                vo -= j
                vp += j/5.0

    # Vertical center follows the visible track profile measured from source motion.
    # Around the center it is nearly flat; this analytic profile avoids borrowing a
    # source frame with a mismatched object configuration.
    def track_y(x):
        return 71.15 - 0.00105*(x-128.0)**2
    oy = np.array([track_y(x) for x in ox], np.float32)
    py = np.array([track_y(x) for x in px], np.float32)

    out = src.copy().astype(np.float32)
    # A clean center-track plate from frame 90; the camera and environment are static.
    clean = src[90].astype(np.float32)
    # Use pristine source material samples close to center, including original shading.
    o_sprite, o_alpha = source_sprite(src[32], orange[32,0], orange[32,1], 7)
    p_sprite, p_alpha = source_sprite(src[34], purple[34,0], purple[34,1], 7)

    for i in range(hit, N):
        frame = src[i].astype(np.float32).copy()
        erase_ball(frame, clean, orange[i,0], orange[i,1], 8)
        erase_ball(frame, clean, purple[i,0], purple[i,1], 8)
        # Back-to-front ordering in the extremely unlikely event of silhouette overlap.
        paste_sprite(frame, o_sprite, o_alpha, float(ox[i]), float(oy[i]))
        paste_sprite(frame, p_sprite, p_alpha, float(px[i]), float(py[i]))
        out[i] = frame

    out = np.clip(np.rint(out), 0, 255).astype(np.uint8)
    save(out, CASE)


if __name__ == '__main__':
    main()
