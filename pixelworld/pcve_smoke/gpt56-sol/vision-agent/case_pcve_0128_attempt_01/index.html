+"""Generate the requested counterfactual video.
+
+Frames 1--38 are copied exactly.  Beginning at frame 39 the purple marble is
+removed.  The yellow marble is unaffected until the point at which it would
+have met the purple marble a second time; from there its free motion through
+the shallow trough is continued with a trajectory fitted to its approach.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def purple_center(frame: np.ndarray) -> tuple[float, float]:
+    """Locate the saturated violet body (the highlight is enclosed later)."""
+    r, g, b = [frame[..., i].astype(np.int16) for i in range(3)]
+    seed = (
+        (b > 70) & (r > 45) & (g < 115)
+        & (b > g + 28) & (b > r + 8)
+    ).astype(np.uint8)
+    seed[:53] = 0
+    seed[84:] = 0
+    seed[:, :95] = 0
+    seed[:, 195:] = 0
+    count, labels, stats, centers = cv2.connectedComponentsWithStats(seed)
+    choices = [i for i in range(1, count) if stats[i, cv2.CC_STAT_AREA] >= 12]
+    if not choices:
+        raise RuntimeError("purple marble was not found")
+    i = max(choices, key=lambda j: stats[j, cv2.CC_STAT_AREA])
+    return tuple(float(v) for v in centers[i])
+
+
+def yellow_center(frame: np.ndarray, expected_x: float) -> tuple[float, float]:
+    """Estimate the gold sphere center inside a small predicted window."""
+    x0 = max(0, int(round(expected_x)) - 9)
+    x1 = min(frame.shape[1], int(round(expected_x)) + 10)
+    y0, y1 = 62, 82
+    crop = frame[y0:y1, x0:x1].astype(np.int16)
+    r, g, b = crop[..., 0], crop[..., 1], crop[..., 2]
+    # The marble is gold; the stricter brightness clause rejects the wood.
+    mask = (r > 155) & (g > 92) & (r > g + 22) & (g > b + 28)
+    yy, xx = np.where(mask)
+    # Use only the upper/body pixels.  Their x centroid is robust, while the
+    # geometric center follows the known surface and is more stable in y.
+    if len(xx) < 8:
+        raise RuntimeError("yellow marble was not found")
+    cx = float(np.median(xx + x0))
+    cy = 72.05 - 0.00130 * (cx - 128.0) ** 2
+    return cx, cy
+
+
+def ellipse_mask(shape, cx, cy, rx=7.0, ry=7.5, shadow=True):
+    """Mask a sphere, its antialiased fringe, and its soft contact shadow."""
+    h, w = shape
+    yy, xx = np.mgrid[:h, :w]
+    body = ((xx - cx) / rx) ** 2 + ((yy - cy) / ry) ** 2 <= 1.0
+    if shadow:
+        contact = ((xx - cx) / 8.0) ** 2 + ((yy - (cy + 6.4)) / 3.0) ** 2 <= 1.0
+        body |= contact
+    return (body.astype(np.uint8) * 255)
+
+
+def inpaint_objects(frame: np.ndarray, masks: list[np.ndarray]) -> np.ndarray:
+    """Remove small objects while retaining the white/wood surface edges."""
+    mask = np.maximum.reduce(masks) if len(masks) > 1 else masks[0]
+    # Navier-Stokes continuation preserves the long, gently curving edges of
+    # the trough better than a flat color fill.  A modest radius avoids pulling
+    # colors from across both the white liner and wooden rim.
+    return cv2.inpaint(frame, mask, 3.0, cv2.INPAINT_NS)
+
+
+def make_yellow_sprite(source: np.ndarray):
+    """Extract a clean, softly antialiased gold sphere from source frame 55."""
+    frame = source[54]
+    cx, cy = 119.0, 72.0
+    radius = 6
+    x0, x1 = int(cx) - radius, int(cx) + radius + 1
+    y0, y1 = int(cy) - radius, int(cy) + radius + 1
+    rgb = frame[y0:y1, x0:x1].copy()
+    yy, xx = np.mgrid[y0:y1, x0:x1]
+    dist = np.sqrt((xx - cx) ** 2 + (yy - cy) ** 2)
+    alpha = np.clip(6.15 - dist, 0.0, 1.0).astype(np.float32)
+    return rgb, alpha, cx - x0, cy - y0
+
+
+def add_yellow(frame: np.ndarray, sprite, cx: float, cy: float) -> np.ndarray:
+    """Place the extracted sphere at a subpixel position plus contact shadow."""
+    rgb, alpha, anchor_x, anchor_y = sprite
+    out = frame.astype(np.float32)
+    h, w = frame.shape[:2]
+
+    # Soft shadow is applied first.  Its center follows the contact point, and
+    # is deliberately subtle because most of the source shadow lies under the
+    # sphere rather than extending to one side.
+    yy, xx = np.mgrid[:h, :w]
+    shadow = np.exp(-0.5 * (((xx - cx) / 4.8) ** 2 + ((yy - (cy + 6.0)) / 1.35) ** 2))
+    strength = (0.20 * shadow)[..., None]
+    out *= (1.0 - strength)
+
+    # Shift the 13x13 extracted source sprite at subpixel precision.  Warping
+    # both premultiplied RGB and alpha keeps the fringe clean.
+    sh, sw = alpha.shape
+    x0 = int(np.floor(cx - anchor_x))
+    y0 = int(np.floor(cy - anchor_y))
+    frac_x = float(cx - anchor_x - x0)
+    frac_y = float(cy - anchor_y - y0)
+    mat = np.array([[1.0, 0.0, frac_x], [0.0, 1.0, frac_y]], np.float32)
+    a = cv2.warpAffine(alpha, mat, (sw + 1, sh + 1), flags=cv2.INTER_LINEAR,
+                       borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+    premul = rgb.astype(np.float32) * alpha[..., None]
+    p = cv2.warpAffine(premul, mat, (sw + 1, sh + 1), flags=cv2.INTER_LINEAR,
+                       borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+    ys = slice(y0, y0 + sh + 1)
+    xs = slice(x0, x0 + sw + 1)
+    aa = a[..., None]
+    out[ys, xs] = out[ys, xs] * (1.0 - aa) + p
+    return np.clip(np.rint(out), 0, 255).astype(np.uint8)
+
+
+def generate() -> np.ndarray:
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    frames = source.copy()
+    sprite = make_yellow_sprite(source)
+
+    # Approximate factual yellow centers, needed only to remove it after the
+    # counterfactual path separates.  Tracking is initialized from its stable
+    # late position and then done locally in every source frame.
+    factual_guess = 121.0
+
+    for index in range(38, len(frames)):  # source frame 39 onward
+        frame = source[index]
+        pcx, pcy = purple_center(frame)
+        masks = [ellipse_mask(frame.shape[:2], pcx, pcy)]
+
+        # Until frame 55 the yellow motion is factual and can remain pixel
+        # perfect.  Frame 56 is the onset of the collision that is absent here.
+        if index >= 55:
+            ycx, ycy = yellow_center(frame, factual_guess)
+            factual_guess = ycx
+            masks.append(ellipse_mask(frame.shape[:2], ycx, ycy, 7.0, 7.7))
+
+        edited = inpaint_objects(frame, masks)
+
+        if index >= 55:
+            # Damped-free central oscillation fitted to frames 43--55.  There
+            # is negligible visible damping over the remaining 1.7 seconds.
+            t = (index + 1) - 43.0  # frame 43 is the left turning point
+            cx = 128.0 - 23.82 * np.cos(0.0970 * t)
+            cy = 72.05 - 0.00130 * (cx - 128.0) ** 2
+            edited = add_yellow(edited, sprite, cx, cy)
+
+        frames[index] = edited
+
+    assert frames.shape == (96, 144, 256, 3)
+    assert frames.dtype == np.uint8
+    assert np.array_equal(frames[:38], source[:38])
+    return frames
+
+
+if __name__ == "__main__":
+    result = generate()
+    artifact_io.save(result, CASE_DIR)
+    print("saved", result.shape, result.dtype)
