    plate=cv2.inpaint(cv2.cvtColor(frame,cv2.COLOR_RGB2BGR),mask,3,cv2.INPAINT_TELEA)
    plate=cv2.cvtColor(plate,cv2.COLOR_BGR2RGB)

    # Texture frame indexed by travelled distance preserves physically slower
    # spin.  At rest, hold the final orientation without temporal shimmer.
    ti=int(np.clip(round(q[i]),0,N-1))
    tx,ty=sx[ti],sy[ti]
    r=18
    x0=max(0,int(np.floor(tx-r))); x1=min(W,int(np.ceil(tx+r+1)))
    y0=max(0,int(np.floor(ty-r))); y1=min(H,int(np.ceil(ty+r+1)))
    patch=src[ti,y0:y1,x0:x1].copy()
    ph,pw=patch.shape[:2]
    py,px=np.ogrid[:ph,:pw]
    # Soft object matte includes soccer ball and its contact shadow, but fades
    # over 1.5 pixels to avoid cutout edges on the textured grass.
    ex=(px-(tx-x0))/15.0; ey=(py-(ty-y0+1))/12.0
    dist=np.sqrt(ex*ex+ey*ey)
    alpha=np.clip((1.08-dist)/0.13,0,1).astype(np.float32)
    alpha=cv2.GaussianBlur(alpha,(0,0),0.55)

    # Fractional subpixel placement avoids stutter during final deceleration.
    nx,ny=cx[i],cy[i]
    dx=nx-tx; dy=ny-ty
    M=np.float32([[1,0,dx],[0,1,dy]])
    warped=cv2.warpAffine(patch,M,(W,H),flags=cv2.INTER_LINEAR,borderMode=cv2.BORDER_CONSTANT,borderValue=0)
    wa=cv2.warpAffine(alpha,M,(W,H),flags=cv2.INTER_LINEAR,borderMode=cv2.BORDER_CONSTANT,borderValue=0)
    out[i]=np.clip(plate*(1-wa[...,None])+warped*wa[...,None],0,255).astype(np.uint8)

artifact_io.save(out, ROOT)
