+#!/usr/bin/env python3
+"""Generate the 10x-blue-mallet-mass counterfactual.
+
+The camera and the table are static.  We retain the source through the first
+contact, remove the three source mallets after that point, and composite the
+same source-rendered mallets along mass-dependent collision trajectories.
+The collision response uses the 1-D elastic mass ratios projected onto the
+observed direction of travel.  Thus the heavy blue mallet keeps 9/11 of its
+speed at a collision with a normal mallet, while transferring 20/11 of the
+ordinary equal-mass response to the struck mallet.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+
+
+def soft_shape_mask(kind: str, shape=(144, 256)) -> np.ndarray:
+    """Hand-fit alpha silhouette at the object's frame-0 source location."""
+    m = np.zeros(shape, np.uint8)
+    if kind == "red":
+        # broad circular foot and its raised grip
+        cv2.ellipse(m, (140, 70), (10, 5), 0, 0, 360, 255, -1)
+        cv2.ellipse(m, (140, 64), (4, 6), 0, 0, 360, 255, -1)
+    elif kind == "blue":
+        cv2.ellipse(m, (167, 61), (10, 5), 0, 0, 360, 255, -1)
+        cv2.ellipse(m, (167, 55), (4, 6), 0, 0, 360, 255, -1)
+    elif kind == "white":
+        cv2.ellipse(m, (106, 82), (10, 5), 0, 0, 360, 255, -1)
+        cv2.ellipse(m, (106, 76), (4, 6), 0, 0, 360, 255, -1)
+    else:
+        raise ValueError(kind)
+    # A short feather retains the antialiasing already present in the source.
+    return cv2.GaussianBlur(m, (3, 3), 0.55).astype(np.float32) / 255.0
+
+
+def source_positions(i: int):
+    """Approximate centers of the factual source objects (for clean-up)."""
+    v0 = np.array([-1.43, 0.55], np.float32)
+    b0 = np.array([167.0, 61.0], np.float32)
+    r0 = np.array([140.0, 70.0], np.float32)
+    w0 = np.array([106.0, 82.0], np.float32)
+
+    b = b0 + v0 * min(i, 14.25)
+    if i <= 14:
+        r = r0
+    else:
+        r = r0 + np.array([-1.78, 0.68], np.float32) * min(i - 14.5, 15.4)
+
+    if i <= 30:
+        w = w0
+    else:
+        w = w0 + np.array([-2.03, 0.78], np.float32) * (i - 30)
+    return b, r, w
+
+
+def counterfactual_positions(t: float):
+    """Projected collision solution for masses (blue, red, white)=(10,1,1)."""
+    # Frame 14.5 is the observed first blue/red contact.
+    tc = 14.5
+    b_contact = np.array([146.27, 68.02], np.float32)
+    r_start = np.array([140.0, 70.0], np.float32)
+    white_stop = np.array([112.6, 80.45], np.float32)
+    white_start = np.array([106.0, 82.0], np.float32)
+
+    v_blue_in = np.array([-1.43, 0.55], np.float32)
+    # Calibrated from the equal-mass factual red trajectory.  Raising the
+    # striker from 1x to 10x multiplies transferred speed by 20/11.
+    v_red_equal = np.array([-1.78, 0.68], np.float32)
+    v_red_fast = v_red_equal * (20.0 / 11.0)
+    v_blue_mid = v_blue_in * (9.0 / 11.0)
+
+    # Red reaches the resting white mallet here and gives it its velocity.
+    t_rw = tc + (white_stop[0] - r_start[0]) / v_red_fast[0]
+
+    # It then waits until the heavy blue mallet catches it.  Contact occurs
+    # with the projected mallet centers about eight pixels apart.
+    desired_blue_x = white_stop[0] + 8.0
+    t_br2 = tc + (desired_blue_x - b_contact[0]) / v_blue_mid[0]
+    blue_second = b_contact + v_blue_mid * (t_br2 - tc)
+
+    if t <= t_rw:
+        red = r_start + v_red_fast * max(0.0, t - tc)
+        white = white_start.copy()
+    else:
+        red = white_stop.copy()
+        white = white_start + v_red_fast * (t - t_rw)
+
+    if t <= t_br2:
+        blue = b_contact + v_blue_mid * max(0.0, t - tc)
+    else:
+        # A second 10:1 collision: heavy blue keeps another 9/11 and sends
+        # the red mallet forward at 20/11 of blue's incoming speed.
+        blue = blue_second + (v_blue_mid * (9.0 / 11.0)) * (t - t_br2)
+        red = white_stop + (v_blue_mid * (20.0 / 11.0)) * (t - t_br2)
+    return blue, red, white
+
+
+def cleanup_mask(i: int, shape=(144, 256)) -> np.ndarray:
+    """Mask factual objects and their compact contact shadows for inpainting."""
+    m = np.zeros(shape, np.uint8)
+    for p in source_positions(i):
+        x, y = np.rint(p).astype(int)
+        # Union of foot, grip, and the soft lower-right source shadow.
+        cv2.ellipse(m, (x, y), (12, 7), 0, 0, 360, 255, -1)
+        cv2.ellipse(m, (x, y - 6), (6, 7), 0, 0, 360, 255, -1)
+        cv2.ellipse(m, (x + 5, y + 4), (11, 5), 0, 0, 360, 255, -1)
+    return m
+
+
+def translate_layer(rgb: np.ndarray, alpha: np.ndarray, delta) -> tuple[np.ndarray, np.ndarray]:
+    dx, dy = map(float, delta)
+    mat = np.array([[1.0, 0.0, dx], [0.0, 1.0, dy]], np.float32)
+    moved_rgb = cv2.warpAffine(
+        rgb, mat, (rgb.shape[1], rgb.shape[0]), flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT, borderValue=(0, 0, 0)
+    )
+    moved_a = cv2.warpAffine(
+        alpha, mat, (alpha.shape[1], alpha.shape[0]), flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT, borderValue=0
+    )
+    return moved_rgb, moved_a
+
+
+def composite_object(frame: np.ndarray, source0: np.ndarray, alpha0: np.ndarray,
+                     original_center, center) -> np.ndarray:
+    """Composite shadow and source-rendered mallet at a subpixel center."""
+    h, w = frame.shape[:2]
+    cx, cy = map(float, center)
+    if cx < -15 or cx > w + 15 or cy < -12 or cy > h + 12:
+        return frame
+
+    # Shadow follows the light direction seen in the factual rendering.
+    sh = np.zeros((h, w), np.uint8)
+    cv2.ellipse(sh, (int(round(cx + 5)), int(round(cy + 4))), (11, 4),
+                7, 0, 360, 42, -1)
+    sh = cv2.GaussianBlur(sh, (7, 7), 2.0).astype(np.float32) / 255.0
+    out = frame.astype(np.float32)
+    out *= (1.0 - sh[..., None] * 0.48)
+
+    sprite, a = translate_layer(source0, alpha0,
+                                np.asarray(center) - np.asarray(original_center))
+    aa = np.clip(a, 0.0, 1.0)[..., None]
+    out = sprite.astype(np.float32) * aa + out * (1.0 - aa)
+    return np.clip(out, 0, 255).astype(np.uint8)
+
+
+def generate() -> np.ndarray:
+    src = np.load(ROOT / "source_pixels.npz")["rgb"]
+    assert src.shape == (96, 144, 256, 3) and src.dtype == np.uint8
+
+    source0 = src[0]
+    masks = {
+        "red": soft_shape_mask("red"),
+        "blue": soft_shape_mask("blue"),
+        "white": soft_shape_mask("white"),
+    }
+    original = {
+        "red": np.array([140.0, 70.0]),
+        "blue": np.array([167.0, 61.0]),
+        "white": np.array([106.0, 82.0]),
+    }
+
+    out = src.copy()
+    # Frames 0..14 are the factual approach and first-contact history.  Mass
+    # has no visible consequence before impulse exchange.
+    for i in range(15, 96):
+        mask = cleanup_mask(i)
+        # Telea reconstructs the lavender surface and continues nearby rink
+        # markings through the small removed silhouettes.
+        clean = cv2.inpaint(src[i], mask, 3.0, cv2.INPAINT_TELEA)
+        blue, red, white = counterfactual_positions(float(i))
+
+        # Back-to-front order agrees with the projected direction of travel;
+        # during contacts the forward (red/white) object naturally occludes.
+        frame = clean
+        frame = composite_object(frame, source0, masks["blue"], original["blue"], blue)
+        frame = composite_object(frame, source0, masks["red"], original["red"], red)
+        frame = composite_object(frame, source0, masks["white"], original["white"], white)
+        out[i] = frame
+    return out
+
+
+if __name__ == "__main__":
+    frames = generate()
+    artifact_io.save(frames, ROOT)
+    print(f"saved {frames.shape[0]} RGB frames to {ROOT}")
