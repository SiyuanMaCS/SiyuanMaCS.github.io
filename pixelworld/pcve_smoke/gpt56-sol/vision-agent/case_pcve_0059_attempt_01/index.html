+"""Generate the 5x-mass milk-carton counterfactual.
+
+The scene is a nearly one-dimensional rigid-body collision.  The red carton
+first pushes the cup, and that pair reaches the green/white milk carton at
+about frame 10.  With equal-looking source objects the source's post-impact
+speed is consistent with a three-body collision.  Changing the last body's
+mass from 1 to 5 changes that speed by 3/7.  Its lower impact speed, followed
+by the same tabletop friction, gives a much shorter stopping distance.
+
+The source renderer is reused for all appearance.  After impact, a clean
+plate is assembled from source frames in which each part of the table is
+uncovered, and a source foreground state is shifted onto the simulated
+trajectory.  This keeps antialiasing, contact ordering, shadows, and all
+unaffected scene details photographic rather than redrawing the objects.
+"""
+
+from pathlib import Path
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+
+
+def green_center(frame: np.ndarray) -> float:
+    """Subpixel x-center of the milk carton's saturated green label."""
+    f = frame.astype(np.float32)
+    m = ((f[:, :, 1] > 52) &
+         (f[:, :, 1] > 1.18 * f[:, :, 0]) &
+         (f[:, :, 1] > 1.04 * f[:, :, 2]))
+    m[:54] = False
+    m[91:] = False
+    y, x = np.where(m)
+    return float(x.mean())
+
+
+def clean_plate(src: np.ndarray) -> np.ndarray:
+    """Make an object-free background from two genuinely exposed views."""
+    # At the end, x<200 has been uncovered; at the beginning, x>202 is clear.
+    left = src[95].astype(np.float32)
+    right = src[0].astype(np.float32)
+    plate = left.copy()
+    for x in range(src.shape[2]):
+        # Broad smooth seam, situated in empty tabletop in both source views.
+        a = np.clip((x - 196.0) / 10.0, 0.0, 1.0)
+        a = a * a * (3.0 - 2.0 * a)
+        plate[:, x] = (1.0 - a) * left[:, x] + a * right[:, x]
+    return np.clip(plate + 0.5, 0, 255).astype(np.uint8)
+
+
+def foreground_rgba(frame: np.ndarray, plate: np.ndarray, x0: int, x1: int):
+    """Extract the rendered three-object contact group with soft source edges."""
+    h, w = frame.shape[:2]
+    # Pixel difference gives an excellent matte because the background is
+    # uniform, while retaining the subtle contact shadows under the objects.
+    d = np.max(np.abs(frame.astype(np.int16) - plate.astype(np.int16)), axis=2)
+    roi = np.zeros((h, w), np.uint8)
+    roi[45:92, max(0, x0):min(w, x1)] = 1
+
+    core = ((d >= 8) & (roi > 0)).astype(np.uint8)
+    # Fill tiny holes inside neutral/white carton faces but do not expand far
+    # enough to merge unrelated chair or tabletop pixels.
+    core = cv2.morphologyEx(core, cv2.MORPH_CLOSE,
+                            cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (3, 3)))
+    support = cv2.dilate(core, np.ones((3, 3), np.uint8), iterations=1)
+    alpha = np.clip((d.astype(np.float32) - 1.5) / 10.0, 0.0, 1.0)
+    alpha *= support
+    alpha = cv2.GaussianBlur(alpha, (3, 3), 0.42)
+    alpha *= roi
+    return frame, np.clip(alpha, 0.0, 1.0)
+
+
+def translated_over(dst: np.ndarray, rgb: np.ndarray, alpha: np.ndarray,
+                    dx: float) -> np.ndarray:
+    """Subpixel translate an RGBA foreground and composite it over dst."""
+    h, w = dst.shape[:2]
+    M = np.float32([[1, 0, dx], [0, 1, 0]])
+    moved_rgb = cv2.warpAffine(rgb, M, (w, h), flags=cv2.INTER_LINEAR,
+                               borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+    moved_a = cv2.warpAffine(alpha, M, (w, h), flags=cv2.INTER_LINEAR,
+                             borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+    a = moved_a[:, :, None].astype(np.float32)
+    result = moved_rgb.astype(np.float32) * a + dst.astype(np.float32) * (1-a)
+    return np.clip(result + 0.5, 0, 255).astype(np.uint8)
+
+
+def generate() -> np.ndarray:
+    src = np.load(ROOT / "source_pixels.npz")["rgb"]
+    assert src.shape == (96, 144, 256, 3) and src.dtype == np.uint8
+    plate = clean_plate(src)
+    greens = np.array([green_center(f) for f in src])
+
+    out = src.copy()
+    contact_frame = 9
+    contact_x = greens[contact_frame]
+
+    for t in range(10, 96):
+        n = float(t - contact_frame)
+
+        # After the impact, total mass is 1+1+5 rather than 1+1+1.
+        # 2.70 px/frame is the source incoming momentum divided by 7.
+        # Coulomb friction then brings the compact group to rest in ~11 frames.
+        v0, decel = 2.70, 0.25
+        stop_t = v0 / decel
+        q = min(n, stop_t)
+        displacement = v0 * q - 0.5 * decel * q * q
+        target_green = contact_x + displacement
+
+        # Advance through the source's short contact/settling sequence faster;
+        # its geometry supplies realistic occlusion while position comes from
+        # the altered-mass dynamics above.
+        s = min(35, 10 + int(round(1.55 * (t - 10))))
+        source_green = greens[s]
+        dx = target_green - source_green
+
+        # Replace only the moving-object corridor, softly at its boundaries.
+        base = src[t].astype(np.float32)
+        corridor = np.zeros((144, 256), np.float32)
+        corridor[48:91, 132:256] = 1.0
+        corridor = cv2.GaussianBlur(corridor, (9, 9), 1.6)[:, :, None]
+        base = (base * (1.0 - corridor) + plate.astype(np.float32) * corridor)
+        base = np.clip(base + 0.5, 0, 255).astype(np.uint8)
+
+        # The source group extends from about 30 px left of green to 14 px
+        # right.  Padding includes its antialiased edges and grounding shadow.
+        x0 = int(np.floor(source_green - 34))
+        x1 = int(np.ceil(source_green + 16))
+        fg, a = foreground_rgba(src[s], plate, x0, x1)
+        out[t] = translated_over(base, fg, a, dx)
+
+    return np.ascontiguousarray(out, dtype=np.uint8)
+
+
+if __name__ == "__main__":
+    frames = generate()
+    artifact_io.save(frames, ROOT)
+    print("saved", frames.shape, frames.dtype, "to", ROOT)
