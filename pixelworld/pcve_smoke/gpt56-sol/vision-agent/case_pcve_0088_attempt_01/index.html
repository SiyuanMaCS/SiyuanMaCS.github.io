+"""Generate the high-friction cue-ball counterfactual.
+
+The camera is locked and the only source motion is the two pool balls.  In the
+factual clip the cue ball has an approximately constant image-space
+deceleration of 0.127 px/frame^2 before it reaches the object ball.  Multiplying
+that acceleration by ten makes it stop after about seven pixels, so the object
+ball is never struck.  We remove the cue ball from the initial clean scene and
+re-composite its original, shaded appearance along that shorter trajectory.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def translated(image: np.ndarray, dx: float, dy: float) -> np.ndarray:
+    """Subpixel-translate a full-frame float image without wrapping."""
+    matrix = np.array([[1.0, 0.0, dx], [0.0, 1.0, dy]], dtype=np.float32)
+    return cv2.warpAffine(
+        image,
+        matrix,
+        (image.shape[1], image.shape[0]),
+        flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=0,
+    )
+
+
+def make_frames() -> np.ndarray:
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    first = source[0]
+    height, width = first.shape[:2]
+
+    # Build the unobstructed felt directly underneath the initial cue ball.
+    # The small inpaint region lies wholly on the smoothly shaded table cloth;
+    # the stationary dark object ball and every other scene pixel stay intact.
+    cue_center = np.array([85.5, 94.5], dtype=np.float32)
+    yy, xx = np.mgrid[:height, :width]
+    remove_mask = (
+        ((xx - cue_center[0]) / 10.2) ** 2
+        + ((yy - (cue_center[1] + 0.7)) / 11.0) ** 2
+        <= 1.0
+    ).astype(np.uint8) * 255
+    clean = cv2.inpaint(first, remove_mask, 4.0, cv2.INPAINT_TELEA)
+
+    # Derive a soft, source-faithful matte from the change caused by the ball.
+    # It includes the contact shadow and antialiased edge, not just a white disk.
+    difference = np.max(
+        np.abs(first.astype(np.int16) - clean.astype(np.int16)), axis=2
+    ).astype(np.float32)
+    vicinity = (
+        ((xx - cue_center[0]) / 11.0) ** 2
+        + ((yy - (cue_center[1] + 0.5)) / 12.0) ** 2
+        <= 1.0
+    )
+    seed = ((difference > 2.0) & vicinity).astype(np.float32)
+    alpha = cv2.GaussianBlur(seed, (5, 5), 0.9)
+    alpha[difference > 12.0] = 1.0
+    alpha *= vicinity.astype(np.float32)
+    alpha = np.clip(alpha, 0.0, 1.0)
+
+    # Source pre-impact fit (pixels per frame): v0 ~= 4.24 and a ~= 0.127.
+    # Kinetic friction is proportional to deceleration, hence 10*a here.
+    direction = np.array([0.868, -0.497], dtype=np.float32)
+    direction /= np.linalg.norm(direction)
+    initial_speed = 4.24
+    acceleration = 10.0 * 0.127
+    stop_time = initial_speed / acceleration
+    stop_distance = initial_speed * initial_speed / (2.0 * acceleration)
+
+    clean_f = clean.astype(np.float32)
+    first_f = first.astype(np.float32)
+    output = np.empty_like(source)
+
+    for frame_index in range(len(source)):
+        t = float(frame_index)
+        if t < stop_time:
+            distance = initial_speed * t - 0.5 * acceleration * t * t
+        else:
+            distance = stop_distance
+        dx, dy = direction * distance
+
+        moved_alpha = translated(alpha, float(dx), float(dy))[:, :, None]
+        moved_ball = translated(first_f, float(dx), float(dy))
+        composed = moved_ball * moved_alpha + clean_f * (1.0 - moved_alpha)
+        output[frame_index] = np.clip(np.rint(composed), 0, 255).astype(np.uint8)
+
+    # At intervention time the state is exactly the factual initial state.
+    output[0] = first
+    return output
+
+
+if __name__ == "__main__":
+    frames = make_frames()
+    artifact_io.save(frames, CASE_DIR)
+    print(
+        f"saved {frames.shape[0]} frames at {frames.shape[2]}x{frames.shape[1]} "
+        "(cue ball stops before collision)"
+    )
