+"""Generate the requested six-times-initial-speed counterfactual video.
+
+The source scene is static apart from a basketball.  This script removes the
+factual basketball from each source frame, then composites the same ball along
+a trajectory whose initial velocity (both x and y components) is multiplied by
+six.  Gravity, the observed floor height, and rebound coefficient are retained.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+from scipy import ndimage
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+N, H, W = 96, 144, 256
+
+
+def detect_source_centers(frames):
+    """Track the source ball with a circle detector and a trajectory prior."""
+    centers = []
+    for i, rgb in enumerate(frames):
+        gray = cv2.cvtColor(rgb, cv2.COLOR_RGB2GRAY)
+        circles = cv2.HoughCircles(
+            gray,
+            cv2.HOUGH_GRADIENT,
+            dp=1,
+            minDist=20,
+            param1=80,
+            param2=14,
+            minRadius=8,
+            maxRadius=14,
+        )
+        # The prior only disambiguates floor texture at the late, low bounces.
+        expected_x = 127.5 + 0.90 * min(i, 12) + 0.25 * max(i - 12, 0)
+        candidates = [] if circles is None else circles[0]
+        candidates = [
+            c
+            for c in candidates
+            if 105 < c[0] < 190 and 5 < c[1] < 100 and 8 <= c[2] <= 14
+        ]
+        if candidates:
+            c = min(candidates, key=lambda q: abs(float(q[0]) - expected_x))
+            centers.append((float(c[0]), float(c[1])))
+        elif centers:
+            centers.append((expected_x, centers[-1][1]))
+        else:
+            centers.append((127.5, 21.7))
+    return centers
+
+
+def clean_source_frames(frames, centers):
+    """Remove the factual ball and its soft contact shadow frame by frame."""
+    clean = np.empty_like(frames)
+    for i, (rgb, (cx, cy)) in enumerate(zip(frames, centers)):
+        if i == 0:
+            # Frame zero is output verbatim, so its cleaned version is unused.
+            clean[i] = rgb
+            continue
+        mask = np.zeros((H, W), np.uint8)
+        cv2.circle(mask, (round(cx), round(cy)), 15, 255, -1, cv2.LINE_AA)
+        if cy > 62:
+            sy = min(96, round(cy + 12))
+            cv2.ellipse(
+                mask, (round(cx + 1), sy), (16, 6), 0, 0, 360, 255, -1, cv2.LINE_AA
+            )
+        # TELEA works especially cleanly here because the wall is smooth and
+        # the floor grain is locally linear.
+        bgr = cv2.cvtColor(rgb, cv2.COLOR_RGB2BGR)
+        restored = cv2.inpaint(bgr, mask, 4, cv2.INPAINT_TELEA)
+        clean[i] = cv2.cvtColor(restored, cv2.COLOR_BGR2RGB)
+    return clean
+
+
+def make_ball_sprite(frame0):
+    """Extract the original antialiased basketball as an extended RGB sprite."""
+    cx, cy = 127.5, 21.7
+    size = 35
+    sprite = cv2.getRectSubPix(frame0, (size, size), (cx, cy))
+    yy, xx = np.mgrid[:size, :size]
+    rr = np.hypot(xx - (size - 1) / 2, yy - (size - 1) / 2)
+    alpha = np.clip((12.35 - rr) / 1.15, 0.0, 1.0).astype(np.float32)
+
+    # Extrapolate edge colors into transparent pixels so affine interpolation
+    # cannot pull the original gray wall into the rotating ball edge.
+    solid = rr <= 11.7
+    _, indices = ndimage.distance_transform_edt(~solid, return_indices=True)
+    extended = sprite.copy()
+    outside = ~solid
+    extended[outside] = sprite[indices[0][outside], indices[1][outside]]
+    return extended, alpha
+
+
+def simulate_trajectory():
+    """Semi-analytically integrate the altered rigid-body trajectory."""
+    x0, y0 = 127.5, 21.7
+    # Source fit: vx=.904 px/frame, vy=.454 px/frame, g=.752 px/frame^2.
+    vx, vy = 6.0 * 0.904, 6.0 * 0.454
+    gravity = 0.752
+    floor_y = 82.0
+    restitution = 0.78
+    x, y = x0, y0
+    positions = [(x, y)]
+
+    # Fine substeps keep the collision instant accurate while positions remain
+    # sampled exactly at the requested 24 fps frame times.
+    substeps = 16
+    dt = 1.0 / substeps
+    for _ in range(1, N):
+        for _ in range(substeps):
+            vy += gravity * dt
+            x += vx * dt
+            y += vy * dt
+            if y > floor_y and vy > 0:
+                y = floor_y - (y - floor_y)
+                vy = -vy * restitution
+        positions.append((x, y))
+    return positions
+
+
+def add_shadow(frame, cx, cy):
+    """Composite a subtle floor shadow consistent with the source lighting."""
+    # The ball's observed contact point is around y=94 (center 82 + radius 12).
+    distance = max(0.0, 82.0 - cy)
+    strength = 0.14 * np.exp(-distance / 15.0)
+    if strength < 0.006 or cx < -20 or cx > W + 20:
+        return frame
+    yy, xx = np.mgrid[:H, :W]
+    sx = 7.0 + min(distance, 25.0) * 0.16
+    sy = 2.2 + min(distance, 25.0) * 0.055
+    shadow = np.exp(-0.5 * (((xx - (cx + 1.5)) / sx) ** 2 + ((yy - 94.5) / sy) ** 2))
+    a = (strength * shadow)[..., None]
+    return np.clip(frame.astype(np.float32) * (1.0 - a), 0, 255).astype(np.uint8)
+
+
+def composite_ball(frame, sprite, alpha, cx, cy, angle_deg):
+    """Rotate and alpha-composite the extracted source basketball."""
+    size = sprite.shape[0]
+    center = ((size - 1) / 2, (size - 1) / 2)
+    matrix = cv2.getRotationMatrix2D(center, angle_deg, 1.0)
+    rgb = cv2.warpAffine(
+        sprite,
+        matrix,
+        (size, size),
+        flags=cv2.INTER_CUBIC,
+        borderMode=cv2.BORDER_REPLICATE,
+    )
+    a = cv2.warpAffine(
+        alpha,
+        matrix,
+        (size, size),
+        flags=cv2.INTER_CUBIC,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=0,
+    )
+    a = np.clip(a, 0.0, 1.0)
+
+    left = int(np.floor(cx - center[0]))
+    top = int(np.floor(cy - center[1]))
+    x0, y0 = max(0, left), max(0, top)
+    x1, y1 = min(W, left + size), min(H, top + size)
+    if x0 >= x1 or y0 >= y1:
+        return frame
+    sx0, sy0 = x0 - left, y0 - top
+    sx1, sy1 = sx0 + (x1 - x0), sy0 + (y1 - y0)
+    aa = a[sy0:sy1, sx0:sx1, None]
+    dst = frame[y0:y1, x0:x1].astype(np.float32)
+    src = rgb[sy0:sy1, sx0:sx1].astype(np.float32)
+    frame[y0:y1, x0:x1] = np.clip(dst * (1.0 - aa) + src * aa, 0, 255).astype(
+        np.uint8
+    )
+    return frame
+
+
+def generate():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (N, H, W, 3) and source.dtype == np.uint8
+    source_centers = detect_source_centers(source)
+    clean = clean_source_frames(source, source_centers)
+    sprite, alpha = make_ball_sprite(source[0])
+    trajectory = simulate_trajectory()
+
+    output = np.empty_like(source)
+    output[0] = source[0]  # intervention changes velocity, not initial placement
+    # Rolling rate scales with translational speed: source ~4.5 deg/frame,
+    # therefore the six-times-speed ball turns about 27 deg each frame.
+    for i in range(1, N):
+        cx, cy = trajectory[i]
+        frame = add_shadow(clean[i].copy(), cx, cy)
+        output[i] = composite_ball(frame, sprite, alpha, cx, cy, 27.0 * i)
+    return output
+
+
+if __name__ == "__main__":
+    frames = generate()
+    artifact_io.save(frames, CASE_DIR)
+    print(f"saved {frames.shape} {frames.dtype} to {CASE_DIR}")
