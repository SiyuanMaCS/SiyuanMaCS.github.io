+"""Generate the friction counterfactual for this PixelWorld case.
+
+The source is a locked-off shot.  We reconstruct its static plate from the
+pre-impact frames, recover the red/blue ball trajectories, and retime the
+dynamic motion with lower downhill acceleration.  This keeps the rendered
+objects and scene appearance from the supplied pixels while producing smooth
+sub-frame motion at the requested 24 fps.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def components(mask, min_area=20):
+    """Return useful connected components as (area, center, bbox)."""
+    count, _, stats, centers = cv2.connectedComponentsWithStats(
+        mask.astype(np.uint8), connectivity=8
+    )
+    out = []
+    for label in range(1, count):
+        x, y, w, h, area = stats[label]
+        if area >= min_area:
+            out.append((int(area), centers[label].astype(np.float32), (x, y, w, h)))
+    return out
+
+
+def recover_tracks(source):
+    """Track the two colored dynamic balls directly from supplied RGB pixels."""
+    red_track, blue_track = [], []
+    prev_red = np.array([179.0, 40.5], np.float32)
+    prev_blue = np.array([24.5, 88.5], np.float32)
+
+    for frame in source:
+        hsv = cv2.cvtColor(frame, cv2.COLOR_RGB2HSV)
+        # The red restriction to y<102 rejects reddish wood and furniture.
+        red_mask = (((hsv[..., 0] < 11) | (hsv[..., 0] > 172)) &
+                    (hsv[..., 1] > 105) & (hsv[..., 2] > 65))
+        red_mask[102:] = False
+        reds = [c for c in components(red_mask, 45) if c[2][2] <= 25 and c[2][3] <= 30]
+        if reds:
+            # Continuity disambiguates the ball from small warm-colored details.
+            chosen = min(reds, key=lambda c: np.linalg.norm(c[1] - prev_red))
+            prev_red = chosen[1]
+        red_track.append(prev_red.copy())
+
+        blue_mask = ((hsv[..., 0] > 91) & (hsv[..., 0] < 126) &
+                     (hsv[..., 1] > 65) & (hsv[..., 2] > 45))
+        blues = [c for c in components(blue_mask, 30) if c[2][2] <= 28 and c[2][3] <= 28]
+        if blues:
+            chosen = min(blues, key=lambda c: np.linalg.norm(c[1] - prev_blue))
+            prev_blue = chosen[1]
+            blue_track.append(prev_blue.copy())
+        else:
+            # Once hit, the blue ball travels left beyond the picture edge.
+            prev_blue = np.array([prev_blue[0] - 8.0, prev_blue[1]], np.float32)
+            blue_track.append(prev_blue.copy())
+
+    return np.asarray(red_track), np.asarray(blue_track)
+
+
+def interpolate(track, time):
+    lo = int(np.floor(time))
+    hi = min(lo + 1, len(track) - 1)
+    f = np.float32(time - lo)
+    return track[lo] * (1.0 - f) + track[hi] * f
+
+
+def paste_ball(canvas, source_frame, source_center, target_center, radius):
+    """Translate a native source-rendered ball with a soft antialiased edge."""
+    h, w = canvas.shape[:2]
+    yy, xx = np.mgrid[:h, :w].astype(np.float32)
+    dist = np.hypot(xx - source_center[0], yy - source_center[1])
+    # Full opacity through the rendered body, then a one-pixel feather.  The
+    # slightly larger lower radius retains the ball's native contact shadow.
+    alpha = np.clip((radius + 0.65) - dist, 0.0, 1.0).astype(np.float32)
+    matrix = np.array(
+        [[1.0, 0.0, float(target_center[0] - source_center[0])],
+         [0.0, 1.0, float(target_center[1] - source_center[1])]],
+        dtype=np.float32,
+    )
+    moved_rgb = cv2.warpAffine(
+        source_frame, matrix, (w, h), flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_REPLICATE,
+    )
+    moved_alpha = cv2.warpAffine(
+        alpha, matrix, (w, h), flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT, borderValue=0,
+    )[..., None]
+    result = canvas.astype(np.float32) * (1.0 - moved_alpha)
+    result += moved_rgb.astype(np.float32) * moved_alpha
+    return np.clip(result + 0.5, 0, 255).astype(np.uint8)
+
+
+def make_static_plates(source):
+    # Across these frames the red ball traverses the ramp, so the median is a
+    # clean static plate while the stationary yellow/blue balls remain intact.
+    plate_with_blue = np.median(source[:21], axis=0).astype(np.uint8)
+
+    # A second plate removes the initial blue ball for post-impact frames.
+    # Navier-Stokes inpainting reconstructs only a tiny patch of the table;
+    # the nearby yellow ball is deliberately outside this mask.
+    mask = np.zeros(plate_with_blue.shape[:2], np.uint8)
+    cv2.ellipse(mask, (24, 89), (11, 11), 0, 0, 360, 255, -1)
+    plate_without_blue = cv2.inpaint(plate_with_blue, mask, 3, cv2.INPAINT_NS)
+    return plate_with_blue, plate_without_blue
+
+
+def generate():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3) and source.dtype == np.uint8
+
+    red_track, blue_track = recover_tracks(source)
+    plate_blue, plate_clean = make_static_plates(source)
+
+    # On a shallow incline, tripling Coulomb friction greatly reduces the net
+    # tangential acceleration.  A retained acceleration of about 46% gives a
+    # time/trajectory scale sqrt(.46)=.68.  The impact still occurs, but later
+    # and at lower speed, which is the visible downstream consequence.
+    time_scale = 0.68
+    frames = []
+    for output_index in range(96):
+        source_time = min(time_scale * output_index, 95.0)
+        sample_index = int(np.clip(np.rint(source_time), 0, 95))
+
+        red_pos = interpolate(red_track, source_time)
+        blue_pos = interpolate(blue_track, source_time)
+
+        # Source frame 22 is where the red ball first contacts the blue one.
+        # Before then the exact stationary blue render in the plate is best;
+        # afterward use the clean plate and animate both collision participants.
+        if source_time < 21.5:
+            canvas = plate_blue.copy()
+        else:
+            canvas = plate_clean.copy()
+            if blue_pos[0] > -10.0:
+                canvas = paste_ball(
+                    canvas, source[sample_index], blue_track[sample_index],
+                    blue_pos, radius=8.8,
+                )
+
+        canvas = paste_ball(
+            canvas, source[sample_index], red_track[sample_index],
+            red_pos, radius=8.7,
+        )
+        frames.append(canvas)
+
+    frames = np.asarray(frames, dtype=np.uint8)
+    # The intervention starts at frame 1/time zero, where no displacement has
+    # yet accumulated.  Preserve that factual initial image exactly.
+    frames[0] = source[0]
+    return frames
+
+
+if __name__ == "__main__":
+    rendered = generate()
+    artifact_io.save(rendered, CASE_DIR)
+    print(f"saved {rendered.shape[0]} frames to {CASE_DIR}")
