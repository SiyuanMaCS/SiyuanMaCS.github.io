+"""Generate the 20x-red-ball counterfactual video.
+
+The intervention changes mass, not size or appearance.  Therefore the source is
+preserved exactly through the first contact with the blue ball.  At that contact
+the velocities are replaced by the 1-D rigid-body collision solution for a red
+mass of 20 and blue mass of 1.  The very small affected part of the tabletop is
+restored with spatial inpainting before source-derived ball sprites are placed
+on their counterfactual trajectories.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+N, H, W = 96, 144, 256
+
+
+def ellipse_mask(shape, center, radii, softness=0.0):
+    """Return a float ellipse mask, optionally with an antialiased boundary."""
+    h, w = shape
+    yy, xx = np.mgrid[:h, :w]
+    cx, cy = center
+    rx, ry = radii
+    d = np.sqrt(((xx - cx) / rx) ** 2 + ((yy - cy) / ry) ** 2)
+    if softness <= 0:
+        return (d <= 1.0).astype(np.float32)
+    return np.clip((1.0 + softness - d) / softness, 0.0, 1.0).astype(np.float32)
+
+
+def extract_ball(frame, center):
+    """Extract a source ball into a compact, softly edged RGBA-like sprite."""
+    size = 19
+    rgb = cv2.getRectSubPix(frame, (size, size), center).astype(np.float32)
+    yy, xx = np.mgrid[:size, :size]
+    distance = np.sqrt((xx - 9.0) ** 2 + (yy - 9.0) ** 2)
+    # The rendered balls have an approximately 8 px radius.  The narrow soft
+    # edge retains the source antialiasing without carrying tabletop pixels.
+    alpha = np.clip((8.75 - distance) / 1.0, 0.0, 1.0).astype(np.float32)
+    return rgb, alpha
+
+
+def composite_sprite(frame, sprite, center):
+    """Composite a sprite at a potentially subpixel or off-screen center."""
+    rgb, alpha = sprite
+    matrix = np.float32([[1.0, 0.0, center[0] - 9.0],
+                         [0.0, 1.0, center[1] - 9.0]])
+    layer = cv2.warpAffine(
+        rgb, matrix, (W, H), flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT, borderValue=0,
+    )
+    a = cv2.warpAffine(
+        alpha, matrix, (W, H), flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT, borderValue=0,
+    )[..., None]
+    return frame * (1.0 - a) + layer * a
+
+
+def add_contact_shadow(frame, center, strength=0.27):
+    """Add the compact, soft contact shadow visible beneath each source ball."""
+    # At tabletop height the ball center is near y=88.5 and its shadow is a
+    # narrow ellipse around y=98.  A Gaussian-like profile matches the source.
+    yy, xx = np.mgrid[:H, :W]
+    cx, cy = center
+    q = ((xx - cx) / 7.8) ** 2 + ((yy - (cy + 9.5)) / 2.7) ** 2
+    a = (strength * np.exp(-2.0 * q))[..., None].astype(np.float32)
+    return frame * (1.0 - a)
+
+
+def original_centers(frame_index):
+    """Centers of source balls after contact, measured from source pixels."""
+    red_x = {
+        23: 34.4, 24: 30.7, 25: 27.7, 26: 25.2, 27: 23.4,
+        28: 22.2, 29: 21.6, 30: 21.2, 31: 21.2, 32: 21.2,
+    }
+    # The red ball is at rest for the remainder (tiny render jitter is safely
+    # covered by the mask).
+    rx = red_x.get(frame_index, 21.3)
+    result = [(rx, 88.4, "red")]
+    if frame_index == 23:
+        result.append((14.0, 89.1, "blue"))
+    elif frame_index == 24:
+        result.append((7.6, 89.3, "blue"))
+    elif frame_index == 25:
+        result.append((4.1, 90.0, "blue"))
+    elif frame_index == 26:
+        result.append((1.5, 89.6, "blue"))
+    return result
+
+
+def erase_source_balls(frame, frame_index):
+    """Remove the factual post-collision balls and their contact shadows."""
+    mask = np.zeros((H, W), np.float32)
+    for cx, cy, _kind in original_centers(frame_index):
+        mask = np.maximum(mask, ellipse_mask((H, W), (cx, cy), (10.0, 10.2)))
+        mask = np.maximum(mask, ellipse_mask((H, W), (cx, cy + 10.0), (11.0, 6.0)))
+    # TELEA follows the surrounding horizontal wood grain well for this small
+    # region.  Keeping the mask compact avoids disturbing the nearby yellow ball.
+    binary = (mask > 0).astype(np.uint8) * 255
+    return cv2.inpaint(frame, binary, 4.0, cv2.INPAINT_TELEA).astype(np.float32)
+
+
+def generate():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (N, H, W, 3) and source.dtype == np.uint8
+    output = source.copy()
+
+    # Use clean, fully visible source renderings.  Mass has no visible marker,
+    # so source color, shading, radius and orientation remain unchanged.
+    red_sprite = extract_ball(source[24], (30.7, 88.3))
+    blue_sprite = extract_ball(source[20], (24.5, 88.6))
+
+    # Contact is first visible in source frame 22 (zero-based).  Frames 0..22
+    # remain factual: before collision mass does not change gravitational
+    # acceleration, and frame 22 represents the instant of contact.
+    contact_red_x = 38.8
+    incident_horizontal_speed = 8.6  # px/frame, measured at the ramp exit
+    red_speed = incident_horizontal_speed * (20.0 - 1.0) / (20.0 + 1.0)
+    blue_speed = incident_horizontal_speed * (2.0 * 20.0) / (20.0 + 1.0)
+
+    for fi in range(23, N):
+        canvas = erase_source_balls(source[fi], fi)
+
+        elapsed = fi - 22
+        red_center = (contact_red_x - red_speed * elapsed, 88.5)
+        blue_center = (21.2 - blue_speed * elapsed, 89.0)
+
+        # Draw rear-to-front.  Once a center is more than one radius outside the
+        # image it can no longer contribute any pixels.
+        if blue_center[0] > -10.0:
+            canvas = add_contact_shadow(canvas, blue_center, strength=0.24)
+            canvas = composite_sprite(canvas, blue_sprite, blue_center)
+        if red_center[0] > -10.0:
+            canvas = add_contact_shadow(canvas, red_center, strength=0.28)
+            canvas = composite_sprite(canvas, red_sprite, red_center)
+
+        output[fi] = np.clip(np.rint(canvas), 0, 255).astype(np.uint8)
+
+    return output
+
+
+if __name__ == "__main__":
+    frames = generate()
+    # Required output path: this writes pixels.npz, frame PNGs, inspection
+    # sheets, and the self-contained player.
+    artifact_io.save(frames, CASE_DIR)
