+"""Generate the counterfactual clip: wooden block absent from frame one.
+
+The shot is locked off.  A clean plate is recovered robustly from the 96 source
+frames after masking the two moving objects.  The source ball is retained up to
+the would-have-been impact, then its pre-impact motion is extrapolated and a
+pre-impact ball sprite is composited over the clean plate.  Thus the edit also
+accounts for the missing collision rather than merely painting out the box.
+"""
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+N, H, W = 96, 144, 256
+
+
+def ball_center(frame):
+    """Observed source-ball track (subpixel screen coordinates)."""
+    if frame <= 12:
+        # Least-squares visual track before contact.
+        return 15.5 + 7.05 * frame, 57.0 + 1.30 * frame
+    # After contact the source ball creeps behind the sliding box.
+    t = frame - 12
+    return 100.1 + 0.68 * t, 72.6 + 0.245 * t
+
+
+def box_left(frame):
+    """Conservative source-box track used only to form an erase mask."""
+    if frame <= 10:
+        return 109.0
+    if frame < 37:
+        # The box is accelerated by the collision, then comes to rest.
+        u = (frame - 10) / 27.0
+        return 109.0 + 73.0 * (1.0 - (1.0 - u) ** 1.55)
+    return 182.0
+
+
+def disk_mask(cx, cy, radius, softness=0.0):
+    yy, xx = np.mgrid[:H, :W]
+    m = ((xx - cx) ** 2 + (yy - cy) ** 2 <= radius * radius).astype(np.float32)
+    if softness:
+        m = cv2.GaussianBlur(m, (0, 0), softness)
+        m = np.clip(m, 0.0, 1.0)
+    return m
+
+
+def object_masks():
+    """Broad masks include contact shadows and anti-aliased object boundaries."""
+    masks = np.zeros((N, H, W), np.uint8)
+    for i in range(N):
+        cx, cy = ball_center(i)
+        # Ball plus its small down/right floor shadow.
+        cv2.circle(masks[i], (round(cx), round(cy)), 21, 1, -1)
+        cv2.ellipse(masks[i], (round(cx + 3), round(cy + 18)), (22, 8), 0, 0, 360, 1, -1)
+
+        bx = round(box_left(i))
+        # The actual box is about 57x47.  This deliberately extends left and
+        # down over its soft cast shadow.
+        poly = np.array([
+            [bx - 21, 54], [bx + 62, 54], [bx + 68, 109],
+            [bx + 52, 115], [bx - 24, 111],
+        ], np.int32)
+        cv2.fillPoly(masks[i], [poly], 1)
+    return masks
+
+
+def clean_plate(src, masks):
+    """Temporal median over visible background samples, with a safe fallback."""
+    work = src.astype(np.float32)
+    work[masks.astype(bool)] = np.nan
+    with np.errstate(all="ignore"):
+        plate = np.nanmedian(work, axis=0)
+    fallback = np.median(src.astype(np.float32), axis=0)
+    plate = np.where(np.isfinite(plate), plate, fallback)
+    # A tiny denoise suppresses temporal renderer grain without losing boards.
+    plate = cv2.bilateralFilter(np.clip(plate, 0, 255).astype(np.uint8), 3, 5, 3)
+    return plate
+
+
+def paste_ball(frame, source, src_center, dst_center):
+    """Subpixel-translate a real pre-impact ball with a feathered round matte."""
+    sx, sy = src_center
+    dx, dy = dst_center
+    affine = np.array([[1.0, 0.0, dx - sx], [0.0, 1.0, dy - sy]], np.float32)
+    shifted = cv2.warpAffine(source, affine, (W, H), flags=cv2.INTER_LINEAR,
+                             borderMode=cv2.BORDER_REFLECT_101)
+
+    # Interior radius is opaque; only the one-pixel silhouette fringe blends.
+    yy, xx = np.mgrid[:H, :W]
+    rr = np.sqrt((xx - dx) ** 2 + (yy - dy) ** 2)
+    alpha = np.clip((18.0 - rr) / 1.7, 0.0, 1.0).astype(np.float32)
+    a = alpha[..., None]
+    return np.clip(frame.astype(np.float32) * (1 - a) + shifted.astype(np.float32) * a,
+                   0, 255).astype(np.uint8)
+
+
+def generate():
+    src = np.load(ROOT / "source_pixels.npz")["rgb"]
+    assert src.shape == (N, H, W, 3) and src.dtype == np.uint8
+    masks = object_masks()
+    plate = clean_plate(src, masks)
+    out = src.copy()
+
+    for i in range(N):
+        bx = round(box_left(i))
+        erase = np.zeros((H, W), np.float32)
+        poly = np.array([
+            [bx - 20, 54], [bx + 62, 54], [bx + 69, 109],
+            [bx + 53, 115], [bx - 24, 111],
+        ], np.int32)
+        cv2.fillPoly(erase, [poly], 1.0)
+
+        # At/after the missing collision, remove the factual slowed ball too.
+        if i >= 13:
+            cx, cy = ball_center(i)
+            cv2.circle(erase, (round(cx), round(cy)), 21, 1.0, -1)
+            cv2.ellipse(erase, (round(cx + 3), round(cy + 18)), (22, 8),
+                        0, 0, 360, 1.0, -1)
+
+        erase = cv2.GaussianBlur(erase, (0, 0), 1.15)
+        erase = np.clip(erase, 0.0, 1.0)[..., None]
+        out[i] = np.clip(src[i].astype(np.float32) * (1 - erase) +
+                         plate.astype(np.float32) * erase, 0, 255).astype(np.uint8)
+
+        if i >= 13:
+            # Continue the measured incoming velocity.  Perspective makes the
+            # trajectory descend slightly as it crosses the floor plane.
+            dst = (15.5 + 7.05 * i, 57.0 + 1.30 * i)
+            if dst[0] - 19 < W:
+                # Cycle genuine pre-impact appearances to preserve spin.
+                j = 5 + ((i - 13) % 7)
+                out[i] = paste_ball(out[i], src[j], ball_center(j), dst)
+
+    return out
+
+
+if __name__ == "__main__":
+    frames = generate()
+    artifact_io.save(frames, ROOT)
