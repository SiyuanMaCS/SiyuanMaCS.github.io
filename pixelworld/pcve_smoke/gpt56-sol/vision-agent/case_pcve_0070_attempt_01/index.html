+import numpy as np

+import artifact_io

+CASE_DIR = Path(__file__).resolve().parent

+# Third domino is counted from the left.  With it absent at t=0, domino 2
+# falls into the empty gap and never reaches domino 4, which stays upright.
+# The source footage already contains the exact unaffected camera/background
+# and exact motion of dominoes 1 and 2.  We remove dominoes 3's pixels and
+# suppress the causally-invalid motion of domino 4 with a clean plate, then
+# restore domino 4 from its real, pre-contact source appearance.

+def soft_composite(dst, src, mask, blur=1.15):
+    m = Image.fromarray((mask.astype(np.uint8) * 255)).filter(ImageFilter.GaussianBlur(blur))
+    return np.asarray(Image.composite(Image.fromarray(src), Image.fromarray(dst), m), dtype=np.uint8)
+
+def nearest_clean_background(src, t, x0, y0, x1, y1, shift):
+    """Use the same frame's tabletop/background just to the right of the edit."""
+    out = src[t].copy()
+    patch = src[t, y0:y1, x0+shift:x1+shift].copy()
+    # match tiny horizontal illumination gradient at destination
+    diff = (src[t, y0:y1, x0-3:x0].astype(np.float32).mean((0,1)) -
+            src[t, y0:y1, x0+shift-3:x0+shift].astype(np.float32).mean((0,1)))
+    patch = np.clip(patch.astype(np.float32) + diff, 0, 255).astype(np.uint8)
+    out[y0:y1, x0:x1] = patch
+    return out
+
+def make_frames():

+    frames = src.copy()
+
+    # Build per-frame clean plates.  The strips come from nearby, object-free
+    # table in the same frame, retaining the source's moving compression/noise.
+    # Region 1 covers removed domino 3 and its source shadow (x 135..161).
+    # Region 2 covers domino 4 after source contact; domino 4 is reinserted below.
+    for t in range(len(frames)):
+        f = frames[t]
+        # Removed third domino is upright through frame 29, then falls right.
+        # A conservative swept mask avoids altering domino 2 at x<132.
+        if t < 29:
+            poly = np.zeros((144,256), bool)
+            poly[31:108, 135:160] = True
+        else:
+            # swept footprint of source domino 3 as it rotates clockwise
+            poly = np.zeros((144,256), bool)
+            poly[30:111, 132:196] = True
+        # table/background donor 38 px right (clear before contact; for later
+        # frames the final scene is reconstructed from stable pre-contact plate)
+        base_t = t if t < 29 else 20
+        donor = nearest_clean_background(src, base_t, 128, 27, 199, 112, 45)
+        # retain current frame outside intervention, but use clean plate inside
+        frames[t] = soft_composite(f, donor, poly, 1.25)
+
+    # Domino 4 must remain upright forever. Obtain its actual pixels and shadow
+    # from a median of early factual frames, which removes codec flicker while
+    # preserving photographic edges.  Its stable silhouette occupies x176..197.
+    ref_stack = src[0:16].astype(np.float32)
+    ref = np.median(ref_stack, axis=0).astype(np.uint8)
+    yy, xx = np.mgrid[:144, :256]
+    core = ((xx >= 176) & (xx <= 198) & (yy >= 32) & (yy <= 105))
+    # include the subtle contact shadow under/right of domino 4
+    shadow = (((xx-188)/25.0)**2 + ((yy-102)/10.0)**2 <= 1.0) & (yy >= 88)
+    keep4 = core | shadow
+
+    for t in range(len(frames)):
+        # before source domino 3 reaches it (roughly frame 37), the original
+        # fourth domino is already correct; restoration remains harmless and
+        # guarantees no remnants from the broad removal mask.
+        frames[t] = soft_composite(frames[t], ref, keep4, 0.85)
+
+    # Restore domino 2 over broad cleanup wherever it overlaps after frame 29.
+    # Its source trajectory is causally unchanged because it falls into the gap.
+    # Segment dark neutral object pixels in the left/middle region, then dilate

+    try:
+        import cv2

+            obj = (mean < 125) & ((mx-mn) < 65)
+            region = np.zeros((144,256), np.uint8)
+            region[30:111, 75:154] = 1
+            m = (obj.astype(np.uint8) & region)
+            m = cv2.morphologyEx(m, cv2.MORPH_CLOSE, np.ones((5,5),np.uint8))
+            m = cv2.dilate(m, np.ones((5,5),np.uint8), iterations=1).astype(bool)

+        pass
