from pathlib import Path
import numpy as np
import cv2
from artifact_io import save

CASE_DIR = Path(__file__).resolve().parent
src = np.load(CASE_DIR / \"source_pixels.npz\")[\"rgb\"]
N,H,W,_ = src.shape
yy,xx = np.mgrid[:H,:W]
frames = src.copy()

# Collision model (screen-space): the source behaves close to an equal-mass,
# nearly elastic impact. For a stationary target of 10m, conservation of
# momentum gives v_ball'=-9/11 v and v_block'=2/11 v. Before contact, mass has
# no visible effect, so frames 1..11 remain factual.
BALL_FRAME = 10                    # last clean, fully visible incoming ball
CONTACT_FRAME = 11
ball_src = src[BALL_FRAME].astype(np.float32)
bcx,bcy = 85.5,70.5

# Complete ball silhouette at frame 11 (body and dark outline, no floor).
r = np.sqrt(((xx-bcx)/17.9)**2 + ((yy-bcy)/17.9)**2)
ball_alpha = np.clip((1.035-r)/0.075,0,1).astype(np.float32)[...,None]

# The room and the massive block remain at their initial pose at impact.
# Frame 1 supplies clean floor at the ball's impact location.
room = src[0].astype(np.float32)

# Extract the entire wooden block from frame 1 with a soft polygonal silhouette.
poly = np.array([[108,61],[137,55],[165,62],[164,96],[136,104],[108,96]],np.int32)
bm = np.zeros((H,W),np.uint8)
cv2.fillPoly(bm,[poly],255)
block_alpha = cv2.GaussianBlur(bm.astype(np.float32)/255.0,(3,3),0.5)[...,None]
block_src = src[0].astype(np.float32)

# Texture used only for the narrow strip exposed as the heavy box creeps right.
# Sampling from just left preserves wall/floor perspective much better than a
# flat fill, and feathering confines it to the vacated silhouette.
texture = np.roll(room,22,axis=1)

for i in range(CONTACT_FRAME,N):
    t = i-CONTACT_FRAME

    # Heavy block: only 2/11 of the factual transfer, with floor friction. The
    # displacement approaches ~13 px (versus ~75 px in the source).
    block_dx = min(13.0, 0.34*t - 0.0019*t*t)
    Mb = np.float32([[1,0,block_dx],[0,1,0]])
    ba = cv2.warpAffine(block_alpha,Mb,(W,H),flags=cv2.INTER_LINEAR,
                        borderMode=cv2.BORDER_CONSTANT,borderValue=0)[...,None]
    bo = cv2.warpAffine(block_src,Mb,(W,H),flags=cv2.INTER_LINEAR,
                        borderMode=cv2.BORDER_CONSTANT,borderValue=0)
    # Start from initial room; conceal only the portion of the original box
    # silhouette that is no longer covered by its translated silhouette.
    vacated = np.clip(block_alpha-ba,0,1)
    frame = room*(1-vacated) + texture*vacated
    frame = frame*(1-ba) + bo*ba

    # Ball reaches contact 5 px to the right, then rebounds at 9/11 of its
    # measured incoming speed. Mild rolling resistance gently reduces speed.
    if t == 0:
        ball_dx = 5.0
    else:
        u=t-1
        travel = 3.95*u - 0.018*u*u
        travel = max(0.0, travel)
        ball_dx = 5.0-travel
    ball_dy = min(1.4,0.12*t)  # tiny visual settling onto the floor

    # Rotate consistently with leftward rolling (counter-clockwise).
    angle = max(0,t-1)*13.0
    M = cv2.getRotationMatrix2D((bcx,bcy),angle,1.0)
    M[0,2] += ball_dx
    M[1,2] += ball_dy
    a = cv2.warpAffine(ball_alpha,M,(W,H),flags=cv2.INTER_LINEAR,
                       borderMode=cv2.BORDER_CONSTANT,borderValue=0)[...,None]
    obj = cv2.warpAffine(ball_src,M,(W,H),flags=cv2.INTER_LINEAR,
                         borderMode=cv2.BORDER_CONSTANT,borderValue=0)

    # Soft contact shadow follows the rebounding ball and naturally disappears
    # when the ball exits the left edge.
    cx,cy = bcx+ball_dx,bcy+ball_dy
    sh = np.exp(-2.0*(((xx-cx)/17.0)**2+((yy-(cy+16.0))/3.0)**2))[...,None]
    frame *= (1-0.16*sh)
    frame = frame*(1-a) + obj*a
    frames[i] = np.clip(frame,0,255).astype(np.uint8)

save(frames.astype(np.uint8), CASE_DIR)
