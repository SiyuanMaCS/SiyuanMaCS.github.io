+"""Generate the reduced-friction tennis-ball counterfactual.
+
+The camera and court are static.  The ball is in free flight until it first
+settles onto the near court at source frame 29 (zero-based frame 28), so that
+history is copied exactly.  At contact, friction both supplies a tangential
+impulse and subsequently decelerates the ball.  Both contributions are scaled
+to 0.2 of their measured source values.
+"""
+
+from pathlib import Path
+
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+CONTACT_FRAME = 28  # zero based; source frame 29
+
+
+def chroma(frame):
+    """Tennis-ball green excess, separated from the red court."""
+    f = frame.astype(np.float32)
+    return f[..., 1] - 0.5 * (f[..., 0] + f[..., 2])
+
+
+def rolling_centers(frames):
+    """Subpixel centers of the source ball after ground contact."""
+    centers = []
+    previous_x = 153.0
+    for index in range(CONTACT_FRAME, len(frames)):
+        q = chroma(frames[index])
+        # Once grounded the ball stays on the horizontal line near y=94.
+        lo = max(0, int(previous_x) - 7)
+        hi = min(q.shape[1], int(previous_x) + 8)
+        local = q[89:99, lo:hi]
+        py, px = np.unravel_index(np.argmax(local), local.shape)
+        px += lo
+        py += 89
+
+        # Green-excess weighted centroid gives stable subpixel placement.
+        x0, x1 = max(0, px - 4), min(q.shape[1], px + 5)
+        y0, y1 = max(0, py - 3), min(q.shape[0], py + 4)
+        weights = np.maximum(q[y0:y1, x0:x1], 0.0)
+        yy, xx = np.indices(weights.shape, dtype=np.float32)
+        total = float(weights.sum())
+        if total:
+            cx = float((weights * (xx + x0)).sum() / total)
+            cy = float((weights * (yy + y0)).sum() / total)
+        else:
+            cx, cy = float(px), float(py)
+        centers.append((cx, cy))
+        previous_x = cx
+    return np.asarray(centers, dtype=np.float32)
+
+
+def flight_speed(frames):
+    """Measure horizontal velocity immediately before contact."""
+    xs = []
+    # In these frames the ball is isolated within a narrow predicted window.
+    for index in range(18, CONTACT_FRAME + 1):
+        q = chroma(frames[index])
+        expected_x = 110.0 + (index - 18) * 4.35
+        expected_y = 45.0 + 0.49 * (index - 18) ** 2
+        yy, xx = np.indices(q.shape)
+        window = ((xx - expected_x) ** 2 / 9.0**2 +
+                  (yy - expected_y) ** 2 / 13.0**2) < 1.0
+        masked = np.where(window, q, -1.0e6)
+        py, px = np.unravel_index(np.argmax(masked), masked.shape)
+        x0, x1 = max(0, px - 4), min(q.shape[1], px + 5)
+        y0, y1 = max(0, py - 4), min(q.shape[0], py + 5)
+        weights = np.maximum(q[y0:y1, x0:x1], 0.0)
+        gy, gx = np.indices(weights.shape, dtype=np.float32)
+        xs.append(float((weights * (gx + x0)).sum() / weights.sum()))
+    t = np.arange(len(xs), dtype=np.float64)
+    return float(np.polyfit(t, np.asarray(xs), 1)[0])
+
+
+def source_ground_motion(centers):
+    """Fit initial velocity and Coulomb deceleration on the source court."""
+    # Fit before the final low-speed quantization.  Position at contact is
+    # constrained exactly, leaving velocity and the t^2 coefficient unknown.
+    count = 41
+    t = np.arange(1, count, dtype=np.float64)
+    displacement = centers[1:count, 0].astype(np.float64) - centers[0, 0]
+    velocity, half_accel = np.linalg.lstsq(
+        np.column_stack((t, t * t)), displacement, rcond=None
+    )[0]
+    return float(velocity), float(-2.0 * half_accel)
+
+
+def ellipse_alpha(height, width, cx, cy, rx=3.65, ry=3.05):
+    """Feathered support around the tiny antialiased ball sprite."""
+    yy, xx = np.indices((height, width), dtype=np.float32)
+    distance = np.sqrt(((xx - cx) / rx) ** 2 + ((yy - cy) / ry) ** 2)
+    return np.clip((1.19 - distance) / 0.19, 0.0, 1.0)
+
+
+def erase_ball(frame, cx, cy):
+    """Remove the source ball using the court's horizontal continuation."""
+    h, w = frame.shape[:2]
+    result = frame.astype(np.float32).copy()
+    radius = 6
+    xl = max(0, int(np.floor(cx)) - radius)
+    xr = min(w - 1, int(np.ceil(cx)) + radius)
+
+    # Every affected row is reconstructed between clean samples immediately
+    # to either side.  This retains the horizontal painted line beneath it.
+    reconstructed = result.copy()
+    for y in range(max(0, int(cy) - 5), min(h, int(cy) + 6)):
+        left = result[y, xl]
+        right = result[y, xr]
+        span = max(1, xr - xl)
+        for x in range(xl, xr + 1):
+            a = (x - xl) / span
+            reconstructed[y, x] = (1.0 - a) * left + a * right
+
+    alpha = ellipse_alpha(h, w, cx, cy)[..., None]
+    result = result * (1.0 - alpha) + reconstructed * alpha
+    return result
+
+
+def bilinear_sample(image, sx, sy):
+    """Sample an RGB image at floating-point coordinates."""
+    h, w = image.shape[:2]
+    x0 = np.floor(sx).astype(np.int32)
+    y0 = np.floor(sy).astype(np.int32)
+    x1 = np.clip(x0 + 1, 0, w - 1)
+    y1 = np.clip(y0 + 1, 0, h - 1)
+    x0 = np.clip(x0, 0, w - 1)
+    y0 = np.clip(y0, 0, h - 1)
+    wx = (sx - np.floor(sx))[..., None]
+    wy = (sy - np.floor(sy))[..., None]
+    top = image[y0, x0] * (1.0 - wx) + image[y0, x1] * wx
+    bottom = image[y1, x0] * (1.0 - wx) + image[y1, x1] * wx
+    return top * (1.0 - wy) + bottom * wy
+
+
+def place_translated_ball(clean, source, source_x, source_y, target_x):
+    """Move the original antialiased sprite along its ground line."""
+    h, w = clean.shape[:2]
+    # An entirely off-screen ball needs no sampling.
+    if target_x < -5.0 or target_x > w + 5.0:
+        return clean
+    yy, xx = np.indices((h, w), dtype=np.float32)
+    sx = xx - target_x + source_x
+    sy = yy
+    sprite = bilinear_sample(source.astype(np.float32), sx, sy)
+    alpha = ellipse_alpha(h, w, target_x, source_y)[..., None]
+    return clean * (1.0 - alpha) + sprite * alpha
+
+
+def generate():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3)
+    centers = rolling_centers(source)
+
+    pre_contact_v = flight_speed(source)
+    original_v, original_deceleration = source_ground_motion(centers)
+
+    # A friction impulse caused the source's sudden horizontal speed loss at
+    # impact.  With mu'=0.2mu only 20% of that loss remains.  Ground drag is
+    # likewise one fifth as large.
+    counterfactual_v = pre_contact_v - 0.2 * (pre_contact_v - original_v)
+    counterfactual_deceleration = 0.2 * original_deceleration
+
+    output = source.copy().astype(np.float32)
+    contact_x = float(centers[0, 0])
+    for frame_index in range(CONTACT_FRAME + 1, len(source)):
+        elapsed = float(frame_index - CONTACT_FRAME)
+        target_x = (contact_x + counterfactual_v * elapsed -
+                    0.5 * counterfactual_deceleration * elapsed * elapsed)
+        source_x, source_y = map(float, centers[frame_index - CONTACT_FRAME])
+        clean = erase_ball(source[frame_index], source_x, source_y)
+        output[frame_index] = place_translated_ball(
+            clean, source[frame_index], source_x, source_y, target_x
+        )
+
+    output = np.clip(np.rint(output), 0, 255).astype(np.uint8)
+    print(
+        f"pre-impact vx={pre_contact_v:.3f}, source ground vx={original_v:.3f}, "
+        f"source decel={original_deceleration:.4f}; counterfactual "
+        f"vx={counterfactual_v:.3f}, decel={counterfactual_deceleration:.4f}"
+    )
+    artifact_io.save(output, CASE_DIR)
+
+
+if __name__ == "__main__":
+    generate()
