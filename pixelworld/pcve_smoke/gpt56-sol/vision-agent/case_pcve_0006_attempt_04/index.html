+"""Generate the requested counterfactual air-hockey video.
+
+Frames 1--21 are factual.  At frame 22 the red mallet is removed.  In the
+source it subsequently hits the white puck, so the counterfactual also keeps
+that puck at its pre-contact resting pose.  A clean plate is assembled from
+two source instants: frame 1 supplies the resting puck and frame 40 supplies
+the small area hidden by the red mallet in frame 1.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def soft_polygon(shape, points, feather=4.0):
+    """Return an antialiased/feathered float mask for a polygon."""
+    h, w = shape
+    hard = np.zeros((h, w), np.uint8)
+    cv2.fillPoly(hard, [np.asarray(points, np.int32)], 255)
+    # A distance-field feather stays fully opaque well inside the plate while
+    # avoiding a visible pasted boundary on the smooth purple tabletop.
+    inside = cv2.distanceTransform(hard, cv2.DIST_L2, 5)
+    outside = cv2.distanceTransform(255 - hard, cv2.DIST_L2, 5)
+    signed = inside - outside
+    return np.clip(0.5 + signed / (2.0 * feather), 0.0, 1.0).astype(np.float32)
+
+
+def blend(dst, src, alpha):
+    a = alpha[..., None]
+    return np.clip(dst.astype(np.float32) * (1.0 - a) +
+                   src.astype(np.float32) * a + 0.5, 0, 255).astype(np.uint8)
+
+
+def make_clean_plate(source):
+    """Build tabletop plate with the white puck resting and no red mallet."""
+    plate = source[0].copy()
+
+    # At frame 1 the red mallet hides part of the center-court line.  By
+    # source frame 40 that exact patch is exposed, while both the moving puck
+    # and red mallet are well to its lower-left.  The polygon includes the
+    # mallet body, soft contact shadow, and red color spill, but excludes the
+    # blue mallet.
+    old_red = soft_polygon(
+        plate.shape[:2],
+        [(125, 62), (135, 55), (149, 58), (155, 69),
+         (151, 81), (137, 84), (125, 76)],
+        feather=2.5,
+    )
+    plate = blend(plate, source[39], old_red)
+    return plate
+
+
+def generate():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3) and source.dtype == np.uint8
+
+    result = source.copy()
+    plate = make_clean_plate(source)
+
+    # This is the union of the red mallet trajectory and the factual puck
+    # trajectory.  Its boundary lies in featureless tabletop and deliberately
+    # avoids the yellow rails and unaffected blue mallet.  The plate contains
+    # the stationary counterfactual puck at (about 104, 84).
+    affected = soft_polygon(
+        source.shape[1:3],
+        [(43, 68), (91, 56), (126, 51), (151, 56),
+         (158, 73), (151, 91), (126, 101), (108, 116),
+         (65, 116), (43, 105)],
+        feather=4.0,
+    )
+
+    # The prompt's numbering is one-based: frame 22 is array index 21.
+    for index in range(21, 96):
+        result[index] = blend(result[index], plate, affected)
+
+    return result
+
+
+if __name__ == "__main__":
+    frames = generate()
+    artifact_io.save(frames, CASE_DIR)
+    print("saved", frames.shape, frames.dtype, "to", CASE_DIR)
