+def bilinear_sample(image, sx, sy):
+    """Sample RGB image at float coordinate grids, edge-clamped."""
+    x0=np.floor(sx).astype(np.int32); y0=np.floor(sy).astype(np.int32)
+    fx=(sx-x0)[...,None]; fy=(sy-y0)[...,None]
+    x0=np.clip(x0,0,W-1); y0=np.clip(y0,0,H-1)
+    x1=np.clip(x0+1,0,W-1); y1=np.clip(y0+1,0,H-1)

+    c=image[y1,x0].astype(np.float32); d=image[y1,x1].astype(np.float32)

+def make_clean_plate(frame, cx):
+    """Erase the factual ball/shadow by sampling matching nearby grass.

+    A strip 35 px below the object has the same perspective-scale grass texture.
+    Tiny per-frame color changes are therefore preserved, and feathering makes the
