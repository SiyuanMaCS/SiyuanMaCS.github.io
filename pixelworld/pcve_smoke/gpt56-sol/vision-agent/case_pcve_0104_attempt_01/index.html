+"""Remove the old floor ball and render the resulting no-collision timeline.
+
+The source is a locked-off render.  A clean floor plate is assembled from
+source frames in which each patch is unoccluded.  Until impact, only the old
+green ball is removed.  From impact onward both factual balls are cleared and
+the yellow ball is re-composited on its physically continued leftward path.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def feathered_box(shape, box, feather=3.0):
+    """Return a soft rectangular alpha mask."""
+    h, w = shape
+    x0, y0, x1, y1 = box
+    yy, xx = np.mgrid[:h, :w]
+    inside = np.minimum.reduce((xx - x0, x1 - 1 - xx, yy - y0, y1 - 1 - yy))
+    return np.clip((inside + 1.0) / feather, 0.0, 1.0).astype(np.float32)
+
+
+def composite(base, layer, alpha):
+    a = alpha[..., None].astype(np.float32)
+    return np.clip(base.astype(np.float32) * (1.0 - a) + layer.astype(np.float32) * a,
+                   0, 255).astype(np.uint8)
+
+
+def make_clean_plate(frames):
+    """Build a source-only clean plate, avoiding generated/inpainted texture."""
+    # Frame 40 has no ball around the old ball's starting point.  Frame 0 has
+    # no balls around the two eventual resting points, so those patches clean
+    # frame 40.  Match the tiny temporal RGB drift before inserting them.
+    plate = frames[40].copy()
+    early = frames[0].astype(np.int16)
+    late = frames[40].astype(np.int16)
+    drift = np.median(late[112:136, 8:150] - early[112:136, 8:150], axis=(0, 1))
+    early_matched = np.clip(early + drift[None, None, :], 0, 255).astype(np.uint8)
+
+    # Final green ball and shadow; final yellow ball and shadow.
+    for box in ((7, 82, 53, 109), (95, 80, 141, 109)):
+        a = feathered_box(plate.shape[:2], box, feather=3.0)
+        plate = composite(plate, early_matched, a)
+    return plate
+
+
+def plate_for_frame(plate, frame):
+    """Match subtle exposure drift using an always-clear strip of floor."""
+    roi = (slice(114, 137), slice(145, 235))
+    delta = np.median(frame[roi].astype(np.int16) - plate[roi].astype(np.int16),
+                      axis=(0, 1))
+    return np.clip(plate.astype(np.int16) + delta[None, None, :], 0, 255).astype(np.uint8)
+
+
+def yellow_center(frame, min_x=70):
+    """Locate the high-saturation yellow ball in a post-collision frame."""
+    hsv = cv2.cvtColor(frame, cv2.COLOR_RGB2HSV)
+    mask = ((hsv[..., 0] >= 20) & (hsv[..., 0] <= 45) &
+            (hsv[..., 1] >= 75) & (hsv[..., 2] >= 80)).astype(np.uint8)
+    count, labels, stats, centers = cv2.connectedComponentsWithStats(mask)
+    candidates = []
+    for i in range(1, count):
+        x, y, w, h, area = stats[i]
+        cx, cy = centers[i]
+        if area >= 18 and cx >= min_x and y >= 82:
+            candidates.append((area, float(cx), float(cy), x, y, w, h))
+    if not candidates:
+        raise RuntimeError("Could not locate yellow ball")
+    _, cx, cy, x, y, w, h = max(candidates)
+    # Component pixels omit the shaded rim. Bounding-box center is a more
+    # stable geometric center for compositing the complete seven-pixel ball.
+    return x + (w - 1) / 2.0, y + (h - 1) / 2.0
+
+
+def move_object(source, source_plate, base, src_center, dst_center):
+    """Move the rendered yellow ball and its floor shadow as soft layers."""
+    h, w = source.shape[:2]
+    sx, sy = src_center
+    dx, dy = dst_center[0] - sx, dst_center[1] - sy
+    yy, xx = np.mgrid[:h, :w]
+
+    # Complete ball, including its antialiased shaded rim.
+    r = np.sqrt((xx - sx) ** 2 + (yy - sy) ** 2)
+    ball_alpha = np.clip((8.3 - r) / 1.5, 0.0, 1.0).astype(np.float32)
+
+    # Recover only the genuinely darkened portion of the soft cast shadow.
+    # Its shape is taken from the source frame, not synthetically painted.
+    ex = (xx - (sx + 8.0)) / 17.0
+    ey = (yy - (sy + 5.2)) / 5.0
+    ellipse = np.clip((1.0 - (ex * ex + ey * ey)) * 2.0, 0.0, 1.0)
+    darkness = (source_plate.astype(np.float32).mean(axis=2) -
+                source.astype(np.float32).mean(axis=2))
+    shadow_alpha = ellipse * np.clip((darkness - 0.8) / 2.4, 0.0, 1.0)
+    shadow_alpha *= (1.0 - ball_alpha)
+    shadow_alpha = cv2.GaussianBlur(shadow_alpha.astype(np.float32), (3, 3), 0.55)
+
+    transform = np.array([[1.0, 0.0, dx], [0.0, 1.0, dy]], dtype=np.float32)
+    moved_source = cv2.warpAffine(source, transform, (w, h),
+                                  flags=cv2.INTER_LINEAR,
+                                  borderMode=cv2.BORDER_REPLICATE)
+    moved_shadow = cv2.warpAffine(shadow_alpha, transform, (w, h),
+                                  flags=cv2.INTER_LINEAR,
+                                  borderMode=cv2.BORDER_CONSTANT)
+    moved_ball = cv2.warpAffine(ball_alpha, transform, (w, h),
+                                flags=cv2.INTER_LINEAR,
+                                borderMode=cv2.BORDER_CONSTANT)
+    out = composite(base, moved_source, moved_shadow)
+    out = composite(out, moved_source, moved_ball)
+    return out
+
+
+def render():
+    frames = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert frames.shape == (96, 144, 256, 3) and frames.dtype == np.uint8
+    clean = make_clean_plate(frames)
+    output = frames.copy()
+
+    # Before the factual collision, erase only the stationary old ball and
+    # its right-cast shadow. The incoming yellow ball remains source-exact.
+    old_ball_box = (56, 85, 87, 108)
+    old_alpha = feathered_box(clean.shape[:2], old_ball_box, feather=2.5)
+    for t in range(21):
+        clean_t = plate_for_frame(clean, frames[t])
+        output[t] = composite(frames[t], clean_t, old_alpha)
+
+    # At t=21 the source collision begins. Clear the entire affected floor
+    # corridor, then add back only the yellow ball on a no-collision path.
+    clear_main = feathered_box(clean.shape[:2], (7, 82, 126, 110), feather=2.5)
+    clear_shadow_tail = feathered_box(clean.shape[:2], (122, 89, 141, 110), feather=2.5)
+    clear_alpha = np.maximum(clear_main, clear_shadow_tail)
+
+    x20 = 78.53       # final uncontaminated airborne center (source frame 20)
+    initial_vx = 6.30 # measured incoming leftward speed, pixels/frame
+    retention = 0.875 # floor friction; asymptotic rest center is x=28.1
+
+    for t in range(21, 96):
+        clean_t = plate_for_frame(clean, frames[t])
+        base = composite(frames[t], clean_t, clear_alpha)
+
+        # Use the next source instant at the onset to recover a complete ball
+        # unobscured by the removed green ball. Thereafter use each frame's
+        # own rolling/bouncing appearance and shadow.
+        src_t = min(95, t + 1) if t == 21 else t
+        source = frames[src_t]
+        source_plate = plate_for_frame(clean, source)
+        sx, sy = yellow_center(source)
+
+        n = t - 20
+        desired_x = x20 - initial_vx * (1.0 - retention ** n) / (1.0 - retention)
+        # Preserve the source's subtle floor bounce; at impact place the
+        # complete borrowed ball at the same ground-contact height.
+        desired_y = sy if t > 21 else 91.0
+        output[t] = move_object(source, source_plate, base,
+                                (sx, sy), (desired_x, desired_y))
+
+    return output
+
+
+if __name__ == "__main__":
+    result = render()
+    artifact_io.save(result, CASE_DIR)
+    print("saved", result.shape, result.dtype)
