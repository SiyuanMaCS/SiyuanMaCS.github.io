+def feather_patch(im, patch, mask, x, y):
+    """Alpha composite a small RGBA-like cutout at integer upper-left."""
+    h,w = patch.shape[:2]
+    x0=max(0,x); y0=max(0,y); x1=min(W,x+w); y1=min(H,y+h)
+    if x1<=x0 or y1<=y0: return

+    a=mask[py0:py1,px0:px1,None].astype(np.float32)/255.0
+    roi=im[y0:y1,x0:x1].astype(np.float32)
+    im[y0:y1,x0:x1]=np.clip(patch[py0:py1,px0:px1]*a+roi*(1-a),0,255).astype(np.uint8)
+

+# The ball occludes shelf's left tip around x127..130; frame 41 reveals it.
+base[25:43,123:135] = src[40,25:43,123:135]
+
+# Extract high-quality cutouts. Soft alpha derives from difference to a local
+# clean plate and includes antialiased edge pixels plus contact shading.
+def make_cutout(frame, box, clean_under, threshold=3, dilate=1, blur=1.05):
+    x0,y0,x1,y1=box
+    p=frame[y0:y1,x0:x1].copy()
+    c=clean_under[y0:y1,x0:x1]
+    d=np.max(np.abs(p.astype(np.int16)-c.astype(np.int16)),axis=2).astype(np.uint8)
+    # Smooth confidence ramp, preserving subtle edge blur.
+    a=np.clip((d.astype(np.float32)-threshold)*18.0,0,255).astype(np.uint8)
+    if dilate:
+        core=(d>8).astype(np.uint8)*255
+        core=cv2.dilate(core,np.ones((3,3),np.uint8),iterations=dilate)
+        a=np.maximum(a,(core*0.72).astype(np.uint8))
+    a=cv2.GaussianBlur(a,(0,0),blur)
+    # Suppress patch borders.
+    a[:2]*=0; a[-2:]*=0; a[:,:2]*=0; a[:,-2:]*=0
+    return p,a

+                bvx=min(0.0,bvx+roll_friction)
