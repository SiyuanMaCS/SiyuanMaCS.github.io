+#!/usr/bin/env python3
+"""Generate the restitution counterfactual for this case.
+
+The camera is static, so a clean plate can be recovered from source pixels that
+are not covered by either rigid body.  The bodies retain their source texture;
+only their post-impact transforms are changed.  Velocities around impact imply
+an original relative restitution of about 0.84.  The requested 1.8 multiplier
+therefore gives e ~= 1.51.  For the approximately equal-mass bodies this makes
+the ball reverse at 1.77 px/frame and launches the block at 8.77 px/frame.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+N, H, W = 96, 144, 256
+
+
+def ball_measurements(frames):
+    """Locate the orange ball in every source frame."""
+    centers = []
+    for frame in frames:
+        r, g, b = frame.transpose(2, 0, 1).astype(np.int16)
+        mask = ((r > 125) & (r - g > 48) & (g - b > 10) & (g < 125)).astype(np.uint8)
+        count, labels, stats, centroids = cv2.connectedComponentsWithStats(mask)
+        candidates = [j for j in range(1, count) if stats[j, cv2.CC_STAT_AREA] > 180]
+        j = max(candidates, key=lambda k: stats[k, cv2.CC_STAT_AREA])
+        x, y, w, h, _ = stats[j]
+        # The color mask misses a little of the dark seam and shaded rim.  The
+        # bounding-box center is consequently more stable than the color mean.
+        centers.append((x + (w - 1) / 2.0, y + (h - 1) / 2.0))
+    centers = np.asarray(centers, np.float32)
+    # During the one occluded contact frame the visible orange component is
+    # clipped by the block.  Continue the measured incoming trajectory there.
+    centers[12] = centers[11] + np.median(np.diff(centers[5:12], axis=0), axis=0)
+    return centers
+
+
+def source_block_pose(i):
+    """Approximate the source block's translation from measured optical flow."""
+    t = max(0.0, i - 11.0)
+    dx = max(0.0, 6.80 * t - 0.138 * t * t)
+    dx = min(83.15, dx)
+    return dx, 0.152 * dx
+
+
+def block_polygon(dx=0.0, dy=0.0, scale=1.0):
+    # Silhouette in frame 0: top-left, rear top, right-top, right-bottom,
+    # front-bottom, left-bottom.  Scale about its initial floor contact point.
+    p = np.array([[109, 63], [128, 57], [164, 63],
+                  [164, 94], [137, 103], [109, 96]], np.float32)
+    pivot = np.array([137.0, 103.0], np.float32)
+    return (p - pivot) * scale + pivot + np.array([dx, dy], np.float32)
+
+
+def clean_plate(frames, ball_centers):
+    """Take a per-pixel median only over frames where the pixel is unobscured."""
+    usable = np.ones((N, H, W), dtype=bool)
+    yy, xx = np.ogrid[:H, :W]
+    for i, (cx, cy) in enumerate(ball_centers):
+        # Ball plus its soft floor shadow.
+        usable[i] &= (((xx - cx) / 24.5) ** 2 + ((yy - cy) / 25.5) ** 2 > 1.0)
+        usable[i] &= (((xx - (cx + 4)) / 29.0) ** 2 + ((yy - (cy + 23)) / 10.0) ** 2 > 1.0)
+
+        dx, dy = source_block_pose(i)
+        poly = np.round(block_polygon(dx, dy, 1.0 + 0.0008 * dx)).astype(np.int32)
+        m = np.zeros((H, W), np.uint8)
+        cv2.fillConvexPoly(m, poly, 255)
+        # Include antialiasing and the block's contact/cast shadow.
+        cv2.fillConvexPoly(m, poly + np.array([7, 5]), 255)
+        m = cv2.dilate(m, np.ones((7, 7), np.uint8))
+        usable[i] &= m == 0
+
+    stack = frames.astype(np.float32)
+    stack[~usable] = np.nan
+    plate = np.nanmedian(stack, axis=0)
+    # All scene pixels have many clean observations, but retain a deterministic
+    # fallback should a very conservative mask ever cover one throughout.
+    missing = np.isnan(plate[..., 0])
+    if missing.any():
+        fallback = np.median(frames, axis=0)
+        plate[missing] = fallback[missing]
+    return np.clip(np.rint(plate), 0, 255).astype(np.uint8)
+
+
+def darken_with_shadow(image, center, axes, opacity):
+    """Apply a soft multiplicative contact shadow in-place."""
+    layer = np.zeros((H, W), np.float32)
+    cv2.ellipse(layer, (int(round(center[0])), int(round(center[1]))),
+                (max(1, int(round(axes[0]))), max(1, int(round(axes[1])))),
+                0, 0, 360, 1.0, -1, cv2.LINE_AA)
+    layer = cv2.GaussianBlur(layer, (0, 0), 2.0)
+    factor = 1.0 - opacity * layer[..., None]
+    image[:] = np.clip(image.astype(np.float32) * factor, 0, 255).astype(np.uint8)
+
+
+def normalized_ball(frames, centers, source_time, size=64):
+    """Source-textured ball normalized to a square RGBA sprite.
+
+    Adjacent source observations are aligned before interpolation, avoiding
+    held textures when the rebound is slower than the incoming motion.
+    """
+    source_time = float(np.clip(source_time, 0.0, 11.0))
+    lo = int(np.floor(source_time))
+    hi = min(11, lo + 1)
+    frac = source_time - lo
+    patches = []
+    for k in (lo, hi):
+        # Apparent radius grows gently as the incoming ball approaches camera.
+        radius = 15.7 + 0.155 * k
+        scale = (size * 0.78) / (2.0 * radius)
+        matrix = np.array([[scale, 0, size / 2.0 - scale * centers[k, 0]],
+                           [0, scale, size / 2.0 - scale * centers[k, 1]]], np.float32)
+        patches.append(cv2.warpAffine(frames[k], matrix, (size, size),
+                                      flags=cv2.INTER_LINEAR,
+                                      borderMode=cv2.BORDER_REFLECT_101).astype(np.float32))
+    rgb = patches[0] * (1.0 - frac) + patches[1] * frac
+    y, x = np.mgrid[:size, :size].astype(np.float32)
+    d = np.sqrt((x - size / 2.0) ** 2 + (y - size / 2.0) ** 2)
+    edge = size * 0.39
+    alpha = np.clip(edge + 0.75 - d, 0.0, 1.0)
+    return rgb, alpha
+
+
+def composite_ball(image, frames, centers, source_time, center):
+    sprite, alpha = normalized_ball(frames, centers, source_time)
+    # Radius and therefore image scale are already encoded by the selected
+    # source observation.  Put its normalized 64px canvas at center.
+    x0 = int(round(center[0] - 32))
+    y0 = int(round(center[1] - 32))
+    x1, y1 = x0 + 64, y0 + 64
+    ox0, oy0 = max(0, x0), max(0, y0)
+    ox1, oy1 = min(W, x1), min(H, y1)
+    if ox0 >= ox1 or oy0 >= oy1:
+        return
+    sx0, sy0 = ox0 - x0, oy0 - y0
+    sx1, sy1 = sx0 + ox1 - ox0, sy0 + oy1 - oy0
+    a = alpha[sy0:sy1, sx0:sx1, None]
+    dst = image[oy0:oy1, ox0:ox1].astype(np.float32)
+    src = sprite[sy0:sy1, sx0:sx1]
+    image[oy0:oy1, ox0:ox1] = np.clip(src * a + dst * (1.0 - a), 0, 255).astype(np.uint8)
+
+
+def composite_block(image, source_frame, dx, dy, scale):
+    """Warp the original textured wooden block, with a feathered silhouette."""
+    src_poly = block_polygon()
+    src_mask = np.zeros((H, W), np.uint8)
+    cv2.fillConvexPoly(src_mask, np.round(src_poly).astype(np.int32), 255, cv2.LINE_AA)
+    src_alpha = cv2.GaussianBlur(src_mask.astype(np.float32) / 255.0, (0, 0), 0.42)
+
+    pivot = np.array([137.0, 103.0], np.float32)
+    tx, ty = pivot + np.array([dx, dy], np.float32) - scale * pivot
+    matrix = np.array([[scale, 0, tx], [0, scale, ty]], np.float32)
+    warped = cv2.warpAffine(source_frame, matrix, (W, H), flags=cv2.INTER_LINEAR,
+                            borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+    alpha = cv2.warpAffine(src_alpha, matrix, (W, H), flags=cv2.INTER_LINEAR,
+                           borderMode=cv2.BORDER_CONSTANT, borderValue=0)[..., None]
+    dst = image.astype(np.float32)
+    image[:] = np.clip(warped.astype(np.float32) * alpha + dst * (1.0 - alpha),
+                       0, 255).astype(np.uint8)
+
+
+def generate():
+    frames = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert frames.shape == (N, H, W, 3) and frames.dtype == np.uint8
+    centers = ball_measurements(frames)
+    plate = clean_plate(frames, centers)
+
+    out = np.empty_like(frames)
+    # Nothing can reveal restitution until contact, so preserve the complete
+    # factual history through the impact pose (source frames 1--13).
+    out[:13] = frames[:13]
+
+    impact_center = centers[11] + np.median(np.diff(centers[5:12], axis=0), axis=0)
+    e_original = 0.836
+    e_new = 1.8 * e_original
+    incoming_vx = 7.0
+    ball_vx = 0.5 * (1.0 - e_new) * incoming_vx       # -1.7668 px/frame
+    block_v0 = 0.5 * (1.0 + e_new) * incoming_vx      #  8.7668 px/frame
+    friction = 0.138                                    # source-measured px/frame^2
+    incoming_slope = 0.205
+
+    for i in range(13, N):
+        t = float(i - 12)
+        image = plate.copy()
+
+        # Superelastic rebound of the ball.  Reusing the normalized incoming
+        # texture in reverse naturally supplies perspective shrink and reverse
+        # spin while its center follows the equal-mass impulse result.
+        bx = impact_center[0] + ball_vx * t
+        by = impact_center[1] + ball_vx * incoming_slope * t
+        source_time = 11.0 - (-ball_vx / incoming_vx) * (t - 1.0)
+        source_time = float(np.clip(source_time, 0.0, 11.0))
+        ball_radius = 15.7 + 0.155 * source_time
+        darken_with_shadow(image, (bx + 4.0, by + ball_radius + 2.2),
+                           (ball_radius * 1.15, 3.3), 0.28)
+        composite_ball(image, frames, centers, source_time, (bx, by))
+
+        # The same floor friction acts after the stronger launch.  Its stopping
+        # distance is much larger than in the factual video, carrying the block
+        # through and out of the right side of the fixed camera view.
+        block_t = t
+        dx = max(0.0, block_v0 * block_t - 0.5 * friction * block_t * block_t)
+        dy = 0.152 * dx
+        scale = 1.0 + 0.0008 * dx
+        poly = block_polygon(dx, dy, scale)
+        if poly[:, 0].min() < W and poly[:, 1].min() < H:
+            floor_y = float(poly[:, 1].max() + 2.0)
+            floor_x = float(np.mean(poly[3:6, 0]) + 5.0)
+            darken_with_shadow(image, (floor_x, floor_y),
+                               (30.0 * scale, 4.2 * scale), 0.25)
+            composite_block(image, frames[0], dx, dy, scale)
+
+        out[i] = image
+
+    return out
+
+
+if __name__ == "__main__":
+    result = generate()
+    artifact_io.save(result, CASE_DIR)
+    print(f"saved {result.shape[0]} frames at {result.shape[2]}x{result.shape[1]}")
