+"""Generate the counterfactual: 8x car friction beginning at source frame 8.
+
+The scene camera and ramp are static.  We recover a clean plate from the temporal
+median of the first 20 frames (the moving car visits every pixel briefly), remove
+the factual car from frame 8 onward, and composite the frame-7 car at a physically
+shortened uphill trajectory.  From the measured source trajectory, the car's
+speed just before intervention is about 9 px/frame and its original deceleration
+is about 1 px/frame^2.  Eight times that deceleration gives roughly 5.5 px of
+remaining travel, so the car stops on the ramp during the next two frames.
+"""
+from pathlib import Path
+import cv2
+import numpy as np
+import artifact_io
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def foreground_alpha(frame, clean, box, threshold=4, dilate=1):
+    """Soft matte for changed pixels in box, retaining coherent small details."""
+    h, w = frame.shape[:2]
+    x0, y0, x1, y1 = box
+    x0, y0, x1, y1 = max(0,x0), max(0,y0), min(w,x1), min(h,y1)
+    delta = np.max(np.abs(frame.astype(np.int16)-clean.astype(np.int16)), axis=2)
+    raw = np.zeros((h,w), np.uint8)
+    raw[y0:y1, x0:x1] = (delta[y0:y1, x0:x1] > threshold).astype(np.uint8)
+
+    # Suppress isolated render-grain changes while retaining body, wheels, and shadow.
+    count, labels, stats, _ = cv2.connectedComponentsWithStats(raw, 8)
+    mask = np.zeros_like(raw)
+    for k in range(1, count):
+        if stats[k, cv2.CC_STAT_AREA] >= 3:
+            mask[labels == k] = 255
+    mask = cv2.morphologyEx(mask, cv2.MORPH_CLOSE, np.ones((3,3),np.uint8))
+    if dilate:
+        mask = cv2.dilate(mask, np.ones((3,3),np.uint8), iterations=dilate)
+    return cv2.GaussianBlur(mask.astype(np.float32)/255.0, (3,3), 0.55)
+
+
+def composite(base, rgb, alpha):
+    a = alpha[...,None]
+    return np.clip(base.astype(np.float32)*(1-a) + rgb.astype(np.float32)*a,
+                   0, 255).astype(np.uint8)
+
+
+def shift_layer(rgb, alpha, dx, dy):
+    h,w = alpha.shape
+    matrix = np.float32([[1,0,dx],[0,1,dy]])
+    moved_rgb = cv2.warpAffine(rgb, matrix, (w,h), flags=cv2.INTER_LINEAR,
+                               borderMode=cv2.BORDER_CONSTANT)
+    moved_a = cv2.warpAffine(alpha, matrix, (w,h), flags=cv2.INTER_LINEAR,
+                             borderMode=cv2.BORDER_CONSTANT)
+    return moved_rgb, moved_a
+
+
+def factual_car_box(i):
+    """Conservative factual-car regions, zero-based frame index."""
+    # The car is airborne / ramp-borne through frame 18.
+    if i <= 17:
+        # Track bounds measured from the source; broad margin absorbs wheel shadow.
+        xs = [72,66,57,50,44,39,34,30,26,23,20]  # indices 7..17
+        ys = [72,69,67,64,63,63,64,66,70,76,84]
+        j = i-7
+        return (xs[j]-5, ys[j]-5, xs[j]+47, ys[j]+42)
+    # It lands at lower left and remains there for the rest of the shot.
+    return (0, 88, 59, 139)
+
+
+def generate():
+    source = np.load(CASE_DIR/'source_pixels.npz')['rgb']
+    assert source.shape == (96,144,256,3) and source.dtype == np.uint8
+
+    # Clean static scene. No location is covered by the fast-moving car for half
+    # of these samples, so the median removes it while preserving ramp boundaries.
+    clean = np.median(source[:20].astype(np.float32), axis=0).astype(np.uint8)
+
+    # Use frame 7 (last factual frame) as the sprite: this guarantees visual and
+    # angular continuity at the intervention boundary.
+    sprite_frame = source[6]
+    sprite_alpha = foreground_alpha(sprite_frame, clean, (74,69,124,101), 4, 1)
+
+    out = source.copy()  # frames 1..7 remain bit-identical by construction
+    for i in range(7, 96):
+        # Erase only the factual car and its immediate contact/landing shadow.
+        erase = foreground_alpha(source[i], clean, factual_car_box(i), 4, 1)
+        base = composite(source[i], clean, erase)
+
+        # Eightfold friction: 5 px in the first affected interval, then only one
+        # further pixel before static friction holds the car on the incline.
+        if i == 7:
+            dx, dy = -5.0, -1.8
+        else:
+            dx, dy = -6.0, -2.2
+        car_rgb, car_a = shift_layer(sprite_frame, sprite_alpha, dx, dy)
+        out[i] = composite(base, car_rgb, car_a)
+
+    # Contractual output writer (pixels.npz, PNG frames, inspection, HTML player).
+    artifact_io.save(out, CASE_DIR)
+    return out
+
+
+if __name__ == '__main__':
+    generate()
