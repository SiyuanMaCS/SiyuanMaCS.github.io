+"""Generate the 10x-ball-mass counterfactual for this PixelWorld case.
+
+The intervention changes inertia, not appearance.  Frames through the first
+contact are consequently retained verbatim.  After contact we reuse source
+object pixels at matching floor positions and alter only the trajectories.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+N, H, W = 96, 144, 256

+def ball_measurements(frames):
+    """Measure the orange ball using its stable rendered material color."""
+    centers, radii = [], []
+    previous_x = 15.0
+    yy = np.indices((H, W))[0]
+    for frame in frames:
+        f = frame.astype(np.int16)
+        r, g, b = f[..., 0], f[..., 1], f[..., 2]
+        mask = ((r > 110) & (r - g > 60) & (b < 78) & (yy < 122)).astype(np.uint8)
+        n, labels, stats, centroids = cv2.connectedComponentsWithStats(mask)
+        choices = []
+        for j in range(1, n):
+            x, y, w, h, area = stats[j]
+            if area > 350 and 10 < h < 48 and x < 185:
+                # Continuity rejects similarly colored details in the floor/chest.
+                score = abs((x + w / 2) - previous_x)
+                choices.append((score, j))
+        if not choices:
+            raise RuntimeError("Could not locate ball")
+        j = min(choices)[1]
+        x, y, w, h, _ = stats[j]
+        # The first ball is clipped by the image edge.  Its visible right edge
+        # and height still define the true circle reliably.
+        radius = max(h / 2.0, w / 2.0)
+        if x == 0:
+            cx = x + w - radius
+        else:
+            cx = x + w / 2.0
+        cy = y + h / 2.0
+        centers.append((cx, cy))
+        radii.append(radius + 0.35)
+        previous_x = cx
+    return np.asarray(centers, np.float32), np.asarray(radii, np.float32)
+
+
+# Left edge of the chest in the factual source.  Values were measured from the
+# stable top/front silhouette.  The long plateau is the chest at the wall.
+BOX_LEFT_KEYS = np.asarray(
+    [109, 109, 109, 109, 109, 109, 109, 109, 109, 109, 109, 109,
+     109, 114, 119, 125, 129, 133, 137, 142, 147, 153, 157, 161,
+     164, 166, 168, 170, 172, 174, 175, 177, 178, 180, 181, 181],
+    dtype=np.float32,
+)
+
+
+def box_left(source_index):
+    i = int(np.clip(round(source_index), 0, 35))
+    return float(BOX_LEFT_KEYS[i])
+
+
+def box_polygon(left):
+    """Perspective-aware chest silhouette at a given horizontal position."""
+    p = np.clip((left - 109.0) / 72.0, 0.0, 1.0)
+    top = 55.0 + 10.0 * p
+    pts = np.asarray([
+        [left,             top + 6.0],
+        [left + 19 + 2*p,  top],
+        [left + 55 + 7*p,  top + 7.0],
+        [left + 55 + 7*p,  top + 39 + 4*p],
+        [left + 37 + 1*p,  top + 49 + 5*p],
+        [left,             top + 41 + 2*p],
+    ])
+    return np.round(pts).astype(np.int32)
+
+
+def reconstruct_clean_plate(frames, centers, radii):
+    """Temporal clean plate with moving-object samples explicitly excluded."""
+    valid = np.ones((N, H, W), np.uint8)
+    for i in range(N):
+        # Ball, contact shadow, and a small antialiasing margin.
+        cx, cy = centers[i]
+        rr = radii[i] + 4.0
+        cv2.ellipse(valid[i], (round(cx), round(cy + 2)),
+                    (round(rr), round(rr + 8)), 0, 0, 360, 0, -1)
+
+        left = box_left(i)
+        poly = box_polygon(left).astype(np.float32)
+        # Conservative expansion includes the chest's floor shadow.
+        x0 = max(0, int(poly[:, 0].min()) - 6)
+        x1 = min(W, int(poly[:, 0].max()) + 10)
+        y0 = max(0, int(poly[:, 1].min()) - 5)
+        y1 = min(H, int(poly[:, 1].max()) + 13)
+        valid[i, y0:y1, x0:x1] = 0
+
+    work = frames.astype(np.float32)
+    work[valid == 0] = np.nan
+    with np.errstate(all="ignore"):
+        plate = np.nanmedian(work, axis=0)
+    known = np.isfinite(plate[..., 0])
+    plate = np.nan_to_num(plate, nan=0.0).astype(np.uint8)
+
+    # The tiny portion swept by an object in every source frame is filled from
+    # its textured neighborhood.  TELEA is used only for this unavailable area.
+    missing = (~known).astype(np.uint8) * 255
+    if missing.any():
+        missing = cv2.dilate(missing, np.ones((3, 3), np.uint8), iterations=1)
+        plate = cv2.inpaint(plate, missing, 3, cv2.INPAINT_TELEA)
+    return plate
+
+
+def alpha_composite(dst, src, alpha):
+    a = np.clip(alpha[..., None], 0.0, 1.0).astype(np.float32)
+    return np.clip(dst.astype(np.float32) * (1-a) + src.astype(np.float32) * a,
+                   0, 255).astype(np.uint8)
+
+
+def add_soft_shadow(frame, cx, cy, radius, strength=0.16):
+    mask = np.zeros((H, W), np.float32)
+    center = (int(round(cx + 2)), int(round(cy + radius * 0.78)))
+    axes = (max(2, int(round(radius * 0.92))), max(2, int(round(radius * 0.23))))
+    cv2.ellipse(mask, center, axes, -5, 0, 360, 1.0, -1, cv2.LINE_AA)
+    mask = cv2.GaussianBlur(mask, (0, 0), 2.0)
+    out = frame.astype(np.float32) * (1.0 - strength * mask[..., None])
+    return np.clip(out, 0, 255).astype(np.uint8)
+
+
+def paste_ball(frame, source, source_center, source_radius, target_center):
+    """Translate a source ball with a subpixel, feathered circular matte."""
+    sx, sy = source_center
+    tx, ty = target_center
+    # Warp both pixels and matte; linear resampling preserves source antialiasing.
+    mat = np.asarray([[1.0, 0.0, tx - sx], [0.0, 1.0, ty - sy]], np.float32)
+    moved = cv2.warpAffine(source, mat, (W, H), flags=cv2.INTER_LINEAR,
+                           borderMode=cv2.BORDER_REFLECT_101)
+    yy, xx = np.mgrid[:H, :W].astype(np.float32)
+    dist = np.sqrt((xx - tx) ** 2 + (yy - ty) ** 2)
+    alpha = np.clip((source_radius + 0.65 - dist) * 1.7, 0.0, 1.0)
+    return alpha_composite(frame, moved, alpha)
+
+
+def paste_box(frame, source, left):
+    poly = box_polygon(left)
+    mask = np.zeros((H, W), np.uint8)
+    cv2.fillPoly(mask, [poly], 255, lineType=cv2.LINE_AA)
+    # A one-pixel feather keeps the original antialiased silhouette.
+    alpha = cv2.GaussianBlur(mask.astype(np.float32) / 255.0, (0, 0), 0.38)
+    return alpha_composite(frame, source, alpha)
+
+
+def target_ball_x(frame_index):
+    """Counterfactual ball center after the first (frame-12) contact."""
+    if frame_index <= 24:
+        n = frame_index - 12.0
+        return 97.5 + 5.60*n - 0.015*n*n
+    # Collision with the chest pinned at the wall: damped rebound to the left.
+    n = frame_index - 24.0
+    v0, decel = 2.30, 0.070
+    stop_t = v0 / decel
+    q = min(n, stop_t)
+    return 162.54 - v0*q + 0.5*decel*q*q
+
+
+def generate():
+    source_frames = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source_frames.shape == (N, H, W, 3) and source_frames.dtype == np.uint8
+    centers, radii = ball_measurements(source_frames)
+    plate = reconstruct_clean_plate(source_frames, centers, radii)
+
+    output = np.empty_like(source_frames)
+    # Mass has no visible effect before contact; retain the factual history.
+    output[:13] = source_frames[:13]
+
+    # Exclude the few source balls that overlap the chest when selecting a
+    # position-matched sprite.  This preserves seams, grain and rotation.
+    ball_library = np.asarray(list(range(0, 12)) + list(range(16, N)))
+
+    for t in range(13, N):
+        frame = plate.copy()
+
+        # A 10x ball transfers substantially more momentum.  Replaying the
+        # factual chest displacement at 2x speed is the calibrated consequence.
+        mapped_box_frame = min(35, int(round(12 + 2.0 * (t - 12))))
+        left = box_left(mapped_box_frame)
+        box_source = source_frames[mapped_box_frame]
+
+        # Floor contact shadows are synthesized separately so translated source
+        # backgrounds never leak into the clean plate.
+        poly = box_polygon(left)
+        shadow_cx = float((poly[:, 0].min() + poly[:, 0].max()) / 2 + 3)
+        shadow_cy = float(poly[:, 1].max() - 1)
+        shadow = np.zeros((H, W), np.float32)
+        cv2.ellipse(shadow, (round(shadow_cx), round(shadow_cy)),
+                    (round(31 + 3*(left-109)/72), 6), 0, 0, 360, 1.0, -1,
+                    cv2.LINE_AA)
+        shadow = cv2.GaussianBlur(shadow, (0, 0), 2.2)
+        frame = np.clip(frame.astype(np.float32) *
+                        (1.0 - 0.12 * shadow[..., None]), 0, 255).astype(np.uint8)
+        frame = paste_box(frame, box_source, left)
+
+        bx = target_ball_x(t)
+        # Source perspective gives y, radius, and seam rotation for this x.
+        nearest = ball_library[np.argmin(np.abs(centers[ball_library, 0] - bx))]
+        sx, sy = centers[nearest]
+        br = float(radii[nearest])
+        # The measured source trajectory is a robust perspective map x -> y.
+        by = float(np.interp(bx, centers[:, 0], centers[:, 1]))
+        frame = add_soft_shadow(frame, bx, by, br)
+        frame = paste_ball(frame, source_frames[nearest], (sx, sy), br, (bx, by))
+        output[t] = frame
+
+    return output
+
+
+if __name__ == "__main__":
+    frames = generate()
+    # Required benchmark serializer: writes pixels.npz, PNG sequence and player.
+    artifact_io.save(frames, CASE_DIR)
