+def interpolated_corners(frame_index: int) -> np.ndarray:
+    indices = sorted(KEYS)
+    if frame_index <= indices[0]:
+        return KEYS[indices[0]].copy()
+    for a, b in zip(indices[:-1], indices[1:]):
+        if a <= frame_index <= b:
+            t = (frame_index - a) / (b - a)
+            return KEYS[a] * (1.0 - t) + KEYS[b] * t
+    return KEYS[indices[-1]].copy()
+
+
+def removal_mask(frame: np.ndarray, frame_index: int) -> np.ndarray:
+    """Mask domino 1 plus only its tabletop-colored soft shadow."""
+    h, w = frame.shape[:2]
+    corners = np.rint(interpolated_corners(frame_index)).astype(np.int32)
+
+    core = np.zeros((h, w), np.uint8)
+    cv2.fillConvexPoly(core, corners, 255, lineType=cv2.LINE_AA)
+    # A tiny expansion catches antialiased silhouette pixels without reaching
+    # far into the touching second domino.
+    core = cv2.dilate(core, cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (3, 3)))
+
+    # The area in which domino 1 can cast its low, diffuse shadow.  Dark,
+    # near-neutral pixels are other dominoes and are explicitly protected.
+    shifted = corners + np.array([5, 8], np.int32)
+    shadow_zone = np.zeros_like(core)
+    cv2.fillConvexPoly(shadow_zone, shifted, 255, lineType=cv2.LINE_AA)
+    shadow_zone = cv2.dilate(
+        shadow_zone, cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (13, 9))
+    )
+    rgb = frame.astype(np.int16)
+    tabletop = ((rgb[..., 1] - rgb[..., 0] > 15) &
+                (rgb[..., 1] - rgb[..., 2] > 8) &
+                (rgb[..., 1] > 45))
+    shadow = ((shadow_zone > 0) & tabletop).astype(np.uint8) * 255
+    return cv2.bitwise_or(core, shadow)
+
+
+def generate() -> np.ndarray:
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    output = source.copy()
+    for i in range(FIRST_EDITED_INDEX, len(output)):
+        mask = removal_mask(source[i], i)
+        # TELEA produces a smooth continuation of the softly varying green
+        # tabletop and leaves the unmasked wall and remaining dominoes exact.
+        output[i] = cv2.inpaint(source[i], mask, 7, cv2.INPAINT_TELEA)
+    return output.astype(np.uint8)
+
+
+if __name__ == "__main__":
+    frames = generate()
+    artifact_io.save(frames, CASE_DIR)
+    print(f"saved {len(frames)} edited frames to {CASE_DIR}")
