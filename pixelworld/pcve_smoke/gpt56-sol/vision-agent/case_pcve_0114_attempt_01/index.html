+"""Generate the counterfactual: tennis ball launched at 0.6x source speed.
+
+The camera is locked.  We reconstruct its clean plate from temporal samples,
+then render the same small shaded ball along a scaled-speed rigid-body path.
+"""
+from pathlib import Path
+import numpy as np
+from PIL import Image
+import artifact_io
+
+CASE = Path(__file__).resolve().parent
+N, H, W = 96, 144, 256
+
+
+def smooth_plate(src):
+    """Robust static clean plate; only the ball moves in this shot."""
+    # Temporal median removes the moving ball everywhere except its final rest
+    # position.  Patch that small region from early frames where it was absent.
+    plate = np.median(src, axis=0).astype(np.uint8)
+    plate[87:101, 199:217] = np.median(src[:20, 87:101, 199:217], axis=0).astype(np.uint8)
+    return plate
+
+
+def sample_bilinear(im, x, y):
+    x0 = int(np.floor(x)); y0 = int(np.floor(y))
+    ax = x-x0; ay = y-y0
+    x0 = max(0, min(W-2, x0)); y0 = max(0, min(H-2, y0))
+    return ((1-ay)*(1-ax)*im[y0,x0] + (1-ay)*ax*im[y0,x0+1] +
+            ay*(1-ax)*im[y0+1,x0] + ay*ax*im[y0+1,x0+1])
+
+
+def render_ball(frame, plate, cx, cy):
+    """Draw the source-sized antialiased ball with scene-aware edge blending."""
+    # Source ball has radius about 2 px: warm highlight, lime body, dark rim.
+    x0=max(0,int(np.floor(cx-3.5))); x1=min(W-1,int(np.ceil(cx+3.5)))
+    y0=max(0,int(np.floor(cy-3.5))); y1=min(H-1,int(np.ceil(cy+3.5)))
+    for y in range(y0,y1+1):
+        for x in range(x0,x1+1):
+            # 4x4 supersampling gives stable motion between pixels.
+            cov=0.0; col=np.zeros(3,float)
+            for sy in (0.125,0.375,0.625,0.875):
+                for sx in (0.125,0.375,0.625,0.875):
+                    dx=(x+sx)-cx; dy=(y+sy)-cy; r=(dx*dx+dy*dy)**0.5
+                    if r < 2.32:
+                        # radial tennis-ball shade and upper-left specular light
+                        base=np.array([153.,183.,36.])
+                        shade=1.02 - 0.18*(r/2.32) + 0.10*(-dx-dy)/4.64
+                        c=base*shade
+                        if (dx+0.72)**2+(dy+0.78)**2 < .48**2:
+                            c=np.array([202.,213.,124.])
+                        col += np.clip(c,0,255); cov += 1
+            if cov:
+                a=cov/16.0
+                ball=col/cov
+                # Tiny dark-green optical rim, as in source antialiasing.
+                frame[y,x]=np.clip((1-a)*frame[y,x]+a*ball,0,255).astype(np.uint8)
+    # Soft one-pixel contact shadow once the ball is on court.
+    if cy >= 92.9:
+        for x in range(max(0,int(cx-3)),min(W,int(cx+4))):
+            d=abs((x+.5)-cx)/3
+            if d<1:
+                a=.10*(1-d)
+                frame[96,x]=((1-a)*frame[96,x]).astype(np.uint8)
+

+def main():
+    src=np.load(CASE/'source_pixels.npz')['rgb']
+    plate=smooth_plate(src)
+    out=np.repeat(plate[None],N,axis=0)
+
+    # Fit from source frames 0..27 (pre-impact): x = x0+vx*t,
+    # y = y0+vy*t+0.5*g*t^2. The requested intervention scales initial
+    # velocity only; gravity and court interaction remain unchanged.
+    x0,y0=28.5,59.2
+    vx=4.5109*0.6
+    vy=(-4.5530)*0.6
+    g=2*0.20938
+    ground=93.5
+

+    roots=np.roots([0.5*g,vy,y0-ground])
+    impact=float(max(r for r in roots if r>0))
+    xi=x0+vx*impact
+    # Source has an inelastic impact: vertical velocity becomes zero and the
+    # ball rolls with drag. Scale its fitted post-impact rolling speed too.
+    roll_v0=2.1893*0.6
+    roll_a=0.04485*0.6
+    stop_t=roll_v0/roll_a
+
+    for i in range(N):
+        t=float(i)
+        if t <= impact:
+            x=x0+vx*t
+            y=y0+vy*t+0.5*g*t*t
+        else:
+            tau=min(t-impact,stop_t)
+            x=xi+roll_v0*tau-0.5*roll_a*tau*tau
+            y=ground
+        render_ball(out[i],plate,x,y)
+
+    artifact_io.save(out,CASE)
+
+if __name__ == '__main__':
+    main()
