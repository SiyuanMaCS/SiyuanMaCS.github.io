+"""Remove the football from all source frames and propagate the no-collision outcome.
+
+This uses only the supplied RGB tensor.  The football is removed with texture sampled
+from the same planar floor farther to the right.  A pre-contact cutout of the pink
+ball is then continued leftward after its last clean observation, because without
+the football there is no collision to reverse it.
+"""
+from pathlib import Path
+import cv2
+import numpy as np
+from artifact_io import save
+
+CASE_DIR = Path(__file__).resolve().parent
+N, H, W = 96, 144, 256
+
+
+def pink_mask(frame):
+    hsv = cv2.cvtColor(frame, cv2.COLOR_RGB2HSV)
+    h, s, v = cv2.split(hsv)
+    m = (((h < 13) | (h > 175)) & (s > 28) & (v > 125)).astype(np.uint8)
+    # Keep only the moving ball's region, not the purple basket.
+    m[:43] = 0
+    m[:, 140:] = 0
+    n, lab, stats, _ = cv2.connectedComponentsWithStats(m)
+    if n <= 1:
+        return np.zeros((H, W), np.uint8)
+    candidates = [k for k in range(1, n) if 100 <= stats[k, cv2.CC_STAT_AREA] <= 1000]
+    if not candidates:
+        return np.zeros((H, W), np.uint8)
+    k = max(candidates, key=lambda q: stats[q, cv2.CC_STAT_AREA])
+    obj = (lab == k).astype(np.uint8)
+    obj = cv2.morphologyEx(obj, cv2.MORPH_CLOSE, np.ones((5, 5), np.uint8))
+    obj = cv2.dilate(obj, np.ones((3, 3), np.uint8), iterations=1)
+    return obj
+
+
+def soccer_track(frames):
+    """Smooth conservative center/radius track, initialized from visible circles."""
+    raw = []
+    prior = np.array([100.0, 92.0, 13.0])
+    for frame in frames:
+        gray = cv2.cvtColor(frame, cv2.COLOR_RGB2GRAY)
+        circles = cv2.HoughCircles(gray, cv2.HOUGH_GRADIENT, 1, 12,
+                                   param1=70, param2=18,
+                                   minRadius=10, maxRadius=24)
+        choices = [] if circles is None else [q for q in circles[0]
+                    if 72 < q[0] < 138 and 76 < q[1] < 139]
+        if choices:
+            prior = min(choices, key=lambda q: (q[0]-prior[0])**2 +
+                        (q[1]-prior[1])**2 + 1.5*(q[2]-prior[2])**2)
+        raw.append(prior.copy())
+    raw = np.asarray(raw)
+    # Hough is noisy during overlap. Known observed path is nearly stationary then
+    # rolls toward the camera; robust anchors model it more cleanly.
+    anchors_t = np.array([0, 18, 24, 30, 36, 43, 55, 72, 95], float)
+    anchors_x = np.array([100, 100, 101, 102, 103, 104, 104, 102, 102], float)
+    anchors_y = np.array([92, 92, 95, 104, 110, 113, 114, 114, 114], float)
+    anchors_r = np.array([13, 13, 14, 16, 18, 19, 20, 20, 20], float)
+    t = np.arange(N)
+    return np.c_[np.interp(t, anchors_t, anchors_x),
+                 np.interp(t, anchors_t, anchors_y),
+                 np.interp(t, anchors_t, anchors_r)]
+
+
+def floor_fill(frame, cx, cy, rx, ry):
+    """Replace an ellipse using a feathered sample from the same floor plane."""
+    yy, xx = np.ogrid[:H, :W]
+    hard = (((xx-cx)/rx)**2 + ((yy-cy)/ry)**2 <= 1).astype(np.float32)
+    alpha = cv2.GaussianBlur(hard, (0, 0), 1.6)[..., None]
+    # Sampling 65 px right and slightly lower preserves local plank direction and
+    # avoids both balls throughout the shot. Static camera makes this temporally stable.
+    M = np.float32([[1, 0, -65], [0, 1, -8]])
+    donor = cv2.warpAffine(frame, M, (W, H), flags=cv2.INTER_LINEAR,
+                           borderMode=cv2.BORDER_REFLECT)
+    return np.clip(frame.astype(np.float32)*(1-alpha) +
+                   donor.astype(np.float32)*alpha, 0, 255).astype(np.uint8)
+
+
+def make_ball_sprite(frame, mask):
+    ys, xs = np.nonzero(mask)
+    x0, x1 = max(0, xs.min()-3), min(W, xs.max()+4)
+    y0, y1 = max(0, ys.min()-3), min(H, ys.max()+4)
+    rgb = frame[y0:y1, x0:x1].copy()
+    a = cv2.GaussianBlur(mask[y0:y1, x0:x1].astype(np.float32), (0, 0), .8)
+    return rgb, np.clip(a, 0, 1), ((x0+x1-1)/2, (y0+y1-1)/2)
+
+
+def paste_sprite(dst, sprite, alpha, source_center, target_center):
+    sh, sw = alpha.shape
+    x0 = int(round(target_center[0] - (source_center[0] % 1) - sw/2))
+    y0 = int(round(target_center[1] - (source_center[1] % 1) - sh/2))
+    x1, y1 = x0+sw, y0+sh
+    dx0, dy0 = max(0, x0), max(0, y0)
+    dx1, dy1 = min(W, x1), min(H, y1)
+    if dx1 <= dx0 or dy1 <= dy0:
+        return dst
+    sx0, sy0 = dx0-x0, dy0-y0
+    sx1, sy1 = sx0+(dx1-dx0), sy0+(dy1-dy0)
+    a = alpha[sy0:sy1, sx0:sx1, None]
+    dst[dy0:dy1, dx0:dx1] = np.clip(
+        dst[dy0:dy1, dx0:dx1].astype(float)*(1-a) +
+        sprite[sy0:sy1, sx0:sx1].astype(float)*a, 0, 255).astype(np.uint8)
+    return dst
+
+
+def main():
+    frames = np.load(CASE_DIR / 'source_pixels.npz')['rgb'].copy()
+    track = soccer_track(frames)
+
+    # Last clean pre-contact pink ball appearance and center (source frame 12, 0-based 11).
+    pm = pink_mask(frames[11])
+    sprite, salpha, source_center = make_ball_sprite(frames[11], pm)
+
+    out = np.empty_like(frames)
+    for i, frame in enumerate(frames):
+        cx, cy, r = track[i]
+        # Extra vertical coverage removes contact shadows and every edge pixel.
+        edited = floor_fill(frame, cx, cy, r+4.0, r+6.0)
+
+        if i >= 12:
+            # Remove the factual post-contact pink ball before inserting its
+            # counterfactual continuation. A slightly broad mask also removes its shadow.
+            m = pink_mask(frame)
+            if m.any():
+                ys, xs = np.nonzero(m)
+                pcx, pcy = xs.mean(), ys.mean()
+                prx = max(14.0, (xs.max()-xs.min()+1)/2 + 4)
+                pry = max(14.0, (ys.max()-ys.min()+1)/2 + 4)
+                edited = floor_fill(edited, pcx, pcy, prx, pry)
+
+            # Before collision it traversed right-to-left at ~14 px/frame. Continue
+            # this inertia; it exits the left edge during frame 17.
+            tx = 82.6 - 13.6*(i-11)
+            ty = 65.4 + 0.15*(i-11)
+            edited = paste_sprite(edited, sprite, salpha, source_center, (tx, ty))
+        out[i] = edited
+
+    assert out.shape == (96, 144, 256, 3) and out.dtype == np.uint8
+    save(out, CASE_DIR)
+
+
+if __name__ == '__main__':
+    main()

\"\"\"Remove the football from all source frames and propagate the no-collision outcome.

This uses only the supplied RGB tensor.  The football is removed with texture sampled
from the same planar floor farther to the right.  A pre-contact cutout of the pink
ball is then continued leftward after its last clean observation, because without
the football there is no collision to reverse it.

from pathlib import Path
import cv2
import numpy as np
from artifact_io import save

CASE_DIR = Path(__file__).resolve().parent
N, H, W = 96, 144, 256


def pink_mask(frame):
    hsv = cv2.cvtColor(frame, cv2.COLOR_RGB2HSV)
    h, s, v = cv2.split(hsv)
    m = (((h < 13) | (h > 175)) & (s > 28) & (v > 125)).astype(np.uint8)
    # Keep only the moving ball's region, not the purple basket.
    m[:43] = 0
    m[:, 140:] = 0
    n, lab, stats, _ = cv2.connectedComponentsWithStats(m)
    if n <= 1:
        return np.zeros((H, W), np.uint8)
    candidates = [k for k in range(1, n) if 100 <= stats[k, cv2.CC_STAT_AREA] <= 1000]
    if not candidates:
        return np.zeros((H, W), np.uint8)
    k = max(candidates, key=lambda q: stats[q, cv2.CC_STAT_AREA])
    obj = (lab == k).astype(np.uint8)
    obj = cv2.morphologyEx(obj, cv2.MORPH_CLOSE, np.ones((5, 5), np.uint8))
    obj = cv2.dilate(obj, np.ones((3, 3), np.uint8), iterations=1)
    return obj


def soccer_track(frames):
    \"\"\"Smooth conservative center/radius track, initialized from visible circles.\"\"\"
    raw = []
    prior = np.array([100.0, 92.0, 13.0])
    for frame in frames:
        gray = cv2.cvtColor(frame, cv2.COLOR_RGB2GRAY)
        circles = cv2.HoughCircles(gray, cv2.HOUGH_GRADIENT, 1, 12,
                                   param1=70, param2=18,
                                   minRadius=10, maxRadius=24)
        choices = [] if circles is None else [q for q in circles[0]
                    if 72 < q[0] < 138 and 76 < q[1] < 139]
        if choices:
            prior = min(choices, key=lambda q: (q[0]-prior[0])**2 +
                        (q[1]-prior[1])**2 + 1.5*(q[2]-prior[2])**2)
        raw.append(prior.copy())
    raw = np.asarray(raw)
    # Hough is noisy during overlap. Known observed path is nearly stationary then
    # rolls toward the camera; robust anchors model it more cleanly.
    anchors_t = np.array([0, 18, 24, 30, 36, 43, 55, 72, 95], float)
    anchors_x = np.array([100, 100, 101, 102, 103, 104, 104, 102, 102], float)
    anchors_y = np.array([92, 92, 95, 104, 110, 113, 114, 114, 114], float)
    anchors_r = np.array([13, 13, 14, 16, 18, 19, 20, 20, 20], float)
    t = np.arange(N)
    return np.c_[np.interp(t, anchors_t, anchors_x),
                 np.interp(t, anchors_t, anchors_y),
                 np.interp(t, anchors_t, anchors_r)]


def floor_fill(frame, cx, cy, rx, ry):
    \"\"\"Replace an ellipse using a feathered sample from the same floor plane.\"\"\"
    yy, xx = np.ogrid[:H, :W]
    hard = (((xx-cx)/rx)**2 + ((yy-cy)/ry)**2 <= 1).astype(np.float32)
    alpha = cv2.GaussianBlur(hard, (0, 0), 1.6)[..., None]
    # Sampling 65 px right and slightly lower preserves local plank direction and
    # avoids both balls throughout the shot. Static camera makes this temporally stable.
    M = np.float32([[1, 0, -65], [0, 1, -8]])
    donor = cv2.warpAffine(frame, M, (W, H), flags=cv2.INTER_LINEAR,
                           borderMode=cv2.BORDER_REFLECT)
    return np.clip(frame.astype(np.float32)*(1-alpha) +
                   donor.astype(np.float32)*alpha, 0, 255).astype(np.uint8)


def make_ball_sprite(frame, mask):
    ys, xs = np.nonzero(mask)
    x0, x1 = max(0, xs.min()-3), min(W, xs.max()+4)
    y0, y1 = max(0, ys.min()-3), min(H, ys.max()+4)
    rgb = frame[y0:y1, x0:x1].copy()
    a = cv2.GaussianBlur(mask[y0:y1, x0:x1].astype(np.float32), (0, 0), .8)
    return rgb, np.clip(a, 0, 1), ((x0+x1-1)/2, (y0+y1-1)/2)


def paste_sprite(dst, sprite, alpha, source_center, target_center):
    sh, sw = alpha.shape
    x0 = int(round(target_center[0] - (source_center[0] % 1) - sw/2))
    y0 = int(round(target_center[1] - (source_center[1] % 1) - sh/2))
    x1, y1 = x0+sw, y0+sh
    dx0, dy0 = max(0, x0), max(0, y0)
    dx1, dy1 = min(W, x1), min(H, y1)
    if dx1 <= dx0 or dy1 <= dy0:
        return dst
    sx0, sy0 = dx0-x0, dy0-y0
    sx1, sy1 = sx0+(dx1-dx0), sy0+(dy1-dy0)
    a = alpha[sy0:sy1, sx0:sx1, None]
    dst[dy0:dy1, dx0:dx1] = np.clip(
        dst[dy0:dy1, dx0:dx1].astype(float)*(1-a) +
        sprite[sy0:sy1, sx0:sx1].astype(float)*a, 0, 255).astype(np.uint8)
    return dst


def main():
    frames = np.load(CASE_DIR / 'source_pixels.npz')['rgb'].copy()
    track = soccer_track(frames)

    # Last clean pre-contact pink ball appearance and center (source frame 12, 0-based 11).
    pm = pink_mask(frames[11])
    sprite, salpha, source_center = make_ball_sprite(frames[11], pm)

    out = np.empty_like(frames)
    for i, frame in enumerate(frames):
        cx, cy, r = track[i]
        # Extra vertical coverage removes contact shadows and every edge pixel.
        edited = floor_fill(frame, cx, cy, r+4.0, r+6.0)

        if i >= 12:
            # Remove the factual post-contact pink ball before inserting its
            # counterfactual continuation. A slightly broad mask also removes its shadow.
            m = pink_mask(frame)
            if m.any():
                ys, xs = np.nonzero(m)
                pcx, pcy = xs.mean(), ys.mean()
                prx = max(14.0, (xs.max()-xs.min()+1)/2 + 4)
                pry = max(14.0, (ys.max()-ys.min()+1)/2 + 4)
                edited = floor_fill(edited, pcx, pcy, prx, pry)

            # Before collision it traversed right-to-left at ~14 px/frame. Continue
            # this inertia; it exits the left edge during frame 17.
            tx = 82.6 - 13.6*(i-11)
            ty = 65.4 + 0.15*(i-11)
            edited = paste_sprite(edited, sprite, salpha, source_center, (tx, ty))
        out[i] = edited

    assert out.shape == (96, 144, 256, 3) and out.dtype == np.uint8
    save(out, CASE_DIR)


if __name__ == '__main__':
    main()
