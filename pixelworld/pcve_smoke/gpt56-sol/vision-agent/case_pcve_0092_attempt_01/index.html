+The camera is locked.  We retain source imagery and source-rendered ball sprites,
+remove only the factual cue ball after the new first collision, and advance the

+import artifact_io

+def cue_centers(frames):
+    """Track the bright cue ball (largest compact neutral component on felt)."""
+    out = []
+    for im in frames:
+        hsv = cv2.cvtColor(im, cv2.COLOR_RGB2HSV)
+        m = ((im[..., 0] > 140) & (im[..., 1] > 140) &
+             (im[..., 2] > 130) & (hsv[..., 1] < 55)).astype(np.uint8)
+        m[:45] = 0; m[116:] = 0; m[:, :55] = 0; m[:, 205:] = 0
+        n, lab, stats, cen = cv2.connectedComponentsWithStats(m)
+        candidates = []
+        for k in range(1, n):
+            x, y, w, h, area = stats[k]
+            if 18 <= area <= 80 and w <= 13 and h <= 13:
+                candidates.append((area, cen[k]))
+        if candidates:
+            out.append(max(candidates, key=lambda z: z[0])[1])
+        else:
+            out.append(out[-1].copy())
+    return np.asarray(out, np.float32)
+
+
+def extract_sprite(actual, clean, cx, cy, half=8):
+    """Extract a small source-rendered object and its soft edge/shadow matte."""
+    cx, cy = int(round(cx)), int(round(cy))
+    ys = slice(cy-half, cy+half+1); xs = slice(cx-half, cx+half+1)
+    a = actual[ys, xs].copy()
+    b = clean[ys, xs]
+    delta = np.max(np.abs(a.astype(np.int16)-b.astype(np.int16)), axis=2)
+    core = (delta > 7).astype(np.uint8)
+    core = cv2.morphologyEx(core, cv2.MORPH_CLOSE, np.ones((3,3),np.uint8))
+    # Keep the component(s) around the center plus the cast shadow below it.
+    keep = np.zeros_like(core)
+    keep[max(0,half-5):min(2*half+1,half+8),
+         max(0,half-6):min(2*half+1,half+7)] = 1
+    core &= keep
+    matte = cv2.GaussianBlur(core.astype(np.float32), (3,3), 0.55)
+    matte[core > 0] = np.maximum(matte[core > 0], .92)
+    return a, np.clip(matte, 0, 1)
+
+
+def yellowize(sprite, alpha, clean_patch):
+    """Turn the source white sphere yellow while retaining lighting and shadow."""
+    s = sprite.astype(np.float32)
+    hsv = cv2.cvtColor(sprite, cv2.COLOR_RGB2HSV)
+    # Sphere is bright and nearly neutral; cast shadow remains source-green.
+    sphere = (alpha > .15) & (hsv[...,1] < 75) & (np.max(s,axis=2) > 105)
+    lum = cv2.cvtColor(sprite, cv2.COLOR_RGB2GRAY).astype(np.float32)
+    shade = np.clip(lum / 192.0, .52, 1.24)
+    base = np.array([238., 184., 24.], np.float32)
+    colored = base[None,None,:] * shade[...,None]
+    # Preserve the tiny source highlight as warm near-white specular light.
+    hi = np.clip((lum-202.)/45., 0, 1)[...,None]
+    colored = colored*(1-hi) + np.array([255.,244.,184.])*hi
+    s[sphere] = colored[sphere]
+    return np.clip(s,0,255).astype(np.uint8)
+

+def composite(frame, sprite, alpha, center):
+    h, w = alpha.shape
+    cx, cy = int(round(center[0])), int(round(center[1]))
+    x0, y0 = cx-w//2, cy-h//2
+    x1, y1 = x0+w, y0+h
+    fx0, fy0, fx1, fy1 = max(0,x0),max(0,y0),min(256,x1),min(144,y1)
+    if fx0>=fx1 or fy0>=fy1: return
+    sx0, sy0 = fx0-x0, fy0-y0
+    aa = alpha[sy0:sy0+fy1-fy0, sx0:sx0+fx1-fx0, None]
+    ss = sprite[sy0:sy0+fy1-fy0, sx0:sx0+fx1-fx0].astype(np.float32)
+    dst = frame[fy0:fy1,fx0:fx1].astype(np.float32)
+    frame[fy0:fy1,fx0:fx1] = np.clip(dst*(1-aa)+ss*aa,0,255).astype(np.uint8)
+

+def erase_cue(im, center, delicate=False):
+    """Inpaint factual cue ball. Early on, avoid touching the adjacent black ball."""
+    cx, cy = map(int, np.round(center))
+    mask = np.zeros(im.shape[:2], np.uint8)
+    if delicate:

+        yy, xx = np.ogrid[:144,:256]
+        near = (xx-cx)**2 + (yy-cy)**2 <= 7**2
+        bright = (hsv[...,1] < 65) & (np.max(im,axis=2)>112)
+        mask[near & bright] = 255
+        mask = cv2.dilate(mask, np.ones((3,3),np.uint8), iterations=1)
+    else:

+    return cv2.inpaint(im, mask, 3, cv2.INPAINT_TELEA)

+def erase_black(im, center):

+    cx,cy=map(int,np.round(center))
+    cv2.circle(mask,(cx,cy),5,255,-1)
+    cv2.ellipse(mask,(cx+1,cy+3),(4,3),0,0,360,255,-1)
+    return cv2.inpaint(im,mask,3,cv2.INPAINT_TELEA)
+

+def build():

+    # Source-rendered stopped cue sprite, taken precisely at the new impact point.
+    cue_source_index = 8

+    yellow_rest = target - direction*7.15

+    # tail frames after advancing its source response.
+    black_center = np.array([134.0,66.75],np.float32)

+        else:

    cv2.ellipse(mask,(cx+1,cy+3),(4,3),0,0,360,255,-1)
    return cv2.inpaint(im,mask,3,cv2.INPAINT_TELEA)


def build():
    cues = cue_centers(SRC)

    # Source-rendered stopped cue sprite, taken precisely at the new impact point.
    cue_source_index = 8
    cue_center = cues[cue_source_index]
    cue_rgb, cue_a = extract_sprite(SRC[cue_source_index], SRC[95],
                                    cue_center[0], cue_center[1], 8)
    yellow_rgb = yellowize(cue_rgb, cue_a, None)

    # Prompt geometry, measured in frame 1: P = target + 1/4(cue-target).
    initial_cue = cues[0]
    target = np.array([127.65, 70.65], np.float32)
    yellow_start = target + .25*(initial_cue-target)
    direction = (target-initial_cue)/np.linalg.norm(target-initial_cue)
    yellow_rest = target - direction*7.15

    # Actual last-frame black-ball sprite, used only for the three extrapolated
    # tail frames after advancing its source response.
    black_center = np.array([134.0,66.75],np.float32)
    black_clean = erase_black(SRC[95].copy(), black_center)
    black_rgb, black_a = extract_sprite(SRC[95], black_clean,
                                        black_center[0],black_center[1],7)

    result=[]
    for i in range(96):
        if i <= 8:
            frame=SRC[i].copy()
        elif i == 9:
            frame=erase_cue(SRC[i].copy(),cues[i],delicate=True)
            composite(frame,cue_rgb,cue_a,cue_center)
        else:
            # Target receives the transferred impulse three frames sooner.
            j=min(i+3,95)
            frame=erase_cue(SRC[j].copy(),cues[j],delicate=(j<18))
            composite(frame,cue_rgb,cue_a,cue_center)
            if i>=93:
                # Continue the returning black ball rather than freezing it.
                frame=erase_black(frame,black_center)
                n=i-92
                pos=black_center + n*np.array([-.72,.46],np.float32)
                composite(frame,black_rgb,black_a,pos)

        # Inserted one-ball exists from frame 1. It is struck at frame 9,
        # transfers its velocity to the target over two frames, then rests.
        if i <= 8:
            yp=yellow_start
        elif i == 9:
            yp=yellow_start*.55 + yellow_rest*.45
        else:
            yp=yellow_rest
        composite(frame,yellow_rgb,cue_a,yp)
        result.append(frame)
    return np.asarray(result,dtype=np.uint8)
