+Only the two mallets participating in the collision are re-composited.  The
+unaltered source frames remain the clean reference for the room, table, white
+mallet and lighting.

+import cv2
+import numpy as np
+import artifact_io
+
+CASE_DIR = Path(__file__).resolve().parent
+COLLISION_ANCHOR = 14  # zero-based: source frame 15, last pre-impulse pose
+NEW_RESTITUTION_RATIO = 0.30
+# For equal masses with one initially at rest, post-impact speed fractions are
+# (1-e)/2 and (1+e)/2.  The source collision is visually near-elastic.
+BLUE_FRACTION = (1.0 - NEW_RESTITUTION_RATIO) / 2.0  # 0.35
+RED_FRACTION = (1.0 + NEW_RESTITUTION_RATIO) / 2.0   # 0.65
+

+def nearest_component(mask, expected, min_area=5):
+    """Return the component nearest expected as a uint8 mask and centroid."""
+    count, labels, stats, centers = cv2.connectedComponentsWithStats(mask)
+    choices = []
+    ex = np.asarray(expected, np.float32)
+    for label in range(1, count):
+        area = int(stats[label, cv2.CC_STAT_AREA])
+        if area >= min_area:
+            distance = float(np.linalg.norm(centers[label] - ex))
+            choices.append((distance, -area, label))
+    if not choices:
+        raise RuntimeError(f"Could not locate object near {expected}")
+    label = min(choices)[2]
+    return (labels == label).astype(np.uint8), centers[label].astype(np.float32)
+
+
+def track_mallets(frames):
+    """Track saturated blue/red mallet bodies in the table ROI."""
+    blue_centers, red_centers = [], []
+    prev_b = np.array([167.0, 58.5], np.float32)
+    prev_r = np.array([140.3, 67.8], np.float32)
+    for frame in frames:
+        rgb = frame.astype(np.int16)
+        r, g, b = rgb[..., 0], rgb[..., 1], rgb[..., 2]
+        roi = np.zeros(frame.shape[:2], np.uint8)
+        roi[45:108, 85:185] = 1
+        blue_raw = (((b-r) > 35) & ((b-g) > 10) & (b > 95) & (roi > 0)).astype(np.uint8)
+        red_raw = (((r-g) > 30) & ((r-b) > 18) & (r > 105) & (roi > 0)).astype(np.uint8)
+        _, cb = nearest_component(blue_raw, prev_b)
+        _, cr = nearest_component(red_raw, prev_r)
+        blue_centers.append(cb); red_centers.append(cr)
+        prev_b, prev_r = cb, cr
+    return np.asarray(blue_centers), np.asarray(red_centers)
+
+
+def object_alpha(frame, center, color):
+    """Soft matte around one mallet, retaining highlights and edge antialiasing."""
+    rgb = frame.astype(np.int16)
+    r, g, b = rgb[..., 0], rgb[..., 1], rgb[..., 2]
+    yy, xx = np.mgrid[:frame.shape[0], :frame.shape[1]]
+    cx, cy = center
+    local = (((xx-cx)/12.0)**2 + ((yy-cy)/9.0)**2) <= 1.0
+    if color == 'blue':
+        chroma = ((b-r) > 24) & ((b-g) > 4) & (b > 85)
+    else:
+        chroma = ((r-g) > 24) & ((r-b) > 12) & (r > 95)
+    core = (local & chroma).astype(np.uint8)
+    # Close tiny highlight holes, then include the rendered antialiased fringe.
+    core = cv2.morphologyEx(core, cv2.MORPH_CLOSE,
+                            cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (3, 3)))
+    solid = cv2.dilate(core, cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (5, 5)))
+    # Restrict dilation to the mallet-sized footprint so rink markings cannot attach.
+    solid = (solid * local.astype(np.uint8))
+    alpha = cv2.GaussianBlur(solid.astype(np.float32), (0, 0), 0.70)
+    return np.clip(alpha, 0.0, 1.0)
+
+
+def shifted_layer(frame, alpha, shift):
+    """Translate premultiplied RGB and alpha with subpixel interpolation."""
+    h, w = alpha.shape
+    matrix = np.array([[1.0, 0.0, float(shift[0])],
+                       [0.0, 1.0, float(shift[1])]], np.float32)
+    premul = frame.astype(np.float32) * alpha[..., None]
+    moved_rgb = cv2.warpAffine(premul, matrix, (w, h), flags=cv2.INTER_LINEAR,
+                               borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+    moved_a = cv2.warpAffine(alpha, matrix, (w, h), flags=cv2.INTER_LINEAR,
+                             borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+    return moved_rgb, np.clip(moved_a, 0.0, 1.0)
+
+
+def composite(base, layer_rgb, layer_alpha):
+    a = layer_alpha[..., None]
+    return base * (1.0-a) + layer_rgb
+
+
+def generate():
+    source = np.load(CASE_DIR / 'source_pixels.npz')['rgb']
+    if source.shape != (96, 144, 256, 3) or source.dtype != np.uint8:
+        raise ValueError(f'Unexpected source tensor: {source.shape}, {source.dtype}')
+    blue, red = track_mallets(source)
+    out = source.copy()
+    b0, r0 = blue[COLLISION_ANCHOR], red[COLLISION_ANCHOR]
+
+    for i in range(COLLISION_ANCHOR + 1, len(source)):
+        # The original red trajectory supplies the observed drag/settling profile.
+        original_departure = red[i] - r0
+        desired_r = r0 + RED_FRACTION * original_departure
+        desired_b = b0 + BLUE_FRACTION * original_departure
+        shift_r = desired_r - red[i]
+        shift_b = desired_b - blue[i]
+
+        alpha_b = object_alpha(source[i], blue[i], 'blue')
+        alpha_r = object_alpha(source[i], red[i], 'red')
+
+        # Remove original rendered mallets.  A slightly larger union includes their
+        # soft contact shadows; TELEA reconstructs the smooth lavender table and
+        # thin white rink graphics beneath them.
+        removal = np.maximum(alpha_b, alpha_r)
+        removal = cv2.dilate((removal > 0.025).astype(np.uint8),
+                             cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (5, 5)))
+        clean = cv2.inpaint(source[i], removal, 3.0, cv2.INPAINT_TELEA).astype(np.float32)
+
+        red_rgb, red_a = shifted_layer(source[i], alpha_r, shift_r)
+        blue_rgb, blue_a = shifted_layer(source[i], alpha_b, shift_b)
+        # Red is composited first; at contact this preserves the source-like blue
+        # upper-right / red lower-left ordering.
+        clean = composite(clean, red_rgb, red_a)
+        clean = composite(clean, blue_rgb, blue_a)
+        out[i] = np.clip(np.rint(clean), 0, 255).astype(np.uint8)
+
+    # The restitution parameter cannot affect free flight before first contact.
+    assert np.array_equal(out[:COLLISION_ANCHOR+1], source[:COLLISION_ANCHOR+1])
+    return out
+
+
+if __name__ == '__main__':
+    frames = generate()
+    artifact_io.save(frames, CASE_DIR)

    blue, red = track_mallets(source)
    out = source.copy()
    b0, r0 = blue[COLLISION_ANCHOR], red[COLLISION_ANCHOR]

    for i in range(COLLISION_ANCHOR + 1, len(source)):
        # The original red trajectory supplies the observed drag/settling profile.
        original_departure = red[i] - r0
        desired_r = r0 + RED_FRACTION * original_departure
        desired_b = b0 + BLUE_FRACTION * original_departure
        shift_r = desired_r - red[i]
        shift_b = desired_b - blue[i]

        alpha_b = object_alpha(source[i], blue[i], 'blue')
        alpha_r = object_alpha(source[i], red[i], 'red')

        # Remove original rendered mallets.  A slightly larger union includes their
        # soft contact shadows; TELEA reconstructs the smooth lavender table and
        # thin white rink graphics beneath them.
        removal = np.maximum(alpha_b, alpha_r)
        removal = cv2.dilate((removal > 0.025).astype(np.uint8),
                             cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (5, 5)))
        clean = cv2.inpaint(source[i], removal, 3.0, cv2.INPAINT_TELEA).astype(np.float32)

        red_rgb, red_a = shifted_layer(source[i], alpha_r, shift_r)
        blue_rgb, blue_a = shifted_layer(source[i], alpha_b, shift_b)
        # Red is composited first; at contact this preserves the source-like blue
        # upper-right / red lower-left ordering.
        clean = composite(clean, red_rgb, red_a)
        clean = composite(clean, blue_rgb, blue_a)
        out[i] = np.clip(np.rint(clean), 0, 255).astype(np.uint8)

    # The restitution parameter cannot affect free flight before first contact.
    assert np.array_equal(out[:COLLISION_ANCHOR+1], source[:COLLISION_ANCHOR+1])
    return out
