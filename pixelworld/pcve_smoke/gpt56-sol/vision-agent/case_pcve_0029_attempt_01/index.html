+#!/usr/bin/env python3
+"""Generate the restitution counterfactual for this PixelWorld case.
+
+The source ball follows a clean ballistic trajectory.  The first source impact gives
+an original normal restitution of about 0.80; the requested intervention therefore
+uses 0.96.  Up to that impact the factual RGB frames are retained exactly.  After
+it, the moving source ball/shadow are clean-plated from nearby source frames and a
+source-derived, rotating ball plus physically coupled contact shadow are composited
+on the altered trajectory.
+"""
+from pathlib import Path
+import math
+import cv2
+import numpy as np
+from artifact_io import save
+
+CASE_DIR = Path(__file__).resolve().parent
+N, H, W = 96, 144, 256
+
+
+def circle_centers(frames):
+    xs, ys = [], []
+    for frame in frames:
+        gray = cv2.cvtColor(frame, cv2.COLOR_RGB2GRAY)
+        cs = cv2.HoughCircles(gray, cv2.HOUGH_GRADIENT, 1, 20,
+                              param1=60, param2=18, minRadius=8, maxRadius=14)
+        if cs is None:
+            xs.append(np.nan); ys.append(np.nan)
+        else:
+            xs.append(float(cs[0, 0, 0])); ys.append(float(cs[0, 0, 1]))
+    return np.asarray(xs), np.asarray(ys)
+
+
+def clean_plate(frames, centers):
+    """Remove only the source ball and its local floor shadow.
+
+    For each target pixel use the nearest-time source sample in which the moving
+    object is absent.  This preserves static texture and the source's mild exposure
+    changes substantially better than a global median.
+    """
+    out = frames.copy()
+    yy, xx = np.mgrid[:H, :W]
+    masks = []
+    for i, (x, y) in enumerate(centers):
+        # Ball matte; at contact also includes the small source contact shadow.
+        ball = ((xx - x) / 13.4) ** 2 + ((yy - y) / 13.2) ** 2 < 1
+        floor = 82.46 + .045 * i
+        sh = (((xx - (x + 1.2)) / 16.0) ** 2 +
+              ((yy - (floor + 4.2)) / 5.8) ** 2 < 1) & (y > floor - 25)
+        masks.append(ball | sh)
+    masks = np.asarray(masks)
+
+    for i in range(12, N):
+        m = masks[i]
+        ys0, xs0 = np.where(m)
+        if not len(xs0):
+            continue
+        y0, y1 = max(0, ys0.min()-2), min(H, ys0.max()+3)
+        x0, x1 = max(0, xs0.min()-2), min(W, xs0.max()+3)
+        local = m[y0:y1, x0:x1]
+        # Choose source pixels from the temporally closest object-free sample.
+        best = np.full(local.shape, -1, np.int16)
+        for j in sorted(range(N), key=lambda j: abs(j-i)):
+            available = local & (~masks[j, y0:y1, x0:x1]) & (best < 0)
+            best[available] = j
+            if np.all((~local) | (best >= 0)):
+                break
+        oy, ox = np.where(local)
+        jj = best[oy, ox]
+        valid = jj >= 0
+        out[i, y0+oy[valid], x0+ox[valid]] = frames[jj[valid], y0+oy[valid], x0+ox[valid]]
+        # Tiny anti-aliased boundary repair, confined to uncertain edge pixels.
+        ring = cv2.dilate(local.astype(np.uint8), np.ones((3,3),np.uint8), 1).astype(bool) & ~local
+        # no replacement on ring: its source alpha is already near zero
+    return out
+
+
+def trajectory():
+    # Pixel-space projectile parameters fitted to factual free-flight centers.
+    g = 0.7353                 # px / frame^2
+    y, vy = 21.599, 0.5574
+    e_new = 0.80 * 1.20        # requested 1.2x restitution
+    dt = 1/80
+    pos = np.zeros(N, np.float64)
+    vel = np.zeros(N, np.float64)
+    contacts = []
+    sample = 0
+    t = 0.0
+    while sample < N:
+        if t + 1e-10 >= sample:
+            pos[sample], vel[sample] = y, vy
+            sample += 1
+            if sample == N: break
+        floor_next = 82.46 + .045 * (t + dt)
+        yn = y + vy*dt + .5*g*dt*dt
+        vn = vy + g*dt
+        if yn >= floor_next and vn > .045:
+            # Resolve impact at this short step; sub-frame error is < .01 px.
+            yn = floor_next
+            vn = .045 - e_new * (vn - .045)
+            contacts.append(t + dt)
+        y, vy, t = yn, vn, t + dt
+    # Horizontal path has a slight perspective deceleration in the source.
+    ii = np.arange(N, dtype=np.float64)
+    x = 127.51 + .3634*ii - .0011053*ii*ii
+    return x, pos, vel, contacts
+
+
+def extract_ball(frames, sx, sy):
+    """Extract source ball patches and soft alpha mattes for source-derived rendering."""
+    patches, alphas = [], []
+    yy, xx = np.mgrid[:29, :29]
+    radial = np.sqrt((xx-14)**2 + (yy-14)**2)
+    for i in range(N):
+        cx, cy = sx[i], sy[i]
+        M = np.array([[1,0,14-cx],[0,1,14-cy]], np.float32)
+        p = cv2.warpAffine(frames[i], M, (29,29), flags=cv2.INTER_CUBIC,
+                           borderMode=cv2.BORDER_REFLECT_101)
+        # Slightly feathered circular silhouette. Radius 11.55 matches source.
+        a = np.clip((12.25-radial)/1.15, 0, 1).astype(np.float32)
+        patches.append(p); alphas.append(a)
+    return np.asarray(patches), np.asarray(alphas)
+
+
+def paste_rgba(dst, rgb, alpha, cx, cy):
+    h, w = alpha.shape
+    x0 = int(math.floor(cx - (w-1)/2)); y0 = int(math.floor(cy - (h-1)/2))
+    x1, y1 = x0+w, y0+h
+    dx0, dy0 = max(0,x0), max(0,y0); dx1, dy1 = min(W,x1), min(H,y1)
+    if dx0 >= dx1 or dy0 >= dy1: return
+    px0, py0 = dx0-x0, dy0-y0; px1, py1 = px0+dx1-dx0, py0+dy1-dy0
+    a = alpha[py0:py1,px0:px1,None]
+    reg = dst[dy0:dy1,dx0:dx1].astype(np.float32)
+    dst[dy0:dy1,dx0:dx1] = np.clip(reg*(1-a) + rgb[py0:py1,px0:px1]*a,0,255).astype(np.uint8)
+
+
+def render():
+    src = np.load(CASE_DIR/'source_pixels.npz')['rgb']
+    assert src.shape == (N,H,W,3) and src.dtype == np.uint8
+    sx, sy = circle_centers(src)
+    # Hough misses a few compressed contact frames; interpolate just those.
+    good = np.isfinite(sx)
+    sx = np.interp(np.arange(N), np.flatnonzero(good), sx[good])
+    sy = np.interp(np.arange(N), np.flatnonzero(good), sy[good])
+    plate = clean_plate(src, np.c_[sx,sy])
+    ball, alpha = extract_ball(src, sx, sy)
+    x, y, vy, contacts = trajectory()
+
+    out = src.copy()  # exact factual history through frame 12's first contact
+    for i in range(13, N):
+        frame = plate[i].copy()
+        floor = 82.46 + .045*i
+        clearance = max(0.0, floor-y[i])
+
+        # Physically coupled floor shadow: compact/dark at contact, broader and
+        # lighter aloft. A warm-neutral multiply preserves the wooden texture.
+        sigma_x = 5.4 + min(clearance,55)*.055
+        sigma_y = 1.35 + min(clearance,55)*.018
+        strength = .25*math.exp(-clearance/22.0) + .018
+        yy, xx = np.mgrid[:H,:W]
+        sh = np.exp(-.5*(((xx-(x[i]+1.5))/sigma_x)**2 +
+                         ((yy-(floor+4.0))/sigma_y)**2)) * strength
+        frame = np.clip(frame.astype(np.float32)*(1-sh[...,None]),0,255).astype(np.uint8)
+
+        # The source already supplies natural rotation. Pick a source patch by a
+        # continued angular phase, avoiding squashy contact frames.
+        tex_idx = int((i*1.35) % 29)  # frames 0..28 are clean, round appearances
+        paste_rgba(frame, ball[tex_idx], alpha[tex_idx], x[i], y[i])
+        out[i] = frame
+    return out, contacts
+
+
+if __name__ == '__main__':
+    frames, contacts = render()
+    save(frames, CASE_DIR)
+    print('saved', frames.shape, frames.dtype, 'altered impact frames:',
+          ', '.join(f'{t:.2f}' for t in contacts))

#"'!/usr/bin/env python3
"""Generate the restitution counterfactual for this PixelWorld case.

The source ball follows a clean ballistic trajectory.  The first source impact gives
an original normal restitution of about 0.80; the requested intervention therefore
uses 0.96.  Up to that impact the factual RGB frames are retained exactly.  After
it, the moving source ball/shadow are clean-plated from nearby source frames and a
source-derived, rotating ball plus physically coupled contact shadow are composited
on the altered trajectory.
"""
from pathlib import Path
import math
import cv2
import numpy as np
from artifact_io import save

CASE_DIR = Path(__file__).resolve().parent
N, H, W = 96, 144, 256


def circle_centers(frames):
    xs, ys = [], []
    for frame in frames:
        gray = cv2.cvtColor(frame, cv2.COLOR_RGB2GRAY)
        cs = cv2.HoughCircles(gray, cv2.HOUGH_GRADIENT, 1, 20,
                              param1=60, param2=18, minRadius=8, maxRadius=14)
        if cs is None:
            xs.append(np.nan); ys.append(np.nan)
        else:
            xs.append(float(cs[0, 0, 0])); ys.append(float(cs[0, 0, 1]))
    return np.asarray(xs), np.asarray(ys)


def clean_plate(frames, centers):
    """Remove only the source ball and its local floor shadow.

    For each target pixel use the nearest-time source sample in which the moving
    object is absent.  This preserves static texture and the source'"'s mild exposure
    changes substantially better than a global median.
    \"\"\"
    out = frames.copy()
    yy, xx = np.mgrid[:H, :W]
    masks = []
    for i, (x, y) in enumerate(centers):
        # Ball matte; at contact also includes the small source contact shadow.
        ball = ((xx - x) / 13.4) ** 2 + ((yy - y) / 13.2) ** 2 < 1
        floor = 82.46 + .045 * i
        sh = (((xx - (x + 1.2)) / 16.0) ** 2 +
              ((yy - (floor + 4.2)) / 5.8) ** 2 < 1) & (y > floor - 25)
        masks.append(ball | sh)
    masks = np.asarray(masks)

    for i in range(12, N):
        m = masks[i]
        ys0, xs0 = np.where(m)
        if not len(xs0):
            continue
        y0, y1 = max(0, ys0.min()-2), min(H, ys0.max()+3)
        x0, x1 = max(0, xs0.min()-2), min(W, xs0.max()+3)
        local = m[y0:y1, x0:x1]
        # Choose source pixels from the temporally closest object-free sample.
        best = np.full(local.shape, -1, np.int16)
        for j in sorted(range(N), key=lambda j: abs(j-i)):
            available = local & (~masks[j, y0:y1, x0:x1]) & (best < 0)
            best[available] = j
            if np.all((~local) | (best >= 0)):
                break
        oy, ox = np.where(local)
        jj = best[oy, ox]
        valid = jj >= 0
        out[i, y0+oy[valid], x0+ox[valid]] = frames[jj[valid], y0+oy[valid], x0+ox[valid]]
        # Tiny anti-aliased boundary repair, confined to uncertain edge pixels.
        ring = cv2.dilate(local.astype(np.uint8), np.ones((3,3),np.uint8), 1).astype(bool) & ~local
        # no replacement on ring: its source alpha is already near zero
    return out


def trajectory():
    # Pixel-space projectile parameters fitted to factual free-flight centers.
    g = 0.7353                 # px / frame"'^2
    y, vy = 21.599, 0.5574
    e_new = 0.80 * 1.20        # requested 1.2x restitution
    dt = 1/80
    pos = np.zeros(N, np.float64)
    vel = np.zeros(N, np.float64)
    contacts = []
    sample = 0
    t = 0.0
    while sample < N:
        if t + 1e-10 >= sample:
            pos[sample], vel[sample] = y, vy
            sample += 1
            if sample == N: break
        floor_next = 82.46 + .045 * (t + dt)
        yn = y + vy*dt + .5*g*dt*dt
        vn = vy + g*dt
        if yn >= floor_next and vn > .045:
            # Resolve impact at this short step; sub-frame error is < .01 px.
            yn = floor_next
            vn = .045 - e_new * (vn - .045)
            contacts.append(t + dt)
        y, vy, t = yn, vn, t + dt
    # Horizontal path has a slight perspective deceleration in the source.
    ii = np.arange(N, dtype=np.float64)
    x = 127.51 + .3634*ii - .0011053*ii*ii
    return x, pos, vel, contacts


def extract_ball(frames, sx, sy):
    """Extract source ball patches and soft alpha mattes for source-derived rendering."""
    patches, alphas = [], []
    yy, xx = np.mgrid[:29, :29]
    radial = np.sqrt((xx-14)**2 + (yy-14)**2)
    for i in range(N):
        cx, cy = sx[i], sy[i]
        M = np.array([[1,0,14-cx],[0,1,14-cy]], np.float32)
        p = cv2.warpAffine(frames[i], M, (29,29), flags=cv2.INTER_CUBIC,
                           borderMode=cv2.BORDER_REFLECT_101)
        # Slightly feathered circular silhouette. Radius 11.55 matches source.
        a = np.clip((12.25-radial)/1.15, 0, 1).astype(np.float32)
        patches.append(p); alphas.append(a)
    return np.asarray(patches), np.asarray(alphas)


def paste_rgba(dst, rgb, alpha, cx, cy):
    h, w = alpha.shape
    x0 = int(math.floor(cx - (w-1)/2)); y0 = int(math.floor(cy - (h-1)/2))
    x1, y1 = x0+w, y0+h
    dx0, dy0 = max(0,x0), max(0,y0); dx1, dy1 = min(W,x1), min(H,y1)
    if dx0 >= dx1 or dy0 >= dy1: return
    px0, py0 = dx0-x0, dy0-y0; px1, py1 = px0+dx1-dx0, py0+dy1-dy0
    a = alpha[py0:py1,px0:px1,None]
    reg = dst[dy0:dy1,dx0:dx1].astype(np.float32)
    dst[dy0:dy1,dx0:dx1] = np.clip(reg*(1-a) + rgb[py0:py1,px0:px1]*a,0,255).astype(np.uint8)


def render():
    src = np.load(CASE_DIR/'"'source_pixels.npz')['rgb']
    assert src.shape == (N,H,W,3) and src.dtype == np.uint8
    sx, sy = circle_centers(src)
    # Hough misses a few compressed contact frames; interpolate just those.
    good = np.isfinite(sx)
    sx = np.interp(np.arange(N), np.flatnonzero(good), sx[good])
    sy = np.interp(np.arange(N), np.flatnonzero(good), sy[good])
    plate = clean_plate(src, np.c_[sx,sy])
    ball, alpha = extract_ball(src, sx, sy)
    x, y, vy, contacts = trajectory()

    out = src.copy()  # exact factual history through frame 12's first contact
    for i in range(13, N):
        frame = plate[i].copy()
        floor = 82.46 + .045*i
        clearance = max(0.0, floor-y[i])

        # Physically coupled floor shadow: compact/dark at contact, broader and
        # lighter aloft. A warm-neutral multiply preserves the wooden texture.
        sigma_x = 5.4 + min(clearance,55)*.055
        sigma_y = 1.35 + min(clearance,55)*.018
        strength = .25*math.exp(-clearance/22.0) + .018
        yy, xx = np.mgrid[:H,:W]
        sh = np.exp(-.5*(((xx-(x[i]+1.5))/sigma_x)**2 +
                         ((yy-(floor+4.0))/sigma_y)**2)) * strength
        frame = np.clip(frame.astype(np.float32)*(1-sh[...,None]),0,255).astype(np.uint8)

        # The source already supplies natural rotation. Pick a source patch by a
        # continued angular phase, avoiding squashy contact frames.
        tex_idx = int((i*1.35) % 29)  # frames 0..28 are clean, round appearances
        paste_rgba(frame, ball[tex_idx], alpha[tex_idx], x[i], y[i])
        out[i] = frame
    return out, contacts


if __name__ == '__main__':
    frames, contacts = render()
    save(frames, CASE_DIR)
    print('saved', frames.shape, frames.dtype, 'altered impact frames:',
          ', '.join(f'{t:.2f}' for t in contacts))
