    x = 126.0 - 17.0*ease
    rad = math.radians(angle)
    half_extent_y = abs(math.sin(rad))*18.0 + abs(math.cos(rad))*10.5
    bounce = 3.2*math.sin(v*3.0*math.pi)*(1.0-v)
    y = 117.5 - half_extent_y - max(0.0, bounce)
    return x, y, angle, 1.0, True

    src = np.load(CASE_DIR/'source_pixels.npz')['rgb']
    assert src.shape == (N, H, W, 3) and src.dtype == np.uint8

    for i in range(EXECUTION_INDEX, N):
        frame = plate.copy()
        cx, cy, angle, airborne, floor_contact = car_pose(i)
        if i > 45:
            frame = add_shadow(frame, sprite, cx, cy, angle, airborne, floor_contact)
        frame = composite_rgba(frame, sprite, cx, cy, angle)
        out[i] = frame
    assert np.array_equal(out[:EXECUTION_INDEX], src[:EXECUTION_INDEX])
    return out
