+"""Remove the red die from every frame of the supplied 4-second shot.
+
+The camera is locked.  A robust clean plate can therefore be recovered from
+the opening seven frames: during those frames both dice move far enough that
+the per-pixel median contains the uncovered set.  The red die is tracked from
+its salmon surfaces, converted to a solid silhouette, and replaced by this
+plate.  A small, soft mask also removes the die's contact/drop shadow.  Bright
+neutral pixels belonging to the nearby white die are explicitly protected.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+from scipy.ndimage import binary_fill_holes
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def clean_plate(src: np.ndarray) -> np.ndarray:
+    # Frames 0..6 provide multiple unobstructed samples of every point on the
+    # narrow dice path.  Seven samples make the median insensitive to either
+    # moving object while retaining the source's painterly, compressed detail.
+    return np.median(src[:7].astype(np.float32), axis=0).astype(np.uint8)
+
+
+def largest_component(mask: np.ndarray) -> np.ndarray:
+    count, labels, stats, _ = cv2.connectedComponentsWithStats(mask, 8)
+    out = np.zeros_like(mask)
+    candidates = []
+    for label in range(1, count):
+        x, y, w, h, area = stats[label]
+        # The red die is the only sizeable salmon component in this tight ROI.
+        if area >= 35 and 112 <= x <= 125 and y < 100:
+            candidates.append((area, label))
+    if candidates:
+        out[labels == max(candidates)[1]] = 1
+    return out
+
+
+def red_die_mask(frame: np.ndarray) -> tuple[np.ndarray, tuple[int, int, int, int]]:
+    rgb = frame.astype(np.int16)
+    r, g, b = rgb[..., 0], rgb[..., 1], rgb[..., 2]
+    yy, xx = np.indices(r.shape)
+    salmon = (
+        (r > 112)
+        & (r > 1.16 * g)
+        & (r > 1.10 * b)
+        & (xx >= 112)
+        & (xx < 136)
+        & (yy < 102)
+    ).astype(np.uint8)
+    salmon = cv2.morphologyEx(salmon, cv2.MORPH_CLOSE, np.ones((3, 3), np.uint8))
+    core = largest_component(salmon)
+    ys, xs = np.nonzero(core)
+    if len(xs) == 0:
+        raise RuntimeError("red die tracker lost the object")
+
+    # A convex silhouette includes black pips and shaded faces that do not pass
+    # the chroma threshold.  A one-pixel expansion captures antialiased edges.
+    points = np.column_stack((xs, ys)).astype(np.int32)
+    hull = cv2.convexHull(points)
+    solid = np.zeros_like(core)
+    cv2.fillConvexPoly(solid, hull, 255)
+    solid = binary_fill_holes(solid > 0).astype(np.uint8)
+    solid = cv2.dilate(solid, np.ones((3, 3), np.uint8), iterations=1)
+    return solid, (int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max()))
+
+
+def white_die_protection(frame: np.ndarray, plate: np.ndarray) -> np.ndarray:
+    """Select neutral foreground detail so masking cannot nick the white die."""
+    f = frame.astype(np.int16)
+    p = plate.astype(np.int16)
+    mx, mn = f.max(axis=2), f.min(axis=2)
+    delta = np.abs(f - p).max(axis=2)
+    yy, xx = np.indices(mx.shape)
+    neutral_fg = (
+        (delta > 11)
+        & (mx > 105)
+        & ((mx - mn) < 65)
+        & (xx >= 114)
+        & (xx < 153)
+        & (yy < 103)
+    ).astype(np.uint8)
+    neutral_fg = cv2.morphologyEx(neutral_fg, cv2.MORPH_CLOSE, np.ones((3, 3), np.uint8))
+    neutral_fg = cv2.dilate(neutral_fg, np.ones((3, 3), np.uint8), iterations=1)
+    return neutral_fg.astype(bool)
+
+
+def soft_shadow_alpha(bounds: tuple[int, int, int, int], shape: tuple[int, int]) -> np.ndarray:
+    """Approximate the small projected/contact shadow cast by the red die."""
+    x0, y0, x1, y1 = bounds
+    h, w = shape
+    shadow = np.zeros((h, w), np.uint8)
+
+    # The visible shadow falls below the die.  It is subtle while airborne and
+    # stronger once the die is close to the circular table (roughly y >= 58).
+    if y1 >= 56:
+        cx = (x0 + x1) // 2
+        cy = min(h - 1, y1 + 4)
+        rx = max(7, (x1 - x0 + 1) // 2 + 4)
+        ry = 4 if y1 < 77 else 5
+        cv2.ellipse(shadow, (cx, cy), (rx, ry), 0, 0, 360, 255, -1)
+        shadow = cv2.GaussianBlur(shadow, (7, 7), 1.5)
+        strength = 0.72 if y1 < 77 else 0.88
+        return shadow.astype(np.float32) / 255.0 * strength
+    return shadow.astype(np.float32)
+
+
+def generate() -> np.ndarray:
+    src = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert src.shape == (96, 144, 256, 3) and src.dtype == np.uint8
+    plate = clean_plate(src)
+    result = src.copy()
+
+    for i, frame in enumerate(src):
+        solid, bounds = red_die_mask(frame)
+
+        # Full-opacity interior with a two-pixel antialiasing fringe.
+        hard = (solid * 255).astype(np.uint8)
+        fringe = cv2.GaussianBlur(hard, (5, 5), 0.9).astype(np.float32) / 255.0
+        alpha = np.maximum(solid.astype(np.float32), fringe)
+        alpha = np.maximum(alpha, soft_shadow_alpha(bounds, solid.shape))
+
+        # When the dice overlap, retain all identified white-die pixels.
+        protect = white_die_protection(frame, plate)
+        alpha[protect] = 0.0
+
+        a = alpha[..., None]
+        composed = frame.astype(np.float32) * (1.0 - a) + plate.astype(np.float32) * a
+        result[i] = np.clip(np.rint(composed), 0, 255).astype(np.uint8)
+
+    return result
+
+
+if __name__ == "__main__":
+    frames = generate()
+    artifact_io.save(frames, CASE_DIR)
+    print(f"saved {len(frames)} frames to {CASE_DIR}")
