+from pathlib import Path
+
+import numpy as np
+from scipy.ndimage import map_coordinates
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def smoothstep(edge0, edge1, value):
+    value = np.clip((value - edge0) / (edge1 - edge0), 0.0, 1.0)
+    return value * value * (3.0 - 2.0 * value)
+
+
+def make_sprite(source, background, frame_index, anchor, radii):
+    """Recover a small antialiased RGBA sprite from a clean source location."""
+    cx, cy = anchor
+    half_w, half_h = 11, 10
+    x0, y0 = int(np.floor(cx)) - half_w, int(np.floor(cy)) - half_h
+    x1, y1 = x0 + 2 * half_w + 1, y0 + 2 * half_h + 1
+
+    observed = source[frame_index, y0:y1, x0:x1].astype(np.float32)
+    under = background[y0:y1, x0:x1].astype(np.float32)
+    difference = np.max(np.abs(observed - under), axis=2)
+
+    yy, xx = np.mgrid[y0:y1, x0:x1]
+    support = ((xx - cx) / radii[0]) ** 2 + ((yy - cy) / radii[1]) ** 2
+    # The solid stone differs strongly from the floor.  The shallow threshold
+    # retains its soft contact shadow and the smoothstep keeps the matte clean.
+    alpha = smoothstep(1.5, 14.0, difference)
+    alpha *= smoothstep(1.08, 0.88, support)
+    alpha[alpha < 0.025] = 0.0
+
+    # Unmix edge pixels from their original floor so they can be placed over
+    # the differently colored rings without carrying a pale fringe.
+    safe_alpha = np.maximum(alpha[..., None], 0.045)
+    foreground = (observed - (1.0 - alpha[..., None]) * under) / safe_alpha
+    foreground = np.clip(foreground, 0.0, 255.0)
+    foreground[alpha == 0] = 0
+    return {
+        "rgb": foreground,
+        "alpha": alpha.astype(np.float32),
+        "origin": (x0, y0),
+        "anchor": (cx, cy),
+    }
+
+
+def draw_sprite(canvas, sprite, position):
+    """Bilinearly place a sprite at a floating-point image position."""
+    rgb, alpha = sprite["rgb"], sprite["alpha"]
+    sh, sw = alpha.shape
+    local_anchor_x = sprite["anchor"][0] - sprite["origin"][0]
+    local_anchor_y = sprite["anchor"][1] - sprite["origin"][1]
+    px, py = position
+
+    left = int(np.floor(px - local_anchor_x)) - 1
+    top = int(np.floor(py - local_anchor_y)) - 1
+    right, bottom = left + sw + 2, top + sh + 2
+    x0, y0 = max(0, left), max(0, top)
+    x1, y1 = min(canvas.shape[1], right), min(canvas.shape[0], bottom)
+    if x1 <= x0 or y1 <= y0:
+        return
+
+    yy, xx = np.mgrid[y0:y1, x0:x1]
+    sx = xx - px + local_anchor_x
+    sy = yy - py + local_anchor_y
+    coordinates = np.stack([sy, sx])
+    sampled_alpha = map_coordinates(alpha, coordinates, order=1, mode="constant", cval=0.0)
+    sampled_rgb = np.stack(
+        [map_coordinates(rgb[..., channel], coordinates, order=1, mode="constant", cval=0.0)
+         for channel in range(3)],
+        axis=2,
+    )
+    a = sampled_alpha[..., None]
+    region = canvas[y0:y1, x0:x1].astype(np.float32)
+    canvas[y0:y1, x0:x1] = np.clip(sampled_rgb * a + region * (1.0 - a), 0, 255).astype(np.uint8)
+
+
+def first_contact(red_start, red_velocity, yellow_start, yellow_velocity, distance):
+    offset = yellow_start - red_start
+    relative_velocity = yellow_velocity - red_velocity
+    roots = np.roots([
+        np.dot(relative_velocity, relative_velocity),
+        2.0 * np.dot(offset, relative_velocity),
+        np.dot(offset, offset) - distance * distance,
+    ])
+    positive = [float(root.real) for root in roots if abs(root.imag) < 1e-7 and root.real >= 0]
+    return min(positive)
+
+
+def generate():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+
+    # During the first 35 frames each moving stone visits any given pixel only
+    # briefly, so a temporal median produces the original empty, detailed rink.
+    background = np.median(source[:35].astype(np.float32), axis=0).round().astype(np.uint8)
+
+    # Least-squares trajectories measured from the isolated stones in frames
+    # 1--37 (coordinates are at the source's 256x144 delivery resolution).
+    red_start = np.array([47.2927, 67.1146], dtype=np.float64)
+    red_original_velocity = np.array([1.97171, 0.14981], dtype=np.float64)
+    yellow_start = np.array([212.3657, 78.4709], dtype=np.float64)
+    yellow_velocity = np.array([-2.12613, -0.14099], dtype=np.float64)
+
+    # The requested intervention applies from source frame 1 (time zero).
+    red_velocity = 0.5 * red_original_velocity
+    contact_distance = 13.5
+    contact_time = first_contact(
+        red_start, red_velocity, yellow_start, yellow_velocity, contact_distance
+    )
+    red_contact = red_start + red_velocity * contact_time
+    yellow_contact = yellow_start + yellow_velocity * contact_time
+
+    # Equal rigid stones, with restitution inferred from the slow separation in
+    # the factual collision.  Tangential motion is retained.
+    normal = (yellow_contact - red_contact) / contact_distance
+    restitution = 0.075
+    closing_speed = np.dot(red_velocity - yellow_velocity, normal)
+    impulse = 0.5 * (1.0 + restitution) * closing_speed * normal
+    red_after = red_velocity - impulse
+    yellow_after = yellow_velocity + impulse
+
+    # Frame-one source sprites contain the original shading, bevels, handles,
+    # antialiasing, and contact shadows on a clean portion of the sheet.
+    red_sprite = make_sprite(source, background, 0, red_start, (8.3, 6.3))
+    yellow_sprite = make_sprite(source, background, 0, yellow_start, (8.7, 7.4))
+
+    frames = np.repeat(background[None, ...], 96, axis=0)
+    for frame_index in range(96):
+        t = float(frame_index)
+        if t <= contact_time:
+            red_position = red_start + red_velocity * t
+            yellow_position = yellow_start + yellow_velocity * t
+        else:
+            elapsed = t - contact_time
+            red_position = red_contact + red_after * elapsed
+            yellow_position = yellow_contact + yellow_after * elapsed
+
+        # Painter's order follows the camera depth (larger image y is nearer).
+        objects = [
+            (red_position[1], red_sprite, red_position),
+            (yellow_position[1], yellow_sprite, yellow_position),
+        ]
+        for _, sprite, position in sorted(objects, key=lambda item: item[0]):
+            draw_sprite(frames[frame_index], sprite, position)
+
+    # At t=0 the intervention changes velocity, not pose or appearance.  Keep
+    # the exact factual frame to avoid even a one-pixel transition at the cut.
+    frames[0] = source[0]
+    return frames
+
+
+if __name__ == "__main__":
+    result = generate()
+    assert result.shape == (96, 144, 256, 3)
+    assert result.dtype == np.uint8
+    artifact_io.save(result, CASE_DIR)
