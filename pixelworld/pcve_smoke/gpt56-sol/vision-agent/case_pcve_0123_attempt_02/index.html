+"""Generate the mass-edited marble video for this case.
+
+The two marbles have the same radius, but from frame 1 the purple marble has
+mass 0.3 while the yellow marble retains mass 1.  Their motion before contact
+is unchanged by mass.  At the first contact, the measured incoming speeds and
+the restitution visible in the source are used in the unequal-mass collision
+equations.  The light marble is consequently thrown back up and off the open
+right end of the trough; the yellow marble continues with a much lower speed.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+N, H, W = 96, 144, 256
+
+
+def surface_y(x):
+    """Measured centerline of a radius-6 marble resting in the white trough."""
+    return 72.15 - 0.00135 * (x - 128.0) ** 2
+
+
+def extract_delta(source, background, frame, cx, cy, radius=13):
+    """Extract a source-rendered marble as a feathered additive sprite."""
+    cx, cy = int(round(cx)), int(round(cy))
+    y0, y1 = cy - radius, cy + radius + 1
+    x0, x1 = cx - radius, cx + radius + 1
+    delta = source[frame, y0:y1, x0:x1].astype(np.float32)
+    delta -= background[y0:y1, x0:x1].astype(np.float32)
+
+    yy, xx = np.mgrid[-radius:radius + 1, -radius:radius + 1]
+    # Include the small contact shadow below the sphere, while softly removing
+    # unrelated compression variation at the outside of the crop.
+    sphere = np.clip((10.7 - np.hypot(xx, yy)) / 2.0, 0.0, 1.0)
+    shadow = np.clip((1.0 - (xx / 10.5) ** 2 - ((yy - 6.0) / 5.5) ** 2) * 2.2,
+                     0.0, 1.0)
+    activity = np.max(np.abs(delta), axis=2)
+    evidence = np.clip((activity - 0.7) / 5.0, 0.0, 1.0)
+    alpha = np.maximum(sphere, shadow) * evidence
+    return delta, alpha[..., None]
+
+
+def stamp_delta(image, delta, alpha, cx, cy, shadow_strength=1.0):
+    """Composite a sprite centered at a subpixel location using translation."""
+    radius = delta.shape[0] // 2
+    # Subpixel translation prevents visible one-pixel stepping at low speeds.
+    ix, iy = int(np.floor(cx)), int(np.floor(cy))
+    fx, fy = float(cx - ix), float(cy - iy)
+    mat = np.float32([[1, 0, fx], [0, 1, fy]])
+    d = cv2.warpAffine(delta, mat, (delta.shape[1], delta.shape[0]),
+                       flags=cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)
+    a = cv2.warpAffine(alpha, mat, (alpha.shape[1], alpha.shape[0]),
+                       flags=cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)
+    if a.ndim == 2:
+        a = a[..., None]
+
+    # In flight there is no contact shadow.  The lower part is selectively
+    # faded, preserving the sphere itself and its antialiased boundary.
+    if shadow_strength < 0.999:
+        yy = np.arange(-radius, radius + 1)[:, None]
+        lower = np.clip((yy - 3.0) / 4.0, 0.0, 1.0)[..., None]
+        a = a * (1.0 - lower * (1.0 - shadow_strength))
+
+    x0, y0 = ix - radius, iy - radius
+    x1, y1 = x0 + d.shape[1], y0 + d.shape[0]
+    sx0, sy0 = max(0, -x0), max(0, -y0)
+    sx1, sy1 = d.shape[1] - max(0, x1 - W), d.shape[0] - max(0, y1 - H)
+    x0, y0, x1, y1 = max(0, x0), max(0, y0), min(W, x1), min(H, y1)
+    if x0 >= x1 or y0 >= y1:
+        return
+    region = image[y0:y1, x0:x1]
+    region += d[sy0:sy1, sx0:sx1] * a[sy0:sy1, sx0:sx1]
+
+
+def source_centers(source):
+    """Locate the factual marbles so their old renderings can be removed."""
+    yellow = np.zeros((N, 2), np.float32)
+    purple = np.zeros((N, 2), np.float32)
+    # Tracking is only needed from the edited collision frame onward.  Color
+    # centroids are robust for purple; yellow uses a continuity-constrained
+    # local color centroid to avoid the similarly colored wooden support.
+    py, pp = np.array([120.0, 72.0]), np.array([135.5, 71.0])
+    for i in range(34, N):
+        hsv = cv2.cvtColor(source[i], cv2.COLOR_RGB2HSV)
+        pm = cv2.inRange(hsv, np.array([118, 70, 55]),
+                        np.array([170, 255, 255]))
+        pm[:48] = 0
+        pm[82:] = 0
+        ys, xs = np.where(pm > 0)
+        near = (xs > pp[0] - 18) & (xs < pp[0] + 18)
+        if near.any():
+            pp = np.array([xs[near].mean(), ys[near].mean()])
+
+        ym = cv2.inRange(hsv, np.array([8, 85, 70]),
+                        np.array([34, 255, 255]))
+        ym[:48] = 0
+        ym[82:] = 0
+        ys, xs = np.where(ym > 0)
+        near = ((xs > py[0] - 18) & (xs < py[0] + 18) &
+                (ys > py[1] - 12) & (ys < py[1] + 12))
+        if near.any():
+            py = np.array([xs[near].mean(), ys[near].mean()])
+        yellow[i], purple[i] = py, pp
+    return yellow, purple
+
+
+def counterfactual_paths():
+    """Simulate both centers after the first collision (frame 35 onward)."""
+    ypath = np.full((N, 2), np.nan, np.float32)
+    ppath = np.full((N, 2), np.nan, np.float32)
+
+    # Contact geometry measured at frames 34-35. Incoming tangential speeds
+    # are about +6 and -6 px/frame. Restitution 0.67 reproduces the factual
+    # equal-mass rebound. Apply the 1-D rigid-body impulse with m_purple=.3.
+    my, mp, restitution = 1.0, 0.3, 0.67
+    uy, up = 6.0, -6.0
+    vy = ((my - restitution * mp) * uy +
+          (1.0 + restitution) * mp * up) / (my + mp)
+    vp = ((mp - restitution * my) * up +
+          (1.0 + restitution) * my * uy) / (my + mp)
+
+    xy, xp = 120.1, 135.6
+    launched = False
+    yp = surface_y(xp)
+    vyp = 0.0
+    for i in range(34, N):
+        ypath[i] = (xy, surface_y(xy))
+        ppath[i] = (xp, yp if launched else surface_y(xp))
+
+        # Eight substeps give a smooth turn around the trough bottom.
+        for _ in range(8):
+            dt = 1.0 / 8.0
+            # Constant tangential gravity closely fits the measured source
+            # acceleration on each nearly planar half of the trough.
+            if abs(xy - 128.0) < 3.5:
+                ay = -0.060 * (xy - 128.0) - 0.012 * vy
+            else:
+                ay = -0.235 * np.sign(xy - 128.0) - 0.012 * vy
+            vy += ay * dt
+            xy += vy * dt
+
+            if not launched:
+                if abs(xp - 128.0) < 3.5:
+                    ap = -0.060 * (xp - 128.0) - 0.012 * vp
+                else:
+                    ap = -0.235 * np.sign(xp - 128.0) - 0.012 * vp
+                vp += ap * dt
+                xp += vp * dt
+                if xp >= 237.0:
+                    # Leave tangent to the rising right end. Image y points
+                    # down, so the rightward launch has an upward component.
+                    xp = 237.0
+                    yp = surface_y(xp)
+                    vyp = -0.00270 * (xp - 128.0) * vp
+                    launched = True
+            else:
+                xp += vp * dt
+                yp += vyp * dt
+                vyp += 0.48 * dt
+                vp *= (1.0 - 0.002 * dt)
+
+    return ypath, ppath
+
+
+def main():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (N, H, W, 3) and source.dtype == np.uint8
+
+    # Median over moving-object frames yields an exceptionally clean plate;
+    # all static scene texture, the plant, and the full white trough survive.
+    background = np.median(source[:73], axis=0).astype(np.uint8)
+    old_y, old_p = source_centers(source)
+    ypath, ppath = counterfactual_paths()
+
+    # Source-derived, anti-aliased appearances from clear mid-ramp frames.
+    yd, ya = extract_delta(source, background, 24, 69, 65)
+    pd, pa = extract_delta(source, background, 24, 186, 65)
+
+    output = source.copy()
+    for i in range(34, N):
+        # Keep source temporal texture. Replace only the factual marbles and
+        # their contact shadows with the static clean plate, using a feather.
+        frame = source[i].astype(np.float32)
+        mask = np.zeros((H, W), np.float32)
+        for center in (old_y[i], old_p[i]):
+            cx, cy = int(round(center[0])), int(round(center[1]))
+            cv2.ellipse(mask, (cx, cy + 2), (12, 11), 0, 0, 360, 1.0, -1)
+        mask = cv2.GaussianBlur(mask, (5, 5), 1.0)[..., None]
+        frame = frame * (1.0 - mask) + background.astype(np.float32) * mask
+
+        stamp_delta(frame, yd, ya, float(ypath[i, 0]), float(ypath[i, 1]))
+        px, py = map(float, ppath[i])
+        if px < W + 14:
+            airborne = max(0.0, min(1.0, (px - 237.0) / 8.0))
+            stamp_delta(frame, pd, pa, px, py, shadow_strength=1.0-airborne)
+        output[i] = np.clip(np.rint(frame), 0, 255).astype(np.uint8)
+
+    # Required deliverables: pixels.npz, frames, inspection images, and player.
+    artifact_io.save(output, CASE_DIR)
+
+
+if __name__ == "__main__":
+    main()
