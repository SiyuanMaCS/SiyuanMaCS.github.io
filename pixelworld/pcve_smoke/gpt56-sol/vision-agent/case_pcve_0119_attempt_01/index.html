+def soft_disk(cx, cy, radius, feather=1.5):
+    yy, xx = np.mgrid[:H, :W]
+    distance = np.hypot(xx - cx, yy - cy)
+    return np.clip((radius + feather - distance) / feather, 0.0, 1.0)
+
+
+def soft_ellipse(cx, cy, rx, ry, feather=1.2):
+    yy, xx = np.mgrid[:H, :W]
+    distance = np.sqrt(((xx - cx) / rx) ** 2 + ((yy - cy) / ry) ** 2)
+    return np.clip((1.0 + feather / max(rx, ry) - distance) * max(rx, ry) / feather, 0.0, 1.0)
+
+
+def over(frame, rgb, alpha):
+    return frame * (1.0 - alpha[..., None]) + rgb * alpha[..., None]
+
+
+def paste_patch(frame, patch, alpha, center_x, center_y):
+    ph, pw = patch.shape[:2]
+    left = int(round(center_x - (pw - 1) / 2))
+    top = int(round(center_y - (ph - 1) / 2))
+    x0, y0 = max(0, left), max(0, top)
+    x1, y1 = min(W, left + pw), min(H, top + ph)
+    if x0 >= x1 or y0 >= y1:
+        return
+    sx0, sy0 = x0 - left, y0 - top
+    sx1, sy1 = sx0 + x1 - x0, sy0 + y1 - y0
+    a = alpha[sy0:sy1, sx0:sx1, None]
+    frame[y0:y1, x0:x1] = frame[y0:y1, x0:x1] * (1.0 - a) + patch[sy0:sy1, sx0:sx1] * a
+

+def clean_plate(source):

+    # ball location to recover the wall and the shelf edge.

+        q = (x - 136) / 13.0
+        plate[:, x] = (1 - q) * source[95, :, 136] + q * source[0, :, 149]
+    return plate

+def simulate_ball():
+    """Counterfactual trajectory with 1.5x the inferred restitution.
+
+    The car starts transferring momentum at frame 20.  The ball rolls off the
+    shelf, then uses e=0.51 (1.5 * the source estimate 0.34) at floor impacts.
+    Tangential damping occurs only at an impact; ordinary rolling friction is
+    applied after the last small bounce.
+    """
+    x = np.full(N, 137.0)
+    y = np.full(N, 28.0)
+    fall_frame = 30
+    # Accelerating shelf roll caused by the car/ball impact.
+    for i in range(20, fall_frame + 1):
+        t = i - 19.0
+        x[i] = 137.0 - 0.45 * t - 0.075 * t * t
+
+    px, py = x[fall_frame], 28.0
+    vx = -(0.45 + 0.15 * (fall_frame - 19.0))
+    vy = 0.45
+    gravity = 1.55
+    restitution = 0.34 * 1.5
+    ground = 117.0
+    grounded = False
+    dt = 1.0 / 16.0
+    for i in range(fall_frame + 1, N):
+        for _ in range(16):
+            if grounded:
+                # Coulomb-like rolling resistance, expressed per frame.
+                drag = 0.040 * dt
+                vx = np.sign(vx) * max(0.0, abs(vx) - drag)
+                px += vx * dt
+                py = ground
+                continue
+            vy += gravity * dt
+            px += vx * dt
+            py += vy * dt
+            if py >= ground and vy > 0:
+                py = ground
+                incoming = vy
+                vx *= 0.82
+                rebound = incoming * restitution
+                if rebound < 1.35:
+                    vy = 0.0
+                    grounded = True
+                else:
+                    vy = -rebound
+        x[i], y[i] = px, py
+    return x, y
+

+def make_sprites(source, plate):
+    # Car alpha is estimated against the temporal median, where its starting
+    # location is unobstructed for most of the clip.
+    median = np.median(source.astype(np.float32), axis=0)
+    car_box = (181, 21, 224, 41)
+    x0, y0, x1, y1 = car_box
+    car = source[0, y0:y1, x0:x1].astype(np.float32)
+    diff = np.max(np.abs(car - median[y0:y1, x0:x1]), axis=2)
+    car_alpha = np.clip((diff - 2.5) / 15.0, 0, 1)
+    car_alpha = ndimage.gaussian_filter(ndimage.maximum_filter(car_alpha, 3), 0.55)
+    car_alpha = np.clip(car_alpha, 0, 1)
+
+    # Fully visible, differently rotated ball appearances from the free-fall
+    # section.  A geometric antialiased matte avoids carrying wall pixels.
+    bx, by = source_ball_centers()
+    balls = []
+    size = 19
+    yy, xx = np.mgrid[:size, :size]
+    matte = np.clip((8.35 - np.hypot(xx - 9, yy - 9)) / 1.25, 0, 1)
+    for index in range(36, 44):
+        cx, cy = int(round(bx[index])), int(round(by[index]))
+        crop = source[index, cy - 9:cy + 10, cx - 9:cx + 10].astype(np.float32)
+        balls.append((crop, matte))
+    return (car, car_alpha), balls
+

+def render():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (N, H, W, 3) and source.dtype == np.uint8
+    plate = clean_plate(source)
+    (car_sprite, car_alpha), ball_sprites = make_sprites(source, plate)
+    old_x, old_y = source_ball_centers()
+    ball_x, ball_y = simulate_ball()
+    frames = source.astype(np.float32).copy()
+
+    # Preserve the factual history exactly through frame 20 (source frame 20,
+    # one-based).  Restitution cannot have a visible effect before contact.
+    for i in range(20, N):
+        frame = frames[i]
+
+        # Remove factual ball and its factual cast/contact shadow.
+        mask = soft_disk(old_x[i], old_y[i], 9.5, 1.6)
+        if old_y[i] < 105:
+            mask = np.maximum(mask, soft_ellipse(old_x[i] - 6, old_y[i] + 10, 12, 15, 2.0))
+        else:
+            mask = np.maximum(mask, soft_ellipse(old_x[i] + 3, 120, 13, 5, 1.8))
+        frame[:] = over(frame, plate, mask)
+
+        # Remove the factual car after impact.  This region is static wall and
+        # shelf in the clean plate, so a soft rectangular matte is seamless.
+        car_remove = np.zeros((H, W), np.float32)
+        car_remove[18:43, 136:177] = 1.0
+        car_remove = ndimage.gaussian_filter(car_remove, 1.0)
+        frame[:] = over(frame, plate, np.clip(car_remove, 0, 1))
+
+        # The heavier car is affected less: it brakes sooner and comes to rest
+        # about four pixels to the right of its factual resting point.
+        if i == 20:
+            car_cx = 160.0
+        elif i == 21:
+            car_cx = 159.4
+        elif i == 22:
+            car_cx = 159.0
+        elif i == 23:
+            car_cx = 158.8
+        else:
+            car_cx = 158.8
+        paste_patch(frame, car_sprite, car_alpha, car_cx, 30.5)
+
+        # Counterfactual shadow: wall shadow while airborne, compact floor
+        # shadow near contact.  Draw on an alpha layer and blur it optically.
+        shadow = Image.new("L", (W, H), 0)
+        from PIL import ImageDraw
+        draw = ImageDraw.Draw(shadow)
+        cx, cy = ball_x[i], ball_y[i]
+        if cy < 103:
+            draw.ellipse((cx - 17, cy - 3, cx + 5, cy + 24), fill=42)
+            shadow = shadow.filter(ImageFilter.GaussianBlur(3.0))
+        else:
+            proximity = np.clip((cy - 98) / 19.0, 0, 1)
+            draw.ellipse((cx - 8, 119, cx + 13, 125), fill=int(62 * proximity))
+            shadow = shadow.filter(ImageFilter.GaussianBlur(2.0))
+        sa = np.asarray(shadow, dtype=np.float32) / 255.0
+        shadow_color = np.array([112, 82, 59], np.float32)
+        frame[:] = over(frame, np.broadcast_to(shadow_color, frame.shape), sa)
+
+        # Rotation advances with horizontal travel.  These source-derived
+        # appearances preserve the exact star, cyan band, shading, and scale.
+        phase = int(round((137.0 - cx) / 1.9)) % len(ball_sprites)
+        ball_sprite, ball_alpha = ball_sprites[phase]
+        paste_patch(frame, ball_sprite, ball_alpha, cx, cy)
+
+    return np.clip(np.rint(frames), 0, 255).astype(np.uint8)
+
