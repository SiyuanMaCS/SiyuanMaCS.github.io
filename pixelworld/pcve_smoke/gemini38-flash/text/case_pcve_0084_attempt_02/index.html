+def main():

+    src_data = np.load(case_dir / "source_pixels.npz")
+    frames = src_data["rgb"].copy()  # shape (96, 144, 256, 3), uint8
+    n_frames, h, w, c = frames.shape
+
+    # Frame 18 (1-based) corresponds to index 17 (0-based)
+    exec_idx = 17
+
+    # Construct clean background for the table:
+    # Early frames (0..8) have clean table where apple stops in late frames (x < 100)
+    # Late frames (75..95) have clean table where apple started in early frames (x > 105)
+    early_clean = np.median(frames[0:8], axis=0).astype(np.float32)
+    late_clean = np.median(frames[75:95], axis=0).astype(np.float32)
+
+    weight_early = np.zeros((h, w, 1), dtype=np.float32)
+    for x in range(w):
+        if x <= 98:
+            weight_early[:, x] = 1.0
+        elif x >= 108:
+            weight_early[:, x] = 0.0
+        else:
+            weight_early[:, x] = (108 - x) / 10.0
+
+    clean_bg = late_clean * (1.0 - weight_early) + early_clean * weight_early
+    clean_bg = np.clip(np.round(clean_bg), 0, 255).astype(np.uint8)
+
+    # Process from frame 18 (index 17) onwards
+    for t in range(exec_idx, n_frames):
+        cur_frame = frames[t]
+        diff = np.abs(cur_frame.astype(float) - clean_bg.astype(float)).sum(axis=-1)
+
+        # Segment the apple: located on the table (y in [65, 102], x < 145)
+        core_mask = (diff > 12) & (np.arange(w)[None, :] < 145) & (np.arange(h)[:, None] >= 65) & (np.arange(h)[:, None] <= 102)
+
+        # Dilate mask to cover anti-aliasing and contact shadows
+        dilated = binary_dilation(core_mask, iterations=4)
+
+        # Feathered blending along the boundary for seamless inpainting
+        dist_out = distance_transform_edt(~dilated)
+        alpha = np.clip(1.0 - (dist_out - 1.0) / 2.0, 0.0, 1.0)
+        alpha = np.where(dilated, 1.0, alpha)[:, :, None]
+
+        inpainted = cur_frame.astype(np.float32) * (1.0 - alpha) + clean_bg.astype(np.float32) * alpha
+        frames[t] = np.clip(np.round(inpainted), 0, 255).astype(np.uint8)
+
+    artifact_io.save(frames, case_dir)
