+
+Prompt: From frame 1 onwards, increase the tennis ball's friction coefficient to 3 times its original value.
+
+Analysis:
+- The tennis ball begins rolling along the horizontal surface at frame 28 (0-indexed).
+- In the factual rollout, the ball decelerates due to rolling friction from frame 28 to frame 77,
+  stopping at x ≈ 208.0 after rolling 55.0 pixels over 49 frames.
+- Under 3x friction, the deceleration triples:
+  * Rolling duration is reduced by 3x: 49 / 3 ≈ 16.33 frames (stops at frame ~44).
+  * Rolling distance is reduced by 3x: 55.0 / 3 ≈ 18.33 pixels (stops at x ≈ 171.33).
+  * At each intermediate time tau = t - 28, the trajectory scales faithfully according to
+    x(tau) = x_start + (x_orig(3 * tau) - x_start) / 3.
+"""
+from pathlib import Path
+import numpy as np
+from scipy.ndimage import shift
+import artifact_io
+
+
+def main():
+    case_dir = Path(__file__).parent.resolve()
+    source_npz = np.load(case_dir / "source_pixels.npz")
+    src = source_npz["rgb"]  # shape (96, 144, 256, 3), uint8
+    n_frames, h, w, _ = src.shape
+
+    # Compute static background estimate across all frames
+    bg = np.median(src, axis=0).astype(np.uint8)
+
+    # Track ball positions across the source rollout
+    source_xs = []
+    for t in range(28, n_frames):
+        diff = np.max(np.abs(src[t].astype(float) - bg.astype(float)), axis=-1)
+        coords = np.argwhere(diff > 25)
+        source_xs.append(coords[:, 1].mean())
+    source_xs = np.array(source_xs)
+
+    # Identify stop time and rollout parameters in the original video
+    t_orig_stop = np.where(source_xs >= 207.99)[0][0]  # 49 frames after frame 28
+    x_start = source_xs[0]                              # 153.0
+    x_stop_orig = source_xs[t_orig_stop]               # 208.0
+
+    # 3x friction dynamics
+    tau_stop = t_orig_stop / 3.0                        # ~16.333 frames
+    x_stop_new = x_start + (x_stop_orig - x_start) / 3.0  # ~171.333 pixels
+
+    out_frames = np.zeros_like(src)
+    # Frames 0..28 are identical to source (intervention takes effect upon surface contact)
+    out_frames[:29] = src[:29]
+
+    for f in range(29, n_frames):
+        tau = f - 28
+        if tau < tau_stop:
+            t_orig = 3.0 * tau
+            x_interp = np.interp(t_orig, np.arange(len(source_xs)), source_xs)
+            x_target = x_start + (x_interp - x_start) / 3.0
+            # Match donor ball appearance and rotation to distance traveled
+            s_idx = np.argmin(np.abs(source_xs - x_target))
+            donor_f = 28 + s_idx
+        else:
+            x_target = x_stop_new
+            donor_f = 80  # Fully stationary ball
+
+        frame = src[f].copy()
+
+        # Clean out the original ball position in frame f using background
+        d_orig = np.max(np.abs(frame.astype(float) - bg.astype(float)), axis=-1)
+        orig_coords = np.argwhere(d_orig > 15)
+        if len(orig_coords) > 0:
+            ymin = max(0, orig_coords[:, 0].min() - 2)
+            ymax = min(h - 1, orig_coords[:, 0].max() + 2)
+            xmin = max(0, orig_coords[:, 1].min() - 2)
+            xmax = min(w - 1, orig_coords[:, 1].max() + 2)
+            frame[ymin:ymax+1, xmin:xmax+1] = bg[ymin:ymax+1, xmin:xmax+1]
+

+        donor_coords = np.argwhere(d_mag > 15)
