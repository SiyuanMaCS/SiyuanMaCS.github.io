    # unaffected blue mallet or the first collision.

            difference = first - frame
            occlusion = np.clip(np.maximum(difference[..., 1], difference[..., 2]) / 30., 0, 1)

                # damped rest observed at this same rail in the factual video.
                px = rim_x + 4.0*(1-np.exp(-(dt-impact_dt)/13.))

            col = cv2.warpAffine(premultiplied.astype(np.float32), matrix, (w, h), flags=cv2.INTER_LINEAR)
            frame = frame * (1-al[..., None]) + col
            trace.append([t, float(px), float(py), float(scale)])

    frames = np.stack(output)
    artifact_io.save(frames, ROOT)
    print('Saved', frames.shape, frames.dtype)
    print('Trajectory samples:', trace[::8])
