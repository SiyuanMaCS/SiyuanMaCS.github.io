+from pathlib import Path
+
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+FIRST_EDIT_FRAME = 17  # Zero-based index of source frame 18.
+
+
+def dilate(mask, iterations):
+    """8-connected binary dilation without relying on extra libraries."""
+    result = mask.copy()
+    for _ in range(iterations):
+        padded = np.pad(result, 1, mode="constant", constant_values=False)
+        expanded = result.copy()
+        for row_offset in (-1, 0, 1):
+            for col_offset in (-1, 0, 1):
+                if row_offset == 0 and col_offset == 0:
+                    continue
+                expanded |= padded[
+                    1 + row_offset : padded.shape[0] - 1 + row_offset,
+                    1 + col_offset : padded.shape[1] - 1 + col_offset,
+                ]
+        result = expanded
+    return result
+
+
+def main():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    frames = source.copy()
+
+    # After the apple leaves the frame, this region shows only the stable
+    # textured background. A temporal median removes incidental pixel noise.
+    clean_background = np.median(source[50:], axis=0).astype(np.uint8)
+
+    # The apple is the only changing object in the lower-right of the scene.
+    # A slightly larger outer region keeps the inpaint footprint away from the
+    # independent mechanism on the left.
+    core_roi = np.zeros(source.shape[1:3], dtype=bool)
+    core_roi[78:98, 150:256] = True
+    outer_roi = np.zeros_like(core_roi)
+    outer_roi[77:99, 147:256] = True
+
+    for frame_index in range(FIRST_EDIT_FRAME, source.shape[0]):
+        difference = np.abs(
+            source[frame_index].astype(np.int16) - clean_background.astype(np.int16)
+        ).sum(axis=2)
+        core = (difference > 7) & core_roi
+        if not np.any(core):
+            continue
+
+        # Cover the soft shadow and antialiased edge pixels as well.
+        inpaint_mask = dilate(core, 3) & outer_roi
+        frames[frame_index][inpaint_mask] = clean_background[inpaint_mask]
+
+    frames = frames.astype(np.uint8)
+    artifact_io.save(frames, CASE_DIR)
+
+
+if __name__ == "__main__":
+    main()
