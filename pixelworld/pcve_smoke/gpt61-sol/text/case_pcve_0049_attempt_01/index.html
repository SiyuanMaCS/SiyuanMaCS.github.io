+"""Mass counterfactual for the complete source timeline.
+
+Use the original approach, infer its low-restitution contact, and solve the
+same impulse with the yellow body's mass scaled by 0.2.  All appearance is
+sampled from the source; the ice, camera, and stone geometry stay unchanged.
+"""
+from pathlib import Path
+import cv2
+import numpy as np
+from PIL import Image
+from scipy.ndimage import (
+    label, binary_closing, binary_fill_holes, gaussian_filter,
+)
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+
+def read_source():
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    movie = []
+    while True:
+        ok, bgr = cap.read()
+        if not ok:
+            break
+        movie.append(bgr[:, :, ::-1].copy())
+    cap.release()
+    assert len(movie) == 96
+    return movie
+
+
+def track_handles(rgb):
+    tracks = []
+    yy, xx = np.mgrid[:144, :256]
+    for i, a in enumerate(rgb.astype(float)):
+        r, g, b = a.transpose(2, 0, 1)
+        guess = 47.5 + 1.97*i if i < 38 else 122 - .16*(i-38)
+        mr = ((r > 175) & (g < 105) & (b < 125)
+              & (abs(xx-guess) < 4)
+              & (yy < 66.5+.065*(guess-47.5)+1.5))
+        wr = np.maximum(r-g-90, 0)*mr
+        my = (g-b > 45) & (r-b > 35) & (g > 160)
+        wy = np.maximum(g-b-45, 0)*my
+        tracks.append([(wr*xx).sum()/wr.sum(),
+                       (wy*xx).sum()/wy.sum()])
+    return np.asarray(tracks)
+
+
+def sprite(frame, clean_strip, x):
+    # A separate uncollided stone at frame 38 (zero-based 37).
+    # Coordinates here are the original high-resolution source coordinates.
+    cy = 73.5 + .069*(x-123)
+    x0 = round((x-7)*5)
+    y0 = round((cy-7)*5)
+    patch = frame[y0:y0+75, x0:x0+75].astype(float)
+    bg = clean_strip[y0-315:y0-315+75, x0:x0+75].astype(float)
+    Y, X = np.mgrid[:75, :75]
+    X = X/5+x0/5
+    Y = Y/5+y0/5
+    # Geometry bounds the body; background subtraction excludes ice and
+    # its rings. Fill the contiguous stone, not its reflection on the ice.
+    geometric = ((X-x)/5.2)**2 + ((Y-(cy-1.3))/3.4)**2 < 1
+    foreground = (np.max(abs(patch-bg), axis=-1) > 12) & geometric
+    labs, _ = label(foreground)
+    counts = np.bincount(labs.ravel())
+    counts[0] = 0
+    foreground = labs == counts.argmax()
+    foreground = binary_fill_holes(binary_closing(foreground, iterations=2))
+    alpha = gaussian_filter(foreground.astype(float), .45)
+    alpha[alpha < .01] = 0
+
+    # Recover a neutral contact-shadow matte without transporting red
+    # painted ice along with the stone to its new location.
+    lum = np.dot(patch, [.2126, .7152, .0722])
+    blum = np.dot(bg, [.2126, .7152, .0722])
+    shade = np.clip(1-lum/np.maximum(blum, 1), 0, .38)
+    foot = cy+2.15
+    shade_region = ((X-x)/6.1)**2+((Y-(foot+.65))/2.25)**2 < 1
+    shade *= shade_region * (1-alpha)
+    shade = gaussian_filter(shade, 1.0)
+    # At the lower rim the soft floor contact fills any undersampled edge.
+    contact = .10*np.exp(-.5*((X-x)/4.7)**2
+                        -.5*((Y-(foot+.1))/.65)**2)
+    shade = np.maximum(shade, contact*(1-alpha))
+    return {'rgb': patch, 'alpha': alpha, 'shadow': shade,
+            'x0': x0, 'y0': y0, 'x': x}
+
+
+def stamp(canvas, obj, target_x):
+    dx = (target_x-obj['x'])*5
+    dy = .0685*dx
+    # Subpixel shifts at the native resolution retain the original smooth
+    # antialiasing when the finished composition is downsampled once.
+    left = int(np.floor(obj['x0']+dx))
+    top = int(np.floor(obj['y0']+dy))
+    mat = np.array([[1, 0, obj['x0']+dx-left],
+                    [0, 1, obj['y0']+dy-top]], dtype=np.float32)
+    size = (77, 77)
+    alpha = cv2.warpAffine(obj['alpha'].astype(np.float32), mat, size)
+    premult = obj['rgb']*obj['alpha'][..., None]
+    premult = cv2.warpAffine(premult.astype(np.float32), mat, size)
+    shade = cv2.warpAffine(obj['shadow'].astype(np.float32), mat, size)
+    H, W = canvas.shape[:2]
+    l, t, r, b = max(0, left), max(0, top), min(W, left+77), min(H, top+77)
+    if l >= r or t >= b:
+        return
+    sx, sy = l-left, t-top
+    sl = np.s_[sy:sy+b-t, sx:sx+r-l]
+    area = canvas[t:b, l:r]
+    area *= 1-shade[sl][..., None]
+    area *= 1-alpha[sl][..., None]
+    area += premult[sl]
+
+
+def clear_old_stones(canvas, clean_strip, xr, xy):
+    for x in [xr, xy]:
+        left = max(0, int((x-8.1)*5))
+        right = min(1280, int((x+8.1)*5)+1)
+        top, bottom = 325, 422
+        # Soft edges only outside the source object's full shadow bounds.
+        y, q = np.mgrid[top:bottom, left:right]
+        edge = np.minimum.reduce([q-left, right-1-q, y-top, bottom-1-y])
+        a = np.clip(edge/5, 0, 1)[..., None]
+        clean = clean_strip[top-315:bottom-315, left:right]
+        canvas[top:bottom, left:right] = (
+            (1-a)*canvas[top:bottom, left:right] + a*clean)
+
+
+def generate():
+    factual = np.load(ROOT / 'source_pixels.npz')['rgb']
+    hi = read_source()
+    # These early frames contain a clean observation of every pixel near
+    # the collision; moving stones disappear in the temporal median.
+    clean_strip = np.median(np.stack([a[315:430] for a in hi[:28]]), axis=0)
+    tr = track_handles(factual)
+    t = np.arange(8, 36)
+    vin = np.array([np.polyfit(t, tr[8:36, j], 1)[0] for j in range(2)])
+    tout = np.arange(40, 70)
+    vout = np.array([np.polyfit(tout, tr[40:70, j], 1)[0] for j in range(2)])
+    # Correct for the small ice deceleration in the fitted postimpact window.
+    decel = .00125
+    incident_relative = vin[0]-vin[1]
+    e = np.clip((vout[1]-vout[0]+2*decel*16)/incident_relative, 0, 1)
+    masses = np.array([1., .2])
+    impulse = (1+e)*incident_relative/(1/masses[0]+1/masses[1])
+    velocity = vin + np.array([-impulse/masses[0], impulse/masses[1]])
+
+    # Frame 39 is the contact image, before any visible separation. Equal
+    # radii and initial velocities are unchanged by the mass intervention.
+    # The following time samples carry the changed collision response.
+    pivot = 38
+    red = sprite(hi[37], clean_strip, 120.8)
+    yellow = sprite(hi[37], clean_strip, 134.8)
+    # Handle center and geometric center are offset by a fraction of a pixel.
+    initial = tr[pivot] + np.array([.27, .24])
+    frames = factual.copy()
+    positions = []
+    for i in range(pivot+1, 96):
+        dt = i-pivot
+        travel = velocity*dt - .5*decel*dt*dt*np.sign(velocity)
+        p = initial+travel
+        canvas = hi[i].astype(np.float32)
+        old = tr[i]+np.array([.27, .24])
+        clear_old_stones(canvas, clean_strip, *old)
+        # The yellow body is farther right/down and therefore in front
+        # during the few frames when the two antialiased silhouettes touch.
+        stamp(canvas, red, p[0])
+        stamp(canvas, yellow, p[1])
+        frames[i] = np.asarray(Image.fromarray(np.clip(canvas, 0, 255)
+                             .astype(np.uint8)).resize((256, 144),
+                              Image.Resampling.LANCZOS))
+        positions.append(p)
+    assert frames.shape == (96, 144, 256, 3)
+    assert np.array_equal(frames[0], factual[0])
+    artifact_io.save(frames, ROOT)
+    print('Original approach velocities (pixels/frame):', vin)
+    print('Calibrated restitution:', e)
+    print('New post-contact velocities:', velocity)
+    print('Final stone centers:', positions[-1])
+    print('Saved all 96 frames and output artifacts.')
+
+
+if __name__ == '__main__':
+    generate()

Use the original approach, infer its low-restitution contact, and solve the
same impulse with the yellow body's mass scaled by 0.2.  All appearance is

from pathlib import Path
import cv2
import numpy as np
from PIL import Image
from scipy.ndimage import (
    label, binary_closing, binary_fill_holes, gaussian_filter,
)
import artifact_io

ROOT = Path(__file__).resolve().parent


def read_source():
    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
    movie = []
    while True:
        ok, bgr = cap.read()
        if not ok:
            break
        movie.append(bgr[:, :, ::-1].copy())
    cap.release()
    assert len(movie) == 96
    return movie


def track_handles(rgb):
    tracks = []
    yy, xx = np.mgrid[:144, :256]
    for i, a in enumerate(rgb.astype(float)):
        r, g, b = a.transpose(2, 0, 1)
        guess = 47.5 + 1.97*i if i < 38 else 122 - .16*(i-38)
        mr = ((r > 175) & (g < 105) & (b < 125)
              & (abs(xx-guess) < 4)
              & (yy < 66.5+.065*(guess-47.5)+1.5))
        wr = np.maximum(r-g-90, 0)*mr
        my = (g-b > 45) & (r-b > 35) & (g > 160)
        wy = np.maximum(g-b-45, 0)*my
        tracks.append([(wr*xx).sum()/wr.sum(),
                       (wy*xx).sum()/wy.sum()])
    return np.asarray(tracks)


def sprite(frame, clean_strip, x):
    # A separate uncollided stone at frame 38 (zero-based 37).
    # Coordinates here are the original high-resolution source coordinates.
    cy = 73.5 + .069*(x-123)
    x0 = round((x-7)*5)
    y0 = round((cy-7)*5)
    patch = frame[y0:y0+75, x0:x0+75].astype(float)
    bg = clean_strip[y0-315:y0-315+75, x0:x0+75].astype(float)
    Y, X = np.mgrid[:75, :75]
    X = X/5+x0/5
    Y = Y/5+y0/5
    # Geometry bounds the body; background subtraction excludes ice and
    # its rings. Fill the contiguous stone, not its reflection on the ice.
    geometric = ((X-x)/5.2)**2 + ((Y-(cy-1.3))/3.4)**2 < 1
    foreground = (np.max(abs(patch-bg), axis=-1) > 12) & geometric
    labs, _ = label(foreground)
    counts = np.bincount(labs.ravel())
    counts[0] = 0
    foreground = labs == counts.argmax()
    foreground = binary_fill_holes(binary_closing(foreground, iterations=2))
    alpha = gaussian_filter(foreground.astype(float), .45)
    alpha[alpha < .01] = 0

    # Recover a neutral contact-shadow matte without transporting red
    # painted ice along with the stone to its new location.
    lum = np.dot(patch, [.2126, .7152, .0722])
    blum = np.dot(bg, [.2126, .7152, .0722])
    shade = np.clip(1-lum/np.maximum(blum, 1), 0, .38)
    foot = cy+2.15
    shade_region = ((X-x)/6.1)**2+((Y-(foot+.65))/2.25)**2 < 1
    shade *= shade_region * (1-alpha)
    shade = gaussian_filter(shade, 1.0)
    # At the lower rim the soft floor contact fills any undersampled edge.
    contact = .10*np.exp(-.5*((X-x)/4.7)**2
                        -.5*((Y-(foot+.1))/.65)**2)
    shade = np.maximum(shade, contact*(1-alpha))
    return {'rgb': patch, 'alpha': alpha, 'shadow': shade,
            'x0': x0, 'y0': y0, 'x': x}


def stamp(canvas, obj, target_x):
    dx = (target_x-obj['x'])*5
    dy = .0685*dx
    # Subpixel shifts at the native resolution retain the original smooth
    # antialiasing when the finished composition is downsampled once.
    left = int(np.floor(obj['x0']+dx))
    top = int(np.floor(obj['y0']+dy))
    mat = np.array([[1, 0, obj['x0']+dx-left],
                    [0, 1, obj['y0']+dy-top]], dtype=np.float32)
    size = (77, 77)
    alpha = cv2.warpAffine(obj['alpha'].astype(np.float32), mat, size)
    premult = obj['rgb']*obj['alpha'][..., None]
    premult = cv2.warpAffine(premult.astype(np.float32), mat, size)
    shade = cv2.warpAffine(obj['shadow'].astype(np.float32), mat, size)
    H, W = canvas.shape[:2]
    l, t, r, b = max(0, left), max(0, top), min(W, left+77), min(H, top+77)
    if l >= r or t >= b:
        return
    sx, sy = l-left, t-top
    sl = np.s_[sy:sy+b-t, sx:sx+r-l]
    area = canvas[t:b, l:r]
    area *= 1-shade[sl][..., None]
    area *= 1-alpha[sl][..., None]
    area += premult[sl]


def clear_old_stones(canvas, clean_strip, xr, xy):
    for x in [xr, xy]:
        left = max(0, int((x-8.1)*5))
        right = min(1280, int((x+8.1)*5)+1)
        top, bottom = 325, 422
        # Soft edges only outside the source object's full shadow bounds.
        y, q = np.mgrid[top:bottom, left:right]
        edge = np.minimum.reduce([q-left, right-1-q, y-top, bottom-1-y])
        a = np.clip(edge/5, 0, 1)[..., None]
        clean = clean_strip[top-315:bottom-315, left:right]
        canvas[top:bottom, left:right] = (
            (1-a)*canvas[top:bottom, left:right] + a*clean)


def generate():
    factual = np.load(ROOT / 'source_pixels.npz')['rgb']
    hi = read_source()
    # These early frames contain a clean observation of every pixel near
    # the collision; moving stones disappear in the temporal median.
    clean_strip = np.median(np.stack([a[315:430] for a in hi[:28]]), axis=0)
    tr = track_handles(factual)
    t = np.arange(8, 36)
    vin = np.array([np.polyfit(t, tr[8:36, j], 1)[0] for j in range(2)])
    tout = np.arange(40, 70)
    vout = np.array([np.polyfit(tout, tr[40:70, j], 1)[0] for j in range(2)])
    # Correct for the small ice deceleration in the fitted postimpact window.
    decel = .00125
    incident_relative = vin[0]-vin[1]
    e = np.clip((vout[1]-vout[0]+2*decel*16)/incident_relative, 0, 1)
    masses = np.array([1., .2])
    impulse = (1+e)*incident_relative/(1/masses[0]+1/masses[1])
    velocity = vin + np.array([-impulse/masses[0], impulse/masses[1]])

    # Frame 39 is the contact image, before any visible separation. Equal
    # radii and initial velocities are unchanged by the mass intervention.
    # The following time samples carry the changed collision response.
    pivot = 38
    red = sprite(hi[37], clean_strip, 120.8)
    yellow = sprite(hi[37], clean_strip, 134.8)
    # Handle center and geometric center are offset by a fraction of a pixel.
    initial = tr[pivot] + np.array([.27, .24])
    frames = factual.copy()
    positions = []
    for i in range(pivot+1, 96):
        dt = i-pivot
        travel = velocity*dt - .5*decel*dt*dt*np.sign(velocity)
        p = initial+travel
        canvas = hi[i].astype(np.float32)
        old = tr[i]+np.array([.27, .24])
        clear_old_stones(canvas, clean_strip, *old)
        # The yellow body is farther right/down and therefore in front
        # during the few frames when the two antialiased silhouettes touch.
        stamp(canvas, red, p[0])
        stamp(canvas, yellow, p[1])
        frames[i] = np.asarray(Image.fromarray(np.clip(canvas, 0, 255)
                             .astype(np.uint8)).resize((256, 144),
                              Image.Resampling.LANCZOS))
        positions.append(p)
    assert frames.shape == (96, 144, 256, 3)
    assert np.array_equal(frames[0], factual[0])
    artifact_io.save(frames, ROOT)
    print('Original approach velocities (pixels/frame):', vin)
    print('Calibrated restitution:', e)
    print('New post-contact velocities:', velocity)
    print('Final stone centers:', positions[-1])
    print('Saved all 96 frames and output artifacts.')


if __name__ == '__main__':
    generate()
