+"""Sixfold initial linear speed; static scene and original vertical bounce track.
+
+All imagery is sampled from the supplied source. The collision response is the
+solid-sphere tangential impulse (I=2mr^2/5), with the original initial spin.
+Original gravity and normal restitution leave the vertical trajectory unchanged.
+"""
+from pathlib import Path
+import numpy as np
+import cv2
+from PIL import Image
+from scipy.ndimage import label, binary_fill_holes, gaussian_filter, map_coordinates
+from scipy.optimize import least_squares
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+
+def circle_pose(image, plate, guess):
+    cx, cy, radius = guess
+    x0, x1 = max(0, int(cx-radius-9)), min(image.shape[1], int(cx+radius+10))
+    y0, y1 = max(0, int(cy-radius-8)), min(image.shape[0], int(cy+radius+10))
+    difference = np.max(np.abs(image[y0:y1,x0:x1].astype(float)-plate[y0:y1,x0:x1]),axis=2)
+    mask = difference > 24
+    lab, _ = label(mask)
+    sizes = np.bincount(lab.ravel()); sizes[0] = 0
+    mask = binary_fill_holes(lab == sizes.argmax())
+    ys, xs = np.where(mask)
+    points=[]
+    # Upper silhouette avoids the contact shadow below the ball.
+    for yy in np.unique(ys):
+        if yy > ys.min() + radius*1.05:
+            continue
+        xx=xs[ys==yy]
+        points.extend([(xx.min()+x0-.5, yy+y0), (xx.max()+x0+.5, yy+y0)])
+    p=np.array(points)
+    fit=least_squares(lambda q:np.linalg.norm(p-q[:2],axis=1)-q[2],
+                      [cx,cy,radius],loss='soft_l1',f_scale=1.5).x
+    return fit
+
+
+def main():
+    source=np.load(ROOT/'source_pixels.npz')['rgb']
+    cap=cv2.VideoCapture(str(ROOT/'source_video.mp4'))
+    hi=[]
+    for i in range(42):
+        ok, frame=cap.read()
+        if not ok: raise RuntimeError('Incomplete source decode')
+        hi.append(frame[:,:,::-1].copy())
+    cap.release()
+    # A source-derived clean plate. Frame 21 exposes the initial ball location;
+    # the rest of frame 1, including every grain of the wood, is retained.
+    plate=hi[0].copy()
+    plate[40:180,565:715]=hi[20][40:180,565:715]
+    platef=plate.astype(float)
+    low_plate=np.array(Image.fromarray(plate).resize((256,144),Image.Resampling.LANCZOS))
+    # Use color segmentation for an initial guess, then fit the physical edge.
+    poses=[]
+    for i,img in enumerate(hi):
+        a=source[i].astype(float)
+        mask=(a[:,:,0]>100)&(a[:,:,0]>1.6*a[:,:,1])&(a[:,:,0]>1.6*a[:,:,2])
+        mask[:,:110]=False; mask[:,180:]=False
+        lab,_=label(mask); size=np.bincount(lab.ravel());size[0]=0
+        y,x=np.where(lab==size.argmax())
+        guess=[(x.mean()+.4)*5, ((y.min()+y.max())/2+.4)*5,56]
+        poses.append(circle_pose(img,platef,guess))
+    poses=np.array(poses)
+    # Smooth horizontal projected velocity fitted to the unobstructed first
+    # flight. No scaling of gravity, time, or angular velocity is performed.
+    times=np.arange(10,dtype=float)
+    x_coeff=np.polyfit(times,poses[:10,0]/5-.4,2)
+    x0=poses[0,0]/5-.4
+    v0=x_coeff[1]*6
+    accel_x=2*x_coeff[0]*6
+    # Spin measured from local RGB texture motion before impact.
+    initial_omega=-.079
+    damping=.991
+    first_hit=12.0
+    x_hit=x0+v0*first_hit+.5*accel_x*first_hit**2
+    r_contact=poses[12,2]/5
+    incoming_v=v0+accel_x*first_hit
+    after_v=(5*incoming_v+2*r_contact*initial_omega)/7
+    angular_extra=0.
+    x=x0
+    vx=after_v
+    trace=[]
+    output=[]
+    yylo,xxlo=np.mgrid[:144,:256]
+    for i in range(96):
+        if i<=12:
+            x=x0+v0*i+.5*accel_x*i*i
+        else:
+            x+=vx
+            # A solid ball's post-impact spin accelerates to the rolling speed.
+            original_omega=.031 * damping**(i-13)
+            angular_extra+=vx/r_contact-original_omega
+            vx*=damping
+        if i>=len(hi) or x>272:
+            output.append(low_plate.copy())

+        # Restrict it to the floor and remove the sphere from this field.
+        dark=(low_plate.astype(float).mean(2)-source[i].astype(float).mean(2)) / np.maximum(1,low_plate.mean(2))

+        dark=np.where(valid & (dark>.012),np.clip(dark,0,.65),0)

+        # sampling as the supplied RGB tensor, for faithful subpixel edges.
+        large=platef.copy()

+            ux=gx-target_x;uy=gy-cy
+            dist=np.sqrt(ux*ux+uy*uy)
+            # Radius inferred at the .25-alpha silhouette threshold.
+            alpha=np.clip((r+.4-dist)/1.8,0,1)
+            theta=angular_extra if i>12 else 0.
+            ct,st=np.cos(theta),np.sin(theta)
+            sx=cx+ct*ux+st*uy
+            sy=cy-st*ux+ct*uy
+            old_dist=np.sqrt((sx-cx)**2+(sy-cy)**2)
+            old_alpha=np.clip((r+.4-old_dist)/1.8,0,1)
+            rgb=np.stack([map_coordinates(hi[i][:,:,j].astype(float),[sy,sx],order=1,mode='nearest') for j in range(3)],-1)
+            old_bg=np.stack([map_coordinates(platef[:,:,j],[sy,sx],order=1,mode='nearest') for j in range(3)],-1)
+            rgb=(rgb-(1-old_alpha[:,:,None])*old_bg)/np.maximum(old_alpha[:,:,None],.035)
+            if i>12:
+                # Rotate markings, not illumination. Fit a low-order diffuse
+                # light field robustly so the dark seams are not mistaken for
+                # a directional change in illumination.
+                nx=(sx-cx)/r;ny=(sy-cy)/r
+                nz=np.sqrt(np.maximum(0,1-nx*nx-ny*ny))
+                core=(nx*nx+ny*ny<.78)
+                A=np.stack([np.ones_like(nx),nx,ny,nz],-1)
+                B=A[core]; colors=rgb[core]
+                coeff=[]
+                for j in range(3):
+                    q=np.linalg.lstsq(B,colors[:,j],rcond=None)[0]
+                    q=least_squares(lambda q:B@q-colors[:,j],q,loss='soft_l1',f_scale=6,max_nfev=20).x
+                    coeff.append(q)
+                coeff=np.array(coeff).T
+                old_light=np.maximum(18,A@coeff)
+                newnx=ux/r;newny=uy/r;newnz=np.sqrt(np.maximum(0,1-newnx*newnx-newny*newny))
+                newA=np.stack([np.ones_like(nx),newnx,newny,newnz],-1)
+                rgb*=np.maximum(18,newA@coeff)/old_light
+            rgb=np.clip(rgb,0,255)
+            large[by0:by1,bx0:bx1]=large[by0:by1,bx0:bx1]*(1-alpha[:,:,None])+rgb*alpha[:,:,None]
+        frame=np.array(Image.fromarray(np.clip(np.rint(large),0,255).astype(np.uint8)).resize((256,144),Image.Resampling.LANCZOS))
+        if i==0:frame=source[0].copy()
+        output.append(frame)
+        trace.append([i,x,source_y,(incoming_v if i==12 else vx)])
+    frames=np.array(output,dtype=np.uint8)
+    artifact_io.save(frames,ROOT)
+    np.savetxt(ROOT/'motion_trace.csv',np.array(trace),delimiter=',',header='frame,x,y,vx',comments='')
+    print('Saved',frames.shape,'uint8 RGB frames; initial horizontal velocity',round(v0,4),'px/frame; post-impact velocity',round(after_v,4),'px/frame')
+    print('Ball exits view at frame', next(i for i,t in enumerate(trace) if t[1]>268))
+
+if __name__=='__main__':
+    main()

\"\"\"Sixfold initial linear speed; static scene and original vertical bounce track.

All imagery is sampled from the supplied source. The collision response is the
solid-sphere tangential impulse (I=2mr"'^2/5), with the original initial spin.
Original gravity and normal restitution leave the vertical trajectory unchanged.

from pathlib import Path
import numpy as np
import cv2
from PIL import Image
from scipy.ndimage import label, binary_fill_holes, gaussian_filter, map_coordinates
from scipy.optimize import least_squares
import artifact_io

ROOT = Path(__file__).resolve().parent


def circle_pose(image, plate, guess):
    cx, cy, radius = guess
    x0, x1 = max(0, int(cx-radius-9)), min(image.shape[1], int(cx+radius+10))
    y0, y1 = max(0, int(cy-radius-8)), min(image.shape[0], int(cy+radius+10))
    difference = np.max(np.abs(image[y0:y1,x0:x1].astype(float)-plate[y0:y1,x0:x1]),axis=2)
    mask = difference > 24
    lab, _ = label(mask)
    sizes = np.bincount(lab.ravel()); sizes[0] = 0
    mask = binary_fill_holes(lab == sizes.argmax())
    ys, xs = np.where(mask)
    points=[]
    # Upper silhouette avoids the contact shadow below the ball.
    for yy in np.unique(ys):
        if yy > ys.min() + radius*1.05:
            continue
        xx=xs[ys==yy]
        points.extend([(xx.min()+x0-.5, yy+y0), (xx.max()+x0+.5, yy+y0)])
    p=np.array(points)
    fit=least_squares(lambda q:np.linalg.norm(p-q[:2],axis=1)-q[2],
                      [cx,cy,radius],loss='"'soft_l1',f_scale=1.5).x
    return fit
