+"""Mass-only counterfactual, reconstructed from the supplied RGB source.
+
+The two curling stones run on a single perspective-projected straight lane.
+Only the yellow mass changes (1 -> 3); geometry, launch velocities, friction
+and the very inelastic contact remain unchanged. The factual frames before
+contact are therefore retained verbatim. No simulation/rendering engine is
+used: contact is an analytic impulse and sprites are source-pixel mattes.
+"""
+from pathlib import Path
+import numpy as np
+from PIL import Image
+import cv2
+from scipy.ndimage import label, binary_fill_holes, binary_closing, gaussian_filter
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+
+def largest(mask):
+    lab, _ = label(mask)
+    sizes = np.bincount(lab.ravel())
+    sizes[0] = 0
+    return lab == int(sizes.argmax())
+
+
+def body_axis(image, approximate):
+    # The neutral-gray lower stone body is independent of the colored target
+    # painted on the ice, so it supplies an unambiguous translation anchor.
+    x, y = approximate
+    x0, y0 = int(x * 5) - 36, int(y * 5) - 32
+    crop = image[y0:y0+64, x0:x0+72].astype(np.float32)
+    r, g, b = crop.transpose(2, 0, 1)
+    neutral = ((r < 180) & (g < 181) & (b < 188)
+               & (np.abs(r-g) < 24) & (np.abs(b-g) < 25))
+    yy, xx = np.where(largest(neutral))
+    return np.array([x0 + (xx.min()+xx.max())/2,
+                     y0 + (yy.min()+yy.max())/2], dtype=float)
+
+
+def extract_asset(hi, background, stone):
+    ref = 28
+    approximate_ref = [(102.8, 71.5), (153.3, 74.7)][stone]
+    approximate_last = [(120.7, 73.5), (134.5, 74.3)][stone]
+    ref_axis = body_axis(hi[ref], approximate_ref)
+    last_axis = body_axis(hi[37], approximate_last)
+    # A clean silhouette is acquired before the stones overlap the red target.
+    x0, y0 = np.round(ref_axis).astype(int) - [39, 35]
+    crop = hi[ref, y0:y0+72, x0:x0+80]
+    clean = background[y0:y0+72, x0:x0+80]
+    diff = np.max(np.abs(crop.astype(float)-clean.astype(float)), axis=2)
+    mask = binary_fill_holes(binary_closing(largest(diff > 22), iterations=1))
+    # Source-resolution antialiasing gives a sharp ~10px stone at output size.
+    alpha = gaussian_filter(mask.astype(np.float32), .55)
+    alpha[alpha < .008] = 0
+    alpha[alpha > .992] = 1
+    # Translate this source-derived silhouette to the last pre-contact frame.
+    # Keep that frame's actual texture, including the handle's orientation.
+    yy, xx = np.indices(mask.shape, dtype=np.float32)
+    shift = last_axis-ref_axis
+    sx = (xx+x0+shift[0]).astype(np.float32)
+    sy = (yy+y0+shift[1]).astype(np.float32)
+    observed = cv2.remap(hi[37].astype(np.float32), sx, sy,
+                         cv2.INTER_LINEAR, borderMode=cv2.BORDER_REFLECT)
+    bg = cv2.remap(background.astype(np.float32), sx, sy,
+                   cv2.INTER_LINEAR, borderMode=cv2.BORDER_REFLECT)
+    # Remove the old ice/target contribution at the antialiased boundary.
+    denom = np.maximum(alpha[..., None], .18)
+    foreground = np.clip((observed-(1-alpha[...,None])*bg)/denom, 0, 255)
+    foreground[alpha < .18] = observed[alpha < .18]
+    premult = foreground * alpha[..., None]
+    local_axis = ref_axis - [x0, y0]
+    return premult, alpha, local_axis, last_axis / 5
+
+
+def motion(frame, initial, stone):
+    # Velocities are measured from the factual constant-speed approach.
+    u = np.array([2.05, -2.08])
+    mass = np.array([1., 3.])
+    e = .095  # measured small rebound / incoming relative speed
+    J = (1+e)*(u[0]-u[1]) / (1/mass[0]+1/mass[1])
+    v = u + np.array([-J/mass[0], J/mass[1]])
+    contact_start, duration = 37.77, .40
+    s = frame-37.
+    before = contact_start-37.
+    if frame <= contact_start:
+        dx = u[stone]*s
+    else:
+        z = min((frame-contact_start)/duration, 1.)
+        # A smooth distributed contact impulse; integral of smoothstep.
+        integral = z**3-.5*z**4
+        dx = u[stone]*before + duration*(u[stone]*z + (v[stone]-u[stone])*integral)
+        if z == 1:
+            elapsed = frame-contact_start-duration
+            # The same weak ice resistance as in the original. Coulomb
+            # acceleration is independent of mass because N=m*g.
+            deceleration = .0013
+            elapsed = min(elapsed, abs(v[stone])/deceleration)
+            dx += v[stone]*elapsed - np.sign(v[stone])*.5*deceleration*elapsed**2
+    x = initial[0]+dx
+    # Source lane projection: stones moving left also move slightly upward.
+    y = initial[1]+.065*dx
+    return np.array([x,y])
+
+
+def composite(canvas, asset, position):
+    premult, alpha, anchor, _ = asset
+    origin = position*5-anchor
+    x0, y0 = np.floor(origin).astype(int)
+    frac = origin - [x0,y0]
+    height, width = alpha.shape
+    # One-pixel pad allows fractional translation without edge truncation.
+    M = np.array([[1,0,frac[0]+1],[0,1,frac[1]+1]], dtype=np.float32)
+    rgb = cv2.warpAffine(premult, M, (width+2,height+2), flags=cv2.INTER_LINEAR)
+    a = cv2.warpAffine(alpha, M, (width+2,height+2), flags=cv2.INTER_LINEAR)
+    x0 -= 1; y0 -= 1
+    x1, y1 = x0+width+2, y0+height+2
+    bx0, by0, bx1, by1 = max(x0,0),max(y0,0),min(x1,1280),min(y1,720)
+    sx0, sy0 = bx0-x0,by0-y0
+    rgb = rgb[sy0:sy0+by1-by0,sx0:sx0+bx1-bx0]
+    a = a[sy0:sy0+by1-by0,sx0:sx0+bx1-bx0]
+    canvas[by0:by1,bx0:bx1] = rgb + canvas[by0:by1,bx0:bx1]*(1-a[...,None])
+
+
+def main():
+    original = np.load(ROOT/'source_pixels.npz')['rgb']
+    cap = cv2.VideoCapture(str(ROOT/'source_video.mp4'))
+    decoded = []
+    while True:
+        ok, frame = cap.read()
+        if not ok:
+            break
+        decoded.append(cv2.cvtColor(frame, cv2.COLOR_BGR2RGB))
+    cap.release()
+    hi = np.asarray(decoded)
+    assert hi.shape == (96,720,1280,3)
+    background = np.median(hi[:28], axis=0).astype(np.uint8)
+    assets = [extract_asset(hi, background, j) for j in range(2)]
+    output = original.copy()
+    # The two factual stones only occupy this ROI after contact. Replacing it
+    # with observed empty ice removes both their factual paths and shadows.
+    x0,x1,y0,y1 = 535,760,325,420
+    trajectories = []
+    for t in range(38,96):
+        canvas = hi[t].astype(np.float32)
+        canvas[y0:y1,x0:x1] = background[y0:y1,x0:x1]
+        positions = [motion(t, assets[j][3], j) for j in range(2)]
+        # Both stones are on the ice; source depth order is retained.
+        for j in range(2):
+            composite(canvas, assets[j], positions[j])
+        output[t] = np.asarray(Image.fromarray(np.clip(np.round(canvas),0,255).astype(np.uint8))
+                               .resize((256,144), Image.Resampling.LANCZOS))
+        trajectories.append([t,*positions[0],*positions[1]])
+    assert output.shape == (96,144,256,3) and output.dtype == np.uint8
+    artifact_io.save(output, ROOT)
+    np.savetxt(ROOT/'motion.csv', np.array(trajectories), delimiter=',',
+               header='frame_0based,red_x,red_y,yellow_x,yellow_y', comments='')
+    print('Saved 96 RGB frames at 24 fps.')
+    print('Body anchors:', assets[0][3], assets[1][3])
+    print('Final positions:', trajectories[-1])
+
+
+if __name__ == '__main__':
+    main()

\"\"\"Mass-only counterfactual, reconstructed from the supplied RGB source.

The two curling stones run on a single perspective-projected straight lane.
Only the yellow mass changes (1 -> 3); geometry, launch velocities, friction
and the very inelastic contact remain unchanged. The factual frames before
contact are therefore retained verbatim. No simulation/rendering engine is
used: contact is an analytic impulse and sprites are source-pixel mattes.
\"\"\"
from pathlib import Path
import numpy as np
from PIL import Image
import cv2
from scipy.ndimage import label, binary_fill_holes, binary_closing, gaussian_filter
import artifact_io

ROOT = Path(__file__).resolve().parent


def largest(mask):
    lab, _ = label(mask)
    sizes = np.bincount(lab.ravel())
    sizes[0] = 0
    return lab == int(sizes.argmax())


def body_axis(image, approximate):
    # The neutral-gray lower stone body is independent of the colored target
    # painted on the ice, so it supplies an unambiguous translation anchor.
    x, y = approximate
    x0, y0 = int(x * 5) - 36, int(y * 5) - 32
    crop = image[y0:y0+64, x0:x0+72].astype(np.float32)
    r, g, b = crop.transpose(2, 0, 1)
    neutral = ((r < 180) & (g < 181) & (b < 188)
               & (np.abs(r-g) < 24) & (np.abs(b-g) < 25))
    yy, xx = np.where(largest(neutral))
    return np.array([x0 + (xx.min()+xx.max())/2,
                     y0 + (yy.min()+yy.max())/2], dtype=float)


def extract_asset(hi, background, stone):
    ref = 28
    approximate_ref = [(102.8, 71.5), (153.3, 74.7)][stone]
    approximate_last = [(120.7, 73.5), (134.5, 74.3)][stone]
    ref_axis = body_axis(hi[ref], approximate_ref)
    last_axis = body_axis(hi[37], approximate_last)
    # A clean silhouette is acquired before the stones overlap the red target.
    x0, y0 = np.round(ref_axis).astype(int) - [39, 35]
    crop = hi[ref, y0:y0+72, x0:x0+80]
    clean = background[y0:y0+72, x0:x0+80]
    diff = np.max(np.abs(crop.astype(float)-clean.astype(float)), axis=2)
    mask = binary_fill_holes(binary_closing(largest(diff > 22), iterations=1))
    # Source-resolution antialiasing gives a sharp ~10px stone at output size.
    alpha = gaussian_filter(mask.astype(np.float32), .55)
    alpha[alpha < .008] = 0
    alpha[alpha > .992] = 1
    # Translate this source-derived silhouette to the last pre-contact frame.
    # Keep that frame's actual texture, including the handle's orientation.
    yy, xx = np.indices(mask.shape, dtype=np.float32)
    shift = last_axis-ref_axis
    sx = (xx+x0+shift[0]).astype(np.float32)
    sy = (yy+y0+shift[1]).astype(np.float32)
    observed = cv2.remap(hi[37].astype(np.float32), sx, sy,
                         cv2.INTER_LINEAR, borderMode=cv2.BORDER_REFLECT)
    bg = cv2.remap(background.astype(np.float32), sx, sy,
                   cv2.INTER_LINEAR, borderMode=cv2.BORDER_REFLECT)
    # Remove the old ice/target contribution at the antialiased boundary.
    denom = np.maximum(alpha[..., None], .18)
    foreground = np.clip((observed-(1-alpha[...,None])*bg)/denom, 0, 255)
    foreground[alpha < .18] = observed[alpha < .18]
    premult = foreground * alpha[..., None]
    local_axis = ref_axis - [x0, y0]
    return premult, alpha, local_axis, last_axis / 5


def motion(frame, initial, stone):
    # Velocities are measured from the factual constant-speed approach.
    u = np.array([2.05, -2.08])
    mass = np.array([1., 3.])
    e = .095  # measured small rebound / incoming relative speed
    J = (1+e)*(u[0]-u[1]) / (1/mass[0]+1/mass[1])
    v = u + np.array([-J/mass[0], J/mass[1]])
    contact_start, duration = 37.77, .40
    s = frame-37.
    before = contact_start-37.
    if frame <= contact_start:
        dx = u[stone]*s
    else:
        z = min((frame-contact_start)/duration, 1.)
        # A smooth distributed contact impulse; integral of smoothstep.
        integral = z**3-.5*z**4
        dx = u[stone]*before + duration*(u[stone]*z + (v[stone]-u[stone])*integral)
        if z == 1:
            elapsed = frame-contact_start-duration
            # The same weak ice resistance as in the original. Coulomb
            # acceleration is independent of mass because N=m*g.
            deceleration = .0013
            elapsed = min(elapsed, abs(v[stone])/deceleration)
            dx += v[stone]*elapsed - np.sign(v[stone])*.5*deceleration*elapsed**2
    x = initial[0]+dx
    # Source lane projection: stones moving left also move slightly upward.
    y = initial[1]+.065*dx
    return np.array([x,y])


def composite(canvas, asset, position):
    premult, alpha, anchor, _ = asset
    origin = position*5-anchor
    x0, y0 = np.floor(origin).astype(int)
    frac = origin - [x0,y0]
    height, width = alpha.shape
    # One-pixel pad allows fractional translation without edge truncation.
    M = np.array([[1,0,frac[0]+1],[0,1,frac[1]+1]], dtype=np.float32)
    rgb = cv2.warpAffine(premult, M, (width+2,height+2), flags=cv2.INTER_LINEAR)
    a = cv2.warpAffine(alpha, M, (width+2,height+2), flags=cv2.INTER_LINEAR)
    x0 -= 1; y0 -= 1
    x1, y1 = x0+width+2, y0+height+2
    bx0, by0, bx1, by1 = max(x0,0),max(y0,0),min(x1,1280),min(y1,720)
    sx0, sy0 = bx0-x0,by0-y0
    rgb = rgb[sy0:sy0+by1-by0,sx0:sx0+bx1-bx0]
    a = a[sy0:sy0+by1-by0,sx0:sx0+bx1-bx0]
    canvas[by0:by1,bx0:bx1] = rgb + canvas[by0:by1,bx0:bx1]*(1-a[...,None])


def main():
    original = np.load(ROOT/'source_pixels.npz')['rgb']
    cap = cv2.VideoCapture(str(ROOT/'source_video.mp4'))
    decoded = []
    while True:
        ok, frame = cap.read()
        if not ok:
            break
        decoded.append(cv2.cvtColor(frame, cv2.COLOR_BGR2RGB))
    cap.release()
    hi = np.asarray(decoded)
    assert hi.shape == (96,720,1280,3)
    background = np.median(hi[:28], axis=0).astype(np.uint8)
    assets = [extract_asset(hi, background, j) for j in range(2)]
    output = original.copy()
    # The two factual stones only occupy this ROI after contact. Replacing it
    # with observed empty ice removes both their factual paths and shadows.
    x0,x1,y0,y1 = 535,760,325,420
    trajectories = []
    for t in range(38,96):
        canvas = hi[t].astype(np.float32)
        canvas[y0:y1,x0:x1] = background[y0:y1,x0:x1]
        positions = [motion(t, assets[j][3], j) for j in range(2)]
        # Both stones are on the ice; source depth order is retained.
        for j in range(2):
            composite(canvas, assets[j], positions[j])
        output[t] = np.asarray(Image.fromarray(np.clip(np.round(canvas),0,255).astype(np.uint8))
                               .resize((256,144), Image.Resampling.LANCZOS))
        trajectories.append([t,*positions[0],*positions[1]])
    assert output.shape == (96,144,256,3) and output.dtype == np.uint8
    artifact_io.save(output, ROOT)
    np.savetxt(ROOT/'motion.csv', np.array(trajectories), delimiter=',',
               header='frame_0based,red_x,red_y,yellow_x,yellow_y', comments='')
    print('Saved 96 RGB frames at 24 fps.')
    print('Body anchors:', assets[0][3], assets[1][3])
    print('Final positions:', trajectories[-1])


if __name__ == '__main__':
    main()
