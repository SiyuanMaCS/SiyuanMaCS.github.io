+"""Reduced-mass counterfactual, reconstructed from RGB and impulse mechanics.
+
+The scenery is recovered from unobstructed source times.  Marble appearance,
+including their contact reflections, is sampled along the source's own tracks;
+only the two post-impact trajectories change. No external renderer is used.
+"""
+from pathlib import Path
+import numpy as np
+import cv2
+from scipy.ndimage import label, binary_fill_holes, binary_closing
+from scipy.optimize import least_squares
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+SCALE = 5
+
+def load_source():
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    frames = []
+    # Early moving objects plus one unobstructed frame of the initial positions.
+    for i in range(41):
+        ok, frame = cap.read()
+        if not ok:
+            raise RuntimeError('Incomplete source')
+        frames.append(frame[..., ::-1].copy())
+    cap.release()
+    return np.stack(frames)
+
+
+def track_low(a):
+    yy, xx = np.indices(a.shape[1:3])
+    tr = np.empty((len(a), 2, 2))
+    for t, frame in enumerate(a):
+        p = frame.astype(np.int16)
+        masks = [(p[...,0] > 130) & (p[...,1] > 95) & (p[...,2] < 100),
+                 (p[...,1] - p[...,0] > 35) & (p[...,2] - p[...,0] > 30)]
+        for obj, m in enumerate(masks):
+            m &= (yy > 54) & (yy < 79)
+            lab, _ = label(m)
+            sizes = np.bincount(lab.ravel()); sizes[0] = 0
+            y, x = np.where(lab == np.argmax(sizes))
+            tr[t,obj] = [x.mean() + (.85 if obj == 0 else .35), y.mean()+.45]
+    return tr
+
+
+def make_background(h):
+    b = h[0].copy()
+    # These regions are unobstructed later in the source; all other scenery
+    # remains literally the original first-frame RGB.
+    b[270:470, 1030:1190] = h[40,270:470,1030:1190]
+    b[320:480, 620:715] = h[40,320:480,620:715]
+    return b
+
+
+def geometry(src, bg, obj, guess):
+    """Estimate the silhouette with a robust bounded ellipse fit, in source px."""
+    cx0, cy0 = guess * SCALE
+    rx0 = (9.15 if obj == 0 else 5.05) * SCALE
+    ry0 = (8.2 if obj == 0 else 4.6) * SCALE
+    if obj == 1 and cx0/SCALE > 130.5:
+        rx0, ry0 = 4.3*SCALE, 4.35*SCALE
+    xlo, xhi = max(0,int(cx0-rx0-6)), min(src.shape[1],int(cx0+rx0+7))
+    diff = np.max(abs(src.astype(np.int16)-bg.astype(np.int16)), axis=2)
+    pts = []
+    # The upper and lateral arcs are free of contact shadows and reflections.
+    for y in range(int(cy0-ry0-6), int(cy0+ry0*.40)):
+        xs = np.flatnonzero(diff[y,xlo:xhi] > 28) + xlo
+        if len(xs) > 5:
+            pts += [(xs[0]-.5, y), (xs[-1]+.5, y)]
+    pts = np.asarray(pts)
+    def res(z):
+        return ((pts[:,0]-z[0])/z[2])**2 + ((pts[:,1]-z[1])/z[3])**2 - 1
+    fit = least_squares(res, [cx0,cy0,rx0,ry0], loss='soft_l1', f_scale=.04,
+        bounds=([cx0-5,cy0-5,rx0*.85,ry0*.85],
+                [cx0+5,cy0+5,rx0*1.12,ry0*1.12]))
+    return fit.x
+
+
+def sprite(src, bg, geom, obj):
+    cx,cy,rx,ry = geom
+    # 34x40 screen-pixel local patch, including reflection and contact shadow.
+    y,x = np.indices((200,170),dtype=np.float32)
+    dx,dy = x-85, y-65
+    mx,my = (cx+dx).astype(np.float32),(cy+dy).astype(np.float32)
+    col = cv2.remap(src, mx,my,cv2.INTER_LINEAR,borderMode=cv2.BORDER_REFLECT_101).astype(np.float32)
+    oldbg = cv2.remap(bg,mx,my,cv2.INTER_LINEAR,borderMode=cv2.BORDER_REFLECT_101).astype(np.float32)
+    d = col-oldbg
+    rho = (dx/rx)**2+(dy/ry)**2
+    # Detect the object separately from effects to avoid copying its background.
+    strong = (np.max(abs(d),axis=2)>13)&(rho < 1.08)
+    strong = binary_closing(strong, structure=np.ones((3,3)))
+    strong = binary_fill_holes(strong)
+    alpha = cv2.GaussianBlur(strong.astype(np.float32),(3,3),.48)
+    # Restrict additive reflected color to the strip beneath this marble.
+    below = (dy > ry*.78) & (dy < ry+95)
+    shadow_width = rx*1.22
+    if obj == 0:
+        shadow_width = rx*1.35
+    effect = below & (abs(dx) < shadow_width) & (alpha < .08)
+    # Do not displace tiny codec fluctuations of the stationary world.
+    signal = np.max(abs(d),axis=2)
+    fade = np.clip((signal-5)/10,0,1)
+    edge = np.clip((shadow_width-abs(dx))/5,0,1)
+    bottom = np.clip((ry+95-dy)/10,0,1)
+    residual = d * (effect*fade*edge*bottom)[...,None]
+    return col, alpha, residual
+
+
+def composite(canvas, sp, cx,cy):
+    col,alpha,residual = sp
+    x0=int(np.floor(cx))-85; y0=int(np.floor(cy))-65
+    # Subpixel placement keeps slow deceleration smooth at the 256px output.
+    fx,fy=cx-np.floor(cx),cy-np.floor(cy)
+    mat=np.float32([[1,0,fx],[0,1,fy]])
+    col=cv2.warpAffine(col,mat,(170,200),flags=cv2.INTER_LINEAR)
+    alpha=cv2.warpAffine(alpha,mat,(170,200),flags=cv2.INTER_LINEAR)
+    residual=cv2.warpAffine(residual,mat,(170,200),flags=cv2.INTER_LINEAR)
+    xa,ya=max(x0,0),max(y0,0)
+    xb,yb=min(x0+170,canvas.shape[1]),min(y0+200,canvas.shape[0])
+    sy=slice(ya-y0,yb-y0);sx=slice(xa-x0,xb-x0)
+    a=alpha[sy,sx,None]
+    region=canvas[ya:yb,xa:xb]
+    region += residual[sy,sx]
+    region[:]=region*(1-a)+col[sy,sx]*a
+
+
+def main():
+    low=np.load(ROOT/'source_pixels.npz')['rgb']
+    high=load_source(); bg=make_background(high)
+    tr=track_low(low)
+    ref_indices={0:list(range(12,34)),1:[0]+list(range(13,39))}
+    geoms={}; sprites={}
+    for obj in range(2):
+        for t in ref_indices[obj]:
+            guess=tr[t,obj].copy()
+            if obj==0:
+                guess[1]=64.28+.0455*(224.38-guess[0])
+            else:
+                guess[1]=72.73+.0265*(131.52-guess[0])
+            g=geometry(high[t],bg,obj,guess)
+            geoms[obj,t]=g
+            sprites[obj,t]=sprite(high[t],bg,g,obj)
+    # Perspective-linear track coordinate, inferred from the small marble's
+    # constant post-impact deceleration in the source.
+    L=1542.0
+    q=lambda x:x/(1-x/L)
+    screen=lambda x:x/(1+x/L)
+    collision=13.6
+    tb=np.arange(2,13)-collision
+    pre=np.polyfit(tb,q(tr[2:13,0,0]),2)
+    tb=np.arange(20,33)-collision
+    post=np.polyfit(tb,q(tr[20:33,0,0]),2)
+    ts=np.arange(15,39)-collision
+    smallpost=np.polyfit(ts,q(tr[15:39,1,0]),2)
+    u=pre[1]; vb=post[1]; vs=smallpost[1]
+    original_mass_ratio=vs/(u-vb)
+    restitution=np.clip((vb-vs)/abs(u),.1,.95)
+    ratio=original_mass_ratio*.125
+    newvb=(ratio-restitution)/(ratio+1)*u
+    newvs=(1+restitution)*ratio/(ratio+1)*u
+    # Measured contact/friction losses are mass-independent accelerations.
+    ab=max(.16,2*post[0]); ass=max(.25,2*smallpost[0])
+    xb0=screen(pre[2]); xs0=131.52
+    def trajectory(x0,v,acc,t):
+        dt=min(max(t,0),abs(v)/acc)
+        return screen(q(x0)+v*dt+.5*acc*dt*dt)
+    frames=low.copy()
+    positions=[]
+    for t in range(14,96):
+        elapsed=t-collision
+        xb=trajectory(xb0,newvb,ab,elapsed)
+        xs=trajectory(xs0,newvs,ass,elapsed)
+        positions.append([t,xb,xs])
+        canvas=bg.astype(np.float32)
+        for obj,xwant in [(0,xb),(1,xs)]:
+            indices=ref_indices[obj]
+            # Appearance is interpolated by displacement: light, silhouette,
+            # rolling texture and reflections are drawn from nearby source RGB.
+            xx=np.array([geoms[obj,k][0]/SCALE for k in indices])
+            order=np.argsort(xx);xx=xx[order]
+            ii=np.searchsorted(xx,xwant)
+            ia=order[max(0,min(len(xx)-1,ii-1))]
+            ib=order[max(0,min(len(xx)-1,ii))]
+            ka,kb=indices[ia],indices[ib]
+            xa,xbref=geoms[obj,ka][0]/SCALE,geoms[obj,kb][0]/SCALE
+            blend=np.clip((xwant-xa)/(xbref-xa),0,1) if xbref!=xa else 0
+            sp=tuple((1-blend)*a+blend*b for a,b in zip(sprites[obj,ka],sprites[obj,kb]))
+            y=(1-blend)*geoms[obj,ka][1]+blend*geoms[obj,kb][1]
+            composite(canvas,sp,xwant*SCALE,y)
+        canvas=np.clip(canvas,0,255).astype(np.uint8)
+        frames[t]=cv2.resize(canvas,(256,144),interpolation=cv2.INTER_AREA)
+    # First source frame is the unchanged visible state at the mass intervention.
+    artifact_io.save(frames,ROOT)
+    np.save(ROOT/'motion.npy',np.asarray(positions))
+    print('Source mass ratio / restitution:',original_mass_ratio,restitution)
+    print('Counterfactual velocities:',newvb,newvs)
+    print('Final marble x:',positions[-1][1:])
+    print('Saved',frames.shape,frames.dtype)
+
+if __name__=='__main__':
+    main()

The scenery is recovered from unobstructed source times.  Marble appearance,

import cv2

from scipy.optimize import least_squares

def load_source():
    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
    frames = []
    # Early moving objects plus one unobstructed frame of the initial positions.
    for i in range(41):
        ok, frame = cap.read()
        if not ok:
            raise RuntimeError('Incomplete source')
        frames.append(frame[..., ::-1].copy())
    cap.release()
    return np.stack(frames)


def track_low(a):
    yy, xx = np.indices(a.shape[1:3])
    tr = np.empty((len(a), 2, 2))
    for t, frame in enumerate(a):
        p = frame.astype(np.int16)
        masks = [(p[...,0] > 130) & (p[...,1] > 95) & (p[...,2] < 100),
                 (p[...,1] - p[...,0] > 35) & (p[...,2] - p[...,0] > 30)]
        for obj, m in enumerate(masks):
            m &= (yy > 54) & (yy < 79)
            lab, _ = label(m)
            sizes = np.bincount(lab.ravel()); sizes[0] = 0
            y, x = np.where(lab == np.argmax(sizes))
            tr[t,obj] = [x.mean() + (.85 if obj == 0 else .35), y.mean()+.45]
    return tr


def make_background(h):
    b = h[0].copy()
    # These regions are unobstructed later in the source; all other scenery
    # remains literally the original first-frame RGB.
    b[270:470, 1030:1190] = h[40,270:470,1030:1190]
    b[320:480, 620:715] = h[40,320:480,620:715]
    return b


def geometry(src, bg, obj, guess):
    \"\"\"Estimate the silhouette with a robust bounded ellipse fit, in source px.\"\"\"
    cx0, cy0 = guess * SCALE
    rx0 = (9.15 if obj == 0 else 5.05) * SCALE
    ry0 = (8.2 if obj == 0 else 4.6) * SCALE
    if obj == 1 and cx0/SCALE > 130.5:
        rx0, ry0 = 4.3*SCALE, 4.35*SCALE
    xlo, xhi = max(0,int(cx0-rx0-6)), min(src.shape[1],int(cx0+rx0+7))
    diff = np.max(abs(src.astype(np.int16)-bg.astype(np.int16)), axis=2)
    pts = []
    # The upper and lateral arcs are free of contact shadows and reflections.
    for y in range(int(cy0-ry0-6), int(cy0+ry0*.40)):
        xs = np.flatnonzero(diff[y,xlo:xhi] > 28) + xlo
        if len(xs) > 5:
            pts += [(xs[0]-.5, y), (xs[-1]+.5, y)]
    pts = np.asarray(pts)
    def res(z):
        return ((pts[:,0]-z[0])/z[2])**2 + ((pts[:,1]-z[1])/z[3])**2 - 1
    fit = least_squares(res, [cx0,cy0,rx0,ry0], loss='soft_l1', f_scale=.04,
        bounds=([cx0-5,cy0-5,rx0*.85,ry0*.85],
                [cx0+5,cy0+5,rx0*1.12,ry0*1.12]))
    return fit.x


def sprite(src, bg, geom, obj):
    cx,cy,rx,ry = geom
    # 34x40 screen-pixel local patch, including reflection and contact shadow.
    y,x = np.indices((200,170),dtype=np.float32)
    dx,dy = x-85, y-65
    mx,my = (cx+dx).astype(np.float32),(cy+dy).astype(np.float32)
    col = cv2.remap(src, mx,my,cv2.INTER_LINEAR,borderMode=cv2.BORDER_REFLECT_101).astype(np.float32)
    oldbg = cv2.remap(bg,mx,my,cv2.INTER_LINEAR,borderMode=cv2.BORDER_REFLECT_101).astype(np.float32)
    d = col-oldbg
    rho = (dx/rx)**2+(dy/ry)**2
    # Detect the object separately from effects to avoid copying its background.
    strong = (np.max(abs(d),axis=2)>13)&(rho < 1.08)
    strong = binary_closing(strong, structure=np.ones((3,3)))
    strong = binary_fill_holes(strong)
    alpha = cv2.GaussianBlur(strong.astype(np.float32),(3,3),.48)
    # Restrict additive reflected color to the strip beneath this marble.
    below = (dy > ry*.78) & (dy < ry+95)
    shadow_width = rx*1.22
    if obj == 0:
        shadow_width = rx*1.35
    effect = below & (abs(dx) < shadow_width) & (alpha < .08)
    # Do not displace tiny codec fluctuations of the stationary world.
    signal = np.max(abs(d),axis=2)
    fade = np.clip((signal-5)/10,0,1)
    edge = np.clip((shadow_width-abs(dx))/5,0,1)
    bottom = np.clip((ry+95-dy)/10,0,1)
    residual = d * (effect*fade*edge*bottom)[...,None]
    return col, alpha, residual


def composite(canvas, sp, cx,cy):
    col,alpha,residual = sp
    x0=int(np.floor(cx))-85; y0=int(np.floor(cy))-65
    # Subpixel placement keeps slow deceleration smooth at the 256px output.
    fx,fy=cx-np.floor(cx),cy-np.floor(cy)
    mat=np.float32([[1,0,fx],[0,1,fy]])
    col=cv2.warpAffine(col,mat,(170,200),flags=cv2.INTER_LINEAR)
    alpha=cv2.warpAffine(alpha,mat,(170,200),flags=cv2.INTER_LINEAR)
    residual=cv2.warpAffine(residual,mat,(170,200),flags=cv2.INTER_LINEAR)
    xa,ya=max(x0,0),max(y0,0)

    a=alpha[sy,sx,None]
    region=canvas[ya:yb,xa:xb]
    region += residual[sy,sx]
    region[:]=region*(1-a)+col[sy,sx]*a
