+    return ndimage.shift(array, (dy, dx, 0), order=1,

+        # Alignment of consecutive cutouts avoids double edges during

+                    (dy+cb[1],dx+cb[0]),order=1,mode='constant',cval=0,prefilter=False))
+        # Preserve the exact source timeline for every unaffected pixel.
+        # Only original and edited footprints use the recovered clean plate.
+        affected = masks[t] | (support > 0.001)
+        plate = f[t].copy()
+        plate[affected] = background[affected]
+        out[t] = np.clip(np.rint(plate+foreground),0,255).astype(np.uint8)
+
+    save(out, ROOT)
+    print('Saved', out.shape, out.dtype)
+

+if __name__ == '__main__':
+    main()

\"\"\"Counterfactual edit using source RGB cutouts and measured ramp dynamics.

No renderer is used: foreground appearance, tumbling, and cast shadows are
resampled from the source. Initial velocity changes, while ramp acceleration
and gravitational fall time remain unchanged.
\"\"\"
from pathlib import Path
import numpy as np
from scipy import ndimage
from artifact_io import save

ROOT = Path(__file__).resolve().parent


def shift_rgb(array, dx, dy):
    return ndimage.shift(array, (dy, dx, 0), order=1,
                         mode='constant', cval=0, prefilter=False)


def main():
    src = np.load(ROOT / 'source_pixels.npz')['rgb']
    n, h, w, _ = src.shape
    f = src.astype(np.float32)
    # Every part of the moving object's footprint is exposed elsewhere in
    # the same video. The late stationary car is removed using early floor.
    background = np.median(f[26:90], axis=0)
    background[99:, :66] = np.median(f[:8, 99:, :66], axis=0)

    centers = []
    masks = []
    residuals = []
    for k in range(n):
        delta = f[k] - background
        d = np.max(np.abs(delta), axis=2)
        # A source-derived moving-object ROI, including its floor shadow.
        if k <= 17:
            approx_x = [173,159,146,133,121,110,101,92,84,78,71,65,60,55,49,44,39,35][k]
            approx_y = [109,104,99,95,91,87,83,80,77,75,72,70,70,71,75,81,89,100][k]
        else:
            approx_x, approx_y = 28, 121
        yy, xx = np.mgrid[:h, :w]
        roi = ((xx > approx_x-27) & (xx < approx_x+25) &
               (yy > approx_y-23) & (yy < approx_y+23))
        if 13 <= k <= 19:
            roi |= ((xx > 14) & (xx < 64) & (yy > 121))
        lab, count = ndimage.label((d > 12) & roi)
        sizes = np.bincount(lab.ravel())
        sizes[0] = 0
        keep = np.flatnonzero(sizes >= 18)
        raw = np.isin(lab, keep)
        # Close the holes from dark tires matching the dark floor, then keep
        # the small antialiased perimeter and soft source shadow intact.
        mask = ndimage.binary_fill_holes(ndimage.binary_closing(raw, iterations=1))
        mask = ndimage.binary_dilation(mask, iterations=2) & ndimage.binary_dilation(roi, iterations=1)
        masks.append(mask)
        residuals.append(delta * mask[..., None])
        # Alignment of consecutive cutouts avoids double edges during
        # fractional-time interpolation. Exclude detached floor shadows.
        if k <= 17:
            obj_labels, _ = ndimage.label(raw & (yy < approx_y+22))
            ss = np.bincount(obj_labels.ravel()); ss[0] = 0
            obj = obj_labels == ss.argmax()
        elif k == 18:
            obj = raw & (yy < 123)
        elif k == 19:
            obj = raw & (yy < 124)
        else:
            obj = raw
        y, x = np.where(obj)
        centers.append([x.mean(), y.mean()])
    centers = np.asarray(centers)
    residuals = np.asarray(residuals)
    masks = np.asarray(masks)

    # Registration of the early source car gives a constant uphill
    # acceleration of -0.909 pixels/frame"'^2 and initial vx -15.13.
    v0 = 15.13337463
    decel = 0.90892228
    v_new = 0.95*v0
    # First loss of edge support lies between source samples 10 and 11.
    edge_time = 10.7
    distance = v0*edge_time - 0.5*decel*edge_time**2
    arrival = (v_new - np.sqrt(v_new*v_new-2*decel*distance))/decel
    source_launch_v = v0-decel*edge_time
    edited_launch_v = v_new-decel*arrival
    launch_ratio = edited_launch_v/source_launch_v
    print(f'"'Initial velocity ratio .95; edge arrival {edge_time:.3f} -> {arrival:.3f} frames; airborne velocity ratio {launch_ratio:.4f}')

    # Source car horizontal position during the fall (shadow-independent).
    # Smoothing suppresses the subpixel centroid bias from changing occlusion.
    fall_x = centers[:23,0].copy()
    fall_x[18] = 31.0
    fall_x[19] = 29.0
    fall_x[20] = centers[20,0]
    fall_x[21:] = centers[22,0]
    launch_x = np.interp(edge_time, np.arange(len(fall_x)), fall_x)
    end_x = fall_x[-1]
    ground_direction_y = 0.359
    out = np.empty_like(src)

    for t in range(n):
        if t == 0:
            out[t] = src[t]
            continue
        if t <= arrival:
            travel = v_new*t - .5*decel*t*t
            # The same ramp positions correspond to a different time, not
            # a global .95 playback rate (gravity must remain unchanged).
            s = (v0-np.sqrt(max(0.,v0*v0-2*decel*travel)))/decel
            dx = dy = 0.
        else:
            s = edge_time + t-arrival
            sx = np.interp(min(s,22.), np.arange(len(fall_x)), fall_x)
            # Change the planar velocity of the airborne body. Source pose
            # is retained at equal elapsed gravitational fall time.
            dx = (1-launch_ratio)*(launch_x-sx)
            # Less horizontal momentum means less floor sliding, too.
            if s > 19:
                dx += .35*(1-launch_ratio**2)*min(1.,(s-19)/3.)
            dy = ground_direction_y*dx
        s = min(s,n-1)
        a = int(np.floor(s)); b = min(a+1,n-1); u = s-a
        center = (1-u)*centers[a] + u*centers[b]
        ca = center-centers[a]
        cb = center-centers[b]
        foreground = ((1-u)*shift_rgb(residuals[a], dx+ca[0], dy+ca[1]) +
                      u*shift_rgb(residuals[b], dx+cb[0], dy+cb[1]))
        support = ((1-u)*ndimage.shift(masks[a].astype(np.float32),
                    (dy+ca[1],dx+ca[0]),order=1,mode='constant',cval=0,prefilter=False) +
                   u*ndimage.shift(masks[b].astype(np.float32),
                    (dy+cb[1],dx+cb[0]),order=1,mode='constant',cval=0,prefilter=False))
        # Preserve the exact source timeline for every unaffected pixel.
        # Only original and edited footprints use the recovered clean plate.
        affected = masks[t] | (support > 0.001)
        plate = f[t].copy()
        plate[affected] = background[affected]
        out[t] = np.clip(np.rint(plate+foreground),0,255).astype(np.uint8)

    save(out, ROOT)
    print('Saved', out.shape, out.dtype)


if __name__ == '__main__':
    main()
