+
+The static camera and all textures come from the supplied clip.  A projective
+cuboid fitted to the first two dominoes is used only to continue the first
+one's now-unimpeded fall.  There is no external renderer or learned model.

+import cv2
+from scipy.ndimage import gaussian_filter
+from scipy.optimize import least_squares
+from scipy.integrate import solve_ivp

+H, W = 144, 256

+def world_vertices(angle):
+    e = [-.4 * np.cos(angle), .4 * np.sin(angle), 0.]
+    bottom = np.array([e, [0., 0., 0.], [0., 0., 1.], [e[0], e[1], 1.]])
+    return np.concatenate([bottom, bottom + [3*np.sin(angle), 3*np.cos(angle), 0.]])

+def fit_camera():
+    ids = [0, 1, 2, 4, 5, 6, 7]
+    first = np.array([[58.5,97.8],[68.1,100.6],[75.1,90.2],
+                      [67.2,38.7],[77.6,41.1],[84.2,35.8],[74.6,33.4]])
+    second = np.array([[103.3,101.3],[112.5,101.4],[115.,90.5],
+                       [101.3,39.],[111.2,39.],[115.1,34.5],[105.8,34.2]])
+    init = np.array([[23.,-3.,11.,68.],[0.,-21.,-10.,100.],[0.,0.,0.,1.]])
+    def unpack(p):
+        return np.r_[p[:11], 1.].reshape(3,4)

+        q = np.c_[v, np.ones(len(v))] @ P.T
+        return q[:,:2] / q[:,2,None]
+    def residual(p):
+        P=unpack(p)
+        return np.r_[(project(world_vertices(p[11])[ids], P)-first).ravel(),
+                     (project((world_vertices(0)+[p[12],0,0])[ids], P)-second).ravel()]
+    fit=least_squares(residual,np.r_[init.ravel()[:11],.14,1.65],max_nfev=2000)
+    return unpack(fit.x), float(fit.x[11])

+def project(v):

+def polygon_mask(points, scale=SS):

+    cv2.fillConvexPoly(mask,np.round(np.asarray(points)*scale*16).astype(np.int32),255,shift=4)
+    return mask.astype(np.float32)/255

+def signed_area(p):

+def paste_face(image, points, tex):

+def shadow(image, angle):

+    # Area-light shadows: a compact cast silhouette plus a wider, softer lobe.
+    shadow_sum=np.zeros((H*SS,W*SS),np.float32)
+    for dx,dz,opacity,blur in [(.10,-.33,.38,2.2),(.06,-.60,.18,5.0)]:

+    hi=polygon_mask(hull)

+
+
+def render_mesh(angle):
+    image=shadow(BGHI.copy(),angle)
+    v=world_vertices(angle)
+    # Measurement correction smoothly decreases as the body leaves its
+    # initial pose; it does not alter the final floor-contact geometry.
+    mix=max(0,1-(angle-A0)/(np.pi/2-A0))
+    points=project(v)+CORR*mix
+    # Lighting on the broad face follows its normal rotating toward the light.

+    # Preserve the source's warmer illumination on the now upward-facing face.

+    left=LEFT*(.92+.06*r)
+    right=RIGHT*(1-.03*r)
+    faces=[([0,1,5,4],front),([1,2,6,5],right),
+           ([4,5,6,7],TOP),([3,0,4,7],left)]
+    visible=[]
+    for ids,tex in faces:
+        pp=points[ids]
+        if signed_area(pp)<-.01:
+            depth=np.mean((np.c_[v[ids],np.ones(4)]@P.T)[:,2])
+            visible.append((depth,pp,tex))
+    for _,pp,tex in sorted(visible,key=lambda row:row[0],reverse=True):
+        image=paste_face(image,pp,tex)
+    return cv2.resize(image,(W,H),interpolation=cv2.INTER_AREA)
+
+
+def main():
+    frames=[]
+    for i in range(96):
+        t=i/24
+        angle=angle_at(t)
+        if i<20:

+            m=source_first_mask(S[i])[...,None]
+            image=image*(1-m)+S[i].astype(float)*m
+        else:
+            # Temporal supersampling only during the fast unsupported motion.
+            offsets=[-.17/24,0.,.17/24] if t<impact+.25 else [0.]
+            image=np.mean([render_mesh(angle_at(max(0,t+o))) for o in offsets],axis=0)
+        frames.append(np.clip(np.rint(image),0,255).astype(np.uint8))
+    frames=np.stack(frames)
+    artifact_io.save(frames,ROOT)
+    print('Saved',frames.shape,'RGB; unsupported first domino contacts floor at',round(impact,3),'s')
+    print('Unaffected third/fourth upright; final first reach:',project(world_vertices(np.pi/2))[4:].round(2).tolist())
+
+if __name__=='__main__':
+    main()
