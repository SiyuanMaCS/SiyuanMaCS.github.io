+transferred velocities are scaled together, while the actual contact locations,

+def inverse_map(flow, weight, grid):
+    """Invert a forward displacement field with fixed-point iterations."""
+    x = grid[0] - weight * flow[..., 0]
+    y = grid[1] - weight * flow[..., 1]
+    for _ in range(3):
+        sampled = cv2.remap(flow, x, y, cv2.INTER_LINEAR,
+                            borderMode=cv2.BORDER_REPLICATE)
+        x = grid[0] - weight * sampled[..., 0]
+        y = grid[1] - weight * sampled[..., 1]
+    return x, y
+

+def interpolate(a, b, forward, backward, fraction, grid):
+    xa, ya = inverse_map(forward, fraction, grid)
+    xb, yb = inverse_map(backward, 1.0 - fraction, grid)
+    wa = cv2.remap(a, xa, ya, cv2.INTER_LINEAR,
+                   borderMode=cv2.BORDER_REPLICATE).astype(np.float32)
+    wb = cv2.remap(b, xb, yb, cv2.INTER_LINEAR,
+                   borderMode=cv2.BORDER_REPLICATE).astype(np.float32)
+    return np.clip(np.rint((1.0-fraction)*wa + fraction*wb), 0, 255).astype(np.uint8)

        k = int(np.floor(phase + 1e-9))
        fraction = float(phase - k)
        if index == 0:
            out.append(source[0].copy())
            continue
        if fraction < 1e-8:
            crop = crops[k]
        else:
            if k not in flows:

                              iterations=6, poly_n=7, poly_sigma=1.5, flags=0)
                f = cv2.calcOpticalFlowFarneback(gray[k], gray[k+1], None, **params)
                r = cv2.calcOpticalFlowFarneback(gray[k+1], gray[k], None, **params)
                flows[k] = (f, r)
            f, r = flows[k]
            crop = interpolate(crops[k], crops[k+1], f, r, fraction, grid)
        frame = base.copy()
        frame[y0:y1, x0:x1] = np.clip(np.rint(
            feather * crop + (1.0-feather)*base[y0:y1, x0:x1]), 0, 255).astype(np.uint8)
        out.append(np.array(Image.fromarray(frame).resize(
            (spec['width'], spec['height']), Image.Resampling.LANCZOS)))
        if index % 24 == 0:
            print(f'Rendered {index+1}/{n}: physical source phase {phase:.2f}', flush=True)
    frames = np.stack(out).astype(np.uint8)
    # Save every constructed candidate immediately, before validation/encoding.
    artifact_io.save(frames, ROOT)
    np.savez_compressed(ROOT / 'edited_rgb.npz', rgb=frames)
    # Provide a standard, silent MP4 in addition to the required RGB artifacts.
    if shutil.which('ffmpeg'):

               str(ROOT / 'edited_video.mp4')]
        subprocess.run(cmd, input=frames.tobytes(), check=True)
    print('Saved 96 RGB frames at 256x144, 24 fps.', flush=True)
    return frames
