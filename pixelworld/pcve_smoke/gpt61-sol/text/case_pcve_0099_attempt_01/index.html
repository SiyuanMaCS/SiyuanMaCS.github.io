+"""Source-pixel counterfactual: the incoming tennis ball has fivefold mass.
+
+The source's free motion and unchanged set are kept. A small impulse/contact
+model handles the changed impact; only the two balls and their floor shadows
+are composited afterwards. No learned generation or external renderer is used.
+"""
+from pathlib import Path
+import numpy as np
+import cv2
+from PIL import Image
+from scipy.ndimage import binary_closing, binary_fill_holes, gaussian_filter
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+SOURCE = np.load(ROOT / 'source_pixels.npz')['rgb']
+W, H, SCALE = 256, 144, 5
+
+def changed_motion():
+    # Units are downsampled pixels and source-frame intervals. These forces
+    # retain the measured freefall and ground rolling losses from the source.
+    radius_sum, restitution = 10.4, .37
+    floor_restitution = .48
+    rolling_drag, rolling_loss = .13634493, .11178125
+    gravity, floor_friction = 2.4, .01
+    masses = np.array([5., 1.])
+    xy = np.array([[78.7957, 76.8602], [66.2133, 93.9067]])
+    velocity = np.array([[-5.9, 15.4], [0., 0.]])
+    dt = .005
+    result = []
+    for step in range(75*200 + 1):
+        if step % 200 == 0:
+            result.append(xy.copy())
+        floor = 94.66-.0105*xy[:,0]
+        heights = np.maximum(0., floor-xy[:,1])
+        near = np.exp(-heights/.75)
+        # Coulomb and rolling losses do not decrease simply because mass grows.
+        decel = (.014+(rolling_drag-.014)*near)*np.abs(velocity[:,0]) + rolling_loss*near
+        velocity[:,0] = np.sign(velocity[:,0])*np.maximum(0., np.abs(velocity[:,0])-decel*dt)
+        velocity[:,1] += gravity*dt
+        xy += velocity*dt
+        floor = 94.66-.0105*xy[:,0]
+        for k in range(2):
+            if xy[k,1] > floor[k]:
+                xy[k,1] = floor[k]
+                if velocity[k,1] > 0:
+                    v = velocity[k,0]
+                    velocity[k,0] = np.sign(v)*max(0., abs(v)-floor_friction*(1+floor_restitution)*velocity[k,1])
+                    velocity[k,1] = -floor_restitution*velocity[k,1] if velocity[k,1] > 1.2 else 0.
+        d = xy[0]-xy[1]
+        length = np.linalg.norm(d)
+        if length < radius_sum:
+            n = d/max(length, 1.e-9)
+            closing = np.dot(velocity[0]-velocity[1], n)
+            constrained = floor[1]-xy[1,1]<.2 and n[1]<0
+            inverse_effective_mass = 1/masses[0] + (n[0]**2 if constrained else 1.)/masses[1]
+            if closing < 0:
+                impulse = -(1+restitution)*closing/inverse_effective_mass
+                velocity[0] += impulse*n/masses[0]
+                velocity[1] -= impulse*n/masses[1]
+                if constrained:
+                    velocity[1,1] = 0.
+            xy[0] += n*(radius_sum-length)*.5
+            xy[1] -= n*(radius_sum-length)*.5
+            if constrained:
+                xy[1,1] = floor[1]
+    return np.array(result)
+
+# We use decoded 5x source crops for antialiased texture transport. Source RGB
+# frames were made with Lanczos downsampling, which is used on our edits too.
+video = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+CACHE = {}
+def high_frame(t):
+    if t not in CACHE:
+        video.set(cv2.CAP_PROP_POS_FRAMES, int(t))
+        ok, bgr = video.read()
+        if not ok:
+            raise RuntimeError('Could not decode source frame '+str(t))
+        CACHE[t] = bgr[...,::-1].copy()
+    return CACHE[t]
+
+
+def ball_sprite(t, guess):
+    f = high_frame(t)
+    cx, cy = guess
+    half = 7
+    left, top = int((cx-half)*SCALE), int((cy-half)*SCALE)
+    crop = f[top:top+half*2*SCALE,left:left+half*2*SCALE].astype(float)
+    r,g,b = crop.transpose(2,0,1)
+    mask = (g>r+5)&(g>b+20)
+    mask = binary_fill_holes(binary_closing(mask, iterations=2))
+    contours = cv2.findContours(mask.astype(np.uint8),cv2.RETR_EXTERNAL,cv2.CHAIN_APPROX_NONE)[0]
+    co = max(contours,key=cv2.contourArea)
+    (x,y),(ew,eh),angle = cv2.fitEllipse(co)
+    yy,xx = np.mgrid[:crop.shape[0],:crop.shape[1]]
+    a = np.deg2rad(angle)
+    u = (xx-x)*np.cos(a)+(yy-y)*np.sin(a)
+    v = -(xx-x)*np.sin(a)+(yy-y)*np.cos(a)
+    # One source-sized sphere, unchanged by its new mass.
+    rho = np.sqrt((u/(ew/2+.25))**2+(v/(eh/2+.25))**2)
+    alpha = np.clip((1-rho)*min(ew,eh)/2+.5,0,1)
+    return crop, alpha, ((left+x)/SCALE, (top+y)/SCALE)
+

+NEW = [ball_sprite(t,guess) for t,guess in [
+    (22,(81.,92.)), (23,(84.8,89.6)), (24,(88.3,89.)),
+    (25,(92.,90.6)), (26,(95.8,93.4)), (27,(98.,91.2)),

+    (34,(109.8,93.8)), (35,(110.2,93.4)), (36,(110.3,93.7))]]

+# Counterfactual region lies on the plain, stationary foreground floor. Build
+# its clean plate from source moments where each location is unoccupied.
+plate = high_frame(0).copy()
+plate[82*SCALE:105*SCALE,54*SCALE:81*SCALE] = high_frame(36)[82*SCALE:105*SCALE,54*SCALE:81*SCALE]
+Y, X = np.mgrid[:H*SCALE,:W*SCALE].astype(np.float32)
+X /= SCALE
+Y /= SCALE

+def add_shadow(canvas, center, height, old=False):
+    x,y = center
+    ground = 94.66-.0105*x+.42
+    # Same oblique source lighting; airborne shadows separate and soften.
+    cx = x+4.0+.30*height
+    cy = ground+4.0+.025*height
+    rx, ry = 6.8+.06*height, 2.35+.02*height
+    q = ((X-cx)/rx)**2+((Y-cy)/ry)**2
+    opacity = .35*np.exp(-.95*q)*np.exp(-height/55.)
+    opacity[q>7.] = 0
+    canvas *= 1-opacity[...,None]
+
+
+def paint_ball(canvas, sprite, dst):
+    crop, alpha, source_center = sprite
+    # Place the ellipse's texture at its fitted center, subpixel accurately.
+    cx,cy = source_center
+    # Crop origin can be reconstructed from fitted sphere coordinates and
+    # the alpha ellipse center, avoiding source rounding offsets.
+    yy,xx = np.mgrid[:alpha.shape[0],:alpha.shape[1]]
+    sx = np.sum(xx*alpha)/np.sum(alpha)
+    sy = np.sum(yy*alpha)/np.sum(alpha)
+    tx,ty = dst[0]*SCALE-sx, dst[1]*SCALE-sy
+    mat = np.float32([[1,0,tx],[0,1,ty]])
+    premult = cv2.warpAffine((crop*alpha[...,None]).astype(np.float32),mat,(W*SCALE,H*SCALE),flags=cv2.INTER_LINEAR)
+    opacity = cv2.warpAffine(alpha.astype(np.float32),mat,(W*SCALE,H*SCALE),flags=cv2.INTER_LINEAR)
+    canvas *= 1-opacity[...,None]
+    canvas += premult
+

+def main():
+    motion = changed_motion()
+    frames = SOURCE.copy()
+    # Mass itself has no visible marking. Equal gravity and unchanged initial
+    # velocity make frames preceding the first contact identical to the source.
+    for t in range(21,96):
+        canvas = high_frame(t).astype(np.float32).copy()
+        # Replace the entire original balls' swept path and their cast shadows,
+        # leaving the table, support, room and all other source pixels intact.
+        canvas[82*SCALE:107*SCALE,:128*SCALE] = plate[82*SCALE:107*SCALE,:128*SCALE]
+        pair = motion[t-20] + np.array([.42,.42])
+        for k in (1,0):
+            height = max(0.,94.66-.0105*(pair[k,0]-.42)+.42-pair[k,1])
+            add_shadow(canvas,pair[k],height,old=k==1)
+        # Use successive factual orientations during the heavier ball's short
+        # leftward roll. The ball's material color/diameter are unchanged.
+        index = min(len(NEW)-1,max(0,int((t-21)*.9)))
+        if t >= 37:
+            index = len(NEW)-1
+        if pair[1,0] > -9:
+            paint_ball(canvas,OLD,pair[1])
+        paint_ball(canvas,NEW[index],pair[0])
+        edited = np.array(Image.fromarray(np.clip(canvas,0,255).astype(np.uint8)).resize((W,H),Image.Resampling.LANCZOS))
+        # Preserve source byte-for-byte outside the foreground edit.
+        frames[t,81:108,:129] = edited[81:108,:129]
+    artifact_io.save(frames,ROOT)
+    np.save(ROOT / '.tmp' / 'counterfactual_motion.npy',motion)
+    print('Saved',frames.shape,'RGB frames. Final ball positions:',motion[-1].tolist())
+
+if __name__ == '__main__':
+    main()

\"\"\"Source-pixel counterfactual: the incoming tennis ball has fivefold mass.

The source's free motion and unchanged set are kept. A small impulse/contact
model handles the changed impact; only the two balls and their floor shadows
are composited afterwards. No learned generation or external renderer is used.

import cv2
from PIL import Image
from scipy.ndimage import binary_closing, binary_fill_holes, gaussian_filter

def changed_motion():
    # Units are downsampled pixels and source-frame intervals. These forces
    # retain the measured freefall and ground rolling losses from the source.
    radius_sum, restitution = 10.4, .37
    floor_restitution = .48
    rolling_drag, rolling_loss = .13634493, .11178125
    gravity, floor_friction = 2.4, .01
    masses = np.array([5., 1.])
    xy = np.array([[78.7957, 76.8602], [66.2133, 93.9067]])
    velocity = np.array([[-5.9, 15.4], [0., 0.]])
    dt = .005
    result = []
    for step in range(75*200 + 1):
        if step % 200 == 0:
            result.append(xy.copy())
        floor = 94.66-.0105*xy[:,0]
        heights = np.maximum(0., floor-xy[:,1])
        near = np.exp(-heights/.75)
        # Coulomb and rolling losses do not decrease simply because mass grows.
        decel = (.014+(rolling_drag-.014)*near)*np.abs(velocity[:,0]) + rolling_loss*near
        velocity[:,0] = np.sign(velocity[:,0])*np.maximum(0., np.abs(velocity[:,0])-decel*dt)
        velocity[:,1] += gravity*dt
        xy += velocity*dt
        floor = 94.66-.0105*xy[:,0]
        for k in range(2):
            if xy[k,1] > floor[k]:
                xy[k,1] = floor[k]
                if velocity[k,1] > 0:
                    v = velocity[k,0]
                    velocity[k,0] = np.sign(v)*max(0., abs(v)-floor_friction*(1+floor_restitution)*velocity[k,1])
                    velocity[k,1] = -floor_restitution*velocity[k,1] if velocity[k,1] > 1.2 else 0.
        d = xy[0]-xy[1]
        length = np.linalg.norm(d)
        if length < radius_sum:
            n = d/max(length, 1.e-9)
            closing = np.dot(velocity[0]-velocity[1], n)
            constrained = floor[1]-xy[1,1]<.2 and n[1]<0
            inverse_effective_mass = 1/masses[0] + (n[0]**2 if constrained else 1.)/masses[1]
            if closing < 0:
                impulse = -(1+restitution)*closing/inverse_effective_mass
                velocity[0] += impulse*n/masses[0]
                velocity[1] -= impulse*n/masses[1]
                if constrained:
                    velocity[1,1] = 0.
            xy[0] += n*(radius_sum-length)*.5
            xy[1] -= n*(radius_sum-length)*.5
            if constrained:
                xy[1,1] = floor[1]
