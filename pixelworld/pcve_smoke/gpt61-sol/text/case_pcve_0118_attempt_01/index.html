+"""Counterfactual rigid-body edit, using only the supplied RGB timeline.
+
+The ball has the same dimensions and texture, but twenty times the inertia.
+The initial car approach is unchanged. At contact, the finite car impulse is
+shared with the much heavier ball; it can no longer push the ball to the edge.
+All sprite radiance and contact shadows are extracted from source pixels.
+"""
+from pathlib import Path
+import numpy as np
+from scipy.ndimage import map_coordinates
+from artifact_io import save
+
+ROOT = Path(__file__).resolve().parent
+
+
+def shifted(field, dx):
+    y, x = np.indices(field.shape[:2], dtype=np.float64)
+    return np.stack([
+        map_coordinates(field[..., c], [y, x-dx], order=1,
+                        mode='constant', cval=0.0, prefilter=False)
+        for c in range(3)
+    ], axis=-1)
+
+
+def main():
+    source = np.load(ROOT / 'source_pixels.npz')['rgb']
+    rgb = source.astype(np.float64)
+    initial = rgb[0]
+    contact = 20
+
+    # Reconstruct the local clean plate. The initial car is absent from this
+    # part of the source at contact. Use those pixels, not a painted table.
+    plate = initial.copy()
+    plate[24:40, 183:225] = np.median(rgb[35:, 24:40, 183:225], axis=0)
+
+    # The ball's background is exposed once it has left the tabletop. The
+    # stationary source car obscures a few columns on the lower right; bridge
+    # just those columns between the two actually exposed background samples.
+    exposed = np.median(rgb[45:], axis=0)
+    ball_clean = initial.copy()
+    ball_clean[18:38, 124:149] = exposed[18:38, 124:149]
+    for y in range(28, 38):
+        for x in range(138,149):
+            fraction = (x-137) / (149-137)
+            ball_clean[y,x] = (1-fraction)*exposed[y,137] + fraction*initial[y,149]
+
+    # Restrict the radiance residual to the local object and its own shadow.
+    # Soft margins prevent compression noise at a cut boundary from appearing
+    # as a box, while preserving the original silhouette's antialiasing.
+    by, bx = np.indices((144,256))
+    ball_window = np.zeros((144,256), float)
+    ball_window[19:37,126:148] = 1.0
+    ball_window[19,126:148] = .5
+    ball_window[36,126:148] = .5
+    ball_window[19:37,126] *= .5
+    ball_window[19:37,147] *= .5
+    ball_delta = (initial-ball_clean)*ball_window[...,None]
+    plate -= ball_delta
+    ball_scene = plate + ball_delta
+
+    # The car at contact is a clean native-pixel sprite including its wheels
+    # and contact shadow. Before contact, the ball is still at its initial pose.
+    car_window = np.zeros((144,256), float)
+    car_window[25:40,142:180] = 1.0
+    car_window[25,142:180] = .25
+    car_window[39,142:180] = .25
+    car_window[25:40,142] = .5
+    car_window[25:40,179] = .5
+    car_delta = (rgb[contact]-ball_scene)*car_window[...,None]
+
+    # Precontact registration gives an incoming speed of 1.27 pixels/frame.
+    # The visible ball's effective original inertia is small relative to the
+    # car. Multiplying it by twenty produces a ball/car mass ratio of four.
+    # A damped plastic/rubber impact (e=0.25) arrests the car. Rolling resistance
+    # dissipates the ball's small transferred velocity before the table edge.
+    car_mass, original_ball_mass = 1.0, 0.2
+    ball_mass = 20.0 * original_ball_mass
+    e, incoming = .25, -1.27
+    ball_velocity = (1+e)*car_mass/(car_mass+ball_mass)*incoming
+    car_velocity = (car_mass-e*ball_mass)/(car_mass+ball_mass)*incoming
+    rolling_drag, car_drag = .052, .098
+
+    result = source.copy()
+    ball_x, car_x = 0.0, 0.0
+    substeps = 12
+    for frame in range(contact+1,96):
+        for _ in range(substeps):
+            step = 1/substeps
+            old_v = ball_velocity
+            ball_velocity = np.sign(old_v)*max(0,abs(old_v)-rolling_drag*step)
+            ball_x += .5*(old_v+ball_velocity)*step
+            old_v = car_velocity
+            car_velocity = np.sign(old_v)*max(0,abs(old_v)-car_drag*step)
+            car_x += .5*(old_v+car_velocity)*step
+        scene = plate + shifted(ball_delta,ball_x) + shifted(car_delta,car_x)
+
+        # Preserve the actual current source everywhere except the causal
+        # region. This keeps the potted plant, room, table, and floor identical
+        # and removes the factual falling ball and its later ground shadow.
+        out = rgb[frame].copy()
+        out[17:41,123:225] = scene[17:41,123:225]
+        # The source falling/rolling ball and shadow stay inside this corridor;
+        # here the initial image is a complete clean, unobstructed background.
+        out[41:128,68:132] = initial[41:128,68:132]
+        result[frame] = np.clip(np.rint(out),0,255).astype(np.uint8)
+
+    save(result, ROOT)
+    print(f'Saved {result.shape[0]} RGB frames, 256x144, 24 fps.')
+    print(f'Ball mass factor=20; final displacement={ball_x:.3f}px; car displacement={car_x:.3f}px')
+
+
+if __name__ == '__main__':
+    main()

\"\"\"Counterfactual rigid-body edit, using only the supplied RGB timeline.

The ball has the same dimensions and texture, but twenty times the inertia.
The initial car approach is unchanged. At contact, the finite car impulse is
shared with the much heavier ball; it can no longer push the ball to the edge.
All sprite radiance and contact shadows are extracted from source pixels.

from pathlib import Path
import numpy as np
from scipy.ndimage import map_coordinates
from artifact_io import save

ROOT = Path(__file__).resolve().parent


def shifted(field, dx):
    y, x = np.indices(field.shape[:2], dtype=np.float64)
    return np.stack([
        map_coordinates(field[..., c], [y, x-dx], order=1,
                        mode='constant', cval=0.0, prefilter=False)
        for c in range(3)
    ], axis=-1)


def main():
    source = np.load(ROOT / 'source_pixels.npz')['rgb']
    rgb = source.astype(np.float64)
    initial = rgb[0]
    contact = 20

    # Reconstruct the local clean plate. The initial car is absent from this
    # part of the source at contact. Use those pixels, not a painted table.
    plate = initial.copy()
    plate[24:40, 183:225] = np.median(rgb[35:, 24:40, 183:225], axis=0)

    # The ball's background is exposed once it has left the tabletop. The
    # stationary source car obscures a few columns on the lower right; bridge
    # just those columns between the two actually exposed background samples.
    exposed = np.median(rgb[45:], axis=0)
    ball_clean = initial.copy()
    ball_clean[18:38, 124:149] = exposed[18:38, 124:149]
    for y in range(28, 38):
        for x in range(138,149):
            fraction = (x-137) / (149-137)
            ball_clean[y,x] = (1-fraction)*exposed[y,137] + fraction*initial[y,149]

    # Restrict the radiance residual to the local object and its own shadow.
    # Soft margins prevent compression noise at a cut boundary from appearing
    # as a box, while preserving the original silhouette's antialiasing.
    by, bx = np.indices((144,256))
    ball_window = np.zeros((144,256), float)
    ball_window[19:37,126:148] = 1.0
    ball_window[19,126:148] = .5
    ball_window[36,126:148] = .5
    ball_window[19:37,126] *= .5
    ball_window[19:37,147] *= .5
    ball_delta = (initial-ball_clean)*ball_window[...,None]
    plate -= ball_delta
    ball_scene = plate + ball_delta

    # The car at contact is a clean native-pixel sprite including its wheels
    # and contact shadow. Before contact, the ball is still at its initial pose.
    car_window = np.zeros((144,256), float)
    car_window[25:40,142:180] = 1.0
    car_window[25,142:180] = .25
    car_window[39,142:180] = .25
    car_window[25:40,142] = .5
    car_window[25:40,179] = .5
    car_delta = (rgb[contact]-ball_scene)*car_window[...,None]

    # Precontact registration gives an incoming speed of 1.27 pixels/frame.
    # The visible ball's effective original inertia is small relative to the
    # car. Multiplying it by twenty produces a ball/car mass ratio of four.
    # A damped plastic/rubber impact (e=0.25) arrests the car. Rolling resistance
    # dissipates the ball's small transferred velocity before the table edge.
    car_mass, original_ball_mass = 1.0, 0.2
    ball_mass = 20.0 * original_ball_mass
    e, incoming = .25, -1.27
    ball_velocity = (1+e)*car_mass/(car_mass+ball_mass)*incoming
    car_velocity = (car_mass-e*ball_mass)/(car_mass+ball_mass)*incoming
    rolling_drag, car_drag = .052, .098

    result = source.copy()
    ball_x, car_x = 0.0, 0.0
    substeps = 12
    for frame in range(contact+1,96):
        for _ in range(substeps):
            step = 1/substeps
            old_v = ball_velocity
            ball_velocity = np.sign(old_v)*max(0,abs(old_v)-rolling_drag*step)
            ball_x += .5*(old_v+ball_velocity)*step
            old_v = car_velocity
            car_velocity = np.sign(old_v)*max(0,abs(old_v)-car_drag*step)
            car_x += .5*(old_v+car_velocity)*step
        scene = plate + shifted(ball_delta,ball_x) + shifted(car_delta,car_x)

        # Preserve the actual current source everywhere except the causal
        # region. This keeps the potted plant, room, table, and floor identical
        # and removes the factual falling ball and its later ground shadow.
        out = rgb[frame].copy()
        out[17:41,123:225] = scene[17:41,123:225]
        # The source falling/rolling ball and shadow stay inside this corridor;
        # here the initial image is a complete clean, unobstructed background.
        out[41:128,68:132] = initial[41:128,68:132]
        result[frame] = np.clip(np.rint(out),0,255).astype(np.uint8)

    save(result, ROOT)
    print(f'Saved {result.shape[0]} RGB frames, 256x144, 24 fps.')
    print(f'Ball mass factor=20; final displacement={ball_x:.3f}px; car displacement={car_x:.3f}px')

    main()
