+
+All scene geometry, object appearance and motion calibration come from the supplied
+video.  Gravity is unchanged: reducing the initial uphill speed reduces the climb
+height quadratically, rather than merely slowing down the factual fall.

+from pathlib import Path
+import numpy as np
+import cv2
+from scipy import ndimage
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+
+def smoothstep(x):
+    x = np.clip(x, 0., 1.)
+    return x*x*(3.-2.*x)
+
+
+def sprite(image, background, bounds):
+    x0, y0, x1, y1 = bounds
+    crop = image[y0:y1, x0:x1].astype(np.float32)
+    back = background[y0:y1, x0:x1]
+    difference = np.max(np.abs(crop-back), axis=2)
+    # Only the largest foreground component is the car; retain its contact shadow.
+    seed = difference > 13
+    seed = ndimage.binary_closing(seed, iterations=1)
+    labels, count = ndimage.label(seed)
+    if count:
+        sizes = ndimage.sum(seed, labels, np.arange(1, count+1))
+        seed = labels == (np.argmax(sizes)+1)
+    seed = ndimage.binary_fill_holes(seed)
+    near = ndimage.binary_dilation(seed, iterations=2)
+    seed |= near & (difference > 7)
+    seed = ndimage.binary_fill_holes(seed)
+    # Source boundary pixels already carry antialiasing. Do not blur the texture.
+    alpha = seed.astype(np.float32)
+    return crop*alpha[..., None], alpha, (x0,y0)
+
+
+def place(item, center, original_center, shear=0., scale=1.):
+    color, alpha, origin = item
+    ox, oy = origin
+    cx, cy = original_center
+    tx, ty = center
+    # Vertical shear changes projected pitch but keeps the roof's verticals vertical.
+    matrix = np.float32([[scale, 0, tx + scale*(ox-cx)],
+                         [-shear, 1, ty + oy-cy-shear*(ox-cx)]])
+    p = cv2.warpAffine(color, matrix, (256,144), flags=cv2.INTER_LINEAR,
+                       borderMode=cv2.BORDER_CONSTANT)
+    a = cv2.warpAffine(alpha, matrix, (256,144), flags=cv2.INTER_LINEAR,
+                       borderMode=cv2.BORDER_CONSTANT)
+    return p, a
+

    dys = np.array([0., -5.110, -10.105, -14.850, -19.115])
    center0 = np.array([172.5,109.0])
    refs = []
    for i, (dx,dy) in enumerate(zip(deltas,dys)):
        refs.append(sprite(factual[i], clean,
                           (int(148+dx),int(97+dy),int(198+dx),int(125+dy))))
    ground = sprite(factual[25], clean, (4,105,54,136))
    ground_center = np.array([28.5,120.5])

    # Measured ramp rise in the projection. The lower edge spans the wheelbase:
    # rear wheels leave first and the car levels onto the table as front wheels exit.
    ramp_slope = .3608
    initial_v = -15.03422619 * .6
    slope_acceleration = .8779762
    position = 0.
    velocity = initial_v
    result = []
    positions = []
    for i in range(96):
        x = center0[0]+position
        pitch = float(smoothstep((position-7.)/27.))
        y_ramp = center0[1]+ramp_slope*position
        y = (1-pitch)*y_ramp + pitch*ground_center[1]
        center = (x,y)
        if position <= 0:
            # Sample the actual uphill poses by spatial progress, not source time.
            # Reversing this sampling also correctly reverses the wheel movement.
            phase = float(np.interp(-position, -deltas, np.arange(len(deltas))))
            j = min(int(phase), len(refs)-2)
            blend = phase-j
            p0,a0 = place(refs[j],center,center0+[deltas[j],dys[j]])
            p1,a1 = place(refs[j+1],center,center0+[deltas[j+1],dys[j+1]])
            p = p0*(1-blend)+p1*blend
            a = a0*(1-blend)+a1*blend
        else:
            # Use the source car for the entire rollback, then its actual flat-table
            # pose. Bidirectional pitch warps keep the short transition coherent.
            p0,a0 = place(refs[0],center,center0,
                           shear=.265*pitch,scale=1+.045*pitch)
            p1,a1 = place(ground,center,ground_center,
                           shear=-.265*(1-pitch),scale=1-.045*(1-pitch))
            p = p0*(1-pitch)+p1*pitch
            a = a0*(1-pitch)+a1*pitch
        frame = np.clip(clean*(1-a[...,None])+p,0,255).round().astype(np.uint8)
        if i == 0:
            frame = factual[0].copy()
        result.append(frame)
        positions.append((position,velocity,pitch,x,y))
        # Small substeps retain the exact 0.6 initial speed and avoid discretization
        # error at the turnaround and transition from slope to horizontal tabletop.
        for _ in range(100):
            q = float(smoothstep((position-7.)/27.))
            # Once on the table, weak rolling friction replaces gravity down-slope.
            accel = slope_acceleration*(1-q) - .045*q*np.sign(velocity)
            position += velocity*.01 + .5*accel*.0001
            velocity += accel*.01
    frames = np.stack(result)
    artifact_io.save(frames, ROOT)
    # Numeric audit is reproducible and does not depend on any visual tools.
    positions = np.array(positions)
    np.savez_compressed(ROOT/'"'motion_audit.npz', states=positions,
                        frames_rgb_shape=np.array(frames.shape))
    print('Saved',frames.shape,frames.dtype)
    print('Initial speed ratio:',initial_v/(-15.03422619))
    print('Turnaround frame:',int(np.argmin(positions[:,0])),
          'climb displacement:',round(float(positions[:,0].min()),3))
    for k in [0,1,5,10,15,20,23,25,28,32,95]:
        print(k, np.round(positions[k],3).tolist())


if __name__ == '__main__':
    main()
