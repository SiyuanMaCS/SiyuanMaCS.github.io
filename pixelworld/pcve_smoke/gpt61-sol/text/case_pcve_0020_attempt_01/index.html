    return np.array([np.interp(t,key_t,key_x),np.interp(t,key_t,key_y),np.interp(t,key_t,key_r)])

def red_layer(t):
    src=frame_at(t)
    cx,cy,r=red_pose(t)
    alpha=disk(cx,cy,r)
    # Fill only pixels that the original foreground football had occluded.
    qx=(X-cx)/r; qy=(Y-cy)/r
    underside=sample(ref,ref_pose[0]+qx*ref_pose[2],
                         ref_pose[1]+qy*ref_pose[2])
    # The source ball is lit differently as it moves forward: retain the local
    # source illumination while borrowing only the inaccessible underside.
    underside += np.array([5,9,11])
    covered=foot_alpha
    tex=blend(src,underside,covered)
    # Actual dark residual retains source contact/reflection appearance.
    residual=np.minimum(src-F0,0)
    shadow_support=np.exp(-(((X-(cx-3))/(r*1.0))**2+
                           ((Y-(cy+r+5))/(r*.8))**2)*2)
    shadow_support*=(1-foot_alpha)*(1-alpha)
    residual*=shadow_support[...,None]
    return tex,alpha,residual

def heavy_ball(dst, dx, dy):
    # The original impulse would have displaced the football about 30 pixels.
    # With 10x inertia, the much smaller impulse-induced speed is extinguished
    # quickly by the unchanged floor friction.  The size/material do not change.
    sf=1+dy*.006
    cx,cy=FC+[dx,dy]
    a=disk(cx,cy,FR*sf)
    roll=dx/FR*.65
    cr,sr=np.cos(roll),np.sin(roll)
    xx=(X-cx)/sf; yy=(Y-cy)/sf
    sx=FC[0]+cr*xx+sr*yy
    sy=FC[1]-sr*xx+cr*yy
    texture=sample(F0,sx,sy)
    shadow=sample(foot_shadow,X-dx,Y-dy)
    dst=np.clip(dst+shadow,0,255)
    return blend(dst,texture,a)

def generate():
    out=SOURCE.copy()
    # Momentum for a 10:1 mass ratio and e~0.7 reverses the lighter ball's
    # normal velocity. Friction then brings it gently to rest. Source poses
    # trace the corresponding backward path and retain perspective lighting.
    collision=22
    final_tau=14.05
    duration=24.0
    roi=(X>=63)&(X<=132)&(Y>=49)
    feather=np.clip(np.minimum(X-61,134-X)/2,0,1)*np.clip((Y-47)/2,0,1)
    feather*=roi
    for t in range(collision+1,N):
        elapsed=t-collision
        s=min(elapsed/duration,1.)
        # Constant rolling resistance: decelerating reverse travel.
        progress=2*s-s*s
        tau=collision-(collision-final_tau)*progress
        tex,alpha,shadow=red_layer(tau)
        scene=np.clip(clean+shadow,0,255)
        scene=blend(scene,tex,alpha)
        # Small, mass-dependent recoil of the heavy football. No mass label,
        # enlargement, or change in gravitational acceleration is introduced.
        q=min(elapsed/13.,1.)
        displacement=2*q-q*q
        scene=heavy_ball(scene,.28*displacement,1.45*displacement)
        out[t]=np.clip(blend(SOURCE[t].astype(float),scene,feather),0,255).round().astype(np.uint8)
    return out
