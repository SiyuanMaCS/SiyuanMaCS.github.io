+
+The straw-bearing cup receives a fivefold mass multiplier at t=0.  Its
+geometry, material and initial state are unchanged.  Contact with the can
+has the same low restitution as the source; sliding friction is mass
+independent.  The diminished cup motion never reaches the glass bottle.

+ROOT = Path(__file__).resolve().parent

+def read_video():
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    imgs = []
+    while True:
+        ok, bgr = cap.read()
+        if not ok:
+            break
+        imgs.append(cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB))
+    cap.release()
+    return imgs

+def polygon_mask(shape, vertices):
+    m = np.zeros(shape, np.uint8)
+    cv2.fillPoly(m, [np.round(np.array(vertices)*5).astype(np.int32)], 255)
+    return m.astype(np.float32)/255

+def alpha_outline(rgb, background, envelope, inner):
+    """RGB-difference edge matting, with geometric filling of white interiors."""
+    delta = np.max(np.abs(rgb.astype(float)-background.astype(float)), axis=2)
+    seed = (delta > 11) & (envelope > 0)
+    seed |= inner > 0
+    seed = ndimage.binary_closing(seed, iterations=2)
+    seed = ndimage.binary_fill_holes(seed)
+    labels, n = ndimage.label(seed)
+    areas = ndimage.sum(seed, labels, range(1, n+1))
+    if n:
+        seed = labels == (int(np.argmax(areas))+1)
+    # Fine source-resolution edge only: retain the antialiasing in the RGB.
+    alpha = cv2.GaussianBlur(seed.astype(np.float32), (3, 3), .45)
+    return np.minimum(alpha, envelope)
+
+
+def sprite(rgb, alpha, bounds):
+    x0,y0,x1,y1 = [int(round(v*5)) for v in bounds]
+    a = alpha[y0:y1, x0:x1].copy()
+    tex = rgb[y0:y1, x0:x1].astype(np.float32)
+    return (tex*a[...,None], a, x0, y0)
+
+
+def composite(canvas, spr, dx, dy=0):
+    premul, alpha, x0, y0 = spr
+    h,w = alpha.shape
+    tx = x0 + dx*5
+    ty = y0 + dy*5
+    ix,iy = int(np.floor(tx)),int(np.floor(ty))
+    matrix = np.array([[1,0,tx-ix+2],[0,1,ty-iy+2]], np.float32)
+    aa = cv2.warpAffine(alpha, matrix, (w+4,h+4), flags=cv2.INTER_LINEAR)
+    pp = cv2.warpAffine(premul, matrix, (w+4,h+4), flags=cv2.INTER_LINEAR)
+    ix-=2; iy-=2
+    lx=max(0,ix); ly=max(0,iy)
+    rx=min(canvas.shape[1],ix+w+4); ry=min(canvas.shape[0],iy+h+4)
+    a=aa[ly-iy:ry-iy,lx-ix:rx-ix,None]
+    p=pp[ly-iy:ry-iy,lx-ix:rx-ix]
+    canvas[ly:ry,lx:rx] = canvas[ly:ry,lx:rx]*(1-a)+p
+
+
+def shadow_sprite(rgb, bg, bounds, body_alpha):
+    x0,y0,x1,y1 = [int(round(v*5)) for v in bounds]
+    src=rgb[y0:y1,x0:x1].astype(np.float32)
+    clean=bg[y0:y1,x0:x1].astype(np.float32)
+    # Ground-contact shadows are transported as transmission, not pasted wood.
+    trans = np.clip(np.mean(src/(clean+1e-3),axis=2), .65, 1)
+    yy=np.arange(y0,y1)[:,None]/5
+    ground=np.clip((yy-80.9)/.8,0,1)*np.clip((83.5-yy)/.7,0,1)
+    strength=(1-trans)*ground
+    strength*=1-body_alpha[y0:y1,x0:x1]
+    strength=cv2.GaussianBlur(strength,(5,5),.65)
+    return strength, x0, y0
+
+
+def cast_shadow(canvas, spr, dx):
+    s,x0,y0=spr
+    h,w=s.shape
+    tx=x0+dx*5; ix=int(np.floor(tx))
+    matrix=np.array([[1,0,tx-ix+2],[0,1,0]],np.float32)
+    a=cv2.warpAffine(s,matrix,(w+4,h),flags=cv2.INTER_LINEAR)
+    ix-=2
+    lo=max(0,ix);hi=min(canvas.shape[1],ix+w+4)
+    canvas[y0:y0+h,lo:hi]*=1-a[:,lo-ix:hi-ix,None]
+

+        'incoming_pixels_per_frame':float(impact_speed),

+
+if __name__=='__main__':
+    main()

def polygon_mask(shape, vertices):
    m = np.zeros(shape, np.uint8)
    cv2.fillPoly(m, [np.round(np.array(vertices)*5).astype(np.int32)], 255)
    return m.astype(np.float32)/255


def alpha_outline(rgb, background, envelope, inner):
    \"\"\"RGB-difference edge matting, with geometric filling of white interiors.\"\"\"
    delta = np.max(np.abs(rgb.astype(float)-background.astype(float)), axis=2)
    seed = (delta > 11) & (envelope > 0)
    seed |= inner > 0
    seed = ndimage.binary_closing(seed, iterations=2)
    seed = ndimage.binary_fill_holes(seed)
    labels, n = ndimage.label(seed)
    areas = ndimage.sum(seed, labels, range(1, n+1))
    if n:
        seed = labels == (int(np.argmax(areas))+1)
    # Fine source-resolution edge only: retain the antialiasing in the RGB.
    alpha = cv2.GaussianBlur(seed.astype(np.float32), (3, 3), .45)
    return np.minimum(alpha, envelope)


def sprite(rgb, alpha, bounds):
    x0,y0,x1,y1 = [int(round(v*5)) for v in bounds]
    a = alpha[y0:y1, x0:x1].copy()
    tex = rgb[y0:y1, x0:x1].astype(np.float32)
    return (tex*a[...,None], a, x0, y0)


def composite(canvas, spr, dx, dy=0):
    premul, alpha, x0, y0 = spr
    h,w = alpha.shape
    tx = x0 + dx*5
    ty = y0 + dy*5
    ix,iy = int(np.floor(tx)),int(np.floor(ty))
    matrix = np.array([[1,0,tx-ix+2],[0,1,ty-iy+2]], np.float32)
    aa = cv2.warpAffine(alpha, matrix, (w+4,h+4), flags=cv2.INTER_LINEAR)
    pp = cv2.warpAffine(premul, matrix, (w+4,h+4), flags=cv2.INTER_LINEAR)
    ix-=2; iy-=2
    lx=max(0,ix); ly=max(0,iy)
    rx=min(canvas.shape[1],ix+w+4); ry=min(canvas.shape[0],iy+h+4)
    a=aa[ly-iy:ry-iy,lx-ix:rx-ix,None]
    p=pp[ly-iy:ry-iy,lx-ix:rx-ix]
    canvas[ly:ry,lx:rx] = canvas[ly:ry,lx:rx]*(1-a)+p


def shadow_sprite(rgb, bg, bounds, body_alpha):
    x0,y0,x1,y1 = [int(round(v*5)) for v in bounds]
    src=rgb[y0:y1,x0:x1].astype(np.float32)
    clean=bg[y0:y1,x0:x1].astype(np.float32)
    # Ground-contact shadows are transported as transmission, not pasted wood.
    trans = np.clip(np.mean(src/(clean+1e-3),axis=2), .65, 1)
    yy=np.arange(y0,y1)[:,None]/5
    ground=np.clip((yy-80.9)/.8,0,1)*np.clip((83.5-yy)/.7,0,1)
    strength=(1-trans)*ground
    strength*=1-body_alpha[y0:y1,x0:x1]
    strength=cv2.GaussianBlur(strength,(5,5),.65)
    return strength, x0, y0


def cast_shadow(canvas, spr, dx):
    s,x0,y0=spr
    h,w=s.shape
    tx=x0+dx*5; ix=int(np.floor(tx))
    matrix=np.array([[1,0,tx-ix+2],[0,1,0]],np.float32)
    a=cv2.warpAffine(s,matrix,(w+4,h),flags=cv2.INTER_LINEAR)
    ix-=2
    lo=max(0,ix);hi=min(canvas.shape[1],ix+w+4)
    canvas[y0:y0+h,lo:hi]*=1-a[:,lo-ix:hi-ix,None]
