        source_height = max(0, floor_y-sy)
        dx_shadow = cx-sx + .95*(height-source_height)
        shifted_shadow = translate(shadow, dx_shadow, 0)
        # Airborne shadows are a little softer and less intense.
        if height > source_height+1:
            shifted_shadow = cv2.GaussianBlur(shifted_shadow, (3, 3), .55)
            shifted_shadow *= max(.70, 1-.035*(height-source_height))
        out[t] += shifted_shadow + shifted_body

    frames = np.clip(np.rint(out), 0, 255).astype(np.uint8)
    artifact_io.save(frames, ROOT)
    # A conventional MP4 in addition to the required exact RGB artifacts.
    writer = cv2.VideoWriter(str(ROOT/'"'edited_video.mp4'),
                             cv2.VideoWriter_fourcc(*'mp4v'), 24, (256, 144))
    if not writer.isOpened():
        raise RuntimeError('Cannot open video encoder')
    for f in frames:
        writer.write(cv2.cvtColor(f, cv2.COLOR_RGB2BGR))
    writer.release()
    return frames


if __name__ == '__main__':
    frames = make_video()
    print(f'Saved {len(frames)} RGB frames, 256x144, 24 fps; edited_video.mp4')
