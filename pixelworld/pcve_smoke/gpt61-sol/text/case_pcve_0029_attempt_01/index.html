+"""Restitution counterfactual using source RGB layers and analytic ballistic motion.
+No learned rendering or external physics engine is used.
+"""
+from pathlib import Path
+import numpy as np
+from scipy.ndimage import label, map_coordinates, gaussian_filter
+from scipy.optimize import least_squares
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+def track_ball(frames):
+    positions = []
+    for a in frames:
+        r, g, b = a.transpose(2, 0, 1)
+        mask = (r > g * 1.7) & (r > b * 1.7) & (r > 85)
+        mask[:, :100] = False
+        mask[:, 180:] = False
+        mask[103:] = False
+        labels, n = label(mask)
+        k = max(range(1, n + 1), key=lambda j: np.count_nonzero(labels == j))
+        sy, sx = np.where(labels == k)
+        top = sy.min()
+        boundary = []
+        for y in range(top, min(top + 18, 103)):
+            xs = np.where(labels[y] == k)[0]
+            if len(xs) > 3:
+                boundary.extend([(xs.min() - .5, y), (xs.max() + .5, y)])
+        boundary = np.asarray(boundary)
+        result = least_squares(
+            lambda v: np.linalg.norm(boundary - v[:2], axis=1) - v[2],
+            [sx.mean(), top + 11, 11], loss='soft_l1', f_scale=.3)
+        positions.append(result.x)
+    return np.asarray(positions)
+
+
+def get_motion(p):
+    # Fit free flight, rather than scaling bounce height (energy is quadratic).
+    windows = [(0, 11), (13, 31), (33, 47), (49, 61), (63, 72), (74, 82), (83, 89)]
+    coeffs = [np.polyfit(np.arange(a, b + 1), p[a:b + 1, 1], 2)
+              for a, b in windows]
+    times = []
+    e = []
+    for j, (c1, c2) in enumerate(zip(coeffs[:-1], coeffs[1:])):
+        roots = np.roots(c1 - c2)
+        t = min(roots, key=lambda v: abs(v - windows[j][1]))
+        vi = np.polyval(np.polyder(c1), t)
+        vo = np.polyval(np.polyder(c2), t)
+        times.append(float(t))
+        e.append(-vo / vi)
+    # The initial bounce includes the initial sliding-to-rolling impulse.
+    first_t = times[0]
+    floor = float(np.median([np.polyval(coeffs[i], t) for i, t in enumerate(times)]))
+    gravity = float(np.median([2 * c[0] for c in coeffs[1:]]))
+    outgoing = -float(np.polyval(np.polyder(coeffs[1]), first_t)) * 1.2
+    restitution = float(np.median(e[1:])) * 1.2
+
+    # Tangential source dynamics, with collision losses at counterfactual times.
+    events = np.asarray(times + [90.2])
+    def xs(v, event_times):
+        x0, speed0, drag, firstloss, loss = v
+        out = []
+        for T in np.arange(96):
+            x, speed, last = x0, speed0, 0.
+            for k, time in enumerate(list(np.asarray(event_times)[np.asarray(event_times) < T]) + [T]):
+                dt = time - last
+                x += speed * (-np.expm1(-drag * dt)) / drag
+                speed *= np.exp(-drag * dt)
+                speed *= firstloss if k == 0 else loss
+                last = time
+            out.append(x)
+        return np.asarray(out)
+    fit = least_squares(lambda v: xs(v, events) - p[:, 0],
+                        [127.4, .95, .005, .4, .92],
+                        bounds=([126, .7, .000001, .2, .6], [129, 1.3, .04, .7, 1.05]))
+    target_y = p[:, 1].copy()
+    new_events = [first_t]
+    last, up = first_t, outgoing
+    for t in range(12, 96):
+        while t >= last + 2 * up / gravity:
+            last += 2 * up / gravity
+            up *= restitution
+            new_events.append(last)
+        dt = t - last
+        target_y[t] = floor - up * dt + .5 * gravity * dt * dt
+    target_x = xs(fit.x, new_events)
+    target_x[:12] = p[:12, 0]
+    # At the first contact, preserve its exactly sampled image; no impulse has
+    # yet created a visible positional separation at this temporal resolution.
+    target_y[12] = p[12, 1]
+    target_x[12] = p[12, 0]
+    print('Estimated gravity (px/frame^2):', gravity)
+    print('Original later restitution:', restitution / 1.2, 'edited:', restitution)
+    print('Edited impacts (zero-based frame times):', new_events)
+    return np.c_[target_x, target_y], floor
+
+
+def interp(a, sy, sx):
+    return np.stack([map_coordinates(a[:, :, c], [sy, sx], order=1,
+                                      mode='nearest', prefilter=False)
+                     for c in range(3)], axis=-1)
+
+
+def generate():
+    src = np.load(ROOT / 'source_pixels.npz')['rgb'].astype(np.float64)
+    p = track_ball(src)
+    target, floor = get_motion(p)
+    height, width = src.shape[1:3]
+    yy, xx = np.indices((height, width), dtype=float)
+
+    # Initial source view contains a clear, unoccluded floor. The initial ball
+    # is removed using later observations of exactly the same wall pixels.
+    bg = src[0].copy()
+    initial_hole = np.hypot(xx - p[0, 0], yy - p[0, 1]) < 14
+    clean_samples = src.copy()
+    for i, (x, y, r) in enumerate(p):
+        clean_samples[i, np.hypot(xx-x, yy-y) < 14] = np.nan
+    wall = np.nanmedian(clean_samples, axis=0)
+    bg[initial_hole] = wall[initial_hole]
+    # A tightly bounded circular silhouette provides source-derived antialias
+    # coverage; preserve original diffuse texture and highlights in the RGB.
+    layers = []
+    floor_layers = []
+    for i, (x, y, _) in enumerate(p):
+        rad = np.hypot(xx - x, yy - y)
+        # Estimate edge color from nearby interior pixels, then solve optical
+        # coverage against the clear original backdrop at each edge pixel.
+        sx = x + (xx-x) * 9.6 / np.maximum(rad, 1e-9)
+        sy = y + (yy-y) * 9.6 / np.maximum(rad, 1e-9)
+        edge = interp(src[i], sy, sx)
+        d = edge - bg
+        alpha = np.clip(np.sum((src[i]-bg)*d, axis=-1) /
+                        np.maximum(np.sum(d*d, axis=-1), 1), 0, 1)
+        alpha[rad < 10.25] = 1
+        # Limit floor-shadow contamination of alpha beyond the sphere edge.
+        u = np.clip((rad - 10.4) / 1.65, 0, 1)
+        geo = 1 - u*u*(3-2*u)
+        alpha = np.where(rad < 10.25, 1, np.minimum(alpha, geo + .12))
+        alpha[rad >= 12.1] = 0
+        alpha[alpha < .014] = 0
+        premul = src[i] - (1-alpha[:, :, None]) * bg
+        color = premul / np.maximum(alpha[:, :, None], .01)
+        color = np.clip(color, [72, 20, 10], [197, 117, 101])
+        premul = color * alpha[:, :, None]
+        layers.append((premul, alpha))
+
+        # Appearance residual on the actual floor: cast shadows, warm reflected
+        # light, and the weak mirror image all come from source RGB, not decals.
+        residual = src[i] - bg
+        residual[rad < 12.4] = 0
+        residual[yy < 82] = 0
+        local = np.clip((xx-(x-39))/7, 0, 1) * np.clip(((x+48)-xx)/7, 0, 1)
+        residual *= local[:, :, None]
+        residual = gaussian_filter(residual, sigma=(.65, .65, 0))
+        floor_layers.append(residual)
+
+    frames = []
+    for i in range(96):
+        if i <= 12:
+            frames.append(src[i].astype(np.uint8))
+            continue
+        nx, ny = target[i]
+        ox, oy, _ = p[i]
+        image = src[i].copy()
+        old_radius = np.hypot(xx-ox, yy-oy)
+        image[old_radius < 13.2] = bg[old_radius < 13.2]
+        # Clean all original ground response, with a tapered boundary outside
+        # its support, and then place the response appropriate to new height.
+        clean = np.clip((xx-(ox-43))/8, 0, 1) * np.clip(((ox+53)-xx)/8, 0, 1)
+        clean *= (yy >= 82)
+        image = image * (1-clean[:, :, None]) + bg * clean[:, :, None]
+        # Select an observed flight height; early pre-impact samples provide
+        # entirely clear-floor templates when the new ball is high.
+        costs = np.abs(p[:, 1]-ny)
+        j = int(np.argmin(costs))
+        if ny < p[0, 1]:
+            j = 0
+        residual = interp(floor_layers[j], yy, xx-(nx-p[j, 0]))
+        image += residual
+
+        premul, alpha = layers[i]
+        sx = xx - (nx-ox)
+        sy = yy - (ny-oy)
+        moved_color = interp(premul, sy, sx)
+        moved_alpha = map_coordinates(alpha, [sy, sx], order=1, mode='constant',
+                                       cval=0, prefilter=False)
+        image = image * (1-moved_alpha[:, :, None]) + moved_color
+        frames.append(np.rint(np.clip(image, 0, 255)).astype(np.uint8))
+    frames = np.asarray(frames, dtype=np.uint8)
+    np.savez_compressed(ROOT / 'motion_analysis.npz', factual=p, edited=target)
+    artifact_io.save(frames, ROOT)
+    print('Saved complete RGB counterfactual:', frames.shape)
+
+if __name__ == '__main__':
+    generate()

\"\"\"Restitution counterfactual using source RGB layers and analytic ballistic motion.
No learned rendering or external physics engine is used.
\"\"\"
from pathlib import Path
import numpy as np
from scipy.ndimage import label, map_coordinates, gaussian_filter
from scipy.optimize import least_squares
import artifact_io

ROOT = Path(__file__).resolve().parent

def track_ball(frames):
    positions = []
    for a in frames:
        r, g, b = a.transpose(2, 0, 1)
        mask = (r > g * 1.7) & (r > b * 1.7) & (r > 85)
        mask[:, :100] = False
        mask[:, 180:] = False
        mask[103:] = False
        labels, n = label(mask)
        k = max(range(1, n + 1), key=lambda j: np.count_nonzero(labels == j))
        sy, sx = np.where(labels == k)
        top = sy.min()
        boundary = []
        for y in range(top, min(top + 18, 103)):
            xs = np.where(labels[y] == k)[0]
            if len(xs) > 3:
                boundary.extend([(xs.min() - .5, y), (xs.max() + .5, y)])
        boundary = np.asarray(boundary)
        result = least_squares(
            lambda v: np.linalg.norm(boundary - v[:2], axis=1) - v[2],
            [sx.mean(), top + 11, 11], loss='soft_l1', f_scale=.3)
        positions.append(result.x)
    return np.asarray(positions)


def get_motion(p):
    # Fit free flight, rather than scaling bounce height (energy is quadratic).
    windows = [(0, 11), (13, 31), (33, 47), (49, 61), (63, 72), (74, 82), (83, 89)]
    coeffs = [np.polyfit(np.arange(a, b + 1), p[a:b + 1, 1], 2)
              for a, b in windows]
    times = []
    e = []
    for j, (c1, c2) in enumerate(zip(coeffs[:-1], coeffs[1:])):
        roots = np.roots(c1 - c2)
        t = min(roots, key=lambda v: abs(v - windows[j][1]))
        vi = np.polyval(np.polyder(c1), t)
        vo = np.polyval(np.polyder(c2), t)
        times.append(float(t))
        e.append(-vo / vi)
    # The initial bounce includes the initial sliding-to-rolling impulse.
    first_t = times[0]
    floor = float(np.median([np.polyval(coeffs[i], t) for i, t in enumerate(times)]))
    gravity = float(np.median([2 * c[0] for c in coeffs[1:]]))
    outgoing = -float(np.polyval(np.polyder(coeffs[1]), first_t)) * 1.2
    restitution = float(np.median(e[1:])) * 1.2

    # Tangential source dynamics, with collision losses at counterfactual times.
    events = np.asarray(times + [90.2])
    def xs(v, event_times):
        x0, speed0, drag, firstloss, loss = v
        out = []
        for T in np.arange(96):
            x, speed, last = x0, speed0, 0.
            for k, time in enumerate(list(np.asarray(event_times)[np.asarray(event_times) < T]) + [T]):
                dt = time - last
                x += speed * (-np.expm1(-drag * dt)) / drag
                speed *= np.exp(-drag * dt)
                speed *= firstloss if k == 0 else loss
                last = time
            out.append(x)
        return np.asarray(out)
    fit = least_squares(lambda v: xs(v, events) - p[:, 0],
                        [127.4, .95, .005, .4, .92],
                        bounds=([126, .7, .000001, .2, .6], [129, 1.3, .04, .7, 1.05]))
    target_y = p[:, 1].copy()
    new_events = [first_t]
    last, up = first_t, outgoing
    for t in range(12, 96):
        while t >= last + 2 * up / gravity:
            last += 2 * up / gravity
            up *= restitution
            new_events.append(last)
        dt = t - last
        target_y[t] = floor - up * dt + .5 * gravity * dt * dt
    target_x = xs(fit.x, new_events)
    target_x[:12] = p[:12, 0]
    # At the first contact, preserve its exactly sampled image; no impulse has
    # yet created a visible positional separation at this temporal resolution.
    target_y[12] = p[12, 1]
    target_x[12] = p[12, 0]
    print('Estimated gravity (px/frame"'^2):'"', gravity)
    print('Original later restitution:', restitution / 1.2, 'edited:', restitution)
    print('Edited impacts (zero-based frame times):', new_events)
    return np.c_[target_x, target_y], floor


def interp(a, sy, sx):
    return np.stack([map_coordinates(a[:, :, c], [sy, sx], order=1,
                                      mode='nearest', prefilter=False)
                     for c in range(3)], axis=-1)


def generate():
    src = np.load(ROOT / 'source_pixels.npz')['rgb'].astype(np.float64)
    p = track_ball(src)
    target, floor = get_motion(p)
    height, width = src.shape[1:3]
    yy, xx = np.indices((height, width), dtype=float)

    # Initial source view contains a clear, unoccluded floor. The initial ball
    # is removed using later observations of exactly the same wall pixels.
    bg = src[0].copy()
    initial_hole = np.hypot(xx - p[0, 0], yy - p[0, 1]) < 14
    clean_samples = src.copy()
    for i, (x, y, r) in enumerate(p):
        clean_samples[i, np.hypot(xx-x, yy-y) < 14] = np.nan
    wall = np.nanmedian(clean_samples, axis=0)
    bg[initial_hole] = wall[initial_hole]
    # A tightly bounded circular silhouette provides source-derived antialias
    # coverage; preserve original diffuse texture and highlights in the RGB.
    layers = []
    floor_layers = []
    for i, (x, y, _) in enumerate(p):
        rad = np.hypot(xx - x, yy - y)
        # Estimate edge color from nearby interior pixels, then solve optical
        # coverage against the clear original backdrop at each edge pixel.
        sx = x + (xx-x) * 9.6 / np.maximum(rad, 1e-9)
        sy = y + (yy-y) * 9.6 / np.maximum(rad, 1e-9)
        edge = interp(src[i], sy, sx)
        d = edge - bg
        alpha = np.clip(np.sum((src[i]-bg)*d, axis=-1) /
                        np.maximum(np.sum(d*d, axis=-1), 1), 0, 1)
        alpha[rad < 10.25] = 1
        # Limit floor-shadow contamination of alpha beyond the sphere edge.
        u = np.clip((rad - 10.4) / 1.65, 0, 1)
        geo = 1 - u*u*(3-2*u)
        alpha = np.where(rad < 10.25, 1, np.minimum(alpha, geo + .12))
        alpha[rad >= 12.1] = 0
        alpha[alpha < .014] = 0
        premul = src[i] - (1-alpha[:, :, None]) * bg
        color = premul / np.maximum(alpha[:, :, None], .01)
        color = np.clip(color, [72, 20, 10], [197, 117, 101])
        premul = color * alpha[:, :, None]
        layers.append((premul, alpha))

        # Appearance residual on the actual floor: cast shadows, warm reflected
        # light, and the weak mirror image all come from source RGB, not decals.
        residual = src[i] - bg
        residual[rad < 12.4] = 0
        residual[yy < 82] = 0
        local = np.clip((xx-(x-39))/7, 0, 1) * np.clip(((x+48)-xx)/7, 0, 1)
        residual *= local[:, :, None]
        residual = gaussian_filter(residual, sigma=(.65, .65, 0))
        floor_layers.append(residual)

    frames = []
    for i in range(96):
        if i <= 12:
            frames.append(src[i].astype(np.uint8))
            continue
        nx, ny = target[i]
        ox, oy, _ = p[i]
        image = src[i].copy()
        old_radius = np.hypot(xx-ox, yy-oy)
        image[old_radius < 13.2] = bg[old_radius < 13.2]
        # Clean all original ground response, with a tapered boundary outside
        # its support, and then place the response appropriate to new height.
        clean = np.clip((xx-(ox-43))/8, 0, 1) * np.clip(((ox+53)-xx)/8, 0, 1)
        clean *= (yy >= 82)
        image = image * (1-clean[:, :, None]) + bg * clean[:, :, None]
        # Select an observed flight height; early pre-impact samples provide
        # entirely clear-floor templates when the new ball is high.
        costs = np.abs(p[:, 1]-ny)
        j = int(np.argmin(costs))
        if ny < p[0, 1]:
            j = 0
        residual = interp(floor_layers[j], yy, xx-(nx-p[j, 0]))
        image += residual

        premul, alpha = layers[i]
        sx = xx - (nx-ox)
        sy = yy - (ny-oy)
        moved_color = interp(premul, sy, sx)
        moved_alpha = map_coordinates(alpha, [sy, sx], order=1, mode='constant',
                                       cval=0, prefilter=False)
        image = image * (1-moved_alpha[:, :, None]) + moved_color
        frames.append(np.rint(np.clip(image, 0, 255)).astype(np.uint8))
    frames = np.asarray(frames, dtype=np.uint8)
    np.savez_compressed(ROOT / 'motion_analysis.npz', factual=p, edited=target)
    artifact_io.save(frames, ROOT)
    print('Saved complete RGB counterfactual:', frames.shape)

if __name__ == '__main__':
    generate()
