+"""Counterfactual volleyball: 1.5x initial speed, unchanged rug friction.
+Recover geometry, rolling paint, and contact shadow using only source RGB.
+"""
+from pathlib import Path
+import cv2
+import numpy as np
+from scipy.optimize import minimize_scalar
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+# Numerically recovered perspective/rolling model. Motion is almost lossless
+# on the floor, followed by constant rolling resistance on the rug.
+PARAM = np.array([80.3634067,120.63123026,12.23699931,
+                  87.47556554,-134.26730279,1.23927615,
+                  2.50343243,.29888453])
+AXIS = np.array([np.cos(.1236)*np.cos(.28475),
+                 np.sin(.1236)*np.cos(.28475),np.sin(.28475)])
+ROLL = 15.20358
+
+
+def distance(t, speed_factor=1):
+    v=PARAM[5]*speed_factor
+    entry=PARAM[7]/speed_factor
+    braking=np.clip(np.asarray(t)-entry,0,v/PARAM[6])
+    return v*np.minimum(t,entry)+v*braking-.5*PARAM[6]*braking**2
+
+
+def pose(s):
+    return np.array([PARAM[0]+PARAM[3]*s/(1+s),
+                     PARAM[1]+PARAM[4]*s/(1+s),PARAM[2]/(1+s)])
+
+
+def read_source():
+    cap=cv2.VideoCapture(str(ROOT/'source_video.mp4'))
+    frames=[]
+    while True:
+        ok,a=cap.read()
+        if not ok: break
+        frames.append(a[:,:,::-1])
+    cap.release()
+    return np.asarray(frames)
+
+
+def sample(im,x,y):
+    return cv2.remap(im,np.asarray(x,dtype=np.float32),
+                     np.asarray(y,dtype=np.float32),cv2.INTER_LINEAR,
+                     borderMode=cv2.BORDER_REFLECT_101)
+
+
+def calibrate_phases(src,ps,ss):
+    grid=np.linspace(-.73,.73,27)
+    u,v=np.meshgrid(grid,grid)
+    m=u*u+v*v<.55
+    ns=np.stack([u[m],v[m],np.sqrt(1-u[m]**2-v[m]**2)],axis=1)
+    colors=[]
+    for im,(cx,cy,r) in zip(src,ps):
+        co=sample(im.astype(np.float32),(5*(cx+r*ns[:,0])).reshape(-1,1),
+                  (5*(cy+r*ns[:,1])).reshape(-1,1)).reshape(-1,3)
+        colors.append(co/np.maximum(co.sum(axis=1,keepdims=True),1))
+    phases=[0.]
+    for i in range(19):
+        def error(angle):
+            nn=ns@cv2.Rodrigues(-AXIS*angle)[0].T
+            good=nn[:,2]>.1
+            cx,cy,r=ps[i]
+            co=sample(src[i].astype(np.float32),(5*(cx+r*nn[:,0])).reshape(-1,1),
+                      (5*(cy+r*nn[:,1])).reshape(-1,1)).reshape(-1,3)
+            co/=np.maximum(co.sum(axis=1,keepdims=True),1)
+            return np.mean(np.minimum(np.abs(co[good]-colors[i+1][good]),.15))
+        expected=ROLL*(ss[i+1]-ss[i])
+        fit=minimize_scalar(error,bounds=(max(-.025,expected-.18),expected+.18),
+                            method='bounded',options={'xatol':.0007})
+        phases.append(phases[-1]+fit.x)
+    return np.array(phases)
+
+
+def main():
+    src=read_source()
+    assert src.shape==(96,720,1280,3)
+    factual=np.load(ROOT/'source_pixels.npz')['rgb']
+    background=np.median(src[20:],axis=0).astype(np.float32)
+    # The only persistent foreground in the last frames is the stopped ball.
+    # Its footprint is visible unobstructed during the opening frames.
+    background[270:400,505:650]=np.median(src[:6,270:400,505:650],axis=0)
+    ss=distance(np.arange(20)/24)
+    ps=np.array([pose(s) for s in ss])
+    phases=calibrate_phases(src[:20],ps,ss)
+    print('Source roll phases:',np.round(phases,3))
+    # Small geometric Lambert field, used solely to transfer source paint
+    # across viewing angles without transferring moving highlights.
+    light=np.array([-.38,-.55,.745]);light/=np.linalg.norm(light)
+    def illumination(n):
+        return .64+.36*np.maximum(0,n@light)
+    # Numeric contact-shadow mattes normalized to a sphere radius.
+    shadow_maps=[]
+    for k in range(20):
+        cx,cy,r=ps[k]*5
+        x0=max(0,int(cx-2.6*r));x1=min(1280,int(cx+1.3*r))
+        y0=max(0,int(cy-1.3*r));y1=min(720,int(cy+1.6*r))
+        yy,xx=np.mgrid[y0:y1,x0:x1]
+        ratio=np.mean(src[k,y0:y1,x0:x1].astype(np.float32),axis=2)/np.maximum(
+            np.mean(background[y0:y1,x0:x1],axis=2),1)
+        # Shadows are dark and achromatic; remove all sphere paint.
+        rr=np.hypot((xx-cx)/r,(yy-cy)/r)
+        ring=np.clip((rr-1.01)/.065,0,1)
+        dark=np.clip(1-ratio,0,.68)*ring
+        rgb=src[k,y0:y1,x0:x1].astype(float)
+        chroma=np.max(rgb,axis=2)-np.min(rgb,axis=2)
+        dark*=np.clip((65-chroma)/25,0,1)
+        dark=cv2.GaussianBlur(dark.astype(np.float32),(5,5),.8)
+        shadow_maps.append((dark,x0,y0))
+    results=[]
+    for frame in range(96):
+        if frame==0:
+            results.append(factual[0].copy())
+            continue
+        s=float(distance(frame/24,1.5));cx,cy,r=pose(s)
+        phase=float(np.interp(min(s,ss[-1]),ss,phases))
+        if s>ss[-1]:phase=phases[-1]+ROLL*(s-ss[-1])
+        canvas=background.copy()
+        # Shadow follows the ball on the unchanged floor/rug.
+        k=int(np.argmin(np.abs(ss-min(s,ss[-1]))))
+        oldx,oldy,oldr=ps[k]*5
+        shadow,x0,y0=shadow_maps[k]
+        bx0=max(0,int((cx-2.7*r)*5));bx1=min(1280,int((cx+1.5*r)*5)+1)
+        by0=max(0,int((cy-1.4*r)*5));by1=min(720,int((cy+1.7*r)*5)+1)
+        yy,xx=np.mgrid[by0:by1,bx0:bx1]
+        sh=sample(shadow,oldx+(xx-cx*5)*oldr/(r*5)-x0,
+                  oldy+(yy-cy*5)*oldr/(r*5)-y0)
+        # remap outside a crop must be transparent, not reflected.
+        sx=oldx+(xx-cx*5)*oldr/(r*5)-x0
+        sy=oldy+(yy-cy*5)*oldr/(r*5)-y0
+        sh*=((sx>=0)&(sx<shadow.shape[1]-1)&(sy>=0)&(sy<shadow.shape[0]-1))
+        canvas[by0:by1,bx0:bx1]*=(1-sh[:,:,None])
+        # Paint is recovered from all available source orientations. Backward
+        # sphere warps retain the actual colored curved volleyball panels.
+        bx0=max(0,int((cx-r)*5)-2);bx1=min(1280,int((cx+r)*5)+3)
+        by0=max(0,int((cy-r)*5)-2);by1=min(720,int((cy+r)*5)+3)
+        yy,xx=np.mgrid[by0:by1,bx0:bx1]
+        u=(xx-cx*5)/(r*5);v=(yy-cy*5)/(r*5)
+        rr=u*u+v*v
+        nn=np.stack([u,v,np.sqrt(np.maximum(0,1-rr))],axis=-1)
+        total=np.zeros((*rr.shape,3),dtype=np.float32)
+        weight=np.zeros(rr.shape,dtype=np.float32)
+        delta=np.angle(np.exp(1j*(phase-phases)))
+        nearest=np.argsort(np.abs(delta))[:5]
+        for j in nearest:
+            oldnorm=nn@cv2.Rodrigues(-AXIS*delta[j])[0].T
+            ox,oy,orr=ps[j]*5
+            paint=sample(src[j].astype(np.float32),ox+orr*oldnorm[:,:,0],oy+orr*oldnorm[:,:,1])
+            correction=illumination(nn)/np.maximum(illumination(oldnorm),.1)
+            paint*=correction[:,:,None]
+            w=np.exp(-.5*(delta[j]/.23)**2)*np.clip((oldnorm[:,:,2]-.025)/.18,0,1)
+            # Use only genuinely visible surfaces, never background pixels.
+            total+=paint*w[:,:,None];weight+=w
+        # Rare limb samples: a close orientation remains the best appearance.
+        missing=weight<1e-7
+        if np.any(missing):
+            j=nearest[0];ox,oy,orr=ps[j]*5
+            paint=sample(src[j].astype(np.float32),ox+orr*u*.98,oy+orr*v*.98)
+            total[missing]=paint[missing];weight[missing]=1
+        paint=total/weight[:,:,None]
+        coverage=np.clip((1-np.sqrt(rr))*r*5+.5,0,1)
+        patch=canvas[by0:by1,bx0:bx1]
+        canvas[by0:by1,bx0:bx1]=patch*(1-coverage[:,:,None])+paint*coverage[:,:,None]
+        small=cv2.resize(np.clip(canvas,0,255).astype(np.uint8),(256,144),interpolation=cv2.INTER_AREA)
+        results.append(small)
+    results=np.asarray(results,dtype=np.uint8)
+    artifact_io.save(results,ROOT)
+    print('Saved:',results.shape,'; counterfactual stop position:',pose(float(distance(4,1.5))))
+
+if __name__=='__main__':main()

def calibrate_phases(src,ps,ss):
    grid=np.linspace(-.73,.73,27)

    shadow_maps=[]
    for k in range(20):
        cx,cy,r=ps[k]*5
        x0=max(0,int(cx-2.6*r));x1=min(1280,int(cx+1.3*r))
        y0=max(0,int(cy-1.3*r));y1=min(720,int(cy+1.6*r))
        yy,xx=np.mgrid[y0:y1,x0:x1]
        ratio=np.mean(src[k,y0:y1,x0:x1].astype(np.float32),axis=2)/np.maximum(
            np.mean(background[y0:y1,x0:x1],axis=2),1)
        # Shadows are dark and achromatic; remove all sphere paint.
        rr=np.hypot((xx-cx)/r,(yy-cy)/r)
        ring=np.clip((rr-1.01)/.065,0,1)
        dark=np.clip(1-ratio,0,.68)*ring
        rgb=src[k,y0:y1,x0:x1].astype(float)
        chroma=np.max(rgb,axis=2)-np.min(rgb,axis=2)
        dark*=np.clip((65-chroma)/25,0,1)
        dark=cv2.GaussianBlur(dark.astype(np.float32),(5,5),.8)
        shadow_maps.append((dark,x0,y0))
    results=[]
    for frame in range(96):
        if frame==0:
            results.append(factual[0].copy())
            continue
        s=float(distance(frame/24,1.5));cx,cy,r=pose(s)
        phase=float(np.interp(min(s,ss[-1]),ss,phases))
        if s>ss[-1]:phase=phases[-1]+ROLL*(s-ss[-1])
        canvas=background.copy()
        # Shadow follows the ball on the unchanged floor/rug.
        k=int(np.argmin(np.abs(ss-min(s,ss[-1]))))
        oldx,oldy,oldr=ps[k]*5
        shadow,x0,y0=shadow_maps[k]
        bx0=max(0,int((cx-2.7*r)*5));bx1=min(1280,int((cx+1.5*r)*5)+1)
        by0=max(0,int((cy-1.4*r)*5));by1=min(720,int((cy+1.7*r)*5)+1)
        yy,xx=np.mgrid[by0:by1,bx0:bx1]
        sh=sample(shadow,oldx+(xx-cx*5)*oldr/(r*5)-x0,
                  oldy+(yy-cy*5)*oldr/(r*5)-y0)
        # remap outside a crop must be transparent, not reflected.
        sx=oldx+(xx-cx*5)*oldr/(r*5)-x0
        sy=oldy+(yy-cy*5)*oldr/(r*5)-y0
        sh*=((sx>=0)&(sx<shadow.shape[1]-1)&(sy>=0)&(sy<shadow.shape[0]-1))
        canvas[by0:by1,bx0:bx1]*=(1-sh[:,:,None])
        # Paint is recovered from all available source orientations. Backward
        # sphere warps retain the actual colored curved volleyball panels.
        bx0=max(0,int((cx-r)*5)-2);bx1=min(1280,int((cx+r)*5)+3)
        by0=max(0,int((cy-r)*5)-2);by1=min(720,int((cy+r)*5)+3)
        yy,xx=np.mgrid[by0:by1,bx0:bx1]
        u=(xx-cx*5)/(r*5);v=(yy-cy*5)/(r*5)
        rr=u*u+v*v
        nn=np.stack([u,v,np.sqrt(np.maximum(0,1-rr))],axis=-1)
        total=np.zeros((*rr.shape,3),dtype=np.float32)
        weight=np.zeros(rr.shape,dtype=np.float32)
        delta=np.angle(np.exp(1j*(phase-phases)))
        nearest=np.argsort(np.abs(delta))[:5]
        for j in nearest:
            oldnorm=nn@cv2.Rodrigues(-AXIS*delta[j])[0].T
            ox,oy,orr=ps[j]*5
            paint=sample(src[j].astype(np.float32),ox+orr*oldnorm[:,:,0],oy+orr*oldnorm[:,:,1])
            correction=illumination(nn)/np.maximum(illumination(oldnorm),.1)
            paint*=correction[:,:,None]
            w=np.exp(-.5*(delta[j]/.23)**2)*np.clip((oldnorm[:,:,2]-.025)/.18,0,1)
            # Use only genuinely visible surfaces, never background pixels.
            total+=paint*w[:,:,None];weight+=w
        # Rare limb samples: a close orientation remains the best appearance.
        missing=weight<1e-7
        if np.any(missing):
            j=nearest[0];ox,oy,orr=ps[j]*5
            paint=sample(src[j].astype(np.float32),ox+orr*u*.98,oy+orr*v*.98)
            total[missing]=paint[missing];weight[missing]=1
        paint=total/weight[:,:,None]
        coverage=np.clip((1-np.sqrt(rr))*r*5+.5,0,1)
        patch=canvas[by0:by1,bx0:bx1]
        canvas[by0:by1,bx0:bx1]=patch*(1-coverage[:,:,None])+paint*coverage[:,:,None]
        small=cv2.resize(np.clip(canvas,0,255).astype(np.uint8),(256,144),interpolation=cv2.INTER_AREA)
        results.append(small)
    results=np.asarray(results,dtype=np.uint8)
    artifact_io.save(results,ROOT)
    print('Saved:',results.shape,'; counterfactual stop position:',pose(float(distance(4,1.5))))

if __name__=='__main__':main()
