+"""Remove the purple marble and reconstruct the unimpeded gold-marble motion.
+
+Only source RGB samples are used. Geometry is numeric; the remaining marble's
+surface, contact shadow, and reflection are sampled from its source appearance.
+"""
+from pathlib import Path
+import json
+import cv2
+import numpy as np
+from scipy.ndimage import label
+from PIL import Image
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+
+def tracks(rgb):
+    yy, xx = np.mgrid[:144, :256]
+    gold, purple = [], []
+    for f in rgb:
+        r, g, b = f.astype(np.float32).transpose(2, 0, 1)
+        masks = [(r-b > 45) & (g-b > 30) & (yy < 76),
+                 (b > r*1.1) & (r > g*1.25) & (b > 50) & (yy < 80)]
+        for mask, dest in zip(masks, (gold, purple)):
+            lab, n = label(mask)
+            candidates = []
+            for i in range(1, n+1):
+                y, x = np.where(lab == i)
+                if 35 < len(x) < 150 and np.ptp(y) < 15:
+                    candidates.append((len(x), x.mean(), y.mean()))
+            _, x, y = max(candidates)
+            # Account for the gold surface's slightly off-center bright region.
+            dest.append((x + (0.27 if dest is gold else 0.0),
+                         y + (0.20 if dest is gold else 0.0)))
+    return np.array(gold), np.array(purple)
+
+
+def soft_ellipse(X, Y, cx, cy, rx, ry, feather=0.15):
+    d = np.sqrt(((X-cx)/rx)**2 + ((Y-cy)/ry)**2)
+    z = np.clip((1+feather-d)/feather, 0, 1)
+    return z*z*(3-2*z)
+
+
+def footprint(X, Y, c):
+    x, y = c
+    shift = (x-127.5)*0.033
+    sphere = soft_ellipse(X, Y, x, y, 7.4, 7.4, .16)
+    shadow = soft_ellipse(X, Y, x+shift, y+7.2, 11.5, 4.8, .25)
+    # The lower glass surface carries a faint, blurred reflected marble.
+    reflection = soft_ellipse(X, Y, x, 89.0, 10.0, 10.0, .2)
+    return np.maximum.reduce([sphere, shadow, reflection])
+
+
+def clean_background(hi, gold, purple):
+    chosen = np.array([0, 8, 14, 19, 23, 27, 31, 33, 38, 44, 50, 56, 62, 72, 95])
+    top, bottom = 220, 510
+    stack = hi[chosen, top:bottom].astype(np.float32)
+    for j, t in enumerate(chosen):
+        for c in (gold[t], purple[t]):
+            x, y = c
+            x0, x1 = max(0, int((x-19)*5)), min(1280, int((x+20)*5))
+            y0 = max(top, int((y-10)*5))
+            stack[j, y0-top:bottom-top, x0:x1] = np.nan
+    bg = hi[0].copy()
+    bg[top:bottom] = np.nanmedian(stack, axis=0).round().astype(np.uint8)
+    return bg
+
+
+def free_motion(gold):
+    p = gold.copy()
+    # The source's incoming acceleration is 0.233 pixels/frame^2. Opposing
+    # slope/rolling-friction forces reproduce that acceleration, while allowing
+    # a dissipative ascent, turnaround and return with no marble collision.
+    x = gold[33, 0]
+    v = float(np.mean(np.diff(gold[29:34, 0])))
+    for t in range(34, 96):
+        for _ in range(100):
+            left = 1/(1+np.exp((x-111.5)/2.0))
+            right = 1/(1+np.exp((143.5-x)/2.0))
+            gravity = .453*(left-right)
+            resistance = .070 + .150*(left+right)
+            if abs(v) < resistance*.01 and abs(gravity) < resistance:
+                v = 0.0
+            else:
+                if v == 0:
+                    v = gravity*.01
+                nv = v + (gravity-resistance*np.sign(v))*.01
+                if nv*v < 0 and abs(gravity) < resistance:
+                    nv = 0.0
+                x += (v+nv)*.005
+                v = nv
+        q = min(x, 255-x)
+        y = np.interp(q, gold[:34, 0], gold[:34, 1])
+        p[t] = x, y
+    return p
+
+
+def main():
+    rgb = np.load(ROOT/'source_pixels.npz')['rgb']
+    gold, purple = tracks(rgb)
+    cap = cv2.VideoCapture(str(ROOT/'source_video.mp4'))
+    decoded = []
+    while True:
+        ok, image = cap.read()
+        if not ok:
+            break
+        decoded.append(cv2.cvtColor(image, cv2.COLOR_BGR2RGB))
+    cap.release()
+    hi = np.stack(decoded)
+    assert hi.shape == (96, 720, 1280, 3)
+    bg = clean_background(hi, gold, purple)
+    alt = free_motion(gold)
+
+    # Work at native resolution, then use the source's exact Lanczos reduction.
+    top, bottom = 245, 505
+    gy, gx = np.mgrid[top:bottom, :1280]
+    X = (gx+.5)/5-.5
+    Y = (gy+.5)/5-.5
+    bgstrip = bg[top:bottom].astype(np.float32)
+    clean33 = hi[33].astype(np.float32)
+    pm = footprint(X, Y, purple[33])
+    protect = soft_ellipse(X, Y, *gold[33], 7.2, 7.2, .1)
+    pm *= 1-protect
+    clean33[top:bottom] = clean33[top:bottom]*(1-pm[...,None])+bgstrip*pm[...,None]
+
+    # Normalized native-resolution sprites contain the real source marble,
+    # shadows and reflection. Adjacent source samples interpolate smoothly.
+    ly, lx = np.mgrid[-45:166, -85:86]
+    localx, localy = lx/5., ly/5.
+    env = soft_ellipse(localx, localy, 0, 0, 7.25, 7.25, .1)
+    opaque = soft_ellipse(localx, localy, 0, 0, 5.30, 5.30, .105)
+    sprites = []
+    for t in range(34):
+        cx, cy = (gold[t]+.5)*5-.5
+        mapx = (lx+cx).astype(np.float32)
+        mapy = (ly+cy).astype(np.float32)
+        image = clean33 if t == 33 else hi[t].astype(np.float32)
+        sample = cv2.remap(image, mapx, mapy, cv2.INTER_CUBIC,
+                           borderMode=cv2.BORDER_REPLICATE)
+        back = cv2.remap(bg.astype(np.float32), mapx, mapy, cv2.INTER_CUBIC,
+                         borderMode=cv2.BORDER_REPLICATE)
+        delta = sample-back
+        # Suppress codec noise, not subtle shadows or specular pixels.
+        magnitude = np.max(np.abs(delta), axis=2)
+        weight = np.clip((magnitude-1.0)/3.0, 0, 1)
+        fp = footprint(localx+gold[t,0], localy+gold[t,1], gold[t])
+        delta *= (weight*fp)[..., None]
+        sprites.append((delta, back))
+
+    def draw_marble(image, c):
+        q = min(c[0], 255-c[0])
+        ix = int(np.searchsorted(gold[:34, 0], q))
+        ix = min(33, max(1, ix))
+        mix = np.clip((q-gold[ix-1,0])/(gold[ix,0]-gold[ix-1,0]), 0, 1)
+        delta = sprites[ix-1][0]*(1-mix)+sprites[ix][0]*mix
+        back = sprites[ix-1][1]*(1-mix)+sprites[ix][1]*mix
+        if c[0] > 127.5:
+            # Keep the marble's lighting orientation, but mirror the ground
+            # projection so the shadow/reflection fits the opposite slope.
+            delta = delta*env[...,None] + delta[:, ::-1]*(1-env[...,None])
+        cx, cy = (c+.5)*5-.5
+        x0 = int(np.floor(cx))-85
+        y0 = int(np.floor(cy))-45
+        w, h = 171, 211
+        dst = image[y0:y0+h, x0:x0+w]
+        # Subpixel translation of premultiplied residuals and alpha.
+        mat = np.array([[1,0,cx-np.floor(cx)], [0,1,cy-np.floor(cy)]], np.float32)
+        warp = lambda a: cv2.warpAffine(a, mat, (w,h), flags=cv2.INTER_CUBIC,
+                                      borderMode=cv2.BORDER_CONSTANT)
+        d = warp(delta)
+        a = np.clip(warp(opaque), 0, 1)[...,None]
+        premulback = warp(back*opaque[...,None])
+        dst[:] = dst + d + premulback - a*dst
+
+    output = []
+    for t in range(96):
+        image = hi[t].astype(np.float32)
+        mask = footprint(X, Y, purple[t])
+        if t <= 33:
+            # Protect the unaffected gold ball before the missing collision.
+            protect = soft_ellipse(X, Y, *gold[t], 7.2, 7.2, .1)
+            mask *= 1-protect
+        else:
+            mask = np.maximum(mask, footprint(X, Y, gold[t]))
+        image[top:bottom] = image[top:bottom]*(1-mask[...,None])+bgstrip*mask[...,None]
+        if t >= 34:
+            draw_marble(image, alt[t])
+        native = np.clip(image, 0, 255).round().astype(np.uint8)
+        output.append(np.asarray(Image.fromarray(native).resize((256,144), Image.Resampling.LANCZOS)))
+    output = np.stack(output)
+    artifact_io.save(output, ROOT)
+    (ROOT/'motion_diagnostics.json').write_text(json.dumps({
+        'gold_source': gold.tolist(), 'gold_counterfactual': alt.tolist(),
+        'purple_source': purple.tolist(), 'frames': 96, 'fps': 24,
+    }, indent=2))
+    print('Saved complete counterfactual:', output.shape, output.dtype)
+    print('Counterfactual gold x:', [(t, round(float(alt[t,0]),2)) for t in range(33,96,6)])
+
+
+if __name__ == '__main__':
+    main()

\"\"\"Remove the purple marble and reconstruct the unimpeded gold-marble motion.

Only source RGB samples are used. Geometry is numeric; the remaining marble's
surface, contact shadow, and reflection are sampled from its source appearance.
\"\"\"
from pathlib import Path
import json
import cv2
import numpy as np
from scipy.ndimage import label
from PIL import Image
import artifact_io

ROOT = Path(__file__).resolve().parent


def tracks(rgb):
    yy, xx = np.mgrid[:144, :256]
    gold, purple = [], []
    for f in rgb:
        r, g, b = f.astype(np.float32).transpose(2, 0, 1)
        masks = [(r-b > 45) & (g-b > 30) & (yy < 76),
                 (b > r*1.1) & (r > g*1.25) & (b > 50) & (yy < 80)]
        for mask, dest in zip(masks, (gold, purple)):
            lab, n = label(mask)
            candidates = []
            for i in range(1, n+1):
                y, x = np.where(lab == i)
                if 35 < len(x) < 150 and np.ptp(y) < 15:
                    candidates.append((len(x), x.mean(), y.mean()))
            _, x, y = max(candidates)
            # Account for the gold surface's slightly off-center bright region.
            dest.append((x + (0.27 if dest is gold else 0.0),
                         y + (0.20 if dest is gold else 0.0)))
    return np.array(gold), np.array(purple)


def soft_ellipse(X, Y, cx, cy, rx, ry, feather=0.15):
    d = np.sqrt(((X-cx)/rx)**2 + ((Y-cy)/ry)**2)
    z = np.clip((1+feather-d)/feather, 0, 1)
    return z*z*(3-2*z)


def footprint(X, Y, c):
    x, y = c
    shift = (x-127.5)*0.033
    sphere = soft_ellipse(X, Y, x, y, 7.4, 7.4, .16)
    shadow = soft_ellipse(X, Y, x+shift, y+7.2, 11.5, 4.8, .25)
    # The lower glass surface carries a faint, blurred reflected marble.
    reflection = soft_ellipse(X, Y, x, 89.0, 10.0, 10.0, .2)
    return np.maximum.reduce([sphere, shadow, reflection])


def clean_background(hi, gold, purple):
    chosen = np.array([0, 8, 14, 19, 23, 27, 31, 33, 38, 44, 50, 56, 62, 72, 95])
    top, bottom = 220, 510
    stack = hi[chosen, top:bottom].astype(np.float32)
    for j, t in enumerate(chosen):
        for c in (gold[t], purple[t]):
            x, y = c
            x0, x1 = max(0, int((x-19)*5)), min(1280, int((x+20)*5))
            y0 = max(top, int((y-10)*5))
            stack[j, y0-top:bottom-top, x0:x1] = np.nan
    bg = hi[0].copy()
    bg[top:bottom] = np.nanmedian(stack, axis=0).round().astype(np.uint8)
    return bg


def free_motion(gold):
    p = gold.copy()
    # The source's incoming acceleration is 0.233 pixels/frame"'^2. Opposing
    # slope/rolling-friction forces reproduce that acceleration, while allowing
    # a dissipative ascent, turnaround and return with no marble collision.
    x = gold[33, 0]
    v = float(np.mean(np.diff(gold[29:34, 0])))
    for t in range(34, 96):
        for _ in range(100):
            left = 1/(1+np.exp((x-111.5)/2.0))
            right = 1/(1+np.exp((143.5-x)/2.0))
            gravity = .453*(left-right)
            resistance = .070 + .150*(left+right)
            if abs(v) < resistance*.01 and abs(gravity) < resistance:
                v = 0.0
            else:
                if v == 0:
                    v = gravity*.01
                nv = v + (gravity-resistance*np.sign(v))*.01
                if nv*v < 0 and abs(gravity) < resistance:
                    nv = 0.0
                x += (v+nv)*.005
                v = nv
        q = min(x, 255-x)
        y = np.interp(q, gold[:34, 0], gold[:34, 1])
        p[t] = x, y
    return p


def main():
    rgb = np.load(ROOT/'"'source_pixels.npz')['rgb']
    gold, purple = tracks(rgb)
    cap = cv2.VideoCapture(str(ROOT/'source_video.mp4'))
    decoded = []
    while True:
        ok, image = cap.read()
        if not ok:
            break
        decoded.append(cv2.cvtColor(image, cv2.COLOR_BGR2RGB))
    cap.release()
    hi = np.stack(decoded)
    assert hi.shape == (96, 720, 1280, 3)
    bg = clean_background(hi, gold, purple)
    alt = free_motion(gold)

    # Work at native resolution, then use the source's exact Lanczos reduction.
    top, bottom = 245, 505
    gy, gx = np.mgrid[top:bottom, :1280]
    X = (gx+.5)/5-.5
    Y = (gy+.5)/5-.5
    bgstrip = bg[top:bottom].astype(np.float32)
    clean33 = hi[33].astype(np.float32)
    pm = footprint(X, Y, purple[33])
    protect = soft_ellipse(X, Y, *gold[33], 7.2, 7.2, .1)
    pm *= 1-protect
    clean33[top:bottom] = clean33[top:bottom]*(1-pm[...,None])+bgstrip*pm[...,None]

    # Normalized native-resolution sprites contain the real source marble,
    # shadows and reflection. Adjacent source samples interpolate smoothly.
    ly, lx = np.mgrid[-45:166, -85:86]
    localx, localy = lx/5., ly/5.
    env = soft_ellipse(localx, localy, 0, 0, 7.25, 7.25, .1)
    opaque = soft_ellipse(localx, localy, 0, 0, 5.30, 5.30, .105)
    sprites = []
    for t in range(34):
        cx, cy = (gold[t]+.5)*5-.5
        mapx = (lx+cx).astype(np.float32)
        mapy = (ly+cy).astype(np.float32)
        image = clean33 if t == 33 else hi[t].astype(np.float32)
        sample = cv2.remap(image, mapx, mapy, cv2.INTER_CUBIC,
                           borderMode=cv2.BORDER_REPLICATE)
        back = cv2.remap(bg.astype(np.float32), mapx, mapy, cv2.INTER_CUBIC,
                         borderMode=cv2.BORDER_REPLICATE)
        delta = sample-back
        # Suppress codec noise, not subtle shadows or specular pixels.
        magnitude = np.max(np.abs(delta), axis=2)
        weight = np.clip((magnitude-1.0)/3.0, 0, 1)
        fp = footprint(localx+gold[t,0], localy+gold[t,1], gold[t])
        delta *= (weight*fp)[..., None]
        sprites.append((delta, back))

    def draw_marble(image, c):
        q = min(c[0], 255-c[0])
        ix = int(np.searchsorted(gold[:34, 0], q))
        ix = min(33, max(1, ix))
        mix = np.clip((q-gold[ix-1,0])/(gold[ix,0]-gold[ix-1,0]), 0, 1)
        delta = sprites[ix-1][0]*(1-mix)+sprites[ix][0]*mix
        back = sprites[ix-1][1]*(1-mix)+sprites[ix][1]*mix
        if c[0] > 127.5:
            # Keep the marble's lighting orientation, but mirror the ground
            # projection so the shadow/reflection fits the opposite slope.
            delta = delta*env[...,None] + delta[:, ::-1]*(1-env[...,None])
        cx, cy = (c+.5)*5-.5
        x0 = int(np.floor(cx))-85
        y0 = int(np.floor(cy))-45
        w, h = 171, 211
        dst = image[y0:y0+h, x0:x0+w]
        # Subpixel translation of premultiplied residuals and alpha.
        mat = np.array([[1,0,cx-np.floor(cx)], [0,1,cy-np.floor(cy)]], np.float32)
        warp = lambda a: cv2.warpAffine(a, mat, (w,h), flags=cv2.INTER_CUBIC,
                                      borderMode=cv2.BORDER_CONSTANT)
        d = warp(delta)
        a = np.clip(warp(opaque), 0, 1)[...,None]
        premulback = warp(back*opaque[...,None])
        dst[:] = dst + d + premulback - a*dst

    output = []
    for t in range(96):
        image = hi[t].astype(np.float32)
        mask = footprint(X, Y, purple[t])
        if t <= 33:
            # Protect the unaffected gold ball before the missing collision.
            protect = soft_ellipse(X, Y, *gold[t], 7.2, 7.2, .1)
            mask *= 1-protect
        else:
            mask = np.maximum(mask, footprint(X, Y, gold[t]))
        image[top:bottom] = image[top:bottom]*(1-mask[...,None])+bgstrip*mask[...,None]
        if t >= 34:
            draw_marble(image, alt[t])
        native = np.clip(image, 0, 255).round().astype(np.uint8)
        output.append(np.asarray(Image.fromarray(native).resize((256,144), Image.Resampling.LANCZOS)))
    output = np.stack(output)
    artifact_io.save(output, ROOT)
    (ROOT/'motion_diagnostics.json').write_text(json.dumps({
        'gold_source': gold.tolist(), 'gold_counterfactual': alt.tolist(),
        'purple_source': purple.tolist(), 'frames': 96, 'fps': 24,
    }, indent=2))
    print('Saved complete counterfactual:', output.shape, output.dtype)
    print('Counterfactual gold x:', [(t, round(float(alt[t,0]),2)) for t in range(33,96,6)])


if __name__ == '__main__':
    main()
