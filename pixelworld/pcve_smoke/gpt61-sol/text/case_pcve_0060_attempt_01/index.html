+"""Edit the initial impulse, retaining the original scene and pixel textures.
+
+Motion is measured in output pixels per source frame.  The two contacts in the
+source allow the ground drag and the first collision's velocity transfer to be
+estimated independently.  Only the initial velocity is changed, not friction
+or the speed of the rest of the video.
+"""
+from pathlib import Path
+
+import cv2
+import numpy as np
+from scipy import ndimage
+from scipy.optimize import brentq, least_squares
+
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+
+
+def track_can(frames):
+    result = []
+    for frame in frames:
+        r, g, b = frame.astype(float).transpose(2, 0, 1)
+        mask = (r > 1.3 * g) & (r > 1.3 * b) & (r > 70)
+        mask[:60] = False
+        mask[85:] = False
+        labels, count = ndimage.label(mask)
+        sizes = np.bincount(labels.ravel())
+        sizes[0] = 0
+        yy, xx = np.where(labels == sizes.argmax())
+        left, right = max(0, xx.min() - 2), min(256, xx.max() + 3)
+        weight = np.maximum(r[65:75, left:right] - g[65:75, left:right] - 6, 0)
+        result.append(np.sum(weight.sum(0) * np.arange(left, right)) / weight.sum())
+    return np.asarray(result)
+
+
+def displacement(v, t, drag, quadratic):
+    t = np.asarray(t, dtype=float)
+    angle = np.arctan(v * np.sqrt(quadratic / drag))
+    frequency = np.sqrt(drag * quadratic)
+    tau = np.minimum(np.maximum(t, 0), angle / frequency)
+    return np.log(np.cos(angle - frequency * tau) / np.cos(angle)) / quadratic
+
+
+def velocity(v, t, drag, quadratic):
+    angle = np.arctan(v * np.sqrt(quadratic / drag))
+    return np.sqrt(drag / quadratic) * np.tan(
+        np.maximum(0, angle - np.sqrt(drag * quadratic) * np.asarray(t))
+    )
+
+
+def estimate_dynamics(source):
+    x = track_can(source)
+    regions = [(0, 3), (4, 10), (12, 33)]
+
+    def residual(p):
+        c, q = p[:2]
+        return np.concatenate([
+            p[2 + 2 * j] + displacement(p[3 + 2 * j], np.arange(hi - lo + 1), c, q)
+            - x[lo:hi + 1]
+            for j, (lo, hi) in enumerate(regions)
+        ])
+
+    fit = least_squares(
+        residual, [.219, .00045, x[0], 19.5, x[4], 9.4, x[12], 4.9],
+        bounds=([.001, 1e-7, -np.inf, 0, -np.inf, 0, -np.inf, 0],
+                [1, .02, np.inf, 30, np.inf, 30, np.inf, 30]),
+        max_nfev=1000,
+    )
+    c, q, start, original_v, _, post_v, _, _ = fit.x
+    # Contact when the can's right rim meets the carton's left face.
+    contact_x = 89.0
+    old_contact = brentq(
+        lambda t: start + displacement(original_v, t, c, q) - contact_x, 0, 10
+    )
+    new_contact = brentq(
+        lambda t: start + displacement(original_v * .5, t, c, q) - contact_x, 0, 20
+    )
+    old_impact_v = velocity(original_v, old_contact, c, q)
+    new_impact_v = velocity(original_v * .5, new_contact, c, q)
+    old_post_can_v = velocity(post_v, old_contact - 4, c, q)
+
+    # Track the upper light face of the carton before the bottle collision.
+    box_centers = []
+    for t, approx in [(0, 100), (4, 109), (5, 119), (6, 129),
+                      (7, 138), (8, 147), (9, 156)]:
+        luminance = source[t].astype(float).mean(2)
+        cols = np.arange(approx - 6, approx + 7)
+        w = np.maximum(luminance[62:68, cols] - 175, 0)
+        box_centers.append(float((w.sum(0) * cols).sum() / w.sum()))
+    box_centers = np.asarray(box_centers)
+    old_post_box_v = least_squares(
+        lambda p: box_centers[0]
+        + displacement(p[0], np.arange(4, 10) - old_contact, c, q)
+        - box_centers[1:], [10.5],
+    ).x[0]
+    impulse_scale = new_impact_v / old_impact_v
+    dynamics = dict(
+        drag=c, quadratic=q, start=start, initial_v=original_v * .5,
+        contact=new_contact, contact_x=contact_x,
+        post_can_v=old_post_can_v * impulse_scale,
+        post_box_v=old_post_box_v * impulse_scale,
+    )
+    print('Estimated counterfactual dynamics:', dynamics)
+    print('Can tracking fit RMS:', float(np.sqrt(np.mean(fit.fun ** 2))))
+    return dynamics
+
+
+def polygon_matte(points):
+    scale = 8
+    canvas = np.zeros((144 * scale, 256 * scale), np.uint8)
+    coords = np.round(np.asarray(points) * scale).astype(np.int32)
+    cv2.fillPoly(canvas, [coords], 255)
+    return cv2.resize(canvas, (256, 144), interpolation=cv2.INTER_AREA).astype(float) / 255
+
+
+def sprite(source_frame, clean_frame, points, box):
+    """A premultiplied pixel cutout plus its soft, multiplicative ground shadow."""
+    x0, y0, x1, y1 = box
+    a = polygon_matte(points)
+    region = np.zeros((144, 256), dtype=bool)
+    region[y0:y1, x0:x1] = True
+    a *= region
+    delta = source_frame - clean_frame
+    # Recover foreground RGB at antialiased edges, instead of transplanting
+    # the old background into the moving silhouette.
+    color = np.divide(source_frame - (1 - a[..., None]) * clean_frame,
+                      a[..., None], out=np.zeros_like(source_frame),
+                      where=a[..., None] > 1e-5)
+    color = np.clip(color, 0, 255)
+    premultiplied = color * a[..., None]
+    strength = np.max(np.abs(delta), axis=2)
+    soft_support = np.clip((strength - 2) / 5, 0, 1) * region
+    # Residual texture/shading outside the solid object (including rim AA).
+    residual = (delta - (premultiplied - a[..., None] * clean_frame))
+    residual *= soft_support[..., None]
+    shadow = np.divide(residual, np.maximum(clean_frame, 24))
+    return a.astype(np.float32), premultiplied.astype(np.float32), shadow.astype(np.float32)
+
+
+def translated(array, shift):
+    return cv2.warpAffine(array, np.array([[1., 0., shift], [0., 1., 0.]], np.float32),
+                          (256, 144), flags=cv2.INTER_LINEAR,
+                          borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+
+
+def composite(frame, item, shift):
+    alpha, foreground, shadow = item
+    a = translated(alpha, shift)[..., None]
+    f = translated(foreground, shift)
+    s = translated(shadow, shift)
+    return np.clip(frame * (1 + s) * (1 - a) + f, 0, 255)
+
+
+def main():
+    source = np.load(ROOT / 'source_pixels.npz')['rgb']
+    d = estimate_dynamics(source)
+    first = source[0].astype(float)
+    # Later factual frames expose exactly the pixels behind the two initially
+    # moving objects.  The bottle is kept from frame zero, including its shadow.
+    exposed = np.median(source[45:95].astype(float), axis=0)
+    clean = first.copy()
+    for x0, y0, x1, y1 in [(20, 62, 39, 85), (90, 58, 110, 85)]:
+        clean[y0:y1, x0:x1] = exposed[y0:y1, x0:x1]
+
+    can = sprite(first, clean,
+                 [(25, 64.2), (31.5, 64.2), (33.1, 65.2), (33.6, 67),
+                  (33.6, 80.2), (32.1, 81.8), (25.2, 81.8), (23.6, 80.3),
+                  (23.3, 67.2), (24, 65.3)],
+                 (20, 62, 39, 85))
+    carton = sprite(first, clean,
+                    [(94, 62), (102.8, 61.8), (105.2, 62.5),
+                     (104.7, 79), (104.4, 81.5), (96, 81.6),
+                     (94.6, 79.8), (94.1, 68), (93.7, 65)],
+                    (90, 58, 110, 85))
+
+    # Feather the reconstruction boundary, preserving original unrelated
+    # foreground animation and all static scene details outside the motion band.
+    reconstruction = np.zeros((144, 256), np.float32)
+    reconstruction[54:90, 18:249] = 1
+    reconstruction = cv2.GaussianBlur(reconstruction, (5, 5), .65)[..., None]
+    result = []
+    trajectory = []
+    for t in range(96):
+        c, q = d['drag'], d['quadratic']
+        if t < d['contact']:
+            can_x = d['start'] + displacement(d['initial_v'], t, c, q)
+            box_shift = 0.
+        else:
+            elapsed = t - d['contact']
+            can_x = d['contact_x'] + displacement(d['post_can_v'], elapsed, c, q)
+            box_shift = displacement(d['post_box_v'], elapsed, c, q)
+        canvas = source[t].astype(float) * (1 - reconstruction) + clean * reconstruction
+        canvas = composite(canvas, carton, float(box_shift))
+        canvas = composite(canvas, can, float(can_x - d['start']))
+        rgb = np.round(canvas).clip(0, 255).astype(np.uint8)
+        if t == 0:
+            rgb = source[0].copy()
+        result.append(rgb)
+        trajectory.append((float(can_x), float(100.4242 + box_shift)))
+
+    frames = np.stack(result)
+    artifact_io.save(frames, ROOT)
+    print('Saved', frames.shape, frames.dtype)
+    print('Final can/carton centers:', trajectory[-1])
+    print('Bottle remains at its initial position; there is no second collision.')
+
+
+if __name__ == '__main__':
+    main()
