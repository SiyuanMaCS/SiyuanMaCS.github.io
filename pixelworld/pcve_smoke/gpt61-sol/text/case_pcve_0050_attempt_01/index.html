+"""Double the red curling stone's launch velocity on the original timeline.
+
+Motion is fitted from RGB observations, in a one-dimensional ground coordinate
+with a weak perspective projection.  The earlier equal-mass impact is resolved
+analytically.  Rendering uses only masked, non-learned source RGB cutouts.
+"""
+from pathlib import Path
+import json
+import cv2
+import numpy as np
+from PIL import Image
+from scipy.ndimage import (label, binary_closing, binary_fill_holes,
+                           gaussian_filter, map_coordinates)
+from scipy.optimize import least_squares, brentq
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+
+def tracks(frames):
+    clean = np.median(frames, axis=0).astype(float)
+    # Neither stone is in this region at time zero. Preserve the painted rings.
+    clean[62:88, 103:152] = frames[0, 62:88, 103:152]
+    obs = []
+    for image in frames[:38]:
+        mask = binary_closing(np.max(np.abs(image.astype(float)-clean), axis=2)>32)
+        labs, count = label(mask)
+        points = []
+        for k in range(1, count+1):
+            yy, xx = np.where(labs == k)
+            if len(xx)>20:
+                points.append([xx.mean(), yy.mean()])
+        points.sort()
+        if len(points) != 2:
+            raise RuntimeError('Could not locate the two source stones')
+        obs.append(points)
+    return np.asarray(obs)
+
+
+def fit_motion(obs):
+    time = np.arange(len(obs), dtype=float)[:, None]
+    def model(p):
+        q, friction, vr, vy, xr, xy = p
+        x = np.array([xr, xy])
+        s = x/(1+q*x)
+        v = np.array([vr, vy])
+        s = s + time*v - .5*time**2*friction*np.sign(v)
+        return s/(1-q*s)
+    fit = least_squares(lambda p: (model(p)-obs[:, :, 0]).ravel(),
+                        [.0004,.001,1.86,-1.87,47.5,212.9], max_nfev=1000)
+    return fit.x
+
+
+def biggest(mask):
+    labs, _ = label(mask)
+    size = np.bincount(labs.ravel())
+    size[0] = 0
+    return binary_fill_holes(labs == size.argmax())
+
+
+def sprite(image, background, center):
+    """Extract stone and soft contact shadow at the source's native resolution."""
+    cx, cy = center
+    x0, y0 = int(round(cx))-43, int(round(cy))-33
+    rgb = image[y0:y0+73, x0:x0+87].astype(float)
+    back = background[y0:y0+73, x0:x0+87].astype(float)
+    difference = np.max(np.abs(rgb-back), axis=-1)
+    core = biggest(binary_closing(difference>25))
+    alpha = gaussian_filter(core.astype(float), .55)
+    # Remove the original background from the antialiased silhouette rather
+    # than moving a rectangular ice-coloured tile across the target rings.
+    premult = rgb - (1-alpha[...,None])*back
+    premult = np.clip(premult, 0, 255*alpha[...,None])
+    # The remaining low-contrast portion of the cast shadow adapts to whatever
+    # rink colour is under it. Suppress tiny video-compression differences.
+    darkness = np.maximum(0, (back.mean(-1)-rgb.mean(-1)-2)/back.mean(-1))
+    shadow = gaussian_filter(np.clip(darkness, 0, .4)*(1-alpha), .8)
+    yy, xx = np.where(core)
+    anchor = np.array([xx.mean(), yy.mean()])
+    return {'alpha':alpha, 'rgb':premult, 'shadow':shadow,
+            'anchor':anchor, 'xy':anchor+np.array([x0,y0])}
+
+
+def composite(canvas, item, destination, scale):
+    anchor = item['anchor']
+    h, w = item['alpha'].shape
+    left = destination[0]-anchor[0]*scale
+    top = destination[1]-anchor[1]*scale
+    x0 = max(0, int(np.floor(left))-2)
+    y0 = max(0, int(np.floor(top))-2)
+    x1 = min(canvas.shape[1], int(np.ceil(left+w*scale))+2)
+    y1 = min(canvas.shape[0], int(np.ceil(top+h*scale))+2)
+    if x0>=x1 or y0>=y1:
+        return
+    y, x = np.mgrid[y0:y1,x0:x1]
+    coords = [(y-top)/scale, (x-left)/scale]
+    def warp(a):
+        return map_coordinates(a, coords, order=1, mode='constant', cval=0)
+    alpha = warp(item['alpha'])
+    shadow = warp(item['shadow'])
+    rgb = np.stack([warp(item['rgb'][...,c]) for c in range(3)], axis=-1)
+    crop = canvas[y0:y1,x0:x1]
+    crop *= 1-shadow[...,None]
+    crop *= 1-alpha[...,None]
+    crop += rgb
+
+
+def main():
+    frames = np.load(ROOT/'source_pixels.npz')['rgb']
+    obs = tracks(frames)
+    q, friction, vr, vy, xr, xy = fit_motion(obs)
+    project = lambda s: s/(1-q*s)
+    ground = lambda x: x/(1+q*x)
+    initial = ground(np.array([xr,xy]))
+    # The source stones are the same size. Their near-zero rebound in the
+    # factual footage gives a low restitution; the red launch changes only
+    # its translational velocity, not the yellow launch or rink friction.
+    diameter = 10.0/(1+q*127.5)**2
+    def approach(t):
+        return initial + np.array([2*vr,vy])*t - .5*friction*t*t*np.array([1,-1])
+    impact_time = brentq(lambda t: np.diff(approach(t))[0]-diameter, 0, 38)
+    impact = approach(impact_time)
+    incoming = np.array([2*vr-friction*impact_time,vy+friction*impact_time])
+    restitution = .07
+    cm = incoming.mean()
+    relative = incoming[0]-incoming[1]
+    outgoing = cm + restitution*relative/2*np.array([-1,1])
+    positions = []
+    for t in range(96):
+        if t<=impact_time:
+            s = approach(t)
+        else:
+            dt = t-impact_time
+            # Coulomb ground friction, with a clamp preventing reversal.
+            moving_time = np.minimum(dt, np.abs(outgoing)/friction)
+            s = impact + outgoing*moving_time - .5*friction*moving_time**2*np.sign(outgoing)
+        positions.append(project(s))
+    positions = np.asarray(positions)
+
+    capture = cv2.VideoCapture(str(ROOT/'source_video.mp4'))
+    native = []
+    for t in range(31):
+        ok, bgr = capture.read()
+        if not ok:
+            raise RuntimeError('Incomplete source video')
+        native.append(bgr[...,::-1].copy())
+    capture.release()
+    background = native[0].copy()
+    # Later source pixels reveal the full initial stone footprints, including
+    # their shadows. All other ice and rink geometry remain source pixels.
+    background[303:379,195:290] = native[20][303:379,195:290]
+    background[351:445,1015:1120] = native[20][351:445,1015:1120]
+    banks = [[],[]]
+    for t in range(31):
+        for j in range(2):
+            # Low-resolution measurements are pixel-centred; convert them to
+            # the native five-times resolution for precision in the cutouts.
+            approx = obs[t,j]*5+2
+            banks[j].append(sprite(native[t],background,approx))
+    # Use the undisturbed rolling appearance. The factual source shows no
+    # fragments, deformation, or appreciable rotation caused by the collision.
+    slope = [np.polyfit(obs[:30,j,0],obs[:30,j,1],1)[0] for j in range(2)]
+    result = np.empty_like(frames)
+    for t in range(96):
+        if t==0:
+            result[t] = frames[0]
+            continue
+        canvas = background.astype(float)
+        for j in range(2):
+            # Source texture progression is tied to time (not the doubled
+            # translational velocity). Hold the late uniform rolling surface.
+            k = min(t,30)
+            item = banks[j][k]
+            source_x = obs[k,j,0]
+            target_x = positions[t,j]
+            scale = (1+q*target_x)/(1+q*source_x)
+            shift_x = (target_x-source_x)*5
+            destination = item['xy'] + np.array([shift_x,shift_x*slope[j]])
+            composite(canvas,item,destination,scale)
+        result[t] = np.asarray(Image.fromarray(np.clip(np.rint(canvas),0,255).astype(np.uint8))
+                               .resize((256,144),Image.Resampling.LANCZOS))
+    artifact_io.save(result,ROOT)
+    (ROOT/'motion_report.json').write_text(json.dumps({
+        'initial_red_source_px_per_frame':float(vr*(1+q*xr)**2),
+        'initial_red_edited_px_per_frame':float(2*vr*(1+q*xr)**2),
+        'impact_time_frames':float(impact_time),
+        'impact_time_seconds':float(impact_time/24),
+        'impact_centers':positions[round(impact_time)].tolist(),
+        'last_centers':positions[-1].tolist(),
+        'positions':positions.tolist(),
+        'shape':list(result.shape)},indent=2))
+    print('Saved complete 96-frame edit. Impact at %.3f s; final x = %s' %
+          (impact_time/24,np.round(positions[-1],2)))
+
+
+if __name__ == '__main__':
+    main()
