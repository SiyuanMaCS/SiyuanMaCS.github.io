+def physics():
+    # Perspective parameters estimated from sphere silhouette sizes. Coordinates
+    # describe the horizontal table, not image-plane overlaps.
+    F=240.; horizon=20.; vx=150.; H=4.96
+    Rs=1.; Rb=.800
+    start=22
+    ys=STAR[start,1];yb=SOCCER[start,1]
+    rs=(ys-horizon)/(H-Rs);rb=(yb-horizon)/(H/Rb-1)
+    p=np.array([[(STAR[start,0]-vx)/rs,F/rs],
+                [(SOCCER[start,0]-vx)*Rb/rb,F*Rb/rb]])
+    v=np.array([[.16,-.38],[0.,0.]])
+    normal=p[1]-p[0];normal/=np.linalg.norm(normal)
+    # 10:1 rather than the factual approximately equal masses. The fixed table
+    # boundary rebound has already occurred and is mass independent.
+    relative=np.dot(v[0]-v[1],normal)
+    impulse=(1+.25)*relative/(1/10+1/1)
+    v[0]-=impulse*normal/10
+    v[1]+=impulse*normal

+    travel=np.zeros((96,2)); angles=np.zeros((96,2))

+        d=np.zeros(2)
+        for _ in range(12):

+            nextspeed=np.maximum(0,speed-.0105/12)
+            vn=v*(nextspeed/np.maximum(speed,1e-9))[:,None]
+            dp=(v+vn)/24
+            p+=dp;d+=np.linalg.norm(dp,axis=1)
+            v=vn
+        travel[t]=travel[t-1]+d
+        for i,R in enumerate([Rs,Rb]):

+            xx=vx+p[i,0]*rr/R
+            yy=horizon+(H/R-1)*rr
+            tracks[t,i]=[xx,yy,rr]
+        angles[t]=travel[t]/np.array([Rs,Rb])
+    return tracks,angles

+def sphere(canvas,tex,cx,cy,r,rot=None):
+    h,w=canvas.shape[:2]
+    x0=max(0,int(np.floor(cx-r-1))); x1=min(w,int(np.ceil(cx+r+1)))

+        # Smooth reflection avoids holes or solid, invented surfaces.

+        shade_ratio=illumination(n)/illumination(sample)

+    broad=np.exp(-2*(((xx-cx+1.5)/(r*1.16))**2+((yy-cy-r*1.12)/(r*.45))**2))
+    core=np.exp(-2*(((xx-cx+2)/(r*.86))**2+((yy-cy-r*1.25)/(r*.27))**2))
+    alpha=.17*broad+.43*core
+    canvas[:]=canvas*(1-alpha[:,:,None])+np.array([93,57,31])*alpha[:,:,None]

    Rs=1.; Rb=.800
    start=22
    ys=STAR[start,1];yb=SOCCER[start,1]
    rs=(ys-horizon)/(H-Rs);rb=(yb-horizon)/(H/Rb-1)
    p=np.array([[(STAR[start,0]-vx)/rs,F/rs],
                [(SOCCER[start,0]-vx)*Rb/rb,F*Rb/rb]])
    v=np.array([[.16,-.38],[0.,0.]])
    normal=p[1]-p[0];normal/=np.linalg.norm(normal)
    # 10:1 rather than the factual approximately equal masses. The fixed table
    # boundary rebound has already occurred and is mass independent.
    relative=np.dot(v[0]-v[1],normal)
    impulse=(1+.25)*relative/(1/10+1/1)
    v[0]-=impulse*normal/10
    v[1]+=impulse*normal
    tracks=np.zeros((96,2,3));tracks[:start+1,0]=STAR[:start+1];tracks[:start+1,1]=SOCCER[:start+1]
    travel=np.zeros((96,2)); angles=np.zeros((96,2))
    for t in range(start+1,96):
        # Small substeps keep rolling deceleration smooth at the final stop.
        d=np.zeros(2)
        for _ in range(12):
            speed=np.linalg.norm(v,axis=1)
            nextspeed=np.maximum(0,speed-.0105/12)
            vn=v*(nextspeed/np.maximum(speed,1e-9))[:,None]
            dp=(v+vn)/24
            p+=dp;d+=np.linalg.norm(dp,axis=1)
            v=vn
        travel[t]=travel[t-1]+d
        for i,R in enumerate([Rs,Rb]):
            rr=F*R/max(p[i,1],.3)
            xx=vx+p[i,0]*rr/R
            yy=horizon+(H/R-1)*rr
            tracks[t,i]=[xx,yy,rr]
        angles[t]=travel[t]/np.array([Rs,Rb])
    return tracks,angles

def sphere(canvas,tex,cx,cy,r,rot=None):
    h,w=canvas.shape[:2]
    x0=max(0,int(np.floor(cx-r-1))); x1=min(w,int(np.ceil(cx+r+1)))
    y0=max(0,int(np.floor(cy-r-1))); y1=min(h,int(np.ceil(cy+r+1)))
    if x1<=x0 or y1<=y0:return
    yy,xx=np.mgrid[y0:y1,x0:x1]
    u=(xx-cx)/r; vv=(yy-cy)/r
    dist=np.sqrt(u*u+vv*vv)
    n=np.stack([u,-vv,np.sqrt(np.maximum(0,1-dist*dist))],-1)
    if rot is None:
        sample=n
        shade_ratio=np.ones(n.shape[:2])
    else:
        sample=n@rot
        # Repeated source-view decoration also covers unobserved rear hemisphere.
        # Smooth reflection avoids holes or solid, invented surfaces.
        sample[:,:,0]=np.where(sample[:,:,2]<0,-sample[:,:,0],sample[:,:,0])
        shade_ratio=illumination(n)/illumination(sample)
    coords=np.array([(1-sample[:,:,1])*(TEX-1)/2,(sample[:,:,0]+1)*(TEX-1)/2])
    color=np.stack([map_coordinates(tex[:,:,c],coords,order=1,mode='nearest') for c in range(3)],-1)
    color=np.clip(color*shade_ratio[:,:,None],0,255)
    # Subpixel coverage at output's 3x working resolution.
    alpha=np.clip((1-dist)*r+.5,0,1)
    dst=canvas[y0:y1,x0:x1]
    dst[:]=dst*(1-alpha[:,:,None])+color*alpha[:,:,None]

def shadow(canvas,cx,cy,r):
    yy,xx=np.mgrid[:canvas.shape[0],:canvas.shape[1]]
    # Broad penumbra plus compact warm contact shadow, consistent with the wood.
    broad=np.exp(-2*(((xx-cx+1.5)/(r*1.16))**2+((yy-cy-r*1.12)/(r*.45))**2))
    core=np.exp(-2*(((xx-cx+2)/(r*.86))**2+((yy-cy-r*1.25)/(r*.27))**2))
    alpha=.17*broad+.43*core
    canvas[:]=canvas*(1-alpha[:,:,None])+np.array([93,57,31])*alpha[:,:,None]
