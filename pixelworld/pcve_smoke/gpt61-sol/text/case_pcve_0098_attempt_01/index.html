+"""Mass-intervention edit, using measured motion and original RGB materials.
+
+The 40x mass change has no visible effect before contact.  Collision impulses
+are recalculated, rather than slowing the incoming marble or enlarging the blue
+one.  All rendering is pixel resampling and analytic alpha compositing.
+"""
+from pathlib import Path
+import json
+
+import cv2
+import numpy as np
+from PIL import Image
+from scipy.ndimage import gaussian_filter, map_coordinates
+
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+
+
+def read_material_frames():
+    cap = cv2.VideoCapture(str(ROOT / "source_video.mp4"))
+    result = {}
+    for t in range(31):
+        ok, frame = cap.read()
+        if not ok:
+            raise RuntimeError("Could not decode the source video")
+        if t in (0, 22, 30):
+            result[t] = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
+    cap.release()
+    return result
+
+
+def disk_alpha(x, y, cx, cy, radius):
+    # Coordinates here are original-resolution pixel coordinates.
+    distance = np.hypot(x - cx, y - cy)
+    return np.clip((radius - distance) / 0.7 + 0.5, 0, 1)
+
+
+def sample_rgb(image, x, y):
+    return np.stack([
+        map_coordinates(image[..., k], [y, x], order=1,
+                        mode="nearest", prefilter=False)
+        for k in range(3)
+    ], axis=-1)
+
+
+def generate():
+    source = np.load(ROOT / "source_pixels.npz")["rgb"]
+    info = json.loads((ROOT / "source_info.json").read_text())
+    assert info["execution_frame_1based"] == 1
+    material = read_material_frames()
+    scale = material[0].shape[1] / 256
+    assert scale == 5
+
+    # Static camera: use the original high-resolution room as the plate.
+    # Replace only the initially occupied red-marble area with a later clear
+    # observation of exactly the same surface.
+    plate = material[0].astype(np.float32)
+    y, x = np.mgrid[140:290, 810:985]
+    erase = np.maximum(np.abs((x / scale - 179) / 12),
+                       np.abs((y / scale - 43) / 14))
+    feather = np.clip((1.18 - erase) / 0.18, 0, 1)[..., None]
+    plate[140:290, 810:985] = (
+        plate[140:290, 810:985] * (1 - feather)
+        + material[30][140:290, 810:985] * feather
+    )
+
+    # Measured ground-level center and radius of the original red marble at
+    # source frame 23 (zero-index 22).  Reuse its shaded/glossy sphere pixels.
+    red_center = np.array([194.73, 440.54])
+    red_radius = 39.60
+    red_crop_origin = np.array([140, 390])
+    red_pixels = material[22][390:565, 140:270].astype(np.float32)
+
+    sy, sx = np.mgrid[390:565, 140:270]
+    red_alpha = disk_alpha(sx, sy, *red_center, red_radius)
+    difference = red_pixels - material[0][390:565, 140:270].astype(np.float32)
+    difference = gaussian_filter(difference, (0.7, 0.7, 0))
+    # Contact shadow and the reddish reflection are transferred as RGB
+    # changes, not as a square of the old wooden surface.
+    dx = (sx - red_center[0]) / scale
+    dy = (sy - red_center[1]) / scale
+    shadow_region = np.clip((1.15 - np.hypot((dx - 0.7) / 10.4,
+                                             (dy - 9.5) / 12.0)) / 0.28, 0, 1)
+    shadow_region *= np.clip((dy - 3.0) / 2.0, 0, 1)
+    shadow_region *= 1 - red_alpha
+    # Do not capture any of the departing factual blue marble on the left.
+    shadow_region *= np.where(sy < 480, np.clip((sx - 163) / 7, 0, 1), 1)
+    shadow_region *= np.clip((-difference[..., 1] - 3) / 7, 0, 1)
+    shadow_delta = difference * shadow_region[..., None]
+
+    # Collision calibration from the observed equal-mass event.  Velocities
+    # are pixels / source frame.  Both marbles decelerate at about 0.61
+    # px/frame^2 on the wooden floor.  Increasing mass doesn't change that
+    # acceleration: frictional force increases with normal force.
+    contact = 21.575
+    friction = 0.61
+    incoming_red = -12.88
+    factual_red_after = -4.82
+    factual_blue_after = -7.76
+    mass_ratio = (incoming_red - factual_red_after) / factual_blue_after
+    restitution = (factual_red_after - factual_blue_after) / -incoming_red
+    heavy_mass = 40 * mass_ratio
+    red_velocity = ((1 - restitution * heavy_mass) / (1 + heavy_mass)) * incoming_red
+    blue_velocity = ((1 + restitution) / (1 + heavy_mass)) * incoming_red
+    red_contact_x = 40.93
+    blue_contact_x = 24.85
+
+    def slide(v, dt):
+        moving_time = min(max(dt, 0.0), abs(v) / friction)
+        return np.sign(v) * (abs(v) * moving_time
+                             - 0.5 * friction * moving_time**2)
+
+    # The blue marble's calculated recoil is just 0.11 output pixels.  Apply
+    # that subpixel shift to its source material, leaving its size unchanged.
+    by, bx = np.mgrid[390:500, 65:185]
+    blue_center = np.array([124.11, 442.42])
+    blue_radius = 40.45
+    original_blue = material[0][390:500, 65:185].astype(np.float32)
+    blue_a0 = disk_alpha(bx, by, *blue_center, blue_radius)
+    row_half_width = np.sqrt(np.maximum(0, blue_radius**2
+                                       - (by - blue_center[1])**2)) + 9
+    left = sample_rgb(material[0].astype(np.float32),
+                      blue_center[0] - row_half_width, by)
+    right = sample_rgb(material[0].astype(np.float32),
+                       blue_center[0] + row_half_width, by)
+    blend = np.clip((bx - blue_center[0] + row_half_width)
+                    / (2 * row_half_width), 0, 1)[..., None]
+    hidden_ground = left * (1 - blend) + right * blend
+
+    frames = source.copy()
+    trajectory = []
+    for t in range(96):
+        # No physical consequence before impact, so these frames are exact
+        # originals even though the mass has already changed at time zero.
+        if t <= 21:
+            trajectory.append({"frame": t, "source_unchanged": True})
+            continue
+        dt = t - contact
+        rx = red_contact_x + slide(red_velocity, dt)
+        bxpos = blue_contact_x + slide(blue_velocity, dt)
+        ground_y = 88.108 - 0.029 * (rx - 38.946)
+        # Incoming fall continues through the contact, at reduced vertical
+        # velocity, then the marble is supported by the floor.
+        ry = min(ground_y, 84.80 + 6.6 * dt + 1.5 * dt**2)
+        radius = 7.9238 - 0.0093 * (rx - 38.923)
+        canvas = plate.copy()
+
+        blue_dx = (bxpos - blue_contact_x) * scale
+        blue_dy = -(bxpos - blue_contact_x) * 0.029 * scale
+        shifted_blue = sample_rgb(original_blue,
+                                  bx - blue_dx - 65,
+                                  by - blue_dy - 390)
+        blue_a1 = disk_alpha(bx, by,
+                             blue_center[0] + blue_dx,
+                             blue_center[1] + blue_dy, blue_radius)
+        canvas[390:500, 65:185] += (
+            blue_a1[..., None] * (shifted_blue - hidden_ground)
+            - blue_a0[..., None] * (original_blue - hidden_ground)
+        )
+
+        # Transfer the floor interaction first, followed by the original
+        # sphere. Sampling at 1280x720 retains fractional-pixel trajectories.
+        cx, cy = rx * scale, ry * scale
+        floor_cy = ground_y * scale
+        yy, xx = np.mgrid[int(cy - 48):int(floor_cy + 114),
+                          int(cx - 62):int(cx + 65)]
+        shadow_x = xx - cx + red_center[0] - red_crop_origin[0]
+        shadow_y = yy - floor_cy + red_center[1] - red_crop_origin[1]
+        sampled_shadow = sample_rgb(shadow_delta, shadow_x, shadow_y)
+        height = ground_y - ry
+        sampled_shadow *= np.exp(-height / 8)
+        ys = slice(int(cy - 48), int(floor_cy + 114))
+        xs = slice(int(cx - 62), int(cx + 65))
+        patch = canvas[ys, xs] + sampled_shadow
+        factor = red_radius / (radius * scale)
+        tex_x = (xx - cx) * factor + red_center[0] - red_crop_origin[0]
+        tex_y = (yy - cy) * factor + red_center[1] - red_crop_origin[1]
+        texture = sample_rgb(red_pixels, tex_x, tex_y)
+        alpha = disk_alpha(xx, yy, cx, cy, radius * scale)[..., None]
+        canvas[ys, xs] = patch * (1 - alpha) + texture * alpha
+        image = Image.fromarray(np.clip(canvas, 0, 255).round().astype(np.uint8))
+        frames[t] = np.asarray(image.resize((256, 144), Image.Resampling.LANCZOS))
+        trajectory.append({"frame": t, "red_x": float(rx), "red_y": float(ry),
+                           "blue_x": float(bxpos)})
+
+    assert frames.shape == (96, 144, 256, 3)
+    assert frames.dtype == np.uint8
+    assert np.array_equal(frames[:22], source[:22])
+    # Save this candidate before any further inspection or work.
+    artifact_io.save(frames, ROOT)
+    (ROOT / "trajectory.json").write_text(json.dumps({
+        "mass_multiplier": 40,
+        "contact_frame_zero_based": contact,
+        "coefficient_of_restitution": restitution,
+        "post_impact_red_velocity": red_velocity,
+        "post_impact_blue_velocity": blue_velocity,
+        "trajectory": trajectory,
+    }, indent=2))
+    print("Saved 96 RGB frames at 256x144, 24 fps.")
+    print("Post-contact velocities:", red_velocity, blue_velocity)
+    print("Final centers:", trajectory[-1])
+
+
+if __name__ == "__main__":
+    generate()
