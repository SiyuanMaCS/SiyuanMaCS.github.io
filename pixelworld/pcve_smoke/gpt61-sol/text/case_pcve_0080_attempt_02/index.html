+"""Pixel-only friction counterfactual, with source-derived motion and sprites.
+
+The intervention changes Coulomb contact friction, not the impact's normal
+restitution, the scene, the other body's friction, or the air-drag term.
+"""
+from pathlib import Path
+import numpy as np
+from scipy.ndimage import map_coordinates
+from scipy.optimize import least_squares
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+
+def smoothstep(x):
+    x = np.clip(x, 0.0, 1.0)
+    return x*x*(3.0-2.0*x)
+
+
+def generate():
+    src = np.load(ROOT / 'source_pixels.npz')['rgb'].astype(np.float64)
+    n,h,w,_ = src.shape
+    yy,xx = np.mgrid[:h,:w]
+    # The camera and terrain are stationary. Both ends of the source provide
+    # unobstructed views of opposite halves of the ball's swept ground region.
+    blend = smoothstep((xx-102)/8)
+    clean = src[0]*(1-blend[...,None])+src[-1]*blend[...,None]
+
+    # Subpixel left silhouette, shared sphere geometry, and fitted centers.
+    observations=[]
+    for t,frame in enumerate(src):
+        for y in range(75,88):
+            d=np.max(np.abs(frame[y]-clean[y]),axis=1)
+            xs=np.flatnonzero(d[:134]>30)
+            if xs.size:
+                j=int(xs[0])
+                edge=j-1+(30-d[j-1])/max(1,d[j]-d[j-1])
+                observations.append((t,y,edge))
+    obs=np.asarray(observations)
+    guess=np.array([np.median([x+np.sqrt(max(0,10.85**2-(y-81)**2))
+                              for tt,y,x in observations if tt==t])
+                    for t in range(n)])
+    def edge_residual(p):
+        return (p[obs[:,0].astype(int)]-
+                p[97]*np.sqrt(np.maximum(.01,1-((obs[:,1]-p[96])/p[98])**2))-
+                obs[:,2])
+    fit=least_squares(edge_residual,np.r_[guess,81.04,10.87,10.23],
+                      bounds=(np.r_[np.zeros(n),80,9.5,10],
+                              np.r_[np.full(n,140),82,11.5,12]),
+                      loss='soft_l1')
+    center=fit.x[:n]
+
+    # v'=-mu*g-c*v^2 fits the observed deceleration to subpixel accuracy.
+    # Time is measured in source-frame intervals. Three times mu changes only
+    # the constant contact acceleration. Analytical integration clamps at rest.
+    def trajectory(t,p):
+        x0,v,a,c=p
+        theta=np.arctan(v*np.sqrt(c/a))
+        stop=theta/np.sqrt(a*c)
+        u=np.minimum(np.maximum(t,0),stop)
+        distance=np.log(np.cos(theta-u*np.sqrt(a*c))/np.cos(theta))/c
+        return x0-distance
+    time=np.arange(14,63)-13
+    motion=least_squares(lambda p: trajectory(time,[center[13],*p])-center[14:63],
+                         [2.7,.031,.017],
+                         bounds=([.1,.00001,.00001],[6,.5,.5]))
+    factual=np.r_[center[13],motion.x]
+    modified=factual.copy(); modified[2]*=3.0
+    target=trajectory(np.arange(n)-13,modified)
+    target[:14]=center[:14]
+
+    # Exclude the second, small reddish sphere and its shadow from soccer
+    # masks. Its normal impact and its subsequent travel are unaffected.
+    small=[]
+    for t,frame in enumerate(src):
+        r,g,b=frame.transpose(2,0,1)
+        m=(r>g+15)&(r>b+12)&(yy>=71)&(yy<95)&(xx>center[t]+17)
+        sy,sx=np.where(m)
+        if len(sx):
+            small.append((sx.min()-2,sy.min()-2,sx.max()+8,sy.max()+7))
+        else:
+            small.append(None)
+
+    def region(t):
+        dx=xx-center[t]
+        # Broad feather includes antialiasing, cast shadow and soft ambient
+        # contact shadow, but does not touch the pink platform above.
+        mask=(smoothstep((dx+17)/4)*smoothstep((27-dx)/5)*
+              smoothstep((yy-66)/4)*smoothstep((109-yy)/8))
+        if small[t] is not None:
+            x0,y0,x1,y1=small[t]
+            mask[(xx>=x0)&(xx<=x1)&(yy>=y0)&(yy<=y1)]=0
+        return mask
+
+    masks=[region(t) for t in range(n)]
+    residuals=[(src[t]-clean)*masks[t][...,None] for t in range(n)]
+
+    out=src.copy()
+    # Reuse the original texture at equal distance rolled. Interpolate two
+    # source textures after subpixel registration, preventing orientation
+    # jumps, and hold the final texture once the high-friction ball stops.
+    for t in range(14,n):
+        dest=target[t]
+        q=int(np.flatnonzero(center[13:63]>=dest)[-1]+13)
+        q=min(q,61)
+        f=np.clip((center[q]-dest)/max(1e-6,center[q]-center[q+1]),0,1)
+        background=src[t]*(1-masks[t][...,None])+clean*masks[t][...,None]
+        warped=np.zeros((h,w,3))
+        for s,weight in [(q,1-f),(q+1,f)]:
+            if weight<=0: continue
+            coords=np.array([yy,xx-(dest-center[s])])
+            for c in range(3):
+                warped[...,c]+=weight*map_coordinates(residuals[s][...,c],
+                    coords,order=1,mode='constant',cval=0,prefilter=False)
+        out[t]=background+warped
+    out=np.clip(np.rint(out),0,255).astype(np.uint8)
+    out[:14]=src[:14].astype(np.uint8)
+    artifact_io.save(out,ROOT)
+    stop_frame=13+np.arctan(modified[1]*np.sqrt(modified[3]/modified[2]))/np.sqrt(modified[2]*modified[3])
+    print('Fitted original contact deceleration / drag:',motion.x)
+    print('Soccer center: initial %.3f, original final %.3f, edited final %.3f'%
+          (center[0],center[-1],target[-1]))
+    print('Edited stop time: %.3f seconds'%(stop_frame/24))
+    print('Saved RGB frames:',out.shape,out.dtype)
+    return out
+
+
+if __name__=='__main__':
+    generate()

\"\"\"Pixel-only friction counterfactual, with source-derived motion and sprites.

The intervention changes Coulomb contact friction, not the impact's normal
restitution, the scene, the other body's friction, or the air-drag term.
\"\"\"
from pathlib import Path
import numpy as np
from scipy.ndimage import map_coordinates
from scipy.optimize import least_squares
import artifact_io

ROOT = Path(__file__).resolve().parent


def smoothstep(x):
    x = np.clip(x, 0.0, 1.0)
    return x*x*(3.0-2.0*x)


def generate():
    src = np.load(ROOT / 'source_pixels.npz')['rgb'].astype(np.float64)
    n,h,w,_ = src.shape
    yy,xx = np.mgrid[:h,:w]
    # The camera and terrain are stationary. Both ends of the source provide
    # unobstructed views of opposite halves of the ball's swept ground region.
    blend = smoothstep((xx-102)/8)
    clean = src[0]*(1-blend[...,None])+src[-1]*blend[...,None]

    # Subpixel left silhouette, shared sphere geometry, and fitted centers.
    observations=[]
    for t,frame in enumerate(src):
        for y in range(75,88):
            d=np.max(np.abs(frame[y]-clean[y]),axis=1)
            xs=np.flatnonzero(d[:134]>30)
            if xs.size:
                j=int(xs[0])
                edge=j-1+(30-d[j-1])/max(1,d[j]-d[j-1])
                observations.append((t,y,edge))
    obs=np.asarray(observations)
    guess=np.array([np.median([x+np.sqrt(max(0,10.85**2-(y-81)**2))
                              for tt,y,x in observations if tt==t])
                    for t in range(n)])
    def edge_residual(p):
        return (p[obs[:,0].astype(int)]-
                p[97]*np.sqrt(np.maximum(.01,1-((obs[:,1]-p[96])/p[98])**2))-
                obs[:,2])
    fit=least_squares(edge_residual,np.r_[guess,81.04,10.87,10.23],
                      bounds=(np.r_[np.zeros(n),80,9.5,10],
                              np.r_[np.full(n,140),82,11.5,12]),
                      loss='soft_l1')
    center=fit.x[:n]

    # v'=-mu*g-c*v"'^2 fits the observed deceleration to subpixel accuracy.
    # Time is measured in source-frame intervals. Three times mu changes only
    # the constant contact acceleration. Analytical integration clamps at rest.
    def trajectory(t,p):
        x0,v,a,c=p
        theta=np.arctan(v*np.sqrt(c/a))
        stop=theta/np.sqrt(a*c)
        u=np.minimum(np.maximum(t,0),stop)
        distance=np.log(np.cos(theta-u*np.sqrt(a*c))/np.cos(theta))/c
        return x0-distance
    time=np.arange(14,63)-13
    motion=least_squares(lambda p: trajectory(time,[center[13],*p])-center[14:63],
                         [2.7,.031,.017],
                         bounds=([.1,.00001,.00001],[6,.5,.5]))
    factual=np.r_[center[13],motion.x]
    modified=factual.copy(); modified[2]*=3.0
    target=trajectory(np.arange(n)-13,modified)
    target[:14]=center[:14]

    # Exclude the second, small reddish sphere and its shadow from soccer
    # masks. Its normal impact and its subsequent travel are unaffected.
    small=[]
    for t,frame in enumerate(src):
        r,g,b=frame.transpose(2,0,1)
        m=(r>g+15)&(r>b+12)&(yy>=71)&(yy<95)&(xx>center[t]+17)
        sy,sx=np.where(m)
        if len(sx):
            small.append((sx.min()-2,sy.min()-2,sx.max()+8,sy.max()+7))
        else:
            small.append(None)

    def region(t):
        dx=xx-center[t]
        # Broad feather includes antialiasing, cast shadow and soft ambient
        # contact shadow, but does not touch the pink platform above.
        mask=(smoothstep((dx+17)/4)*smoothstep((27-dx)/5)*
              smoothstep((yy-66)/4)*smoothstep((109-yy)/8))
        if small[t] is not None:
            x0,y0,x1,y1=small[t]
            mask[(xx>=x0)&(xx<=x1)&(yy>=y0)&(yy<=y1)]=0
        return mask

    masks=[region(t) for t in range(n)]
    residuals=[(src[t]-clean)*masks[t][...,None] for t in range(n)]

    out=src.copy()
    # Reuse the original texture at equal distance rolled. Interpolate two
    # source textures after subpixel registration, preventing orientation
    # jumps, and hold the final texture once the high-friction ball stops.
    for t in range(14,n):
        dest=target[t]
        q=int(np.flatnonzero(center[13:63]>=dest)[-1]+13)
        q=min(q,61)
        f=np.clip((center[q]-dest)/max(1e-6,center[q]-center[q+1]),0,1)
        background=src[t]*(1-masks[t][...,None])+clean*masks[t][...,None]
        warped=np.zeros((h,w,3))
        for s,weight in [(q,1-f),(q+1,f)]:
            if weight<=0: continue
            coords=np.array([yy,xx-(dest-center[s])])
            for c in range(3):
                warped[...,c]+=weight*map_coordinates(residuals[s][...,c],
                    coords,order=1,mode='"'constant',cval=0,prefilter=False)
        out[t]=background+warped
    out=np.clip(np.rint(out),0,255).astype(np.uint8)
    out[:14]=src[:14].astype(np.uint8)
    artifact_io.save(out,ROOT)
    stop_frame=13+np.arctan(modified[1]*np.sqrt(modified[3]/modified[2]))/np.sqrt(modified[2]*modified[3])
    print('Fitted original contact deceleration / drag:',motion.x)
    print('Soccer center: initial %.3f, original final %.3f, edited final %.3f'%
          (center[0],center[-1],target[-1]))
    print('Edited stop time: %.3f seconds'%(stop_frame/24))
    print('Saved RGB frames:',out.shape,out.dtype)
    return out


if __name__=='__main__':
    generate()
