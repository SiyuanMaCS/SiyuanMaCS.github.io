+"""Remove the yellow ball and re-solve the surviving ball's rolling motion.
+
+All appearance comes from the supplied RGB sequence. Background reconstruction
+is a visibility-aware temporal median; sphere and shadow sprites are extracted
+from the surviving source sphere, not synthesized with an external renderer.
+"""
+from pathlib import Path
+import numpy as np
+from scipy.ndimage import label, map_coordinates, gaussian_filter
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+
+def track(frames):
+    yy, xx = np.indices(frames.shape[1:3])
+    tracks = []
+    for im in frames.astype(float):
+        r, g, b = im.transpose(2, 0, 1)
+        band = (yy > 48) & (yy < 83)
+        colors = [(r > 130) & (g > 90) & (b < .65*g) & (r > 1.05*g),
+                  (b > 100) & (b > 1.3*r) & (b > 1.15*g)]
+        row = []
+        for color in colors:
+            labs, _ = label(color & band)
+            sizes = np.bincount(labs.ravel()); sizes[0] = 0
+            y, x = np.where(labs == sizes.argmax())
+            row.append([x.mean(), y.mean()])
+        tracks.append(row)
+    return np.asarray(tracks)
+
+
+def plate_from_sequence(source, positions):
+    yy, xx = np.indices(source.shape[1:3])
+    visible = np.ones(source.shape[:3], bool)
+    for t, balls in enumerate(positions):
+        for x, y in balls:
+            # Includes penumbrae and the very faint lower reflected image.
+            visible[t] &= ~((abs(xx-x) < 25) & (yy > y-10) & (yy < 103))
+    plate = np.nanmedian(np.where(visible[..., None], source, np.nan), axis=0)
+    assert np.isfinite(plate).all()
+    return plate.astype(np.float32)
+
+
+def counterfactual_motion(positions):
+    """Pixel-space rolling mechanics calibrated to the original slopes.
+
+    The two straight inclines flank a short nearly level valley. Original
+    downhill acceleration is .23 pixels/frame**2; uphill deceleration includes
+    rolling resistance. No yellow-ball impulses remain in this simulation.
+    """
+    xy = positions[:, 1].copy()
+    x = float(xy[32, 0])
+    # Quadratic fit avoids quantization in the RGB centroids.
+    fit = np.polyfit(np.arange(10, 33), xy[10:33, 0], 2)
+    v = float(2*fit[0]*32 + fit[1])
+    dt = 1/100
+    for t in range(33, len(xy)):
+        for _ in range(100):
+            # Rounded joins, preserving the source's straight side slopes.
+            left = np.clip((114.0-x)/2.0, 0, 1)
+            right = np.clip((x-141.5)/2.0, 0, 1)
+            slope = left-right
+            resistance = .225 * abs(slope)
+            gravity = .46 * slope
+            floor_drag = (.017 + .085*np.exp(-(v/1.7)**2))*(1-abs(slope))
+            accel = gravity - resistance*np.tanh(v/.13) - floor_drag*v
+            v += accel*dt
+            x += v*dt
+        # Sphere-center projection along the unchanged V-shaped track.
+        y = 70.88 - .1395*max(114-x, x-141.5, 0)
+        xy[t] = [x, y]
+    return xy
+
+
+def fade_box(xx, yy, x, y):
+    horizontal = np.clip((25-abs(xx-x))/3, 0, 1)
+    top = np.clip((yy-(y-10))/2, 0, 1)
+    bottom = np.clip((103-yy)/3, 0, 1)
+    return horizontal*top*bottom
+
+
+def sprite_library(source, plate, tracks):
+    """Registered source sphere colors plus multiplicative contact shadows."""
+    # Small supersampled sprites preserve the downsampled source highlights.
+    scale = 4
+    coords = np.arange(-8, 8+1/scale, 1/scale)
+    dy, dx = np.meshgrid(coords, coords, indexing='ij')
+    radius = np.sqrt(dx*dx+dy*dy)
+    sprites = []
+    for t in range(len(source)):
+        x, y = tracks[t, 1]
+        y -= .22  # chromatic centroid is slightly below the geometric center
+        # Estimate opacity from circular coverage. Interior colors are exact.
+        alpha = np.clip((5.55-radius)*scale+.5, 0, 1)
+        pos = [y+dy, x+dx]
+        rgb = np.stack([map_coordinates(source[t,:,:,c].astype(float), pos,
+                                       order=1, mode='nearest') for c in range(3)], -1)
+        bg = np.stack([map_coordinates(plate[:,:,c], pos, order=1,
+                                      mode='nearest') for c in range(3)], -1)
+        # Unmatte the narrow edge against its factual background. At the
+        # silhouette use the neighboring interior color to avoid matte halos.
+        a_geom = np.clip(5.65-radius+.5, .12, 1)
+        raw = (rgb-(1-a_geom[...,None])*bg)/a_geom[...,None]
+        inner_dx = dx*np.minimum(1, 4.7/np.maximum(radius, .01))
+        inner_dy = dy*np.minimum(1, 4.7/np.maximum(radius, .01))
+        inner = np.stack([map_coordinates(source[t,:,:,c].astype(float),
+                  [y+inner_dy, x+inner_dx], order=1, mode='nearest') for c in range(3)],-1)
+        mix = np.clip((radius-4.7)/.8, 0, 1)[..., None]
+        rgb = np.clip(raw*(1-mix)+inner*mix, 0, 255)
+        sprites.append((rgb.astype(np.float32), alpha.astype(np.float32)))
+    return sprites, coords
+
+
+def render_survivor(im, source, plate, tracks, sprites, coords, t, location, velocity):
+    yy, xx = np.indices(im.shape[:2])
+    x, y = location
+    # Match source light/material context by distance along either incline.
+    match_x = x if x >= 127.75 else 255.5-x
+    source_x = tracks[:,1,0]
+    target_speed = abs(velocity)
+    source_speed = np.r_[0, abs(np.diff(source_x))]
+    cost = ((source_x-match_x)/13)**2 + .3*((source_speed-target_speed)/3)**2
+    j = int(cost.argmin())
+    sx, sy = tracks[j,1]
+    # Transport source shadows as ratios, so the original ramp grain and
+    # lighting, rather than a translated patch of background, show through.
+    mirror = x < 127.75
+    qx = sx + (-(xx-x) if mirror else (xx-x))
+    qy = sy + yy-y
+    sampled = np.stack([map_coordinates(source[j,:,:,c].astype(float), [qy,qx],
+                              order=1, mode='nearest') for c in range(3)],-1)
+    original_bg = np.stack([map_coordinates(plate[:,:,c], [qy,qx], order=1,
+                                          mode='nearest') for c in range(3)],-1)
+    attenuation = np.clip(sampled/np.maximum(original_bg,1), .24, 1)
+    body_distance = np.sqrt((xx-x)**2+(yy-(y-.22))**2)
+    # Primary shadow below the tangent point, and faint source reflection.
+    soft = np.clip((yy-(y+5.0))/1.2,0,1)*np.clip((y+12-yy)/2,0,1)
+    soft *= np.clip((23-abs(xx-x))/3,0,1)
+    reflection = np.clip((yy-83)/2,0,1)*np.clip((103-yy)/5,0,1)
+    reflection *= np.clip((12-abs(xx-x))/3,0,1)
+    shade = np.maximum(soft, reflection)
+    shade *= np.clip((body_distance-5.2)*2,0,1)
+    im *= 1-shade[...,None]*(1-attenuation)
+
+    rgb, alpha = sprites[j]
+    # Sampling the premultiplied layer prevents colored silhouette fringes.
+    cy = (yy-(y-.22)-coords[0])*4
+    cx = (xx-x-coords[0])*4
+    a = map_coordinates(alpha,[cy,cx],order=1,mode='constant',cval=0)
+    color = np.stack([map_coordinates(rgb[:,:,c]*alpha,[cy,cx],order=1,
+                             mode='constant',cval=0) for c in range(3)],-1)
+    im[:] = im*(1-a[...,None])+color
+
+
+def generate():
+    source = np.load(ROOT/'source_pixels.npz')['rgb']
+    tracks = track(source)
+    plate = plate_from_sequence(source, tracks)
+    motion = counterfactual_motion(tracks)
+    sprites, coords = sprite_library(source, plate, tracks)
+    yy, xx = np.indices(source.shape[1:3])
+    result = []
+    for t in range(len(source)):
+        im = source[t].astype(np.float32)
+        # Remove both source moving layers with feathered, exact-view repairs.
+        weights = np.zeros(im.shape[:2],np.float32)
+        for x,y in tracks[t]:
+            weights = np.maximum(weights,fade_box(xx,yy,x,y))
+        im = im*(1-weights[...,None])+plate*weights[...,None]
+        if t <= 32:
+            # Purple is causally unaffected before the first original contact.
+            # Copy its complete factual layer, including reflections/shadows.
+            x,y = tracks[t,1]
+            purple = fade_box(xx,yy,x,y)
+            im += (source[t].astype(float)-plate)*purple[...,None]
+        else:
+            v = motion[t,0]-motion[t-1,0]
+            render_survivor(im,source,plate,tracks,sprites,coords,t,motion[t],v)
+        result.append(np.clip(np.rint(im),0,255).astype(np.uint8))
+    frames = np.stack(result)
+    artifact_io.save(frames, ROOT)
+    # Also supply a conventional full-length MP4 for convenient playback.
+    import imageio.v2 as imageio
+    with imageio.get_writer(ROOT/'edited_video.mp4',fps=24,codec='libx264',
+                            quality=9,macro_block_size=1,pixelformat='yuv420p') as writer:
+        for frame in frames: writer.append_data(frame)
+    print('Saved',frames.shape,frames.dtype)
+    print('Survivor path:',[(t,*np.round(motion[t],2)) for t in [0,32,40,48,56,64,72,80,88,95]])
+    return frames
+
+
+if __name__ == '__main__':
+    generate()

        y = 70.88 - .1395*max(114-x, x-141.5, 0)
        xy[t] = [x, y]
    return xy


def fade_box(xx, yy, x, y):
    horizontal = np.clip((25-abs(xx-x))/3, 0, 1)
    top = np.clip((yy-(y-10))/2, 0, 1)
    bottom = np.clip((103-yy)/3, 0, 1)
    return horizontal*top*bottom


def sprite_library(source, plate, tracks):
    \"\"\"Registered source sphere colors plus multiplicative contact shadows.\"\"\"
    # Small supersampled sprites preserve the downsampled source highlights.
    scale = 4
    coords = np.arange(-8, 8+1/scale, 1/scale)
    dy, dx = np.meshgrid(coords, coords, indexing='ij')
    radius = np.sqrt(dx*dx+dy*dy)
    sprites = []
    for t in range(len(source)):
        x, y = tracks[t, 1]
        y -= .22  # chromatic centroid is slightly below the geometric center
        # Estimate opacity from circular coverage. Interior colors are exact.
        alpha = np.clip((5.55-radius)*scale+.5, 0, 1)
        pos = [y+dy, x+dx]
        rgb = np.stack([map_coordinates(source[t,:,:,c].astype(float), pos,
                                       order=1, mode='nearest') for c in range(3)], -1)
        bg = np.stack([map_coordinates(plate[:,:,c], pos, order=1,
                                      mode='nearest') for c in range(3)], -1)
        # Unmatte the narrow edge against its factual background. At the
        # silhouette use the neighboring interior color to avoid matte halos.
        a_geom = np.clip(5.65-radius+.5, .12, 1)
        raw = (rgb-(1-a_geom[...,None])*bg)/a_geom[...,None]
        inner_dx = dx*np.minimum(1, 4.7/np.maximum(radius, .01))
        inner_dy = dy*np.minimum(1, 4.7/np.maximum(radius, .01))
        inner = np.stack([map_coordinates(source[t,:,:,c].astype(float),
                  [y+inner_dy, x+inner_dx], order=1, mode='nearest') for c in range(3)],-1)
        mix = np.clip((radius-4.7)/.8, 0, 1)[..., None]
        rgb = np.clip(raw*(1-mix)+inner*mix, 0, 255)
        sprites.append((rgb.astype(np.float32), alpha.astype(np.float32)))
    return sprites, coords


def render_survivor(im, source, plate, tracks, sprites, coords, t, location, velocity):
    yy, xx = np.indices(im.shape[:2])
    x, y = location
    # Match source light/material context by distance along either incline.
    match_x = x if x >= 127.75 else 255.5-x
    source_x = tracks[:,1,0]
    target_speed = abs(velocity)
    source_speed = np.r_[0, abs(np.diff(source_x))]
    cost = ((source_x-match_x)/13)**2 + .3*((source_speed-target_speed)/3)**2
    j = int(cost.argmin())
    sx, sy = tracks[j,1]
    # Transport source shadows as ratios, so the original ramp grain and
    # lighting, rather than a translated patch of background, show through.
    mirror = x < 127.75
    qx = sx + (-(xx-x) if mirror else (xx-x))
    qy = sy + yy-y
    sampled = np.stack([map_coordinates(source[j,:,:,c].astype(float), [qy,qx],
                              order=1, mode='nearest') for c in range(3)],-1)
    original_bg = np.stack([map_coordinates(plate[:,:,c], [qy,qx], order=1,
                                          mode='nearest') for c in range(3)],-1)
    attenuation = np.clip(sampled/np.maximum(original_bg,1), .24, 1)
    body_distance = np.sqrt((xx-x)**2+(yy-(y-.22))**2)
    # Primary shadow below the tangent point, and faint source reflection.
    soft = np.clip((yy-(y+5.0))/1.2,0,1)*np.clip((y+12-yy)/2,0,1)
    soft *= np.clip((23-abs(xx-x))/3,0,1)
    reflection = np.clip((yy-83)/2,0,1)*np.clip((103-yy)/5,0,1)
    reflection *= np.clip((12-abs(xx-x))/3,0,1)
    shade = np.maximum(soft, reflection)
    shade *= np.clip((body_distance-5.2)*2,0,1)
    im *= 1-shade[...,None]*(1-attenuation)

    rgb, alpha = sprites[j]
    # Sampling the premultiplied layer prevents colored silhouette fringes.
    cy = (yy-(y-.22)-coords[0])*4
    cx = (xx-x-coords[0])*4
    a = map_coordinates(alpha,[cy,cx],order=1,mode='constant',cval=0)
    color = np.stack([map_coordinates(rgb[:,:,c]*alpha,[cy,cx],order=1,
                             mode='constant',cval=0) for c in range(3)],-1)
    im[:] = im*(1-a[...,None])+color


def generate():
    source = np.load(ROOT/'source_pixels.npz')['rgb']
    tracks = track(source)
    plate = plate_from_sequence(source, tracks)
    motion = counterfactual_motion(tracks)
    sprites, coords = sprite_library(source, plate, tracks)
    yy, xx = np.indices(source.shape[1:3])
    result = []
    for t in range(len(source)):
        im = source[t].astype(np.float32)
        # Remove both source moving layers with feathered, exact-view repairs.
        weights = np.zeros(im.shape[:2],np.float32)
        for x,y in tracks[t]:
            weights = np.maximum(weights,fade_box(xx,yy,x,y))
        im = im*(1-weights[...,None])+plate*weights[...,None]
        if t <= 32:
            # Purple is causally unaffected before the first original contact.
            # Copy its complete factual layer, including reflections/shadows.
            x,y = tracks[t,1]
            purple = fade_box(xx,yy,x,y)
            im += (source[t].astype(float)-plate)*purple[...,None]
        else:
            v = motion[t,0]-motion[t-1,0]
            render_survivor(im,source,plate,tracks,sprites,coords,t,motion[t],v)
        result.append(np.clip(np.rint(im),0,255).astype(np.uint8))
    frames = np.stack(result)
    artifact_io.save(frames, ROOT)
    # Also supply a conventional full-length MP4 for convenient playback.
    import imageio.v2 as imageio
    with imageio.get_writer(ROOT/'edited_video.mp4',fps=24,codec='libx264',
                            quality=9,macro_block_size=1,pixelformat='yuv420p') as writer:
        for frame in frames: writer.append_data(frame)
    print('Saved',frames.shape,frames.dtype)
    print('Survivor path:',[(t,*np.round(motion[t],2)) for t in [0,32,40,48,56,64,72,80,88,95]])
    return frames
