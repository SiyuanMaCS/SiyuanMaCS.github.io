+"""Halve the red curling stone's initial velocity and rebuild the edited timeline.
+Uses measured RGB trajectories, a dissipative two-body impact and pixel layers.
+"""
+from pathlib import Path
+import json
+import cv2
+import numpy as np
+from PIL import Image
+from scipy.ndimage import gaussian_filter
+from artifact_io import save
+
+ROOT = Path(__file__).resolve().parent
+

+def decode():
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    frames = []
+    while True:
+        ok, bgr = cap.read()
+        if not ok:
+            break
+        frames.append(bgr[:, :, ::-1].astype(np.float32))
+    cap.release()
+    return np.stack(frames)
+
+
+def clean_background(src):
+    # The camera and the painted ice are stationary. Recover the two initially
+    # covered patches from later frames, when both stones have moved away.
+    bg = src[0].copy()
+    for x0, x1, y0, y1 in [(170, 305, 295, 385), (985, 1140, 345, 455)]:
+        patch = np.median(src[12:30, y0:y1, x0:x1], axis=0)
+        bg[y0:y1, x0:x1] = patch
+    return bg
+
+
+def track(src, bg):
+    pos = np.zeros((len(src), 2, 2), dtype=np.float64)
+    for i, frame in enumerate(src):
+        for c in (0, 1):
+            px = (47 + 1.98 * min(i, 38) - .15 * max(0, i-38)
+                  if c == 0 else 214 - 2.15 * min(i, 38) + .18 * max(0, i-38))
+            py = 66.4 + .061*(px-47) if c == 0 else 76.5 + .06*(px-214)
+            x0, x1 = int((px-8)*5), int((px+8)*5)
+            y0, y1 = int((py-4)*5), int((py+4)*5)
+            p = frame[y0:y1, x0:x1]
+            d = p - bg[y0:y1, x0:x1]
+            yy, xx = np.mgrid[y0:y1, x0:x1]
+            if c == 0:
+                weight = np.maximum(0, d[:,:,0]-d[:,:,1]) * np.clip((p[:,:,0]-p[:,:,1]-55)/50, 0, 1)
+            else:
+                weight = np.maximum(0, .5*(d[:,:,0]+d[:,:,1])-d[:,:,2]) * np.clip((p[:,:,1]-p[:,:,2]-50)/50, 0, 1)
+            pos[i,c] = [(weight*xx).sum()/weight.sum(), (weight*yy).sum()/weight.sum()]
+    return pos
+
+
+def remap(image, mx, my):
+    return cv2.remap(image, mx.astype(np.float32), my.astype(np.float32),
+                     cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)
+
+
+class StoneLayers:
+    def __init__(self, src, bg, pos, c):
+        self.src, self.bg, self.pos, self.c = src, bg, pos, c
+        self.cache = {}
+        # Estimate coverage from aligned RGB observations against differently
+        # colored parts of the target. This separates stone and soft reflection
+        # from the underlying ice without replacing the scene's painted rings.
+        self.rect = (190,300,300,390) if c == 0 else (990,1140,345,455)
+        x0,x1,y0,y1 = self.rect
+        yy,xx = np.mgrid[y0:y1,x0:x1]
+        self.xx, self.yy = xx, yy
+        observations, backgrounds = [], []
+        for i in range(0,29,2):
+            shift = pos[i,c]-pos[0,c]
+            observations.append(remap(src[i], xx+shift[0], yy+shift[1]))
+            backgrounds.append(remap(bg, xx+shift[0], yy+shift[1]))
+        observations, backgrounds = np.array(observations), np.array(backgrounds)
+        db = backgrounds-backgrounds.mean(0)
+        df = observations-observations.mean(0)
+        transparency = np.clip((db*df).sum((0,3))/((db*db).sum((0,3))+1e-4),0,1)
+        alpha = 1-gaussian_filter(transparency,.65)
+        delta = src[0,y0:y1,x0:x1]-bg[y0:y1,x0:x1]
+        contrast = np.max(abs(delta),axis=2)
+        # Suppress codec residue in truly uncovered ice.
+        alpha *= np.clip((contrast-3)/11,0,1)
+        alpha[alpha>.93] = 1
+        self.alpha = np.clip(alpha,0,1).astype(np.float32)
+
+    def layer(self, index):
+        if index in self.cache:
+            return self.cache[index]
+        x0,x1,y0,y1 = self.rect
+        shift = self.pos[index,self.c]-self.pos[0,self.c]
+        mx,my = self.xx+shift[0], self.yy+shift[1]
+        fg = remap(self.src[index],mx,my)
+        ice = remap(self.bg,mx,my)
+        alpha = self.alpha.copy()
+        # Handles and stone bodies are opaque; preserve true source shading.
+        saturation = np.max(fg,2)-np.min(fg,2)
+        contrast = np.max(abs(fg-ice),2)
+        core = (saturation>65)&(contrast>65)&(alpha>.4)
+        alpha[core] = 1
+        premult = fg-(1-alpha[:,:,None])*ice
+        premult[alpha<.005] = 0
+        self.cache[index] = (premult.astype(np.float32),alpha)
+        return self.cache[index]
+
+    def paste(self, canvas, source_time, position):
+        lo = int(np.floor(source_time))
+        hi = min(lo+1,29)
+        frac = source_time-lo
+        p0,a0 = self.layer(lo)
+        if frac > 1e-6:
+            p1,a1 = self.layer(hi)
+            premult = (1-frac)*p0+frac*p1
+            alpha = (1-frac)*a0+frac*a1
+        else:
+            premult,alpha = p0,a0
+        x0,x1,y0,y1 = self.rect
+        delta = position-self.pos[0,self.c]
+        tx0,ty0 = int(np.floor(x0+delta[0]))-2, int(np.floor(y0+delta[1]))-2
+        tx1,ty1 = tx0+(x1-x0)+5, ty0+(y1-y0)+5
+        tx0,ty0 = max(tx0,0),max(ty0,0)
+        tx1,ty1 = min(tx1,canvas.shape[1]),min(ty1,canvas.shape[0])
+        yy,xx = np.mgrid[ty0:ty1,tx0:tx1]
+        mx,my = xx-delta[0]-x0, yy-delta[1]-y0
+        p = remap(premult,mx,my)
+        a = remap(alpha,mx,my)
+        patch = canvas[ty0:ty1,tx0:tx1]
+        patch[:] = patch*(1-a[:,:,None])+p
+
+
+def interpolate(pos, c, t):
+    lo=int(np.floor(t)); hi=min(lo+1,len(pos)-1)
+    return pos[lo,c]*(hi-t)+pos[hi,c]*(t-lo) if hi!=lo else pos[lo,c].copy()
+
+
+def trajectories(pos):
+    # The original impact is between source indices 38 and 40.
+    # Estimate the mass ratio and restitution from measured incident/rebound
+    # velocities. Velocities are native-image pixels per output frame.
+    ur = np.polyfit(np.arange(12,30),pos[12:30,0,0],1)[0]
+    uy = np.polyfit(np.arange(25,39),pos[25:39,1,0],1)[0]
+    from scipy.optimize import least_squares, brentq
+    rebounds=[]
+    for c in (0,1):
+        t=np.arange(40,96)-40
+        x=pos[40:96,c,0]
+        def residual(z):
+            return z[0]+z[1]*(-np.expm1(-z[2]*t))/z[2]-x
+        fit=least_squares(residual,[x[0],-1 if c==0 else 1,.008],
+                          bounds=([-np.inf,-np.inf,.0001],[np.inf,np.inf,.1]))
+        rebounds.append((fit.x[1],fit.x[2]))
+    vr,vy=rebounds[0][0],rebounds[1][0]
+    mass_ratio=(ur-vr)/(vy-uy)
+    restitution=np.clip((vy-vr)/(ur-uy),.05,.3)
+    # Red is retimed only prior to the new collision. Yellow follows the
+    # unchanged factual history until the causal influence reaches it.
+    yslope=np.polyfit(pos[12:38,1,0],pos[12:38,1,1],1)[0]
+    yvel=np.polyfit(np.arange(28,39),pos[28:39,1,0],1)[0]
+    def incident(t):
+        red=interpolate(pos,0,t*.5)
+        if t<=38:
+            yellow=interpolate(pos,1,t)
+        else:
+            dx=yvel*(t-38)
+            yellow=pos[38,1]+np.array([dx,yslope*dx])
+        return np.array([red,yellow])
+    contact_sep=pos[38,1,0]-pos[38,0,0]
+    tc=brentq(lambda t: incident(t)[1,0]-incident(t)[0,0]-contact_sep,40,60)
+    at_contact=incident(tc)
+    h=.2
+    velocities=(incident(tc+h)-incident(tc-h))/(2*h)
+    u,v=velocities[:,0]
+    after_r=(u+mass_ratio*v-mass_ratio*restitution*(u-v))/(1+mass_ratio)
+    after_y=(u+mass_ratio*v+restitution*(u-v))/(1+mass_ratio)
+    rslope=velocities[0,1]/u
+    rates=np.array([rebounds[0][1],rebounds[1][1]])
+    result=[]
+    for i in range(96):
+        if i<=tc:
+            p=incident(i)
+        else:
+            dt=i-tc
+            dx=np.array([after_r,after_y])*(-np.expm1(-rates*dt))/rates
+            p=at_contact+np.stack([dx,dx*np.array([rslope,yslope])],1)
+        result.append(p)
+    report={'collision_frame_1based':float(tc+1),'initial_red_speed_factor':.5,
+            'mass_ratio_yellow_to_red':float(mass_ratio),'restitution':float(restitution),
+            'postimpact_horizontal_velocity_pixels_per_frame':[float(after_r/5),float(after_y/5)],
+            'final_positions_pixels':(np.array(result)[-1]/5).tolist()}
+    return np.array(result),tc,report
+
+
+def main():
+    source=decode()
+    bg=clean_background(source)
+    measured=track(source,bg)
+    path,tc,report=trajectories(measured)
+    stones=[StoneLayers(source,bg,measured,c) for c in (0,1)]
+    output=[]
+    for i in range(96):
+        canvas=bg.copy()
+        # Source appearance is interpolated smoothly in local object space.
+        red_time=min(.5*min(i,tc),29)
+        yellow_time=min(i,29)
+        for c in np.argsort(path[i,:,1]):
+            stones[c].paste(canvas,red_time if c==0 else yellow_time,path[i,c])
+        frame=np.array(Image.fromarray(np.clip(np.rint(canvas),0,255).astype(np.uint8)).resize((256,144),Image.Resampling.LANCZOS))
+        output.append(frame)
+    frames=np.stack(output)
+    # At execution time speed changes, never the initial configuration.
+    frames[0]=np.load(ROOT/'source_pixels.npz')['rgb'][0]
+    save(frames,ROOT)
+    (ROOT/'motion_report.json').write_text(json.dumps(report,indent=2))
+    print(json.dumps(report,indent=2))
+    print('Saved complete RGB video:',frames.shape,frames.dtype)
+
+
+if __name__=='__main__':
+    main()

Uses measured RGB trajectories, a dissipative two-body impact and pixel layers.

from pathlib import Path
import json
import cv2
import numpy as np
from PIL import Image
from scipy.ndimage import gaussian_filter
from artifact_io import save

ROOT = Path(__file__).resolve().parent


def decode():
    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
    frames = []
    while True:
        ok, bgr = cap.read()
        if not ok:
            break
        frames.append(bgr[:, :, ::-1].astype(np.float32))
    cap.release()
    return np.stack(frames)


def clean_background(src):
    # The camera and the painted ice are stationary. Recover the two initially
    # covered patches from later frames, when both stones have moved away.
    bg = src[0].copy()
    for x0, x1, y0, y1 in [(170, 305, 295, 385), (985, 1140, 345, 455)]:
        patch = np.median(src[12:30, y0:y1, x0:x1], axis=0)
        bg[y0:y1, x0:x1] = patch
    return bg


def track(src, bg):
    pos = np.zeros((len(src), 2, 2), dtype=np.float64)
    for i, frame in enumerate(src):
        for c in (0, 1):
            px = (47 + 1.98 * min(i, 38) - .15 * max(0, i-38)
                  if c == 0 else 214 - 2.15 * min(i, 38) + .18 * max(0, i-38))
            py = 66.4 + .061*(px-47) if c == 0 else 76.5 + .06*(px-214)
            x0, x1 = int((px-8)*5), int((px+8)*5)
            y0, y1 = int((py-4)*5), int((py+4)*5)
            p = frame[y0:y1, x0:x1]
            d = p - bg[y0:y1, x0:x1]
            yy, xx = np.mgrid[y0:y1, x0:x1]
            if c == 0:
                weight = np.maximum(0, d[:,:,0]-d[:,:,1]) * np.clip((p[:,:,0]-p[:,:,1]-55)/50, 0, 1)
            else:
                weight = np.maximum(0, .5*(d[:,:,0]+d[:,:,1])-d[:,:,2]) * np.clip((p[:,:,1]-p[:,:,2]-50)/50, 0, 1)
            pos[i,c] = [(weight*xx).sum()/weight.sum(), (weight*yy).sum()/weight.sum()]
    return pos


def remap(image, mx, my):
    return cv2.remap(image, mx.astype(np.float32), my.astype(np.float32),
                     cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)


class StoneLayers:
    def __init__(self, src, bg, pos, c):
        self.src, self.bg, self.pos, self.c = src, bg, pos, c
        self.cache = {}
        # Estimate coverage from aligned RGB observations against differently
        # colored parts of the target. This separates stone and soft reflection
        # from the underlying ice without replacing the scene's painted rings.
        self.rect = (190,300,300,390) if c == 0 else (990,1140,345,455)
        x0,x1,y0,y1 = self.rect
        yy,xx = np.mgrid[y0:y1,x0:x1]
        self.xx, self.yy = xx, yy
        observations, backgrounds = [], []
        for i in range(0,29,2):
            shift = pos[i,c]-pos[0,c]
            observations.append(remap(src[i], xx+shift[0], yy+shift[1]))
            backgrounds.append(remap(bg, xx+shift[0], yy+shift[1]))
        observations, backgrounds = np.array(observations), np.array(backgrounds)
        db = backgrounds-backgrounds.mean(0)
        df = observations-observations.mean(0)
        transparency = np.clip((db*df).sum((0,3))/((db*db).sum((0,3))+1e-4),0,1)
        alpha = 1-gaussian_filter(transparency,.65)
        delta = src[0,y0:y1,x0:x1]-bg[y0:y1,x0:x1]
        contrast = np.max(abs(delta),axis=2)
        # Suppress codec residue in truly uncovered ice.
        alpha *= np.clip((contrast-3)/11,0,1)
        alpha[alpha>.93] = 1
        self.alpha = np.clip(alpha,0,1).astype(np.float32)

    def layer(self, index):
        if index in self.cache:
            return self.cache[index]
        x0,x1,y0,y1 = self.rect
        shift = self.pos[index,self.c]-self.pos[0,self.c]
        mx,my = self.xx+shift[0], self.yy+shift[1]
        fg = remap(self.src[index],mx,my)
        ice = remap(self.bg,mx,my)
        alpha = self.alpha.copy()
        # Handles and stone bodies are opaque; preserve true source shading.
        saturation = np.max(fg,2)-np.min(fg,2)
        contrast = np.max(abs(fg-ice),2)
        core = (saturation>65)&(contrast>65)&(alpha>.4)
        alpha[core] = 1
        premult = fg-(1-alpha[:,:,None])*ice
        premult[alpha<.005] = 0
        self.cache[index] = (premult.astype(np.float32),alpha)
        return self.cache[index]

    def paste(self, canvas, source_time, position):
        lo = int(np.floor(source_time))
        hi = min(lo+1,29)
        frac = source_time-lo
        p0,a0 = self.layer(lo)
        if frac > 1e-6:
            p1,a1 = self.layer(hi)
            premult = (1-frac)*p0+frac*p1
            alpha = (1-frac)*a0+frac*a1
        else:
            premult,alpha = p0,a0
        x0,x1,y0,y1 = self.rect
        delta = position-self.pos[0,self.c]
        tx0,ty0 = int(np.floor(x0+delta[0]))-2, int(np.floor(y0+delta[1]))-2
        tx1,ty1 = tx0+(x1-x0)+5, ty0+(y1-y0)+5
        tx0,ty0 = max(tx0,0),max(ty0,0)
        tx1,ty1 = min(tx1,canvas.shape[1]),min(ty1,canvas.shape[0])
        yy,xx = np.mgrid[ty0:ty1,tx0:tx1]
        mx,my = xx-delta[0]-x0, yy-delta[1]-y0
        p = remap(premult,mx,my)
        a = remap(alpha,mx,my)
        patch = canvas[ty0:ty1,tx0:tx1]
        patch[:] = patch*(1-a[:,:,None])+p


def interpolate(pos, c, t):
    lo=int(np.floor(t)); hi=min(lo+1,len(pos)-1)
    return pos[lo,c]*(hi-t)+pos[hi,c]*(t-lo) if hi"'!=lo else pos[lo,c].copy()


def trajectories(pos):
    # The original impact is between source indices 38 and 40.
    # Estimate the mass ratio and restitution from measured incident/rebound
    # velocities. Velocities are native-image pixels per output frame.
    ur = np.polyfit(np.arange(12,30),pos[12:30,0,0],1)[0]
    uy = np.polyfit(np.arange(25,39),pos[25:39,1,0],1)[0]
    from scipy.optimize import least_squares, brentq
    rebounds=[]
    for c in (0,1):
        t=np.arange(40,96)-40
        x=pos[40:96,c,0]
        def residual(z):
            return z[0]+z[1]*(-np.expm1(-z[2]*t))/z[2]-x
        fit=least_squares(residual,[x[0],-1 if c==0 else 1,.008],
                          bounds=([-np.inf,-np.inf,.0001],[np.inf,np.inf,.1]))
        rebounds.append((fit.x[1],fit.x[2]))
    vr,vy=rebounds[0][0],rebounds[1][0]
    mass_ratio=(ur-vr)/(vy-uy)
    restitution=np.clip((vy-vr)/(ur-uy),.05,.3)
    # Red is retimed only prior to the new collision. Yellow follows the
    # unchanged factual history until the causal influence reaches it.
    yslope=np.polyfit(pos[12:38,1,0],pos[12:38,1,1],1)[0]
    yvel=np.polyfit(np.arange(28,39),pos[28:39,1,0],1)[0]
    def incident(t):
        red=interpolate(pos,0,t*.5)
        if t<=38:
            yellow=interpolate(pos,1,t)
        else:
            dx=yvel*(t-38)
            yellow=pos[38,1]+np.array([dx,yslope*dx])
        return np.array([red,yellow])
    contact_sep=pos[38,1,0]-pos[38,0,0]
    tc=brentq(lambda t: incident(t)[1,0]-incident(t)[0,0]-contact_sep,40,60)
    at_contact=incident(tc)
    h=.2
    velocities=(incident(tc+h)-incident(tc-h))/(2*h)
    u,v=velocities[:,0]
    after_r=(u+mass_ratio*v-mass_ratio*restitution*(u-v))/(1+mass_ratio)
    after_y=(u+mass_ratio*v+restitution*(u-v))/(1+mass_ratio)
    rslope=velocities[0,1]/u
    rates=np.array([rebounds[0][1],rebounds[1][1]])
    result=[]
    for i in range(96):
        if i<=tc:
            p=incident(i)
        else:
            dt=i-tc
            dx=np.array([after_r,after_y])*(-np.expm1(-rates*dt))/rates
            p=at_contact+np.stack([dx,dx*np.array([rslope,yslope])],1)
        result.append(p)
    report={'"'collision_frame_1based':float(tc+1),'initial_red_speed_factor':.5,
            'mass_ratio_yellow_to_red':float(mass_ratio),'restitution':float(restitution),
            'postimpact_horizontal_velocity_pixels_per_frame':[float(after_r/5),float(after_y/5)],
            'final_positions_pixels':(np.array(result)[-1]/5).tolist()}
    return np.array(result),tc,report


def main():
    source=decode()
    bg=clean_background(source)
    measured=track(source,bg)
    path,tc,report=trajectories(measured)
    stones=[StoneLayers(source,bg,measured,c) for c in (0,1)]
    output=[]
    for i in range(96):
        canvas=bg.copy()
        # Source appearance is interpolated smoothly in local object space.
        red_time=min(.5*min(i,tc),29)
        yellow_time=min(i,29)
        for c in np.argsort(path[i,:,1]):
            stones[c].paste(canvas,red_time if c==0 else yellow_time,path[i,c])
        frame=np.array(Image.fromarray(np.clip(np.rint(canvas),0,255).astype(np.uint8)).resize((256,144),Image.Resampling.LANCZOS))
        output.append(frame)
    frames=np.stack(output)
    # At execution time speed changes, never the initial configuration.
    frames[0]=np.load(ROOT/'source_pixels.npz')['rgb'][0]
    save(frames,ROOT)
    (ROOT/'motion_report.json').write_text(json.dumps(report,indent=2))
    print(json.dumps(report,indent=2))
    print('Saved complete RGB video:',frames.shape,frames.dtype)


if __name__=='__main__':
    main()
