+No renderer or physics engine: the rolling/contact dynamics are integrated here,
+and the two marbles' appearances and shadows are resampled from source pixels.
+"""
+from pathlib import Path
+import json
+import numpy as np
+from scipy.ndimage import label, map_coordinates
+from scipy.optimize import minimize
+from artifact_io import save
+
+ROOT = Path(__file__).resolve().parent

+def track_marble(frames):
+    yy, xx = np.indices(frames.shape[1:3])
+    out = []
+    for frame in frames:
+        r, g, b = frame.astype(float).transpose(2, 0, 1)
+        masks = [(b > r * 1.1) & (b > g * 1.2) & (b > 50),
+                 (r > g * 1.05) & (g > b * 1.6) & (g > 70) &
+                 (yy > 52) & (yy < 78) & (xx < 220)]
+        row = []
+        for mask in masks:
+            labs, n = label(mask)
+            counts = np.bincount(labs.ravel()); counts[0] = 0
+            y, x = np.where(labs == np.argmax(counts))
+            row.append((x.mean(), y.mean()))
+        out.append(row)
+    return np.asarray(out)
+
+
+def simulate(params, masses=(1., 1.), substeps=60):
+    # Units are image pixels and original frame intervals. Solid-sphere inertia.
+    gravity, friction, resistance, restitution, diameter, speed, spin = params
+    inertia = .4
+    x = [137.04, 118.50]
+    v = [-speed, speed]
+    w = [-spin, spin]
+    inv = [1. / masses[0], 1. / masses[1]]
+    result = [x.copy()]
+    dt = 1. / substeps
+    for frame in range(34, 96):
+        for step in range(substeps):
+            for j in range(2):
+                grav = gravity if x[j] < 113.4 else -gravity if x[j] > 141.4 else 0.
+                slip = v[j] - w[j]
+                # Static contact enforces rolling. Sliding contact dissipates slip
+                # without spuriously changing the inertia when mass is increased.
+                if abs(slip) < (3.5 * friction + abs(grav)) * dt * 1.3:
+                    v[j] = (v[j] + inertia * w[j]) / (1. + inertia)
+                    w[j] = v[j]
+                    if abs(v[j]) < resistance * dt / 1.4 and abs(grav) < resistance:
+                        v[j] = w[j] = 0.
+                    else:
+                        rr = resistance * np.sign(v[j]) if abs(v[j]) > .00001 else 0.
+                        acc = (grav - rr) / (1. + inertia)
+                        v[j] += acc * dt
+                        w[j] = v[j]
+                else:
+                    ff = -friction * np.sign(slip)
+                    v[j] += (grav + ff) * dt
+                    w[j] -= ff * dt / inertia
+                x[j] += v[j] * dt
+            if x[0] - x[1] < diameter:
+                overlap = diameter - (x[0] - x[1])
+                x[0] += overlap * inv[0] / sum(inv)
+                x[1] -= overlap * inv[1] / sum(inv)
+                relative = v[0] - v[1]
+                if relative < 0.:
+                    impulse = -(1. + restitution) * relative / sum(inv)
+                    v[0] += impulse * inv[0]
+                    v[1] -= impulse * inv[1]
+                    # Smooth spherical contact is central: spin is retained.
+        result.append(x.copy())
+    return np.asarray(result)
+
+
+def fit_dynamics(centers):
+    truth = centers[33:, :, 0]
+    # Fit mostly to the first rebound and return, avoiding pixel-centroid biases
+    # dominating the late static equilibrium.
+    weights = np.ones_like(truth)
+    weights[30:] *= .35
+    def objective(p):
+        prediction = simulate(p, substeps=32)
+        # Removing the small color-dependent center bias makes the fit symmetric.
+        target = truth - np.mean(truth, axis=1, keepdims=True) + 127.77
+        return np.mean((prediction - target) ** 2 * weights)
+    initial = [.34, .45, .10, .68, 11.8, 5.95, 6.35]
+    bounds = [(.29, .42), (.25, .7), (.025, .22), (.45, .85),
+              (11.1, 12.5), (5.55, 6.35), (5.9, 6.8)]
+    fit = minimize(objective, initial, method='Nelder-Mead', bounds=bounds,
+                   options={'maxiter': 380, 'xatol': .001, 'fatol': .0001})
+    print('Recovered contact parameters:', fit.x.round(5).tolist(),
+          'source tracking MSE:', round(float(fit.fun), 4), flush=True)
+    return fit.x
+

+def remove_moving_objects(source, centers):
+    """Recover static pixels by temporal visibility, including cast shadows."""
+    yy, xx = np.indices(source.shape[1:3])
+    masked = source.astype(np.float32)
+    erase = []
+    for t, c in enumerate(centers):
+        mask = np.zeros(yy.shape, bool)
+        for j, (x, y) in enumerate(c):
+            # Covers both the antialiased sphere and its displaced soft shadow.
+            sphere = ((xx-x)/7.7)**2 + ((yy-y)/7.8)**2 < 1
+            shadow_x = x + (x - 127.5) * .075
+            shadow = ((xx-shadow_x)/16.)**2 + ((yy-y-8.5)/6.5)**2 < 1
+            mask |= sphere | shadow
+        erase.append(mask)
+        masked[t][mask] = np.nan
+    import warnings
+    with warnings.catch_warnings():
+        warnings.simplefilter('ignore', RuntimeWarning)
+        bg = np.nanmedian(masked, axis=0)
+    missing = np.isnan(bg[:, :, 0])
+    bg[missing] = source[0][missing]
+    return np.clip(bg, 0, 255), np.asarray(erase)
+
+
+def height(x):
+    if x < 113.4:
+        return 70.96 + .1395 * (x - 113.4)
+    if x > 141.4:
+        return 70.96 - .1390 * (x - 141.4)
+    return 70.96
+

+def sphere_template(source, background, centers, t, j, flip=False):
+    x, y = centers[t, j]
+    # Small bias between chromatic centroid and the geometric sphere center.
+    x += .35 if j == 1 else -.12
+    y += .04
+    gy, gx = np.mgrid[-10:17, -20:21].astype(float)
+    sample_x = x - gx if flip else x + gx
+    sample_y = y + gy
+    coords = [sample_y, sample_x]
+    tex = np.stack([map_coordinates(source[t, :, :, k].astype(float), coords,
+                                   order=1, mode='nearest') for k in range(3)], -1)
+    clean = np.stack([map_coordinates(background[:, :, k], coords,
+                                     order=1, mode='nearest') for k in range(3)], -1)
+    # Supersampled silhouette. The source itself supplies color, highlights and
+    # subtle mottling; mass does not change the sphere's radius or material.
+    alpha = np.zeros(gx.shape)
+    for ay in (-.375, -.125, .125, .375):
+        for ax in (-.375, -.125, .125, .375):
+            alpha += (((gx+ax)/5.77)**2 + ((gy+ay)/5.83)**2 <= 1)
+    alpha /= 16.
+    color = (tex - (1-alpha[:, :, None]) * clean) / np.maximum(alpha[:, :, None], .08)
+    color = np.clip(color, 0, 255)
+    # The cast shadow is source-derived attenuation, not a pasted background.
+    ratio = np.median((tex+3.)/(clean+3.), axis=2)
+    attenuation = np.clip(1-ratio, 0, .80)
+    attenuation[(gy < 5.3) | (gx**2+(gy*.99)**2 < 6.0**2)] = 0.
+    attenuation[attenuation < .025] = 0.
+    return color, alpha, attenuation
+
+
+def paste_object(frame, x, y, sprite):
+    color, alpha, shadow = sprite
+    iy, ix = np.indices(frame.shape[:2])
+    u, v = ix-x+20., iy-y+10.
+    use = (u >= 0) & (u <= 40) & (v >= 0) & (v <= 26)
+    sy, sx = v[use], u[use]
+    a = map_coordinates(alpha, [sy, sx], order=1, mode='constant')
+    s = map_coordinates(shadow, [sy, sx], order=1, mode='constant')
+    c = np.stack([map_coordinates(color[:, :, k], [sy, sx], order=1,
+                                 mode='nearest') for k in range(3)], -1)
+    under = frame[use] * (1-s[:, None])
+    frame[use] = under * (1-a[:, None]) + c * a[:, None]
+
+
+def generate():
+    source = np.load(ROOT/'source_pixels.npz')['rgb']
+    centers = track_marble(source)
+    params = fit_dynamics(centers)
+    positions = simulate(params, masses=(5., 1.), substeps=100)
+    # Reconcile the fitted symmetric release with the actual observed centers.
+    positions += centers[33, :, 0] - positions[0]
+    background, erase = remove_moving_objects(source, centers)
+    output = source.copy()
+    cache = {}
+    for t in range(34, 96):
+        frame = source[t].astype(float)
+        frame[erase[t]] = background[erase[t]]
+        coords = positions[t-33]
+        # A frame's two shadows are drawn under both marbles; draw farther one
+        # first. Source shading is selected at the closest matching ramp position.
+        for j in (0, 1):
+            x = coords[j]
+            mirrored = j == 0 and x < 127.5
+            reference_x = 255.0-x if mirrored else x
+            matches = np.abs(centers[:, j, 0]-reference_x)
+            # Prefer an early moving sample when several identical positions exist.
+            ref = int(np.argmin(matches + np.arange(96)*.00001))
+            key = (ref, j, mirrored)
+            if key not in cache:
+                cache[key] = sphere_template(source, background, centers, ref, j, mirrored)
+            paste_object(frame, x + (.35 if j == 1 else -.12), height(x), cache[key])
+        output[t] = np.clip(np.rint(frame), 0, 255).astype(np.uint8)
+    # Frame 1 begins at the unchanged position; increased inertia is first visible
+    # at the first marble-to-marble impact, not as a spurious gravity change.
+    save(output, ROOT)
+    np.save(ROOT/'counterfactual_positions.npy', positions)
+    (ROOT/'generation_notes.json').write_text(json.dumps({
+        'mass_multiplier': 5, 'intervention_frame_1based': 1,
+        'contact_parameters': params.tolist(),
+        'purple_x_range': [float(positions[:,0].min()),float(positions[:,0].max())],
+        'yellow_x_range': [float(positions[:,1].min()),float(positions[:,1].max())]
+    }, indent=2))
+    print('Saved 96 complete RGB frames and player.', flush=True)
+    for t in (33,34,38,42,46,50,56,64,72,80,88,95):
+        print('frame',t,'marble x',positions[t-33].round(2).tolist(), flush=True)
+
+if __name__ == '__main__':
+    generate()

            sphere = ((xx-x)/7.7)**2 + ((yy-y)/7.8)**2 < 1
            shadow_x = x + (x - 127.5) * .075
            shadow = ((xx-shadow_x)/16.)**2 + ((yy-y-8.5)/6.5)**2 < 1
            mask |= sphere | shadow
        erase.append(mask)
        masked[t][mask] = np.nan
    import warnings
    with warnings.catch_warnings():
        warnings.simplefilter('ignore', RuntimeWarning)
        bg = np.nanmedian(masked, axis=0)
    missing = np.isnan(bg[:, :, 0])
    bg[missing] = source[0][missing]
    return np.clip(bg, 0, 255), np.asarray(erase)


def height(x):
    if x < 113.4:
        return 70.96 + .1395 * (x - 113.4)
    if x > 141.4:
        return 70.96 - .1390 * (x - 141.4)
    return 70.96


def sphere_template(source, background, centers, t, j, flip=False):
    x, y = centers[t, j]
    # Small bias between chromatic centroid and the geometric sphere center.
    x += .35 if j == 1 else -.12
    y += .04
    gy, gx = np.mgrid[-10:17, -20:21].astype(float)
    sample_x = x - gx if flip else x + gx
    sample_y = y + gy
    coords = [sample_y, sample_x]
    tex = np.stack([map_coordinates(source[t, :, :, k].astype(float), coords,
                                   order=1, mode='nearest') for k in range(3)], -1)
    clean = np.stack([map_coordinates(background[:, :, k], coords,
                                     order=1, mode='nearest') for k in range(3)], -1)
    # Supersampled silhouette. The source itself supplies color, highlights and
    # subtle mottling; mass does not change the sphere's radius or material.
    alpha = np.zeros(gx.shape)
    for ay in (-.375, -.125, .125, .375):
        for ax in (-.375, -.125, .125, .375):
            alpha += (((gx+ax)/5.77)**2 + ((gy+ay)/5.83)**2 <= 1)
    alpha /= 16.
    color = (tex - (1-alpha[:, :, None]) * clean) / np.maximum(alpha[:, :, None], .08)
    color = np.clip(color, 0, 255)
    # The cast shadow is source-derived attenuation, not a pasted background.
    ratio = np.median((tex+3.)/(clean+3.), axis=2)
    attenuation = np.clip(1-ratio, 0, .80)
    attenuation[(gy < 5.3) | (gx**2+(gy*.99)**2 < 6.0**2)] = 0.
    attenuation[attenuation < .025] = 0.
    return color, alpha, attenuation


def paste_object(frame, x, y, sprite):
    color, alpha, shadow = sprite
    iy, ix = np.indices(frame.shape[:2])
    u, v = ix-x+20., iy-y+10.
    use = (u >= 0) & (u <= 40) & (v >= 0) & (v <= 26)
    sy, sx = v[use], u[use]
    a = map_coordinates(alpha, [sy, sx], order=1, mode='constant')
    s = map_coordinates(shadow, [sy, sx], order=1, mode='constant')
    c = np.stack([map_coordinates(color[:, :, k], [sy, sx], order=1,
                                 mode='nearest') for k in range(3)], -1)
    under = frame[use] * (1-s[:, None])
    frame[use] = under * (1-a[:, None]) + c * a[:, None]


def generate():
    source = np.load(ROOT/'source_pixels.npz')['rgb']
    centers = track_marble(source)
    params = fit_dynamics(centers)
    positions = simulate(params, masses=(5., 1.), substeps=100)
    # Reconcile the fitted symmetric release with the actual observed centers.
    positions += centers[33, :, 0] - positions[0]
    background, erase = remove_moving_objects(source, centers)
    output = source.copy()
    cache = {}
    for t in range(34, 96):
        frame = source[t].astype(float)
        frame[erase[t]] = background[erase[t]]
        coords = positions[t-33]
        # A frame's two shadows are drawn under both marbles; draw farther one
        # first. Source shading is selected at the closest matching ramp position.
        for j in (0, 1):
            x = coords[j]
            mirrored = j == 0 and x < 127.5
            reference_x = 255.0-x if mirrored else x
            matches = np.abs(centers[:, j, 0]-reference_x)
            # Prefer an early moving sample when several identical positions exist.
            ref = int(np.argmin(matches + np.arange(96)*.00001))
            key = (ref, j, mirrored)
            if key not in cache:
                cache[key] = sphere_template(source, background, centers, ref, j, mirrored)
            paste_object(frame, x + (.35 if j == 1 else -.12), height(x), cache[key])
        output[t] = np.clip(np.rint(frame), 0, 255).astype(np.uint8)
    # Frame 1 begins at the unchanged position; increased inertia is first visible
    # at the first marble-to-marble impact, not as a spurious gravity change.
    save(output, ROOT)
    np.save(ROOT/'counterfactual_positions.npy', positions)
    (ROOT/'generation_notes.json').write_text(json.dumps({
        'mass_multiplier': 5, 'intervention_frame_1based': 1,
        'contact_parameters': params.tolist(),
        'purple_x_range': [float(positions[:,0].min()),float(positions[:,0].max())],
        'yellow_x_range': [float(positions[:,1].min()),float(positions[:,1].max())]
    }, indent=2))
    print('Saved 96 complete RGB frames and player.', flush=True)
    for t in (33,34,38,42,46,50,56,64,72,80,88,95):
        print('frame',t,'marble x',positions[t-33].round(2).tolist(), flush=True)

if __name__ == '__main__':
    generate()
