+"""Counterfactual carton removal using only source RGB, matting and mechanics.
+
+The sliding can, carton and bottle have nearly inelastic collisions. Their
+observed velocity changes determine relative masses; deleting the intervening
+carton eliminates its first collision and brings the can straight to the bottle.
+No models or external assets are used.
+"""
+from pathlib import Path
+import numpy as np
+from scipy.ndimage import map_coordinates, gaussian_filter
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+SOURCE = np.load(ROOT / 'source_pixels.npz')['rgb']
+H, W = SOURCE.shape[1:3]
+Y, X = np.indices((H, W))
+
+
+def clean_background():
+    # All occluded parts of the two stationary plant pots and the tabletop
+    # become visible later; the final resting objects' patch is visible early.
+    b = np.median(SOURCE[40:].astype(np.float32), axis=0)
+    b[:, 204:] = np.median(SOURCE[:4, :, 204:].astype(np.float32), axis=0)
+    return b
+
+
+BG = clean_background()
+
+
+def can_position(frame):
+    a = SOURCE[frame].astype(float)
+    redness = np.clip(a[69:75, :, 0] - a[69:75, :, 1] - 23, 0, None).mean(0)
+    select = redness > 20
+    return float(np.sum(np.arange(W)[select] * redness[select]) / redness[select].sum())
+
+
+CAN_X = np.array([can_position(i) for i in range(36)])
+# The source bottle is seen unobstructed at the beginning.
+BOTTLE_X = 169.37
+
+
+def shape_alpha(kind, cx, x0, y0, width, height):
+    """Pixel-integrated projected rigid silhouettes; edges are antialiased."""
+    scale = 6
+    sy = y0 + (np.arange(height * scale) + .5) / scale - .5
+    sx = x0 + (np.arange(width * scale) + .5) / scale - .5
+    yy, xx = np.meshgrid(sy, sx, indexing='ij')
+    q = xx - cx
+    if kind == 'can':
+        # Rounded ends of the upright can, including its pale upper rim.
+        q -= .12
+        top = ((q / 5.16) ** 2 + ((yy - 65.25) / 1.13) ** 2 <= 1)
+        mid = (abs(q) <= 5.16) & (yy >= 65.25) & (yy <= 80.45)
+        bot = ((q / 5.16) ** 2 + ((yy - 80.45) / 1.12) ** 2 <= 1)
+        alpha = (top | mid | bot).astype(float)
+    else:
+        cap = (q >= -2.87) & (q <= 1.22) & (yy >= 63.78) & (yy <= 66.04)
+        neck = (q >= -2.98) & (q <= 1.65) & (yy > 66.04) & (yy < 66.52)
+        shoulder_r = 1.65 + np.clip((yy - 66.15) / .90, 0, 1) * 3.28
+        shoulder = (q >= -2.98) & (q <= shoulder_r) & (yy >= 66.15) & (yy <= 67.15)
+        body = (q >= -2.98) & (q <= 4.93) & (yy >= 67.0) & (yy <= 80.95)
+        bottom = ((q - .975) / 3.955) ** 2 + ((yy - 80.95) / 1.48) ** 2 <= 1
+        alpha = (cap | neck | shoulder | body | bottom).astype(float)
+    alpha = alpha.reshape(height, scale, width, scale).mean((1, 3))
+    return gaussian_filter(alpha, .28)
+
+
+def extract_sprite(frame, kind, cx):
+    x0, y0, width, height = int(np.floor(cx)) - 9, 62, 22, 23
+    alpha = shape_alpha(kind, cx, x0, y0, width, height)
+    rgb = SOURCE[frame, y0:y0 + height, x0:x0 + width].astype(float)
+    b = BG[y0:y0 + height, x0:x0 + width]
+    # Unmix antialiased edges rather than copying their old background.
+    foreground = (rgb - (1 - alpha[:, :, None]) * b) / np.maximum(alpha[:, :, None], .018)
+    foreground = np.clip(foreground, 0, 255)
+    premult = foreground * alpha[:, :, None]
+    return (premult, alpha, x0, y0, cx)
+
+
+CANS = [extract_sprite(i, 'can', CAN_X[i]) for i in range(36)]
+BOTTLE = extract_sprite(0, 'bottle', BOTTLE_X)
+
+
+def warp_sprite(sprite, cx):
+    premult, alpha, x0, y0, old_cx = sprite
+    shift = cx - old_cx
+    left = max(0, int(np.floor(x0 + shift)) - 1)
+    right = min(W, int(np.ceil(x0 + shift + alpha.shape[1])) + 1)
+    if right <= left:
+        return None
+    yy, xx = np.indices((alpha.shape[0], right - left), dtype=float)
+    coords = np.array([yy, xx + left - x0 - shift])
+    a = map_coordinates(alpha, coords, order=1, mode='constant', cval=0)
+    p = np.stack([map_coordinates(premult[:, :, c], coords, order=1,
+                                  mode='constant', cval=0) for c in range(3)], axis=2)
+    return left, right, y0, a, p
+
+
+def shadow(canvas, cx, kind):
+    if cx < -18 or cx > W + 18:
+        return
+    strength = .145 if kind == 'can' else .11
+    sigma_x = 5.7 if kind == 'can' else 5.1
+    a = strength * np.exp(-.5 * (((X - cx + 2.1) / sigma_x) ** 2 + ((Y - 82.2) / .62) ** 2))
+    canvas *= (1 - a[:, :, None])
+
+
+def paste_sprite(canvas, sprite, cx, brightness=1):
+    warped = warp_sprite(sprite, cx)
+    if warped is None:
+        return
+    left, right, top, alpha, premult = warped
+    region = canvas[top:top + len(alpha), left:right]
+    # Small spatial light variation inferred from the observed green label.
+    if brightness != 1:
+        premult = np.minimum(premult * brightness, alpha[:, :, None] * 255)
+    region[:] = premult + region * (1 - alpha[:, :, None])
+
+
+def paste_can(canvas, cx):
+    # Interpolate source appearance in *position*, preserving the label,
+    # surface highlights and the spatial change of illumination.
+    if cx < CAN_X[0]:
+        lo = hi = 0
+    elif cx >= CAN_X[-1]:
+        lo = hi = 35
+    else:
+        hi = int(np.searchsorted(CAN_X, cx))
+        lo = hi - 1
+    if lo == hi:
+        paste_sprite(canvas, CANS[lo], cx)
+        return
+    mix = (cx - CAN_X[lo]) / (CAN_X[hi] - CAN_X[lo])
+    a = canvas.copy()
+    b = canvas.copy()
+    paste_sprite(a, CANS[lo], cx)
+    paste_sprite(b, CANS[hi], cx)
+    canvas[:] = a * (1 - mix) + b * mix
+
+
+def trajectory():
+    # Fitted source pre-impact motion, in pixels per sampled frame.
+    x0, v0, a_can = 28.40, 19.4571, .36642
+    # Collision fits: can/carton mass .922, bottle/can mass 1.009.
+    # Coupled source deceleration is .2231, which is also consistent with
+    # the bottle's friction. Direct can/bottle motion has weighted friction.
+    bottle_mass = 1.00893
+    a_bottle = .22310
+    contact_gap = 8.50
+    x_contact = BOTTLE_X - contact_gap
+    collision = (v0 - np.sqrt(v0*v0 - 2*a_can*(x_contact-x0))) / a_can
+    incoming_v = v0 - a_can * collision
+    joint_v = incoming_v / (1 + bottle_mass)
+    joint_a = (a_can + bottle_mass * a_bottle) / (1 + bottle_mass)
+    t = np.arange(96, dtype=float)
+    free = x0 + v0*t - .5*a_can*t*t
+    dt = np.clip(t-collision, 0, joint_v/joint_a)
+    shared_motion = joint_v*dt - .5*joint_a*dt*dt
+    can = np.where(t < collision, free, x_contact + shared_motion)
+    bottle = BOTTLE_X + shared_motion
+    # Before any intervention-caused contact, match actual source sampling.
+    can[:4] = CAN_X[:4]
+    return can, bottle, collision, joint_v, joint_a
+
+
+def generate():
+    can, bottle, collision, velocity, deceleration = trajectory()
+    result = []
+    for i in range(96):
+        canvas = SOURCE[i].astype(float).copy()
+        # A narrow fixed-camera clean plate replaces only the moving-object
+        # strip. It includes their contact shadows and the carton's straw.
+        canvas[55:85, 15:249] = BG[55:85, 15:249]
+        # Feather outside the object's region, leaving all other RGB intact.
+        for x in (14, 249):
+            canvas[55:85, x] = .5 * BG[55:85, x] + .5 * canvas[55:85, x]
+        if i < collision:
+            # This object has not yet been contacted and remains exactly as
+            # observed. Restore its real translucent glass and contact shadow.
+            canvas[62:85, 161:181] = SOURCE[i, 62:85, 161:181]
+        else:
+            shadow(canvas, bottle[i], 'bottle')
+        shadow(canvas, can[i], 'can')
+        if i < 4:
+            # Exact original can pixels before the factual carton collision;
+            # no carton pixels fall inside these crops.
+            ranges = [(16, 40), (35, 60), (55, 79), (75, 93)]
+            left, right = ranges[i]
+            canvas[62:84, left:right] = SOURCE[i, 62:84, left:right]
+        else:
+            paste_can(canvas, can[i])
+        if i >= collision:
+            light = 1 + .075 * np.clip((bottle[i] - BOTTLE_X) / 70, 0, 1)
+            paste_sprite(canvas, BOTTLE, bottle[i], brightness=light)
+        result.append(np.uint8(np.clip(np.round(canvas), 0, 255)))
+    print(f'Direct collision: frame {collision:.3f}; joint speed {velocity:.3f}; deceleration {deceleration:.4f}')
+    print('Can centers:', np.round(can[[0,3,4,7,8,12,16,20,24,35,95]], 2).tolist())
+    print('Bottle centers:', np.round(bottle[[0,3,4,7,8,12,16,20,24,35,95]], 2).tolist())
+    frames = np.stack(result)
+    artifact_io.save(frames, ROOT)
+    # The RGB package and player are the primary lossless output. Also supply
+    # a conventional video for convenient playback if local ffmpeg is present.
+    import shutil, subprocess
+    ffmpeg = shutil.which('ffmpeg')
+    if ffmpeg:
+        p = subprocess.Popen([ffmpeg, '-y', '-loglevel', 'error', '-f', 'rawvideo',
+            '-vcodec', 'rawvideo', '-pix_fmt', 'rgb24', '-s', '256x144', '-r', '24',
+            '-i', '-', '-an', '-c:v', 'libx264', '-crf', '16', '-pix_fmt', 'yuv420p',
+            '-movflags', '+faststart', str(ROOT / 'edited_video.mp4')], stdin=subprocess.PIPE)
+        p.communicate(frames.tobytes())
+        if p.returncode:
+            raise RuntimeError('Video encoding failed')
+    return frames
+
+
+if __name__ == '__main__':
+    generate()

def clean_background():
    # All occluded parts of the two stationary plant pots and the tabletop
    # become visible later; the final resting objects' patch is visible early.
    b = np.median(SOURCE[40:].astype(np.float32), axis=0)
    b[:, 204:] = np.median(SOURCE[:4, :, 204:].astype(np.float32), axis=0)
    return b

def can_position(frame):
    a = SOURCE[frame].astype(float)
    redness = np.clip(a[69:75, :, 0] - a[69:75, :, 1] - 23, 0, None).mean(0)

def shape_alpha(kind, cx, x0, y0, width, height):
    \"\"\"Pixel-integrated projected rigid silhouettes; edges are antialiased.\"\"\"
    scale = 6
    sy = y0 + (np.arange(height * scale) + .5) / scale - .5
    sx = x0 + (np.arange(width * scale) + .5) / scale - .5
    yy, xx = np.meshgrid(sy, sx, indexing='ij')
    q = xx - cx
    if kind == 'can':
        # Rounded ends of the upright can, including its pale upper rim.
        q -= .12
        top = ((q / 5.16) ** 2 + ((yy - 65.25) / 1.13) ** 2 <= 1)
        mid = (abs(q) <= 5.16) & (yy >= 65.25) & (yy <= 80.45)
        bot = ((q / 5.16) ** 2 + ((yy - 80.45) / 1.12) ** 2 <= 1)
        alpha = (top | mid | bot).astype(float)
    else:

        neck = (q >= -2.98) & (q <= 1.65) & (yy > 66.04) & (yy < 66.52)

        bottom = ((q - .975) / 3.955) ** 2 + ((yy - 80.95) / 1.48) ** 2 <= 1
        alpha = (cap | neck | shoulder | body | bottom).astype(float)
    alpha = alpha.reshape(height, scale, width, scale).mean((1, 3))
    return gaussian_filter(alpha, .28)


def extract_sprite(frame, kind, cx):
    x0, y0, width, height = int(np.floor(cx)) - 9, 62, 22, 23
    alpha = shape_alpha(kind, cx, x0, y0, width, height)
    rgb = SOURCE[frame, y0:y0 + height, x0:x0 + width].astype(float)
    b = BG[y0:y0 + height, x0:x0 + width]
    # Unmix antialiased edges rather than copying their old background.
    foreground = (rgb - (1 - alpha[:, :, None]) * b) / np.maximum(alpha[:, :, None], .018)
    foreground = np.clip(foreground, 0, 255)
    premult = foreground * alpha[:, :, None]
    return (premult, alpha, x0, y0, cx)

def warp_sprite(sprite, cx):
    premult, alpha, x0, y0, old_cx = sprite
    shift = cx - old_cx
    left = max(0, int(np.floor(x0 + shift)) - 1)
    right = min(W, int(np.ceil(x0 + shift + alpha.shape[1])) + 1)
    if right <= left:
        return None
    yy, xx = np.indices((alpha.shape[0], right - left), dtype=float)
    coords = np.array([yy, xx + left - x0 - shift])
    a = map_coordinates(alpha, coords, order=1, mode='constant', cval=0)
    p = np.stack([map_coordinates(premult[:, :, c], coords, order=1,
                                  mode='constant', cval=0) for c in range(3)], axis=2)
    return left, right, y0, a, p


def shadow(canvas, cx, kind):
    if cx < -18 or cx > W + 18:
        return
    strength = .145 if kind == 'can' else .11
    sigma_x = 5.7 if kind == 'can' else 5.1
    a = strength * np.exp(-.5 * (((X - cx + 2.1) / sigma_x) ** 2 + ((Y - 82.2) / .62) ** 2))
    canvas *= (1 - a[:, :, None])


def paste_sprite(canvas, sprite, cx, brightness=1):
    warped = warp_sprite(sprite, cx)
    if warped is None:
        return
    left, right, top, alpha, premult = warped
    region = canvas[top:top + len(alpha), left:right]
    # Small spatial light variation inferred from the observed green label.
    if brightness "'!= 1:
        premult = np.minimum(premult * brightness, alpha[:, :, None] * 255)
    region[:] = premult + region * (1 - alpha[:, :, None])


def paste_can(canvas, cx):
    # Interpolate source appearance in *position*, preserving the label,
    # surface highlights and the spatial change of illumination.
    if cx < CAN_X[0]:
        lo = hi = 0
    elif cx >= CAN_X[-1]:
        lo = hi = 35
    else:
        hi = int(np.searchsorted(CAN_X, cx))
        lo = hi - 1
    if lo == hi:
        paste_sprite(canvas, CANS[lo], cx)
        return
    mix = (cx - CAN_X[lo]) / (CAN_X[hi] - CAN_X[lo])
    a = canvas.copy()
    b = canvas.copy()
    paste_sprite(a, CANS[lo], cx)
    paste_sprite(b, CANS[hi], cx)
    canvas[:] = a * (1 - mix) + b * mix


def trajectory():
    # Fitted source pre-impact motion, in pixels per sampled frame.
    x0, v0, a_can = 28.40, 19.4571, .36642
    # Collision fits: can/carton mass .922, bottle/can mass 1.009.
    # Coupled source deceleration is .2231, which is also consistent with
    # the bottle'"'s friction. Direct can/bottle motion has weighted friction.
    bottle_mass = 1.00893
    a_bottle = .22310
    contact_gap = 8.50
    x_contact = BOTTLE_X - contact_gap
    collision = (v0 - np.sqrt(v0*v0 - 2*a_can*(x_contact-x0))) / a_can
    incoming_v = v0 - a_can * collision
    joint_v = incoming_v / (1 + bottle_mass)
    joint_a = (a_can + bottle_mass * a_bottle) / (1 + bottle_mass)
    t = np.arange(96, dtype=float)
    free = x0 + v0*t - .5*a_can*t*t
    dt = np.clip(t-collision, 0, joint_v/joint_a)
    shared_motion = joint_v*dt - .5*joint_a*dt*dt
    can = np.where(t < collision, free, x_contact + shared_motion)
    bottle = BOTTLE_X + shared_motion
    # Before any intervention-caused contact, match actual source sampling.
    can[:4] = CAN_X[:4]
    return can, bottle, collision, joint_v, joint_a
