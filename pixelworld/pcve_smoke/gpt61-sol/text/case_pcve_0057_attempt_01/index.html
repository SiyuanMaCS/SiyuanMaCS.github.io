+"""Counterfactual mass edit using source-pixel sprites and a 1D impulse model.
+
+The camera and environment are unchanged.  Friction is mass-proportional,
+so the can has the same pre-contact velocity. Collision impulses use four
+units of can mass, rather than the source's one unit.
+"""
+from pathlib import Path
+import numpy as np
+from scipy.ndimage import map_coordinates, binary_closing, binary_fill_holes, binary_erosion, maximum_filter, gaussian_filter
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+
+def source_centers(frames):
+    centers = []
+    for image in frames:
+        r, g, b = image.astype(float).transpose(2, 0, 1)
+        masks = [(r > 1.6*g) & (r > 1.5*b) & (r > 70),
+                 (r > 110) & (g > 100) & (b < .72*g) & (r > 1.02*g),
+                 (g > 1.15*r) & (g > 1.15*b)]
+        cc = []
+        for mask in masks:
+            mask[:65] = False
+            mask[83:] = False
+            ys, xs = np.where(mask)
+            cc.append((xs.min()+xs.max())*.5)
+        centers.append(cc)
+    return np.asarray(centers)
+
+
+def trajectory(centers):
+    # Source-calibrated friction, contact width, restitution and relative masses.
+    a = np.array([.23058430, .21879461, .21756756])
+    restitution = .03008767
+    spacing = [11.30807835, 10.33135755]
+    masses = np.array([4., .85173906, 1.00720001])
+    x = centers[3].copy()
+    v = np.array([18.45, 0., 0.])
+    positions = list(centers[:4].copy())
+    velocities = [np.array([19., 0., 0.]) for _ in range(4)]
+    dt = .005
+    for frame in range(4, 96):
+        for _ in range(200):
+            v = np.sign(v)*np.maximum(np.abs(v)-a*dt, 0.)
+            x += v*dt
+            # Iterated nonpenetration resolves the later three-body contact.
+            for __ in range(3):
+                for i, gap in enumerate(spacing):
+                    if x[i+1]-x[i] < gap:
+                        overlap = gap - (x[i+1]-x[i])
+                        total = masses[i]+masses[i+1]
+                        x[i] -= overlap*masses[i+1]/total
+                        x[i+1] += overlap*masses[i]/total
+                        closing = v[i]-v[i+1]
+                        if closing > 0:
+                            impulse = (1+restitution)*closing/(1/masses[i]+1/masses[i+1])
+                            v[i] -= impulse/masses[i]
+                            v[i+1] += impulse/masses[i+1]
+        positions.append(x.copy())
+        velocities.append(v.copy())
+    return np.asarray(positions), np.asarray(velocities)
+

+def clean_plate(source):
+    # Objects vacate these regions in the factual clip; the rest of frame 1
+    # already supplies an unobstructed plate, including the two large vases.
+    plate = source[0].astype(float).copy()
+    late = np.median(source[40:].astype(float), axis=0)
+    for x0, x1, y0, y1 in [(9, 44, 62, 87), (81, 120, 55, 87), (158, 187, 62, 87)]:
+        plate[y0:y1, x0:x1] = late[y0:y1, x0:x1]
+    return plate
+

+def object_sprite(source, plate, centers, obj, sample):
+    cx = centers[sample, obj]
+    x0, x1 = int(np.floor(cx))-8, int(np.floor(cx))+9
+    y0, y1 = (55, 83) if obj == 1 else (62, 83)
+    image = source[sample, y0:y1, x0:x1].astype(float)
+    back = plate[y0:y1, x0:x1]
+    yy, xx = np.mgrid[y0:y1, x0:x1]
+    dist = np.max(np.abs(image-back), axis=2)
+
+    # Per-pixel source/background contrast gives the actual source silhouette.
+    # Limit it to the body and cap/straw to exclude neighboring containers.
+    if obj == 0:
+        roi = (xx >= cx-6.2) & (xx <= cx+6.6) & (yy >= 63) & (yy <= 82)
+    elif obj == 1:
+        roi = (xx >= cx-6.6) & (xx <= cx+6.4) & (yy >= 62) & (yy <= 82)
+        roi |= (np.abs(xx-cx) < 4.6) & (yy >= 55) & (yy < 63)
+    else:
+        roi = (xx >= cx-5.5) & (xx <= cx+6.) & (yy >= 66) & (yy <= 82)
+        roi |= (xx >= cx-4) & (xx <= cx+1.5) & (yy >= 63) & (yy < 67)
+    binary = (dist > 7.5) & roi
+    interior = binary_fill_holes(binary_closing(binary, iterations=1)) & roi
+    full = binary_erosion(interior, iterations=1)
+    alpha = np.where(interior | binary,
+                     np.minimum(1., dist/np.maximum(maximum_filter(dist, size=3), 1)), 0.)
+    alpha[full] = 1.
+    # Every low-contrast pixel surrounded by body belongs to the object too.
+    if obj == 1:
+        core = (np.abs(xx-cx) <= 4.25) & (yy >= 63) & (yy <= 80)
+        alpha[core] = 1.
+    if obj == 2:
+        core = (xx >= cx-3) & (xx <= cx+3.5) & (yy >= 67) & (yy <= 81)
+        alpha[core] = 1.
+
+    alpha = np.clip(alpha, 0., 1.)
+    # Recover non-premultiplied edge color from the known background.
+    foreground = np.clip(back + (image-back)/np.maximum(alpha[..., None], .12), 0., 255.)
+    # The green bottle has a transparent upper glass section. Preserve tint
+    # and specular highlights but let the destination background show through.
+    glass = np.zeros_like(alpha)
+    if obj == 2:
+        glass = ((yy >= 67) & (yy <= 73)).astype(float)*alpha
+        # A thin lower region above the printed green label is glass as well.
+        glass[(yy == 74) & (xx > cx)] = alpha[(yy == 74) & (xx > cx)]
+    ratio = np.clip(image/np.maximum(back, 1.), .5, 1.18)
+    return dict(x0=x0, y0=y0, center=cx, color=foreground,
+                alpha=alpha, glass=glass, ratio=ratio)
+

+def paste_sprite(canvas, sprite, center):
+    shift = center-sprite['center']
+    x0 = sprite['x0']+shift
+    y0 = sprite['y0']
+    h, w = sprite['alpha'].shape
+    lo, hi = max(0, int(np.floor(x0))-1), min(256, int(np.ceil(x0+w))+1)
+    if hi <= lo:
+        return
+    ys, xs = np.mgrid[y0:y0+h, lo:hi]
+    coords = np.array([ys-y0, xs-x0])
+    def sample(arr):
+        if arr.ndim == 2:
+            return map_coordinates(arr, coords, order=1, mode='constant', cval=0.)
+        return np.stack([map_coordinates(arr[..., c], coords, order=1, mode='constant', cval=0.)
+                         for c in range(3)], axis=-1)
+    # Warp premultiplied pixels to avoid dark fringes at subpixel offsets.
+    alpha = sample(sprite['alpha'])
+    premult = sample(sprite['color']*sprite['alpha'][..., None])
+    old = canvas[y0:y0+h, lo:hi]
+    opaque = old*(1-alpha[..., None])+premult
+    glass = sample(sprite['glass'])
+    ratios = sample(sprite['ratio']*sprite['alpha'][..., None])/np.maximum(alpha[..., None], 1e-6)
+    glass_result = old*(1-alpha[..., None])+old*ratios*alpha[..., None]
+    weight = glass/np.maximum(alpha, 1e-6)
+    canvas[y0:y0+h, lo:hi] = opaque*(1-weight[..., None])+glass_result*weight[..., None]
+
+
+def add_shadow(canvas, cx, obj):
+    if cx < -20 or cx > 278:
+        return
+    # Same table and light direction: a source-sized, thin contact shadow.
+    yy, xx = np.mgrid[79:86, :256]
+    offset = -4.0 if obj < 2 else 2.5
+    rx = [10.8, 10.5, 9.5][obj]
+    shade = .105*np.exp(-.5*((xx-(cx+offset))/rx)**2
+                        -.5*((yy-82.15)/.74)**2)
+    shade *= np.clip((yy-80.4)/1.4, 0., 1.)
+    canvas[79:86] *= 1-shade[..., None]
+
+
+def main():
+    source = np.load(ROOT/'source_pixels.npz')['rgb']
+    centers = source_centers(source)
+    positions, velocities = trajectory(centers)
+    plate = clean_plate(source)
+    sprites = {}
+    frames = []
+    for t in range(96):
+        if t <= 3:
+            # Mass alone cannot change free Coulomb sliding acceleration.
+            frames.append(source[t].copy())
+            continue
+        canvas = plate.copy()
+        for obj in range(3):
+            add_shadow(canvas, positions[t, obj], obj)
+        for obj in range(3):
+            cx = positions[t, obj]
+            if cx > 266 or cx < -12:
+                continue
+            sample = int(np.argmin(np.abs(centers[:38, obj]-cx)))
+            # Sample each object's lighting/rotation at its destination along
+            # the same source path. Past the factual endpoint use settled pose.
+            if cx > centers[37, obj]:
+                sample = 40
+            key = (obj, sample)
+            if key not in sprites:
+                sprites[key] = object_sprite(source, plate, centers, obj, sample)
+            paste_sprite(canvas, sprites[key], cx)
+        frames.append(np.clip(np.rint(canvas), 0, 255).astype(np.uint8))
+    frames = np.stack(frames)
+    artifact_io.save(frames, ROOT)
+    print('Saved', frames.shape, frames.dtype)
+    print('4x-mass object positions:', [(t, np.round(positions[t], 2).tolist())
+                                       for t in [0, 3, 4, 7, 10, 14, 18, 24, 95]])
+
+
+if __name__ == '__main__':
+    main()

def source_centers(frames):
    centers = []
    for image in frames:
        r, g, b = image.astype(float).transpose(2, 0, 1)
        masks = [(r > 1.6*g) & (r > 1.5*b) & (r > 70),

def trajectory(centers):
    # Source-calibrated friction, contact width, restitution and relative masses.
    a = np.array([.23058430, .21879461, .21756756])
    restitution = .03008767
    spacing = [11.30807835, 10.33135755]
    masses = np.array([4., .85173906, 1.00720001])
    x = centers[3].copy()
    v = np.array([18.45, 0., 0.])
    positions = list(centers[:4].copy())
    velocities = [np.array([19., 0., 0.]) for _ in range(4)]
    dt = .005
    for frame in range(4, 96):
        for _ in range(200):
            v = np.sign(v)*np.maximum(np.abs(v)-a*dt, 0.)
            x += v*dt
            # Iterated nonpenetration resolves the later three-body contact.
            for __ in range(3):
                for i, gap in enumerate(spacing):
                    if x[i+1]-x[i] < gap:
                        overlap = gap - (x[i+1]-x[i])
                        total = masses[i]+masses[i+1]
                        x[i] -= overlap*masses[i+1]/total
                        x[i+1] += overlap*masses[i]/total
                        closing = v[i]-v[i+1]
                        if closing > 0:
                            impulse = (1+restitution)*closing/(1/masses[i]+1/masses[i+1])
                            v[i] -= impulse/masses[i]
                            v[i+1] += impulse/masses[i+1]
        positions.append(x.copy())
        velocities.append(v.copy())
    return np.asarray(positions), np.asarray(velocities)


def clean_plate(source):

    # already supplies an unobstructed plate, including the two large vases.
    plate = source[0].astype(float).copy()

def object_sprite(source, plate, centers, obj, sample):
    cx = centers[sample, obj]
    x0, x1 = int(np.floor(cx))-8, int(np.floor(cx))+9
    y0, y1 = (55, 83) if obj == 1 else (62, 83)
    image = source[sample, y0:y1, x0:x1].astype(float)
    back = plate[y0:y1, x0:x1]
    yy, xx = np.mgrid[y0:y1, x0:x1]
    dist = np.max(np.abs(image-back), axis=2)

    # Per-pixel source/background contrast gives the actual source silhouette.
    # Limit it to the body and cap/straw to exclude neighboring containers.
    if obj == 0:
        roi = (xx >= cx-6.2) & (xx <= cx+6.6) & (yy >= 63) & (yy <= 82)
    elif obj == 1:

        roi |= (np.abs(xx-cx) < 4.6) & (yy >= 55) & (yy < 63)
    else:

        roi |= (xx >= cx-4) & (xx <= cx+1.5) & (yy >= 63) & (yy < 67)
    binary = (dist > 7.5) & roi
    interior = binary_fill_holes(binary_closing(binary, iterations=1)) & roi
    full = binary_erosion(interior, iterations=1)
    alpha = np.where(interior | binary,

    alpha[full] = 1.
    # Every low-contrast pixel surrounded by body belongs to the object too.

        alpha[core] = 1.
    if obj == 2:

    foreground = np.clip(back + (image-back)/np.maximum(alpha[..., None], .12), 0., 255.)

def paste_sprite(canvas, sprite, center):
    shift = center-sprite['center']
    x0 = sprite['x0']+shift
    y0 = sprite['y0']
    h, w = sprite['alpha'].shape
    lo, hi = max(0, int(np.floor(x0))-1), min(256, int(np.ceil(x0+w))+1)
    if hi <= lo:
        return
    ys, xs = np.mgrid[y0:y0+h, lo:hi]
    coords = np.array([ys-y0, xs-x0])
    def sample(arr):
        if arr.ndim == 2:
            return map_coordinates(arr, coords, order=1, mode='constant', cval=0.)
        return np.stack([map_coordinates(arr[..., c], coords, order=1, mode='constant', cval=0.)
                         for c in range(3)], axis=-1)
    # Warp premultiplied pixels to avoid dark fringes at subpixel offsets.
    alpha = sample(sprite['alpha'])
    premult = sample(sprite['color']*sprite['alpha'][..., None])
    old = canvas[y0:y0+h, lo:hi]
    opaque = old*(1-alpha[..., None])+premult
    glass = sample(sprite['glass'])
    ratios = sample(sprite['ratio']*sprite['alpha'][..., None])/np.maximum(alpha[..., None], 1e-6)
    glass_result = old*(1-alpha[..., None])+old*ratios*alpha[..., None]
    weight = glass/np.maximum(alpha, 1e-6)
    canvas[y0:y0+h, lo:hi] = opaque*(1-weight[..., None])+glass_result*weight[..., None]

    yy, xx = np.mgrid[79:86, :256]

                        -.5*((yy-82.15)/.74)**2)
    shade *= np.clip((yy-80.4)/1.4, 0., 1.)
    canvas[79:86] *= 1-shade[..., None]
