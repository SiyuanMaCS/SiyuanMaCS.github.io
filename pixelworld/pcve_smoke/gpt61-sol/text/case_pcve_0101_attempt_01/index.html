+"""Reconstruct the source contact with a fivefold-mass old tennis ball.
+
+All appearance is sampled from the supplied video.  The ball contact uses
+impulse mechanics with a grounded old ball and the same fitted dissipation
+and restitution as the observed contact.  Gravity does not scale with mass.
+"""

+import numpy as np
+import cv2
+from scipy.ndimage import binary_closing, binary_fill_holes, gaussian_filter
+import artifact_io

+ROOT = Path(__file__).resolve().parent
+N, H, W = 96, 144, 256

+def contact_trajectory(mass=5.0):
+    # Units: output pixels and source-frame intervals. Grounded old sphere
+    # has a normal reaction, so its contact inverse mass is nx**2 / mass.
+    ux, uy, restitution, floor_e = -4.99698358, 13.4310294, .460924244, .567121383
+    drag_old, roll_old = .190783247, .0487779622
+    drag_new, roll_new = .0125154312, .266376795
+    gravity, diameter_contact = 2.55590458, 10.4706675
+    positions = np.array([[66.5, 94.5], [78.7, 76.5]], dtype=float)
+    velocity = np.array([[0., 0.], [ux, uy]])
+    steps = 30
+    dt = 1. / steps
+    output = [positions.copy()]
+    for frame in range(75):
+        for _ in range(steps):
+            velocity[1, 1] += gravity * dt
+            velocity[0, 0] *= np.exp(-drag_old * dt)
+            velocity[0, 0] -= np.sign(velocity[0, 0]) * min(abs(velocity[0, 0]), roll_old * dt)
+            if frame >= 1:
+                velocity[1, 0] *= np.exp(-drag_new * dt)
+                velocity[1, 0] -= np.sign(velocity[1, 0]) * min(abs(velocity[1, 0]), roll_new * dt)
+            positions += velocity * dt
+            positions[0, 1] = 94.5
+            delta = positions[1] - positions[0]
+            length = np.linalg.norm(delta)
+            if length < diameter_contact:
+                normal = delta / length
+                relative_normal_speed = np.dot(velocity[1] - velocity[0], normal)
+                inverse_mass = 1. + normal[0] ** 2 / mass
+                if relative_normal_speed < 0:
+                    impulse = -(1 + restitution) * relative_normal_speed / inverse_mass
+                    velocity[0, 0] -= impulse * normal[0] / mass
+                    velocity[1] += impulse * normal
+                separation = (diameter_contact - length) / inverse_mass
+                positions[0, 0] -= separation * normal[0] / mass
+                positions[1] += separation * normal
+            if positions[1, 1] > 93.6:
+                positions[1, 1] = 93.6
+                if velocity[1, 1] > 0:
+                    velocity[1, 1] = -floor_e * velocity[1, 1] if velocity[1, 1] > .85 else 0
+        output.append(positions.copy())
+    return np.asarray(output)

+def background_from_source(source):
+    # Temporal visibility provides clean floor pixels without inpainting.
+    bg = np.median(source, axis=0).astype(np.float32)
+    bg[83:108, 9:40] = np.median(source[:19, 83:108, 9:40], axis=0)
+    bg[81:108, 100:135] = np.median(source[:15, 81:108, 100:135], axis=0)
+    return bg

+def detect_low_centers(source):
+    centers = np.zeros((N, 2, 2), dtype=float)
+    for t in range(N):
+        r, g, b = source[t].astype(float).transpose(2, 0, 1)
+        m = (g > r * 1.025) & (g > b * 1.18) & (g > 80)
+        m[:69] = False
+        m[:, 123:] = False

+            if k == 0:
+                mask[:, split:] = False
+                mask[:86] = False
+            else:
+                mask[:, :split] = False
+            y, x = np.where(mask)
+            if len(x):

+    return centers

+def get_sprites(source, centers):
+    # Use the supplied native pixels to retain the worn felt and curved seams
+    # when moving the tiny output-resolution objects by fractional pixels.
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    sprites = {}
+    native_centers = centers.copy()
+    side, scale = 100, 5.0
+    for t in range(N):
+        ok, bgr = cap.read()
+        if not ok:
+            raise RuntimeError('Could not decode complete source video')
+        if t not in {0} and t < 21:
+            continue
+        rgb = cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB).astype(np.float32)
+        kinds = [0] if t == 0 else [0, 1]
+        for k in kinds:
+            guess = centers[t, k] * scale
+            # Find the colored connected sphere, not its achromatic shadow.
+            ix, iy = np.rint(guess).astype(int)
+            x0, y0 = ix - 37, iy - 40
+            crop = rgb[y0:iy + 41, x0:ix + 38]
+            rr, gg, bb = crop.transpose(2, 0, 1)
+            yy, xx = np.indices(rr.shape)
+            region = ((xx - 37) / 33.) ** 2 + ((yy - 40) / 37.) ** 2 < 1
+            mask = (gg - bb > 12) & (gg - rr > 1.5) & region
+            count, labels, stats, _ = cv2.connectedComponentsWithStats(mask.astype(np.uint8), 8)
+            if count < 2:
+                raise RuntimeError('Sphere color segmentation failed')
+            mask = labels == (1 + np.argmax(stats[1:, cv2.CC_STAT_AREA]))
+            mask = binary_fill_holes(binary_closing(mask, iterations=2))
+            ys, xs = np.where(mask)
+            cx = x0 + (xs.min() + xs.max()) / 2.
+            cy = y0 + (ys.min() + ys.max()) / 2.
+            native_centers[t, k] = [cx / scale, cy / scale]
+            alpha = gaussian_filter(mask.astype(np.float32), .75)
+            # Normalize to a centered native-resolution sprite. Keep the real
+            # fuzzy silhouette and source motion blur instead of drawing balls.
+            sy, sx = np.indices((side, side), dtype=np.float32)
+            mapx = (sx - (side - 1) / 2 + cx - x0).astype(np.float32)
+            mapy = (sy - (side - 1) / 2 + cy - y0).astype(np.float32)
+            aa = cv2.remap(alpha, mapx, mapy, cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)
+            cc = cv2.remap(crop, mapx, mapy, cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)
+            # Decontaminate the fractional silhouette against its original floor.
+            # This avoids a light-gray halo as the yellow ball enters table shade.
+            src_bg = np.array([np.median(source[t, 105:108, max(0, int(cx/scale)-2):int(cx/scale)+3, j]) for j in range(3)])
+            decontaminated = (cc - (1 - aa[..., None]) * src_bg) / np.maximum(aa[..., None], .12)
+            # Only change the feathered edge. Interior is the unmodified source.
+            edge = aa < .98
+            cc[edge] = np.clip(decontaminated[edge], 0, 255)
+            sprites[t, k] = (cc * aa[..., None], aa)
+    cap.release()
+    return sprites, native_centers

+def shadow_layer(source, bg, sprites, centers, frame, kind):
+    cx, cy = centers[frame, kind]
+    yy, xx = np.indices((H, W), dtype=np.float32)
+    rgba, alpha = sprites[frame, kind]
+    sx = ((xx - cx) * 5 + (alpha.shape[1] - 1) / 2).astype(np.float32)
+    sy = ((yy - cy) * 5 + (alpha.shape[0] - 1) / 2).astype(np.float32)
+    ball_alpha = cv2.remap(alpha, sx, sy, cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)
+    # Shadows are copied as a signed, achromatic floor-lighting residual.
+    diff = source[frame].astype(float) - bg
+    diff = np.minimum(diff, 0)
+    shade = diff.mean(axis=2)
+    shape = (xx > cx - 7) & (xx < cx + 23) & (yy >= 96) & (yy < 106)
+    if frame == 24:
+        shape &= xx > cx - 3
+    shade *= shape * (ball_alpha < .025)
+    shade[np.abs(shade) < 2.] = 0
+    return shade.astype(np.float32), cx
+

+def restore_objects(image, bg, source_centers, t):
+    yy, xx = np.indices((H, W), dtype=np.float32)
+    clean = np.zeros((H, W), dtype=np.float32)
+    for k in range(2):
+        cx, cy = source_centers[t, k]
+        lo_y = min(cy - 8.0, 87.)
+        feather = np.minimum.reduce([xx - (cx - 10), cx + 24 - xx,
+                                     yy - lo_y, 108 - yy])
+        clean = np.maximum(clean, np.clip(feather / 2., 0, 1))
+    return image * (1 - clean[..., None]) + bg * clean[..., None]
+

+def mix_sprite(sprites, kind, times, distances, target):
+    q = np.interp(target, distances, np.asarray(times, float))
+    lo = int(np.floor(q)); hi = min(lo + 1, times[-1]); w = q - lo
+    a, ma = sprites[lo, kind]; b, mb = sprites[hi, kind]
+    return (1 - w) * a + w * b, (1 - w) * ma + w * mb
+

+def place_sprite(image, sprite, center, illumination=1.):
+    rgb, alpha = sprite
+    cx, cy = center
+    # Supersampled small patch avoids edge flicker from subpixel positions.
+    x0 = max(0, int(np.floor(cx)) - 12); x1 = min(W, int(np.floor(cx)) + 13)
+    y0 = max(0, int(np.floor(cy)) - 12); y1 = min(H, int(np.floor(cy)) + 13)
+    s = 5
+    yy, xx = np.indices(((y1-y0)*s, (x1-x0)*s), dtype=np.float32)
+    xx = (xx + .5) / s + x0 - .5
+    yy = (yy + .5) / s + y0 - .5
+    mapx = ((xx - cx) * 5 + (alpha.shape[1] - 1) / 2).astype(np.float32)
+    mapy = ((yy - cy) * 5 + (alpha.shape[0] - 1) / 2).astype(np.float32)
+    aa = cv2.remap(alpha, mapx, mapy, cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)
+    cc = cv2.remap(rgb, mapx, mapy, cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)
+    aa = cv2.resize(aa, (x1-x0, y1-y0), interpolation=cv2.INTER_AREA)
+    cc = cv2.resize(cc, (x1-x0, y1-y0), interpolation=cv2.INTER_AREA)
+    image[y0:y1, x0:x1] = image[y0:y1, x0:x1] * (1 - aa[..., None]) + cc * illumination
+
+
+def main():
+    source = np.load(ROOT / 'source_pixels.npz')['rgb']
+    bg = background_from_source(source)
+    coarse = detect_low_centers(source)
+    sprites, centers = get_sprites(source, coarse)
+    shadow_old = shadow_layer(source, bg, sprites, centers, 0, 0)
+    shadow_new_ground = shadow_layer(source, bg, sprites, centers, 40, 1)
+    shadow_new_air = shadow_layer(source, bg, sprites, centers, 24, 1)
+    trajectory = contact_trajectory(5.)
+    # Source angular textures are indexed by traveled distance, not source time.
+    # The heavier ball rotates only as far as it now rolls. Incoming ball wraps
+    # through the observed complete seam revolution when its rebound goes farther.
+    old_times = list(range(21, 38))
+    old_distances = centers[0, 0, 0] - centers[old_times, 0, 0]
+    old_distances = np.maximum.accumulate(old_distances)
+    new_times = list(range(21, 38))
+    new_distances = centers[new_times, 1, 0] - centers[21, 1, 0]
+    new_distances = np.maximum.accumulate(new_distances)
+    period = 2 * np.pi * 5.3
+    yy, xx = np.indices((H, W), dtype=np.float32)
+    output = source.copy()
+    for t in range(21, N):
+        positions = trajectory[t - 20]
+        image = restore_objects(source[t].astype(np.float32), bg, centers, t)
+        # Stable illumination from the observed floor/table shadow, sampled at
+        # the sphere's ground contact (height and mass leave shape unchanged).
+        lighting = []
+        for k in range(2):
+            x, y = positions[k]
+            local = float(np.mean(bg[98:101, max(0, int(x)-2):min(W, int(x)+3)]))
+            reference = 185. if k == 0 else 181.
+            lighting.append(np.clip(.18 + .82 * local / reference, .56, 1.025))
+        # Reposition both source shadow fields, maintaining the overhead light.
+        for k in range(2):
+            x, y = positions[k]
+            if k == 0:
+                layer, origin = shadow_old
+                shadow = cv2.warpAffine(layer, np.float32([[1, 0, x-origin], [0, 1, 0]]), (W, H), flags=cv2.INTER_LINEAR)
+            else:
+                height = max(0, 93.6-y)
+                blend = np.clip(height / 5., 0, 1)
+                ground, g_origin = shadow_new_ground
+                air, a_origin = shadow_new_air
+                ground = cv2.warpAffine(ground, np.float32([[1, 0, x-g_origin], [0, 1, 0]]), (W, H), flags=cv2.INTER_LINEAR)
+                air = cv2.warpAffine(air, np.float32([[1, 0, x-a_origin], [0, 1, 0]]), (W, H), flags=cv2.INTER_LINEAR)
+                shadow = ground * (1-blend) + air * blend
+            image += shadow[..., None] * lighting[k]
+        traveled_old = max(0, centers[0, 0, 0] - positions[0, 0])
+        if traveled_old < old_distances[0]:
+            w = np.clip(traveled_old / old_distances[0], 0, 1)
+            a, aa = sprites[0, 0]; b, ba = sprites[21, 0]
+            old_sprite = ((1-w)*a+w*b, (1-w)*aa+w*ba)
+        else:
+            old_sprite = mix_sprite(sprites, 0, old_times, old_distances, traveled_old)
+        traveled_new = max(0, positions[1, 0] - trajectory[1, 1, 0])
+        if t == 21:
+            new_sprite = sprites[21, 1]
+        else:
+            phase = traveled_new % period
+            new_sprite = mix_sprite(sprites, 1, new_times, new_distances, phase)
+        place_sprite(image, old_sprite, positions[0], lighting[0])
+        place_sprite(image, new_sprite, positions[1], lighting[1])
+        output[t] = np.clip(np.rint(image), 0, 255).astype(np.uint8)
+    artifact_io.save(output, ROOT)
+    print('Saved 96 RGB frames, 256x144, 24 fps.')
+    print('Old ball final position:', trajectory[-1, 0].round(3).tolist())
+    print('Incoming ball final position:', trajectory[-1, 1].round(3).tolist())
+    print('Unchanged precontact frames:', np.array_equal(output[:21], source[:21]))
+

+if __name__ == '__main__':
+    main()

                    velocity[1] += impulse * normal
                separation = (diameter_contact - length) / inverse_mass
                positions[0, 0] -= separation * normal[0] / mass
                positions[1] += separation * normal
            if positions[1, 1] > 93.6:
                positions[1, 1] = 93.6
                if velocity[1, 1] > 0:
                    velocity[1, 1] = -floor_e * velocity[1, 1] if velocity[1, 1] > .85 else 0
        output.append(positions.copy())
    return np.asarray(output)


def background_from_source(source):
    # Temporal visibility provides clean floor pixels without inpainting.
    bg = np.median(source, axis=0).astype(np.float32)
    bg[83:108, 9:40] = np.median(source[:19, 83:108, 9:40], axis=0)
    bg[81:108, 100:135] = np.median(source[:15, 81:108, 100:135], axis=0)
    return bg


def detect_low_centers(source):
    centers = np.zeros((N, 2, 2), dtype=float)
    for t in range(N):
        r, g, b = source[t].astype(float).transpose(2, 0, 1)
        m = (g > r * 1.025) & (g > b * 1.18) & (g > 80)
        m[:69] = False
        m[:, 123:] = False
        split = 71 if t <= 22 else 65
        for k in range(2):
            mask = m.copy()
            if k == 0:
                mask[:, split:] = False
                mask[:86] = False
            else:
                mask[:, :split] = False
            y, x = np.where(mask)
            if len(x):
                centers[t, k] = [(x.min() + x.max()) / 2, (y.min() + y.max()) / 2]
    return centers


def get_sprites(source, centers):
    # Use the supplied native pixels to retain the worn felt and curved seams
    # when moving the tiny output-resolution objects by fractional pixels.
    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
    sprites = {}
    native_centers = centers.copy()
    side, scale = 100, 5.0
    for t in range(N):
        ok, bgr = cap.read()
        if not ok:
            raise RuntimeError('Could not decode complete source video')
        if t not in {0} and t < 21:
            continue
        rgb = cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB).astype(np.float32)
        kinds = [0] if t == 0 else [0, 1]
        for k in kinds:
            guess = centers[t, k] * scale
            # Find the colored connected sphere, not its achromatic shadow.
            ix, iy = np.rint(guess).astype(int)
            x0, y0 = ix - 37, iy - 40
            crop = rgb[y0:iy + 41, x0:ix + 38]
            rr, gg, bb = crop.transpose(2, 0, 1)
            yy, xx = np.indices(rr.shape)
            region = ((xx - 37) / 33.) ** 2 + ((yy - 40) / 37.) ** 2 < 1
            mask = (gg - bb > 12) & (gg - rr > 1.5) & region
            count, labels, stats, _ = cv2.connectedComponentsWithStats(mask.astype(np.uint8), 8)
            if count < 2:
                raise RuntimeError('Sphere color segmentation failed')
            mask = labels == (1 + np.argmax(stats[1:, cv2.CC_STAT_AREA]))
            mask = binary_fill_holes(binary_closing(mask, iterations=2))
            ys, xs = np.where(mask)
            cx = x0 + (xs.min() + xs.max()) / 2.
            cy = y0 + (ys.min() + ys.max()) / 2.
            native_centers[t, k] = [cx / scale, cy / scale]
            alpha = gaussian_filter(mask.astype(np.float32), .75)
            # Normalize to a centered native-resolution sprite. Keep the real
            # fuzzy silhouette and source motion blur instead of drawing balls.
            sy, sx = np.indices((side, side), dtype=np.float32)
            mapx = sx - (side - 1) / 2 + cx - x0
            mapy = sy - (side - 1) / 2 + cy - y0
            aa = cv2.remap(alpha, mapx, mapy, cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)
            cc = cv2.remap(crop, mapx, mapy, cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)
            # Decontaminate the fractional silhouette against its original floor.
            # This avoids a light-gray halo as the yellow ball enters table shade.
            src_bg = np.array([np.median(source[t, 105:108, max(0, int(cx/scale)-2):int(cx/scale)+3, j]) for j in range(3)])
            decontaminated = (cc - (1 - aa[..., None]) * src_bg) / np.maximum(aa[..., None], .12)
            # Only change the feathered edge. Interior is the unmodified source.
            edge = aa < .98
            cc[edge] = np.clip(decontaminated[edge], 0, 255)
            sprites[t, k] = (cc * aa[..., None], aa)
    cap.release()
    return sprites, native_centers


def shadow_layer(source, bg, sprites, centers, frame, kind):
    cx, cy = centers[frame, kind]
    yy, xx = np.indices((H, W), dtype=np.float32)
    rgba, alpha = sprites[frame, kind]
    sx = (xx - cx) * 5 + (alpha.shape[1] - 1) / 2
    sy = (yy - cy) * 5 + (alpha.shape[0] - 1) / 2
    ball_alpha = cv2.remap(alpha, sx, sy, cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)
    # Shadows are copied as a signed, achromatic floor-lighting residual.
    diff = source[frame].astype(float) - bg
    diff = np.minimum(diff, 0)
    shade = diff.mean(axis=2)
    shape = (xx > cx - 7) & (xx < cx + 23) & (yy >= 96) & (yy < 106)
    if frame == 24:
        shape &= xx > cx - 3
    shade *= shape * (ball_alpha < .025)
    shade[np.abs(shade) < 2.] = 0
    return shade.astype(np.float32), cx


def restore_objects(image, bg, source_centers, t):
    yy, xx = np.indices((H, W), dtype=np.float32)
    clean = np.zeros((H, W), dtype=np.float32)
    for k in range(2):
        cx, cy = source_centers[t, k]
        lo_y = min(cy - 8.0, 87.)
        feather = np.minimum.reduce([xx - (cx - 10), cx + 24 - xx,
                                     yy - lo_y, 108 - yy])
        clean = np.maximum(clean, np.clip(feather / 2., 0, 1))
    return image * (1 - clean[..., None]) + bg * clean[..., None]


def mix_sprite(sprites, kind, times, distances, target):
    q = np.interp(target, distances, np.asarray(times, float))
    lo = int(np.floor(q)); hi = min(lo + 1, times[-1]); w = q - lo
    a, ma = sprites[lo, kind]; b, mb = sprites[hi, kind]
    return (1 - w) * a + w * b, (1 - w) * ma + w * mb


def place_sprite(image, sprite, center, illumination=1.):
    rgb, alpha = sprite
    cx, cy = center
    # Supersampled small patch avoids edge flicker from subpixel positions.
    x0 = max(0, int(np.floor(cx)) - 12); x1 = min(W, int(np.floor(cx)) + 13)
    y0 = max(0, int(np.floor(cy)) - 12); y1 = min(H, int(np.floor(cy)) + 13)
    s = 5
    yy, xx = np.indices(((y1-y0)*s, (x1-x0)*s), dtype=np.float32)
    xx = (xx + .5) / s + x0 - .5
    yy = (yy + .5) / s + y0 - .5
    mapx = (xx - cx) * 5 + (alpha.shape[1] - 1) / 2
    mapy = (yy - cy) * 5 + (alpha.shape[0] - 1) / 2
    aa = cv2.remap(alpha, mapx, mapy, cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)
    cc = cv2.remap(rgb, mapx, mapy, cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)
    aa = cv2.resize(aa, (x1-x0, y1-y0), interpolation=cv2.INTER_AREA)
    cc = cv2.resize(cc, (x1-x0, y1-y0), interpolation=cv2.INTER_AREA)
    image[y0:y1, x0:x1] = image[y0:y1, x0:x1] * (1 - aa[..., None]) + cc * illumination


def main():

    coarse = detect_low_centers(source)
    sprites, centers = get_sprites(source, coarse)

    old_distances = centers[0, 0, 0] - centers[old_times, 0, 0]
    old_distances = np.maximum.accumulate(old_distances)
    new_times = list(range(21, 38))
    new_distances = centers[new_times, 1, 0] - centers[21, 1, 0]
    new_distances = np.maximum.accumulate(new_distances)
    period = 2 * np.pi * 5.3
    yy, xx = np.indices((H, W), dtype=np.float32)
    output = source.copy()

        positions = trajectory[t - 20]
        image = restore_objects(source[t].astype(np.float32), bg, centers, t)
        # Stable illumination from the observed floor/table shadow, sampled at
        # the sphere's ground contact (height and mass leave shape unchanged).
        lighting = []
        for k in range(2):
            x, y = positions[k]
            local = float(np.mean(bg[98:101, max(0, int(x)-2):min(W, int(x)+3)]))
            reference = 185. if k == 0 else 181.
            lighting.append(np.clip(.18 + .82 * local / reference, .56, 1.025))
        # Reposition both source shadow fields, maintaining the overhead light.
        for k in range(2):

            if k == 0:
                layer, origin = shadow_old
                shadow = cv2.warpAffine(layer, np.float32([[1, 0, x-origin], [0, 1, 0]]), (W, H), flags=cv2.INTER_LINEAR)
            else:
                height = max(0, 93.6-y)
                blend = np.clip(height / 5., 0, 1)
                ground, g_origin = shadow_new_ground
                air, a_origin = shadow_new_air
                ground = cv2.warpAffine(ground, np.float32([[1, 0, x-g_origin], [0, 1, 0]]), (W, H), flags=cv2.INTER_LINEAR)
                air = cv2.warpAffine(air, np.float32([[1, 0, x-a_origin], [0, 1, 0]]), (W, H), flags=cv2.INTER_LINEAR)
                shadow = ground * (1-blend) + air * blend
            image += shadow[..., None] * lighting[k]
        traveled_old = max(0, centers[0, 0, 0] - positions[0, 0])
        if traveled_old < old_distances[0]:
            w = np.clip(traveled_old / old_distances[0], 0, 1)
            a, aa = sprites[0, 0]; b, ba = sprites[21, 0]
            old_sprite = ((1-w)*a+w*b, (1-w)*aa+w*ba)

        traveled_new = max(0, positions[1, 0] - trajectory[1, 1, 0])

            new_sprite = sprites[21, 1]
        else:
            phase = traveled_new % period
            new_sprite = mix_sprite(sprites, 1, new_times, new_distances, phase)
        place_sprite(image, old_sprite, positions[0], lighting[0])
        place_sprite(image, new_sprite, positions[1], lighting[1])

    print('Incoming ball final position:', trajectory[-1, 1].round(3).tolist())
    print('Unchanged precontact frames:', np.array_equal(output[:21], source[:21]))
