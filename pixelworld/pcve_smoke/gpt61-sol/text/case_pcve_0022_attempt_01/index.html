+"""Counterfactual removal using source-pixel matting and projective motion.
+No learned inference: the clean wooden surface is a gradient-domain texture
+repair, and the unimpeded apple is a source RGB cutout with a floor shadow.
+"""
+from pathlib import Path
+import numpy as np
+import cv2
+from scipy import ndimage as ndi
+from scipy.sparse import lil_matrix
+from scipy.sparse.linalg import spsolve
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+def smoothstep(x):
+    x = np.clip(x, 0., 1.)
+    return x*x*(3.-2.*x)
+
+def poisson_texture(base, donor, mask):
+    """Keep observed boundary lighting while borrowing nearby wood grain."""
+    yy, xx = np.where(mask)
+    ids = np.full(mask.shape, -1, np.int32)
+    ids[yy, xx] = np.arange(len(xx))
+    mat = lil_matrix((len(xx),len(xx)), dtype=np.float64)
+    rhs = np.zeros((len(xx),3),np.float64)
+    for k,(y,x) in enumerate(zip(yy,xx)):
+        mat[k,k] = 4
+        for dy,dx in ((0,1),(0,-1),(1,0),(-1,0)):
+            qy,qx=y+dy,x+dx
+            rhs[k] += donor[y,x]-donor[qy,qx]
+            j=ids[qy,qx]
+            if j>=0: mat[k,j] = -1
+            else: rhs[k] += base[qy,qx]
+    out=base.copy()
+    out[yy,xx]=spsolve(mat.tocsr(),rhs)
+    return out
+

\"\"\"Counterfactual removal using source-pixel matting and projective motion.
No learned inference: the clean wooden surface is a gradient-domain texture
repair, and the unimpeded apple is a source RGB cutout with a floor shadow.
\"\"\"
from pathlib import Path
import numpy as np
import cv2
from scipy import ndimage as ndi
from scipy.sparse import lil_matrix
from scipy.sparse.linalg import spsolve
import artifact_io

ROOT = Path(__file__).resolve().parent

def smoothstep(x):
    x = np.clip(x, 0., 1.)
    return x*x*(3.-2.*x)

def poisson_texture(base, donor, mask):
    \"\"\"Keep observed boundary lighting while borrowing nearby wood grain.\"\"\"
    yy, xx = np.where(mask)
    ids = np.full(mask.shape, -1, np.int32)
    ids[yy, xx] = np.arange(len(xx))
    mat = lil_matrix((len(xx),len(xx)), dtype=np.float64)
    rhs = np.zeros((len(xx),3),np.float64)
    for k,(y,x) in enumerate(zip(yy,xx)):
        mat[k,k] = 4
        for dy,dx in ((0,1),(0,-1),(1,0),(-1,0)):
            qy,qx=y+dy,x+dx
            rhs[k] += donor[y,x]-donor[qy,qx]
            j=ids[qy,qx]
            if j>=0: mat[k,j] = -1
            else: rhs[k] += base[qy,qx]
    out=base.copy()
    out[yy,xx]=spsolve(mat.tocsr(),rhs)
    return out
