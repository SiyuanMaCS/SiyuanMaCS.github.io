+All plate, material, reflection and geometry samples come from the source.
+"""
+from pathlib import Path
+import cv2
+import numpy as np
+from PIL import Image
+from scipy.ndimage import label, gaussian_filter
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+S = 5
+
+
+def read_source():
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    result = []
+    while True:
+        ok, frame = cap.read()
+        if not ok:
+            break
+        result.append(cv2.cvtColor(frame, cv2.COLOR_BGR2RGB))
+    cap.release()
+    return result
+

+def feather_rect(shape, bounds, feather=6):
+    y, x = np.mgrid[:shape[0], :shape[1]]
+    x0, y0, x1, y1 = bounds
+    d = np.minimum.reduce([x-x0, x1-x, y-y0, y1-y])
+    return np.clip(d / feather, 0, 1).astype(np.float32)
+
+
+def make_plate(src):
+    # Each occluded region has an unobstructed observation elsewhere in time.
+    plate = src[0].astype(np.float32)
+    late = np.median(np.stack(src[40::8]), axis=0).astype(np.float32)
+    for bounds in [(610, 320, 705, 450), (1020, 260, 1195, 415)]:
+        m = feather_rect(plate.shape, bounds)[...,None]
+        plate = plate*(1-m)+late*m
+    return plate, late
+
+
+X = np.array([224.495,218.553,212.205,205.782,199.490,193.367,
+              187.206,181.114,175.106,169.237,163.452,157.769,
+              152.226,146.840,141.905,137.629,133.532,129.586,
+              125.753,122.205,118.865,115.813,112.935,110.335,
+              107.975,105.830,104.013,102.410,101.132,100.054,
+              99.251,98.697,98.419,98.326])
+
+
+def extract_geometry(src, plate):
+    geometry=[]
+    for t,cx in enumerate(X):
+        cy=64.7+.05*(224.5-cx)
+        x0,x1=int(S*(cx-13)),int(S*(cx+13))
+        y0,y1=int(S*(cy-12)),int(S*(cy+12))
+        a=src[t][y0:y1,x0:x1].astype(np.float32)
+        b=plate[y0:y1,x0:x1]
+        yy,xx=np.mgrid[y0:y1,x0:x1]/S
+        geom=((xx-cx)/9.2)**2+((yy-cy)/(7.7+.01*(224.5-cx)))**2<1.2
+        core=(a[...,0]-b[...,0]>32)&(a[...,0]-a[...,2]>43)&geom
+        lab,n=label(core)
+        counts=np.bincount(lab.ravel());counts[0]=0
+        mask=(lab==counts.argmax()).astype(np.uint8)
+        contours,_=cv2.findContours(mask,cv2.RETR_EXTERNAL,cv2.CHAIN_APPROX_NONE)
+        hull=cv2.convexHull(max(contours,key=cv2.contourArea))
+        center,axes,angle=cv2.fitEllipse(hull)
+        # The high-contrast hull is just inside the antialiased silhouette.
+        geometry.append(((center[0]+x0)/S,(center[1]+y0)/S,
+                         axes[1]/(2*S)+.22,axes[0]/(2*S)+.25))
+    return np.array(geometry)
+
+
+# Normalized source sprites are sampled densely to avoid downsampling twice.
+N=257
+uv=np.linspace(-2.8,2.8,N,dtype=np.float32)
+V,U=np.meshgrid(uv,uv,indexing='ij')
+ALPHA=np.clip((1-np.sqrt(U*U+V*V))/.035+.5,0,1)
+
+
+def source_sprite(t,src,plate,geometry):
+    cx,cy,rx,ry=geometry[t]
+    mx=((cx+rx*U)*S).astype(np.float32)
+    my=((cy+ry*V)*S).astype(np.float32)
+    image=cv2.remap(src[t].astype(np.float32),mx,my,cv2.INTER_LINEAR)
+    background=cv2.remap(plate,mx,my,cv2.INTER_LINEAR)
+    # Thin foreground rails are kept as scene geometry, not baked into wood.
+    # Only the small intersection inside the sphere is repaired.
+    if 14 <= t <= 21:
+        for rail in (114.95,135.45):
+            occluded=(abs(mx/S-rail)<.58)&(my/S<70.4)&(ALPHA>.2)
+            for row in range(N):
+                col=np.flatnonzero(occluded[row])
+                if len(col):
+                    lo=max(0,col[0]-2);hi=min(N-1,col[-1]+2)
+                    weight=((col-lo)/(hi-lo))[:,None]
+                    image[row,col]=image[row,lo]*(1-weight)+image[row,hi]*weight
+    # Unmix the observed background at the boundary to retain source AA.
+    premult=image-background*(1-ALPHA[...,None])
+    premult=np.maximum(0,np.minimum(premult,255*ALPHA[...,None]))
+    color=premult/np.maximum(ALPHA[...,None],.001)
+    # Actual soft reflected light and contact shadow on the glossy board.
+    residual=image-background
+    below=(V>1.02).astype(np.float32)
+    reflection_window=np.exp(-((U-.09)/.95)**8)*np.clip((2.65-V)/.25,0,1)
+    reflection_window*=below
+    residual*=reflection_window[...,None]
+    # Eliminate compression changes and isolated ringing away from reflection.
+    residual=gaussian_filter(residual,sigma=(.55,.55,0))
+    return color,residual
+
+
+def no_impact_path():
+    # Fit smooth drag and a separate impact impulse. Removing the latter is
+    # the intervention; friction/rolling resistance remain in the scene.
+    t=np.arange(1,len(X),dtype=float)
+    contact=13.6
+    basis=np.column_stack([np.ones_like(t),t,t*t,t**3,np.maximum(t-contact,0)])
+    p=np.linalg.lstsq(basis,X[1:],rcond=None)[0]
+    # Do not extrapolate beyond zero speed: the rolling ball settles.
+    roots=np.roots([3*p[3],2*p[2],p[1]])
+    stop=max(float(r.real) for r in roots if abs(r.imag)<1e-7)
+    q=np.minimum(np.arange(96,dtype=float),stop)
+    path=p[0]+p[1]*q+p[2]*q*q+p[3]*q*q*q
+    path[:14]=X[:14]
+    return path,stop,p
+
+
+def render_ball(frame,cx,cy,rx,ry,color,reflection):
+    h,w=frame.shape[:2]
+    x0=max(0,int(S*(cx-1.7*rx)));x1=min(w,int(S*(cx+1.7*rx))+1)
+    y0=max(0,int(S*(cy-1.3*ry)));y1=min(h,int(S*(cy+2.65*ry))+1)
+    yy,xx=np.mgrid[y0:y1,x0:x1].astype(np.float32)/S
+    u=(xx-cx)/rx;v=(yy-cy)/ry
+    mx=(u-uv[0])/(uv[-1]-uv[0])*(N-1)
+    my=(v-uv[0])/(uv[-1]-uv[0])*(N-1)
+    alpha=np.clip((1-np.sqrt(u*u+v*v))/.035+.5,0,1)
+    c=cv2.remap(color.astype(np.float32),mx,my,cv2.INTER_LINEAR,borderMode=cv2.BORDER_CONSTANT)
+    refl=cv2.remap(reflection.astype(np.float32),mx,my,cv2.INTER_LINEAR,borderMode=cv2.BORDER_CONSTANT)
+    bg=frame[y0:y1,x0:x1]
+    frame[y0:y1,x0:x1]=np.clip((bg+refl)*(1-alpha[...,None])+c*alpha[...,None],0,255)
+
+
+def main():
+    src=read_source()
+    assert len(src)==96
+    plate,late=make_plate(src)
+    geometry=extract_geometry(src,plate)
+    sprites=[source_sprite(i,src,plate,geometry) for i in range(34)]
+    path,stop,coefficients=no_impact_path()
+    # Projection of the board, estimated directly from the ball's observed
+    # centers and silhouettes. Continue the same perspective slightly left.
+    cyfit=np.polyfit(geometry[4:,0],geometry[4:,1],2)
+    ryfit=np.polyfit(geometry[4:,0],geometry[4:,3],1)
+    output=[]
+    marble_mask=feather_rect(plate.shape,(610,320,705,450))[...,None]
+    removal=feather_rect(plate.shape,(0,265,1195,490),feather=8)[...,None]
+    for t in range(96):
+        frame=src[t].astype(np.float32)
+        if t<14:
+            # The wooden ball's observed motion is unchanged up to impact.
+            frame=frame*(1-marble_mask)+late*marble_mask
+        else:
+            frame=frame*(1-removal)+plate*removal
+            cx=float(path[t]);cy=float(np.polyval(cyfit,cx))
+            rx=9.27;ry=float(np.polyval(ryfit,cx))
+            if cx>=X[-1]:
+                # Interpolate observations at the same rolling distance,
+                # rather than retaining the factual slowdown in time.
+                idx=float(np.interp(cx,X[::-1],np.arange(34)[::-1]))
+                a=min(32,int(idx));b=a+1;weight=idx-a
+                color=sprites[a][0]*(1-weight)+sprites[b][0]*weight
+                reflection=sprites[a][1]*(1-weight)+sprites[b][1]*weight
+            else:
+                color=sprites[-1][0].copy();reflection=sprites[-1][1]
+                # Advance the source material on a spherical surface while
+                # keeping diffuse illumination and specular light stable.
+                smooth=gaussian_filter(color,sigma=(7,7,0))
+                detail=color-smooth
+                angle=(X[-1]-cx)/9.2
+                phi=np.arcsin(np.clip(U,-1,1))
+                latitude=np.sqrt(np.clip(1-V*V,0,1))
+                u2=np.sin(phi+angle)*latitude
+                mx=(u2-uv[0])/(uv[-1]-uv[0])*(N-1)
+                my=(V-uv[0])/(uv[-1]-uv[0])*(N-1)
+                shifted=cv2.remap(detail.astype(np.float32),mx.astype(np.float32),my.astype(np.float32),cv2.INTER_LINEAR)
+                blend=min(1,(X[-1]-cx)/4)
+                color=np.clip(color*(1-blend)+(smooth+shifted)*blend,0,255)
+            render_ball(frame,cx,cy,rx,ry,color,reflection)
+            # Restore the two fixed, narrow front rails where the moving
+            # ball passes behind them. They end at the back board edge.
+            for rail in (114.95,135.45):
+                x0=int(S*(rail-.42));x1=int(S*(rail+.42))+1
+                y0=265;y1=350
+                frame[y0:y1,x0:x1]=plate[y0:y1,x0:x1]
+        low=np.array(Image.fromarray(np.clip(frame,0,255).astype(np.uint8)).resize((256,144),Image.Resampling.LANCZOS))
+        output.append(low)
+    frames=np.stack(output).astype(np.uint8)
+    artifact_io.save(frames,ROOT)
+    print('Saved',frames.shape,'uint8; no-impact rest x=',round(path[-1],3),
+          'at',round(stop/24,3),'s')
+    print('Drag/impact fit:',coefficients)
+
+
+if __name__=='__main__':
+    main()

\"\"\"Remove the marble and reconstruct the wooden ball's no-impact trajectory.
All plate, material, reflection and geometry samples come from the source.
\"\"\"
from pathlib import Path
import cv2
import numpy as np
from PIL import Image
from scipy.ndimage import label, gaussian_filter
import artifact_io

ROOT = Path(__file__).resolve().parent
S = 5


def read_source():
    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
    result = []
    while True:
        ok, frame = cap.read()
        if not ok:
            break
        result.append(cv2.cvtColor(frame, cv2.COLOR_BGR2RGB))
    cap.release()
    return result


def feather_rect(shape, bounds, feather=6):
    y, x = np.mgrid[:shape[0], :shape[1]]
    x0, y0, x1, y1 = bounds
    d = np.minimum.reduce([x-x0, x1-x, y-y0, y1-y])
    return np.clip(d / feather, 0, 1).astype(np.float32)


def make_plate(src):
    # Each occluded region has an unobstructed observation elsewhere in time.
    plate = src[0].astype(np.float32)
    late = np.median(np.stack(src[40::8]), axis=0).astype(np.float32)
    for bounds in [(610, 320, 705, 450), (1020, 260, 1195, 415)]:
        m = feather_rect(plate.shape, bounds)[...,None]
        plate = plate*(1-m)+late*m
    return plate, late


X = np.array([224.495,218.553,212.205,205.782,199.490,193.367,
              187.206,181.114,175.106,169.237,163.452,157.769,

              107.975,105.830,104.013,102.410,101.132,100.054,
              99.251,98.697,98.419,98.326])


def extract_geometry(src, plate):
    geometry=[]
    for t,cx in enumerate(X):
        cy=64.7+.05*(224.5-cx)
        x0,x1=int(S*(cx-13)),int(S*(cx+13))
        y0,y1=int(S*(cy-12)),int(S*(cy+12))
        a=src[t][y0:y1,x0:x1].astype(np.float32)
        b=plate[y0:y1,x0:x1]
        yy,xx=np.mgrid[y0:y1,x0:x1]/S
        geom=((xx-cx)/9.2)**2+((yy-cy)/(7.7+.01*(224.5-cx)))**2<1.2
        core=(a[...,0]-b[...,0]>32)&(a[...,0]-a[...,2]>43)&geom
        lab,n=label(core)
        counts=np.bincount(lab.ravel());counts[0]=0
        mask=(lab==counts.argmax()).astype(np.uint8)
        contours,_=cv2.findContours(mask,cv2.RETR_EXTERNAL,cv2.CHAIN_APPROX_NONE)
        hull=cv2.convexHull(max(contours,key=cv2.contourArea))
        center,axes,angle=cv2.fitEllipse(hull)
        # The high-contrast hull is just inside the antialiased silhouette.
        geometry.append(((center[0]+x0)/S,(center[1]+y0)/S,
                         axes[1]/(2*S)+.22,axes[0]/(2*S)+.25))
    return np.array(geometry)


# Normalized source sprites are sampled densely to avoid downsampling twice.
N=161
uv=np.linspace(-1.75,1.75,N,dtype=np.float32)
V,U=np.meshgrid(uv,uv,indexing='ij')
ALPHA=np.clip((1-np.sqrt(U*U+V*V))/.035+.5,0,1)


def source_sprite(t,src,plate,geometry):
    cx,cy,rx,ry=geometry[t]
    mx=((cx+rx*U)*S).astype(np.float32)
    my=((cy+ry*V)*S).astype(np.float32)
    image=cv2.remap(src[t].astype(np.float32),mx,my,cv2.INTER_LINEAR)
    background=cv2.remap(plate,mx,my,cv2.INTER_LINEAR)
    # Thin foreground rails are kept as scene geometry, not baked into wood.
    # Only the small intersection inside the sphere is repaired.
    if 14 <= t <= 21:
        for rail in (114.95,135.45):
            occluded=(abs(mx/S-rail)<.58)&(my/S<70.4)&(ALPHA>.2)
            for row in range(N):
                col=np.flatnonzero(occluded[row])
                if len(col):
                    lo=max(0,col[0]-2);hi=min(N-1,col[-1]+2)
                    weight=((col-lo)/(hi-lo))[:,None]
                    image[row,col]=image[row,lo]*(1-weight)+image[row,hi]*weight
    # Unmix the observed background at the boundary to retain source AA.
    premult=image-background*(1-ALPHA[...,None])
    premult=np.maximum(0,np.minimum(premult,255*ALPHA[...,None]))
    color=premult/np.maximum(ALPHA[...,None],.001)
    # Actual soft reflected light and contact shadow on the glossy board.
    residual=image-background
    below=(V>1.02).astype(np.float32)
    reflection_window=np.exp(-((U-.09)/.95)**8)*np.clip((2.65-V)/.25,0,1)
    reflection_window*=below
    residual*=reflection_window[...,None]
    # Eliminate compression changes and isolated ringing away from reflection.
    residual=gaussian_filter(residual,sigma=(.55,.55,0))
    return color,residual


def no_impact_path():
    # Fit smooth drag and a separate impact impulse. Removing the latter is
    # the intervention; friction/rolling resistance remain in the scene.
    t=np.arange(1,len(X),dtype=float)
    contact=13.6
    basis=np.column_stack([np.ones_like(t),t,t*t,t**3,np.maximum(t-contact,0)])
    p=np.linalg.lstsq(basis,X[1:],rcond=None)[0]
    # Do not extrapolate beyond zero speed: the rolling ball settles.
    roots=np.roots([3*p[3],2*p[2],p[1]])
    stop=max(float(r.real) for r in roots if abs(r.imag)<1e-7)
    q=np.minimum(np.arange(96,dtype=float),stop)
    path=p[0]+p[1]*q+p[2]*q*q+p[3]*q*q*q
    path[:14]=X[:14]
    return path,stop,p


def render_ball(frame,cx,cy,rx,ry,color,reflection):
    h,w=frame.shape[:2]
    x0=max(0,int(S*(cx-1.7*rx)));x1=min(w,int(S*(cx+1.7*rx))+1)
    y0=max(0,int(S*(cy-1.3*ry)));y1=min(h,int(S*(cy+2.65*ry))+1)
    yy,xx=np.mgrid[y0:y1,x0:x1].astype(np.float32)/S
    u=(xx-cx)/rx;v=(yy-cy)/ry
    mx=(u-uv[0])/(uv[-1]-uv[0])*(N-1)
    my=(v-uv[0])/(uv[-1]-uv[0])*(N-1)
    alpha=np.clip((1-np.sqrt(u*u+v*v))/.035+.5,0,1)
    c=cv2.remap(color.astype(np.float32),mx,my,cv2.INTER_LINEAR,borderMode=cv2.BORDER_CONSTANT)
    refl=cv2.remap(reflection.astype(np.float32),mx,my,cv2.INTER_LINEAR,borderMode=cv2.BORDER_CONSTANT)
    bg=frame[y0:y1,x0:x1]
    frame[y0:y1,x0:x1]=np.clip((bg+refl)*(1-alpha[...,None])+c*alpha[...,None],0,255)


def main():
    src=read_source()
    assert len(src)==96
    plate,late=make_plate(src)
    geometry=extract_geometry(src,plate)
    sprites=[source_sprite(i,src,plate,geometry) for i in range(34)]
    path,stop,coefficients=no_impact_path()
    # Projection of the board, estimated directly from the ball's observed
    # centers and silhouettes. Continue the same perspective slightly left.
    cyfit=np.polyfit(geometry[4:,0],geometry[4:,1],2)
    ryfit=np.polyfit(geometry[4:,0],geometry[4:,3],1)
    output=[]
    marble_mask=feather_rect(plate.shape,(610,320,705,450))[...,None]
    removal=feather_rect(plate.shape,(0,265,1195,490),feather=8)[...,None]
    for t in range(96):
        frame=src[t].astype(np.float32)
        if t<14:
            # The wooden ball's observed motion is unchanged up to impact.
            frame=frame*(1-marble_mask)+late*marble_mask
        else:
            frame=frame*(1-removal)+plate*removal
            cx=float(path[t]);cy=float(np.polyval(cyfit,cx))
            rx=9.27;ry=float(np.polyval(ryfit,cx))
            if cx>=X[-1]:
                # Interpolate observations at the same rolling distance,
                # rather than retaining the factual slowdown in time.
                idx=float(np.interp(cx,X[::-1],np.arange(34)[::-1]))
                a=min(32,int(idx));b=a+1;weight=idx-a
                color=sprites[a][0]*(1-weight)+sprites[b][0]*weight
                reflection=sprites[a][1]*(1-weight)+sprites[b][1]*weight
            else:
                color=sprites[-1][0].copy();reflection=sprites[-1][1]
                # Advance the source material on a spherical surface while
                # keeping diffuse illumination and specular light stable.
                smooth=gaussian_filter(color,sigma=(7,7,0))
                detail=color-smooth
                angle=(X[-1]-cx)/9.2
                phi=np.arcsin(np.clip(U,-1,1))
                latitude=np.sqrt(np.clip(1-V*V,0,1))
                u2=np.sin(phi+angle)*latitude
                mx=(u2-uv[0])/(uv[-1]-uv[0])*(N-1)
                my=(V-uv[0])/(uv[-1]-uv[0])*(N-1)
                shifted=cv2.remap(detail.astype(np.float32),mx,my,cv2.INTER_LINEAR)
                blend=min(1,(X[-1]-cx)/4)
                color=np.clip(color*(1-blend)+(smooth+shifted)*blend,0,255)
            render_ball(frame,cx,cy,rx,ry,color,reflection)
            # Restore the two fixed, narrow front rails where the moving
            # ball passes behind them. They end at the back board edge.
            for rail in (114.95,135.45):
                x0=int(S*(rail-.42));x1=int(S*(rail+.42))+1
                y0=265;y1=350
                frame[y0:y1,x0:x1]=plate[y0:y1,x0:x1]
        low=np.array(Image.fromarray(np.clip(frame,0,255).astype(np.uint8)).resize((256,144),Image.Resampling.LANCZOS))
        output.append(low)
    frames=np.stack(output).astype(np.uint8)
    artifact_io.save(frames,ROOT)
    print('Saved',frames.shape,'uint8; no-impact rest x=',round(path[-1],3),
          'at',round(stop/24,3),'s')
    print('Drag/impact fit:',coefficients)


if __name__=='__main__':
    main()
