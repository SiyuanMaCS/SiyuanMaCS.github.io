+"""Pixel-only counterfactual: reduce toy car's sliding friction to 30%.
+
+The source supplies all textures, the camera, and the launch/drop animation.
+A fitted ground-plane motion model supplies the counterfactual translation.
+There is no new object appearance or modification of the fixed scenery.
+"""
+from pathlib import Path
+import json
+import numpy as np
+import cv2
+from scipy.ndimage import binary_closing, binary_fill_holes, gaussian_filter
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+
+def source_track(frames):
+    centers = []
+    for frame in frames[:24]:
+        r, g, b = frame.astype(float).transpose(2, 0, 1)
+        m = (r > 130) & (r > 1.5*g) & (r > 1.35*b)
+        m[:24] = False
+        m[53:] = False
+        m[:, :52] = False
+        m[:, 211:] = False
+        count, lab, stats, centroids = cv2.connectedComponentsWithStats(m.astype('uint8'))
+        valid = [i for i in range(1, count) if stats[i, cv2.CC_STAT_AREA] > 12]
+        k = max(valid, key=lambda i: stats[i, cv2.CC_STAT_AREA])
+        centers.append(centroids[k])
+    return np.asarray(centers)
+
+
+def clean_background(frames):
+    bg = np.median(frames[28:], axis=0).astype(np.float32)
+    # The final resting car is occluding a small area; early frames show it cleanly.
+    bg[38:57, 179:211] = np.median(frames[:10, 38:57, 179:211], axis=0)
+    return bg
+
+
+def isolate(frame, bg, cx):
+    """Premultiplied source appearance and its soft shadow, numerically matted."""
+    frame = frame.astype(np.float32)
+    diff = np.max(np.abs(frame-bg), axis=2)
+    yy, xx = np.indices(diff.shape)
+    roi = (xx >= cx-15) & (xx <= cx+17) & (yy >= 23) & (yy <= 57)
+    # Solid windows, chassis, wheels and their antialiasing. Fill small holes
+    # rather than allowing a wood-colored gap inside a translated car.
+    solid = binary_fill_holes(binary_closing((diff > 36) & roi,
+                                             structure=np.ones((3, 3))))
+    feather = gaussian_filter(solid.astype(np.float32), .48)
+    # Lower-contrast shadow pixels retain their original background-relative
+    # attenuation; they are not pasted as a rectangular background patch.
+    support = (diff > 4) & roi
+    support = binary_closing(support, structure=np.ones((3, 3)))
+    soft = np.clip((diff-3)/15, 0, 1) * support
+    soft = np.maximum(soft, feather)
+    # Wherever source pixels match the background, keep the ground visible.
+    color = frame * feather[..., None]
+    residual = (frame-bg) * (soft*(1-feather))[..., None]
+    return color, feather, residual
+
+
+def place(base, layers, affine):
+    premul, alpha, shadow = layers
+    w, h = base.shape[1], base.shape[0]
+    kw = dict(flags=cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+    p = cv2.warpAffine(premul, affine, (w, h), **kw)
+    a = cv2.warpAffine(alpha, affine, (w, h), **kw)
+    s = cv2.warpAffine(shadow, affine, (w, h), **kw)
+    return base*(1-a[..., None]) + p + s
+
+
+def main():
+    frames = np.load(ROOT/'source_pixels.npz')['rgb']
+    track = source_track(frames)
+    bg = clean_background(frames)
+    # Source motion is almost horizontal, with modest perspective shrinkage.
+    # Undo the perspective before fitting Coulomb sliding deceleration.
+    vanishing_x = 650.0
+    xland = track[7, 0]
+    D = vanishing_x-xland
+    dx = track[7:24, 0]-xland
+    ground_distance = dx/(1-dx/D)
+    fit = np.polyfit(np.arange(17), ground_distance, 2)
+    acceleration = -2*fit[0]
+    original_v = fit[1]
+    friction_ratio = .3
+    # Same initial speed. Friction loss on the initial support and landing
+    # impulse is reduced, while gravity, restitution and launch remain fixed.
+    initial_acceleration = max(0., -2*np.polyfit(np.arange(4), track[:4, 0], 2)[0])
+    initial_dx = .5*(1-friction_ratio)*initial_acceleration*7**2
+    initial_dx = min(initial_dx, 4.1)
+    new_xland = xland + initial_dx
+    incident_velocity = 10.0
+    new_v = original_v + (1-friction_ratio)*(incident_velocity-original_v) + .735
+    a_new = acceleration*friction_ratio
+    stop_time = new_v/a_new
+    out = []
+    positions = []
+    for i in range(96):
+        base = frames[i].astype(np.float32).copy()
+        # Keep all fixed scene pixels, replacing only the complete source track.
+        base[23:58, 53:211] = bg[23:58, 53:211]
+        if i < 7:
+            # Retain the source gravity-driven flight and changing orientation.
+            src = i
+            cx, cy = track[src]
+            dx_extra = initial_dx*(i/7)**2
+            # At the slightly greater speed the platform edge is reached a
+            # fraction of a frame sooner; vertical gravity is otherwise equal.
+            dy_extra = 0 if i < 4 else .16*(i-3)
+            nx, ny = cx+dx_extra, cy+dy_extra
+            scale = 1.0
+        else:
+            t = min(i-7, stop_time)
+            distance = new_v*t - .5*a_new*t*t
+            nx = new_xland + distance/(1+distance/(vanishing_x-new_xland))
+            # Select the factual appearance at the same *spatial* point,
+            # keeping its perspective, rolling pose, lights and contact shadow.
+            src = int(np.argmin(np.abs(track[7:, 0]-nx)))+7
+            cx, cy = track[src]
+            scale = (vanishing_x-nx)/(vanishing_x-cx)
+            ny = cy - .033*(nx-cx)
+        positions.append([float(nx), float(ny)])
+        if nx-15*scale < 256:
+            affine = np.array([[scale, 0, nx-scale*cx],
+                               [0, scale, ny-scale*cy]], dtype=np.float32)
+            base = place(base, isolate(frames[src], bg, cx), affine)
+        out.append(np.clip(np.rint(base), 0, 255).astype(np.uint8))
+    # The intervention is a physical parameter change, not an instantaneous
+    # visual jump: preserve the exact initial configuration.
+    out[0] = frames[0].copy()
+    out = np.asarray(out, dtype=np.uint8)
+    artifact_io.save(out, ROOT)
+    (ROOT/'motion_model.json').write_text(json.dumps({
+        'friction_multiplier': friction_ratio,
+        'source_plane_acceleration_px_per_frame2': acceleration,
+        'counterfactual_plane_acceleration_px_per_frame2': a_new,
+        'positions': positions,
+        'notes': 'Ground-plane friction fit; source textures, gravity and fixed scenery retained.'
+    }, indent=2))
+    print('Saved', out.shape, 'RGB frames. Ground acceleration:', acceleration, '->', a_new)
+    print('Counterfactual car centers:', [(i, positions[i]) for i in [0, 6, 7, 12, 18, 24, 30, 95]])
+
+
+if __name__ == '__main__':
+    main()

            # fraction of a frame sooner; vertical gravity is otherwise equal.
            dy_extra = 0 if i < 4 else .16*(i-3)
            nx, ny = cx+dx_extra, cy+dy_extra
            scale = 1.0
        else:
            t = min(i-7, stop_time)
            distance = new_v*t - .5*a_new*t*t
            nx = new_xland + distance/(1+distance/(vanishing_x-new_xland))
            # Select the factual appearance at the same *spatial* point,
            # keeping its perspective, rolling pose, lights and contact shadow.
            src = int(np.argmin(np.abs(track[7:, 0]-nx)))+7
            cx, cy = track[src]
            scale = (vanishing_x-nx)/(vanishing_x-cx)
            ny = cy - .033*(nx-cx)
        positions.append([float(nx), float(ny)])
        if nx-15*scale < 256:
            affine = np.array([[scale, 0, nx-scale*cx],
                               [0, scale, ny-scale*cy]], dtype=np.float32)
            base = place(base, isolate(frames[src], bg, cx), affine)
        out.append(np.clip(np.rint(base), 0, 255).astype(np.uint8))
    # The intervention is a physical parameter change, not an instantaneous
    # visual jump: preserve the exact initial configuration.
    out[0] = frames[0].copy()
    out = np.asarray(out, dtype=np.uint8)
    artifact_io.save(out, ROOT)
    (ROOT/'motion_model.json').write_text(json.dumps({
        'friction_multiplier': friction_ratio,
        'source_plane_acceleration_px_per_frame2': acceleration,
        'counterfactual_plane_acceleration_px_per_frame2': a_new,
        'positions': positions,
        'notes': 'Ground-plane friction fit; source textures, gravity and fixed scenery retained.'
    }, indent=2))
    print('Saved', out.shape, 'RGB frames. Ground acceleration:', acceleration, '->', a_new)
    print('Counterfactual car centers:', [(i, positions[i]) for i in [0, 6, 7, 12, 18, 24, 30, 95]])


if __name__ == '__main__':
    main()
