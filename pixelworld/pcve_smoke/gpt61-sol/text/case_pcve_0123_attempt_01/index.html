+"""Counterfactual mass edit, using source pixels and a fitted rigid-sphere model.
+
+No external renderer is used: contacts are integrated with numpy, and both
+marbles and their contact shadows are composited from the supplied footage.
+"""
+from pathlib import Path
+import numpy as np
+import cv2
+from scipy.ndimage import binary_erosion
+from scipy.optimize import least_squares
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+
+def read_source():
+    low = np.load(ROOT / 'source_pixels.npz')['rgb']
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    frames = []
+    centers = []
+    for t in range(96):
+        ok, bgr = cap.read()
+        if not ok:
+            raise RuntimeError('Source video is incomplete')
+        rgb = bgr[:, :, ::-1].copy()
+        frames.append(rgb)
+        # Color segmentation is used only numerically, for physical tracking.
+        roi = rgb[240:390].astype(np.float32)
+        r, g, b = roi.transpose(2, 0, 1)
+        gold = (r > 145) & (g > b + 55) & (r - g < 75)
+        gold[:, 825:] = False
+        purple = (b > g + 35) & (b > r + 15)
+        cc = []
+        for mask in (gold, purple):
+            n, labs, stats, cent = cv2.connectedComponentsWithStats(mask.astype(np.uint8))
+            idx = 1 + np.argmax(stats[1:, 4])
+            cc.append((cent[idx] + [0, 240]) / 5)
+        centers.append(cc)
+    cap.release()
+    return low, np.asarray(frames), np.asarray(centers)
+

+def trajectory(centers, purple_mass):
+    # Coefficients calibrated against both factual rebound cycles. Time units
+    # are source frames; distances are output pixels. Rolling inertia is 2/5 mrÂ².
+    restitution = 0.612059966
+    flat_friction = 0.0200929956
+    ramp_friction = 0.444322383
+    rolling_loss = 0.0547849549
+    incoming_spin = 2.24059149
+    ramp_edge = 112.872099
+    contact_x = 122.139129
+    gravity_tangent = 0.377626739
+    inertia_ratio = 0.386973212
+    incoming_speed = 6.16943833
+    midpoint = centers[33, :, 0].mean()
+    separation = 2 * (midpoint - contact_x)
+    x = centers[33, :, 0].copy()
+    v = np.array([incoming_speed, -incoming_speed])
+    spin = np.array([incoming_spin, -incoming_spin])
+    mass = np.array([1., purple_mass])
+    positions = [x.copy()]
+    velocities = [v.copy()]
+    spins = [spin.copy()]
+    dt = 1 / 120
+    contacts = []
+    for frame in range(33, 95):
+        for step in range(120):
+            slope = np.where(x < ramp_edge, 1., np.where(x > 2 * midpoint - ramp_edge, -1., 0.))
+            mu = np.where(slope != 0, ramp_friction, flat_friction)
+            slip = v - spin
+            # Solve the tangential no-slip impulse, clipped to Coulomb friction.
+            friction = np.clip(-slip / ((1 + 1 / inertia_ratio) * dt), -mu, mu)
+            rolling = np.where(np.abs(v) > .001, -rolling_loss * np.sign(v), 0.)
+            v += (gravity_tangent * slope + friction + rolling) * dt
+            spin += (-friction / inertia_ratio + rolling) * dt
+            x += v * dt
+            if x[1] - x[0] < separation:
+                overlap = separation - (x[1] - x[0])
+                x += np.array([-mass[1], mass[0]]) * overlap / mass.sum()
+                if v[0] > v[1]:
+                    impulse = (1 + restitution) * (v[0] - v[1]) / (1 / mass[0] + 1 / mass[1])
+                    v += np.array([-impulse / mass[0], impulse / mass[1]])
+                    contacts.append(frame + step * dt)
+            stopped = (np.abs(v) < rolling_loss * dt) & (np.abs(v - spin) < .01) & (slope == 0)
+            v[stopped] = 0.
+            spin[stopped] = 0.
+        positions.append(x.copy())
+        velocities.append(v.copy())
+        spins.append(spin.copy())
+    return np.array(positions), np.array(velocities), np.array(spins), contacts
+
+
+def ellipse_from_source(image, background, center):
+    cx, cy = center * 5
+    ix, iy = int(cx), int(cy)
+    a = image[iy-35:iy+32, ix-38:ix+38].astype(float)
+    bg = background[iy-35:iy+32, ix-38:ix+38].astype(float)
+    diff = np.max(np.abs(a-bg), axis=2)
+    mask = diff > 28
+    edge = mask & ~binary_erosion(mask)
+    yy, xx = np.nonzero(edge)
+    xx = xx + ix - 38
+    yy = yy + iy - 35
+    keep = (yy < cy + 22) & (((xx-cx)/35)**2 + ((yy-cy)/35)**2 < 1)
+    xx, yy = xx[keep], yy[keep]
+    fit = least_squares(
+        lambda p: np.sqrt(((xx-p[0])/p[2])**2 + ((yy-p[1])/p[3])**2)-1,
+        [cx, cy-.5, 26, 26], loss='soft_l1', f_scale=.1,
+        bounds=([cx-3,cy-3,23,23],[cx+3,cy+3,29,29]))
+    return fit.x
+
+
+def make_sprite(image, background, tracked_center):
+    cx, cy, rx, ry = ellipse_from_source(image, background, tracked_center)
+    # Canonical sphere matting, independent of the source track underneath it.
+    yy, xx = np.indices((65,65), dtype=np.float32)
+    qx, qy = xx-32, yy-32
+    mx = (cx + qx * rx / 26).astype(np.float32)
+    my = (cy + qy * ry / 26).astype(np.float32)
+    raw = cv2.remap(image.astype(np.float32),mx,my,cv2.INTER_LINEAR)
+    bg = cv2.remap(background.astype(np.float32),mx,my,cv2.INTER_LINEAR)
+    radius = np.sqrt(qx*qx + qy*qy)
+    alpha = np.clip(26.45-radius,0,1).astype(np.float32)
+    premult = np.clip(raw-bg*(1-alpha[:,:,None]),0,255*alpha[:,:,None])
+    premult[alpha==0] = 0
+    return premult, alpha
+
+
+def make_shadow(image, background, center):
+    cx,cy = center*5
+    yy,xx = np.indices((105,131),dtype=np.float32)
+    qx,qy=xx-65,yy-35
+    mx=(cx+qx).astype(np.float32)
+    my=(cy+qy).astype(np.float32)
+    a=cv2.remap(image.astype(np.float32),mx,my,cv2.INTER_LINEAR)
+    b=cv2.remap(background.astype(np.float32),mx,my,cv2.INTER_LINEAR)
+    ratio=(a+3)/(b+3)-1
+    # Shadow and local reflection below the sphere; no sphere pixels transported.
+    fade=np.clip((qy-25)/3,0,1)*np.clip((68-qy)/12,0,1)*np.clip((64-np.abs(qx))/12,0,1)
+    ratio[np.max(np.abs(a-b),axis=2)<5]=0
+    ratio=np.clip(ratio,-.85,.15)*fade[:,:,None]
+    return cv2.GaussianBlur(ratio,(3,3),.55)
+
+
+def stamp(frame, premult, alpha, x,y, blur=0.):
+    h,w=alpha.shape
+    # Numeric exposure integration, retaining a crisp contour at low velocities.
+    taps=[0.] if blur<.3 else [-.5,0.,.5]
+    p=np.zeros_like(premult);a=np.zeros_like(alpha)
+    for offset in taps:
+        M=np.float32([[1,0,offset*blur],[0,1,0]])
+        p+=cv2.warpAffine(premult,M,(w,h),flags=cv2.INTER_LINEAR)/len(taps)
+        a+=cv2.warpAffine(alpha,M,(w,h),flags=cv2.INTER_LINEAR)/len(taps)
+    left=int(np.floor(x-w//2));top=int(np.floor(y-h//2))
+    dx=x-(left+w//2);dy=y-(top+h//2)
+    M=np.float32([[1,0,dx],[0,1,dy]])
+    p=cv2.warpAffine(p,M,(w,h),flags=cv2.INTER_LINEAR)
+    a=cv2.warpAffine(a,M,(w,h),flags=cv2.INTER_LINEAR)
+    frame[top:top+h,left:left+w]=frame[top:top+h,left:left+w]*(1-a[:,:,None])+p
+
+
+def shadow_stamp(frame,ratio,x,y):
+    h,w=ratio.shape[:2]
+    left=int(np.floor(x-65));top=int(np.floor(y-35))
+    M=np.float32([[1,0,x-(left+65)],[0,1,y-(top+35)]])
+    ratio=cv2.warpAffine(ratio,M,(w,h),flags=cv2.INTER_LINEAR)
+    region=frame[top:top+h,left:left+w]
+    frame[top:top+h,left:left+w]=np.clip(region*(1+ratio),0,255)
+
+
+def generate():
+    low,hi,centers=read_source()
+    cf_x,cf_v,cf_spin,contacts=trajectory(centers,.3)
+    print('Counterfactual collision times (s):',np.array(contacts)/24)
+    # A static camera lets the empty central track be recovered directly from
+    # the first frame. The initial marbles are far outside the edited corridor.
+    background=hi[0]
+    sprites=[]
+    for j in range(2):
+        sprites.append([make_sprite(hi[t],background,centers[t,j]) for t in range(38,50)])
+    shadow_bank=[]
+    valid=list(range(16,32))+list(range(37,50))
+    for j in range(2):
+        shadow_bank.append([(centers[t,j,0],make_shadow(hi[t],background,centers[t,j])) for t in valid])
+    # Fit the source camera's projection of the ramps. Flat center section.
+    y_floor=np.mean(centers[52:56,:,1],axis=0)
+    edge_y=np.mean(centers[33,:,1])
+    mid=centers[33,:,0].mean()
+    ramp_edge=112.872099
+    ramp_slope=.139
+    phases=np.zeros(2)
+    results=low.copy()
+    # No pre-contact dynamics depend on mass: preserve frames 1--34 exactly.
+    for t in range(34,96):
+        k=t-33
+        frame=hi[t].astype(np.float32)
+        # Remove both factual marbles and all of their shadows/reflections.
+        for j in range(2):
+            cx,cy=centers[t,j]*5
+            left=max(0,int(cx)-105);right=min(1280,int(cx)+106)
+            top=max(0,int(cy)-60);bottom=min(720,int(cy)+105)
+            yy,xx=np.indices((bottom-top,right-left),dtype=np.float32)
+            d=np.sqrt(((xx+left-cx)/102)**2+((yy+top-(cy+21))/81)**2)
+            fade=np.clip((1-d)*12,0,1)[:,:,None]
+            frame[top:bottom,left:right]=frame[top:bottom,left:right]*(1-fade)+background[top:bottom,left:right]*fade
+        render_y=[]
+        for j in range(2):
+            x=cf_x[k,j]
+            outside=max(ramp_edge-x,x-(2*mid-ramp_edge),0.)
+            y=y_floor[j]-ramp_slope*outside
+            render_y.append(y)
+            bank=shadow_bank[j]
+            nearest=min(bank,key=lambda item:abs(item[0]-x))[1]
+            shadow_stamp(frame,nearest,x*5,y*5)
+        for j in range(2):
+            phases[j]+=cf_spin[k,j]
+            # Small source texture variations reproduce rotation without rotating
+            # the fixed lighting or highlight as a flat image.
+            phase=(phases[j]/4)%12
+            i=int(phase);f=phase-i
+            p0,a0=sprites[j][i]
+            p1,a1=sprites[j][(i+1)%12]
+            p=p0*(1-f)+p1*f;alpha=a0*(1-f)+a1*f
+            stamp(frame,p,alpha,cf_x[k,j]*5,render_y[j]*5,
+                  blur=min(abs(cf_v[k,j])*5*.22,10))
+        results[t]=cv2.resize(np.clip(frame,0,255).astype(np.uint8),(256,144),interpolation=cv2.INTER_AREA)
+    # Required candidate commit. All submitted RGB frames are stored by save.
+    artifact_io.save(results,ROOT)
+    # A conventional MP4 is also useful alongside the required RGB artifact.
+    writer=cv2.VideoWriter(str(ROOT/'edited_video.mp4'),cv2.VideoWriter_fourcc(*'mp4v'),24,(256,144))
+    for frame in results:
+        writer.write(frame[:,:,::-1])
+    writer.release()
+    print('Saved',results.shape,results.dtype)
+
+
+if __name__=='__main__':
+    generate()

\"\"\"Counterfactual mass edit, using source pixels and a fitted rigid-sphere model.

No external renderer is used: contacts are integrated with numpy, and both
marbles and their contact shadows are composited from the supplied footage.
\"\"\"
from pathlib import Path
import numpy as np
import cv2
from scipy.ndimage import binary_erosion
from scipy.optimize import least_squares
import artifact_io

ROOT = Path(__file__).resolve().parent


def read_source():
    low = np.load(ROOT / 'source_pixels.npz')['rgb']
    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
    frames = []
    centers = []
    for t in range(96):
        ok, bgr = cap.read()
        if not ok:
            raise RuntimeError('Source video is incomplete')
        rgb = bgr[:, :, ::-1].copy()
        frames.append(rgb)
        # Color segmentation is used only numerically, for physical tracking.
        roi = rgb[240:390].astype(np.float32)
        r, g, b = roi.transpose(2, 0, 1)
        gold = (r > 145) & (g > b + 55) & (r - g < 75)
        gold[:, 825:] = False
        purple = (b > g + 35) & (b > r + 15)
        cc = []
        for mask in (gold, purple):
            n, labs, stats, cent = cv2.connectedComponentsWithStats(mask.astype(np.uint8))
            idx = 1 + np.argmax(stats[1:, 4])
            cc.append((cent[idx] + [0, 240]) / 5)
        centers.append(cc)
    cap.release()
    return low, np.asarray(frames), np.asarray(centers)


def trajectory(centers, purple_mass):
    # Coefficients calibrated against both factual rebound cycles. Time units
    # are source frames; distances are output pixels. Rolling inertia is 2/5 mrÂ².
    restitution = 0.612059966
    flat_friction = 0.0200929956
    ramp_friction = 0.444322383
    rolling_loss = 0.0547849549
    incoming_spin = 2.24059149
    ramp_edge = 112.872099
    contact_x = 122.139129
    gravity_tangent = 0.377626739
    inertia_ratio = 0.386973212
    incoming_speed = 6.16943833
    midpoint = centers[33, :, 0].mean()
    separation = 2 * (midpoint - contact_x)
    x = centers[33, :, 0].copy()
    v = np.array([incoming_speed, -incoming_speed])
    spin = np.array([incoming_spin, -incoming_spin])
    mass = np.array([1., purple_mass])
    positions = [x.copy()]
    velocities = [v.copy()]
    spins = [spin.copy()]
    dt = 1 / 120
    contacts = []
    for frame in range(33, 95):
        for step in range(120):
            slope = np.where(x < ramp_edge, 1., np.where(x > 2 * midpoint - ramp_edge, -1., 0.))
            mu = np.where(slope "'!= 0, ramp_friction, flat_friction)
            slip = v - spin
            # Solve the tangential no-slip impulse, clipped to Coulomb friction.
            friction = np.clip(-slip / ((1 + 1 / inertia_ratio) * dt), -mu, mu)
            rolling = np.where(np.abs(v) > .001, -rolling_loss * np.sign(v), 0.)
            v += (gravity_tangent * slope + friction + rolling) * dt
            spin += (-friction / inertia_ratio + rolling) * dt
            x += v * dt
            if x[1] - x[0] < separation:
                overlap = separation - (x[1] - x[0])
                x += np.array([-mass[1], mass[0]]) * overlap / mass.sum()
                if v[0] > v[1]:
                    impulse = (1 + restitution) * (v[0] - v[1]) / (1 / mass[0] + 1 / mass[1])
                    v += np.array([-impulse / mass[0], impulse / mass[1]])
                    contacts.append(frame + step * dt)
            stopped = (np.abs(v) < rolling_loss * dt) & (np.abs(v - spin) < .01) & (slope == 0)
            v[stopped] = 0.
            spin[stopped] = 0.
        positions.append(x.copy())
        velocities.append(v.copy())
        spins.append(spin.copy())
    return np.array(positions), np.array(velocities), np.array(spins), contacts


def ellipse_from_source(image, background, center):
    cx, cy = center * 5
    ix, iy = int(cx), int(cy)
    a = image[iy-35:iy+32, ix-38:ix+38].astype(float)
    bg = background[iy-35:iy+32, ix-38:ix+38].astype(float)
    diff = np.max(np.abs(a-bg), axis=2)
    mask = diff > 28
    edge = mask & ~binary_erosion(mask)
    yy, xx = np.nonzero(edge)
    xx = xx + ix - 38
    yy = yy + iy - 35
    keep = (yy < cy + 22) & (((xx-cx)/35)**2 + ((yy-cy)/35)**2 < 1)
    xx, yy = xx[keep], yy[keep]
    fit = least_squares(
        lambda p: np.sqrt(((xx-p[0])/p[2])**2 + ((yy-p[1])/p[3])**2)-1,
        [cx, cy-.5, 26, 26], loss='"'soft_l1', f_scale=.1,
        bounds=([cx-3,cy-3,23,23],[cx+3,cy+3,29,29]))
    return fit.x


def make_sprite(image, background, tracked_center):
    cx, cy, rx, ry = ellipse_from_source(image, background, tracked_center)
    # Canonical sphere matting, independent of the source track underneath it.
    yy, xx = np.indices((65,65), dtype=np.float32)
    qx, qy = xx-32, yy-32
    mx = (cx + qx * rx / 26).astype(np.float32)
    my = (cy + qy * ry / 26).astype(np.float32)
    raw = cv2.remap(image.astype(np.float32),mx,my,cv2.INTER_LINEAR)
    bg = cv2.remap(background.astype(np.float32),mx,my,cv2.INTER_LINEAR)
    radius = np.sqrt(qx*qx + qy*qy)
    alpha = np.clip(26.45-radius,0,1).astype(np.float32)
    premult = np.clip(raw-bg*(1-alpha[:,:,None]),0,255*alpha[:,:,None])
    premult[alpha==0] = 0
    return premult, alpha


def make_shadow(image, background, center):
    cx,cy = center*5
    yy,xx = np.indices((105,131),dtype=np.float32)
    qx,qy=xx-65,yy-35
    mx=(cx+qx).astype(np.float32)
    my=(cy+qy).astype(np.float32)
    a=cv2.remap(image.astype(np.float32),mx,my,cv2.INTER_LINEAR)
    b=cv2.remap(background.astype(np.float32),mx,my,cv2.INTER_LINEAR)
    ratio=(a+3)/(b+3)-1
    # Shadow and local reflection below the sphere; no sphere pixels transported.
    fade=np.clip((qy-25)/3,0,1)*np.clip((68-qy)/12,0,1)*np.clip((64-np.abs(qx))/12,0,1)
    ratio[np.max(np.abs(a-b),axis=2)<5]=0
    ratio=np.clip(ratio,-.85,.15)*fade[:,:,None]
    return cv2.GaussianBlur(ratio,(3,3),.55)


def stamp(frame, premult, alpha, x,y, blur=0.):
    h,w=alpha.shape
    # Numeric exposure integration, retaining a crisp contour at low velocities.
    taps=[0.] if blur<.3 else [-.5,0.,.5]
    p=np.zeros_like(premult);a=np.zeros_like(alpha)
    for offset in taps:
        M=np.float32([[1,0,offset*blur],[0,1,0]])
        p+=cv2.warpAffine(premult,M,(w,h),flags=cv2.INTER_LINEAR)/len(taps)
        a+=cv2.warpAffine(alpha,M,(w,h),flags=cv2.INTER_LINEAR)/len(taps)
    left=int(np.floor(x-w//2));top=int(np.floor(y-h//2))
    dx=x-(left+w//2);dy=y-(top+h//2)
    M=np.float32([[1,0,dx],[0,1,dy]])
    p=cv2.warpAffine(p,M,(w,h),flags=cv2.INTER_LINEAR)
    a=cv2.warpAffine(a,M,(w,h),flags=cv2.INTER_LINEAR)
    frame[top:top+h,left:left+w]=frame[top:top+h,left:left+w]*(1-a[:,:,None])+p


def shadow_stamp(frame,ratio,x,y):
    h,w=ratio.shape
    left=int(np.floor(x-65));top=int(np.floor(y-35))
    M=np.float32([[1,0,x-(left+65)],[0,1,y-(top+35)]])
    ratio=cv2.warpAffine(ratio,M,(w,h),flags=cv2.INTER_LINEAR)
    region=frame[top:top+h,left:left+w]
    frame[top:top+h,left:left+w]=np.clip(region*(1+ratio),0,255)

    # the first frame. The initial marbles are far outside the edited corridor.
    background=hi[0]
    sprites=[]
    for j in range(2):

    shadow_bank=[]

    for j in range(2):
        shadow_bank.append([(centers[t,j,0],make_shadow(hi[t],background,centers[t,j])) for t in valid])
    # Fit the source camera's projection of the ramps. Flat center section.
    y_floor=np.mean(centers[52:56,:,1],axis=0)
    edge_y=np.mean(centers[33,:,1])
    mid=centers[33,:,0].mean()
    ramp_edge=112.872099
    ramp_slope=.139
    phases=np.zeros(2)
    results=low.copy()
    # No pre-contact dynamics depend on mass: preserve frames 1--34 exactly.
    for t in range(34,96):
        k=t-33
        frame=hi[t].astype(np.float32)
        # Remove both factual marbles and all of their shadows/reflections.
        for j in range(2):
            cx,cy=centers[t,j]*5
            left=max(0,int(cx)-105);right=min(1280,int(cx)+106)
            top=max(0,int(cy)-60);bottom=min(720,int(cy)+105)
            yy,xx=np.indices((bottom-top,right-left),dtype=np.float32)
            d=np.sqrt(((xx+left-cx)/102)**2+((yy+top-(cy+21))/81)**2)
            fade=np.clip((1-d)*12,0,1)[:,:,None]
            frame[top:bottom,left:right]=frame[top:bottom,left:right]*(1-fade)+background[top:bottom,left:right]*fade
        render_y=[]
        for j in range(2):
            x=cf_x[k,j]
            outside=max(ramp_edge-x,x-(2*mid-ramp_edge),0.)
            y=y_floor[j]-ramp_slope*outside
            render_y.append(y)
            bank=shadow_bank[j]
            nearest=min(bank,key=lambda item:abs(item[0]-x))[1]
            shadow_stamp(frame,nearest,x*5,y*5)
        for j in range(2):
            phases[j]+=cf_spin[k,j]
            # Small source texture variations reproduce rotation without rotating
            # the fixed lighting or highlight as a flat image.
            phase=(phases[j]/4)%12
            i=int(phase);f=phase-i
            p0,a0=sprites[j][i]
            p1,a1=sprites[j][(i+1)%12]

            stamp(frame,p,alpha,cf_x[k,j]*5,render_y[j]*5,
                  blur=min(abs(cf_v[k,j])*5*.22,10))
        results[t]=cv2.resize(np.clip(frame,0,255).astype(np.uint8),(256,144),interpolation=cv2.INTER_AREA)
    # Required candidate commit. All submitted RGB frames are stored by save.
    artifact_io.save(results,ROOT)
    # A conventional MP4 is also useful alongside the required RGB artifact.
    writer=cv2.VideoWriter(str(ROOT/'edited_video.mp4'),cv2.VideoWriter_fourcc(*'mp4v'),24,(256,144))
    for frame in results:
        writer.write(frame[:,:,::-1])
    writer.release()
    print('Saved',results.shape,results.dtype)


if __name__=='__main__':
    generate()
