+
+The mass change preserves gravitational acceleration and the existing descent.
+At contact the inverse-mass impulse changes: red retains almost all its speed,
+while the blue sphere receives a larger impulse. Floor deceleration is inferred
+from the original trajectories (and is independent of the sphere's mass).
+All rendering uses RGB source pixels, analytic geometry and interpolation.
+"""

+import numpy as np
+from scipy.ndimage import label, map_coordinates, distance_transform_edt
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+def object_centers(src):
+    centers = np.full((len(src), 2, 2), np.nan)
+    for t, image in enumerate(src):
+        r, g, b = image.astype(float).transpose(2, 0, 1)
+        masks = [(r > 1.8*g) & (r > 1.9*b) & (r > 80),
+                 (b > 1.5*r) & (b > 1.15*g) & (b > 55)]
+        for k, mask in enumerate(masks):
+            lab, _ = label(mask)
+            counts = np.bincount(lab.ravel())
+            counts[0] = 0
+            if len(counts) <= 1 or counts.max() < 5:
+                continue
+            y, x = np.where(lab == counts.argmax())
+            if k == 0:
+                centers[t,k] = (x.mean()+.35, y.mean()-(1.5 if t==21 else 1.1))
+            else:
+                # Once clipped by the image boundary, fit the center from the
+                # last unconstrained trajectory rather than the visible centroid.
+                centers[t,k] = (x.mean()+.2, y.mean()-.3)
+        if t >= 24:
+            centers[t,1] = (7.75-6.3*(t-24)+.325*(t-24)**2,
+                            89.4) if t < 36 else (-45.,89.4)
+            if t >= 28:
+                centers[t,1,0] = -20.-6.*(t-28)
+    return centers
+
+YY, XX = np.mgrid[0:144,0:256]
+
+def ellipse(cx, cy, rx, ry):
+    return ((XX-cx)/rx)**2 + ((YY-cy)/ry)**2
+
+
+def background(src, centers):
+    """Recover the actual stationary set from unobstructed source observations."""
+    valid = np.ones(src.shape[:3], dtype=bool)
+    for t in range(len(src)):
+        for k in range(2):
+            cx,cy=centers[t,k]
+            if not np.isfinite(cx):
+                continue
+            rad = (7.3+.04*t) if k==0 and t<21 else 8.2
+            hidden = ellipse(cx,cy,rad+1.25,8.7+1.1) < 1
+            hidden |= ellipse(cx+.6,cy+8.3,10.0,4.) < 1
+            if k==1 or t>=22:
+                hidden |= ellipse(cx+2.2,cy+15.,10.0,8.) < 1
+            valid[t] &= ~hidden
+    # Use medians of only clear observations, preserving the photographed grain.
+    import warnings
+    with warnings.catch_warnings():
+        warnings.simplefilter('ignore', RuntimeWarning)
+        bg=np.nanmedian(np.where(valid[...,None],src.astype(float),np.nan),axis=0)
+    unknown=~np.isfinite(bg[...,0])
+    # A very small region swept continuously by the contacting spheres is never
+    # visible. Reconstruct only that missing stationary wood grain harmonically.
+    if unknown.any():

+        bg[unknown]=bg[indices[0][unknown],indices[1][unknown]]
+        for _ in range(280):

+            neighbor=(padded[:-2,1:-1]+padded[2:,1:-1]+
+                      padded[1:-1,:-2]+padded[1:-1,2:])/4
+            bg[unknown]=neighbor[unknown]
+    print('Background reconstructed; unobserved pixels:',int(unknown.sum()))
+    return bg

+def coverage(cx,cy,rx,ry):
+    a=np.zeros((144,256),float)
+    for dy in (np.arange(8)+.5)/8-.5:
+        for dx in (np.arange(8)+.5)/8-.5:
+            a += (((XX+dx-cx)/rx)**2+((YY+dy-cy)/ry)**2 <= 1)
+    return a/64
+

+def make_sphere(src,bg,t,cx,cy,rx,ry):
+    """An antialiased, premultiplied source sprite plus its ground reflection."""
+    a=coverage(cx,cy,rx,ry)
+    premul=src[t].astype(float)-(1-a[...,None])*bg
+    # Dividing the pixel's background out at silhouettes prevents a wooden halo.
+    premul=np.clip(premul,0,255*a[...,None])
+    premul[a==0]=0
+    # The source contains both a soft contact shadow and a faint floor reflection.
+    e=ellipse(cx+2,cy+15.,10.5,8.)
+    smask=np.clip((1.15-e)/.35,0,1)*(1-a)
+    smask[YY < cy+ry-.3]=0
+    shadow=(src[t].astype(float)-bg)*smask[...,None]
+    # Don't transport insignificant compression noise in the surrounding set.
+    shadow[np.max(np.abs(shadow),axis=-1)<2]=0
+    return (a,premul,shadow,cx,cy)
+
+
+def paste(image,sprite,x,y):
+    a,premul,shadow,cx,cy=sprite
+    grid=np.array([YY-(y-cy),XX-(x-cx)])
+    shifted_a=map_coordinates(a,grid,order=1,mode='constant',cval=0)
+    shifted_rgb=np.stack([map_coordinates(premul[...,c],grid,order=1,
+                           mode='constant',cval=0) for c in range(3)],axis=-1)
+    shifted_shadow=np.stack([map_coordinates(shadow[...,c],grid,order=1,
+                              mode='constant',cval=0) for c in range(3)],axis=-1)
+    image += shifted_shadow
+    image *= 1-shifted_a[...,None]
+    image += shifted_rgb
+

+def generate():

+    centers=object_centers(src)
+    bg=background(src,centers)
+    # Fully lit source spheres at the foot of the ramp; size/material unchanged.
+    red=make_sphere(src,bg,24,30.95,88.35,8.15,8.25)
+    blue=make_sphere(src,bg,0,24.6,88.45,8.10,8.15)
+    frames=src.copy()
+    # In a two-body contact, doubling the factual impulse and weighting by the
+    # new inverse masses gives 2m_red/(m_red+m_blue)=40/21 for the blue impulse.
+    mass_ratio=20.
+    impulse_blue=2*mass_ratio/(mass_ratio+1)
+    incoming_speed=12.4
+    factual_red_speed=5.0
+    factual_blue_speed=7.95
+    red_speed=incoming_speed-(incoming_speed-factual_red_speed)*2/(mass_ratio+1)
+    blue_speed=factual_blue_speed*impulse_blue
+    red_x0=centers[21,0,0]-incoming_speed
+    red_x0+=(centers[22,0,0]-red_x0)*2/(mass_ratio+1)
+    blue_x0=centers[21,1,0]+(centers[22,1,0]-centers[21,1,0])*impulse_blue
+    print('Post-impact speed (pixels/frame):',red_speed,blue_speed)
+    print('Frame 22 centers:',red_x0,blue_x0)
+    # Before the first collision the mass cancels in the gravity and rolling
+    # equations. The intervention is instantaneous at t=0 but appearance and
+    # trajectory are identical until contact, so retain the source up through 21.
+    for t in range(22,96):
+        f=src[t].astype(float)
+        # All counterfactually affected source pixels lie in this small floor ROI.
+        # The remainder of the set and the ramp are copied without alteration.
+        f[76:116,:57]=bg[76:116,:57]
+        n=t-22
+        red_x=red_x0-red_speed*n+.5*.65*n*n
+        blue_x=blue_x0-blue_speed*n+.5*.65*n*n

+            red_y=88.25+.3*(1-np.exp(-1.2*n))

+            if blue_x>-22:

+    print('Saved complete counterfactual:',frames.shape,frames.dtype)

def coverage(cx,cy,rx,ry):
    a=np.zeros((144,256),float)
    for dy in (np.arange(8)+.5)/8-.5:
        for dx in (np.arange(8)+.5)/8-.5:
            a += (((XX+dx-cx)/rx)**2+((YY+dy-cy)/ry)**2 <= 1)
    return a/64


def make_sphere(src,bg,t,cx,cy,rx,ry):
    \"\"\"An antialiased, premultiplied source sprite plus its ground reflection.\"\"\"
    a=coverage(cx,cy,rx,ry)
    premul=src[t].astype(float)-(1-a[...,None])*bg
    # Dividing the pixel's background out at silhouettes prevents a wooden halo.
    premul=np.clip(premul,0,255*a[...,None])
    premul[a==0]=0
    # The source contains both a soft contact shadow and a faint floor reflection.
    e=ellipse(cx+2,cy+15.,10.5,8.)
    smask=np.clip((1.15-e)/.35,0,1)*(1-a)
    smask[YY < cy+ry-.3]=0
    shadow=(src[t].astype(float)-bg)*smask[...,None]
    # Don't transport insignificant compression noise in the surrounding set.
    shadow[np.max(np.abs(shadow),axis=-1)<2]=0
    return (a,premul,shadow,cx,cy)


def paste(image,sprite,x,y):
    a,premul,shadow,cx,cy=sprite
    grid=np.array([YY-(y-cy),XX-(x-cx)])
    shifted_a=map_coordinates(a,grid,order=1,mode='constant',cval=0)
    shifted_rgb=np.stack([map_coordinates(premul[...,c],grid,order=1,
                           mode='constant',cval=0) for c in range(3)],axis=-1)
    shifted_shadow=np.stack([map_coordinates(shadow[...,c],grid,order=1,
                              mode='constant',cval=0) for c in range(3)],axis=-1)
    image += shifted_shadow
    image *= 1-shifted_a[...,None]
    image += shifted_rgb
