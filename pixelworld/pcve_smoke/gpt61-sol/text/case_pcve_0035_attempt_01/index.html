+"""Counterfactual bowling edit, using only source RGB and pixel geometry.
+
+A fivefold contact-friction deceleration arrests the ball before the pin deck.
+The perspective-scaled moving ball, its contact shadow and reflection are taken
+from the source, not redrawn; the unstruck pins retain their original state.
+"""
+from pathlib import Path
+import numpy as np
+from PIL import Image
+import cv2
+from scipy.ndimage import label
+from artifact_io import save
+
+ROOT = Path(__file__).resolve().parent
+
+
+def smooth_window(coords, low, high, feather):
+    return np.clip((coords-low)/feather, 0, 1) * np.clip((high-coords)/feather, 0, 1)
+
+
+def generate():
+    original = np.load(ROOT/'source_pixels.npz')['rgb']
+    cap = cv2.VideoCapture(str(ROOT/'source_video.mp4'))
+    early, late = [], []
+    for i in range(96):
+        ok, bgr = cap.read()
+        if not ok:
+            raise RuntimeError('Incomplete source video')
+        rgb = bgr[..., ::-1].copy()
+        if i < 36:
+            early.append(rgb)
+        if i >= 56 and i % 4 == 0:
+            late.append(rgb[440:650, 600:680])
+    cap.release()
+    clean_crop = np.median(np.stack(late), axis=0).astype(np.float32)
+    first = early[0].astype(np.float32)
+    # The late source exposes the exact wood grain hidden by the starting ball.
+    yy, xx = np.mgrid[440:650, 600:680]
+    erase = (smooth_window(xx, 608, 673, 7) *
+             smooth_window(yy, 478, 633, 8))
+    background = first.copy()
+    background[440:650, 600:680] = (
+        first[440:650, 600:680]*(1-erase[..., None]) +
+        clean_crop*erase[..., None])
+
+    # Identify the ball from its blue material, in the known central lane.
+    centers, radii = [], []
+    for f in early:
+        p = f[350:550, 610:665].astype(float)
+        r, g, b = p.transpose(2, 0, 1)
+        mask = (b > 1.6*r) & (b > 1.3*g) & (b > 65)
+        lab, n = label(mask)
+        k = 1 + np.argmax(np.bincount(lab.ravel())[1:])
+        y, x = np.where(lab == k)
+        centers.append([x.mean()+610, y.mean()+350])
+        radii.append(np.sqrt(len(x)/np.pi))
+    centers, radii = np.array(centers), np.array(radii)
+    # Radius/depth gives the perspective horizon without scene metadata.
+    slope, intercept = np.polyfit(centers[:, 1], radii, 1)
+    horizon = -intercept/slope
+    times = np.arange(36)/24.
+    depth0 = centers[0, 1]-horizon
+    distance = depth0/(centers[:, 1]-horizon)-1
+    q = np.polyfit(times[8:], distance[8:], 2)
+    original_accel = max(0.05, -2*q[0])
+    initial_speed = q[1]
+    acceleration = 5*original_accel
+    stop_time = initial_speed/acceleration
+
+    # Freeze only causally affected scene pixels, not the whole source video.
+    Y, X = np.mgrid[:720, :1280]
+    region = (smooth_window(X, 555, 729, 10)*
+              smooth_window(Y, 343, 650, 10))
+    region_low = np.array(Image.fromarray((region*255).astype(np.uint8)).resize(
+        (256, 144), Image.Resampling.LANCZOS)).astype(float)/255
+    region_low = np.clip(region_low, 0, 1)[..., None]
+
+    # Source-local appearance residuals include the actual glossy sphere,
+    # floor shadow and reflection. Feathering prevents texture seams.
+    residuals = []
+    for j in range(20):
+        cx, cy = centers[j]
+        rad = radii[j]+1.5
+        kscale = (cy-horizon)/depth0
+        bottom = horizon + (633-horizon)*kscale
+        width = rad+5
+        m = (smooth_window(xx, cx-width, cx+width, 4)*
+             smooth_window(yy, cy-rad-4, bottom, 4))
+        residuals.append((early[j][440:650,600:680].astype(np.float32)-clean_crop)*m[...,None])
+
+    output = []
+    trajectory = []
+    for i in range(96):
+        if i == 0:
+            output.append(original[0].copy())
+            trajectory.append(float(centers[0,1]/5))
+            continue
+        t = min(i/24., stop_time)
+        displacement = initial_speed*t - 0.5*acceleration*t*t
+        scale = 1/(1+displacement)
+        target_y = horizon + depth0*scale
+        target_x = 639.5 + (centers[0,0]-639.5)*scale
+        # A distance-indexed rolling appearance avoids spinning after stopping.
+        virtual = np.interp(target_y, centers[:20,1][::-1], np.arange(20)[::-1])
+        j = int(np.clip(round(virtual), 0, 19))
+        sc = (target_y-horizon)/(centers[j,1]-horizon)
+        map_x = ((xx-target_x)/sc + centers[j,0]-600).astype(np.float32)
+        map_y = ((yy-horizon)/sc+horizon-440).astype(np.float32)
+        residual = cv2.remap(residuals[j], map_x, map_y,
+                             cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)
+        edited = background.copy()
+        edited[440:650,600:680] += residual
+        low = np.array(Image.fromarray(np.clip(np.rint(edited),0,255).astype(np.uint8)).resize(
+            (256,144), Image.Resampling.LANCZOS))
+        frame = original[i]*(1-region_low)+low*region_low
+        output.append(np.clip(np.rint(frame),0,255).astype(np.uint8))
+        trajectory.append(target_y/5)
+    frames = np.stack(output)
+    save(frames, ROOT)
+    print('Saved:', frames.shape, frames.dtype)
+    print('Friction ratio: 5; stop time %.3f seconds; final image y %.3f' %
+          (stop_time, trajectory[-1]))
+    print('Initial frame exact:', np.array_equal(frames[0], original[0]))
+    print('Ball trajectory samples:', np.round(np.array(trajectory)[::4],3))
+    return frames
+
+
+if __name__ == '__main__':
+    generate()

\"\"\"Counterfactual bowling edit, using only source RGB and pixel geometry.

A fivefold contact-friction deceleration arrests the ball before the pin deck.
The perspective-scaled moving ball, its contact shadow and reflection are taken
from the source, not redrawn; the unstruck pins retain their original state.
\"\"\"
from pathlib import Path
import numpy as np
from PIL import Image
import cv2
from scipy.ndimage import label
from artifact_io import save

ROOT = Path(__file__).resolve().parent


def smooth_window(coords, low, high, feather):
    return np.clip((coords-low)/feather, 0, 1) * np.clip((high-coords)/feather, 0, 1)


def generate():
    original = np.load(ROOT/'source_pixels.npz')['rgb']
    cap = cv2.VideoCapture(str(ROOT/'source_video.mp4'))
    early, late = [], []
    for i in range(96):
        ok, bgr = cap.read()
        if not ok:
            raise RuntimeError('Incomplete source video')
        rgb = bgr[..., ::-1].copy()
        if i < 36:
            early.append(rgb)
        if i >= 56 and i % 4 == 0:
            late.append(rgb[440:650, 600:680])
    cap.release()
    clean_crop = np.median(np.stack(late), axis=0).astype(np.float32)
    first = early[0].astype(np.float32)
    # The late source exposes the exact wood grain hidden by the starting ball.
    yy, xx = np.mgrid[440:650, 600:680]
    erase = (smooth_window(xx, 608, 673, 7) *
             smooth_window(yy, 478, 633, 8))
    background = first.copy()
    background[440:650, 600:680] = (
        first[440:650, 600:680]*(1-erase[..., None]) +
        clean_crop*erase[..., None])

    # Identify the ball from its blue material, in the known central lane.
    centers, radii = [], []
    for f in early:
        p = f[350:550, 610:665].astype(float)
        r, g, b = p.transpose(2, 0, 1)
        mask = (b > 1.6*r) & (b > 1.3*g) & (b > 65)
        lab, n = label(mask)
        k = 1 + np.argmax(np.bincount(lab.ravel())[1:])
        y, x = np.where(lab == k)
        centers.append([x.mean()+610, y.mean()+350])
        radii.append(np.sqrt(len(x)/np.pi))
    centers, radii = np.array(centers), np.array(radii)
    # Radius/depth gives the perspective horizon without scene metadata.
    slope, intercept = np.polyfit(centers[:, 1], radii, 1)
    horizon = -intercept/slope
    times = np.arange(36)/24.
    depth0 = centers[0, 1]-horizon
    distance = depth0/(centers[:, 1]-horizon)-1
    q = np.polyfit(times[8:], distance[8:], 2)
    original_accel = max(0.05, -2*q[0])
    initial_speed = q[1]
    acceleration = 5*original_accel
    stop_time = initial_speed/acceleration

    # Freeze only causally affected scene pixels, not the whole source video.
    Y, X = np.mgrid[:720, :1280]
    region = (smooth_window(X, 555, 729, 10)*
              smooth_window(Y, 343, 650, 10))
    region_low = np.array(Image.fromarray((region*255).astype(np.uint8)).resize(
        (256, 144), Image.Resampling.LANCZOS)).astype(float)/255
    region_low = np.clip(region_low, 0, 1)[..., None]

    # Source-local appearance residuals include the actual glossy sphere,
    # floor shadow and reflection. Feathering prevents texture seams.
    residuals = []
    for j in range(20):
        cx, cy = centers[j]
        rad = radii[j]+1.5
        kscale = (cy-horizon)/depth0
        bottom = horizon + (633-horizon)*kscale
        width = rad+5
        m = (smooth_window(xx, cx-width, cx+width, 4)*
             smooth_window(yy, cy-rad-4, bottom, 4))
        residuals.append((early[j][440:650,600:680].astype(np.float32)-clean_crop)*m[...,None])

    output = []
    trajectory = []
    for i in range(96):
        if i == 0:
            output.append(original[0].copy())
            trajectory.append(float(centers[0,1]/5))
            continue
        t = min(i/24., stop_time)
        displacement = initial_speed*t - 0.5*acceleration*t*t
        scale = 1/(1+displacement)
        target_y = horizon + depth0*scale
        target_x = 639.5 + (centers[0,0]-639.5)*scale
        # A distance-indexed rolling appearance avoids spinning after stopping.
        virtual = np.interp(target_y, centers[:20,1][::-1], np.arange(20)[::-1])
        j = int(np.clip(round(virtual), 0, 19))
        sc = (target_y-horizon)/(centers[j,1]-horizon)
        map_x = ((xx-target_x)/sc + centers[j,0]-600).astype(np.float32)
        map_y = ((yy-horizon)/sc+horizon-440).astype(np.float32)
        residual = cv2.remap(residuals[j], map_x, map_y,
                             cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)
        edited = background.copy()
        edited[440:650,600:680] += residual
        low = np.array(Image.fromarray(np.clip(np.rint(edited),0,255).astype(np.uint8)).resize(
            (256,144), Image.Resampling.LANCZOS))
        frame = original[i]*(1-region_low)+low*region_low
        output.append(np.clip(np.rint(frame),0,255).astype(np.uint8))
        trajectory.append(target_y/5)
    frames = np.stack(output)
    save(frames, ROOT)
    print('Saved:', frames.shape, frames.dtype)
    print('Friction ratio: 5; stop time %.3f seconds; final image y %.3f' %
          (stop_time, trajectory[-1]))
    print('Initial frame exact:', np.array_equal(frames[0], original[0]))
    print('Ball trajectory samples:', np.round(np.array(trajectory)[::4],3))
    return frames


if __name__ == '__main__':
    generate()
