+"""Counterfactual edit: a tenfold mass increase, without a size change.
+
+All appearance is reconstructed from the supplied video.  Motion after contact
+uses a perspective-correct one-dimensional elastic collision calculation.
+"""
+from pathlib import Path
+import numpy as np
+import cv2
+from PIL import Image
+from scipy.ndimage import gaussian_filter, binary_dilation, binary_fill_holes
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+SOURCE = np.load(ROOT / 'source_pixels.npz')['rgb']
+H, W = 720, 1280
+K = .0038  # perspective growth measured from the three source trajectories
+XREF = 114.15
+CONTACT = 29.5
+RESTITUTION = .98
+
+
+def q_from_x(x):
+    d = XREF - x
+    return d / (1 + K * d)
+
+
+def x_from_q(q):
+    return XREF - q / (1 - K * q)
+
+
+def scale_at(x):
+    return 1 + K * (XREF - x)
+
+
+def colored_bbox(im, kind):
+    a = im.astype(np.int16)
+    r, g, b = a.transpose(2, 0, 1)
+    if kind == 'red':
+        m = (r-g > 38) & (r-b > 30)
+    else:
+        m = (b-r > 38) & (b-g > 18)
+    roi = np.zeros((H, W), bool)
+    roi[240:460, 460:905] = True
+    m &= roi
+    n, lab, stats, _ = cv2.connectedComponentsWithStats(m.astype(np.uint8))
+    if n < 2:
+        raise RuntimeError('Missing source object')
+    j = 1 + np.argmax(stats[1:, cv2.CC_STAT_AREA])
+    x,y,w,h,_ = stats[j]
+    return (int(x),int(y),int(x+w),int(y+h)), lab == j
+
+
+def white_source_shift(t):
+    # Measured white foreground centroids, including its factual rail contact.
+    times = [0,29,30,31,32,36,40,44,48,52,60,72,95]
+    xs = [0,0,-.5,-2.3,-4.4,-12,-20.4,-28.4,-37.4,-38,-36.9,-35.4,-34.7]
+    ys = [0,0,.35,1.1,1.85,4.55,7.8,10.9,15.4,15.7,15.1,14.9,14.1]
+    return np.interp(t,times,xs)*5, np.interp(t,times,ys)*5
+
+
+def exclusion(im, t, components=False):
+    mask = np.zeros((H,W),np.uint8)
+    rects = {}
+    for kind in ['red','blue']:
+        (x0,y0,x1,y1),_ = colored_bbox(im,kind)
+        # Include subpixel edges and ambient contact shadows, not just pigment.
+        x0,x1 = max(0,x0-15),min(W,x1+18)
+        y0,y1 = max(0,y0-12),min(H,y1+25)
+        mask[y0:y1,x0:x1] = 1
+        rects[kind] = (x0,y0,x1,y1)
+    dx,dy = white_source_shift(t)
+    x0,x1 = int(491+dx),int(580+dx)
+    y0,y1 = int(373+dy),int(450+dy)
+    mask[y0:y1,x0:x1] = 1
+    rects['white'] = (x0,y0,x1,y1)
+    return (mask,rects) if components else mask
+
+
+def decode():
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    result=[]
+    for t in range(96):
+        ok,bgr=cap.read()
+        if not ok:
+            raise RuntimeError('Incomplete source')
+        result.append(cv2.cvtColor(bgr,cv2.COLOR_BGR2RGB))
+    cap.release()
+    return result
+
+
+def clean_plate(high):
+    # Every edited location is exposed in another source frame.  Excluding all
+    # three objects avoids median ghosts from the two long stationary intervals.
+    sample=[0,8,16,24,28,32,36,40,44,48,56,64,80,95]
+    masks=[exclusion(high[t],t) for t in sample]
+    plate=np.empty((H,W,3),np.float32)
+    for y in range(0,H,40):
+        vals=np.stack([high[t][y:y+40].astype(np.float32) for t in sample])
+        valid=np.stack([m[y:y+40] for m in masks])
+        vals[valid.astype(bool)] = np.nan
+        import warnings
+        with warnings.catch_warnings():
+            warnings.simplefilter('ignore',RuntimeWarning)
+            tile=np.nanmedian(vals,axis=0)
+        # If the wide safety rectangles covered the same floor pixel in every
+        # sample, interpolate only those small holes from exposed neighbors.
+        missing=np.isnan(tile[:,:,0])
+        if missing.any():
+            tile=np.nan_to_num(tile).astype(np.uint8)
+            tile=cv2.inpaint(tile,missing.astype(np.uint8),5,cv2.INPAINT_TELEA)
+        plate[y:y+40]=tile
+    return plate
+
+
+def make_sprite(im, plate, kind, crop):
+    x0,y0,x1,y1=crop
+    rgb=im[y0:y1,x0:x1].astype(np.float32)
+    floor=plate[y0:y1,x0:x1]
+    d=rgb-floor
+    if kind in ('red','blue'):
+        r,g,b=rgb.transpose(2,0,1)
+        if kind=='red':
+            body=(r-g>24)&(r-b>15)
+        else:
+            body=(b-r>25)&(b-g>9)
+        n,lab,stats,_=cv2.connectedComponentsWithStats(body.astype(np.uint8))
+        j=1+np.argmax(stats[1:,cv2.CC_STAT_AREA])
+        body=binary_fill_holes(lab==j)
+        # The desaturated bright rim belongs to the mallet too.
+        body=binary_dilation(body,iterations=2)
+        alpha=gaussian_filter(body.astype(np.float32),.65)
+    else:
+        # Numerical matte of the white body against its revealed floor.  The
+        # top is neutral white; the cylindrical front is slightly violet-gray.
+        strength=d.mean(2)
+        body=(strength>6)&(rgb.min(2)>217)
+        n,lab,stats,_=cv2.connectedComponentsWithStats(body.astype(np.uint8))
+        j=1+np.argmax(stats[1:,cv2.CC_STAT_AREA])
+        body=binary_fill_holes(lab==j)
+        # Prevent a decal or a compression island from becoming foreground.
+        body=binary_dilation(body,iterations=1)
+        alpha=gaussian_filter(body.astype(np.float32),.65)
+    # Capture the remaining local lighting change, especially the soft shadow.
+    # Leave the static floor lettering on the plate, rather than on the sprite.
+    delta=np.where((np.max(np.abs(d),2)>4)[...,None],d,0)
+    delta*= (1-alpha[...,None])
+    # Ignore codec noise at the crop border and smoothly taper the shadow.
+    hh,ww=alpha.shape
+    yy,xx=np.mgrid[:hh,:ww]
+    taper=np.clip(np.minimum.reduce([xx/5,yy/5,(ww-1-xx)/5,(hh-1-yy)/5]),0,1)
+    delta*=taper[...,None]
+    return {'rgb':rgb,'alpha':alpha,'delta':delta,'origin':(x0,y0)}
+
+
+def render_sprite(canvas, sprite, anchor, position):
+    ax,ay=anchor; px,py=position
+    s=scale_at(px)/scale_at(ax)
+    ox,oy=sprite['origin']
+    # All coordinates in the affine transformation are source high-resolution
+    # pixels; subpixel translations are retained until the final Lanczos resize.
+    tx=px*5+s*(ox-ax*5)
+    ty=py*5+s*(oy-ay*5)
+    mat=np.array([[s,0,tx],[0,s,ty]],np.float32)
+    alpha=cv2.warpAffine(sprite['alpha'],mat,(W,H),flags=cv2.INTER_LINEAR)
+    premul=cv2.warpAffine(sprite['rgb']*sprite['alpha'][...,None],mat,(W,H),flags=cv2.INTER_LINEAR)
+    delta=cv2.warpAffine(sprite['delta'],mat,(W,H),flags=cv2.INTER_LINEAR)
+    canvas[:]=canvas*(1-alpha[...,None])+premul+delta
+
+
+def source_centroid(t,kind):
+    a=SOURCE[t].astype(float);r,g,b=a.transpose(2,0,1)
+    m=((r>1.6*g)&(r>1.6*b)) if kind=='red' else ((b>1.7*r)&(b>1.15*g))
+    m[:50]=False;m[90:]=False;m[:,:95]=False;m[:,179:]=False
+    yy,xx=np.where(m)
+    return float(xx.mean()),float(yy.mean())
+
+
+def generate():
+    high=decode()
+    plate=clean_plate(high)
+    white_anchor=(105.2,80.6)
+    red_anchor=source_centroid(24,'red')
+    blue_anchor=source_centroid(28,'blue')
+    white=make_sprite(high[0],plate,'white',(491,373,580,450))
+    rb,_=colored_bbox(high[24],'red')
+    bb,_=colored_bbox(high[28],'blue')
+    red=make_sprite(high[24],plate,'red',(rb[0]-9,rb[1]-7,rb[2]+13,rb[3]+22))
+    blue=make_sprite(high[28],plate,'blue',(bb[0]-9,bb[1]-7,bb[2]+13,bb[3]+22))
+
+    # Equal original masses. The intervention changes only white's inertial
+    # mass to 10; geometry, appearance, and all initial states stay unchanged.
+    m_r,m_w,m_b=1.,10.,1.
+    incident=1.94
+    vr=(m_r-RESTITUTION*m_w)/(m_r+m_w)*incident
+    vw=(1+RESTITUTION)*m_r/(m_r+m_w)*incident
+    qr0=q_from_x(XREF)
+    qw0=q_from_x(white_anchor[0])
+    qb0=q_from_x(blue_anchor[0])
+    # The red/blue contact separation is directly measured from their first
+    # factual collision, with their true size and camera projection.
+    contact_red_x=blue_anchor[0]-6.0
+    qr_second=q_from_x(contact_red_x)
+    second_time=CONTACT+(qr_second-qr0)/vr
+    vr2=(m_r-RESTITUTION*m_b)/(m_r+m_b)*vr
+    vb2=(1+RESTITUTION)*m_r/(m_r+m_b)*vr
+
+    def trajectory(t):
+        elapsed=t-CONTACT
+        qw=qw0+vw*elapsed
+        if t<second_time:
+            qr=qr0+vr*elapsed
+            qb=qb0
+        else:
+            dt=t-second_time
+            qr=qr_second+vr2*dt
+            qb=qb0+vb2*dt
+        xw,xr,xb=map(x_from_q,[qw,qr,qb])
+        # The objects all travel along the same line on the tabletop.  Their
+        # differing stem heights are transformed with the perspective scale.
+        def y_for(x,anchor):
+            ax,ay=anchor
+            ground0=124.0-.37*ax
+            stem=ground0-ay
+            return 124.0-.37*x-stem*scale_at(x)/scale_at(ax)
+        return ((xw,y_for(xw,white_anchor)),(xr,y_for(xr,(XREF,77.55))),
+                (xb,y_for(xb,blue_anchor)))
+
+    output=SOURCE.copy()
+    for t in range(30,96):
+        canvas=high[t].astype(np.float32)
+        _,rects=exclusion(high[t],t,components=True)
+        erase=np.zeros((H,W),np.float32)
+        names=['red','white']+(['blue'] if t>=second_time else [])
+        for name in names:
+            x0,y0,x1,y1=rects[name]
+            erase[y0:y1,x0:x1]=1
+        erase=gaussian_filter(erase,1.6)
+        canvas=canvas*(1-erase[...,None])+plate*erase[...,None]
+        wp,rp,bp=trajectory(t)
+        # Rearmost first, consistent with the source camera's occlusion order.
+        if t>=second_time:
+            render_sprite(canvas,blue,blue_anchor,bp)
+        render_sprite(canvas,red,red_anchor,rp)
+        render_sprite(canvas,white,white_anchor,wp)
+        image=Image.fromarray(np.clip(canvas,0,255).astype(np.uint8))
+        output[t]=np.array(image.resize((256,144),Image.Resampling.LANCZOS))
+    artifact_io.save(output,ROOT)
+    # A normal shareable MP4 complements the required lossless RGB artifacts.
+    import imageio_ffmpeg
+    writer=imageio_ffmpeg.write_frames(str(ROOT/'edited_video.mp4'),(256,144),fps=24,
+                                       codec='libx264',quality=9,pix_fmt_in='rgb24',
+                                       pix_fmt_out='yuv420p',macro_block_size=1)
+    writer.send(None)
+    for frame in output:
+        writer.send(frame)
+    writer.close()
+    print('Saved 96 frames; mass ratio = 10; secondary collision at frame',round(second_time,2))
+    print('Final positions (white, red, blue):',trajectory(95))
+
+
+if __name__=='__main__':
+    generate()

\"\"\"Counterfactual edit: a tenfold mass increase, without a size change.

All appearance is reconstructed from the supplied video.  Motion after contact
uses a perspective-correct one-dimensional elastic collision calculation.
\"\"\"
from pathlib import Path
import numpy as np
import cv2
from PIL import Image
from scipy.ndimage import gaussian_filter, binary_dilation, binary_fill_holes
import artifact_io

ROOT = Path(__file__).resolve().parent
SOURCE = np.load(ROOT / 'source_pixels.npz')['rgb']
H, W = 720, 1280
K = .0038  # perspective growth measured from the three source trajectories
XREF = 114.15
CONTACT = 29.5
RESTITUTION = .98


def q_from_x(x):
    d = XREF - x
    return d / (1 + K * d)


def x_from_q(q):
    return XREF - q / (1 - K * q)


def scale_at(x):
    return 1 + K * (XREF - x)


def colored_bbox(im, kind):
    a = im.astype(np.int16)
    r, g, b = a.transpose(2, 0, 1)
    if kind == 'red':
        m = (r-g > 38) & (r-b > 30)
    else:
        m = (b-r > 38) & (b-g > 18)
    roi = np.zeros((H, W), bool)
    roi[240:460, 460:905] = True
    m &= roi
    n, lab, stats, _ = cv2.connectedComponentsWithStats(m.astype(np.uint8))
    if n < 2:
        raise RuntimeError('Missing source object')
    j = 1 + np.argmax(stats[1:, cv2.CC_STAT_AREA])
    x,y,w,h,_ = stats[j]
    return (int(x),int(y),int(x+w),int(y+h)), lab == j


def white_source_shift(t):
    # Measured white foreground centroids, including its factual rail contact.
    times = [0,29,30,31,32,36,40,44,48,52,60,72,95]
    xs = [0,0,-.5,-2.3,-4.4,-12,-20.4,-28.4,-37.4,-38,-36.9,-35.4,-34.7]
    ys = [0,0,.35,1.1,1.85,4.55,7.8,10.9,15.4,15.7,15.1,14.9,14.1]
    return np.interp(t,times,xs)*5, np.interp(t,times,ys)*5


def exclusion(im, t, components=False):
    mask = np.zeros((H,W),np.uint8)
    rects = {}
    for kind in ['red','blue']:
        (x0,y0,x1,y1),_ = colored_bbox(im,kind)
        # Include subpixel edges and ambient contact shadows, not just pigment.
        x0,x1 = max(0,x0-15),min(W,x1+18)
        y0,y1 = max(0,y0-12),min(H,y1+25)
        mask[y0:y1,x0:x1] = 1
        rects[kind] = (x0,y0,x1,y1)
    dx,dy = white_source_shift(t)
    x0,x1 = int(491+dx),int(580+dx)
    y0,y1 = int(373+dy),int(450+dy)
    mask[y0:y1,x0:x1] = 1
    rects['white'] = (x0,y0,x1,y1)
    return (mask,rects) if components else mask


def decode():
    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
    result=[]
    for t in range(96):
        ok,bgr=cap.read()
        if not ok:
            raise RuntimeError('Incomplete source')
        result.append(cv2.cvtColor(bgr,cv2.COLOR_BGR2RGB))
    cap.release()
    return result


def clean_plate(high):
    # Every edited location is exposed in another source frame.  Excluding all
    # three objects avoids median ghosts from the two long stationary intervals.
    sample=[0,8,16,24,28,32,36,40,44,48,56,64,80,95]
    masks=[exclusion(high[t],t) for t in sample]
    plate=np.empty((H,W,3),np.float32)
    for y in range(0,H,40):
        vals=np.stack([high[t][y:y+40].astype(np.float32) for t in sample])
        valid=np.stack([m[y:y+40] for m in masks])
        vals[valid.astype(bool)] = np.nan
        import warnings
        with warnings.catch_warnings():
            warnings.simplefilter('ignore',RuntimeWarning)
            tile=np.nanmedian(vals,axis=0)
        # If the wide safety rectangles covered the same floor pixel in every
        # sample, interpolate only those small holes from exposed neighbors.
        missing=np.isnan(tile[:,:,0])
        if missing.any():
            tile=np.nan_to_num(tile).astype(np.uint8)
            tile=cv2.inpaint(tile,missing.astype(np.uint8),5,cv2.INPAINT_TELEA)
        plate[y:y+40]=tile
    return plate


def make_sprite(im, plate, kind, crop):
    x0,y0,x1,y1=crop
    rgb=im[y0:y1,x0:x1].astype(np.float32)
    floor=plate[y0:y1,x0:x1]
    d=rgb-floor
    if kind in ('red','blue'):
        r,g,b=rgb.transpose(2,0,1)
        if kind=='red':
            body=(r-g>24)&(r-b>15)
        else:
            body=(b-r>25)&(b-g>9)
        n,lab,stats,_=cv2.connectedComponentsWithStats(body.astype(np.uint8))
        j=1+np.argmax(stats[1:,cv2.CC_STAT_AREA])
        body=binary_fill_holes(lab==j)
        # The desaturated bright rim belongs to the mallet too.
        body=binary_dilation(body,iterations=2)
        alpha=gaussian_filter(body.astype(np.float32),.65)
    else:
        # Numerical matte of the white body against its revealed floor.  The
        # top is neutral white; the cylindrical front is slightly violet-gray.
        strength=d.mean(2)
        body=(strength>6)&(rgb.min(2)>217)
        n,lab,stats,_=cv2.connectedComponentsWithStats(body.astype(np.uint8))
        j=1+np.argmax(stats[1:,cv2.CC_STAT_AREA])
        body=binary_fill_holes(lab==j)
        # Prevent a decal or a compression island from becoming foreground.
        body=binary_dilation(body,iterations=1)
        alpha=gaussian_filter(body.astype(np.float32),.65)
    # Capture the remaining local lighting change, especially the soft shadow.
    # Leave the static floor lettering on the plate, rather than on the sprite.
    delta=np.where((np.max(np.abs(d),2)>4)[...,None],d,0)
    delta*= (1-alpha[...,None])
    # Ignore codec noise at the crop border and smoothly taper the shadow.
    hh,ww=alpha.shape
    yy,xx=np.mgrid[:hh,:ww]
    taper=np.clip(np.minimum.reduce([xx/5,yy/5,(ww-1-xx)/5,(hh-1-yy)/5]),0,1)
    delta*=taper[...,None]
    return {'rgb':rgb,'alpha':alpha,'delta':delta,'origin':(x0,y0)}


def render_sprite(canvas, sprite, anchor, position):
    ax,ay=anchor; px,py=position
    s=scale_at(px)/scale_at(ax)
    ox,oy=sprite['origin']
    # All coordinates in the affine transformation are source high-resolution
    # pixels; subpixel translations are retained until the final Lanczos resize.
    tx=px*5+s*(ox-ax*5)
    ty=py*5+s*(oy-ay*5)
    mat=np.array([[s,0,tx],[0,s,ty]],np.float32)
    alpha=cv2.warpAffine(sprite['alpha'],mat,(W,H),flags=cv2.INTER_LINEAR)
    premul=cv2.warpAffine(sprite['rgb']*sprite['alpha'][...,None],mat,(W,H),flags=cv2.INTER_LINEAR)
    delta=cv2.warpAffine(sprite['delta'],mat,(W,H),flags=cv2.INTER_LINEAR)
    canvas[:]=canvas*(1-alpha[...,None])+premul+delta


def source_centroid(t,kind):
    a=SOURCE[t].astype(float);r,g,b=a.transpose(2,0,1)
    m=((r>1.6*g)&(r>1.6*b)) if kind=='red' else ((b>1.7*r)&(b>1.15*g))
    m[:50]=False;m[90:]=False;m[:,:95]=False;m[:,179:]=False
    yy,xx=np.where(m)
    return float(xx.mean()),float(yy.mean())


def generate():
    high=decode()
    plate=clean_plate(high)
    white_anchor=(105.2,80.6)
    red_anchor=source_centroid(24,'red')
    blue_anchor=source_centroid(28,'blue')
    white=make_sprite(high[0],plate,'white',(491,373,580,450))
    rb,_=colored_bbox(high[24],'red')
    bb,_=colored_bbox(high[28],'blue')
    red=make_sprite(high[24],plate,'red',(rb[0]-9,rb[1]-7,rb[2]+13,rb[3]+22))
    blue=make_sprite(high[28],plate,'blue',(bb[0]-9,bb[1]-7,bb[2]+13,bb[3]+22))

    # Equal original masses. The intervention changes only white's inertial
    # mass to 10; geometry, appearance, and all initial states stay unchanged.
    m_r,m_w,m_b=1.,10.,1.
    incident=1.94
    vr=(m_r-RESTITUTION*m_w)/(m_r+m_w)*incident
    vw=(1+RESTITUTION)*m_r/(m_r+m_w)*incident
    qr0=q_from_x(XREF)
    qw0=q_from_x(white_anchor[0])
    qb0=q_from_x(blue_anchor[0])
    # The red/blue contact separation is directly measured from their first
    # factual collision, with their true size and camera projection.
    contact_red_x=blue_anchor[0]-6.0
    qr_second=q_from_x(contact_red_x)
    second_time=CONTACT+(qr_second-qr0)/vr
    vr2=(m_r-RESTITUTION*m_b)/(m_r+m_b)*vr
    vb2=(1+RESTITUTION)*m_r/(m_r+m_b)*vr

    def trajectory(t):
        elapsed=t-CONTACT
        qw=qw0+vw*elapsed
        if t<second_time:
            qr=qr0+vr*elapsed
            qb=qb0
        else:
            dt=t-second_time
            qr=qr_second+vr2*dt
            qb=qb0+vb2*dt
        xw,xr,xb=map(x_from_q,[qw,qr,qb])
        # The objects all travel along the same line on the tabletop.  Their
        # differing stem heights are transformed with the perspective scale.
        def y_for(x,anchor):
            ax,ay=anchor
            ground0=124.0-.37*ax
            stem=ground0-ay
            return 124.0-.37*x-stem*scale_at(x)/scale_at(ax)
        return ((xw,y_for(xw,white_anchor)),(xr,y_for(xr,(XREF,77.55))),
                (xb,y_for(xb,blue_anchor)))

    output=SOURCE.copy()
    for t in range(30,96):
        canvas=high[t].astype(np.float32)
        _,rects=exclusion(high[t],t,components=True)
        erase=np.zeros((H,W),np.float32)
        names=['red','white']+(['blue'] if t>=second_time else [])
        for name in names:
            x0,y0,x1,y1=rects[name]
            erase[y0:y1,x0:x1]=1
        erase=gaussian_filter(erase,1.6)
        canvas=canvas*(1-erase[...,None])+plate*erase[...,None]
        wp,rp,bp=trajectory(t)
        # Rearmost first, consistent with the source camera's occlusion order.
        if t>=second_time:
            render_sprite(canvas,blue,blue_anchor,bp)
        render_sprite(canvas,red,red_anchor,rp)
        render_sprite(canvas,white,white_anchor,wp)
        image=Image.fromarray(np.clip(canvas,0,255).astype(np.uint8))
        output[t]=np.array(image.resize((256,144),Image.Resampling.LANCZOS))
    artifact_io.save(output,ROOT)
    # A normal shareable MP4 complements the required lossless RGB artifacts.
    import imageio_ffmpeg
    writer=imageio_ffmpeg.write_frames(str(ROOT/'edited_video.mp4'),(256,144),fps=24,
                                       codec='libx264',quality=9,pix_fmt_in='rgb24',
                                       pix_fmt_out='yuv420p',macro_block_size=1)
    writer.send(None)
    for frame in output:
        writer.send(frame)
    writer.close()
    print('Saved 96 frames; mass ratio = 10; secondary collision at frame',round(second_time,2))
    print('Final positions (white, red, blue):',trajectory(95))


if __name__=='__main__':
    generate()
