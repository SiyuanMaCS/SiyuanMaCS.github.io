+"""Remove the ball, and reconstruct the car's unimpeded rolling trajectory.
+
+All scene textures and object detail are recovered from the supplied video.
+No external image, model, or renderer is used.
+"""
+from pathlib import Path
+import warnings
+import cv2
+import numpy as np
+from PIL import Image
+from scipy.ndimage import gaussian_filter, map_coordinates
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+
+def resize_rgb(a):
+    return np.asarray(Image.fromarray(np.clip(np.rint(a), 0, 255).astype(np.uint8)).resize(
+        (256, 144), Image.Resampling.LANCZOS))
+
+
+def car_centers(source):
+    # Red paint is separate from the stationary ball by component size.
+    from scipy.ndimage import label
+    out = []
+    for f in source:
+        r, g, b = f.astype(float).transpose(2, 0, 1)
+        yy, xx = np.indices(r.shape)
+        m = (r > 1.6*g) & (r > 1.45*b) & (yy < 39)
+        labs, n = label(m)
+        ids = [i for i in range(1, n+1) if np.count_nonzero(labs == i) > 80]
+        k = max(ids, key=lambda i: np.count_nonzero(labs == i))
+        # Paint bounding box is invariant despite rotation of the wheels.
+        xs = xx[labs == k]
+        out.append((xs.min()+xs.max())/2 + 1.8)
+    return np.array(out)
+
+
+def main():
+    source = np.load(ROOT/'source_pixels.npz')['rgb']
+    cap = cv2.VideoCapture(str(ROOT/'source_video.mp4'))
+    sample_ids = [0, 8, 16, 21, 24, 28, 32, 38, 48, 72, 95]
+    samples = {}
+    # Subpixel positions are measured at source resolution before collision.
+    measured = []
+    centers_lo = car_centers(source)
+    for t in range(96):
+        ok, bgr = cap.read()
+        if not ok:
+            raise RuntimeError('Incomplete source')
+        rgb = bgr[:, :, ::-1]
+        if t in sample_ids:
+            samples[t] = rgb.copy()
+        if t <= 20:
+            f = rgb.astype(np.float32)
+            yy, xx = np.indices(f.shape[:2])
+            r, g, b = f.transpose(2, 0, 1)
+            cx = centers_lo[t]
+            m = (yy >= 129) & (yy < 180) & (xx > (cx-19)*5) & (xx < (cx+19)*5)
+            weight = np.maximum(r-1.5*g, 0) * m * (r > b*1.5)
+            measured.append(float((weight*xx).sum()/weight.sum()/5))
+    cap.release()
+    # Constant rolling resistance is observed before contact at frame 21.
+    fit = np.polyfit(np.arange(21), measured, 2)
+    stop = -fit[1]/(2*fit[0])
+    free = np.polyval(fit, np.minimum(np.arange(96), stop))
+    print('Unimpeded car fit:', fit, 'stop frame:', stop, 'final center:', free[-1])
+    # Track later paint position with the same full-resolution centroid.
+    late_centers = {}
+    for t, img in samples.items():
+        if t <= 20:
+            continue
+        f = img.astype(float)
+        yy, xx = np.indices(f.shape[:2])
+        r, g, b = f.transpose(2, 0, 1)
+        cx = centers_lo[t]
+        m = (yy >= 129) & (yy < 180) & (xx > (cx-19)*5) & (xx < (cx+19)*5)
+        w = np.maximum(r-1.5*g, 0)*m*(r > b*1.5)
+        late_centers[t] = float((w*xx).sum()/w.sum()/5)
+    measured_centers = np.interp(np.arange(96), list(range(21))+list(late_centers), measured+list(late_centers.values()))
+
+    # Work at native resolution in the upper crop to retain anti-aliased toy detail.
+    H, W = 315, 1150
+    Y, X = np.indices((H, W), dtype=np.float32)
+    x, y = (X+.5)/5-.5, (Y+.5)/5-.5
+    layers = []
+    ball_centers = {0:(136.9,27.0),8:(136.9,27.0),16:(136.9,27.0),21:(136.2,27.0),
+                    24:(133.5,27.0),28:(129.5,27.0),32:(126.3,27.1),38:(116.7,41.5)}
+    for t in sample_ids:
+        f = samples[t][:H,:W].astype(np.float32)
+        c = measured_centers[t]
+        # Exclude the car silhouette, its contact reflection, and soft shadow.
+        car = (x>c-19) & (x<c+19) & (y>25.0) & (y<38.0)
+        shadow = (x>c-20) & (x<c+30) & (y<28.0)
+        invalid = car | shadow
+        if t in ball_centers:
+            bx, by = ball_centers[t]
+            ball = ((x-bx)/9.2)**2+((y-by)/8.8)**2 < 1
+            invalid |= ball
+            # Projected soft tabletop shadow trails the ball down and left.
+            invalid |= (x>bx-30)&(x<bx+11)&(y>by+3)&(y<by+32)
+        f[invalid] = np.nan
+        layers.append(f)
+    with warnings.catch_warnings():
+        warnings.simplefilter('ignore', RuntimeWarning)
+        clean = np.nanmedian(np.stack(layers), axis=0)
+    # A very small persistent occlusion at the touching silhouettes is reconstructed
+    # from the smooth wood/tabletop scanlines, without blurring their horizontal edge.
+    for row in range(H):
+        for ch in range(3):
+            good = np.isfinite(clean[row,:,ch])
+            if not good.all():
+                clean[row,~good,ch] = np.interp(np.flatnonzero(~good), np.flatnonzero(good), clean[row,good,ch])
+    first = samples[0].astype(np.float32)
+    background = first.copy()
+    background[:H,:W] = clean
+    # Original first frame is an unobstructed reference for all floor pixels.
+    # Upper reconstruction is feathered into untouched pixels below and right.
+    top_mix = np.minimum(np.clip((62-y)/4,0,1),np.clip((229-x)/5,0,1))
+    background[:H,:W] = clean*top_mix[:,:,None]+first[:H,:W]*(1-top_mix[:,:,None])
+
+    cap = cv2.VideoCapture(str(ROOT/'source_video.mp4'))
+    frames = []
+    # Everything below the tabletop is unchanged when the ball is absent.
+    # Temporal source compression noise outside the swept volume remains intact.
+    fy, fx = np.indices((720,1280), dtype=np.float32)
+    xx, yy = (fx+.5)/5-.5,(fy+.5)/5-.5
+    upper = np.clip((xx-84)/5,0,1)*np.clip((230-xx)/5,0,1)*np.clip((64-yy)/5,0,1)
+    lower = np.clip((xx-64)/5,0,1)*np.clip((145-xx)/5,0,1)*np.clip((yy-37)/5,0,1)*np.clip((141-yy)/5,0,1)
+    edit = np.maximum(upper, lower)
+    for t in range(96):
+        ok, bgr = cap.read()
+        f = bgr[:,:,::-1].astype(np.float32)
+        result = f*(1-edit[:,:,None])+background*edit[:,:,None]
+        c = measured_centers[t]
+        # Difference compositing preserves source surfaces, wheel holes, highlights,
+        # soft shadows, and reflections without inventing a toy-car appearance.
+        delta = f[:H,:W]-clean
+        # Car-only support, with clean background substituted where the ball was.
+        # The silhouette's left edge steps outward from the top to the wheels.
+        minx = np.interp(y, [0,25,27,28,29,30,31,34,35,37,63],
+                         [-16,-16,-5,-8,-14,-17,-18,-18,-17,-20,-20]) + c
+        right = c+19
+        body = np.clip((x-minx)/.45+.5,0,1)*np.clip((right-x)/.7,0,1)
+        body *= np.clip((y-25.0)/1.0,0,1)*np.clip((38.5-y)/1.5,0,1)
+        sh = np.clip((x-(c-15))/3,0,1)*np.clip((c+28-x)/4,0,1)*np.clip((28-y)/2,0,1)
+        # Shadow is achromatic and low contrast; separate it from ball texture.
+        shade = np.clip(delta.mean(axis=2), -15, 0)
+        residual = delta*body[:,:,None]
+        residual += shade[:,:,None]*sh[:,:,None]*(1-body[:,:,None])
+        # For the first 21 frames, car motion is factual and directly preserved.
+        # At contact the counterfactual continues with measured rolling resistance.
+        shift = 0 if t<=20 else float(free[t]-measured_centers[t])
+        if abs(shift)>1e-4:
+            residual = cv2.warpAffine(residual, np.array([[1,0,shift*5],[0,1,0]],np.float32),
+                                      (W,H), flags=cv2.INTER_LANCZOS4, borderMode=cv2.BORDER_CONSTANT)
+        result[:H,:W] += residual
+        frames.append(resize_rgb(result))
+    cap.release()
+    frames = np.stack(frames).astype(np.uint8)
+    artifact_io.save(frames, ROOT)
+    print('Saved', frames.shape, frames.dtype)
+
+
+if __name__ == '__main__':
+    main()

def resize_rgb(a):

def car_centers(source):
    # Red paint is separate from the stationary ball by component size.
    from scipy.ndimage import label
    out = []
    for f in source:
        r, g, b = f.astype(float).transpose(2, 0, 1)
        yy, xx = np.indices(r.shape)
        m = (r > 1.6*g) & (r > 1.45*b) & (yy < 39)
        labs, n = label(m)
        ids = [i for i in range(1, n+1) if np.count_nonzero(labs == i) > 80]
        k = max(ids, key=lambda i: np.count_nonzero(labs == i))
        # Paint bounding box is invariant despite rotation of the wheels.
        xs = xx[labs == k]
        out.append((xs.min()+xs.max())/2 + 1.8)
    return np.array(out)
