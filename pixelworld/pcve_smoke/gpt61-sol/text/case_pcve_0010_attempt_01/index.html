+"""Edit the original rigid-body shot: wooden block mass is multiplied by ten.
+
+The scene is reconstructed from the supplied RGB sequence.  The initial rolling
+approach is unchanged.  A two-body collision is integrated at the observed
+contact time, with Coulomb sliding, rolling inertia and the block's increased
+mass.  Source surface pixels provide the object appearance; there is no external
+renderer or model.
+"""
+from pathlib import Path
+import numpy as np
+from scipy.ndimage import map_coordinates, gaussian_filter
+from PIL import Image, ImageDraw
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+FPS = 24
+H, W = 144, 256
+Y, X = np.mgrid[:H, :W].astype(float)
+
+
+def sample(rgb, x, y):
+    return np.stack([map_coordinates(rgb[..., c], [y, x], order=1,
+                                    mode='nearest') for c in range(3)], -1)
+
+
+def ball_geometry(t):
+    return (15.3 + 6.92*t + .014*t*t,
+            56.25 + 1.31*t + .010*t*t,
+            15.15 + .215*t, 14.55 + .205*t)
+
+
+def polygon_mask(points, scale=4):
+    im = Image.new('L', (W*scale, H*scale))
+    ImageDraw.Draw(im).polygon([(round(x*scale), round(y*scale)) for x,y in points], fill=255)
+    return np.asarray(im.resize((W,H), Image.Resampling.LANCZOS), float)/255
+
+
+def main():
+    source = np.load(ROOT / 'source_pixels.npz')['rgb'].astype(float)
+    # The left-hand tabletop is revealed after the projectile has passed it.
+    # Temporal median retains the original photographic wood texture.
+    clean = np.median(source[26:96], axis=0)
+    base = source[0].copy()
+    erase = np.zeros((H,W), float)
+    erase[38:96, :58] = 1
+    erase = gaussian_filter(erase, .6)
+    base = base*(1-erase[...,None]) + clean*erase[...,None]
+
+    # Background for source sphere mattes is visible in the later shot.
+    sphere_bg = np.median(source[50:96], axis=0)
+    body_polygon = np.array([[109,60.5],[126.5,55.2],[164.5,61.0],
+                             [164.1,95.2],[149.5,103.5],[109.0,94.8]])
+    block_mask = polygon_mask(body_polygon)
+
+    # A smooth local warp moves the textured block and its contact shadow while
+    # leaving the camera, wall, table boundary and remote tabletop untouched.
+    region = polygon_mask([[105,55],[126,49],[169,57],[181,96],
+                           [174,119],[104,112],[98,68]])
+    region = gaussian_filter(region, 4.5)
+    region = np.maximum(region, block_mask)
+
+    # Impact measured at original frame 13 (t=.5 s), immediately after the
+    # factual approach.  Inferred original masses are approximately equal.
+    contact = 12.0
+    initial_x, initial_y, _, _ = ball_geometry(contact)
+    vin = 7.15                 # projected incoming speed, pixels/frame
+    restitution = .78
+    mass_ball, mass_block = 1.0, 10.0
+    vb0 = (1+restitution)*mass_ball/(mass_ball+mass_block)*vin
+    vs0 = (mass_ball-restitution*mass_block)/(mass_ball+mass_block)*vin
+
+    # A sphere keeps some pre-impact spin: sliding friction brings linear and
+    # angular velocities back to rolling. Low rolling resistance then carries
+    # the rebounding projectile back along the tabletop rather than freezing it.
+    sphere_x = initial_x
+    sphere_v = vs0
+    spin_surface_v = vin
+    block_x, block_v = 0.0, vb0
+    track = []
+    for i in range(96):
+        if i < contact:
+            track.append((ball_geometry(i)[0], 0.0))
+            continue
+        if i > contact:
+            for _ in range(32):
+                dt = 1/32
+                slip = sphere_v-spin_surface_v
+                if abs(slip) > .018:
+                    accel = -.31*np.sign(slip)
+                    # I=2/5 mr^2 for a solid rolling sphere.
+                    next_v = sphere_v+accel*dt
+                    next_spin = spin_surface_v-2.5*accel*dt
+                    if (next_v-next_spin)*slip < 0:
+                        next_v = next_spin = (sphere_v+.4*spin_surface_v)/1.4
+                    sphere_v, spin_surface_v = next_v, next_spin
+                else:
+                    speed = max(0, abs(sphere_v)-.0025*dt)
+                    sphere_v = np.sign(sphere_v)*speed
+                    spin_surface_v = sphere_v
+                sphere_x += sphere_v*dt
+                if block_v > 0:
+                    vnext = max(0, block_v-.29*dt)
+                    block_x += (block_v+vnext)*.5*dt
+                    block_v = vnext
+        track.append((sphere_x, block_x))
+
+    frames = np.empty((96,H,W,3), np.uint8)
+    for i, (cx, dx) in enumerate(track):
+        if i < 12:
+            frames[i] = source[i].astype(np.uint8)
+            continue
+        dy = .185*dx
+        # Increased inertia also suppresses the source's large yaw of the cube.
+        theta = np.deg2rad(.32*dx)
+        pivot = np.array([137,82.0])
+        # Smooth inverse transform, including the table-side contact shadow.
+        xr, yr = X-dx*region, Y-dy*region
+        ang = theta*region
+        u, v = xr-pivot[0], yr-pivot[1]
+        xr = pivot[0] + np.cos(ang)*u + np.sin(ang)*v
+        yr = pivot[1] - np.sin(ang)*u + np.cos(ang)*v
+        scene = sample(base,xr,yr)
+        occlusion = map_coordinates(block_mask,[yr,xr],order=1,mode='constant',cval=0)
+
+        # The rebound follows the same plane and perspective as the approach.
+        phase = np.clip((-6.92+np.sqrt(6.92**2+4*.014*max(0,cx-15.3)))/(.028),0,10)
+        sy = 56.25 + .19*(cx-15.3)
+        rx = max(13.2,15.15+.031*(cx-15.3))
+        ry = max(12.7,14.55+.0296*(cx-15.3))
+        rr = np.sqrt(((X-cx)/rx)**2 + ((Y-sy)/ry)**2)
+        alpha = np.clip((1-rr)*min(rx,ry)+.55,0,1)
+
+        k0 = int(np.floor(phase)); k1 = min(10,k0+1); mix = phase-k0
+        colors=[]
+        shadows=[]
+        for k in (k0,k1):
+            sx, yy, arx, ary = ball_geometry(k)
+            tx = sx+(X-cx)*arx/rx
+            ty = yy+(Y-sy)*ary/ry
+            pix = sample(source[k],tx,ty)
+            back = sample(sphere_bg,tx,ty)
+            # Undo the original edge's background contribution before compositing.
+            a0 = np.clip((1-rr)*min(arx,ary)+.55,0,1)
+            color = (pix-back*(1-a0[...,None]))/np.maximum(a0[...,None],.15)
+            colors.append(np.clip(color,0,255))
+            ratio = np.clip(pix/np.maximum(back,1),.35,1.0)
+            shadows.append(ratio)
+        color = (1-mix)*colors[0]+mix*colors[1]
+        shadow = (1-mix)*shadows[0]+mix*shadows[1]
+        # The source soft shadow occupies the floor below and to the right of
+        # the sphere.  Its residual multiplies, rather than replacing, wood grain.
+        sm = np.exp(-(((X-(cx+7))/(rx*1.2))**2 +
+                       ((Y-(sy+ry+3))/(ry*.56))**2)*.8)
+        sm *= (1-alpha) * (1-occlusion)
+        sm *= np.clip((Y-(sy+ry*.35))/5,0,1)
+        scene *= 1-sm[...,None]*(1-shadow)
+        a = alpha*(1-occlusion)
+        scene = scene*(1-a[...,None])+color*a[...,None]
+        frames[i] = np.clip(np.round(scene),0,255).astype(np.uint8)
+    # The first RGB frame is exactly the factual initial state; changing mass
+    # has no instantaneous visual effect on shape, material, or position.
+    frames[0] = source[0].astype(np.uint8)
+    artifact_io.save(frames, ROOT)
+    print('Saved',frames.shape,'at',FPS,'fps')
+    print('Collision sphere velocity:',round(vs0,3),'block velocity:',round(vb0,3))
+    print('Block final displacement:',np.round(track[-1][1],3),
+          'sphere final x:',round(track[-1][0],3))
+
+if __name__ == '__main__':
+    main()

\"\"\"Edit the original rigid-body shot: wooden block mass is multiplied by ten.

The scene is reconstructed from the supplied RGB sequence.  The initial rolling
approach is unchanged.  A two-body collision is integrated at the observed
contact time, with Coulomb sliding, rolling inertia and the block's increased
mass.  Source surface pixels provide the object appearance; there is no external
renderer or model.

from pathlib import Path
import numpy as np
from scipy.ndimage import map_coordinates, gaussian_filter
from PIL import Image, ImageDraw
import artifact_io

ROOT = Path(__file__).resolve().parent
FPS = 24
H, W = 144, 256
Y, X = np.mgrid[:H, :W].astype(float)


def sample(rgb, x, y):
    return np.stack([map_coordinates(rgb[..., c], [y, x], order=1,
                                    mode='nearest') for c in range(3)], -1)


def ball_geometry(t):
    return (15.3 + 6.92*t + .014*t*t,
            56.25 + 1.31*t + .010*t*t,
            15.15 + .215*t, 14.55 + .205*t)


def polygon_mask(points, scale=4):
    im = Image.new('L', (W*scale, H*scale))
    ImageDraw.Draw(im).polygon([(round(x*scale), round(y*scale)) for x,y in points], fill=255)
    return np.asarray(im.resize((W,H), Image.Resampling.LANCZOS), float)/255


def main():
    source = np.load(ROOT / 'source_pixels.npz')['rgb'].astype(float)
    # The left-hand tabletop is revealed after the projectile has passed it.
    # Temporal median retains the original photographic wood texture.
    clean = np.median(source[26:96], axis=0)
    base = source[0].copy()
    erase = np.zeros((H,W), float)
    erase[38:96, :58] = 1
    erase = gaussian_filter(erase, .6)
    base = base*(1-erase[...,None]) + clean*erase[...,None]

    # Background for source sphere mattes is visible in the later shot.
    sphere_bg = np.median(source[50:96], axis=0)
    body_polygon = np.array([[109,60.5],[126.5,55.2],[164.5,61.0],
                             [164.1,95.2],[149.5,103.5],[109.0,94.8]])
    block_mask = polygon_mask(body_polygon)

    # A smooth local warp moves the textured block and its contact shadow while
    # leaving the camera, wall, table boundary and remote tabletop untouched.
    region = polygon_mask([[105,55],[126,49],[169,57],[181,96],
                           [174,119],[104,112],[98,68]])
    region = gaussian_filter(region, 4.5)
    region = np.maximum(region, block_mask)

    # Impact measured at original frame 13 (t=.5 s), immediately after the
    # factual approach.  Inferred original masses are approximately equal.
    contact = 12.0
    initial_x, initial_y, _, _ = ball_geometry(contact)
    vin = 7.15                 # projected incoming speed, pixels/frame
    restitution = .78
    mass_ball, mass_block = 1.0, 10.0
    vb0 = (1+restitution)*mass_ball/(mass_ball+mass_block)*vin
    vs0 = (mass_ball-restitution*mass_block)/(mass_ball+mass_block)*vin

    # A sphere keeps some pre-impact spin: sliding friction brings linear and
    # angular velocities back to rolling. Low rolling resistance then carries
    # the rebounding projectile back along the tabletop rather than freezing it.
    sphere_x = initial_x
    sphere_v = vs0
    spin_surface_v = vin
    block_x, block_v = 0.0, vb0
    track = []
    for i in range(96):
        if i < contact:
            track.append((ball_geometry(i)[0], 0.0))
            continue
        if i > contact:
            for _ in range(32):
                dt = 1/32
                slip = sphere_v-spin_surface_v
                if abs(slip) > .018:
                    accel = -.31*np.sign(slip)
                    # I=2/5 mr"'^2 for a solid rolling sphere.
                    next_v = sphere_v+accel*dt
                    next_spin = spin_surface_v-2.5*accel*dt
                    if (next_v-next_spin)*slip < 0:
                        next_v = next_spin = (sphere_v+.4*spin_surface_v)/1.4
                    sphere_v, spin_surface_v = next_v, next_spin
                else:
                    speed = max(0, abs(sphere_v)-.0025*dt)
                    sphere_v = np.sign(sphere_v)*speed
                    spin_surface_v = sphere_v
                sphere_x += sphere_v*dt
                if block_v > 0:
                    vnext = max(0, block_v-.29*dt)
                    block_x += (block_v+vnext)*.5*dt
                    block_v = vnext
        track.append((sphere_x, block_x))

    frames = np.empty((96,H,W,3), np.uint8)
    for i, (cx, dx) in enumerate(track):
        if i < 12:

            continue
        dy = .185*dx
        # Increased inertia also suppresses the source'"'s large yaw of the cube.
        theta = np.deg2rad(.32*dx)
        pivot = np.array([137,82.0])
        # Smooth inverse transform, including the table-side contact shadow.
        xr, yr = X-dx*region, Y-dy*region
        ang = theta*region
        u, v = xr-pivot[0], yr-pivot[1]
        xr = pivot[0] + np.cos(ang)*u + np.sin(ang)*v
        yr = pivot[1] - np.sin(ang)*u + np.cos(ang)*v
        scene = sample(base,xr,yr)
        occlusion = map_coordinates(block_mask,[yr,xr],order=1,mode='constant',cval=0)

        phase = np.clip((-6.92+np.sqrt(6.92**2+4*.014*max(0,cx-15.3)))/(.028),0,10)
        sy = 56.25 + .19*(cx-15.3)
        rx = max(13.2,15.15+.031*(cx-15.3))
        ry = max(12.7,14.55+.0296*(cx-15.3))
        rr = np.sqrt(((X-cx)/rx)**2 + ((Y-sy)/ry)**2)
        alpha = np.clip((1-rr)*min(rx,ry)+.55,0,1)

        k0 = int(np.floor(phase)); k1 = min(10,k0+1); mix = phase-k0
        colors=[]
        shadows=[]
        for k in (k0,k1):
            sx, yy, arx, ary = ball_geometry(k)
            tx = sx+(X-cx)*arx/rx
            ty = yy+(Y-sy)*ary/ry
            pix = sample(source[k],tx,ty)
            back = sample(sphere_bg,tx,ty)
            # Undo the original edge's background contribution before compositing.
            a0 = np.clip((1-rr)*min(arx,ary)+.55,0,1)
            color = (pix-back*(1-a0[...,None]))/np.maximum(a0[...,None],.15)
            colors.append(np.clip(color,0,255))
            ratio = np.clip(pix/np.maximum(back,1),.35,1.0)
            shadows.append(ratio)
        color = (1-mix)*colors[0]+mix*colors[1]
        shadow = (1-mix)*shadows[0]+mix*shadows[1]
        # The source soft shadow occupies the floor below and to the right of
        # the sphere.  Its residual multiplies, rather than replacing, wood grain.
        sm = np.exp(-(((X-(cx+7))/(rx*1.2))**2 +
                       ((Y-(sy+ry+3))/(ry*.56))**2)*.8)
        sm *= (1-alpha) * (1-occlusion)
        sm *= np.clip((Y-(sy+ry*.35))/5,0,1)
        scene *= 1-sm[...,None]*(1-shadow)
        a = alpha*(1-occlusion)
        scene = scene*(1-a[...,None])+color*a[...,None]
        frames[i] = np.clip(np.round(scene),0,255).astype(np.uint8)
    # The first RGB frame is exactly the factual initial state; changing mass
    # has no instantaneous visual effect on shape, material, or position.
    frames[0] = source[0].astype(np.uint8)
    artifact_io.save(frames, ROOT)
    print('Saved',frames.shape,'at',FPS,'fps')
    print('Collision sphere velocity:',round(vs0,3),'block velocity:',round(vb0,3))
    print('Block final displacement:',np.round(track[-1][1],3),
          'sphere final x:',round(track[-1][0],3))

if __name__ == '__main__':
    main()
