+"""Counterfactual RGB edit: tenfold static/sliding friction on purple.
+
+All geometry, tracking, and appearance are recovered from the supplied video.
+No external renderer or model is used.  The unaffected purple sphere retains
+its initial contact because mu*N exceeds the ramp's downhill force.  The
+orange sphere retains its measured incoming motion, then traverses the bottom,
+climbs the opposite ramp, returns, and loses energy through its own friction.
+"""
+from pathlib import Path
+import numpy as np
+from PIL import Image
+import cv2
+from artifact_io import save
+
+ROOT = Path(__file__).resolve().parent
+FPS = 24
+
+
+def read_source():
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    full = []
+    while True:
+        ok, frame = cap.read()
+        if not ok:
+            break
+        full.append(cv2.cvtColor(frame, cv2.COLOR_BGR2RGB))
+    cap.release()
+    return np.stack(full), np.load(ROOT / 'source_pixels.npz')['rgb']
+
+
+def orange_geometry(a):
+    ai = a.astype(np.int16)
+    mask = ((ai[..., 0] > ai[..., 1] + 35) &
+            (ai[..., 1] > ai[..., 2] + 55)).astype(np.uint8)
+    mask[390:] = 0
+    n, lab, stats, centroids = cv2.connectedComponentsWithStats(mask)
+    j = 1 + np.argmax(stats[1:, 4])
+    x, y, w, h, count = stats[j]
+    # The coloured component excludes the thin, neutral specular cap.
+    return np.array([(x + (w - 1) / 2) / 5 - .4,
+                     (y + (h - 1) / 2) / 5 - .4]), (x, y, w, h)
+
+
+def trajectory(centers):
+    """Integrate the untouched orange ball, in projected pixels/frame units.
+
+    Ramp acceleration is fitted to the factual approach. Opposing-ramp
+    deceleration separates gravity and contact friction. The flat has the
+    smaller normal impulse/contact drag observed in the source's last roll.
+    """
+    x = centers[32, 0]
+    # Quadratic fit avoids estimating speed from the patterned surface.
+    fit = np.polyfit(np.arange(8, 32), centers[8:32, 0], 2)
+    speed = np.polyval(np.polyder(fit), 32)
+    incoming_a = 2 * fit[0]
+    opposing_a = .727
+    gravity = (opposing_a + incoming_a) / 2
+    friction = (opposing_a - incoming_a) / 2
+    flat_friction = .080
+    path = centers[:, 0].copy()
+    velocities = np.zeros(96)
+    for t in range(33, 96):
+        for _ in range(160):
+            dt = 1 / 160
+            g = gravity if x < 113.0 else (-gravity if x > 143.0 else 0.)
+            drag = friction if g else flat_friction
+            if abs(speed) < 1e-8 and abs(g) <= drag:
+                speed = 0.
+            else:
+                direction = np.sign(speed) if abs(speed) > 1e-8 else np.sign(g)
+                new_speed = speed + (g - drag * direction) * dt
+                if speed * new_speed < 0 and abs(g) <= drag:
+                    new_speed = 0.
+                x += .5 * (speed + new_speed) * dt
+                speed = new_speed
+        path[t] = x
+        velocities[t] = speed
+    return path, velocities, gravity, friction
+
+
+class Sprite:
+    def __init__(self, frame, bg, center, box):
+        # Work at original 5x resolution for clean antialiased final edges.
+        cx, cy = (center + .4) * 5
+        self.cx, self.cy = cx, cy
+        self.halfx, self.up, self.down = 70, 40, 135
+        dx, dy = np.meshgrid(np.arange(-self.halfx, self.halfx + 1),
+                             np.arange(-self.up, self.down + 1))
+        mx = (cx + dx).astype(np.float32)
+        my = (cy + dy).astype(np.float32)
+        rgb = cv2.remap(frame, mx, my, cv2.INTER_LINEAR).astype(np.float32)
+        empty = cv2.remap(bg, mx, my, cv2.INTER_LINEAR).astype(np.float32)
+        difference = rgb - empty
+        x, y, w, h = box
+        # Numeric silhouette recovery, including its achromatic white cap.
+        rx = (w - 1) / 2 + 3.0
+        ry = (h - 1) / 2 + 3.5
+        ellipse = (dx / (rx + 1.5)) ** 2 + (dy / (ry + 1.0)) ** 2 < 1
+        changed = np.max(np.abs(difference), axis=2) > 20
+        candidate = (changed & ellipse).astype(np.uint8)
+        candidate = cv2.morphologyEx(candidate, cv2.MORPH_CLOSE,
+                                     np.ones((3, 3), np.uint8))
+        # Fill the small compression holes without incorporating contact shadow.
+        contours, _ = cv2.findContours(candidate, cv2.RETR_EXTERNAL,
+                                       cv2.CHAIN_APPROX_SIMPLE)
+        body = np.zeros(candidate.shape, np.uint8)
+        if contours:
+            cv2.drawContours(body, [max(contours, key=cv2.contourArea)], -1, 1, -1)
+        body = cv2.dilate(body, np.ones((3, 3), np.uint8))
+        body = body * ellipse.astype(np.uint8)
+        self.alpha = cv2.GaussianBlur(body.astype(np.float32), (3, 3), .45)
+        self.rgb = rgb
+        # Multiplicative shadow/reflection transfer leaves the track grain intact.
+        ratio = np.clip(rgb / np.maximum(empty, 1), .12, 1.0)
+        shadow_region = ((dy > ry - 1) & (dy < 132) & (abs(dx) < 64))
+        shadow = np.mean(difference, axis=2) < -4.5
+        ratio = np.where((shadow & shadow_region)[..., None], ratio, 1.)
+        # Smooth only the faint encoded edges, not the contact core.
+        self.ratio = cv2.GaussianBlur(ratio, (3, 3), .5)
+
+    def draw(self, canvas, x, y, flip=False):
+        cx, cy = (np.array([x, y]) + .4) * 5
+        x0 = int(np.floor(cx)) - self.halfx
+        y0 = int(np.floor(cy)) - self.up
+        width, height = self.rgb.shape[1], self.rgb.shape[0]
+        # The sphere stays well inside the frame throughout this trajectory.
+        xx, yy = np.meshgrid(np.arange(width, dtype=np.float32),
+                             np.arange(height, dtype=np.float32))
+        mx = (xx - (cx - np.floor(cx))).astype(np.float32)
+        my = (yy - (cy - np.floor(cy))).astype(np.float32)
+        if flip:
+            mx = width - 1 - mx
+        alpha = cv2.remap(self.alpha, mx, my, cv2.INTER_LINEAR,
+                          borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+        rgb = cv2.remap(self.rgb, mx, my, cv2.INTER_LINEAR,
+                        borderMode=cv2.BORDER_REPLICATE)
+        ratio = cv2.remap(self.ratio, mx, my, cv2.INTER_LINEAR,
+                          borderMode=cv2.BORDER_CONSTANT, borderValue=(1, 1, 1))
+        patch = canvas[y0:y0 + height, x0:x0 + width]
+        patch *= ratio
+        patch[:] = patch * (1 - alpha[..., None]) + rgb * alpha[..., None]
+
+
+def main():
+    full, original = read_source()
+    assert full.shape == (96, 720, 1280, 3)
+    geometry = [orange_geometry(a) for a in full]
+    centers = np.stack([g[0] for g in geometry])
+    path, velocities, gravity, friction = trajectory(centers)
+
+    # Empty the orange's initial footprint using that exact part of the scene
+    # after it has left. The purple is never removed from its initial footprint.
+    bg = full[0].copy()
+    bg[250:345, 80:190] = full[20, 250:345, 80:190]
+    # Orange body templates and their source-derived shadows.  In the flat,
+    # use the clean steady geometry instead of a collision-deformed silhouette.
+    donors = list(range(7, 33)) + list(range(52, 73))
+    sprites = {i: Sprite(full[i], bg, centers[i], geometry[i][1]) for i in donors}
+    frames = []
+    for t in range(96):
+        if t <= 5:
+            frames.append(original[t].copy())
+            continue
+        if t <= 32:
+            canvas = bg.copy()
+            cx = int(round((centers[t, 0] + .4) * 5))
+            # Preserve every incoming orange RGB detail; its collision partner
+            # is outside this rectangle at every pre-collision sample.
+            canvas[245:495, cx - 67:cx + 68] = full[t, 245:495, cx - 67:cx + 68]
+        else:
+            canvas = bg.astype(np.float32)
+            x = path[t]
+            mirror = x > 128
+            donor_x = 256 - x if mirror else x
+            if donor_x >= 113:
+                # Floor is level, and lighting/size match the final source roll.
+                candidates = list(range(52, 73))
+                donor = min(candidates, key=lambda i: abs(centers[i, 0] - donor_x))
+                y = centers[70, 1]
+            else:
+                candidates = list(range(7, 33))
+                donor = min(candidates, key=lambda i: abs(centers[i, 0] - donor_x))
+                # Measured camera projection of the two symmetric ramps.
+                rampfit = np.polyfit(centers[8:32, 0], centers[8:32, 1], 1)
+                y = float(np.polyval(rampfit, donor_x))
+            sprites[donor].draw(canvas, x, y, mirror)
+            canvas = np.clip(np.rint(canvas), 0, 255).astype(np.uint8)
+        small = np.asarray(Image.fromarray(canvas).resize((256, 144), Image.Resampling.LANCZOS))
+        frames.append(small)
+    result = np.stack(frames).astype(np.uint8)
+    save(result, ROOT)
+    print('Saved', result.shape, result.dtype)
+    print('Measured ramp gravity/contact friction:', gravity, friction)
+    print('Purple friction multiplier: 10; static threshold exceeds ramp gravity.')
+    print('Orange positions:', {i: round(float(path[i]), 3) for i in [0, 24, 32, 40, 48, 64, 80, 95]})
+    print('Orange final velocity:', round(float(velocities[-1]), 5))
+
+
+if __name__ == '__main__':
+    main()

All geometry, tracking, and appearance are recovered from the supplied video.
No external renderer or model is used.  The unaffected purple sphere retains
its initial contact because mu*N exceeds the ramp's downhill force.  The
orange sphere retains its measured incoming motion, then traverses the bottom,
climbs the opposite ramp, returns, and loses energy through its own friction.

from pathlib import Path
import numpy as np
from PIL import Image
import cv2
from artifact_io import save

ROOT = Path(__file__).resolve().parent
FPS = 24


def read_source():
    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
    full = []
    while True:
        ok, frame = cap.read()
        if not ok:
            break
        full.append(cv2.cvtColor(frame, cv2.COLOR_BGR2RGB))
    cap.release()
    return np.stack(full), np.load(ROOT / 'source_pixels.npz')['rgb']


def orange_geometry(a):
    ai = a.astype(np.int16)
    mask = ((ai[..., 0] > ai[..., 1] + 35) &
            (ai[..., 1] > ai[..., 2] + 55)).astype(np.uint8)
    mask[390:] = 0
    n, lab, stats, centroids = cv2.connectedComponentsWithStats(mask)
    j = 1 + np.argmax(stats[1:, 4])
    x, y, w, h, count = stats[j]
    # The coloured component excludes the thin, neutral specular cap.
    return np.array([(x + (w - 1) / 2) / 5 - .4,
                     (y + (h - 1) / 2) / 5 - .4]), (x, y, w, h)


def trajectory(centers):
    \"\"\"Integrate the untouched orange ball, in projected pixels/frame units.

    Ramp acceleration is fitted to the factual approach. Opposing-ramp
    deceleration separates gravity and contact friction. The flat has the
    smaller normal impulse/contact drag observed in the source's last roll.
    \"\"\"
    x = centers[32, 0]
    # Quadratic fit avoids estimating speed from the patterned surface.
    fit = np.polyfit(np.arange(8, 32), centers[8:32, 0], 2)
    speed = np.polyval(np.polyder(fit), 32)
    incoming_a = 2 * fit[0]
    opposing_a = .727
    gravity = (opposing_a + incoming_a) / 2
    friction = (opposing_a - incoming_a) / 2
    flat_friction = .080
    path = centers[:, 0].copy()
    velocities = np.zeros(96)
    for t in range(33, 96):
        for _ in range(160):
            dt = 1 / 160
            g = gravity if x < 113.0 else (-gravity if x > 143.0 else 0.)
            drag = friction if g else flat_friction
            if abs(speed) < 1e-8 and abs(g) <= drag:
                speed = 0.
            else:
                direction = np.sign(speed) if abs(speed) > 1e-8 else np.sign(g)
                new_speed = speed + (g - drag * direction) * dt
                if speed * new_speed < 0 and abs(g) <= drag:
                    new_speed = 0.
                x += .5 * (speed + new_speed) * dt
                speed = new_speed
        path[t] = x
        velocities[t] = speed
    return path, velocities, gravity, friction


class Sprite:
    def __init__(self, frame, bg, center, box):
        # Work at original 5x resolution for clean antialiased final edges.
        cx, cy = (center + .4) * 5
        self.cx, self.cy = cx, cy
        self.halfx, self.up, self.down = 70, 40, 135
        dx, dy = np.meshgrid(np.arange(-self.halfx, self.halfx + 1),
                             np.arange(-self.up, self.down + 1))
        mx = (cx + dx).astype(np.float32)
        my = (cy + dy).astype(np.float32)
        rgb = cv2.remap(frame, mx, my, cv2.INTER_LINEAR).astype(np.float32)
        empty = cv2.remap(bg, mx, my, cv2.INTER_LINEAR).astype(np.float32)
        difference = rgb - empty
        x, y, w, h = box
        # Numeric silhouette recovery, including its achromatic white cap.
        rx = (w - 1) / 2 + 3.0
        ry = (h - 1) / 2 + 3.5
        ellipse = (dx / (rx + 1.5)) ** 2 + (dy / (ry + 1.0)) ** 2 < 1
        changed = np.max(np.abs(difference), axis=2) > 20
        candidate = (changed & ellipse).astype(np.uint8)
        candidate = cv2.morphologyEx(candidate, cv2.MORPH_CLOSE,
                                     np.ones((3, 3), np.uint8))
        # Fill the small compression holes without incorporating contact shadow.
        contours, _ = cv2.findContours(candidate, cv2.RETR_EXTERNAL,
                                       cv2.CHAIN_APPROX_SIMPLE)
        body = np.zeros(candidate.shape, np.uint8)
        if contours:
            cv2.drawContours(body, [max(contours, key=cv2.contourArea)], -1, 1, -1)
        body = cv2.dilate(body, np.ones((3, 3), np.uint8))
        body = body * ellipse.astype(np.uint8)
        self.alpha = cv2.GaussianBlur(body.astype(np.float32), (3, 3), .45)
        self.rgb = rgb
        # Multiplicative shadow/reflection transfer leaves the track grain intact.
        ratio = np.clip(rgb / np.maximum(empty, 1), .12, 1.0)
        shadow_region = ((dy > ry - 1) & (dy < 132) & (abs(dx) < 64))
        shadow = np.mean(difference, axis=2) < -4.5
        ratio = np.where((shadow & shadow_region)[..., None], ratio, 1.)
        # Smooth only the faint encoded edges, not the contact core.
        self.ratio = cv2.GaussianBlur(ratio, (3, 3), .5)

    def draw(self, canvas, x, y, flip=False):
        cx, cy = (np.array([x, y]) + .4) * 5
        x0 = int(np.floor(cx)) - self.halfx
        y0 = int(np.floor(cy)) - self.up
        width, height = self.rgb.shape[1], self.rgb.shape[0]
        # The sphere stays well inside the frame throughout this trajectory.
        xx, yy = np.meshgrid(np.arange(width, dtype=np.float32),
                             np.arange(height, dtype=np.float32))
        mx = xx - (cx - np.floor(cx))
        my = yy - (cy - np.floor(cy))
        if flip:
            mx = width - 1 - mx
        alpha = cv2.remap(self.alpha, mx, my, cv2.INTER_LINEAR,
                          borderMode=cv2.BORDER_CONSTANT, borderValue=0)
        rgb = cv2.remap(self.rgb, mx, my, cv2.INTER_LINEAR,
                        borderMode=cv2.BORDER_REPLICATE)
        ratio = cv2.remap(self.ratio, mx, my, cv2.INTER_LINEAR,
                          borderMode=cv2.BORDER_CONSTANT, borderValue=(1, 1, 1))
        patch = canvas[y0:y0 + height, x0:x0 + width]
        patch *= ratio
        patch[:] = patch * (1 - alpha[..., None]) + rgb * alpha[..., None]
