+def poly_mask(points, shape=(CH,CW), blur=.65):
+    im=Image.new('L',(shape[1],shape[0]))
+    p=[(int(round(x*SCALE-X0)),int(round(y*SCALE))) for x,y in points]
+    ImageDraw.Draw(im).polygon(p,fill=255)
+    if blur: im=im.filter(ImageFilter.GaussianBlur(blur))
+    return np.asarray(im,dtype=np.float32)/255.
+
+
+def approximate_white_poly(i):
+    if i in WHITE_POLYS: return WHITE_POLYS[i]
+    y=WHITE_Y[min(i,len(WHITE_Y)-1)]
+    j=min(WHITE_POLYS,key=lambda j:abs(WHITE_Y[j]-y))
+    return [(x,yy+y-WHITE_Y[j]) for x,yy in WHITE_POLYS[j]]
+
+
+def fit_motion():
+    t=np.arange(7,dtype=float)
+    def residual(p):
+        y,v,g,c=p; d=v*t+.5*g*t*t
+        return y+d/(1+c*d)-WHITE_Y[:7]
+    y,v,g,c=least_squares(residual,[7.8,4.2,2,.001],
+                          bounds=([7,2,1,0],[9,6,3,.003])).x
+    floor=78.4
+    distance=(floor-y)/(1-c*(floor-y))
+    source_apex=(59.9-y)/(1-c*(59.9-y))
+    restitution=np.sqrt(2*g*(distance-source_apex)/(v*v+2*g*distance))
+    # Unit is one source-frame interval. Only the initial speed is multiplied.
+    new_v=4*v
+    impact=(-new_v+np.sqrt(new_v*new_v+2*g*distance))/g
+    velocity_at_impact=new_v+g*impact
+    segments=[]
+    start=impact; up=-restitution*velocity_at_impact
+    for _ in range(12):
+        end=start-2*up/g
+        segments.append((start,end,up))
+        start=end; up*=restitution
+        if abs(up)<.3:break
+    def state(t):
+        if t<impact:
+            d=new_v*t+.5*g*t*t; speed=new_v+g*t
+        else:
+            d=distance;speed=0.
+            for t0,t1,u in segments:
+                if t0<=t<t1:
+                    dt=t-t0; d=distance+u*dt+.5*g*dt*dt;speed=u+g*dt;break
+        return y+d/(1+c*d),speed/(1+c*d)**2
+    print('Measured initial velocity:',v,'px/frame; gravity:',g)
+    print('Edited initial velocity:',new_v,'; first impact frame:',impact,
+          '; restitution:',restitution)
+    return state,c,y
+

+        sprites[j]=(crops[j].astype(np.float32),poly_mask(points))

+    def red_mask(i):
+        i=min(i,31); a=crops[i].astype(np.float32)
+        expected=RED_Y[i]
+        # The red object's complete front silhouette is a rounded square.
+        m=(a[:,:,0]>145)&(a[:,:,0]-a[:,:,1]>40)&(XX>=35)&(XX<100)&(YY>(expected-8)*5)&(YY<(expected+8)*5)
+        y,x=np.where(m)

+        # Full silhouette including its pale bevel and three-dimensional upper lip.
+        pts=[(left+.95,top),(right-1.35,top+.1),(right-.15,top+.8),
+             (right,bot-1.0),(right-.85,bot),(left+.7,bot-.25),(left,bot-1.2),(left,top+1.1)]
+        return poly_mask(pts),top,bot
+
+    red_masks=[red_mask(i) for i in range(32)]
+    # Fully exposed donor faces reconstruct the sliver previously occluded by white.
+    red_donors={i:(crops[i].astype(np.float32),red_masks[i][0],red_masks[i][1:]) for i in [1,4,6,15]}
+    state,c,yinit=fit_motion()
+    output=source.copy()
+    trace=[]
+    for i in range(1,30):
+        dest=background.copy()
+        wy,wv=state(float(i))
+        trace.append((i,round(wy,3),round(wv,3)))
+        # Maintain the original red shadow; difference blending does not copy the dish.
+        if i>=4:

+            shadow=np.clip(background.mean(axis=2)-f.mean(axis=2),0,30)
+            scope=np.exp(-((XX-67)/73)**6-((YY-460)/35)**4)
+            # The white die's old shadow is on its separate, rear lane.
+            scope*=np.clip((108-XX)/20,0,1)

+        cover=white_alpha[i]
+        red_tex=red_tex*cover[:,:,None]+crops[i].astype(np.float32)*(1-cover[:,:,None])

+        tex,alpha=sprites[j]
+        oldheight=WHITE_Y[j]

+        olddist=(oldheight-yinit)/(1-c*(oldheight-yinit))
+        newdist=(wy-yinit)/(1-c*(wy-yinit))
+        scale=(1+c*olddist)/(1+c*newdist)

+        dest=dest*(1-walpha[:,:,None])+premul
+        # Red remains foreground, exactly as in the source's two separated lanes.
+        dest=dest*(1-rm[:,:,None])+red_tex*rm[:,:,None]
+        im=Image.fromarray(np.clip(dest,0,255).astype(np.uint8)).resize((40,108),Image.Resampling.LANCZOS)
+        patch=np.asarray(im)
+        # Local crop only: retain every unaffected RGB pixel elsewhere.
+        output[i,:108,110:150]=patch

+    # After the small final rebounds both dice have the factual resting configuration.

def poly_mask(points, shape=(CH,CW), blur=.65):
    im=Image.new('L',(shape[1],shape[0]))
    p=[(int(round(x*SCALE-X0)),int(round(y*SCALE))) for x,y in points]
    ImageDraw.Draw(im).polygon(p,fill=255)
    if blur: im=im.filter(ImageFilter.GaussianBlur(blur))
    return np.asarray(im,dtype=np.float32)/255.


def approximate_white_poly(i):
    if i in WHITE_POLYS: return WHITE_POLYS[i]
    y=WHITE_Y[min(i,len(WHITE_Y)-1)]
    j=min(WHITE_POLYS,key=lambda j:abs(WHITE_Y[j]-y))
    return [(x,yy+y-WHITE_Y[j]) for x,yy in WHITE_POLYS[j]]


def fit_motion():
    t=np.arange(7,dtype=float)
    def residual(p):
        y,v,g,c=p; d=v*t+.5*g*t*t
        return y+d/(1+c*d)-WHITE_Y[:7]
    y,v,g,c=least_squares(residual,[7.8,4.2,2,.001],
                          bounds=([7,2,1,0],[9,6,3,.003])).x
    floor=78.4
    distance=(floor-y)/(1-c*(floor-y))
    source_apex=(59.9-y)/(1-c*(59.9-y))
    restitution=np.sqrt(2*g*(distance-source_apex)/(v*v+2*g*distance))
    # Unit is one source-frame interval. Only the initial speed is multiplied.
    new_v=4*v
    impact=(-new_v+np.sqrt(new_v*new_v+2*g*distance))/g
    velocity_at_impact=new_v+g*impact
    segments=[]
    start=impact; up=-restitution*velocity_at_impact
    for _ in range(12):
        end=start-2*up/g
        segments.append((start,end,up))
        start=end; up*=restitution
        if abs(up)<.3:break
    def state(t):
        if t<impact:
            d=new_v*t+.5*g*t*t; speed=new_v+g*t
        else:
            d=distance;speed=0.
            for t0,t1,u in segments:
                if t0<=t<t1:
                    dt=t-t0; d=distance+u*dt+.5*g*dt*dt;speed=u+g*dt;break
        return y+d/(1+c*d),speed/(1+c*d)**2
    print('Measured initial velocity:',v,'px/frame; gravity:',g)
    print('Edited initial velocity:',new_v,'; first impact frame:',impact,
          '; restitution:',restitution)
    return state,c,y
