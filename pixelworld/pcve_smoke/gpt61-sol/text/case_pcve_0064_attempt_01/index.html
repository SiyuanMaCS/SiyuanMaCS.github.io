+"""Source-pixel counterfactual: delete the cup at source frame ten.
+
+All appearance comes from the supplied RGB timeline.  A one-dimensional rigid
+contact calculation replaces the cup-mediated collision with can/bottle contact.
+No image models or external assets are used.
+"""
+from pathlib import Path
+import numpy as np
+from scipy.ndimage import map_coordinates, gaussian_filter
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+def sample_rgb(a, yy, xx):
+    return np.stack([map_coordinates(a[..., c], [yy, xx], order=1,
+                     mode='nearest', prefilter=False) for c in range(3)], -1)
+
+def clean_plate(source):
+    # The camera and the background geometry are stationary. The later interval
+    # exposes the left part of the tabletop; the early interval exposes the right.
+    left = np.median(source[43:80], axis=0)
+    right = np.median(source[:3], axis=0)
+    plate = left.copy()
+    w = np.clip((np.arange(256) - 185) / 10, 0, 1)[None, :, None]
+    plate = left * (1-w) + right * w
+    return plate
+
+def can_matte():
+    yy, xx = np.indices((144,256))
+    # Coverage of the small upright can, with independent contact shadow below.
+    a = np.zeros((144,256), float)
+    for y in range(64,82):
+        if y == 64:
+            l,r,k = 137.6,144.9,.68
+        elif y == 65:
+            l,r,k = 136.6,145.6,1.
+        elif y == 66:
+            l,r,k = 136.25,145.85,1.
+        elif y <= 79:
+            l,r,k = 135.9,145.75,1.
+        elif y == 80:
+            l,r,k = 136.,145.8,1.
+        else:
+            l,r,k = 136.6,145.5,.88
+        a[y] = np.minimum(np.clip(xx[y]+.5-l,0,1),np.clip(r-(xx[y]-.5),0,1))*k
+    return a
+
+def bottle_matte():
+    # Glass shoulders: mostly reflective, with background transmission at the
+    # rim and clear neck. Green glass at the base is opaque at this resolution.
+    a=np.zeros((144,256),float)
+    profiles = {
+      63:(166.8,170.3,.17), 64:(166.7,170.5,.76),
+      65:(166.7,170.55,1.), 66:(166.5,174.9,.62),
+      67:(166.6,174.9,.93), 68:(166.65,174.8,.96),
+      69:(166.5,174.9,.94), 70:(166.5,174.9,.95),
+      71:(166.5,174.9,.95), 72:(166.45,174.95,.95),
+      73:(166.3,174.9,.98), 74:(166.2,174.95,1.),
+      75:(166.2,174.95,1.), 76:(166.2,174.95,1.),
+      77:(166.4,174.85,1.), 78:(166.7,174.65,1.),
+      79:(166.7,174.6,1.), 80:(166.7,174.5,1.),
+      81:(166.7,174.5,1.), 82:(167.,174.6,.7)
+    }
+    x=np.arange(256)
+    for y,(l,r,k) in profiles.items():
+        a[y]=np.minimum(np.clip(x+.5-l,0,1),np.clip(r-(x-.5),0,1))*k
+    return a
+
+def make_sprite(image, plate, alpha, bounds):
+    x0,y0,x1,y1=bounds
+    yy,xx=np.indices(alpha.shape)
+    crop=(xx>=x0)&(xx<x1)&(yy>=y0)&(yy<y1)
+    # Small signed source residual retains antialiasing, highlights, and shadows;
+    # the matte separates opaque colors from background transmission.
+    residual=image-(1-alpha[...,None])*plate
+    residual[~crop]=0
+    alpha=alpha*crop
+    # Feather the crop, not the object itself, to eliminate codec-noise seams.
+    edge=np.minimum.reduce([xx-x0, x1-1-xx, yy-y0, y1-1-yy]).astype(float)
+    fade=np.clip((edge+1)/2,0,1)*crop
+    return residual*fade[...,None],alpha*fade
+
+def erase_region(frame,plate,mask,cx,xradius,y0,y1):
+    yy,xx=np.indices(mask.shape)
+    wx=np.clip(xradius+.5-np.abs(xx-cx),0,1)
+    wy=np.minimum(np.clip(yy-y0+1,0,1),np.clip(y1-yy,0,1))
+    region=wx*wy
+    mask[:]=np.maximum(mask,region)
+
+def render_sprite(frame,plate,residual,alpha,dx):
+    yy,xx=np.indices(alpha.shape,dtype=float)
+    coords=[yy,xx-dx]
+    aa=map_coordinates(alpha,coords,order=1,mode='constant',cval=0,prefilter=False)
+    rr=np.stack([map_coordinates(residual[...,c],coords,order=1,
+            mode='constant',cval=0,prefilter=False) for c in range(3)],-1)
+    # Transparent sprite representation uses rr = color - (1-alpha)*old plate.
+    # Outside the original bounds both rr and alpha are zero.
+    frame[:] = frame*(1-aa[...,None])+rr
+    return frame
+
+def centers(source):
+    yy,xx=np.indices((144,256))
+    cans=[];bottles=[];cups=[]
+    ref=np.median(source[43:80],0)
+    luma=np.array([.2126,.7152,.0722])
+    for i,a in enumerate(source):
+        r,g,b=np.moveaxis(a,-1,0)
+        m=(r-g>55)&(r-b>60)&(yy>63)&(yy<82)
+        cans.append(float(xx[m].mean()))
+        m=(g-r>15)&(g-b>15)&(yy>60)&(yy<84)
+        bottles.append(float(xx[m].mean()))
+        # Cup's lid follows the can. In overlap use the measured rigid gap.
+        if i<10:
+            d=(a[62]-ref[62])@luma
+            cols=np.where((d>15)&(np.arange(256)>80)&(np.arange(256)<240))[0]
+            cups.append(float(cols.mean()))
+        else:
+            # Track the lid's geometric center even when its straw tilts.
+            cups.append(cans[-1]+13.0)
+    return np.array(cans),np.array(bottles),np.array(cups)
+
+def physical_motion(can0,bottle0,v0,n=87):
+    # Pixels per original frame, fitted to the factual pre-intervention motion.
+    c,b=can0,bottle0
+    vc,vb=v0,0.
+    gap=9.75
+    mc,mb=1.,1.08
+    restitution=.025
+    friction_c,friction_b=.255,.225
+    positions=[]
+    contact_time=None
+    dt=1/160
+    for j in range(n):
+        positions.append((c,b))
+        for k in range(160):
+            oldgap=b-c
+            c+=vc*dt;b+=vb*dt
+            if b-c<gap and vc>vb:
+                p=mc*vc+mb*vb
+                rel=vc-vb
+                vc=(p-mb*restitution*rel)/(mc+mb)
+                vb=(p+mc*restitution*rel)/(mc+mb)
+                correction=gap-(b-c)
+                c-=correction*mb/(mc+mb);b+=correction*mc/(mc+mb)
+                if contact_time is None:contact_time=9+j+(k+1)*dt
+            vc=max(0.,vc-friction_c*dt)
+            vb=max(0.,vb-friction_b*dt)
+    return np.array(positions),contact_time
+
+def main():
+    source=np.load(ROOT/'source_pixels.npz')['rgb'].astype(float)
+    assert source.shape==(96,144,256,3)
+    plate=clean_plate(source)
+    cx,bx,ux=centers(source)
+    # Accurate current position and stable rigid shape anchor from the source.
+    ca=can_matte();ba=bottle_matte()
+    cres,ca=make_sprite(source[9],plate,ca,(133,62,149,86))
+    bres,ba=make_sprite(source[0],plate,ba,(163,61,184,86))
+    # Eliminate residual upper corners from the neighboring factual soda cup.
+    cres[:64]=0;ca[:64]=0
+    # In source images at frame 10 the bottle is still stationary.
+    v0=float(cx[9]-cx[8]-.12)
+    motion,contact=physical_motion(cx[9],bx[9],v0)
+    output=source.copy()
+    for i in range(9,96):
+        frame=source[i].copy()
+        removal=np.zeros((144,256),float)
+        # Original positions are removed including their contact and cast shadows.
+        erase_region(frame,plate,removal,cx[i],9.,62,86)
+        erase_region(frame,plate,removal,ux[i],10.,55,86)
+        erase_region(frame,plate,removal,bx[i]+2.,12.,61,86)
+        # Cup straw can tilt up/right of the lid at its eventual stopping point.
+        if i>=22:
+            erase_region(frame,plate,removal,ux[i]+2.,8.,54,65)
+        frame=frame*(1-removal[...,None])+plate*removal[...,None]
+        can_x,bottle_x=motion[i-9]
+        # First composite the taller bottle, then the can (in front), as in source.
+        frame=render_sprite(frame,plate,bres,ba,bottle_x-bx[0])
+        frame=render_sprite(frame,plate,cres,ca,can_x-cx[9])
+        output[i]=np.clip(np.rint(frame),0,255)
+    frames=output.astype(np.uint8)
+    assert np.array_equal(frames[:9],source[:9].astype(np.uint8))
+    artifact_io.save(frames,ROOT)
+    np.savez_compressed(ROOT/'motion_diagnostics.npz',positions=motion,
+                        source_can=cx,source_bottle=bx,source_cup=ux)
+    print('Saved',frames.shape,'RGB video at 24 fps; direct contact original time frame',contact+1)
+    print('Final can/bottle x:',motion[-1].tolist())
+
+if __name__=='__main__':
+    main()

\"\"\"Source-pixel counterfactual: delete the cup at source frame ten.

All appearance comes from the supplied RGB timeline.  A one-dimensional rigid
contact calculation replaces the cup-mediated collision with can/bottle contact.
No image models or external assets are used.

from pathlib import Path
import numpy as np
from scipy.ndimage import map_coordinates, gaussian_filter
import artifact_io

ROOT = Path(__file__).resolve().parent

def sample_rgb(a, yy, xx):
    return np.stack([map_coordinates(a[..., c], [yy, xx], order=1,
                     mode='nearest', prefilter=False) for c in range(3)], -1)

def clean_plate(source):
    # The camera and the background geometry are stationary. The later interval
    # exposes the left part of the tabletop; the early interval exposes the right.
    left = np.median(source[43:80], axis=0)
    right = np.median(source[:3], axis=0)
    plate = left.copy()
    w = np.clip((np.arange(256) - 185) / 10, 0, 1)[None, :, None]
    plate = left * (1-w) + right * w
    return plate

def can_matte():
    yy, xx = np.indices((144,256))
    # Coverage of the small upright can, with independent contact shadow below.
    a = np.zeros((144,256), float)
    for y in range(64,82):
        if y == 64:
            l,r,k = 137.6,144.9,.68
        elif y == 65:
            l,r,k = 136.6,145.6,1.
        elif y == 66:
            l,r,k = 136.25,145.85,1.
        elif y <= 79:
            l,r,k = 135.9,145.75,1.
        elif y == 80:
            l,r,k = 136.,145.8,1.
        else:
            l,r,k = 136.6,145.5,.88
        a[y] = np.minimum(np.clip(xx[y]+.5-l,0,1),np.clip(r-(xx[y]-.5),0,1))*k
    return a

def bottle_matte():
    # Glass shoulders: mostly reflective, with background transmission at the
    # rim and clear neck. Green glass at the base is opaque at this resolution.
    a=np.zeros((144,256),float)
    profiles = {
      63:(166.8,170.3,.17), 64:(166.7,170.5,.76),
      65:(166.7,170.55,1.), 66:(166.5,174.9,.62),
      67:(166.6,174.9,.93), 68:(166.65,174.8,.96),
      69:(166.5,174.9,.94), 70:(166.5,174.9,.95),
      71:(166.5,174.9,.95), 72:(166.45,174.95,.95),
      73:(166.3,174.9,.98), 74:(166.2,174.95,1.),
      75:(166.2,174.95,1.), 76:(166.2,174.95,1.),
      77:(166.4,174.85,1.), 78:(166.7,174.65,1.),
      79:(166.7,174.6,1.), 80:(166.7,174.5,1.),
      81:(166.7,174.5,1.), 82:(167.,174.6,.7)
    }
    x=np.arange(256)
    for y,(l,r,k) in profiles.items():
        a[y]=np.minimum(np.clip(x+.5-l,0,1),np.clip(r-(x-.5),0,1))*k
    return a

def make_sprite(image, plate, alpha, bounds):
    x0,y0,x1,y1=bounds
    yy,xx=np.indices(alpha.shape)
    crop=(xx>=x0)&(xx<x1)&(yy>=y0)&(yy<y1)
    # Small signed source residual retains antialiasing, highlights, and shadows;
    # the matte separates opaque colors from background transmission.
    residual=image-(1-alpha[...,None])*plate
    residual[~crop]=0
    alpha=alpha*crop
    # Feather the crop, not the object itself, to eliminate codec-noise seams.
    edge=np.minimum.reduce([xx-x0, x1-1-xx, yy-y0, y1-1-yy]).astype(float)
    fade=np.clip((edge+1)/2,0,1)*crop
    return residual*fade[...,None],alpha*fade

def erase_region(frame,plate,mask,cx,xradius,y0,y1):
    yy,xx=np.indices(mask.shape)
    wx=np.clip(xradius+.5-np.abs(xx-cx),0,1)
    wy=np.minimum(np.clip(yy-y0+1,0,1),np.clip(y1-yy,0,1))
    region=wx*wy
    mask[:]=np.maximum(mask,region)

def render_sprite(frame,plate,residual,alpha,dx):
    yy,xx=np.indices(alpha.shape,dtype=float)
    coords=[yy,xx-dx]
    aa=map_coordinates(alpha,coords,order=1,mode='constant',cval=0,prefilter=False)
    rr=np.stack([map_coordinates(residual[...,c],coords,order=1,
            mode='constant',cval=0,prefilter=False) for c in range(3)],-1)
    # Transparent sprite representation uses rr = color - (1-alpha)*old plate.
    # Outside the original bounds both rr and alpha are zero.
    frame[:] = frame*(1-aa[...,None])+rr
    return frame

def centers(source):
    yy,xx=np.indices((144,256))
    cans=[];bottles=[];cups=[]
    ref=np.median(source[43:80],0)
    luma=np.array([.2126,.7152,.0722])
    for i,a in enumerate(source):
        r,g,b=np.moveaxis(a,-1,0)
        m=(r-g>55)&(r-b>60)&(yy>63)&(yy<82)
        cans.append(float(xx[m].mean()))
        m=(g-r>15)&(g-b>15)&(yy>60)&(yy<84)
        bottles.append(float(xx[m].mean()))
        # Cup's lid follows the can. In overlap use the measured rigid gap.
        if i<10:
            d=(a[62]-ref[62])@luma
            cols=np.where((d>15)&(np.arange(256)>80)&(np.arange(256)<240))[0]
            cups.append(float(cols.mean()))
        else:
            # Track the lid's geometric center even when its straw tilts.
            cups.append(cans[-1]+13.0)
    return np.array(cans),np.array(bottles),np.array(cups)

def physical_motion(can0,bottle0,v0,n=87):
    # Pixels per original frame, fitted to the factual pre-intervention motion.
    c,b=can0,bottle0
    vc,vb=v0,0.
    gap=9.75
    mc,mb=1.,1.08
    restitution=.025
    friction_c,friction_b=.255,.225
    positions=[]
    contact_time=None
    dt=1/160
    for j in range(n):
        positions.append((c,b))
        for k in range(160):
            oldgap=b-c
            c+=vc*dt;b+=vb*dt
            if b-c<gap and vc>vb:
                p=mc*vc+mb*vb
                rel=vc-vb
                vc=(p-mb*restitution*rel)/(mc+mb)
                vb=(p+mc*restitution*rel)/(mc+mb)
                correction=gap-(b-c)
                c-=correction*mb/(mc+mb);b+=correction*mc/(mc+mb)
                if contact_time is None:contact_time=9+j+(k+1)*dt
            vc=max(0.,vc-friction_c*dt)
            vb=max(0.,vb-friction_b*dt)
    return np.array(positions),contact_time

def main():
    source=np.load(ROOT/'source_pixels.npz')['rgb'].astype(float)
    assert source.shape==(96,144,256,3)
    plate=clean_plate(source)
    cx,bx,ux=centers(source)
    # Accurate current position and stable rigid shape anchor from the source.
    ca=can_matte();ba=bottle_matte()
    cres,ca=make_sprite(source[9],plate,ca,(133,62,149,86))
    bres,ba=make_sprite(source[0],plate,ba,(163,61,184,86))
    # Eliminate residual upper corners from the neighboring factual soda cup.
    cres[:64]=0;ca[:64]=0
    # In source images at frame 10 the bottle is still stationary.
    v0=float(cx[9]-cx[8]-.12)
    motion,contact=physical_motion(cx[9],bx[9],v0)
    output=source.copy()
    for i in range(9,96):
        frame=source[i].copy()
        removal=np.zeros((144,256),float)
        # Original positions are removed including their contact and cast shadows.
        erase_region(frame,plate,removal,cx[i],9.,62,86)
        erase_region(frame,plate,removal,ux[i],10.,55,86)
        erase_region(frame,plate,removal,bx[i]+2.,12.,61,86)
        # Cup straw can tilt up/right of the lid at its eventual stopping point.
        if i>=22:
            erase_region(frame,plate,removal,ux[i]+2.,8.,54,65)
        frame=frame*(1-removal[...,None])+plate*removal[...,None]
        can_x,bottle_x=motion[i-9]
        # First composite the taller bottle, then the can (in front), as in source.
        frame=render_sprite(frame,plate,bres,ba,bottle_x-bx[0])
        frame=render_sprite(frame,plate,cres,ca,can_x-cx[9])
        output[i]=np.clip(np.rint(frame),0,255)
    frames=output.astype(np.uint8)
    assert np.array_equal(frames[:9],source[:9].astype(np.uint8))
    artifact_io.save(frames,ROOT)
    np.savez_compressed(ROOT/'motion_diagnostics.npz',positions=motion,
                        source_can=cx,source_bottle=bx,source_cup=ux)
    print('Saved',frames.shape,'RGB video at 24 fps; direct contact original time frame',contact+1)
    print('Final can/bottle x:',motion[-1].tolist())

if __name__=='__main__':
    main()
