+"""Source-pixel counterfactual: threefold volleyball contact friction.
+
+All geometry and ball texture are inferred from this case's source video.
+The observed (rolling/sliding) friction slowdown is integrated on its ground
+coordinate, not in perspective screen coordinates. Increasing the contact
+coefficient threefold compresses the velocity profile and divides travel by
+three. The ball, panels, contact shadow and its original silhouette are
+resampled together; the fixed scene is otherwise copied from the source.
+"""
+from pathlib import Path
+import numpy as np
+import cv2
+from PIL import Image
+from scipy.spatial.transform import Rotation, Slerp
+from scipy.ndimage import gaussian_filter
+import artifact_io
+
+CASE = Path(__file__).resolve().parent
+# Numerically fitted high-resolution RGB silhouettes, in output pixel units.
+POSE = np.array([
+ [81.401,121.114,12.126],[85.379,114.529,11.474],
+ [89.251,108.589,10.930],[92.516,102.991,10.218],
+ [95.747,97.968,9.865],[98.703,93.293,9.447],
+ [101.362,88.918,9.202],[103.904,85.059,8.798],
+ [106.226,81.533,8.471],[108.132,78.500,8.190],
+ [109.697,75.953,8.027],[111.091,73.653,7.844],
+ [112.293,71.820,7.648],[113.278,70.321,7.585],
+ [114.179,68.972,7.433],[114.827,67.982,7.339],
+ [115.268,67.224,7.276],[115.590,66.738,7.231],
+ [115.736,66.440,7.205],[115.780,66.384,7.208]], dtype=float)
+# Optical panel correspondences supply the observed rolling axis.
+ROLL = np.array([[.79867035,.20391437,.36735313],
+ [.81089181,.17102341,.33479372],
+ [.78868250,.19019410,.29587239],
+ [.81274595,.16194609,.29783658],
+ [.81505977,.12384620,.29712405]])
+
+
+def read_source():
+    video = cv2.VideoCapture(str(CASE / 'source_video.mp4'))
+    frames = []
+    while True:
+        ok, bgr = video.read()
+        if not ok:
+            break
+        frames.append(cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB))
+    video.release()
+    assert len(frames) == 96
+    return frames
+
+
+def remap(a, x, y):
+    return cv2.remap(a.astype(np.float32), x.astype(np.float32),
+                     y.astype(np.float32), cv2.INTER_LINEAR,
+                     borderMode=cv2.BORDER_REFLECT_101)
+
+
+def region_mask(x, y, p):
+    """Feathered region enclosing the ball and its oblique cast shadow."""
+    cx, cy, r = p
+    ball = np.clip((r + 2.0 - np.hypot(x-cx, y-cy))/1.7, 0, 1)
+    ell = np.sqrt(((x-(cx-1.25*r))/(2.15*r))**2 +
+                  ((y-(cy+.65*r))/(1.95*r))**2)
+    shadow = np.clip((1.12-ell)/.16, 0, 1)
+    return np.maximum(ball, shadow)
+
+
+def make_background(source, grid_x, grid_y):
+    # Late pixels reveal the initial path. Initial pixels reveal the final
+    # resting ball's footprint. Both are the same stationary scene.
+    bg = np.median(np.stack(source[34:48]), axis=0).astype(np.float32)
+    m = region_mask(grid_x, grid_y, POSE[-1]*5)
+    # Enclose the entire final shadow, including its faint lower fringe.
+    m = np.maximum(m, region_mask(grid_x, grid_y,
+                                 np.array([114.9,69.2,8.4])*5))
+    bg = bg*(1-m[...,None]) + source[0].astype(float)*m[...,None]
+    return bg.astype(np.float32)
+
+
+def main():
+    source = read_source()
+    low_source = np.load(CASE/'source_pixels.npz')['rgb']
+    h,w = source[0].shape[:2]
+    Y,X = np.indices((h,w), dtype=np.float32)
+    clean = make_background(source,X,Y)
+    rotations = [Rotation.identity()]
+    for rv in ROLL:
+        rotations.append(Rotation.from_rotvec(rv)*rotations[-1])
+    spherical_time = Slerp(np.arange(6), Rotation.concatenate(rotations))
+
+    # Projective ground coordinate calibrated with the apparent sphere radius.
+    screen_s = (POSE[:,1]-POSE[0,1])/(POSE[-1,1]-POSE[0,1])
+    slope, intercept = np.polyfit(screen_s, POSE[:,2],1)
+    perspective_k = -slope/(intercept+slope)
+    ground_d = screen_s/(1+perspective_k-perspective_k*screen_s)
+
+    shadow_maps = []
+    for k in range(6):
+        # Preserve actual shadow falloff and colour. Separating it from the
+        # sphere allows spherical texture reprojection without warping floor.
+        luma_s = source[k].astype(float).mean(2)
+        luma_b = clean.mean(2)
+        ratio = np.clip((luma_s+1)/(luma_b+1),.15,1)
+        cx,cy,r = POSE[k]*5
+        outside = np.clip((np.hypot(X-cx,Y-cy)-r+.35)/2.0,0,1)
+        stencil = region_mask(X,Y,POSE[k]*5)*outside
+        ratio = 1-(1-ratio)*stencil
+        shadow_maps.append(ratio.astype(np.float32))
+
+    output = np.empty((96,144,256,3),dtype=np.uint8)
+    output[0] = low_source[0]  # unchanged initial state, intervention is physical
+    trajectory = []
+    for t in range(1,96):
+        # v_cf(t)=v_source(3t); hence distance_cf(t)=distance_source(3t)/3.
+        d = np.interp(min(t*3,19), np.arange(20), ground_d)/3.0
+        s = (1+perspective_k)*d/(1+perspective_k*d)
+        equiv = float(np.interp(s,screen_s,np.arange(20)))
+        center = POSE[0,:2] + s*(POSE[-1,:2]-POSE[0,:2])
+        radius = float(np.interp(equiv,np.arange(20),POSE[:,2]))
+        p = np.r_[center,radius]*5
+        trajectory.append([t,*center,radius,equiv])
+        # Remove only the factual moving object's original footprint.
+        factual = POSE[min(t,19)]*5
+        rm = region_mask(X,Y,factual)
+        canvas = source[t].astype(np.float32)
+        canvas = canvas*(1-rm[...,None]) + clean*rm[...,None]
+        cx,cy,r=p
+        xa=max(0,int(cx-4*r-12));xb=min(w,int(cx+1.7*r+12))
+        ya=max(0,int(cy-2.4*r-12));yb=min(h,int(cy+3*r+12))
+        gx,gy=X[ya:yb,xa:xb],Y[ya:yb,xa:xb]
+        nx,ny=(gx-cx)/r,(gy-cy)/r
+        rr=np.hypot(nx,ny)
+        lo=min(4,int(np.floor(equiv))); hi=lo+1; frac=equiv-lo
+        # Interpolate source cast shadows at the same physical ground point.
+        shadow=np.zeros(gx.shape,np.float32)
+        for k,weight in [(lo,1-frac),(hi,frac)]:
+            scx,scy,sr=POSE[k]*5
+            shadow += weight*remap(shadow_maps[k], scx+nx*sr,scy+ny*sr)
+        patch=canvas[ya:yb,xa:xb]*shadow[...,None]
+
+        # Subframe texture interpolation on the sphere, rather than dissolving
+        # two differently positioned/rotated balls. Only source RGB is used.
+        rho=np.minimum(rr,.9999)
+        safe=np.maximum(rr,1e-7)
+        surf=np.stack([nx*rho/safe,ny*rho/safe,
+                       np.sqrt(np.maximum(0,1-rho*rho))],axis=-1)
+        q=spherical_time(equiv)
+        color=np.zeros((*gx.shape,3),np.float32)
+        weights=np.zeros(gx.shape,np.float32)
+        for k,weight in [(lo,1-frac),(hi,frac)]:
+            transform=(rotations[k]*q.inv()).as_matrix()
+            normal=surf@transform.T
+            scx,scy,sr=POSE[k]*5
+            sampled=remap(source[k], scx+normal[...,0]*sr,
+                          scy+normal[...,1]*sr)
+            # Weight by source visibility: newly revealed panels are supplied
+            # by the other side of the optical correspondence.
+            vis=np.clip((normal[...,2]-.015)/.18,0,1)
+            wt=weight*vis
+            color+=sampled*wt[...,None];weights+=wt
+        # At the extreme limb, nearest observed source keeps an opaque sphere.
+        k=int(round(equiv));scx,scy,sr=POSE[k]*5
+        backup=remap(source[k],scx+nx*sr,scy+ny*sr)
+        color=np.where((weights>1e-5)[...,None],
+                       color/np.maximum(weights[...,None],1e-5),backup)
+        alpha=np.clip((1.005-rr)*r/1.25,0,1)
+        patch=patch*(1-alpha[...,None])+color*alpha[...,None]
+        canvas[ya:yb,xa:xb]=patch
+        # Exactly the same Lanczos reduction that created source_pixels.npz.
+        frame=np.array(Image.fromarray(np.clip(canvas,0,255).astype(np.uint8))
+                       .resize((256,144),Image.Resampling.LANCZOS))
+        output[t]=frame
+    artifact_io.save(output,CASE)
+    np.save(CASE/'.tmp'/'edited_trajectory.npy',np.array(trajectory))
+    print('Saved 96 RGB frames. Final ball center:',trajectory[-1][1:3],
+          '; friction multiplier: 3; stop at source frame 8.')
+
+if __name__ == '__main__':
+    main()
