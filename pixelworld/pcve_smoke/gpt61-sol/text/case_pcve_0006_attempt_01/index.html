+"""Remove the mallet at the specified source-timeline intervention.
+
+All textures come from unobstructed pixels of this source shot.  The later
+white-object displacement is the result of the removed mallet's second impact,
+so its pre-impact appearance and position are retained instead.  The blue
+object's earlier interaction precedes the intervention and is not undone.
+"""
+from pathlib import Path
+
+import numpy as np
+from scipy.ndimage import binary_dilation, gaussian_filter
+
+import artifact_io
+
+
+def generate(case_dir):
+    case_dir = Path(case_dir)
+    source = np.load(case_dir / "source_pixels.npz")["rgb"]
+    output = source.copy()
+    height, width = source.shape[1:3]
+    yy, xx = np.indices((height, width))
+
+    # The shot is locked off. Frame 21 supplies the unstruck white object and
+    # the entire later motion corridor, before either object has reached it.
+    plate = source[20].astype(np.float32)
+
+    # Recover the small currently occluded part from later source exposures,
+    # where the mallet has vacated it. This does not import a new scene.
+    late_plate = np.median(source[45:86].astype(np.float32), axis=0)
+    r, g, b = plate.transpose(2, 0, 1)
+    head = (
+        (r > 1.35 * g)
+        & (r > 1.25 * b)
+        & (xx >= 121)
+        & (xx <= 138)
+        & (yy >= 64)
+        & (yy <= 78)
+    )
+    # Include antialiasing, the handle, contact shadow and compression fringe.
+    repair = binary_dilation(head, iterations=3)
+    repair &= (xx <= 139) & (yy >= 63)
+    repair_alpha = gaussian_filter(repair.astype(np.float32), 0.55)
+    repair_alpha[repair] = 1.0
+    plate = plate * (1.0 - repair_alpha[..., None]) + late_plate * repair_alpha[..., None]
+
+    # A time-invariant matte avoids moving inpainting boundaries. Its core is
+    # the union of the mallet sweep and the white object's subsequent motion
+    # and shadows. It excludes the blue object and the yellow table border.
+    mallet_corridor = (
+        (xx >= 96) & (xx <= 139) & (yy >= 63) & (yy <= 99)
+    )
+    white_corridor = (
+        (xx >= 57) & (xx <= 119) & (yy >= 75) & (yy <= 104)
+    )
+    core = mallet_corridor | white_corridor
+    # Soft transition is outside the core, never across a foreground object.
+    outer = binary_dilation(core, iterations=2)
+    matte = gaussian_filter(outer.astype(np.float32), 0.8)
+    matte[core] = 1.0
+    # Do not replace even the outer antialiased silhouette of the blue object.
+    matte[:, 140:] = 0.0
+
+    for t in range(21, len(source)):
+        edited = source[t].astype(np.float32) * (1.0 - matte[..., None])
+        edited += plate * matte[..., None]
+        output[t] = np.clip(np.rint(edited), 0, 255).astype(np.uint8)
+
+    assert np.array_equal(output[:21], source[:21])
+    artifact_io.save(output, case_dir)
+    return output
+
+
+if __name__ == "__main__":
+    generate(Path(__file__).resolve().parent)
