+"""Friction intervention, built only from the supplied RGB timeline.
+The stationary block is unchanged until impact.  Its measured slide is
+reintegrated with five-fold Coulomb deceleration; source surface pixels are
+reprojected onto that shortened slide.  The ball then meets the stopped block.
+"""
+from pathlib import Path
+import numpy as np
+from PIL import Image, ImageDraw
+from scipy.ndimage import map_coordinates, gaussian_filter, label, binary_fill_holes
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+SRC = np.load(ROOT / 'source_pixels.npz')['rgb']
+H, W = SRC.shape[1:3]
+Y, X = np.indices((H, W), dtype=float)
+
+
+def polygon_alpha(vertices):
+    scale = 4
+    im = Image.new('L', (W * scale, H * scale), 0)
+    ImageDraw.Draw(im).polygon([(round(x*scale), round(y*scale)) for x,y in vertices], fill=255)
+    return np.asarray(im.resize((W,H), Image.Resampling.LANCZOS), dtype=float) / 255
+
+
+def shift(a, dx, dy):
+    coords = np.array([Y-dy, X-dx])
+    if a.ndim == 2:
+        return map_coordinates(a, coords, order=1, mode='constant', cval=0)
+    return np.stack([map_coordinates(a[..., c], coords, order=1, mode='constant', cval=0) for c in range(a.shape[2])], axis=-1)
+
+
+def ball_geometry():
+    out = []
+    for i, im in enumerate(SRC.astype(np.int16)):
+        m = (im[...,0]-im[...,1]>48) & (im[...,0]>70) & (im[...,1]-im[...,2]>8)
+        m &= (Y>39)&(Y<119)&(X<185)
+        lab, _ = label(m)
+        core = (im[...,0]-im[...,1]>68)&(im[...,0]>120)&(Y<115)&(Y>39)&(X<185)
+        counts = np.bincount(lab[core]); counts[0]=0
+        m = binary_fill_holes(lab==counts.argmax())
+        yy,xx = np.where(m)
+        left,right,top,bottom = xx.min(),xx.max(),yy.min(),yy.max()
+        cy=(top+bottom)/2
+        ry=(bottom-top+1)/2 + .35
+        rx=(right-left+1)/2 + .35
+        cx=(left+right)/2
+        # During the impulse, the right hemisphere is hidden behind the block.
+        if 12<=i<=14:
+            rx=ry
+            cx=left+rx-.6
+        out.append((cx,cy,rx,ry))
+    return np.array(out)
+
+BALL = ball_geometry()
+# Track the source block's exposed right edge before it comes to rest.
+EDGE = np.full(96, 243.5)
+for i in range(38):
+    lum=SRC[i].astype(float).mean(2)
+    scores=np.maximum(np.diff(lum,axis=1)[72:96,150:246],0).sum(0)
+    EDGE[i]=150+scores.argmax()+.5
+EDGE[:12]=164.5
+EDGE[12]=165.0
+EDGE[13]=170.5
+EDGE[14]=175.5
+EDGE[15]=180.5
+EDGE[16]=185.0
+# Source image outline (top, front, and narrow right face), measured in RGB.
+INITIAL=np.array([[109,60.5],[124,54.5],[164.5,60.5],[164,94.8],[149,103.5],[109,94.5]])
+OUTLINES={
+    11:INITIAL,
+    12:INITIAL+np.array([.5,.1]),
+    13:np.array([[115.7,61.5],[128.5,55.5],[170.5,61.5],[169.7,97.8],[160,104.0],[115.7,95.5]]),
+    14:np.array([[121.0,63.0],[132.0,56.7],[175.5,63.0],[175.0,100.2],[168.0,105.0],[121.0,96.4]]),
+    15:np.array([[125.7,64.5],[136.0,57.7],[180.5,64.5],[180.0,103.2],[175.0,106.5],[125.7,98.0]]),
+    16:np.array([[130.0,66.0],[141.0,58.7],[185.0,65.0],[184.3,104.9],[180.7,106.5],[130.0,100.5]])
+}
+
+
+def clean_background():
+    # Reject all silhouettes and their moving cast shadows before computing a
+    # temporal median. Every exposed table texel comes from the actual source.
+    valid = np.ones((96,H,W),bool)
+    for i,(cx,cy,rx,ry) in enumerate(BALL):
+        disc=((X-cx)/(rx+3))**2+((Y-cy)/(ry+3))**2<1
+        ballshadow=((X-cx-5)/(rx+8))**2+((Y-cy-ry-3)/9)**2<1
+        disp=max(0,EDGE[i]-164.5)
+        block=(X>EDGE[i]-66)&(X<EDGE[i]+19)&(Y>51+.13*disp)&(Y<116+.17*disp)
+        valid[i] &= ~(disc|ballshadow|block)
+    values=SRC.astype(float)
+    values[~valid]=np.nan
+    import warnings
+    with warnings.catch_warnings():
+        warnings.simplefilter('ignore', RuntimeWarning)
+        bg=np.nanmedian(values,axis=0)
+    missing=np.isnan(bg[...,0])
+    bg[missing]=SRC[0][missing]
+    # Restrict restoration to the intervention's footprint; elsewhere retain
+    # the exact table, backdrop, lighting, and framing from the first frame.
+    affected=(X<188)&(Y>39)&(Y<125)
+    return np.where(affected[...,None],bg,SRC[0]).astype(float)
+
+BG=clean_background()
+
+
+def get_ball_layer(i):
+    cx,cy,rx,ry=BALL[i]
+    rr=np.sqrt(((X-cx)/rx)**2+((Y-cy)/ry)**2)
+    alpha=np.clip((1.018-rr)*min(rx,ry),0,1)
+    rgb=SRC[i].astype(float)
+    # Recover the source's soft contact/cast shadow outside the sphere only.
+    support=np.exp(-(((X-cx-4)/(rx+6))**2+((Y-cy-ry-2)/7.5)**2)*2)
+    support *= (Y>cy+ry-2)&(X>cx-rx-7)&(X<cx+rx+11)
+    ratio=np.clip(rgb/np.maximum(BG,1),.35,1)
+    darkness=np.maximum(0,1-ratio)
+    shadow=darkness*support[...,None]*(1-alpha[...,None])
+    return rgb*alpha[...,None],alpha,shadow
+
+
+def get_block_layer(i):
+    outline=OUTLINES[i]
+    alpha=polygon_alpha(outline)
+    # Feather subpixel edges without extending any straight face into the table.
+    rgb=SRC[i].astype(float)
+    right=outline[:,0].max(); bottom=outline[:,1].max()
+    support=np.exp(-(((X-(right-12))/34)**2+((Y-(bottom-1))/8.5)**2)*1.1)
+    support*= (X>outline[:,0].min()-2)&(X<right+18)&(Y>outline[3:,1].min()-6)&(Y<bottom+14)
+    # A thin source-derived shadow along the right side of the standing block.
+    support=np.maximum(support,np.exp(-((X-right-3)/6)**2)*((Y>75)&(Y<bottom))*0.5)
+    ratio=np.clip(rgb/np.maximum(BG,1),.4,1)
+    shadow=np.maximum(0,1-ratio)*support[...,None]*(1-alpha[...,None])
+    return rgb*alpha[...,None],alpha,shadow
+
+BALL_LAYERS={i:get_ball_layer(i) for i in range(11,25)}
+BLOCK_LAYERS={i:get_block_layer(i) for i in range(11,17)}
+
+
+def interpolated_layer(q, kind):
+    lo=int(np.floor(q)); hi=lo+1; fraction=q-lo
+    layers=BALL_LAYERS if kind=='ball' else BLOCK_LAYERS
+    if fraction<1e-6:
+        return layers[lo]
+    if kind=='ball':
+        target=(1-fraction)*BALL[lo,:2]+fraction*BALL[hi,:2]
+        centers=[BALL[lo,:2],BALL[hi,:2]]
+    else:
+        target=(1-fraction)*OUTLINES[lo].mean(0)+fraction*OUTLINES[hi].mean(0)
+        centers=[OUTLINES[lo].mean(0),OUTLINES[hi].mean(0)]
+    result=[]
+    for component in range(3):
+        warped=[]
+        for i,center in zip([lo,hi],centers):
+            dx,dy=target-center
+            warped.append(shift(layers[i][component],dx,dy))
+        result.append(warped[0]*(1-fraction)+warped[1]*fraction)
+    return tuple(result)
+
+
+def main():
+    frames=SRC.copy()
+    # Fit gives approximately 0.23 pixels/frame^2 of source kinetic
+    # deceleration. Multiplying mu by 5 multiplies this term by 5.
+    mu_multiplier=5.0
+    source_deceleration=.23
+    acceleration=source_deceleration*mu_multiplier
+    initial_speed=6.0
+    impact=11.90
+    stopping_time=initial_speed/acceleration
+    records=[]
+    for t in range(12,96):
+        elapsed=min(max(0,t-impact),stopping_time)
+        displacement=initial_speed*elapsed-.5*acceleration*elapsed**2
+        right=164.5+displacement
+        block_q=float(np.interp(right,EDGE[11:17],np.arange(11,17)))
+        # Low residual ball velocity persists until its second contact with
+        # the now-stationary block; then friction/contact remove that velocity.
+        if t<=21: ball_q=float(t)
+        elif t<25:
+            u=t-21
+            ball_q=21+u-u*u/8
+        else: ball_q=23.0
+        bp,ba,bs=interpolated_layer(ball_q,'ball')
+        wp,wa,ws=interpolated_layer(block_q,'block')
+        image=BG*(1-bs)
+        image=image*(1-ba[...,None])+bp
+        image=image*(1-ws)
+        image=image*(1-wa[...,None])+wp
+        frames[t]=np.clip(np.rint(image),0,255).astype(np.uint8)
+        records.append([t,float(displacement),block_q,ball_q])
+    assert frames.shape==(96,144,256,3)
+    assert np.array_equal(frames[:12],SRC[:12])
+    artifact_io.save(frames, ROOT)
+    np.save(ROOT/'.tmp'/'trajectory.npy',np.array(records))
+    print('Saved complete 96-frame counterfactual. Block travel:',round(initial_speed**2/(2*acceleration),2),'pixels; final source pose:',round(records[-1][2],3))
+
+if __name__=='__main__':
+    main()

def polygon_alpha(vertices):
    scale = 4
    im = Image.new('L', (W * scale, H * scale), 0)
    ImageDraw.Draw(im).polygon([(round(x*scale), round(y*scale)) for x,y in vertices], fill=255)
    return np.asarray(im.resize((W,H), Image.Resampling.LANCZOS), dtype=float) / 255


def shift(a, dx, dy):
    coords = np.array([Y-dy, X-dx])
    if a.ndim == 2:
        return map_coordinates(a, coords, order=1, mode='constant', cval=0)
    return np.stack([map_coordinates(a[..., c], coords, order=1, mode='constant', cval=0) for c in range(a.shape[2])], axis=-1)


def ball_geometry():
    out = []
    for i, im in enumerate(SRC.astype(np.int16)):
        m = (im[...,0]-im[...,1]>48) & (im[...,0]>70) & (im[...,1]-im[...,2]>8)
        m &= (Y>39)&(Y<119)&(X<185)

    # temporal median. Every exposed table texel comes from the actual source.
    valid = np.ones((96,H,W),bool)
    for i,(cx,cy,rx,ry) in enumerate(BALL):

def get_ball_layer(i):
    cx,cy,rx,ry=BALL[i]
    rr=np.sqrt(((X-cx)/rx)**2+((Y-cy)/ry)**2)
    alpha=np.clip((1.018-rr)*min(rx,ry),0,1)

    support=np.exp(-(((X-cx-4)/(rx+6))**2+((Y-cy-ry-2)/7.5)**2)*2)
    support *= (Y>cy+ry-2)&(X>cx-rx-7)&(X<cx+rx+11)

    darkness=np.maximum(0,1-ratio)
    shadow=darkness*support[...,None]*(1-alpha[...,None])

def get_block_layer(i):

    rgb=SRC[i].astype(float)

    ratio=np.clip(rgb/np.maximum(BG,1),.4,1)
    shadow=np.maximum(0,1-ratio)*support[...,None]*(1-alpha[...,None])

def interpolated_layer(q, kind):
    lo=int(np.floor(q)); hi=lo+1; fraction=q-lo
    layers=BALL_LAYERS if kind=='ball' else BLOCK_LAYERS
    if fraction<1e-6:
        return layers[lo]
    if kind=='ball':
        target=(1-fraction)*BALL[lo,:2]+fraction*BALL[hi,:2]
        centers=[BALL[lo,:2],BALL[hi,:2]]
    else:
        target=(1-fraction)*OUTLINES[lo].mean(0)+fraction*OUTLINES[hi].mean(0)
        centers=[OUTLINES[lo].mean(0),OUTLINES[hi].mean(0)]
    result=[]
    for component in range(3):
        warped=[]
        for i,center in zip([lo,hi],centers):
            dx,dy=target-center
            warped.append(shift(layers[i][component],dx,dy))
        result.append(warped[0]*(1-fraction)+warped[1]*fraction)
    return tuple(result)
