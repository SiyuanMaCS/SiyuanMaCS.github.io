+
+Only source pixels are used for photographic appearance.  The ball's camera
+projection, velocity, spin, and floor shadow are reconstructed numerically;
+its sliding/rolling dissipation is scaled and its new cabinet impact resolved.

+import cv2
+import numpy as np
+from scipy.ndimage import binary_dilation, gaussian_filter
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+# Measurements of the source ball, in the requested 256 by 144 coordinates.
+CENTERS = np.array([
+ [82.728,120.661,10.279], [86.159,113.900,10.383],
+ [90.444,108.128,9.433], [93.980,102.760,8.948],
+ [96.206,98.236,9.323], [99.231,92.515,8.887],
+ [101.805,88.455,8.661], [104.012,85.004,8.672],
+ [106.308,81.586,8.318], [108.249,78.448,8.062],
+ [109.799,75.947,7.902], [111.201,73.575,7.678],
+ [112.405,71.756,7.528], [113.338,70.265,7.454],
+ [114.224,68.932,7.285], [114.878,67.847,7.206],
+ [115.367,67.145,7.122], [115.737,66.640,7.060],
+ [115.814,66.383,7.080]], dtype=float)
+# Rotation recovered by matching source RGB chromaticity on a sphere.
+PHASE = np.r_[0, np.cumsum([1.029,.973,.937,1.033,.774,.940,.955,.848,
+                            .741,.695,.568,.562,.506,.379,.302,.246,.159,.083])]
+AXIS = np.array([.93237469,.16729209,.32045404])
+AXIS /= np.linalg.norm(AXIS)
+
+
+def rot(theta):
+    x,y,z = AXIS
+    k = np.array([[0,-z,y],[z,0,-x],[-y,x,0]])
+    return np.eye(3)*np.cos(theta) + (1-np.cos(theta))*np.outer(AXIS,AXIS) + np.sin(theta)*k
+
+
+def lighting(n):
+    return np.maximum(.30, .62+.28*n[...,2]+.22*n[...,0]-.10*n[...,1])
+
+
+def background(source):
+    # Every piece of the floor hidden in frame 1 is revealed later.  Conversely,
+    # the eventual source resting spot is unobstructed in the first four frames.
+    clean = np.median(source[32:], axis=0).astype(np.float32)
+    clean[53:80,103:130] = np.median(source[:4,53:80,103:130],axis=0)
+    delta = np.max(np.abs(source.astype(np.float32)-clean[None]),axis=-1)
+    activity = np.max(delta,axis=0)>14
+    roi = np.zeros(activity.shape,bool)
+    roi[53:140,50:130] = True
+    activity &= roi
+    activity = binary_dilation(activity,iterations=2)
+    feather = gaussian_filter(activity.astype(float),.55)
+    # Keep all the unaffected objects, walls, floor, and source compression detail.
+    backgrounds = source.astype(np.float32)*(1-feather[None,:,:,None]) + clean[None]*feather[None,:,:,None]
+    return backgrounds, clean
+
+
+class PhotographicBall:
+    def __init__(self):
+        self.photos=[]
+        cap = cv2.VideoCapture(str(ROOT/'source_video.mp4'))
+        for i in range(19):
+            ok,im=cap.read()
+            if not ok: raise RuntimeError('Missing source video frame')
+            self.photos.append(cv2.cvtColor(im,cv2.COLOR_BGR2RGB).astype(np.float32))
+        cap.release()

+        # into the cabinet/floor shadow.  Separate these from the panel pattern.
+        self.level=np.array([1.,.96,.96,.99,.98,.94,.96,.96,.95,.88,
+                             .91,.93,.94,.93,.89,.86,.82,.82,.81])
+        self.reliability=np.array([.12,.12,.13,.25,.22,.24,.48,.65,.75,
+                                   .72,.88,.92,.96,1.,1.,1.,1.,1.,1.])
+        self.transforms=[rot(t) for t in PHASE]
+
+    def texture(self,normals,theta):
+        # Pull back the desired visible hemisphere to a local body coordinate
+        # system, then sample all the photographic source hemispheres.  Smooth
+        # confidence weights avoid seams and preserve the actual panel layout.
+        local=normals @ rot(theta)
+        color=np.zeros((len(normals),3),np.float32)
+        total=np.zeros(len(normals),np.float32)
+        for i,(cx,cy,r) in enumerate(CENTERS):
+            ns=local @ self.transforms[i].T
+            weight=np.maximum(0,ns[:,2]-.16)**6 * self.reliability[i]
+            if not np.any(weight>1e-6): continue
+            u=((cx+r*ns[:,0])*5).astype(np.float32)
+            v=((cy+r*ns[:,1])*5).astype(np.float32)
+            rgb=cv2.remap(self.photos[i],u[:,None],v[:,None],cv2.INTER_LINEAR,
+                          borderMode=cv2.BORDER_REFLECT)[:,0]
+            rgb=rgb/(lighting(ns)[:,None]*self.level[i])
+            color += rgb*weight[:,None]
+            total += weight
+        # The multiple revolutions in the source cover the sphere. Axis poles
+        # are always visible in at least one photograph; denominator is bounded.
+        return color/np.maximum(total[:,None],1e-7)
+
+    def draw(self,canvas,cx,cy,r,theta,level):
+        scale=4
+        x0=max(0,int(np.floor(cx-r-2)));x1=min(256,int(np.ceil(cx+r+2)))
+        y0=max(0,int(np.floor(cy-r-2)));y1=min(144,int(np.ceil(cy+r+2)))
+        yy,xx=np.mgrid[y0*scale:y1*scale,x0*scale:x1*scale]
+        nx=((xx+.5)/scale-cx)/r
+        ny=((yy+.5)/scale-cy)/r
+        rsq=nx*nx+ny*ny
+        mask=rsq<1
+        normal=np.c_[nx[mask],ny[mask],np.sqrt(np.maximum(0,1-rsq[mask]))]
+        tex=self.texture(normal,theta)
+        rgb=tex*lighting(normal)[:,None]*level
+        # Slight soft contact darkening, present in the source ball's lower edge.
+        rgb*= (1-.09*np.clip((normal[:,1]-.55)/.45,0,1))[:,None]
+        big=np.zeros((*nx.shape,3),np.float32)
+        big[mask]=np.clip(rgb,0,255)
+        alpha=cv2.resize(mask.astype(np.float32),(x1-x0,y1-y0),interpolation=cv2.INTER_AREA)
+        premul=cv2.resize(big,(x1-x0,y1-y0),interpolation=cv2.INTER_AREA)
+        patch=canvas[y0:y1,x0:x1]
+        canvas[y0:y1,x0:x1]=patch*(1-alpha[...,None])+premul
+
+
+def projection(s):
+    # Projective floor-depth coordinate.  Perspective shrinkage comes from the
+    # observed radius ratio, not from an arbitrary linear shrinking animation.
+    b=.48
+    den=1+b*s
+    x0,y0=82.55,120.55
+    xe,ye=115.80,66.40
+    cx=(x0+(xe*(1+b)-x0)*s)/den
+    cy=(y0+(ye*(1+b)-y0)*s)/den
+    radius=10.55/den
+    return cx,cy,radius
+
+
+def dynamics(times):
+    # Source depth velocity stays almost constant for the first .22 s, then
+    # decays to zero under contact friction.  Scale friction, not initial speed.
+    friction_ratio=.30
+    v0=1.98370733
+    original_brake=3.44336087
+    source_coast=.22
+    coast=source_coast/friction_ratio
+    brake=original_brake*friction_ratio
+    wall_depth=2.20
+    dcoast=v0*coast
+    disc=v0*v0-2*brake*(wall_depth-dcoast)
+    hit=coast+(v0-np.sqrt(disc))/brake
+    incoming=np.sqrt(disc)
+    restitution=.46
+    reverse_speed=restitution*incoming
+    settle=hit+reverse_speed/brake
+    state=[]
+    for t in times:
+        if t<coast:
+            s=v0*t;v=v0
+        elif t<hit:
+            d=t-coast;s=dcoast+v0*d-.5*brake*d*d;v=v0-brake*d
+        else:
+            d=min(t-hit,reverse_speed/brake)
+            s=wall_depth-reverse_speed*d+.5*brake*d*d
+            v=-max(0,reverse_speed-brake*d)
+        # Angular damping also gets the 0.30 friction scale.  The observed
+        # source angular history is integrated on that new physical clock.
+        if t<=hit:
+            angle=np.interp(t*friction_ratio,np.arange(19)/24,PHASE)/friction_ratio
+        else:
+            angle_hit=np.interp(hit*friction_ratio,np.arange(19)/24,PHASE)/friction_ratio
+            dt=t-hit
+            # The cabinet impact reverses the roll, with a short slip transient
+            # as tangential floor friction changes its angular velocity.
+            d=min(dt,reverse_speed/brake)
+            travel=reverse_speed*d-.5*brake*d*d
+            angle=angle_hit-11.8*travel + .32*(1-np.exp(-dt/.09))
+        state.append((s,v,angle))
+    return np.array(state),hit,settle
+
+
+def shadow(canvas,cx,cy,r):
+    yy,xx=np.mgrid[:144,:256]
+    # Source light is above and to the right: a leftward soft lobe and a denser
+    # ellipse at the floor contact, both projectively scaled with the ball.
+    q=((xx-(cx-.90*r))/(1.20*r))**2+((yy-(cy+.48*r))/(.52*r))**2
+    soft=.31*np.exp(-1.55*q)
+    q2=((xx-(cx-.18*r))/(.77*r))**2+((yy-(cy+.91*r))/(.18*r))**2
+    contact=.23*np.exp(-1.65*q2)
+    opacity=1-(1-soft)*(1-contact)
+    canvas*=1-opacity[...,None]
+
+
+def main():
+    source=np.load(ROOT/'source_pixels.npz')['rgb']
+    assert source.shape==(96,144,256,3)
+    canvases,clean=background(source)
+    ball=PhotographicBall()
+    times=np.arange(96)/24
+    motion,hit,settle=dynamics(times)
+    frames=[]
+    for i,(s,velocity,angle) in enumerate(motion):
+        if i==0:
+            # Friction changes at t=0; it does not change the initial position,
+            # texture, shadow, or velocity instantaneously.
+            frames.append(source[0].copy());continue
+        cx,cy,r=projection(s)
+        canvas=canvases[i].copy()
+        shadow(canvas,cx,cy,r)
+        # The ball encounters the rear cabinet base but misses the small balls
+        # to its right.  Those photographed objects therefore stay untouched.
+        level=np.interp(cy,[25,40,55,66.4,85,100,120.55],
+                           [.85,.84,.81,.81,.96,.99,1.0])
+        ball.draw(canvas,cx,cy,r,angle,level)
+        frames.append(np.rint(np.clip(canvas,0,255)).astype(np.uint8))
+    frames=np.stack(frames)
+    artifact_io.save(frames,ROOT)
+    print(f'Saved {len(frames)} RGB frames; cabinet contact {hit:.3f}s; rest {settle:.3f}s.')
+    print('Depth / projected center / radius samples:')
+    for i in [0,1,6,12,18,24,30,36,48,72,95]:
+        print(i, np.round(np.r_[motion[i,0],projection(motion[i,0])],3))
+
+if __name__=='__main__':
+    main()

        # system, then sample all the photographic source hemispheres.  Smooth
        # confidence weights avoid seams and preserve the actual panel layout.

                          borderMode=cv2.BORDER_REFLECT)[:,0]

            color += rgb*weight[:,None]

        # are always visible in at least one photograph; denominator is bounded.
        return color/np.maximum(total[:,None],1e-7)

    # observed radius ratio, not from an arbitrary linear shrinking animation.
    b=.48
    den=1+b*s

def dynamics(times):
    # Source depth velocity stays almost constant for the first .22 s, then
    # decays to zero under contact friction.  Scale friction, not initial speed.
    friction_ratio=.30
    v0=1.98370733
    original_brake=3.44336087
    source_coast=.22
    coast=source_coast/friction_ratio
    brake=original_brake*friction_ratio
    wall_depth=2.20
    dcoast=v0*coast
    disc=v0*v0-2*brake*(wall_depth-dcoast)
    hit=coast+(v0-np.sqrt(disc))/brake
    incoming=np.sqrt(disc)
    restitution=.46
    reverse_speed=restitution*incoming
    settle=hit+reverse_speed/brake
    state=[]
    for t in times:
        if t<coast:
            s=v0*t;v=v0
        elif t<hit:
            d=t-coast;s=dcoast+v0*d-.5*brake*d*d;v=v0-brake*d
        else:
            d=min(t-hit,reverse_speed/brake)
            s=wall_depth-reverse_speed*d+.5*brake*d*d
            v=-max(0,reverse_speed-brake*d)
        # Angular damping also gets the 0.30 friction scale.  The observed
        # source angular history is integrated on that new physical clock.
        if t<=hit:
            angle=np.interp(t*friction_ratio,np.arange(19)/24,PHASE)/friction_ratio
        else:
            angle_hit=np.interp(hit*friction_ratio,np.arange(19)/24,PHASE)/friction_ratio
            dt=t-hit
            # The cabinet impact reverses the roll, with a short slip transient
            # as tangential floor friction changes its angular velocity.
            d=min(dt,reverse_speed/brake)
            travel=reverse_speed*d-.5*brake*d*d
            angle=angle_hit-11.8*travel + .32*(1-np.exp(-dt/.09))
        state.append((s,v,angle))
    return np.array(state),hit,settle
