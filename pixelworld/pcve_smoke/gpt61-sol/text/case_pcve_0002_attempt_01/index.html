+
+The physical edit changes inertia, not the size or material of the blue mallet.
+All imagery, including the soft object shadows, is sampled from the source.

+def project(u):
+    return 167.43 - u / (1.0 - .0025 * u)

+def unproject(x):
+    d = 167.43 - x
+    return d / (1.0 + .0025 * d)

+def clean_plate(src, tracks):
+    # Every pixel is recovered from a time at which that part of the table
+    # is unobscured. Very small permanently obscured overlaps are completed
+    # by discrete harmonic RGB interpolation, not by a generative model.
+    f = src.astype(float)
+    excluded = np.zeros(f.shape[:3], dtype=bool)
+    yy, xx = np.indices(f.shape[1:3])
+    for t, a in enumerate(f):
+        for j, tr in enumerate(tracks):
+            x,y = tr[t]
+            roi = (abs(xx-x)<12) & (abs(yy-y)<11)
+            if j == 0:
+                fg = (a[...,2] > a[...,0]*1.22) & (a[...,2] > a[...,1]*1.065)
+            elif j == 1:
+                fg = (a[...,0] > a[...,1]*1.27) & (a[...,0] > a[...,2]*1.32)
+            else:
+                fg = (a[...,0]>217) & (a[...,1]>207) & (a[...,2]-a[...,0]<14)
+            excluded[t] |= binary_dilation(roi & fg, iterations=3)
+    data = np.where(excluded[...,None], np.nan, f)
+    with warnings.catch_warnings():
+        warnings.simplefilter('ignore', RuntimeWarning)
+        plate = np.nanmedian(data, axis=0)
+    missing = ~np.isfinite(plate[...,0])
+    print('Permanently covered plate pixels:', int(missing.sum()))
+    plate[missing] = (216,199,239)
+    for _ in range(450):
+        nb = (np.roll(plate,1,axis=0) + np.roll(plate,-1,axis=0)
+              +np.roll(plate,1,axis=1) + np.roll(plate,-1,axis=1))*.25
+        plate[missing] = nb[missing]
+    # Only the mallets' swept strip needs reconstruction. Leave the rest of
+    # the source untouched, including its original illumination/compression.
+    return plate

+    x,y = anchor
+    # Include the handle, antialiasing, and original contact shadow.
+    x0,x1 = int(np.floor(x))-12, int(np.floor(x))+14
+    y0,y1 = int(np.floor(y))-11, int(np.floor(y))+12
+    image = src[index,y0:y1,x0:x1].astype(float)
+    base = plate[y0:y1,x0:x1]
+    delta = image - base

+    if n:
+        sizes = np.bincount(ll.ravel());sizes[0]=0

+        # noise and remote stationary markings are not allowed into the sprite.
+        active = sizes[ll] >= 3
+    active = binary_dilation(active,iterations=1)
+    feather = np.clip(gaussian_filter(active.astype(float),.42),0,1)
+    delta *= feather[...,None]
+    delta[strength<5] = 0
+    return dict(delta=delta, origin=np.array([x-x0,y-y0]),
+                anchor=np.array(anchor), u=unproject(x))

+def simulate(tracks):
+    # Projected equal-size contact widths correspond to six units on the
+    # tabletop. The source's first collision is at sample 15 (time 14/24).
+    # Collision impulses use precisely a tenfold increase in the blue mass.
+    mass = np.array([10.,1.,1.])
+    inv = 1/mass
+    u = np.array([unproject(tracks[0][14,0]),
+                  unproject(tracks[1][14,0]), unproject(105.48)])
+    v = np.array([1.35,0.,0.])
+    limit = unproject(67.1)
+    restitution = .97
+    radius_gap = 6.0
+    path = np.empty((96,3))
+    vel = np.zeros((96,3))
+    for t in range(15):
+        path[t] = [unproject(tracks[0][t,0]),u[1],u[2]]
+    path[14] = u
+    contacts=[]
+    substeps=180
+    dt=1/substeps
+    for t in range(15,96):
+        for k in range(substeps):
+            # Small sliding/contact losses are mass independent until impact.
+            drag = .0028
+            v = np.sign(v)*np.maximum(abs(v)-drag*dt,0)
+            u += v*dt
+            for _ in range(4):
+                if u[2]>limit:
+                    u[2]=limit
+                    if v[2]>0:
+                        if v[2]>.12:
+                            contacts.append((round(t-1+k*dt,3),'rim',float(v[2])))
+                        v[2] *= -.075
+                for a,b in [(0,1),(1,2)]:
+                    gap=u[b]-u[a]
+                    if gap<radius_gap:
+                        overlap=radius_gap-gap
+                        den=inv[a]+inv[b]
+                        u[a]-=overlap*inv[a]/den
+                        u[b]+=overlap*inv[b]/den
+                        closing=v[a]-v[b]
+                        if closing>0:
+                            impulse=(1+restitution)*closing/den
+                            v[a]-=impulse*inv[a]
+                            v[b]+=impulse*inv[b]
+                            if closing>.1:
+                                contacts.append((round(t-1+k*dt,3),f'{a}-{b}',float(closing)))
+        path[t]=u
+        vel[t]=v
+    print('Contact events:',contacts)
+    print('Final screen positions:',project(path[-1]))
+    return path,vel,contacts
+

+def stamp(canvas, spr, anchor, scale=1., angle=0.):
+    d=spr['delta'];origin=spr['origin']
+    h,w=d.shape[:2]
+    # Inverse affine resampling of source RGB residuals keeps the original
+    # soft shadows and naturally integrates them with the tabletop beneath.
+    ext=int(max(w,h)*scale+5)
+    cx,cy=anchor
+    x0=max(0,int(cx)-ext);x1=min(256,int(cx)+ext+1)
+    y0=max(0,int(cy)-ext);y1=min(144,int(cy)+ext+1)
+    if x1<=x0 or y1<=y0:return
+    yy,xx=np.mgrid[y0:y1,x0:x1]
+    c,s=np.cos(angle),np.sin(angle)
+    rx=(xx-cx)/scale;ry=(yy-cy)/scale
+    sx=c*rx+s*ry+origin[0]
+    sy=-s*rx+c*ry+origin[1]
+    warped=np.stack([map_coordinates(d[...,ch],[sy,sx],order=1,
+                                    mode='constant',cval=0,prefilter=False)
+                     for ch in range(3)],axis=-1)
+    canvas[y0:y1,x0:x1]+=warped
+
+
+def main():
+    src=np.load(ROOT/'source_pixels.npz')['rgb']
+    assert src.shape==(96,144,256,3)
+    tracks=source_tracks(src)

+    # neighboring mallet at contact. Later templates of the red/white mallets
+    # supply the source's small collision-induced change of orientation.
+    templates=[sprite(src,plate,tracks[j][0],0) for j in range(3)]

+    red_late=sprite(src,plate,tracks[1][40],40)
+    path,vel,contacts=simulate(tracks)
+    result=src.copy()
+    # A fixed edit region encloses every original and counterfactual mallet
+    # position, and includes their cast contact shadows.
+    yy,xx=np.indices((144,256))
+    strip=(xx>=54)&(xx<=179)&(yy>=48)&(yy<=103)
+    for t in range(15,96):
+        canvas=src[t].astype(float)
+        canvas[strip]=plate[strip]
+        for j in range(3):
+            x=float(project(path[t,j]))
+            # Projection of the source's straight long-axis slide on the table.
+            initial=templates[j]['anchor']
+            y=initial[1] + .375*(initial[0]-x)
+            sp=templates[j]
+            if j==1 and path[t,j]>unproject(114):
+                sp=red_late
+                y=sp['anchor'][1]+.375*(sp['anchor'][0]-x)
+            if j==2 and path[t,j]>=unproject(72):
+                sp=white_wall
+                y=sp['anchor'][1]+.375*(sp['anchor'][0]-x)
+            # Objects grow only as they approach the camera, not due to mass.
+            scale=(1-.0025*sp['u'])/(1-.0025*path[t,j])
+            stamp(canvas,sp,(x,y),scale)

+    artifact_io.save(result,ROOT)
+    np.savez_compressed(ROOT/'motion_diagnostics.npz', world_position=path,
+                        world_velocity=vel, source_tracks=np.asarray(tracks))
+    print('Saved 96 RGB frames at 256x144, 24 fps.')
+
+if __name__=='__main__':
+    main()

\"\"\"Counterfactual rigid-body edit using source pixels and a numeric contact model.

The physical edit changes inertia, not the size or material of the blue mallet.
All imagery, including the soft object shadows, is sampled from the source.
\"\"\"
from pathlib import Path
import warnings
import numpy as np
from PIL import Image
from scipy.ndimage import binary_dilation, gaussian_filter, map_coordinates, label
import artifact_io

ROOT = Path(__file__).resolve().parent


def project(u):
    return 167.43 - u / (1.0 - .0025 * u)


def unproject(x):
    d = 167.43 - x
    return d / (1.0 + .0025 * d)


def source_tracks(src):
    f = src.astype(float)
    blue = []
    red = []
    for a in f:
        b = (a[..., 2] > a[..., 0] * 1.4) & (a[..., 2] > a[..., 1] * 1.12) & (a[..., 2] > 70)
        b[:, :134] = False
        b[:, 179:] = False
        b[:48] = False
        b[77:] = False
        yy, xx = np.where(b)
        blue.append([xx.mean(), yy.mean()])
        r = (a[..., 0] > a[..., 1] * 1.45) & (a[..., 0] > a[..., 2] * 1.6)
        r[:, :100] = False
        r[:, 155:] = False
        r[:56] = False
        r[88:] = False
        yy, xx = np.where(r)
        red.append([xx.mean(), yy.mean()])
    blue, red = np.asarray(blue), np.asarray(red)
    # Positions of the highlight-bearing part of the white mallet, as opposed
    # to the much larger stationary white border of the tabletop.
    white = np.zeros((96, 2))
    white[:] = (105.48, 80.69)
    measured = np.array([
        [105.12,80.76], [103.03,81.60], [101.17,82.24],
        [98.98,83.02], [97.46,83.71], [95.56,84.20],
        [93.24,85.10], [91.35,86.02], [89.04,86.93],
        [86.85,87.67], [84.92,88.51], [83.21,89.34],
        [81.10,90.20], [78.94,90.92], [76.47,91.88],
        [74.47,92.71], [72.41,93.52], [69.84,94.48],
        [67.89,95.24]])
    white[30:49] = measured
    for t in range(49,96):
        white[t] = (67.06 + 3.65 * (1 - np.exp(-(t-49)/23.0)),
                    95.27 - 1.15 * (1 - np.exp(-(t-49)/23.0)))
    return [blue, red, white]


def clean_plate(src, tracks):
    # Every pixel is recovered from a time at which that part of the table
    # is unobscured. Very small permanently obscured overlaps are completed
    # by discrete harmonic RGB interpolation, not by a generative model.
    f = src.astype(float)
    excluded = np.zeros(f.shape[:3], dtype=bool)
    yy, xx = np.indices(f.shape[1:3])
    for t, a in enumerate(f):
        for j, tr in enumerate(tracks):
            x,y = tr[t]
            roi = (abs(xx-x)<12) & (abs(yy-y)<11)
            if j == 0:
                fg = (a[...,2] > a[...,0]*1.22) & (a[...,2] > a[...,1]*1.065)
            elif j == 1:
                fg = (a[...,0] > a[...,1]*1.27) & (a[...,0] > a[...,2]*1.32)
            else:
                fg = (a[...,0]>217) & (a[...,1]>207) & (a[...,2]-a[...,0]<14)
            excluded[t] |= binary_dilation(roi & fg, iterations=3)
    data = np.where(excluded[...,None], np.nan, f)
    with warnings.catch_warnings():
        warnings.simplefilter('ignore', RuntimeWarning)
        plate = np.nanmedian(data, axis=0)
    missing = ~np.isfinite(plate[...,0])
    print('Permanently covered plate pixels:', int(missing.sum()))
    plate[missing] = (216,199,239)
    for _ in range(450):
        nb = (np.roll(plate,1,axis=0) + np.roll(plate,-1,axis=0)
              +np.roll(plate,1,axis=1) + np.roll(plate,-1,axis=1))*.25
        plate[missing] = nb[missing]
    # Only the mallets' swept strip needs reconstruction. Leave the rest of
    # the source untouched, including its original illumination/compression.
    return plate


def sprite(src, plate, anchor, index):
    x,y = anchor
    # Include the handle, antialiasing, and original contact shadow.
    x0,x1 = int(np.floor(x))-12, int(np.floor(x))+14
    y0,y1 = int(np.floor(y))-11, int(np.floor(y))+12
    image = src[index,y0:y1,x0:x1].astype(float)
    base = plate[y0:y1,x0:x1]
    delta = image - base
    strength = np.max(abs(delta),axis=2)
    active = strength > 9
    ll,n = label(active)
    if n:
        sizes = np.bincount(ll.ravel());sizes[0]=0
        # All meaningful components near the mallet are retained. Compression
        # noise and remote stationary markings are not allowed into the sprite.
        active = sizes[ll] >= 3
    active = binary_dilation(active,iterations=1)
    feather = np.clip(gaussian_filter(active.astype(float),.42),0,1)
    delta *= feather[...,None]
    delta[strength<5] = 0
    return dict(delta=delta, origin=np.array([x-x0,y-y0]),
                anchor=np.array(anchor), u=unproject(x))


def simulate(tracks):
    # Projected equal-size contact widths correspond to six units on the
    # tabletop. The source's first collision is at sample 15 (time 14/24).
    # Collision impulses use precisely a tenfold increase in the blue mass.
    mass = np.array([10.,1.,1.])
    inv = 1/mass
    u = np.array([unproject(tracks[0][14,0]),
                  unproject(tracks[1][14,0]), unproject(105.48)])
    v = np.array([1.35,0.,0.])
    limit = unproject(67.1)
    restitution = .97
    radius_gap = 6.0
    path = np.empty((96,3))
    vel = np.zeros((96,3))
    for t in range(15):
        path[t] = [unproject(tracks[0][t,0]),u[1],u[2]]
    path[14] = u
    contacts=[]
    substeps=180
    dt=1/substeps
    for t in range(15,96):
        for k in range(substeps):
            # Small sliding/contact losses are mass independent until impact.
            drag = .0028
            v = np.sign(v)*np.maximum(abs(v)-drag*dt,0)
            u += v*dt
            for _ in range(4):
                if u[2]>limit:
                    u[2]=limit
                    if v[2]>0:
                        if v[2]>.12:
                            contacts.append((round(t-1+k*dt,3),'rim',float(v[2])))
                        v[2] *= -.075
                for a,b in [(0,1),(1,2)]:
                    gap=u[b]-u[a]
                    if gap<radius_gap:
                        overlap=radius_gap-gap
                        den=inv[a]+inv[b]
                        u[a]-=overlap*inv[a]/den
                        u[b]+=overlap*inv[b]/den
                        closing=v[a]-v[b]
                        if closing>0:
                            impulse=(1+restitution)*closing/den
                            v[a]-=impulse*inv[a]
                            v[b]+=impulse*inv[b]
                            if closing>.1:
                                contacts.append((round(t-1+k*dt,3),f'{a}-{b}',float(closing)))
        path[t]=u
        vel[t]=v
    print('Contact events:',contacts)
    print('Final screen positions:',project(path[-1]))
    return path,vel,contacts


def stamp(canvas, spr, anchor, scale=1., angle=0.):
    d=spr['delta'];origin=spr['origin']
    h,w=d.shape[:2]
    # Inverse affine resampling of source RGB residuals keeps the original
    # soft shadows and naturally integrates them with the tabletop beneath.
    ext=int(max(w,h)*scale+5)
    cx,cy=anchor
    x0=max(0,int(cx)-ext);x1=min(256,int(cx)+ext+1)
    y0=max(0,int(cy)-ext);y1=min(144,int(cy)+ext+1)
    if x1<=x0 or y1<=y0:return
    yy,xx=np.mgrid[y0:y1,x0:x1]
    c,s=np.cos(angle),np.sin(angle)
    rx=(xx-cx)/scale;ry=(yy-cy)/scale
    sx=c*rx+s*ry+origin[0]
    sy=-s*rx+c*ry+origin[1]
    warped=np.stack([map_coordinates(d[...,ch],[sy,sx],order=1,
                                    mode='constant',cval=0,prefilter=False)
                     for ch in range(3)],axis=-1)
    canvas[y0:y1,x0:x1]+=warped


def main():
    src=np.load(ROOT/'source_pixels.npz')['rgb']
    assert src.shape==(96,144,256,3)
    tracks=source_tracks(src)
    plate=clean_plate(src,tracks)
    # The clean initial templates contain only their own mallet, never a
    # neighboring mallet at contact. Later templates of the red/white mallets
    # supply the source's small collision-induced change of orientation.
    templates=[sprite(src,plate,tracks[j][0],0) for j in range(3)]
    white_wall=sprite(src,plate,tracks[2][48],48)
    red_late=sprite(src,plate,tracks[1][40],40)
    path,vel,contacts=simulate(tracks)
    result=src.copy()
    # A fixed edit region encloses every original and counterfactual mallet
    # position, and includes their cast contact shadows.
    yy,xx=np.indices((144,256))
    strip=(xx>=54)&(xx<=179)&(yy>=48)&(yy<=103)
    for t in range(15,96):
        canvas=src[t].astype(float)
        canvas[strip]=plate[strip]
        for j in range(3):
            x=float(project(path[t,j]))
            # Projection of the source's straight long-axis slide on the table.
            initial=templates[j]['anchor']
            y=initial[1] + .375*(initial[0]-x)
            sp=templates[j]
            if j==1 and path[t,j]>unproject(114):
                sp=red_late
                y=sp['anchor'][1]+.375*(sp['anchor'][0]-x)
            if j==2 and path[t,j]>=unproject(72):
                sp=white_wall
                y=sp['anchor'][1]+.375*(sp['anchor'][0]-x)
            # Objects grow only as they approach the camera, not due to mass.
            scale=(1-.0025*sp['u'])/(1-.0025*path[t,j])
            stamp(canvas,sp,(x,y),scale)
        result[t]=np.clip(np.rint(canvas),0,255).astype(np.uint8)
    artifact_io.save(result,ROOT)
    np.savez_compressed(ROOT/'motion_diagnostics.npz', world_position=path,
                        world_velocity=vel, source_tracks=np.asarray(tracks))
    print('Saved 96 RGB frames at 256x144, 24 fps.')

if __name__=='__main__':
    main()
