+"""Counterfactual toy-car slowdown, using only measured source RGB pixels.
+
+The pre-contact path is a uniformly decelerating roll.  Reducing only the
+initial velocity (not friction) reduces the stopping distance to 0.6**2 of
+the free-roll distance.  The car therefore does not reach the yellow toy.
+Source car samples, including wheels and the contact shadow, are registered
+and composited along that new trajectory; no collision/fall is retained.
+"""
+from pathlib import Path
+
+import numpy as np
+from scipy.ndimage import (
+    binary_dilation,
+    binary_fill_holes,
+    gaussian_filter,
+    label,
+    shift,
+)
+
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+
+
+def car_center(frame):
+    r, g, b = frame.astype(float).transpose(2, 0, 1)
+    red = (r > 90) & (r > 1.45 * g) & (r > 1.4 * b)
+    red[:26] = False
+    red[36:] = False
+    red[:, :143] = False
+    components, count = label(red)
+    car = max(range(1, count + 1), key=lambda j: np.count_nonzero(components == j))
+    return np.where(components == car)[1].mean()
+
+
+def smoothstep(x):
+    x = np.clip(x, 0.0, 1.0)
+    return x * x * (3.0 - 2.0 * x)
+
+
+def generate():
+    source = np.load(ROOT / 'source_pixels.npz')['rgb']
+    rgb = source.astype(np.float64)
+    n, h, w, _ = source.shape
+    centers = np.array([car_center(source[t]) for t in range(21)])
+    time = np.arange(len(centers), dtype=float)
+    # Anchor the trajectory exactly at the intervention-frame position.
+    design = np.column_stack((-time, 0.5 * time ** 2))
+    speed, deceleration = np.linalg.lstsq(
+        design, centers - centers[0], rcond=None
+    )[0]
+    new_speed = 0.6 * speed
+    stop_frame = new_speed / deceleration
+    roll_time = np.minimum(np.arange(n, dtype=float), stop_frame)
+    positions = centers[0] - new_speed * roll_time + 0.5 * deceleration * roll_time ** 2
+
+    # The source later exposes the whole original-car footprint.  Reconstruct
+    # just this small, static patch, leaving all other first-frame pixels alone.
+    clean = np.median(rgb[30:36], axis=0)
+    yy, xx = np.mgrid[:h, :w]
+    reveal = (
+        smoothstep((xx - 173) / 4)
+        * smoothstep((242 - xx) / 4)
+        * smoothstep((yy - 21) / 4)
+        * smoothstep((40 - yy) / 3)
+    )
+    background = rgb[0] * (1 - reveal[..., None]) + clean * reveal[..., None]
+
+    # These samples cover the entire revised path before any source contact.
+    # Each residual includes the naturally antialiased car and its shadow.
+    residuals = []
+    for t in range(9):
+        delta = rgb[t] - background
+        distance = np.max(np.abs(delta), axis=2)
+        window = (
+            (yy >= 24) & (yy <= 38)
+            & (xx > centers[t] - 23)
+            & (xx < centers[t] + 31)
+        )
+        components, count = label((distance > 10) & window)
+        largest = max(range(1, count + 1), key=lambda j: np.count_nonzero(components == j))
+        support = binary_fill_holes(components == largest)
+        support = binary_dilation(support, iterations=2)
+        soft = gaussian_filter(support.astype(float), sigma=0.65)
+        # Full-strength object pixels, with feathering confined to the halo.
+        matte = np.minimum(1.0, soft * 1.8)
+        residuals.append(delta * matte[..., None])
+
+    frames = np.empty_like(source)
+    for t, target in enumerate(positions):
+        # Interpolate source poses only after registering their positions.
+        # This avoids the double-car silhouettes of temporal cross-fades.
+        source_time = np.interp(target, centers[:9][::-1], np.arange(9)[::-1])
+        a = int(np.floor(source_time))
+        b = min(a + 1, 8)
+        fraction = source_time - a
+        first = shift(residuals[a], (0, target - centers[a], 0), order=1,
+                      mode='constant', cval=0, prefilter=False)
+        if b != a and fraction > 1e-8:
+            second = shift(residuals[b], (0, target - centers[b], 0), order=1,
+                           mode='constant', cval=0, prefilter=False)
+            object_pixels = first * (1 - fraction) + second * fraction
+        else:
+            object_pixels = first
+        frames[t] = np.clip(np.rint(background + object_pixels), 0, 255).astype(np.uint8)
+    # A speed intervention changes velocity, not the instantaneous image.
+    frames[0] = source[0]
+    artifact_io.save(frames, ROOT)
+    print(f'Original initial speed: {speed:.4f} pixels/frame')
+    print(f'Revised initial speed: {new_speed:.4f} pixels/frame')
+    print(f'Deceleration: {deceleration:.5f} pixels/frame^2')
+    print(f'Stop: frame {stop_frame:.2f}; center x={positions[-1]:.3f}')
+    print(f'Saved {frames.shape[0]} RGB frames at 256x144, 24 fps.')
+
+
+if __name__ == '__main__':
+    generate()
