+"""Counterfactual mass edit using source RGB, geometry and impulse equations."""
+from pathlib import Path
+import numpy as np
+from scipy import ndimage as ndi
+from PIL import Image, ImageDraw
+import artifact_io
+
+ROOT=Path(__file__).resolve().parent
+Y,X=np.mgrid[:144,:256]
+
+def ball_fits(A):
+    fits=[]
+    for i,f in enumerate(A):
+        r,g,b=f.transpose(2,0,1)
+        m=(r-g>50)&(r>100)&(g-b>6)&(Y<115)
+        lab,n=ndi.label(m)
+        ex=15+7.1*i if i<12 else 100+.68*(i-12)
+        ey=56+.196*(ex-15) if i<12 else 73+.345*(ex-100)
+        k=lab[int(ey),int(ex)]
+        m=lab==k
+        yy,xx=np.where(m & ~ndi.binary_erosion(m))
+        q=np.c_[xx*xx,yy*yy,xx,yy]
+        a,b,c,d=np.linalg.lstsq(q,np.ones(len(xx)),rcond=None)[0]
+        cx,cy=-c/(2*a),-d/(2*b)
+        v=1+a*cx*cx+b*cy*cy
+        fits.append([cx,cy,np.sqrt(v/a)+.65,np.sqrt(v/b)+.65])
+    return np.array(fits)
+
+# Screen displacement of the block, extracted from its dark right-hand face.
+CUBE_DX=np.array([0]*13+[5,10,15,20,25,30,35,40,44.5,49,53,56.5,60,63,66,68.5,71,73,74.5,76,77,78,78.5,79]+[79]*59,dtype=float)[:96]
+
+def background(A,fits):
+    valid=np.ones((96,144,256),bool)
+    for i,(cx,cy,rx,ry) in enumerate(fits):
+        # Fully exclude sphere, soft contact shadow, and floor reflection.
+        sphere=((X-cx)/(rx+2.5))**2+((Y-cy)/(ry+2.5))**2<1
+        shadow=((X-cx-5)/(rx*1.55))**2+((Y-cy-ry-2)/(ry*.55+2))**2<1
+        reflection=((X-cx-9)/(rx*1.45))**2+((Y-cy-ry*1.9)/(ry*1.08))**2<1
+        dx=CUBE_DX[i]
+        cube=(X>103+dx)&(X<181+dx)&(Y>51+.18*dx)&(Y<135+.18*dx)
+        valid[i]=~(sphere|shadow|reflection|cube)
+    vals=np.where(valid[...,None],A,np.nan)
+    import warnings
+    with warnings.catch_warnings():
+        warnings.simplefilter('ignore',RuntimeWarning)
+        B=np.nanmedian(vals,axis=0)
+    # Rare holes are filled from the closest valid floor observations.
+    miss=np.isnan(B[:,:,0])
+    if np.any(miss):
+        idx=ndi.distance_transform_edt(miss,return_distances=False,return_indices=True)
+        B[miss]=B[tuple(idx[:,miss])]
+    # Most of the room stays pixel-identical to the initial photograph.
+    erase=ndi.gaussian_filter((~valid[0]).astype(float),1)
+    B=B*erase[...,None]+A[0]*(1-erase[...,None])
+    return B,valid
+
+def sample(array,sx,sy):
+    if array.ndim==2:
+        return ndi.map_coordinates(array,[sy,sx],order=1,mode='constant',cval=0)
+    return np.stack([ndi.map_coordinates(array[:,:,j],[sy,sx],order=1,mode='constant',cval=0) for j in range(array.shape[2])],axis=-1)
+
+
+def build():
+    A=np.load(ROOT/'source_pixels.npz')['rgb'].astype(float)
+    fits=ball_fits(A)
+    B,valid=background(A,fits)
+    return A,fits,B,valid
+
+def cube_layer(A,B,j):
+    """A source-derived residual includes wood, contact shading and reflection."""
+    dx=CUBE_DX[j]
+    x0=105+dx
+    x1=192+dx
+    y0=51+.18*dx
+    # The entire lower patch is needed for the varnished floor's faint reflection.
+    mask=((X-x0)/2).clip(0,1)*((x1-X)/3).clip(0,1)*((Y-y0)/2).clip(0,1)
+    residual=(A[j]-B)*mask[...,None]
+    # Canonicalize translation but keep the original shape, grain and lighting.
+    return sample(residual,X+dx,Y+.18*dx)
+
+
+def ball_layer(A,B,fits,j):
+    cx,cy,rx,ry=fits[j]
+    rho=np.sqrt(((X-cx)/rx)**2+((Y-cy)/ry)**2)
+    alpha=((1-rho)*min(rx,ry)+.65).clip(0,1)
+    # Surface pixels, including their original texture and shading.
+    rgba=np.dstack([A[j]*alpha[...,None],alpha])
+    # Separate floor interactions; this avoids copying floor grain with the ball.
+    floor=(Y>cy+ry*.65)&(np.abs(X-cx)<rx*1.8)&(Y<cy+ry*3.4)
+    envelope=np.exp(-((X-cx-5)/(rx*1.45))**6)
+    envelope*=((cy+ry*3.4-Y)/(ry*.45)).clip(0,1)
+    floor_weight=floor*envelope*(1-alpha)
+    floor_delta=(A[j]-B)*floor_weight[...,None]
+    return rgba,floor_delta
+
+
+def generate():
+    A,fits,B,valid=build()
+    cube_templates={j:cube_layer(A,B,j) for j in [0,16,20,24]}
+    # The later source spheres are unobscured and provide real rolling texture.
+    sphere_templates={j:ball_layer(A,B,fits,j) for j in range(18,96)}
+
+    # Estimate impact coefficients from incoming and outgoing screen velocities.
+    # Momentum conservation gives M_block/M_ball approximately 1.23; source
+    # separation speed gives restitution approximately .64. Only the ball's
+    # mass is changed: shape, incoming velocity, gravity and friction stay fixed.
+    incoming=7.12
+    block_mass=1.23
+    ball_mass=10.0
+    restitution=.64
+    ball_speed=incoming*(ball_mass-restitution*block_mass)/(ball_mass+block_mass)
+    block_speed=incoming*(1+restitution)*ball_mass/(ball_mass+block_mass)
+    collision_frame=12.0
+    # Block floor drag is fitted from the source's decelerating slide.
+    block_drag=.30
+    cx0=99.7
+    cy0=72.72
+    out=[]
+    trajectory=[]
+    for i in range(96):
+        if i<=12:
+            # Mass does not change a body's free motion. First contact still
+            # occurs at the same position and time, with no cosmetic mass cue.
+            out.append(A[i].astype(np.uint8))
+            trajectory.append([i,float(fits[i,0]),float(fits[i,1]),0.0])
+            continue
+        tau=i-collision_frame
+        # A short, resolved contact impulse, not an instantaneous teleport.
+        ramp=.20*(1-np.exp(-tau/.6))
+        s=max(0,tau-ramp)
+        dx_ball=ball_speed*s
+        bx=cx0+dx_ball
+        by=cy0+.215*dx_ball
+        brx=17.85+.0275*dx_ball
+        bry=17.25+.0320*dx_ball
+        slide=min(s,block_speed/block_drag)
+        dx_cube=block_speed*slide-.5*block_drag*slide*slide
+        trajectory.append([i,bx,by,dx_cube])
+        f=B.copy()
+
+        # Increased impulse also advances the block's original yaw response.
+        # Aligned real source patches preserve all three-dimensional texture.
+        rotation_time=12+1.60*tau
+        keys=[0,16,20,24]
+        if rotation_time<=16:
+            ja,jb=0,16
+            mix=np.clip((rotation_time-12)/4,0,1)
+        elif rotation_time<=20:
+            ja,jb=16,20
+            mix=(rotation_time-16)/4
+        else:
+            ja,jb=20,24
+            mix=np.clip((rotation_time-20)/4,0,1)
+        residual=cube_templates[ja]*(1-mix)+cube_templates[jb]*mix
+        # Projection gradually enlarges objects moving into the foreground.
+        scale=1+.00065*max(0,dx_cube-40)
+        px=137.0+dx_cube
+        py=86.0+.18*dx_cube
+        sx=(X-px)/scale+137
+        sy=(Y-py)/scale+86
+        if px-65*scale<256:
+            f+=sample(residual,sx,sy)
+
+        # Render the rolling heavy sphere. Its larger retained momentum, rather
+        # than any change in diameter or material, is the mass intervention.
+        if bx-brx-4<256:
+            # Use a source rolling phase proportional to traversed distance.
+            phase=18+dx_ball/.68
+            j=18+int(round(phase-18))%78
+            cx,cy,rx,ry=fits[j]
+            sx=(X-bx)/brx*rx+cx
+            sy=(Y-by)/bry*ry+cy
+            rgba,shade=sphere_templates[j]
+            floor_patch=sample(shade,sx,sy)
+            # Source shade fades naturally at the extraction patch perimeter.
+            f+=floor_patch
+            sprite=sample(rgba,sx,sy)
+            alpha=sprite[:,:,3].clip(0,1)
+            f=f*(1-alpha[...,None])+sprite[:,:,:3]
+        out.append(np.clip(np.rint(f),0,255).astype(np.uint8))
+    frames=np.stack(out)
+    artifact_io.save(frames,ROOT)
+    # Small numeric audit makes the model and its timeline reproducible.
+    import json
+    (ROOT/'motion_audit.json').write_text(json.dumps({
+        'mass_multiplier':10,'impact_frame_1based':13,
+        'incoming_ball_px_per_frame':incoming,
+        'outgoing_ball_px_per_frame':ball_speed,
+        'outgoing_block_px_per_frame':block_speed,
+        'trajectory_frame_ball_x_ball_y_block_dx':trajectory
+    },indent=2))
+    return frames
+
+
+if __name__=='__main__':
+    frames=generate()
+    print(f'Saved {len(frames)} RGB frames, {frames.shape[2]}x{frames.shape[1]}, 24 fps.')

from pathlib import Path
import numpy as np
from scipy import ndimage as ndi
from PIL import Image, ImageDraw
import artifact_io

ROOT=Path(__file__).resolve().parent

def ball_fits(A):
    fits=[]
    for i,f in enumerate(A):
        r,g,b=f.transpose(2,0,1)
        m=(r-g>50)&(r>100)&(g-b>6)&(Y<115)
        lab,n=ndi.label(m)
        ex=15+7.1*i if i<12 else 100+.68*(i-12)
        ey=56+.196*(ex-15) if i<12 else 73+.345*(ex-100)
        k=lab[int(ey),int(ex)]
        m=lab==k
        yy,xx=np.where(m & ~ndi.binary_erosion(m))
        q=np.c_[xx*xx,yy*yy,xx,yy]
        a,b,c,d=np.linalg.lstsq(q,np.ones(len(xx)),rcond=None)[0]
        cx,cy=-c/(2*a),-d/(2*b)
        v=1+a*cx*cx+b*cy*cy
        fits.append([cx,cy,np.sqrt(v/a)+.65,np.sqrt(v/b)+.65])
    return np.array(fits)

# Screen displacement of the block, extracted from its dark right-hand face.

def background(A,fits):
    valid=np.ones((96,144,256),bool)
    for i,(cx,cy,rx,ry) in enumerate(fits):
        # Fully exclude sphere, soft contact shadow, and floor reflection.
        sphere=((X-cx)/(rx+2.5))**2+((Y-cy)/(ry+2.5))**2<1
        shadow=((X-cx-5)/(rx*1.55))**2+((Y-cy-ry-2)/(ry*.55+2))**2<1
        reflection=((X-cx-9)/(rx*1.45))**2+((Y-cy-ry*1.9)/(ry*1.08))**2<1
        dx=CUBE_DX[i]
        cube=(X>103+dx)&(X<181+dx)&(Y>51+.18*dx)&(Y<135+.18*dx)
        valid[i]=~(sphere|shadow|reflection|cube)
    vals=np.where(valid[...,None],A,np.nan)
    import warnings
    with warnings.catch_warnings():
        warnings.simplefilter('ignore',RuntimeWarning)
        B=np.nanmedian(vals,axis=0)
    # Rare holes are filled from the closest valid floor observations.
    miss=np.isnan(B[:,:,0])
    if np.any(miss):
        idx=ndi.distance_transform_edt(miss,return_distances=False,return_indices=True)
        B[miss]=B[tuple(idx[:,miss])]
    # Most of the room stays pixel-identical to the initial photograph.
    erase=ndi.gaussian_filter((~valid[0]).astype(float),1)
    B=B*erase[...,None]+A[0]*(1-erase[...,None])
    return B,valid

def sample(array,sx,sy):
    if array.ndim==2:
        return ndi.map_coordinates(array,[sy,sx],order=1,mode='constant',cval=0)
    return np.stack([ndi.map_coordinates(array[:,:,j],[sy,sx],order=1,mode='constant',cval=0) for j in range(array.shape[2])],axis=-1)

    fits=ball_fits(A)
    B,valid=background(A,fits)
    return A,fits,B,valid
