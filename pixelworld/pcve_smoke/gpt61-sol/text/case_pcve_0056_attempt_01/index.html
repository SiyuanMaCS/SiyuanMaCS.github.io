+"""Source-pixel curling edit, with an early third-stone intervention.
+
+No assets other than the supplied source are used.  The rink is recovered
+from unoccupied source pixels; the stones are matted from the original
+video.  A one-dimensional equal-mass contact calculation follows the line
+specified in the prompt, using restitution measured from the source impact.
+"""
+from pathlib import Path
+import numpy as np
+import cv2
+from scipy import ndimage
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+S = 5
+
+
+def source_data():
+    frames = np.load(ROOT / 'source_pixels.npz')['rgb']
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    first = clean = None
+    for i in range(13):
+        ok, f = cap.read()
+        if not ok:
+            raise RuntimeError('Source video could not be decoded')
+        if i == 0:
+            first = f[..., ::-1].copy()
+        if i == 12:
+            clean = f[..., ::-1].copy()
+    cap.release()
+    return frames, first, clean
+
+
+def stone_sprite(first, clean, anchor, bounds, blue=False):
+    x0, y0, x1, y1 = bounds
+    donor = first[y0*S:y1*S, x0*S:x1*S].astype(np.float32)
+    bg = clean[y0*S:y1*S, x0*S:x1*S].astype(np.float32)
+    delta = donor - bg
+    d = np.max(np.abs(delta), axis=2)
+    yy, xx = np.mgrid[y0*S:y1*S, x0*S:x1*S].astype(float) / S
+    ax, ay = anchor
+    # The body and its immediate contact shadow, not source codec noise.
+    region = ((xx-ax)/6.1)**2 + ((yy-ay-2.1)/4.7)**2 < 1
+    region |= ((xx-ax-2.8)/7.0)**2 + ((yy-ay-4.4)/2.0)**2 < 1
+    strength = np.clip((d-5)/45, 0, 1) * region
+    solid = (d > 30) & region
+    labels, _ = ndimage.label(solid)
+    sizes = np.bincount(labels.ravel())
+    sizes[0] = 0
+    body = labels == np.argmax(sizes)
+    body = ndimage.binary_fill_holes(body)
+    body = ndimage.gaussian_filter(body.astype(float), 0.7)
+    alpha = np.maximum(strength, body * .98)
+    alpha[d < 6] = 0
+    # Premultiplied foreground, preserving donor shading and edge coverage.
+    premul = delta + bg * alpha[..., None]
+    premul[alpha == 0] = 0
+    if blue:
+        un = premul / np.maximum(alpha[..., None], 1e-5)
+        r, g, b = un.transpose(2, 0, 1)
+        handle = (r-g > 30) & (r-b > 30) & (yy < ay+1.7) & (alpha > .02)
+        recolored = un.copy()
+        recolored[..., 0] = .90*g+8
+        recolored[..., 1] = .70*g+.30*r
+        recolored[..., 2] = r
+        # Recolor only the handle, never the neutral granite or its shadow.
+        premul[handle] = recolored[handle] * alpha[handle, None]
+    premul = np.clip(premul, 0, 255*alpha[..., None])
+    return {
+        'rgba': np.concatenate([premul, alpha[..., None]], axis=2).astype(np.float32),
+        'anchor': np.array([(ax-x0)*S, (ay-y0)*S]),
+    }
+
+
+def get_tracks(frames):
+    xy = np.zeros((len(frames), 2, 2), float)
+    for t, f in enumerate(frames):
+        r, g, b = f.astype(float).transpose(2, 0, 1)
+        for j, mask in enumerate(((r-g > 110) & (r-b > 105),
+                                  (g-b > 70) & (r-b > 70))):
+            yy, xx = np.where(mask)
+            weights = (r-g)[mask] if j == 0 else (g-b)[mask]
+            xy[t, j] = [np.average(xx, weights=weights),
+                        np.average(yy, weights=weights)]
+    return xy
+
+
+def simulate(tracks):
+    # Track ordering: red, inserted blue, yellow. Coordinates are projected
+    # distances along the common line (the transverse offset is zero).
+    red0, yellow0 = tracks[0, 0, 0], tracks[0, 1, 0]
+    blue0 = .25*red0 + .75*yellow0
+    x = np.array([red0, blue0, yellow0], float)
+    radii = np.array([4.60, 4.83, 5.00])
+    v = np.array([1.98, 0, -2.27])
+    changed = np.array([False, True, False])
+    traj = [x.copy()]
+    contacts = []
+    dt = 1/80
+    # The source's two-stone relative speed falls to about a tenth of its
+    # incoming value. Low restitution gives the same granite-on-ice response.
+    restitution = .105
+    drag = .0025  # pixels / source-frame squared, after an altered contact
+    for frame in range(1, 96):
+        for step in range(80):
+            time = frame-1 + (step+1)*dt
+            old = x.copy()
+            for j, src_j in ((0, 0), (2, 1)):
+                if not changed[j]:
+                    k = int(np.floor(time))
+                    a = time-k
+                    x[j] = (1-a)*tracks[k,src_j,0] + a*tracks[min(k+1,95),src_j,0]
+                    v[j] = (x[j]-old[j])/dt
+            for j in range(3):
+                if changed[j]:
+                    v[j] = np.sign(v[j])*max(0., abs(v[j])-drag*dt)
+                    x[j] += v[j]*dt
+            # Sequential impulses with overlap correction, iterated for the
+            # brief three-body contact instead of allowing interpenetration.
+            for _ in range(8):
+                for j in range(2):
+                    gap = x[j+1]-x[j]
+                    required = radii[j]+radii[j+1]
+                    if gap < required:
+                        overlap = required-gap
+                        x[j] -= overlap/2
+                        x[j+1] += overlap/2
+                        approach = v[j]-v[j+1]
+                        if approach > 1e-7:
+                            impulse = (1+restitution)*approach/2
+                            v[j] -= impulse
+                            v[j+1] += impulse
+                            if len(contacts) == 0 or time-contacts[-1][0] > .25:
+                                contacts.append((round(time,3), j, j+1))
+                            changed[j:j+2] = True
+        traj.append(x.copy())
+    return np.array(traj), contacts
+
+
+def paste(canvas, sprite, xy, scale=1.):
+    rgba = sprite['rgba']
+    anchor = sprite['anchor']
+    # Warp the donor at source resolution before area-resampling. This keeps
+    # subpixel motion smooth at the requested small output resolution.
+    h, w = rgba.shape[:2]
+    loc = np.asarray(xy)*S
+    top_left = loc-scale*anchor
+    left, top = np.floor(top_left-2).astype(int)
+    right, bottom = np.ceil(top_left+np.array([w,h])*scale+2).astype(int)
+    left, top = max(0,left), max(0,top)
+    right, bottom = min(canvas.shape[1],right), min(canvas.shape[0],bottom)
+    if right <= left or bottom <= top:
+        return
+    transform = np.array([[scale,0,top_left[0]-left],
+                          [0,scale,top_left[1]-top]], np.float32)
+    warped = cv2.warpAffine(rgba, transform, (right-left,bottom-top),
+                            flags=cv2.INTER_LINEAR,
+                            borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+    a = warped[...,3:4]
+    tile = canvas[top:bottom,left:right]
+    tile[:] = tile*(1-a)+warped[...,:3]
+
+
+def main():
+    src, first, clean = source_data()
+    tracks = get_tracks(src)
+    red_anchor = tracks[0,0]
+    yellow_anchor = tracks[0,1]
+    red = stone_sprite(first, clean, red_anchor, (39,60,60,77))
+    yellow = stone_sprite(first, clean, yellow_anchor, (204,70,226,88))
+    blue = stone_sprite(first, clean, red_anchor, (39,60,60,77), blue=True)
+    background = src[0].copy()
+    # These source patches are empty throughout most of the video. Median
+    # selection removes the original moving bodies and their contact shadows.
+    med = np.median(src,axis=0).astype(np.uint8)
+    background[61:76,39:60] = med[61:76,39:60]
+    background[71:88,204:226] = med[71:88,204:226]
+    path, contacts = simulate(tracks)
+    slope = (yellow_anchor[1]-red_anchor[1]) / (yellow_anchor[0]-red_anchor[0])
+    output = []
+    for t in range(96):
+        # Nearest replication + area reduction is identity on the unaffected
+        # supplied RGB background, while stones retain full-resolution edges.
+        canvas = np.repeat(np.repeat(background,S,axis=0),S,axis=1).astype(np.float32)
+        anchors = [[x, red_anchor[1]+slope*(x-red_anchor[0])] for x in path[t]]
+        # Far-to-near compositing. All bodies stay tangent rather than overlap.
+        for j in np.argsort(np.array(anchors)[:,1]):
+            sprite = [red,blue,yellow][j]
+            scale = 1.065 if j == 1 else 1.
+            paste(canvas,sprite,anchors[j],scale)
+        frame = cv2.resize(canvas,(256,144),interpolation=cv2.INTER_AREA)
+        output.append(np.clip(np.rint(frame),0,255).astype(np.uint8))
+    frames = np.stack(output)
+    artifact_io.save(frames,ROOT)
+    print('Saved',frames.shape,frames.dtype)
+    print('Initial red/blue/yellow x:', path[0].round(3).tolist())
+    print('Contact events (source frame offsets):',contacts)
+    print('Final red/blue/yellow x:',path[-1].round(3).tolist())
+    print('Minimum separations:',np.min(np.diff(path,axis=1),axis=0).round(3).tolist())
+
+
+if __name__ == '__main__':
+    main()

\"\"\"Source-pixel curling edit, with an early third-stone intervention.

No assets other than the supplied source are used.  The rink is recovered
from unoccupied source pixels; the stones are matted from the original
video.  A one-dimensional equal-mass contact calculation follows the line
specified in the prompt, using restitution measured from the source impact.
\"\"\"
from pathlib import Path
import numpy as np
import cv2
from scipy import ndimage
import artifact_io

ROOT = Path(__file__).resolve().parent
S = 5


def source_data():
    frames = np.load(ROOT / 'source_pixels.npz')['rgb']
    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
    first = clean = None
    for i in range(13):
        ok, f = cap.read()
        if not ok:
            raise RuntimeError('Source video could not be decoded')
        if i == 0:
            first = f[..., ::-1].copy()
        if i == 12:
            clean = f[..., ::-1].copy()
    cap.release()
    return frames, first, clean


def stone_sprite(first, clean, anchor, bounds, blue=False):
    x0, y0, x1, y1 = bounds
    donor = first[y0*S:y1*S, x0*S:x1*S].astype(np.float32)
    bg = clean[y0*S:y1*S, x0*S:x1*S].astype(np.float32)
    delta = donor - bg
    d = np.max(np.abs(delta), axis=2)
    yy, xx = np.mgrid[y0*S:y1*S, x0*S:x1*S].astype(float) / S
    ax, ay = anchor
    # The body and its immediate contact shadow, not source codec noise.
    region = ((xx-ax)/6.1)**2 + ((yy-ay-2.1)/4.7)**2 < 1
    region |= (((xx-ax-2.8)/7.0)**2 + ((yy-ay-4.4)/2.0)**2 < 1
    strength = np.clip((d-5)/45, 0, 1) * region
    solid = (d > 30) & region
    labels, _ = ndimage.label(solid)
    sizes = np.bincount(labels.ravel())
    sizes[0] = 0
    body = labels == np.argmax(sizes)
    body = ndimage.binary_fill_holes(body)
    body = ndimage.gaussian_filter(body.astype(float), 0.7)
    alpha = np.maximum(strength, body * .98)
    alpha[d < 6] = 0
    # Premultiplied foreground, preserving donor shading and edge coverage.
    premul = delta + bg * alpha[..., None]
    premul[alpha == 0] = 0
    if blue:
        un = premul / np.maximum(alpha[..., None], 1e-5)
        r, g, b = un.transpose(2, 0, 1)
        handle = (r-g > 30) & (r-b > 30) & (yy < ay+1.7) & (alpha > .02)
        recolored = un.copy()
        recolored[..., 0] = .90*g+8
        recolored[..., 1] = .70*g+.30*r
        recolored[..., 2] = r
        # Recolor only the handle, never the neutral granite or its shadow.
        premul[handle] = recolored[handle] * alpha[handle, None]
    premul = np.clip(premul, 0, 255*alpha[..., None])
    return {
        'rgba': np.concatenate([premul, alpha[..., None]], axis=2).astype(np.float32),
        'anchor': np.array([(ax-x0)*S, (ay-y0)*S]),
    }


def get_tracks(frames):
    xy = np.zeros((len(frames), 2, 2), float)
    for t, f in enumerate(frames):
        r, g, b = f.astype(float).transpose(2, 0, 1)
        for j, mask in enumerate(((r-g > 110) & (r-b > 105),
                                  (g-b > 70) & (r-b > 70))):
            yy, xx = np.where(mask)
            weights = (r-g)[mask] if j == 0 else (g-b)[mask]
            xy[t, j] = [np.average(xx, weights=weights),
                        np.average(yy, weights=weights)]
    return xy


def simulate(tracks):
    # Track ordering: red, inserted blue, yellow. Coordinates are projected
    # distances along the common line (the transverse offset is zero).
    red0, yellow0 = tracks[0, 0, 0], tracks[0, 1, 0]
    blue0 = .25*red0 + .75*yellow0
    x = np.array([red0, blue0, yellow0], float)
    radii = np.array([4.60, 4.83, 5.00])
    v = np.array([1.98, 0, -2.27])
    changed = np.array([False, True, False])
    traj = [x.copy()]
    contacts = []
    dt = 1/80
    # The source's two-stone relative speed falls to about a tenth of its
    # incoming value. Low restitution gives the same granite-on-ice response.
    restitution = .105
    drag = .0025  # pixels / source-frame squared, after an altered contact
    for frame in range(1, 96):
        for step in range(80):
            time = frame-1 + (step+1)*dt
            old = x.copy()
            for j, src_j in ((0, 0), (2, 1)):
                if not changed[j]:
                    k = int(np.floor(time))
                    a = time-k
                    x[j] = (1-a)*tracks[k,src_j,0] + a*tracks[min(k+1,95),src_j,0]
                    v[j] = (x[j]-old[j])/dt
            for j in range(3):
                if changed[j]:
                    v[j] = np.sign(v[j])*max(0., abs(v[j])-drag*dt)
                    x[j] += v[j]*dt
            # Sequential impulses with overlap correction, iterated for the
            # brief three-body contact instead of allowing interpenetration.
            for _ in range(8):
                for j in range(2):
                    gap = x[j+1]-x[j]
                    required = radii[j]+radii[j+1]
                    if gap < required:
                        overlap = required-gap
                        x[j] -= overlap/2
                        x[j+1] += overlap/2
                        approach = v[j]-v[j+1]
                        if approach > 1e-7:
                            impulse = (1+restitution)*approach/2
                            v[j] -= impulse
                            v[j+1] += impulse
                            if len(contacts) == 0 or time-contacts[-1][0] > .25:
                                contacts.append((round(time,3), j, j+1))
                            changed[j:j+2] = True
        traj.append(x.copy())
    return np.array(traj), contacts


def paste(canvas, sprite, xy, scale=1.):
    rgba = sprite['rgba']
    anchor = sprite['anchor']
    # Warp the donor at source resolution before area-resampling. This keeps
    # subpixel motion smooth at the requested small output resolution.
    h, w = rgba.shape[:2]
    loc = np.asarray(xy)*S
    top_left = loc-scale*anchor
    left, top = np.floor(top_left-2).astype(int)
    right, bottom = np.ceil(top_left+np.array([w,h])*scale+2).astype(int)
    left, top = max(0,left), max(0,top)
    right, bottom = min(canvas.shape[1],right), min(canvas.shape[0],bottom)
    if right <= left or bottom <= top:
        return
    transform = np.array([[scale,0,top_left[0]-left],
                          [0,scale,top_left[1]-top]], np.float32)
    warped = cv2.warpAffine(rgba, transform, (right-left,bottom-top),
                            flags=cv2.INTER_LINEAR,
                            borderMode=cv2.BORDER_CONSTANT, borderValue=0)
    a = warped[...,3:4]
    tile = canvas[top:bottom,left:right]
    tile[:] = tile*(1-a)+warped[...,:3]


def main():
    src, first, clean = source_data()
    tracks = get_tracks(src)
    red_anchor = tracks[0,0]
    yellow_anchor = tracks[0,1]
    red = stone_sprite(first, clean, red_anchor, (39,60,60,77))
    yellow = stone_sprite(first, clean, yellow_anchor, (204,70,226,88))
    blue = stone_sprite(first, clean, red_anchor, (39,60,60,77), blue=True)
    background = src[0].copy()
    # These source patches are empty throughout most of the video. Median
    # selection removes the original moving bodies and their contact shadows.
    med = np.median(src,axis=0).astype(np.uint8)
    background[61:76,39:60] = med[61:76,39:60]
    background[71:88,204:226] = med[71:88,204:226]
    path, contacts = simulate(tracks)
    slope = (yellow_anchor[1]-red_anchor[1]) / (yellow_anchor[0]-red_anchor[0])
    output = []
    for t in range(96):
        # Nearest replication + area reduction is identity on the unaffected
        # supplied RGB background, while stones retain full-resolution edges.
        canvas = np.repeat(np.repeat(background,S,axis=0),S,axis=1).astype(np.float32)
        anchors = [[x, red_anchor[1]+slope*(x-red_anchor[0])] for x in path[t]]
        # Far-to-near compositing. All bodies stay tangent rather than overlap.
        for j in np.argsort(np.array(anchors)[:,1]):
            sprite = [red,blue,yellow][j]
            scale = 1.065 if j == 1 else 1.
            paste(canvas,sprite,anchors[j],scale)
        frame = cv2.resize(canvas,(256,144),interpolation=cv2.INTER_AREA)
        output.append(np.clip(np.rint(frame),0,255).astype(np.uint8))
    frames = np.stack(output)
    artifact_io.save(frames,ROOT)
    print('Saved',frames.shape,frames.dtype)
    print('Initial red/blue/yellow x:', path[0].round(3).tolist())
    print('Contact events (source frame offsets):',contacts)
    print('Final red/blue/yellow x:',path[-1].round(3).tolist())
    print('Minimum separations:',np.min(np.diff(path,axis=1),axis=0).round(3).tolist())


if __name__ == '__main__':
    main()
