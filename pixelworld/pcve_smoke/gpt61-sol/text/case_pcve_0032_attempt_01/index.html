+"""Restitution counterfactual, reconstructed from the source RGB sequence.
+
+The camera, set, materials, and pre-contact motion are copied from the source.
+Ball boundaries and ballistic arcs are measured numerically.  The changed
+impact uses one tenth of the observed normal restitution; tangential rolling
+and damping remain unchanged.  No external rendering/simulation is used.
+"""
+from pathlib import Path
+import json
+import numpy as np
+from scipy.ndimage import (label, map_coordinates, gaussian_filter,
+                           distance_transform_edt)
+from scipy.optimize import least_squares
+from scipy.special import ndtr
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+Y, X = np.mgrid[:144, :256].astype(float)
+
+
+def sample(a, ys, xs):
+    if a.ndim == 2:
+        return map_coordinates(a, [ys, xs], order=1, mode='nearest')
+    return np.stack([map_coordinates(a[..., c], [ys, xs], order=1,
+                                    mode='nearest') for c in range(3)], -1)
+
+
+def track_ball(a, background):
+    """Find the strongest falling chroma edge along 96 radial rays."""
+    angles = np.linspace(0, 2*np.pi, 96, endpoint=False)
+    radii = np.arange(9.5, 13, .1)
+    cosa, sina = np.cos(angles), np.sin(angles)
+    result = []
+    for f in a:
+        r, g, b = f.transpose(2, 0, 1)
+        mask = (r > 1.6*g) & (r > 100) & (g < 110) & (b < 110)
+        labs, count = label(mask)
+        areas = np.bincount(labs.ravel()); areas[0] = 0
+        yy, xx = np.where(labs == np.argmax(areas))
+        cx, cy = xx.mean(), yy.mean()
+        chroma = r-g-(background[..., 0]-background[..., 1])
+        for _ in range(3):
+            xs = cx+cosa[:, None]*radii
+            ys = cy+sina[:, None]*radii
+            z = sample(chroma, ys, xs)
+            derivative = (z[:, 6:]-z[:, :-6])/.6
+            j = np.argmin(derivative, axis=1)
+            rad = radii[j+3]
+            valid = derivative[np.arange(96), j] < -10
+            ex, ey = cx+cosa*rad, cy+sina*rad
+            fit = least_squares(
+                lambda p: np.hypot(ex[valid]-p[0], ey[valid]-p[1])-p[2],
+                [cx, cy, 11.1], loss='soft_l1', f_scale=.2).x
+            cx, cy = fit[:2]
+        result.append(fit)
+    return np.array(result)
+
+
+def make_ground_effect(source, background, center):
+    """Extract the real colored reflection, contact shadow and soft cast shadow.
+
+    Only the floor residual is retained; the sphere footprint is filled from
+    surrounding shadow samples so no portion of the old sphere is duplicated.
+    """
+    cx, cy, rad = center
+    rr = np.hypot(X-cx, Y-cy)
+    ratio = source/np.maximum(background, 1)-1
+    occluded = rr < rad+1.4
+    # nearest unoccluded floor samples; their smoothed interpolation is hidden
+    # by the new ball except at its subpixel antialiased edge.
+    _, inds = distance_transform_edt(occluded, return_indices=True)
+    filled = ratio.copy()
+    for c in range(3):
+        near = ratio[inds[0], inds[1], c]
+        filled[..., c][occluded] = near[occluded]
+        smooth = gaussian_filter(filled[..., c], 1.1)
+        filled[..., c][occluded] = smooth[occluded]
+    # The source shadow/reflection sits below the sphere and on the wood.
+    vertical = np.clip((Y-(cy+3))/3, 0, 1)
+    horizontal = np.clip((32-abs(X-cx))/8, 0, 1)
+    lower = np.clip((132-Y)/12, 0, 1)
+    window = vertical*horizontal*lower
+    filled = gaussian_filter(filled, (.55, .55, 0))
+    return filled*window[..., None]
+
+
+def ball_foreground(f, background, center):
+    """Recover RGB foreground against the source underlay, keeping its texture."""
+    cx, cy, rad = center
+    rr = np.hypot(X-cx, Y-cy)
+    alpha = ndtr((rad-rr)/.43)
+    # Remove the known underlying background from the antialiasing mixture.
+    fg = (f-(1-alpha[..., None])*background)/np.maximum(alpha[..., None], .08)
+    # Radial interior extrapolation supplies well-conditioned colors at the
+    # very outer edge rather than amplifying codec noise in nearly clear pixels.
+    scale = np.minimum(1., 9.5/np.maximum(rr, .01))
+    inner = sample(f, cy+(Y-cy)*scale, cx+(X-cx)*scale)
+    stabilize = np.clip((rr-rad+.45)/.8, 0, 1)[..., None]
+    fg = fg*(1-stabilize)+inner*stabilize
+    return np.clip(fg, 0, 200)
+
+
+def main():
+    source = np.load(ROOT/'source_pixels.npz')['rgb'].astype(float)
+    info = json.loads((ROOT/'source_info.json').read_text())
+    intervention = info['execution_frame_1based']-1
+    # Clean wood comes from the early high-altitude frames.  The clean wall
+    # comes from frames in which the sphere has moved well below this region.
+    background = np.median(source[:8], axis=0)
+    background[:63] = np.median(source[35:60, :63], axis=0)
+    centers = track_ball(source, background)
+
+    # Source arcs on either side of the first impact AFTER frame 20.
+    t1 = np.arange(17, 31)
+    t2 = np.arange(33, 48)
+    incoming = np.polyfit(t1, centers[t1, 1], 2)
+    outgoing = np.polyfit(t2, centers[t2, 1], 2)
+    roots = np.roots(incoming-outgoing)
+    contact = float(min(roots, key=lambda t: abs(t-32)))
+    floor = float(np.polyval(incoming, contact))
+    gravity = float(2*incoming[0])
+    incoming_speed = float(np.polyval(np.polyder(incoming), contact))
+    source_e = float(-np.polyval(np.polyder(outgoing), contact)/incoming_speed)
+    new_e = source_e*.1
+
+    # A tiny rebound, then successively smaller bounces.  This is an event-based
+    # ballistic solution, not a reduction of the current midair velocity.
+    def new_height(t):
+        if t < contact:
+            return float(np.polyval(incoming, t))
+        elapsed = t-contact
+        up = incoming_speed*new_e
+        while up > .012:
+            duration = 2*up/gravity
+            if elapsed < duration:
+                return floor-up*elapsed+.5*gravity*elapsed**2
+            elapsed -= duration
+            up *= new_e
+        return floor
+
+    # Tangential rolling damping measured across the complete source.  Fit
+    # perspective shear as well, since image x has a slight dependence on y.
+    tt = np.arange(13, 96, dtype=float)
+    def xmodel(p):
+        return p[0]-p[1]*np.exp(-p[2]*(tt-13))+p[3]*centers[13:, 1]
+    px = least_squares(lambda p: xmodel(p)-centers[13:, 0],
+                       [171., 33., .0115, -.007],
+                       bounds=([155, 1, .001, -.1], [220, 100, .1, .1])).x
+    def xpos(t, y):
+        return px[0]-px[1]*np.exp(-px[2]*(t-13))+px[3]*y
+    # Ensure subpixel continuity at the last factual sample.
+    anchor = int(np.floor(contact))
+    xoffset = centers[anchor, 0]-xpos(anchor, centers[anchor, 1])
+
+    reference = 48  # a source frame at floor contact, with matching lighting
+    ground = make_ground_effect(source[reference], background, centers[reference])
+    result = source.copy()
+    trace = []
+    for t in range(96):
+        if t <= anchor or t < intervention:
+            trace.append([t+1, float(centers[t, 0]), float(centers[t, 1])])
+            continue
+        ny = new_height(t)
+        nx = xpos(t, ny)+xoffset
+        # Preserve a rigid, fixed radius; restitution does not change the ball.
+        nr = float(np.median(centers[17:28, 2]))
+        cx, cy, cr = centers[t]
+        # Remove only the source sphere and its affected floor-light footprint.
+        sphere_erase = np.clip((15-np.hypot(X-cx, Y-cy))/2, 0, 1)
+        floor_erase = np.clip((39-abs(X-cx))/7, 0, 1)
+        floor_erase *= np.clip((Y-65)/5, 0, 1)*np.clip((140-Y)/10, 0, 1)
+        erase = np.maximum(sphere_erase, floor_erase)
+        frame = source[t]*(1-erase[..., None])+background*erase[..., None]
+
+        # Translate the source's contact/reflection field along the floor.
+        gx, gy = X-(nx-centers[reference, 0]), Y-(floor-centers[reference, 1])
+        effect = sample(ground, gy, gx)
+        frame *= np.clip(1+effect, .5, 1.2)
+
+        # Keep the time-varying source ball surface and its orientation.  Only
+        # its center height changes, with antialias-aware foreground compositing.
+        fg = ball_foreground(source[t], background, centers[t])
+        sx = cx+(X-nx)*cr/nr
+        sy = cy+(Y-ny)*cr/nr
+        ball_rgb = sample(fg, sy, sx)
+        a = ndtr((nr-np.hypot(X-nx, Y-ny))/.43)
+        frame = frame*(1-a[..., None])+ball_rgb*a[..., None]
+        result[t] = np.clip(frame, 0, 255)
+        trace.append([t+1, nx, ny])
+
+    frames = np.rint(result).clip(0, 255).astype(np.uint8)
+    # The untouched factual interval is exactly RGB-identical, including grain.
+    assert np.array_equal(frames[:intervention], source[:intervention].astype(np.uint8))
+    assert frames.shape == (96, 144, 256, 3)
+    artifact_io.save(frames, ROOT)
+    (ROOT/'motion_diagnostics.json').write_text(json.dumps({
+        'intervention_frame_1based': intervention+1,
+        'first_affected_impact_frame_1based': contact+1,
+        'floor_center_y': floor, 'gravity_pixels_per_frame2': gravity,
+        'original_restitution_estimate': source_e,
+        'counterfactual_restitution': new_e,
+        'counterfactual_rebound_height_pixels': (incoming_speed*new_e)**2/(2*gravity),
+        'center_trajectory': trace,
+        'output_frames': 96, 'fps': 24,
+    }, indent=2))
+    print('Saved 96 frames; source restitution %.4f -> %.4f; impact %.3f; rebound %.3f px' %
+          (source_e, new_e, contact+1, (incoming_speed*new_e)**2/(2*gravity)))
+
+
+if __name__ == '__main__':
+    main()

\"\"\"Restitution counterfactual, reconstructed from the source RGB sequence.

The camera, set, materials, and pre-contact motion are copied from the source.
Ball boundaries and ballistic arcs are measured numerically.  The changed
impact uses one tenth of the observed normal restitution; tangential rolling
and damping remain unchanged.  No external rendering/simulation is used.

from pathlib import Path
import json
import numpy as np
from scipy.ndimage import (label, map_coordinates, gaussian_filter,
                           distance_transform_edt)
from scipy.optimize import least_squares
from scipy.special import ndtr
import artifact_io

ROOT = Path(__file__).resolve().parent
Y, X = np.mgrid[:144, :256].astype(float)


def sample(a, ys, xs):
    if a.ndim == 2:
        return map_coordinates(a, [ys, xs], order=1, mode='nearest')
    return np.stack([map_coordinates(a[..., c], [ys, xs], order=1,
                                    mode='nearest') for c in range(3)], -1)


def track_ball(a, background):
    \"\"\"Find the strongest falling chroma edge along 96 radial rays.\"\"\"
    angles = np.linspace(0, 2*np.pi, 96, endpoint=False)
    radii = np.arange(9.5, 13, .1)
    cosa, sina = np.cos(angles), np.sin(angles)
    result = []
    for f in a:
        r, g, b = f.transpose(2, 0, 1)
        mask = (r > 1.6*g) & (r > 100) & (g < 110) & (b < 110)
        labs, count = label(mask)
        areas = np.bincount(labs.ravel()); areas[0] = 0
        yy, xx = np.where(labs == np.argmax(areas))
        cx, cy = xx.mean(), yy.mean()
        chroma = r-g-(background[..., 0]-background[..., 1])
        for _ in range(3):
            xs = cx+cosa[:, None]*radii
            ys = cy+sina[:, None]*radii
            z = sample(chroma, ys, xs)
            derivative = (z[:, 6:]-z[:, :-6])/.6
            j = np.argmin(derivative, axis=1)
            rad = radii[j+3]
            valid = derivative[np.arange(96), j] < -10
            ex, ey = cx+cosa*rad, cy+sina*rad
            fit = least_squares(
                lambda p: np.hypot(ex[valid]-p[0], ey[valid]-p[1])-p[2],
                [cx, cy, 11.1], loss='soft_l1', f_scale=.2).x
            cx, cy = fit[:2]
        result.append(fit)
    return np.array(result)


def make_ground_effect(source, background, center):
    \"\"\"Extract the real colored reflection, contact shadow and soft cast shadow.

    Only the floor residual is retained; the sphere footprint is filled from
    surrounding shadow samples so no portion of the old sphere is duplicated.
    \"\"\"
    cx, cy, rad = center
    rr = np.hypot(X-cx, Y-cy)
    ratio = source/np.maximum(background, 1)-1
    occluded = rr < rad+1.4
    # nearest unoccluded floor samples; their smoothed interpolation is hidden
    # by the new ball except at its subpixel antialiased edge.
    _, inds = distance_transform_edt(occluded, return_indices=True)
    filled = ratio.copy()
    for c in range(3):
        near = ratio[inds[0], inds[1], c]
        filled[..., c][occluded] = near[occluded]
        smooth = gaussian_filter(filled[..., c], 1.1)
        filled[..., c][occluded] = smooth[occluded]
    # The source shadow/reflection sits below the sphere and on the wood.
    vertical = np.clip((Y-(cy+3))/3, 0, 1)
    horizontal = np.clip((32-abs(X-cx))/8, 0, 1)
    lower = np.clip((132-Y)/12, 0, 1)
    window = vertical*horizontal*lower
    filled = gaussian_filter(filled, (.55, .55, 0))
    return filled*window[..., None]


def ball_foreground(f, background, center):
    \"\"\"Recover RGB foreground against the source underlay, keeping its texture.\"\"\"
    cx, cy, rad = center
    rr = np.hypot(X-cx, Y-cy)
    alpha = ndtr((rad-rr)/.43)
    # Remove the known underlying background from the antialiasing mixture.
    fg = (f-(1-alpha[..., None])*background)/np.maximum(alpha[..., None], .08)
    # Radial interior extrapolation supplies well-conditioned colors at the
    # very outer edge rather than amplifying codec noise in nearly clear pixels.
    scale = np.minimum(1., 9.5/np.maximum(rr, .01))
    inner = sample(f, cy+(Y-cy)*scale, cx+(X-cx)*scale)
    stabilize = np.clip((rr-rad+.45)/.8, 0, 1)[..., None]
    fg = fg*(1-stabilize)+inner*stabilize
    return np.clip(fg, 0, 200)


def main():
    source = np.load(ROOT/'source_pixels.npz')['rgb'].astype(float)
    info = json.loads((ROOT/'source_info.json').read_text())
    intervention = info['execution_frame_1based']-1
    # Clean wood comes from the early high-altitude frames.  The clean wall
    # comes from frames in which the sphere has moved well below this region.
    background = np.median(source[:8], axis=0)
    background[:63] = np.median(source[35:60, :63], axis=0)
    centers = track_ball(source, background)

    # Source arcs on either side of the first impact AFTER frame 20.
    t1 = np.arange(17, 31)
    t2 = np.arange(33, 48)
    incoming = np.polyfit(t1, centers[t1, 1], 2)
    outgoing = np.polyfit(t2, centers[t2, 1], 2)
    roots = np.roots(incoming-outgoing)
    contact = float(min(roots, key=lambda t: abs(t-32)))
    floor = float(np.polyval(incoming, contact))
    gravity = float(2*incoming[0])
    incoming_speed = float(np.polyval(np.polyder(incoming), contact))
    source_e = float(-np.polyval(np.polyder(outgoing), contact)/incoming_speed)
    new_e = source_e*.1

    # A tiny rebound, then successively smaller bounces.  This is an event-based
    # ballistic solution, not a reduction of the current midair velocity.
    def new_height(t):
        if t < contact:
            return float(np.polyval(incoming, t))
        elapsed = t-contact
        up = incoming_speed*new_e
        while up > .012:
            duration = 2*up/gravity
            if elapsed < duration:
                return floor-up*elapsed+.5*gravity*elapsed**2
            elapsed -= duration
            up *= new_e
        return floor

    # Tangential rolling damping measured across the complete source.  Fit
    # perspective shear as well, since image x has a slight dependence on y.
    tt = np.arange(13, 96, dtype=float)
    def xmodel(p):
        return p[0]-p[1]*np.exp(-p[2]*(tt-13))+p[3]*centers[13:, 1]
    px = least_squares(lambda p: xmodel(p)-centers[13:, 0],
                       [171., 33., .0115, -.007],
                       bounds=([155, 1, .001, -.1], [220, 100, .1, .1])).x
    def xpos(t, y):
        return px[0]-px[1]*np.exp(-px[2]*(t-13))+px[3]*y
    # Ensure subpixel continuity at the last factual sample.
    anchor = int(np.floor(contact))
    xoffset = centers[anchor, 0]-xpos(anchor, centers[anchor, 1])

    reference = 48  # a source frame at floor contact, with matching lighting
    ground = make_ground_effect(source[reference], background, centers[reference])
    result = source.copy()
    trace = []
    for t in range(96):
        if t <= anchor or t < intervention:
            trace.append([t+1, float(centers[t, 0]), float(centers[t, 1])])
            continue
        ny = new_height(t)
        nx = xpos(t, ny)+xoffset
        # Preserve a rigid, fixed radius; restitution does not change the ball.
        nr = float(np.median(centers[17:28, 2]))
        cx, cy, cr = centers[t]
        # Remove only the source sphere and its affected floor-light footprint.
        sphere_erase = np.clip((15-np.hypot(X-cx, Y-cy))/2, 0, 1)
        floor_erase = np.clip((39-abs(X-cx))/7, 0, 1)
        floor_erase *= np.clip((Y-65)/5, 0, 1)*np.clip((140-Y)/10, 0, 1)
        erase = np.maximum(sphere_erase, floor_erase)
        frame = source[t]*(1-erase[..., None])+background*erase[..., None]

        # Translate the source's contact/reflection field along the floor.
        gx, gy = X-(nx-centers[reference, 0]), Y-(floor-centers[reference, 1])
        effect = sample(ground, gy, gx)
        frame *= np.clip(1+effect, .5, 1.2)

        # Keep the time-varying source ball surface and its orientation.  Only
        # its center height changes, with antialias-aware foreground compositing.
        fg = ball_foreground(source[t], background, centers[t])
        sx = cx+(X-nx)*cr/nr
        sy = cy+(Y-ny)*cr/nr
        ball_rgb = sample(fg, sy, sx)
        a = ndtr((nr-np.hypot(X-nx, Y-ny))/.43)
        frame = frame*(1-a[..., None])+ball_rgb*a[..., None]
        result[t] = np.clip(frame, 0, 255)
        trace.append([t+1, nx, ny])

    frames = np.rint(result).clip(0, 255).astype(np.uint8)
    # The untouched factual interval is exactly RGB-identical, including grain.
    assert np.array_equal(frames[:intervention], source[:intervention].astype(np.uint8))
    assert frames.shape == (96, 144, 256, 3)
    artifact_io.save(frames, ROOT)
    (ROOT/'motion_diagnostics.json').write_text(json.dumps({
        'intervention_frame_1based': intervention+1,
        'first_affected_impact_frame_1based': contact+1,
        'floor_center_y': floor, 'gravity_pixels_per_frame2': gravity,
        'original_restitution_estimate': source_e,
        'counterfactual_restitution': new_e,
        'counterfactual_rebound_height_pixels': (incoming_speed*new_e)**2/(2*gravity),
        'center_trajectory': trace,
        'output_frames': 96, 'fps': 24,
    }, indent=2))
    print('Saved 96 frames; source restitution %.4f -> %.4f; impact %.3f; rebound %.3f px' %
          (source_e, new_e, contact+1, (incoming_speed*new_e)**2/(2*gravity)))


if __name__ == '__main__':
    main()
