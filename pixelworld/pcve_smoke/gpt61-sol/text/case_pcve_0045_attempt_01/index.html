+"""Friction intervention, source-textured rigid motion and RGB compositing.
+
+The camera and set are locked.  The original car travels left up a ramp,
+leaves its upper lip, and drops to the tabletop.  We measure its travel and
+fit the original retardation; the friction contribution is multiplied by .2.
+The new exit velocity then drives a ballistic flight and low-friction skid.
+No scene/model assets other than the provided source are used.
+"""
+from pathlib import Path
+import json
+import numpy as np
+import cv2
+from scipy.ndimage import (label, binary_closing, binary_fill_holes,
+                           binary_dilation, gaussian_filter)
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+# Tight source-object ROIs, in output pixel coordinates.  They are simply
+# numerical measurements of temporal RGB changes against the empty ramp.
+BOXES = [
+    (150,100,194,123), (136,94,180,117), (122,89,167,112),
+    (110,85,154,108), (98,81,143,104), (88,77,130,100),
+    (78,73,122,97), (69,70,113,94), (62,67,106,91),
+    (54,65,98,88), (48,62,92,86), (42,61,86,84),
+    (37,60,81,83), (32,61,76,84), (28,63,72,89),
+    (24,66,67,98), (20,73,61,108),
+]
+
+
+def largest(mask):
+    components, n = label(mask)
+    counts = np.bincount(components.ravel())
+    if len(counts) < 2:
+        return mask
+    counts[0] = 0
+    return components == counts.argmax()
+
+
+def make_sprite(image, background, box, scale):
+    x0,y0,x1,y1 = box
+    xs,ys,xe,ye = [int(round(v*scale)) for v in (x0,y0,x1,y1)]
+    rgb = image[ys:ye,xs:xe].astype(np.float32)
+    ref = background[ys:ye,xs:xe].astype(np.float32)
+    diff = np.max(np.abs(rgb-ref), axis=2)
+    # Close small near-identical material patches, retaining the true outer
+    # silhouette; fill the opaque toy including the dark tires and glazing.
+    mask = binary_closing(diff > 23, iterations=max(1,round(.5*scale)))
+    mask = binary_fill_holes(largest(mask))
+    # Subpixel matte edge rather than a rectangular crop.
+    matte = gaussian_filter(mask.astype(np.float32), sigma=.32*scale)
+    yy,xx = np.where(mask)
+    center = np.array([x0+xx.mean()/scale, y0+yy.mean()/scale])
+    return rgb, matte, np.array([x0,y0],dtype=float), center
+
+
+def raster_sprite(sprite, center, out_scale=4):
+    rgb, alpha, origin, old_center = sprite
+    source_scale = rgb.shape[1] / (sprite_width(sprite))
+    # source_scale is saved as an explicit fifth field by the caller.
+
+
+def layer(sprite, position, output_scale=4):
+    rgb, alpha, origin, center, scale = sprite
+    h,w = 144*output_scale, 256*output_scale
+    ratio = output_scale/scale
+    translation = (origin+np.asarray(position)-center)*output_scale
+    transform = np.array([[ratio,0,translation[0]], [0,ratio,translation[1]]],np.float32)
+    a = cv2.warpAffine(alpha,transform,(w,h),flags=cv2.INTER_LINEAR)
+    p = cv2.warpAffine(rgb*alpha[:,:,None],transform,(w,h),flags=cv2.INTER_LINEAR)
+    # Downsampling premultiplied color prevents edge color contamination.
+    a = cv2.resize(a,(256,144),interpolation=cv2.INTER_AREA)
+    p = cv2.resize(p,(256,144),interpolation=cv2.INTER_AREA)
+    return p, a
+
+
+def main():
+    source = np.load(ROOT/'source_pixels.npz')['rgb']
+    n,h,w,_ = source.shape
+    assert (n,h,w)==(96,144,256)
+    scale = 5
+    cap = cv2.VideoCapture(str(ROOT/'source_video.mp4'))
+    high = {}
+    for i in range(31):
+        ok, frame = cap.read()
+        if not ok:
+            raise RuntimeError('Cannot decode the supplied source video')
+        if i <=16 or i==30:
+            high[i] = cv2.cvtColor(frame,cv2.COLOR_BGR2RGB)
+    cap.release()
+    scale = high[0].shape[1]/256
+    sprites = [(*make_sprite(high[i],high[30],BOXES[i],scale),scale)
+               for i in range(17)]
+    # Once on the floor, use its genuine source floor pose.  The reference
+    # is frame zero, where this region was empty, not a synthetic car model.
+    flat = (*make_sprite(high[30],high[0],(3,108,50,132),scale),scale)
+    centers = np.array([s[3] for s in sprites])
+
+    # Empty background recovered temporally, with the final resting car
+    # replaced by the same table pixels from before it arrives there.
+    clean = np.median(source[26:],axis=0).astype(np.float32)
+    clean[108:133,:53] = source[0,108:133,:53]
+    # Restrict replacement to the entire factual car/shadow footprint,
+    # preserving all unrelated background and compression fluctuations.
+    deviation = np.max(np.abs(source.astype(np.float32)-clean),axis=3)
+    sweep = np.any(deviation>24,axis=0)
+    region = np.zeros((144,256),bool)
+    region[59:132,:198]=True
+    sweep &= region
+    sweep = binary_dilation(sweep,iterations=2)
+    clean_alpha = gaussian_filter(sweep.astype(np.float32),.55)
+
+    # Fit the initial source speed and observed uphill acceleration.  The
+    # ramp's gravitational component is not altered by the friction edit.
+    fit = np.polyfit(np.arange(12),centers[:12,0],2)
+    x_initial = centers[0,0]
+    v_initial = float(fit[1])
+    observed_retardation = 2*float(fit[0])
+    gravity_up_ramp = .60*observed_retardation
+    friction_original = observed_retardation-gravity_up_ramp
+    reduced_retardation = gravity_up_ramp+.2*friction_original
+    x_lip = float(centers[12,0])
+    # Positive root of the monotonic first crossing.
+    roots = np.roots([.5*reduced_retardation,v_initial,x_initial-x_lip])
+    exit_time = float(min(r.real for r in roots if abs(r.imag)<1e-6 and r.real>0))
+    exit_vx = v_initial+reduced_retardation*exit_time
+    ramp_slope = float(np.polyfit(centers[:11,0],centers[:11,1],1)[0])
+    exit_y = float(centers[12,1])
+    exit_vy = exit_vx*ramp_slope
+    # Vertical gravity is fitted from the factual free-fall, independent of
+    # surface friction.  Rotating body matte centroids introduce small error.
+    fall_fit = np.polyfit(np.arange(5),centers[12:17,1],2)
+    gravity = float(np.clip(2*fall_fit[0],1.9,2.5))
+    floor_y = float(flat[3][1])
+    flight_time = float((-exit_vy+np.sqrt(exit_vy**2+2*gravity*(floor_y-exit_y)))/gravity)
+    impact_t = exit_time+flight_time
+    impact_x = x_lip+exit_vx*flight_time
+
+    frames = []
+    track = []
+    for i in range(n):
+        if i == 0:
+            # Changing a material coefficient is not an instantaneous pose
+            # displacement; the state at the first frame is exactly factual.
+            frames.append(source[0].copy())
+            track.append([x_initial,float(centers[0,1]),0])
+            continue
+        frame = source[i].astype(np.float32)*(1-clean_alpha[:,:,None])+clean*clean_alpha[:,:,None]
+        if i < exit_time:
+            px = x_initial+v_initial*i+.5*reduced_retardation*i*i
+            # Match source pose to distance, NOT to counterfactual time.
+            s = float(np.interp(px,centers[:13,0][::-1],np.arange(13)[::-1]))
+            ia = int(np.floor(s)); ib = min(ia+1,12); weight=s-ia
+            py = float(np.interp(s,np.arange(13),centers[:13,1]))
+        elif i < impact_t:
+            tau=i-exit_time
+            px=x_lip+exit_vx*tau
+            py=exit_y+exit_vy*tau+.5*gravity*tau*tau
+            # The original changing pitch supplies a consistent view of the
+            # body and wheels during departure and flight.
+            s = min(16.,12.+.80*tau)
+            ia=int(np.floor(s));ib=min(ia+1,16);weight=s-ia
+        else:
+            tau=i-impact_t
+            # Landing retains tangential speed, and lower Coulomb drag lets
+            # the car continue skidding instead of abruptly freezing in place.
+            landing_vx = exit_vx*.72
+            drag = .2*friction_original
+            glide = min(tau,abs(landing_vx)/max(drag,1e-5))
+            px=impact_x+landing_vx*glide+.5*drag*glide*glide
+            py=floor_y-1.1*np.exp(-tau*.9)*abs(np.sin(tau*1.2))
+            ia=ib=16;weight=0.
+        track.append([float(px),float(py),int(i>=exit_time)])
+        # Full-canvas subpixel compositing. Once outside the view the car is
+        # simply occluded by the frame boundary; no fading or teleporting.
+        if -30 < px < 290:
+            if i>=impact_t:
+                p,a=layer(flat,(px,py))
+            else:
+                p0,a0=layer(sprites[ia],(px,py))
+                if ib != ia and weight>1e-5:
+                    p1,a1=layer(sprites[ib],(px,py))
+                    p=p0*(1-weight)+p1*weight;a=a0*(1-weight)+a1*weight
+                else:p,a=p0,a0
+            frame=frame*(1-a[:,:,None])+p
+        frames.append(np.clip(np.rint(frame),0,255).astype(np.uint8))
+
+    frames=np.stack(frames)
+    artifact_io.save(frames,ROOT)
+    # An ordinary H.264 container accompanies the required RGB artifact.
+    import imageio_ffmpeg
+    writer = imageio_ffmpeg.write_frames(str(ROOT/'output.mp4'),(256,144),fps=24,
+             codec='libx264',pix_fmt_in='rgb24',pix_fmt_out='yuv420p',
+             output_params=['-crf','16','-preset','medium','-movflags','+faststart'],
+             macro_block_size=1)
+    writer.send(None)
+    for frame in frames:
+        writer.send(frame)
+    writer.close()
+    (ROOT/'motion_analysis.json').write_text(json.dumps({
+        'original_retardation_px_frame2':observed_retardation,
+        'edited_retardation_px_frame2':reduced_retardation,
+        'friction_ratio':.2,'ramp_exit_frame_zero_based':exit_time,
+        'exit_vx_px_frame':exit_vx,'landing_frame_zero_based':impact_t,
+        'landing_x':impact_x,'centers':track},indent=2))
+    print('Saved',frames.shape,frames.dtype,'exit',round(exit_time,2),
+          'landing',round(impact_t,2),'landing x',round(impact_x,2))
+
+if __name__=='__main__':
+    main()

def largest(mask):
    components, n = label(mask)
    counts = np.bincount(components.ravel())
    if len(counts) < 2:
        return mask
    counts[0] = 0
    return components == counts.argmax()


def make_sprite(image, background, box, scale):
    x0,y0,x1,y1 = box
    xs,ys,xe,ye = [int(round(v*scale)) for v in (x0,y0,x1,y1)]
    rgb = image[ys:ye,xs:xe].astype(np.float32)
    ref = background[ys:ye,xs:xe].astype(np.float32)
    diff = np.max(np.abs(rgb-ref), axis=2)
    # Close small near-identical material patches, retaining the true outer
    # silhouette; fill the opaque toy including the dark tires and glazing.
    mask = binary_closing(diff > 23, iterations=max(1,round(.5*scale)))
    mask = binary_fill_holes(largest(mask))
    # Subpixel matte edge rather than a rectangular crop.
    matte = gaussian_filter(mask.astype(np.float32), sigma=.32*scale)
    yy,xx = np.where(mask)
    center = np.array([x0+xx.mean()/scale, y0+yy.mean()/scale])
    return rgb, matte, np.array([x0,y0],dtype=float), center

def layer(sprite, position, output_scale=4):
    rgb, alpha, origin, center, scale = sprite
    h,w = 144*output_scale, 256*output_scale
    ratio = output_scale/scale
    translation = (origin+np.asarray(position)-center)*output_scale
    transform = np.array([[ratio,0,translation[0]], [0,ratio,translation[1]]],np.float32)
    a = cv2.warpAffine(alpha,transform,(w,h),flags=cv2.INTER_LINEAR)
    p = cv2.warpAffine(rgb*alpha[:,:,None],transform,(w,h),flags=cv2.INTER_LINEAR)
    # Downsampling premultiplied color prevents edge color contamination.
    a = cv2.resize(a,(256,144),interpolation=cv2.INTER_AREA)
    p = cv2.resize(p,(256,144),interpolation=cv2.INTER_AREA)
    return p, a
