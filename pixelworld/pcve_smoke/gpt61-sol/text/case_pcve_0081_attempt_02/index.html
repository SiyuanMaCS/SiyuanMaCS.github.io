+The apple has identical geometry and initial state; only inertial mass changes.
+"""

+import json
+import cv2
+import numpy as np
+from scipy import ndimage

+ROOT = Path(__file__).resolve().parent

+def read_source():
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    frames = []
+    while True:
+        ok, bgr = cap.read()
+        if not ok:
+            break
+        frames.append(cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB))
+    cap.release()
+    return np.asarray(frames)

+def decay_distance(t, v, k, a):
+    """Integrated rolling resistance plus viscous dissipation; no reversal."""
+    stop = np.log1p(v*k/a)/k
+    t = np.clip(t, 0, stop)
+    return (v+a/k)*(-np.expm1(-k*t))/k-a*t/k
+

+def ball_positions(src):
+    positions = []
+    for im in src:
+        region = im[340:475, 200:720]
+        m = region.mean(axis=2) > 145
+        lab, n = ndimage.label(m)
+        sizes = np.bincount(lab.ravel()); sizes[0] = 0
+        obj = np.argmax(sizes)
+        ys, xs = np.where(lab == obj)
+        positions.append(((xs.min()+xs.max())/2+200,
+                          (ys.min()+ys.max())/2+340))
+    return np.array(positions)

+def apple_positions(src, plate):
+    positions = []
+    # A coarse horizontal fit only locates the original fruit for segmentation.
+    for t in range(14, 37):
+        dt = t-14
+        approx = 146 + 6.86*dt - .096*dt*dt
+        x0 = max(0, int(5*approx-45)); x1 = min(1280, int(5*approx+45))
+        y0,y1 = 398,455
+        d = np.max(np.abs(src[t,y0:y1,x0:x1].astype(float)-plate[y0:y1,x0:x1]), axis=2)
+        lab,n=ndimage.label(d>32)
+        sizes=np.bincount(lab.ravel()); sizes[0]=0
+        q=np.argmax(sizes)
+        ys,xs=np.where(lab==q)
+        # Bounding box tracks geometry, not rotating color patches.
+        positions.append(((xs.min()+xs.max())/2+x0,(ys.min()+ys.max())/2+y0))
+    return np.array(positions)

+def crop_layer(src, plate, cx, cy, kind):
+    if kind=='ball':
+        left,right,top,bottom=61,99,62,78
+        rx,ry=79,68
+    else:
+        left,right,top,bottom=28,57,30,38
+        rx,ry=43,32
+    x0=int(np.floor(cx))-left; y0=int(np.floor(cy))-top
+    x1=x0+left+right+1; y1=y0+top+bottom+1
+    ys,xs=np.mgrid[y0:y1,x0:x1]
+    if kind=='ball':
+        body=(xs-cx)**2/54**2+(ys-cy)**2/54**2
+        sh=(xs-cx-28)**2/60**2+(ys-cy-9)**2/61**2
+    else:
+        body=(xs-cx)**2/24**2+(ys-cy)**2/24**2
+        sh=(xs-cx-17)**2/31**2+(ys-cy-8)**2/23**2
+    alpha=np.maximum(np.clip((1.04-body)*12,0,1),np.clip((1.03-sh)*9,0,1)).astype(np.float32)
+    rgb=src[y0:y1,x0:x1].astype(np.float32)
+    ground=plate[y0:y1,x0:x1].astype(np.float32)
+    # Residual compositing carries soft cast shadows while retaining destination floor texture.
+    return (rgb-ground)*alpha[...,None], (x0-cx,y0-cy)
+
+
+def translated_layer(canvas, delta, origin, center):
+    h,w=delta.shape[:2]
+    px=center[0]+origin[0]; py=center[1]+origin[1]
+    x0=int(np.floor(px));y0=int(np.floor(py))
+    transform=np.array([[1.,0.,px-x0],[0.,1.,py-y0]],np.float32)
+    shifted=cv2.warpAffine(delta,transform,(w+2,h+2),flags=cv2.INTER_LINEAR,
+                           borderMode=cv2.BORDER_CONSTANT,borderValue=0)
+    xa=max(0,x0);xb=min(canvas.shape[1],x0+w+2)
+    ya=max(0,y0);yb=min(canvas.shape[0],y0+h+2)
+    if xb>xa and yb>ya:
+        canvas[ya:yb,xa:xb]+=shifted[ya-y0:yb-y0,xa-x0:xb-x0]
+
+
+def interpolate_layer(src,plate,positions,t,kind,offset=0):
+    t=np.clip(t,offset,offset+len(positions)-1)
+    i=int(np.floor(t));j=min(i+1,offset+len(positions)-1);f=t-i
+    c0=positions[i-offset];c1=positions[j-offset]
+    d0,o0=crop_layer(src[i],plate,*c0,kind)
+    d1,o1=crop_layer(src[j],plate,*c1,kind)
+    # Align the two sprites' centers before interpolating their source textures.
+    h,w=d0.shape[:2]
+    dx=o1[0]-o0[0];dy=o1[1]-o0[1]
+    d1=cv2.warpAffine(d1,np.array([[1,0,dx],[0,1,dy]],np.float32),(w,h),flags=cv2.INTER_LINEAR)
+    return d0*(1-f)+d1*f,o0
+
+
+def rotate_ball_body(src,plate,center,angle):
+    delta,origin=crop_layer(src,plate,*center,'ball')
+    h,w=delta.shape[:2]
+    yy,xx=np.mgrid[:h,:w].astype(np.float32)
+    x=(xx+origin[0])/52.2; y=(yy+origin[1])/52.2
+    z=np.sqrt(np.maximum(0,1-x*x-y*y))
+    p=np.stack([x,y,z],axis=-1)
+    axis=np.array([0,.56,.8285],dtype=np.float32);axis/=np.linalg.norm(axis)
+    c,s=np.cos(-angle),np.sin(-angle)
+    q=p*c+np.cross(axis,p)*s+axis*np.sum(p*axis,axis=-1)[...,None]*(1-c)
+    # Reflect the small newly revealed sliver at the limb, continuously.
+    qz=np.abs(q[:,:,2])
+    mapx=(center[0]+52.2*q[:,:,0]).astype(np.float32)
+    mapy=(center[1]+52.2*q[:,:,1]).astype(np.float32)
+    tex=cv2.remap(src.astype(np.float32),mapx,mapy,cv2.INTER_LINEAR,borderMode=cv2.BORDER_REFLECT_101)
+    oldlight=np.maximum(.35,.68+.23*qz-.10*q[:,:,0]-.13*q[:,:,1])
+    newlight=np.maximum(.35,.68+.23*z-.10*x-.13*y)
+    tex=tex*(newlight/oldlight)[...,None]
+    # Keep source shadow independent of rotation.
+    x0=int(round(center[0]+origin[0]));y0=int(round(center[1]+origin[1]))
+    ground=plate[y0:y0+h,x0:x0+w].astype(float)
+    r=np.sqrt(x*x+y*y)
+    body=np.clip((1-r)*52.2,0,1)[...,None]
+    delta=delta*(1-body)+(tex-ground)*body
+    return delta.astype(np.float32),origin
+

+        'initial_frames_preserved':13,'normal_impulse_multiplier':ball_gain,
+        'positions':records},indent=2))
+    print('Saved',frames.shape,frames.dtype,'last positions',records[-1])
+
+if __name__=='__main__':
+    main()

\"\"\"Counterfactual mass edit using source pixels, analytic impact and pixel warps.
The apple has identical geometry and initial state; only inertial mass changes.
\"\"\"
from pathlib import Path
import json
import cv2
import numpy as np
from scipy import ndimage
from artifact_io import save

ROOT = Path(__file__).resolve().parent

def read_source():
    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
    frames = []
    while True:
        ok, bgr = cap.read()
        if not ok:
            break
        frames.append(cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB))
    cap.release()
    return np.asarray(frames)


def decay_distance(t, v, k, a):
    \"\"\"Integrated rolling resistance plus viscous dissipation; no reversal.\"\"\"
    stop = np.log1p(v*k/a)/k
    t = np.clip(t, 0, stop)
    return (v+a/k)*(-np.expm1(-k*t))/k-a*t/k


def ball_positions(src):
    positions = []
    for im in src:
        region = im[340:475, 200:720]
        m = region.mean(axis=2) > 145
        lab, n = ndimage.label(m)
        sizes = np.bincount(lab.ravel()); sizes[0] = 0
        obj = np.argmax(sizes)
        ys, xs = np.where(lab == obj)
        positions.append(((xs.min()+xs.max())/2+200,
                          (ys.min()+ys.max())/2+340))
    return np.array(positions)


def apple_positions(src, plate):
    positions = []
    # A coarse horizontal fit only locates the original fruit for segmentation.
    for t in range(14, 37):
        dt = t-14
        approx = 146 + 6.86*dt - .096*dt*dt
        x0 = max(0, int(5*approx-45)); x1 = min(1280, int(5*approx+45))
        y0,y1 = 398,455
        d = np.max(np.abs(src[t,y0:y1,x0:x1].astype(float)-plate[y0:y1,x0:x1]), axis=2)
        lab,n=ndimage.label(d>32)
        sizes=np.bincount(lab.ravel()); sizes[0]=0
        q=np.argmax(sizes)
        ys,xs=np.where(lab==q)
        # Bounding box tracks geometry, not rotating color patches.
        positions.append(((xs.min()+xs.max())/2+x0,(ys.min()+ys.max())/2+y0))
    return np.array(positions)


def crop_layer(src, plate, cx, cy, kind):
    if kind=='ball':
        left,right,top,bottom=61,99,62,78
        rx,ry=79,68
    else:
        left,right,top,bottom=28,57,30,38
        rx,ry=43,32
    x0=int(np.floor(cx))-left; y0=int(np.floor(cy))-top
    x1=x0+left+right+1; y1=y0+top+bottom+1
    ys,xs=np.mgrid[y0:y1,x0:x1]
    if kind=='ball':
        body=(xs-cx)**2/54**2+(ys-cy)**2/54**2
        sh=(xs-cx-28)**2/60**2+(ys-cy-9)**2/61**2
    else:
        body=(xs-cx)**2/24**2+(ys-cy)**2/24**2
        sh=(xs-cx-17)**2/31**2+(ys-cy-8)**2/23**2
    alpha=np.maximum(np.clip((1.04-body)*12,0,1),np.clip((1.03-sh)*9,0,1)).astype(np.float32)
    rgb=src[y0:y1,x0:x1].astype(np.float32)
    ground=plate[y0:y1,x0:x1].astype(np.float32)
    # Residual compositing carries soft cast shadows while retaining destination floor texture.
    return (rgb-ground)*alpha[...,None], (x0-cx,y0-cy)


def translated_layer(canvas, delta, origin, center):
    h,w=delta.shape[:2]
    px=center[0]+origin[0]; py=center[1]+origin[1]
    x0=int(np.floor(px));y0=int(np.floor(py))
    transform=np.array([[1.,0.,px-x0],[0.,1.,py-y0]],np.float32)
    shifted=cv2.warpAffine(delta,transform,(w+2,h+2),flags=cv2.INTER_LINEAR,
                           borderMode=cv2.BORDER_CONSTANT,borderValue=0)
    xa=max(0,x0);xb=min(canvas.shape[1],x0+w+2)
    ya=max(0,y0);yb=min(canvas.shape[0],y0+h+2)
    if xb>xa and yb>ya:
        canvas[ya:yb,xa:xb]+=shifted[ya-y0:yb-y0,xa-x0:xb-x0]


def interpolate_layer(src,plate,positions,t,kind,offset=0):
    t=np.clip(t,offset,offset+len(positions)-1)
    i=int(np.floor(t));j=min(i+1,offset+len(positions)-1);f=t-i
    c0=positions[i-offset];c1=positions[j-offset]
    d0,o0=crop_layer(src[i],plate,*c0,kind)
    d1,o1=crop_layer(src[j],plate,*c1,kind)
    # Align the two sprites' centers before interpolating their source textures.
    h,w=d0.shape[:2]
    dx=o1[0]-o0[0];dy=o1[1]-o0[1]
    d1=cv2.warpAffine(d1,np.array([[1,0,dx],[0,1,dy]],np.float32),(w,h),flags=cv2.INTER_LINEAR)
    return d0*(1-f)+d1*f,o0


def rotate_ball_body(src,plate,center,angle):
    delta,origin=crop_layer(src,plate,*center,'ball')
    h,w=delta.shape[:2]
    yy,xx=np.mgrid[:h,:w].astype(np.float32)
    x=(xx+origin[0])/52.2; y=(yy+origin[1])/52.2
    z=np.sqrt(np.maximum(0,1-x*x-y*y))
    p=np.stack([x,y,z],axis=-1)
    axis=np.array([0,.56,.8285],dtype=np.float32);axis/=np.linalg.norm(axis)
    c,s=np.cos(-angle),np.sin(-angle)
    q=p*c+np.cross(axis,p)*s+axis*np.sum(p*axis,axis=-1)[...,None]*(1-c)
    # Reflect the small newly revealed sliver at the limb, continuously.
    qz=np.abs(q[:,:,2])
    mapx=(center[0]+52.2*q[:,:,0]).astype(np.float32)
    mapy=(center[1]+52.2*q[:,:,1]).astype(np.float32)
    tex=cv2.remap(src.astype(np.float32),mapx,mapy,cv2.INTER_LINEAR,borderMode=cv2.BORDER_REFLECT_101)
    oldlight=np.maximum(.35,.68+.23*qz-.10*q[:,:,0]-.13*q[:,:,1])
    newlight=np.maximum(.35,.68+.23*z-.10*x-.13*y)
    tex=tex*(newlight/oldlight)[...,None]
    # Keep source shadow independent of rotation.
    x0=int(round(center[0]+origin[0]));y0=int(round(center[1]+origin[1]))
    ground=plate[y0:y0+h,x0:x0+w].astype(float)
    r=np.sqrt(x*x+y*y)
    body=np.clip((1-r)*52.2,0,1)[...,None]
    delta=delta*(1-body)+(tex-ground)*body
    return delta.astype(np.float32),origin
