+"""Counterfactual removal of the block, using only the source RGB sequence.
+
+The table is recovered from pixels exposed elsewhere on the source timeline.
+The free rolling trajectory is fitted before contact; source sphere appearance
+and its soft cast shadow are then reprojected along that trajectory.
+"""
+from pathlib import Path
+import numpy as np
+import cv2
+from scipy.ndimage import label, gaussian_filter, distance_transform_edt
+from scipy.optimize import least_squares
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+
+def sphere_geometry(frame):
+    r, g, b = np.moveaxis(frame, -1, 0)
+    labels, count = label((r-g > 62) & (g-b > 14))
+    component = 1 + np.argmax(np.bincount(labels.ravel())[1:])
+    mask = labels == component
+    ys, xs = np.where(mask)
+    edge = []
+    for y in range(int(ys.min()), int(ys.max())-3):
+        x = np.flatnonzero(mask[y])
+        if len(x) > 3:
+            edge.extend([(x[0]-.4, y), (x[-1]+.4, y)])
+    edge = np.asarray(edge)
+    return least_squares(
+        lambda p: np.hypot(edge[:, 0]-p[0], edge[:, 1]-p[1])-p[2],
+        [xs.mean(), ys.mean(), (xs.max()-xs.min())/2], loss='soft_l1'
+    ).x
+
+
+def restored_background(src, yy, xx):
+    r, g, b = np.moveaxis(src, -1, 0)
+    # Bright exposed tabletop samples, discounting the orange sphere and the
+    # desaturated top face of the wooden block. Median selection avoids codec
+    # extrema while keeping the original table's fine grain.
+    expected_rg = np.clip(27+.23*(yy-50), 25, 46)
+    expected_gb = np.clip(25+.11*(yy-60), 24, 38)
+    score = (g - 2*np.maximum(0, expected_rg-(r-g))
+             - 2*np.maximum(0, expected_gb-(g-b)))
+    score[((r-g) > 60) & ((g-b) > 10)] = -1000
+    order = np.argsort(score, axis=0)[-16:]
+    clean = np.median(np.take_along_axis(src, order[..., None], axis=0), axis=0)
+    clean_left = np.median(src[40:80], axis=0)
+    bg = src[0].copy()
+
+    # Initial ball, its contact shadow, and its diffuse shadow on the table.
+    ball_region = np.exp(-(((xx-23)/29)**8 + ((yy-72)/35)**8))
+    ball_region[xx < 42] = np.maximum(
+        ball_region[xx < 42],
+        np.clip((yy[xx < 42]-38)/5, 0, 1)*np.clip((109-yy[xx < 42])/9, 0, 1)
+    )
+    ball_region = gaussian_filter(ball_region, .7)
+    bg = bg*(1-ball_region[..., None]) + clean_left*ball_region[..., None]
+
+    # Restore the block's volume and the full long soft floor shadow. The
+    # floor immediately around it is also exposed later in the source.
+    core = ((xx >= 106) & (xx <= 176) & (yy >= 53) & (yy <= 107))
+    # Smooth expansion around the opaque object, not a hard rectangle.
+    distance = distance_transform_edt(~core)
+    core_weight = np.clip((5-distance)/4, 0, 1)
+    shadow_weight = (np.clip((xx-99)/9, 0, 1) * np.clip((186-xx)/12, 0, 1)
+                     * np.clip((yy-100)/6, 0, 1) * np.clip((144-yy)/9, 0, 1))
+    weight = np.maximum(core_weight, shadow_weight)
+    bg = bg*(1-weight[..., None]) + clean*weight[..., None]
+    return bg.astype(np.float32)
+
+
+def source_shadow(src, yy, xx, geo):
+    t = 7
+    cx, cy, radius = geo[t]
+    # This frame precedes contact and provides the whole independent ball
+    # shadow, uncontaminated by the block's opaque silhouette.
+    table = np.median(src[40:80], axis=0)
+    transmission = np.clip(src[t]/np.maximum(table, 1), .22, 1.)
+    u, v = (xx-cx)/radius, (yy-cy)/radius
+    d = np.hypot(u, v)
+    envelope = np.clip((1.18-((u-.4)/1.8)**2-((v-1.85)/2.05)**2)/.23, 0, 1)
+    envelope *= np.clip((v-.32)/.20, 0, 1)
+    transmission = 1+(transmission-1)*envelope[..., None]
+    # The hidden part of the shadow lies behind the sphere; a smooth contact
+    # term fills it so no light fringe appears at the sphere's lower contour.
+    hidden = d < 1.035
+    contact = np.exp(-.5*((u-.40)/.58)**2-.5*((v-1.08)/.33)**2)
+    diffuse = np.exp(-.5*((u-.25)/.90)**2-.5*((v-1.65)/1.05)**2)
+    fill = 1-contact[..., None]*np.array([.52, .54, .48])-diffuse[..., None]*np.array([.10, .11, .13])
+    transmission[hidden] = fill[hidden]
+    transmission = gaussian_filter(transmission, sigma=(.45, .45, 0))
+    transmission[np.max(1-transmission, axis=-1) < .012] = 1
+    return transmission.astype(np.float32)
+
+
+def sphere_sprite(src, table, geometry, yy, xx, index):
+    cx, cy, radius = geometry[index]
+    d = np.hypot(xx-cx, yy-cy)
+    interior = d < radius-1.35
+    nearest = distance_transform_edt(~interior, return_distances=False, return_indices=True)
+    inner_color = src[index][nearest[0], nearest[1]]
+    # Estimate the source antialias coverage from the color separation from
+    # the tabletop; preserve RGB lighting and fine surface texture inside.
+    chroma = src[index, ..., 0]-src[index, ..., 1]
+    bg_chroma = table[..., 0]-table[..., 1]
+    fg_chroma = inner_color[..., 0]-inner_color[..., 1]
+    alpha = np.clip((chroma-bg_chroma)/np.maximum(fg_chroma-bg_chroma, 18), 0, 1)
+    alpha[d < radius-1.] = 1
+    alpha[d > radius+1.1] = 0
+    # Unmix the anti-aliased source edge before moving the object.
+    color = np.clip((src[index]-(1-alpha[..., None])*table)/np.maximum(alpha[..., None], .04), 0, 255)
+    color[alpha < .04] = inner_color[alpha < .04]
+    return (color*alpha[..., None]).astype(np.float32), alpha.astype(np.float32)
+
+
+def remap(array, mx, my, border):
+    return cv2.remap(array, mx.astype(np.float32), my.astype(np.float32),
+                     cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT,
+                     borderValue=border)
+
+
+def generate():
+    src = np.load(ROOT/'source_pixels.npz')['rgb'].astype(np.float32)
+    yy, xx = np.indices(src.shape[1:3], dtype=np.float32)
+    geometry = np.array([sphere_geometry(src[t]) for t in range(11)])
+    times = np.arange(1, 11)
+    # Perspective-correct screen displacement of uniform rolling on the
+    # tabletop. No impact impulse or block-induced slowdown is retained.
+    fit = least_squares(lambda p: (p[0]+p[1]*times)/(1-p[2]*times)-geometry[1:, 0],
+                        [14.7, 7., .001]).x
+    y_slope = np.polyfit(geometry[1:, 0], geometry[1:, 1], 1)[0]
+    r_slope = np.polyfit(geometry[1:, 0], geometry[1:, 2], 1)[0]
+    bg = restored_background(src, yy, xx)
+    shadow = source_shadow(src, yy, xx, geometry)
+    table = np.median(src[40:80], axis=0)
+    # Early-frame sprites retain the exact pre-contact color changes. Frame
+    # ten is the last wholly unoccluded sphere before the collision.
+    sprites = [sphere_sprite(src, table, geometry, yy, xx, t) for t in range(11)]
+    frames = []
+    for t in range(96):
+        if t <= 10:
+            cx, cy, radius = geometry[t]
+            reference = t
+        else:
+            rawx = (fit[0]+fit[1]*t)/(1-fit[2]*t)
+            raw10 = (fit[0]+fit[1]*10)/(1-fit[2]*10)
+            cx = geometry[10, 0]+rawx-raw10
+            cy = geometry[10, 1]+y_slope*(cx-geometry[10, 0])
+            radius = geometry[10, 2]+r_slope*(cx-geometry[10, 0])
+            reference = 10
+        if cx-radius > 260:
+            frames.append(np.rint(bg).clip(0, 255).astype(np.uint8))
+            continue
+        scx, scy, sr = geometry[7]
+        mx = scx+(xx-cx)*sr/radius
+        my = scy+(yy-cy)*sr/radius
+        trans = remap(shadow, mx, my, (1., 1., 1.))
+        canvas = bg*trans
+        rcx, rcy, rr = geometry[reference]
+        mx = rcx+(xx-cx)*rr/radius
+        my = rcy+(yy-cy)*rr/radius
+        rgb, a = sprites[reference]
+        moved_rgb = remap(rgb, mx, my, (0., 0., 0.))
+        moved_alpha = remap(a, mx, my, 0.)
+        canvas = canvas*(1-moved_alpha[..., None])+moved_rgb
+        frames.append(np.rint(canvas).clip(0, 255).astype(np.uint8))
+    frames = np.asarray(frames, dtype=np.uint8)
+    artifact_io.save(frames, ROOT)
+    print('Saved', frames.shape, 'RGB counterfactual frames at 24 fps.')
+    print('Pre-contact fitted geometry:', np.round(geometry[[0, 7, 10]], 3))
+    print('Uniform-motion projection:', fit)
+
+
+if __name__ == '__main__':
+    generate()

\"\"\"Counterfactual removal of the block, using only the source RGB sequence.

The table is recovered from pixels exposed elsewhere on the source timeline.
The free rolling trajectory is fitted before contact; source sphere appearance
and its soft cast shadow are then reprojected along that trajectory.
\"\"\"
from pathlib import Path
import numpy as np
import cv2
from scipy.ndimage import label, gaussian_filter, distance_transform_edt
from scipy.optimize import least_squares
import artifact_io

ROOT = Path(__file__).resolve().parent


def sphere_geometry(frame):
    r, g, b = np.moveaxis(frame, -1, 0)
    labels, count = label((r-g > 62) & (g-b > 14))
    component = 1 + np.argmax(np.bincount(labels.ravel())[1:])
    mask = labels == component
    ys, xs = np.where(mask)
    edge = []
    for y in range(int(ys.min()), int(ys.max())-3):
        x = np.flatnonzero(mask[y])
        if len(x) > 3:
            edge.extend([(x[0]-.4, y), (x[-1]+.4, y)])
    edge = np.asarray(edge)
    return least_squares(
        lambda p: np.hypot(edge[:, 0]-p[0], edge[:, 1]-p[1])-p[2],
        [xs.mean(), ys.mean(), (xs.max()-xs.min())/2], loss='soft_l1'
    ).x


def restored_background(src, yy, xx):
    r, g, b = np.moveaxis(src, -1, 0)
    # Bright exposed tabletop samples, discounting the orange sphere and the
    # desaturated top face of the wooden block. Median selection avoids codec
    # extrema while keeping the original table's fine grain.
    expected_rg = np.clip(27+.23*(yy-50), 25, 46)
    expected_gb = np.clip(25+.11*(yy-60), 24, 38)
    score = (g - 2*np.maximum(0, expected_rg-(r-g))
             - 2*np.maximum(0, expected_gb-(g-b)))
    score[((r-g) > 60) & ((g-b) > 10)] = -1000
    order = np.argsort(score, axis=0)[-16:]
    clean = np.median(np.take_along_axis(src, order[..., None], axis=0), axis=0)
    clean_left = np.median(src[40:80], axis=0)
    bg = src[0].copy()

    # Initial ball, its contact shadow, and its diffuse shadow on the table.
    ball_region = np.exp(-(((xx-23)/29)**8 + ((yy-72)/35)**8))
    ball_region[xx < 42] = np.maximum(
        ball_region[xx < 42],
        np.clip((yy[xx < 42]-38)/5, 0, 1)*np.clip((109-yy[xx < 42])/9, 0, 1)
    )
    ball_region = gaussian_filter(ball_region, .7)
    bg = bg*(1-ball_region[..., None]) + clean_left*ball_region[..., None]

    # Restore the block's volume and the full long soft floor shadow. The
    # floor immediately around it is also exposed later in the source.
    core = ((xx >= 106) & (xx <= 176) & (yy >= 53) & (yy <= 107))
    # Smooth expansion around the opaque object, not a hard rectangle.
    distance = distance_transform_edt(~core)
    core_weight = np.clip((5-distance)/4, 0, 1)
    shadow_weight = (np.clip((xx-99)/9, 0, 1) * np.clip((186-xx)/12, 0, 1)
                     * np.clip((yy-100)/6, 0, 1) * np.clip((144-yy)/9, 0, 1))
    weight = np.maximum(core_weight, shadow_weight)
    bg = bg*(1-weight[..., None]) + clean*weight[..., None]
    return bg.astype(np.float32)


def source_shadow(src, yy, xx, geo):
    t = 7
    cx, cy, radius = geo[t]
    # This frame precedes contact and provides the whole independent ball
    # shadow, uncontaminated by the block's opaque silhouette.
    table = np.median(src[40:80], axis=0)
    transmission = np.clip(src[t]/np.maximum(table, 1), .22, 1.)
    u, v = (xx-cx)/radius, (yy-cy)/radius
    d = np.hypot(u, v)
    envelope = np.clip((1.18-((u-.4)/1.8)**2-((v-1.85)/2.05)**2)/.23, 0, 1)
    envelope *= np.clip((v-.32)/.20, 0, 1)
    transmission = 1+(transmission-1)*envelope[..., None]
    # The hidden part of the shadow lies behind the sphere; a smooth contact
    # term fills it so no light fringe appears at the sphere's lower contour.
    hidden = d < 1.035
    contact = np.exp(-.5*((u-.40)/.58)**2-.5*((v-1.08)/.33)**2)
    diffuse = np.exp(-.5*((u-.25)/.90)**2-.5*((v-1.65)/1.05)**2)
    fill = 1-contact[..., None]*np.array([.52, .54, .48])-diffuse[..., None]*np.array([.10, .11, .13])
    transmission[hidden] = fill[hidden]
    transmission = gaussian_filter(transmission, sigma=(.45, .45, 0))
    transmission[np.max(1-transmission, axis=-1) < .012] = 1
    return transmission.astype(np.float32)


def sphere_sprite(src, table, geometry, yy, xx, index):
    cx, cy, radius = geometry[index]
    d = np.hypot(xx-cx, yy-cy)
    interior = d < radius-1.35
    nearest = distance_transform_edt(~interior, return_distances=False, return_indices=True)
    inner_color = src[index][nearest[0], nearest[1]]
    # Estimate the source antialias coverage from the color separation from
    # the tabletop; preserve RGB lighting and fine surface texture inside.
    chroma = src[index, ..., 0]-src[index, ..., 1]
    bg_chroma = table[..., 0]-table[..., 1]
    fg_chroma = inner_color[..., 0]-inner_color[..., 1]
    alpha = np.clip((chroma-bg_chroma)/np.maximum(fg_chroma-bg_chroma, 18), 0, 1)
    alpha[d < radius-1.] = 1
    alpha[d > radius+1.1] = 0
    # Unmix the anti-aliased source edge before moving the object.
    color = np.clip((src[index]-(1-alpha[..., None])*table)/np.maximum(alpha[..., None], .04), 0, 255)
    color[alpha < .04] = inner_color[alpha < .04]
    return (color*alpha[..., None]).astype(np.float32), alpha.astype(np.float32)


def remap(array, mx, my, border):
    return cv2.remap(array, mx.astype(np.float32), my.astype(np.float32),
                     cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT,
                     borderValue=border)


def generate():
    src = np.load(ROOT/'source_pixels.npz')['rgb'].astype(np.float32)
    yy, xx = np.indices(src.shape[1:3], dtype=np.float32)
    geometry = np.array([sphere_geometry(src[t]) for t in range(11)])
    times = np.arange(1, 11)
    # Perspective-correct screen displacement of uniform rolling on the
    # tabletop. No impact impulse or block-induced slowdown is retained.
    fit = least_squares(lambda p: (p[0]+p[1]*times)/(1-p[2]*times)-geometry[1:, 0],
                        [14.7, 7., .001]).x
    y_slope = np.polyfit(geometry[1:, 0], geometry[1:, 1], 1)[0]
    r_slope = np.polyfit(geometry[1:, 0], geometry[1:, 2], 1)[0]
    bg = restored_background(src, yy, xx)
    shadow = source_shadow(src, yy, xx, geometry)
    table = np.median(src[40:80], axis=0)
    # Early-frame sprites retain the exact pre-contact color changes. Frame
    # ten is the last wholly unoccluded sphere before the collision.
    sprites = [sphere_sprite(src, table, geometry, yy, xx, t) for t in range(11)]
    frames = []
    for t in range(96):
        if t <= 10:
            cx, cy, radius = geometry[t]
            reference = t
        else:
            rawx = (fit[0]+fit[1]*t)/(1-fit[2]*t)
            raw10 = (fit[0]+fit[1]*10)/(1-fit[2]*10)
            cx = geometry[10, 0]+rawx-raw10
            cy = geometry[10, 1]+y_slope*(cx-geometry[10, 0])
            radius = geometry[10, 2]+r_slope*(cx-geometry[10, 0])
            reference = 10
        if cx-radius > 260:
            frames.append(np.rint(bg).clip(0, 255).astype(np.uint8))
            continue
        scx, scy, sr = geometry[7]
        mx = scx+(xx-cx)*sr/radius
        my = scy+(yy-cy)*sr/radius
        trans = remap(shadow, mx, my, (1., 1., 1.))
        canvas = bg*trans
        rcx, rcy, rr = geometry[reference]
        mx = rcx+(xx-cx)*rr/radius
        my = rcy+(yy-cy)*rr/radius
        rgb, a = sprites[reference]
        moved_rgb = remap(rgb, mx, my, (0., 0., 0.))
        moved_alpha = remap(a, mx, my, 0.)
        canvas = canvas*(1-moved_alpha[..., None])+moved_rgb
        frames.append(np.rint(canvas).clip(0, 255).astype(np.uint8))
    frames = np.asarray(frames, dtype=np.uint8)
    artifact_io.save(frames, ROOT)
    print('Saved', frames.shape, 'RGB counterfactual frames at 24 fps.')
    print('Pre-contact fitted geometry:', np.round(geometry[[0, 7, 10]], 3))
    print('Uniform-motion projection:', fit)


if __name__ == '__main__':
    generate()
