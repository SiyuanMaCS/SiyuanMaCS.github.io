+"""Counterfactual edit using source pixels and a source-derived rigid sprite.
+
+The white mallet is removed at t=0.  The blue-to-red collision is unaffected;
+the red mallet subsequently traverses the unoccupied surface and encounters
+the near rail rather than transferring its momentum to the white mallet.
+"""
+from pathlib import Path
+import cv2
+import numpy as np
+from PIL import Image
+from scipy.ndimage import gaussian_filter, binary_fill_holes
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+
+
+def read_full_frame(index):
+    cap = cv2.VideoCapture(str(ROOT / "source_video.mp4"))
+    cap.set(cv2.CAP_PROP_POS_FRAMES, index)
+    ok, image = cap.read()
+    cap.release()
+    if not ok:
+        raise RuntimeError(f"Unable to decode source frame {index}")
+    return cv2.cvtColor(image, cv2.COLOR_BGR2RGB)
+
+
+def downsample_float(image):
+    # Source RGB uses this same Lanczos resampling. Float channels preserve
+    # premultiplied alpha at the small object boundaries.
+    return np.stack([
+        np.asarray(Image.fromarray(image[..., k].astype(np.float32)).resize(
+            (256, 144), Image.Resampling.LANCZOS))
+        for k in range(image.shape[2])
+    ], axis=-1)
+
+
+def clean_initial_surface(source):
+    initial = source[0].astype(np.float32)
+    yy, xx = np.mgrid[72:97, 93:118]
+    u, v = (xx - 106) / 20.0, (yy - 83) / 20.0
+    basis = np.stack([u * 0 + 1, u, v, u*u, v*v, u*v], -1)
+    region = initial[72:97, 93:118]
+    valid = ((region[..., 2] - region[..., 0] > 9)
+             & (region[..., 2] - region[..., 1] > 23)
+             & (region.max(-1) < 254))
+    # Exclude the complete head, short handle, and contact shadow.
+    valid &= ~((xx >= 98) & (xx <= 114) & (yy >= 76) & (yy <= 95))
+    coeff = np.linalg.lstsq(basis[valid], region[valid], rcond=None)[0]
+    surface = basis @ coeff
+
+    # Use genuinely revealed source pixels where possible. The red mallet
+    # stops over part of the white head's old footprint, so that tiny overlap
+    # is reconstructed from the surrounding smooth surface RGB gradient.
+    revealed = np.median(source[55:96, 72:97, 93:118].astype(np.float32), 0)
+    reliable = np.max(np.abs(revealed - surface), -1) < 5.0
+    restored = np.where(reliable[..., None], revealed, surface)
+    mask = np.zeros(initial.shape[:2], np.float32)
+    mask[75:95, 98:115] = 1
+    # The source contains a faint, longer highlight/shadow at the handle tip.
+    mask[86:96, 102:112] = 1
+    mask = gaussian_filter(mask, 0.8)
+    mask[77:86, 100:113] = 1
+    mask[87:92, 104:109] = 1
+    background = initial.copy()
+    alpha = mask[72:97, 93:118, None]
+    background[72:97, 93:118] = region*(1-alpha) + restored*alpha
+    return background, mask
+
+
+def red_sprite():
+    full = read_full_frame(28).astype(np.float32)
+    x0, x1, y0, y1 = 535, 645, 345, 431
+    crop = full[y0:y1, x0:x1]
+    r, g, b = crop.transpose(2, 0, 1)
+    colored = ((r-g > 32) & (r-b > -17)).astype(np.uint8)
+    n, labels, stats, _ = cv2.connectedComponentsWithStats(colored)
+    main = 1 + np.argmax(stats[1:, cv2.CC_STAT_AREA])
+    body = binary_fill_holes(labels == main).astype(np.uint8)
+    body = cv2.morphologyEx(body, cv2.MORPH_CLOSE, np.ones((3, 3), np.uint8))
+    body = cv2.dilate(body, np.ones((3, 3), np.uint8))
+    # Subpixel antialiasing comes from source-resolution pixels, not a drawn
+    # approximation of the mallet. Fill color-selection holes in its top.
+    alpha = gaussian_filter(body.astype(np.float32), 0.65)
+    rgba = np.zeros((720, 1280, 4), np.float32)
+    rgba[y0:y1, x0:x1, :3] = crop * alpha[..., None]
+    rgba[y0:y1, x0:x1, 3] = alpha
+    return rgba
+
+
+def red_position(index):
+    elapsed = index - 28
+    # Fit incoming source motion; increasing projected speed is consistent
+    # with the observed motion down this slightly sloping surface.
+    if index <= 51:
+        x = 116.90 - 1.99*elapsed - 0.0105*elapsed*elapsed
+    else:
+        hit_x = 116.90 - 1.99*23 - 0.0105*23*23
+        # Low restitution at the rail and rapid frictional settling.
+        x = hit_x + 2.1 * (1 - np.exp(-(index-51)/6.0))
+    y = 76.30 + 0.366*(116.90-x)
+    return x, y
+
+
+def main():
+    source = np.load(ROOT / "source_pixels.npz")["rgb"]
+    background, white_mask = clean_initial_surface(source)
+    sprite = red_sprite()
+    result = source.astype(np.float32).copy()
+    # Preserve all other source events and all of the initial first collision.
+    for i in range(29):
+        result[i] = result[i]*(1-white_mask[..., None]) + background*white_mask[..., None]
+
+    corridor = np.zeros((144, 256), np.float32)
+    corridor[70:105, 55:126] = 1
+    corridor = gaussian_filter(corridor, 0.65)
+    # Its boundaries are outside the complete factual mallet/shadow paths.
+    # Copy only this narrow static-surface corridor, leaving the blue mallet,
+    # surround, camera, and ambient appearance on the original timeline.
+    for i in range(29, 96):
+        frame = result[i]*(1-corridor[..., None]) + background*corridor[..., None]
+        x, y = red_position(i)
+        scale = 1 + 0.00145*(116.90-x)
+        anchor_x, anchor_y = 116.90*5+2, 76.30*5+2
+        mat = np.array([
+            [scale, 0, x*5+2-scale*anchor_x],
+            [0, scale, y*5+2-scale*anchor_y],
+        ], np.float32)
+        warped = cv2.warpAffine(sprite, mat, (1280, 720),
+                                flags=cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)
+        layer = downsample_float(warped)
+        alpha = np.clip(layer[..., 3], 0, 1)
+        # A short, soft surface contact shadow follows the counterfactual
+        # body. The original source-colored underside is in the sprite.
+        shadow = cv2.warpAffine(alpha, np.array([[1, 0, .6], [0, 1, .7]], np.float32),
+                                (256, 144), flags=cv2.INTER_LINEAR)
+        shadow = gaussian_filter(shadow, 0.6)*0.10*(1-alpha)
+        frame *= (1-shadow[..., None])
+        frame += shadow[..., None]*np.array([105, 83, 139], np.float32)
+        result[i] = frame*(1-alpha[..., None]) + np.clip(layer[..., :3], 0, 255)
+
+    frames = np.clip(np.rint(result), 0, 255).astype(np.uint8)
+    assert frames.shape == (96, 144, 256, 3)
+    artifact_io.save(frames, ROOT)
+    print("Saved complete 96-frame edit at 256x144, 24 fps.")
+
+
+if __name__ == "__main__":
+    main()
