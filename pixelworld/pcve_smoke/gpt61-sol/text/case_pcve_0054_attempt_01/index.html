+"""Rebuild the edited source timeline with pixel-extracted curling stones.
+No external assets: the rink and stones are copied from the supplied video.
+Motion is a planar equal-mass disk model with dissipative contact.
+"""
+from pathlib import Path
+import cv2
+import numpy as np
+from PIL import Image
+from scipy.ndimage import label, binary_fill_holes, binary_dilation, gaussian_filter
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+def load_original():
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    images = []
+    while True:
+        ok, bgr = cap.read()
+        if not ok:
+            break
+        images.append(bgr[:, :, ::-1].copy())
+    cap.release()
+    return np.asarray(images)
+
+
+def cap_positions(original):
+    result = []
+    for f in original:
+        r, g, b = f.astype(float).transpose(2, 0, 1)
+        mr = (r-g > 95) & (r-b > 70)
+        mr[365:] = False  # exclude the rink's painted red ring
+        my = (g-b > 75) & (r-b > 65)
+        positions = []
+        for m in (mr, my):
+            y, x = np.where(m)
+            positions.append([x.mean(), y.mean()])
+        result.append(positions)
+    return np.asarray(result)
+
+
+def make_background(original, positions):
+    # The rink and camera are static. Only these two footprints are dirty in
+    # frame one, and are clean in frames 21--32. All other pixels are retained.
+    background = original[0].astype(np.float32)
+    clean = np.median(original[20:32].astype(np.float32), axis=0)
+    for x1,y1,x2,y2 in ((200,307,285,377),(1023,357,1117,432)):
+        background[y1:y2,x1:x2] = clean[y1:y2,x1:x2]
+    return background
+
+
+def sprite(original, background, region, cap_center):
+    x1,y1,x2,y2 = region
+    src = original[0,y1:y2,x1:x2].astype(np.float32)
+    bg = background[y1:y2,x1:x2]
+    difference = np.max(np.abs(src-bg), axis=2)
+    components,n = label(difference > 18)
+    sizes = np.bincount(components.ravel()); sizes[0] = 0
+    shape = binary_fill_holes(components == np.argmax(sizes))
+    shape = binary_dilation(shape, iterations=1)
+    alpha = gaussian_filter(shape.astype(np.float32), .65)
+    # Support includes the faint local contact shadow, but not the fixed paint.
+    alpha[alpha < .008] = 0
+    offset = np.asarray(cap_center) - [x1,y1]
+    return {'rgb':src, 'bg':bg, 'a':alpha, 'offset':offset}
+
+
+def blue_sprite(red):
+    blue = {k:np.array(v,copy=True) for k,v in red.items()}
+    rgb = blue['rgb']
+    r,g,b = rgb.transpose(2,0,1)
+    yy = np.indices(r.shape)[0]
+    # Preserve gray sidewall, surface illumination and specular highlights.
+    chroma = np.clip((r-g-12)/120,0,1) * np.clip((r-b-8)/105,0,1)
+    top = yy < blue['offset'][1]+9
+    pigment = chroma*top
+    rgb += pigment[...,None]*np.array([-143,52,146],np.float32)
+    blue['rgb'] = np.clip(rgb,0,255)
+    return blue
+
+
+def trajectories(positions):
+    caps = positions / 5.0
+    # World coordinate along the shooting line. The cap/footprint projection
+    # is affine here, as measured from the uncollided motion in the source.
+    red0, yellow0 = caps[0]
+    blue0 = red0 + .25*(yellow0-red0)
+    x = np.array([red0[0], blue0[0], yellow0[0]],dtype=float)
+    speedr = np.polyfit(np.arange(8,35),caps[8:35,0,0],1)[0]
+    speedy = np.polyfit(np.arange(8,35),caps[8:35,1,0],1)[0]
+    v = np.array([speedr,0.0,speedy])
+    restitution = .083
+    radius = 5.0
+    rolling = .0018
+    nsub = 80
+    dt = 1/nsub
+    paths = np.zeros((96,3,2),float)
+    events = []
+    # y offset is stable in the source; blue exactly interpolates the two
+    # initial centers so the requested quarter point is satisfied in frame 1.
+    slope = (yellow0[1]-red0[1])/(yellow0[0]-red0[0])
+    for frame in range(96):
+        if frame:
+            for sub in range(nsub):
+                # Match source motion precisely until each first contact.
+                v = np.sign(v)*np.maximum(np.abs(v)-rolling*dt,0)
+                x += v*dt
+                # Iterative contacts allow the three-stone sequence to settle
+                # without interpenetration or one stone passing through another.
+                for iteration in range(4):
+                    for i in range(2):
+                        j=i+1
+                        distance=x[j]-x[i]
+                        if distance < 2*radius:
+                            correction=(2*radius-distance)/2
+                            x[i]-=correction; x[j]+=correction
+                            if v[i] > v[j]:
+                                impulse=(1+restitution)*(v[i]-v[j])/2
+                                v[i]-=impulse;v[j]+=impulse
+                                if iteration==0:
+                                    events.append((round(frame-1+sub*dt,3),i,j))
+        paths[frame,:,0]=x
+        paths[frame,:,1]=red0[1]+slope*(x-red0[0])
+    paths[0]=[red0,blue0,yellow0]
+    return paths,events
+
+
+def composite(canvas, obj, center):
+    # High-resolution subpixel warping followed by the same Lanczos reduction
+    # used to create source_pixels.npz.
+    x,y = np.asarray(center)*5-obj['offset']
+    ix,iy=int(np.floor(x)),int(np.floor(y))
+    dx,dy=x-ix,y-iy
+    a=obj['a']
+    # The extracted pixels have already been antialiased onto the old rink.
+    # Background-aware premultiplication avoids transporting a pale rectangle.
+    premul = (obj['rgb']-obj['bg'])*a[...,None] + obj['bg']*a[...,None]
+    h,w=a.shape
+    transform=np.array([[1,0,dx+1],[0,1,dy+1]],np.float32)
+    ap=cv2.warpAffine(a,transform,(w+2,h+2),flags=cv2.INTER_LINEAR)
+    cp=cv2.warpAffine(premul,transform,(w+2,h+2),flags=cv2.INTER_LINEAR)
+    ix-=1;iy-=1
+    x1=max(ix,0);y1=max(iy,0);x2=min(ix+w+2,canvas.shape[1]);y2=min(iy+h+2,canvas.shape[0])
+    if x2<=x1 or y2<=y1:return
+    sx=slice(x1-ix,x2-ix);sy=slice(y1-iy,y2-iy)
+    canvas[y1:y2,x1:x2] = cp[sy,sx] + canvas[y1:y2,x1:x2]*(1-ap[sy,sx,None])
+
+
+def main():
+    source=load_original()
+    assert source.shape==(96,720,1280,3)
+    positions=cap_positions(source)
+    background=make_background(source,positions)
+    red=sprite(source,background,(200,307,285,377),positions[0,0])
+    yellow=sprite(source,background,(1023,357,1117,432),positions[0,1])
+    blue=blue_sprite(red)
+    paths,events=trajectories(positions)
+    sprites=[red,blue,yellow]
+    frames=[]
+    for t in range(96):
+        canvas=background.copy()
+        # Far-to-near drawing order follows the shallow perspective.
+        for i in np.argsort(paths[t,:,1]):
+            composite(canvas,sprites[i],paths[t,i])
+        frame=np.asarray(Image.fromarray(np.clip(canvas,0,255).round().astype(np.uint8)).resize((256,144),Image.Resampling.LANCZOS))
+        frames.append(frame)
+    frames=np.asarray(frames,dtype=np.uint8)
+    artifact_io.save(frames,ROOT)
+    np.savez_compressed(ROOT/'.tmp'/'motion.npz',centers=paths)
+    print('Saved',frames.shape,frames.dtype)
+    print('Initial centers:',paths[0].round(3))
+    print('Final centers:',paths[-1].round(3))
+    print('Contact events:',events[:15])
+
+if __name__=='__main__':
+    main()

\"\"\"Rebuild the edited source timeline with pixel-extracted curling stones.
No external assets: the rink and stones are copied from the supplied video.
Motion is a planar equal-mass disk model with dissipative contact.
\"\"\"
from pathlib import Path
import cv2
import numpy as np
from PIL import Image
from scipy.ndimage import label, binary_fill_holes, binary_dilation, gaussian_filter
import artifact_io

ROOT = Path(__file__).resolve().parent

def load_original():
    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
    images = []
    while True:
        ok, bgr = cap.read()
        if not ok:
            break
        images.append(bgr[:, :, ::-1].copy())
    cap.release()
    return np.asarray(images)


def cap_positions(original):
    result = []
    for f in original:
        r, g, b = f.astype(float).transpose(2, 0, 1)
        mr = (r-g > 95) & (r-b > 70)
        mr[365:] = False  # exclude the rink's painted red ring
        my = (g-b > 75) & (r-b > 65)
        positions = []
        for m in (mr, my):
            y, x = np.where(m)
            positions.append([x.mean(), y.mean()])
        result.append(positions)
    return np.asarray(result)


def make_background(original, positions):
    # The rink and camera are static. Only these two footprints are dirty in
    # frame one, and are clean in frames 21--32. All other pixels are retained.
    background = original[0].astype(np.float32)
    clean = np.median(original[20:32].astype(np.float32), axis=0)
    for x1,y1,x2,y2 in ((200,307,285,377),(1023,357,1117,432)):
        background[y1:y2,x1:x2] = clean[y1:y2,x1:x2]
    return background


def sprite(original, background, region, cap_center):
    x1,y1,x2,y2 = region
    src = original[0,y1:y2,x1:x2].astype(np.float32)
    bg = background[y1:y2,x1:x2]
    difference = np.max(np.abs(src-bg), axis=2)
    components,n = label(difference > 18)
    sizes = np.bincount(components.ravel()); sizes[0] = 0
    shape = binary_fill_holes(components == np.argmax(sizes))
    shape = binary_dilation(shape, iterations=1)
    alpha = gaussian_filter(shape.astype(np.float32), .65)
    # Support includes the faint local contact shadow, but not the fixed paint.
    alpha[alpha < .008] = 0
    offset = np.asarray(cap_center) - [x1,y1]
    return {'rgb':src, 'bg':bg, 'a':alpha, 'offset':offset}


def blue_sprite(red):
    blue = {k:np.array(v,copy=True) for k,v in red.items()}
    rgb = blue['rgb']
    r,g,b = rgb.transpose(2,0,1)
    yy = np.indices(r.shape)[0]
    # Preserve gray sidewall, surface illumination and specular highlights.
    chroma = np.clip((r-g-12)/120,0,1) * np.clip((r-b-8)/105,0,1)
    top = yy < blue['offset'][1]+9
    pigment = chroma*top
    rgb += pigment[...,None]*np.array([-143,52,146],np.float32)
    blue['rgb'] = np.clip(rgb,0,255)
    return blue


def trajectories(positions):
    caps = positions / 5.0
    # World coordinate along the shooting line. The cap/footprint projection
    # is affine here, as measured from the uncollided motion in the source.
    red0, yellow0 = caps[0]
    blue0 = red0 + .25*(yellow0-red0)
    x = np.array([red0[0], blue0[0], yellow0[0]],dtype=float)
    speedr = np.polyfit(np.arange(8,35),caps[8:35,0,0],1)[0]
    speedy = np.polyfit(np.arange(8,35),caps[8:35,1,0],1)[0]
    v = np.array([speedr,0.0,speedy])
    restitution = .083
    radius = 5.0
    rolling = .0018
    nsub = 80
    dt = 1/nsub
    paths = np.zeros((96,3,2),float)
    events = []
    # y offset is stable in the source; blue exactly interpolates the two
    # initial centers so the requested quarter point is satisfied in frame 1.
    slope = (yellow0[1]-red0[1])/(yellow0[0]-red0[0])
    for frame in range(96):
        if frame:
            for sub in range(nsub):
                # Match source motion precisely until each first contact.
                v = np.sign(v)*np.maximum(np.abs(v)-rolling*dt,0)
                x += v*dt
                # Iterative contacts allow the three-stone sequence to settle
                # without interpenetration or one stone passing through another.
                for iteration in range(4):
                    for i in range(2):
                        j=i+1
                        distance=x[j]-x[i]
                        if distance < 2*radius:
                            correction=(2*radius-distance)/2
                            x[i]-=correction; x[j]+=correction
                            if v[i] > v[j]:
                                impulse=(1+restitution)*(v[i]-v[j])/2
                                v[i]-=impulse;v[j]+=impulse
                                if iteration==0:
                                    events.append((round(frame-1+sub*dt,3),i,j))
        paths[frame,:,0]=x
        paths[frame,:,1]=red0[1]+slope*(x-red0[0])
    paths[0]=[red0,blue0,yellow0]
    return paths,events


def composite(canvas, obj, center):
    # High-resolution subpixel warping followed by the same Lanczos reduction
    # used to create source_pixels.npz.
    x,y = np.asarray(center)*5-obj['offset']
    ix,iy=int(np.floor(x)),int(np.floor(y))
    dx,dy=x-ix,y-iy
    a=obj['a']
    # The extracted pixels have already been antialiased onto the old rink.
    # Background-aware premultiplication avoids transporting a pale rectangle.
    premul = (obj['rgb']-obj['bg'])*a[...,None] + obj['bg']*a[...,None]
    h,w=a.shape
    transform=np.array([[1,0,dx+1],[0,1,dy+1]],np.float32)
    ap=cv2.warpAffine(a,transform,(w+2,h+2),flags=cv2.INTER_LINEAR)
    cp=cv2.warpAffine(premul,transform,(w+2,h+2),flags=cv2.INTER_LINEAR)
    ix-=1;iy-=1
    x1=max(ix,0);y1=max(iy,0);x2=min(ix+w+2,canvas.shape[1]);y2=min(iy+h+2,canvas.shape[0])
    if x2<=x1 or y2<=y1:return
    sx=slice(x1-ix,x2-ix);sy=slice(y1-iy,y2-iy)
    canvas[y1:y2,x1:x2] = cp[sy,sx] + canvas[y1:y2,x1:x2]*(1-ap[sy,sx,None])


def main():
    source=load_original()
    assert source.shape==(96,720,1280,3)
    positions=cap_positions(source)
    background=make_background(source,positions)
    red=sprite(source,background,(200,307,285,377),positions[0,0])
    yellow=sprite(source,background,(1023,357,1117,432),positions[0,1])
    blue=blue_sprite(red)
    paths,events=trajectories(positions)
    sprites=[red,blue,yellow]
    frames=[]
    for t in range(96):
        canvas=background.copy()
        # Far-to-near drawing order follows the shallow perspective.
        for i in np.argsort(paths[t,:,1]):
            composite(canvas,sprites[i],paths[t,i])
        frame=np.asarray(Image.fromarray(np.clip(canvas,0,255).round().astype(np.uint8)).resize((256,144),Image.Resampling.LANCZOS))
        frames.append(frame)
    frames=np.asarray(frames,dtype=np.uint8)
    artifact_io.save(frames,ROOT)
    np.savez_compressed(ROOT/'.tmp'/'motion.npz',centers=paths)
    print('Saved',frames.shape,frames.dtype)
    print('Initial centers:',paths[0].round(3))
    print('Final centers:',paths[-1].round(3))
    print('Contact events:',events[:15])

if __name__=='__main__':
    main()
