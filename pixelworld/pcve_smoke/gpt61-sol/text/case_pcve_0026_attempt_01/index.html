+"""Counterfactual volleyball roll, reconstructed from source RGB only.
+
+The ball's ground-plane trajectory is recovered using its perspective scale.
+Reducing velocity by .6 with the same resisting acceleration rescales time
+by .6 and travelled ground-plane distance by .36. The stationary scene is
+kept intact, and source sphere/shadow pixels are moved to the new trajectory.
+"""
+from pathlib import Path
+import numpy as np
+import cv2
+from PIL import Image
+from scipy.ndimage import label, binary_closing, binary_fill_holes, gaussian_filter
+from scipy.interpolate import PchipInterpolator
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+
+def resize_rgb(a):
+    return np.asarray(Image.fromarray(a).resize((256, 144), Image.Resampling.LANCZOS))
+
+
+def main():
+    original = np.load(ROOT / 'source_pixels.npz')['rgb']
+    capture = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    early = []
+    last = None
+    n = 0
+    while True:
+        ok, bgr = capture.read()
+        if not ok:
+            break
+        rgb = bgr[:, :, ::-1].copy()
+        if n < 20:
+            early.append(rgb)
+        last = rgb
+        n += 1
+    capture.release()
+    assert n == 96
+    clean = last.copy()
+    # The terminal volleyball footprint is exposed in the initial image.
+    clean[255:440, 490:655] = early[0][255:440, 490:655]
+    H, W = clean.shape[:2]
+    yy, xx = np.mgrid[:H, :W].astype(np.float32)
+    centers = []
+    radii = []
+    hulls = []
+    for i, rgb in enumerate(early):
+        r, g, b = rgb.astype(np.float32).transpose(2, 0, 1)
+        # Numerical color isolation; never relies on supplied object masks.
+        colored = ((r > 115) & (g > 90) & (b < g*.78)) | ((b > r*1.13) & (b > g*1.1) & (b > 55))
+        bright = (np.minimum(np.minimum(r, g), b) > 155) & (np.maximum(np.maximum(r, g), b)-np.minimum(np.minimum(r, g), b) < 78)
+        m = colored | bright
+        m[:280] = False
+        m[:, :340] = False
+        m[:, 630:] = False
+        lab, count = label(m)
+        sizes = np.bincount(lab.ravel())
+        sizes[0] = 0
+        m = lab == sizes.argmax()
+        m = binary_fill_holes(binary_closing(m, iterations=2))
+        contours, _ = cv2.findContours(m.astype(np.uint8), cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_NONE)
+        contour = max(contours, key=cv2.contourArea)
+        ellipse = cv2.fitEllipse(contour)
+        centers.append(np.array(ellipse[0])/5)
+        radii.append(min(ellipse[1])/10)
+        hull = cv2.convexHull(contour)
+        mask = np.zeros((H, W), np.uint8)
+        cv2.fillConvexPoly(mask, hull, 1)
+        mask = cv2.dilate(mask, np.ones((3,3), np.uint8))
+        hulls.append(gaussian_filter(mask.astype(np.float32), .65))
+    centers = np.asarray(centers)
+    radii = np.asarray(radii)
+    # Perspective scale is affine in image height on a horizontal ground plane.
+    scale_coeff = np.polyfit(centers[:, 1], radii, 1)
+    source_scale = np.polyval(scale_coeff, centers[:, 1])
+    # Ground displacement in radius-normalized camera coordinates.
+    world_x = (centers[:, 0]-centers[0, 0])/source_scale
+    world_y = (centers[:, 1]-centers[0, 1])/source_scale
+    # Remove subpixel segmentation jitter while retaining the measured friction curve.
+    direction = np.polyfit(world_x, world_y, 1)[0]
+    q = np.maximum.accumulate(world_x)
+    q[0] = 0
+    qcurve = PchipInterpolator(np.arange(20), q)
+    factor = .6
+    times = np.minimum(np.arange(96)/factor, 19)
+    qnew = factor**2 * qcurve(times)
+    # The original ball comes fully to rest at source frame 19.
+    stop = 19 * factor
+    newscale = source_scale[0] / (1-scale_coeff[0]*direction*qnew)
+    newcenters = np.stack([centers[0,0]+newscale*qnew,
+                           centers[0,1]+newscale*direction*qnew], axis=-1)
+
+    cleanf = clean.astype(np.float32)
+    local_clean = resize_rgb(clean)
+    outputs = original.copy()
+    # Numerical replacement region contains all original and edited ball/shadow pixels.
+    replacement = np.zeros((144,256), bool)
+    replacement[48:144, 46:133] = True
+    for i in range(1,96):
+        # Preserve the initial spin; its stopping time follows the reduced roll.
+        pose = min(i, int(np.ceil(stop)))
+        rgb = early[pose].astype(np.float32)
+        alpha = hulls[pose]
+        center = centers[pose]*5
+        rx = radii[pose]*5
+        # Soft ground shadow from the original rendering, separate from ball color.
+        ratio = rgb / np.maximum(cleanf, 1) - 1
+        lumdelta = (rgb-cleanf).mean(-1)
+        region = np.exp(-(((xx-(center[0]-rx*.85))/(rx*2.0))**6 +
+                          ((yy-(center[1]+rx*.35))/(rx*1.7))**6))
+        # Compression variation is not a cast shadow.
+        shadow_weight = np.clip((-lumdelta-1)/4, 0, 1)*region*(1-alpha)
+        shadow = np.minimum(ratio,0)*shadow_weight[:,:,None]
+        ball = rgb*alpha[:,:,None]
+        s = float(newscale[i]/source_scale[pose])
+        target = newcenters[i]*5
+        affine = np.array([[s,0,target[0]-s*center[0]],
+                           [0,s,target[1]-s*center[1]]],np.float32)
+        # Limit computation to the affected rectangle while maintaining 5x sampling.
+        ox, oy, ow, oh = 230, 240, 435, 480
+        affine[:,2] -= [ox,oy]
+        wa = cv2.warpAffine(alpha, affine, (ow,oh), flags=cv2.INTER_LINEAR)
+        wb = cv2.warpAffine(ball, affine, (ow,oh), flags=cv2.INTER_LINEAR)
+        ws = cv2.warpAffine(shadow, affine, (ow,oh), flags=cv2.INTER_LINEAR)
+        base = cleanf[oy:oy+oh,ox:ox+ow]
+        patch = base*(1+ws)*(1-wa[:,:,None])+wb
+        rendered = clean.copy()
+        rendered[oy:oy+oh,ox:ox+ow] = np.clip(np.rint(patch),0,255).astype(np.uint8)
+        small = resize_rgb(rendered)
+        outputs[i,replacement] = small[replacement]
+    outputs[0] = original[0]
+    artifact_io.save(outputs, ROOT)
+    np.savez_compressed(ROOT / 'motion_diagnostics.npz', original_centers=centers,
+                        edited_centers=newcenters, original_world_displacement=q,
+                        edited_world_displacement=qnew, stopping_frame=stop)
+    print('Saved 96 frames, 256x144, 24 fps.')
+    print('Initial center:',newcenters[0], 'Final center:',newcenters[-1])
+    print('Ground distance ratio:',qnew[-1]/q[-1], 'Stopping frame:',stop)
+
+
+if __name__ == '__main__':
+    main()

\"\"\"Counterfactual volleyball roll, reconstructed from source RGB only.

The ball's ground-plane trajectory is recovered using its perspective scale.
Reducing velocity by .6 with the same resisting acceleration rescales time
by .6 and travelled ground-plane distance by .36. The stationary scene is
kept intact, and source sphere/shadow pixels are moved to the new trajectory.
\"\"\"
from pathlib import Path
import numpy as np
import cv2
from PIL import Image
from scipy.ndimage import label, binary_closing, binary_fill_holes, gaussian_filter
from scipy.interpolate import PchipInterpolator
import artifact_io

ROOT = Path(__file__).resolve().parent


def resize_rgb(a):
    return np.asarray(Image.fromarray(a).resize((256, 144), Image.Resampling.LANCZOS))


def main():
    original = np.load(ROOT / 'source_pixels.npz')['rgb']
    capture = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
    early = []
    last = None
    n = 0
    while True:
        ok, bgr = capture.read()
        if not ok:
            break
        rgb = bgr[:, :, ::-1].copy()
        if n < 20:
            early.append(rgb)
        last = rgb
        n += 1
    capture.release()
    assert n == 96
    clean = last.copy()
    # The terminal volleyball footprint is exposed in the initial image.
    clean[255:440, 490:655] = early[0][255:440, 490:655]
    H, W = clean.shape[:2]
    yy, xx = np.mgrid[:H, :W].astype(np.float32)
    centers = []
    radii = []
    hulls = []
    for i, rgb in enumerate(early):
        r, g, b = rgb.astype(np.float32).transpose(2, 0, 1)
        # Numerical color isolation; never relies on supplied object masks.
        colored = ((r > 115) & (g > 90) & (b < g*.78)) | ((b > r*1.13) & (b > g*1.1) & (b > 55))
        bright = (np.minimum(np.minimum(r, g), b) > 155) & (np.maximum(np.maximum(r, g), b)-np.minimum(np.minimum(r, g), b) < 78)
        m = colored | bright
        m[:280] = False
        m[:, :340] = False
        m[:, 630:] = False
        lab, count = label(m)
        sizes = np.bincount(lab.ravel())
        sizes[0] = 0
        m = lab == sizes.argmax()
        m = binary_fill_holes(binary_closing(m, iterations=2))
        contours, _ = cv2.findContours(m.astype(np.uint8), cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_NONE)
        contour = max(contours, key=cv2.contourArea)
        ellipse = cv2.fitEllipse(contour)
        centers.append(np.array(ellipse[0])/5)
        radii.append(min(ellipse[1])/10)
        hull = cv2.convexHull(contour)
        mask = np.zeros((H, W), np.uint8)
        cv2.fillConvexPoly(mask, hull, 1)
        mask = cv2.dilate(mask, np.ones((3,3), np.uint8))
        hulls.append(gaussian_filter(mask.astype(np.float32), .65))
    centers = np.asarray(centers)
    radii = np.asarray(radii)
    # Perspective scale is affine in image height on a horizontal ground plane.
    scale_coeff = np.polyfit(centers[:, 1], radii, 1)
    source_scale = np.polyval(scale_coeff, centers[:, 1])
    # Ground displacement in radius-normalized camera coordinates.
    world_x = (centers[:, 0]-centers[0, 0])/source_scale
    world_y = (centers[:, 1]-centers[0, 1])/source_scale
    # Remove subpixel segmentation jitter while retaining the measured friction curve.
    direction = np.polyfit(world_x, world_y, 1)[0]
    q = np.maximum.accumulate(world_x)
    q[0] = 0
    qcurve = PchipInterpolator(np.arange(20), q)
    factor = .6
    times = np.minimum(np.arange(96)/factor, 19)
    qnew = factor**2 * qcurve(times)
    # The original ball comes fully to rest at source frame 19.
    stop = 19 * factor
    newscale = source_scale[0] / (1-scale_coeff[0]*direction*qnew)
    newcenters = np.stack([centers[0,0]+newscale*qnew,
                           centers[0,1]+newscale*direction*qnew], axis=-1)

    cleanf = clean.astype(np.float32)
    local_clean = resize_rgb(clean)
    outputs = original.copy()
    # Numerical replacement region contains all original and edited ball/shadow pixels.
    replacement = np.zeros((144,256), bool)
    replacement[48:144, 46:133] = True
    for i in range(1,96):
        # Preserve the initial spin; its stopping time follows the reduced roll.
        pose = min(i, int(np.ceil(stop)))
        rgb = early[pose].astype(np.float32)
        alpha = hulls[pose]
        center = centers[pose]*5
        rx = radii[pose]*5
        # Soft ground shadow from the original rendering, separate from ball color.
        ratio = rgb / np.maximum(cleanf, 1) - 1
        lumdelta = (rgb-cleanf).mean(-1)
        region = np.exp(-(((xx-(center[0]-rx*.85))/(rx*2.0))**6 +
                          ((yy-(center[1]+rx*.35))/(rx*1.7))**6))
        # Compression variation is not a cast shadow.
        shadow_weight = np.clip((-lumdelta-1)/4, 0, 1)*region*(1-alpha)
        shadow = np.minimum(ratio,0)*shadow_weight[:,:,None]
        ball = rgb*alpha[:,:,None]
        s = float(newscale[i]/source_scale[pose])
        target = newcenters[i]*5
        affine = np.array([[s,0,target[0]-s*center[0]],
                           [0,s,target[1]-s*center[1]]],np.float32)
        # Limit computation to the affected rectangle while maintaining 5x sampling.
        ox, oy, ow, oh = 230, 240, 435, 480
        affine[:,2] -= [ox,oy]
        wa = cv2.warpAffine(alpha, affine, (ow,oh), flags=cv2.INTER_LINEAR)
        wb = cv2.warpAffine(ball, affine, (ow,oh), flags=cv2.INTER_LINEAR)
        ws = cv2.warpAffine(shadow, affine, (ow,oh), flags=cv2.INTER_LINEAR)
        base = cleanf[oy:oy+oh,ox:ox+ow]
        patch = base*(1+ws)*(1-wa[:,:,None])+wb
        rendered = clean.copy()
        rendered[oy:oy+oh,ox:ox+ow] = np.clip(np.rint(patch),0,255).astype(np.uint8)
        small = resize_rgb(rendered)
        outputs[i,replacement] = small[replacement]
    outputs[0] = original[0]
    artifact_io.save(outputs, ROOT)
    np.savez_compressed(ROOT / 'motion_diagnostics.npz', original_centers=centers,
                        edited_centers=newcenters, original_world_displacement=q,
                        edited_world_displacement=qnew, stopping_frame=stop)
    print('Saved 96 frames, 256x144, 24 fps.')
    print('Initial center:',newcenters[0], 'Final center:',newcenters[-1])
    print('Ground distance ratio:',qnew[-1]/q[-1], 'Stopping frame:',stop)


if __name__ == '__main__':
    main()
