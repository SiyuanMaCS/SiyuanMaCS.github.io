+"""Render the high-friction counterfactual from the supplied RGB timeline.
+
+The camera is fixed.  The blue mallet is the first moving body, and its
+collision with the red mallet initiates all subsequent puck motion.  An
+increased Coulomb friction force dissipates the initial blue sliding velocity
+before contact.  Consequently neither the red mallet nor the puck is launched.
+Only the blue mallet and its existing contact shadow need to be reprojected.
+No external assets, learned models, or simulation/rendering engines are used.
+"""
+
+from pathlib import Path
+
+import numpy as np
+from scipy.ndimage import shift
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def generate():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3)
+    initial = source[0].astype(np.float64)
+
+    # By this interval the mallet has completely vacated its initial location.
+    # Median RGB suppresses the minor MP4 quantization fluctuations without
+    # modifying the remaining scene.  The plate is used only in this tiny area.
+    exposed_plate = np.median(source[35:75].astype(np.float64), axis=0)
+    y0, y1, x0, x1 = 50, 70, 159, 178
+    sy, sx = np.mgrid[y0:y1, x0:x1]
+    feather = np.minimum.reduce([
+        (sx - x0) / 2.0,
+        (x1 - 1 - sx) / 2.0,
+        (sy - y0) / 2.0,
+        (y1 - 1 - sy) / 2.0,
+        np.ones_like(sx, dtype=float),
+    ])
+    feather = np.clip(feather, 0.0, 1.0)
+    layer = np.zeros_like(initial)
+    layer[y0:y1, x0:x1] = (
+        initial[y0:y1, x0:x1] - exposed_plate[y0:y1, x0:x1]
+    ) * feather[..., None]
+    clean_scene = initial - layer
+
+    # Initial projected sliding velocity, measured from the initial four
+    # factual frames.  Direction is preserved under uniform isotropic friction.
+    velocity = np.array([-33.0, 12.75])  # image pixels/second
+    speed = np.linalg.norm(velocity)
+    direction = velocity / speed
+    original_deceleration = speed / (25.0 * 0.40)
+    new_deceleration = original_deceleration * 25.0
+    stop_time = speed / new_deceleration
+
+    frames = np.empty_like(source)
+    for index in range(len(frames)):
+        t = index / 24.0
+        moving_time = min(t, stop_time)
+        distance = (speed * moving_time
+                    - 0.5 * new_deceleration * moving_time ** 2)
+        dx, dy = direction * distance
+        if index == 0:
+            # Changing a material parameter introduces no position discontinuity.
+            frames[index] = source[index]
+            continue
+        # Signed RGB reprojection carries both the original blue shading and
+        # its soft contact shadow. Cubic interpolation avoids double-resampling
+        # and keeps the very small original mallet visually sharp.
+        translated = shift(layer, (dy, dx, 0), order=3,
+                           mode="constant", cval=0.0, prefilter=True)
+        frames[index] = np.clip(np.rint(clean_scene + translated),
+                                0, 255).astype(np.uint8)
+
+    artifact_io.save(frames, CASE_DIR)
+    print(f"Saved {len(frames)} frames, 256x144, 24 fps.")
+    print(f"Blue mallet stops at {stop_time:.3f}s; displacement "
+          f"({direction[0] * speed * stop_time / 2:.2f}, "
+          f"{direction[1] * speed * stop_time / 2:.2f}) px.")
+    return frames
+
+
+if __name__ == "__main__":
+    generate()
