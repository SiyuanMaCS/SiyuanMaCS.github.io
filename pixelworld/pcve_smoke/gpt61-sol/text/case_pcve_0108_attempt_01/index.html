+"""Counterfactual initial-velocity edit using measured motion and RGB layers.
+
+Only the original local RGB sequence is used.  A perspective-aware constant
+friction fit to the large marble's pre- and post-impact motion estimates its
+free path.  Reducing initial velocity reduces stopping distance quadratically;
+no impact occurs in the edited sequence.  Source RGB foreground, contact shadow
+and reflection layers are motion-compensated to the shortened trajectory.
+"""
+from pathlib import Path
+import json
+import numpy as np
+from scipy.ndimage import label, map_coordinates
+from scipy.optimize import minimize_scalar
+import artifact_io
+
+CASE = Path(__file__).resolve().parent
+
+
+def track_big(frames):
+    yy, xx = np.indices(frames.shape[1:3])
+    positions = []
+    for frame in frames:
+        r, g, b = frame.astype(float).transpose(2, 0, 1)
+        gold = ((r > 150) & (g > 105) & (g > .63*r) &
+                (b < .86*g) & (yy > 52) & (yy < 81))
+        components, n = label(gold)
+        counts = np.bincount(components.ravel())
+        counts[0] = 0
+        component = counts.argmax()
+        y, x = np.where(components == component)
+        positions.append([x.mean(), y.mean()])
+    return np.asarray(positions)
+
+
+def fit_motion(x):
+    # q is a world-distance coordinate up to a constant scale; the pinhole
+    # map is x = x0 - q/(1-perspective*q).
+    x0 = x[0]
+    early = np.arange(1, 14)
+    late = np.arange(15, 33)
+    ts = np.r_[early, late].astype(float)
+    design = np.zeros((len(ts), 5))
+    design[:, 0] = ts**2
+    design[:len(early), 1] = early
+    design[:len(early), 2] = 1
+    design[len(early):, 3] = late
+    design[len(early):, 4] = 1
+
+    def fit(p):
+        distance = x0-x
+        q = distance/(1+p*distance)
+        values = q[ts.astype(int)]
+        coefficients = np.linalg.lstsq(design, values, rcond=None)[0]
+        error = np.mean((design@coefficients-values)**2)
+        return error, coefficients
+
+    result = minimize_scalar(lambda p: fit(p)[0], bounds=(0, .004),
+                             method='bounded', options={'xatol': 1e-10})
+    p = result.x
+    error, coefficients = fit(p)
+    quadratic, slope, intercept = coefficients[:3]
+    roots = np.roots([quadratic, slope, intercept])
+    origin = float(min(roots, key=abs))
+    velocity = slope + 2*quadratic*origin
+    deceleration = -2*quadratic
+    return p, origin, velocity, deceleration, error
+
+
+def soft_interval(values, lower, upper, feather=2.):
+    return (np.clip((values-lower)/feather, 0, 1)*
+            np.clip((upper-values)/feather, 0, 1))
+
+
+def generate():
+    source = np.load(CASE/'source_pixels.npz')['rgb']
+    spec = json.loads((CASE/'task_spec.json').read_text())
+    positions = track_big(source)
+    p, origin, v, a, error = fit_motion(positions[:, 0])
+    # Median of early frames retains the unmoved small marble and static scene,
+    # while removing the moving marble and its moving reflection/shadow.
+    background = np.median(source[:13].astype(float), axis=0)
+    yy, xx = np.indices(source.shape[1:3], dtype=float)
+    layers = []
+    for j in range(13):
+        cx, cy = positions[j]
+        u, w = xx-cx, yy-cy
+        # Feather outside (not inside) the actual source sphere silhouette.
+        sphere = soft_interval(u, -13.5, 13.5)*soft_interval(w, -11., 12.)
+        reflection = (soft_interval(u, -15., 15., 3.)*
+                      soft_interval(w, 7., 23., 3.))
+        shadow = (soft_interval(u, -38., 6., 4.)*
+                  soft_interval(w, 5., 13., 2.))
+        support = np.maximum.reduce([sphere, reflection, shadow])
+        layers.append((source[j].astype(float)-background)*support[..., None])
+
+    def shifted_layer(j, cx, cy):
+        sx = xx-(cx-positions[j, 0])
+        sy = yy-(cy-positions[j, 1])
+        return np.stack([map_coordinates(layers[j][..., c], [sy, sx],
+                                         order=1, mode='constant', cval=0,
+                                         prefilter=False)
+                         for c in range(3)], axis=-1)
+
+    output = []
+    motion = []
+    stop_time = .7*v/a
+    free_source_x = positions[:13, 0]
+    for frame_index in range(spec['frames']):
+        elapsed = max(0., frame_index-origin) if frame_index else 0.
+        elapsed = min(elapsed, stop_time)
+        q = .7*v*elapsed-.5*a*elapsed**2
+        cx = positions[0, 0]-q/(1-p*q)
+        # Select the identical physical location on the factual free-motion
+        # path.  Register adjacent RGB samples before blending, avoiding ghost
+        # silhouettes, and stop both translation and rolling when at rest.
+        jfloat = float(np.interp(cx, free_source_x[::-1], np.arange(13)[::-1]))
+        j = min(int(jfloat), 11)
+        blend = jfloat-j
+        cy = float(np.interp(jfloat, np.arange(13), positions[:13, 1]))
+        layer = ((1-blend)*shifted_layer(j, cx, cy) +
+                 blend*shifted_layer(j+1, cx, cy))
+        rendered = np.clip(np.rint(background+layer), 0, 255).astype(np.uint8)
+        if frame_index == 0:
+            rendered = source[0].copy()
+        output.append(rendered)
+        motion.append([cx, cy, jfloat])
+
+    frames = np.asarray(output, dtype=np.uint8)
+    # Save the complete candidate before any subsequent output checks.
+    artifact_io.save(frames, CASE)
+    details = {'initial_speed_scale': .7,
+               'perspective_coefficient': p,
+               'source_initial_speed': v,
+               'deceleration_per_frame': a,
+               'fit_rms_world_pixels': float(np.sqrt(error)),
+               'stop_time_seconds': (stop_time+origin)/spec['fps'],
+               'final_center_rgb_centroid': motion[-1][:2],
+               'small_marble_affected': False,
+               'motion': motion}
+    (CASE/'motion_report.json').write_text(json.dumps(details, indent=2))
+    print(json.dumps({k: val for k, val in details.items() if k != 'motion'}, indent=2))
+    print('Saved', frames.shape, frames.dtype)
+
+
+if __name__ == '__main__':
+    generate()

                (b < .86*g) & (yy > 52) & (yy < 81))
        components, n = label(gold)
        counts = np.bincount(components.ravel())
        counts[0] = 0
        component = counts.argmax()
        y, x = np.where(components == component)
        positions.append([x.mean(), y.mean()])
    return np.asarray(positions)


def fit_motion(x):
    # q is a world-distance coordinate up to a constant scale; the pinhole
    # map is x = x0 - q/(1-perspective*q).
    x0 = x[0]
    early = np.arange(1, 14)
    late = np.arange(15, 33)
    ts = np.r_[early, late].astype(float)
    design = np.zeros((len(ts), 5))
    design[:, 0] = ts**2
    design[:len(early), 1] = early
    design[:len(early), 2] = 1
    design[len(early):, 3] = late
    design[len(early):, 4] = 1

    def fit(p):
        distance = x0-x
        q = distance/(1+p*distance)
        values = q[ts.astype(int)]
        coefficients = np.linalg.lstsq(design, values, rcond=None)[0]
        error = np.mean((design@coefficients-values)**2)
        return error, coefficients

    result = minimize_scalar(lambda p: fit(p)[0], bounds=(0, .004),
                             method='bounded', options={'xatol': 1e-10})
    p = result.x
    error, coefficients = fit(p)
    quadratic, slope, intercept = coefficients[:3]
    roots = np.roots([quadratic, slope, intercept])
    origin = float(min(roots, key=abs))
    velocity = slope + 2*quadratic*origin
    deceleration = -2*quadratic
    return p, origin, velocity, deceleration, error


def soft_interval(values, lower, upper, feather=2.):
    return (np.clip((values-lower)/feather, 0, 1)*
            np.clip((upper-values)/feather, 0, 1))


def generate():
    source = np.load(CASE/'source_pixels.npz')['rgb']
    spec = json.loads((CASE/'task_spec.json').read_text())
    positions = track_big(source)
    p, origin, v, a, error = fit_motion(positions[:, 0])
    # Median of early frames retains the unmoved small marble and static scene,
    # while removing the moving marble and its moving reflection/shadow.
    background = np.median(source[:13].astype(float), axis=0)
    yy, xx = np.indices(source.shape[1:3], dtype=float)
    layers = []
    for j in range(13):
        cx, cy = positions[j]
        u, w = xx-cx, yy-cy
        # Feather outside (not inside) the actual source sphere silhouette.
        sphere = soft_interval(u, -13.5, 13.5)*soft_interval(w, -11., 12.)
        reflection = (soft_interval(u, -15., 15., 3.)*
                      soft_interval(w, 7., 23., 3.))
        shadow = (soft_interval(u, -38., 6., 4.)*
                  soft_interval(w, 5., 13., 2.))
        support = np.maximum.reduce([sphere, reflection, shadow])
        layers.append((source[j].astype(float)-background)*support[..., None])

    def shifted_layer(j, cx, cy):
        sx = xx-(cx-positions[j, 0])
        sy = yy-(cy-positions[j, 1])
        return np.stack([map_coordinates(layers[j][..., c], [sy, sx],
                                         order=1, mode='constant', cval=0,
                                         prefilter=False)
                         for c in range(3)], axis=-1)

    output = []
    motion = []
    stop_time = .7*v/a
    free_source_x = positions[:13, 0]
    for frame_index in range(spec['frames']):
        elapsed = max(0., frame_index-origin) if frame_index else 0.
        elapsed = min(elapsed, stop_time)
        q = .7*v*elapsed-.5*a*elapsed**2
        cx = positions[0, 0]-q/(1-p*q)
        # Select the identical physical location on the factual free-motion
        # path.  Register adjacent RGB samples before blending, avoiding ghost
        # silhouettes, and stop both translation and rolling when at rest.
        jfloat = float(np.interp(cx, free_source_x[::-1], np.arange(13)[::-1]))
        j = min(int(jfloat), 11)
        blend = jfloat-j
        cy = float(np.interp(jfloat, np.arange(13), positions[:13, 1]))
        layer = ((1-blend)*shifted_layer(j, cx, cy) +
                 blend*shifted_layer(j+1, cx, cy))
        rendered = np.clip(np.rint(background+layer), 0, 255).astype(np.uint8)
        if frame_index == 0:
            rendered = source[0].copy()
        output.append(rendered)
        motion.append([cx, cy, jfloat])

    frames = np.asarray(output, dtype=np.uint8)
    # Save the complete candidate before any subsequent output checks.
    artifact_io.save(frames, CASE)
    details = {'initial_speed_scale': .7,
               'perspective_coefficient': p,
               'source_initial_speed': v,
               'deceleration_per_frame': a,
               'fit_rms_world_pixels': float(np.sqrt(error)),
               'stop_time_seconds': (stop_time+origin)/spec['fps'],
               'final_center_rgb_centroid': motion[-1][:2],
               'small_marble_affected': False,
               'motion': motion}
    (CASE/'motion_report.json').write_text(json.dumps(details, indent=2))
    print(json.dumps({k: val for k, val in details.items() if k "'!= '"'motion'}, indent=2))
    print('Saved', frames.shape, frames.dtype)


if __name__ == '__main__':
    generate()
