+
+All geometry and textures are estimated from the supplied factual movie.  The
+camera is fixed.  The two dice have different ground-plane depths, and their
+vertical motions are independent; the red die occludes the white one in some
+factual frames without transferring a detectable lateral impulse to it.

+import numpy as np

+from scipy.ndimage import gaussian_filter, binary_dilation, label
+import artifact_io

\"\"\"Source-pixel reconstruction of the low-restitution red-die timeline.

All geometry and textures are estimated from the supplied factual movie.  The
camera is fixed.  The two dice have different ground-plane depths, and their
vertical motions are independent; the red die occludes the white one in some

import cv2
import numpy as np
from PIL import Image
from scipy.ndimage import gaussian_filter, binary_dilation, label
import artifact_io

ROOT = Path(__file__).resolve().parent
SCALE = 5
# Decode just the high-resolution source-pixel region needed for the edit.
X0, X1, Y1 = 100, 156, 108


def main():
    rgb = np.load(ROOT / 'source_pixels.npz')['rgb']
    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
    patches = []
    for t in range(96):
        ok, image = cap.read()
        if not ok:
            raise RuntimeError('Incomplete source video')
        image = cv2.cvtColor(image, cv2.COLOR_BGR2RGB)
        patches.append(image[:Y1*SCALE, X0*SCALE:X1*SCALE])
    cap.release()
    src = np.asarray(patches, dtype=np.uint8)
    H, W = src.shape[1:3]
    yy, xx = np.mgrid[:H, :W].astype(float)
    xx = xx/SCALE + X0
    yy /= SCALE
    floor = src[0].astype(float)
    # The last two seconds expose the empty falling trajectories.  The floor
    # at time zero is exposed and supplies the unoccluded checkerboard detail.
    background = np.median(src[32:], axis=0)
    background[yy >= 69] = floor[yy >= 69]

    def hull_mask(points, dilate=1):
        mask = np.zeros((H, W), np.uint8)
        if len(points) > 2:
            hull = cv2.convexHull(points.astype(np.int32))
            cv2.fillConvexPoly(mask, hull, 1)
        if dilate:
            mask = binary_dilation(mask, iterations=dilate).astype(np.uint8)
        return gaussian_filter(mask.astype(float), .65)

    def red_mask(t, cy):
        im = src[t].astype(float)
        r, g, b = im.transpose(2, 0, 1)
        chroma = (r-g) - (background[:,:,0]-background[:,:,1])
        sel = ((chroma > 20) & (r > 145) &
               (xx >= 118.0) & (xx < 129.6) &
               (yy > cy-8) & (yy < cy+8))
        labs, n = label(sel)
        if n:
            sizes = np.bincount(labs.ravel()); sizes[0] = 0
            sel = labs == np.argmax(sizes)
        y, x = np.where(sel)
        mask = hull_mask(np.stack([x, y], axis=1), 2)
        return mask, (y.min()/SCALE, y.max()/SCALE)

    def white_mask(t):
        im = src[t].astype(float)
        r,g,b = im.transpose(2,0,1)
        sel = ((r > 165) & (g > 165) & (b > 155) &
               (abs(r-g) < 18) & (abs(g-b) < 18) &
               (xx >= 124.8) & (xx < 140.6))
        if t >= 5:

    # Entire white-die textures at frames 6 and 15 are unoccluded.  These
    # restore portions that the factual red rebound incorrectly hid.
    wm6, wb6 = whites[6]
    wm15, wb15 = whites[15]
    wm27, wb27 = whites[27]
    rm27, rb27 = reds[27]
    rm15, _ = reds[15]

    def move_layer(t, mask, dy, dx=0, sy=1, cy=0):
        mat = np.array([[1.,0.,dx*SCALE],
                        [0.,sy,((1-sy)*cy+dy)*SCALE]], np.float32)
        alpha = cv2.warpAffine(mask.astype(np.float32),mat,(W,H),
                              flags=cv2.INTER_LINEAR,borderMode=cv2.BORDER_CONSTANT)

        prem = cv2.warpAffine(prem,mat,(W,H),flags=cv2.INTER_LINEAR,
                             borderMode=cv2.BORDER_CONSTANT)
        return prem, alpha

    rest = src[27].astype(float)

    clean = background.copy()
    # Frame 15 exposes the side of the red die's ground shadow while the white
    # die is airborne. Preserve those source shadows rather than painting over
    # the patterned floor with a flat color.
    exposed = ((yy >= 78) & (rm15 < .02) & (wm15 < .02))

    # Remove final red body to allow its small impact-induced settling rotation.
    rest = rest*(1-rm27[:,:,None]) + background*rm27[:,:,None]

    # Multiplying restitution by .05 makes the new flight only .55 frame long
    # and ~.055 pixels tall, so it has landed by sampled frame 6.
    impact = 5.45823132
    gravity = 1.4725

    e_original = 7.98/v_impact

    outgoing = e_new*v_impact

        # Local soft shadow fades as the white die rises.  The fixed red
        # contact shadow below y=88 is retained from the source.

        shadow = 9.0*np.exp(-white_height/5.0)*np.exp(
            -.5*((xx-133.0)/6.2)**2-.5*((yy-83.6)/1.65)**2)
        im = np.clip(im-shadow[:,:,None],0,255)
        im = over(im,wp,wa)

        # Composite exactly observed white pixels, except regions where the
        # factual red die hid it. A complete donor texture occupies those holes.
        red_centers = [82.92,76.58,71.57,68.14,66.22,66.86,68.98,
                       72.76,78.32,85.48,84.05,81.5,80.64,81.62,
                       84.35,86.19,84.95,85.35,86.34,86.4,86.37]

        visible = current_mask*(1-binary_dilation(om>.04,iterations=1))
        im = over(im,src[t].astype(float)*visible[:,:,None],visible)

        # rotational settling response. Source pose layers are vertically
        # registered to the contact plane and blend over a quarter second.
        if t < 9:

            mix = (t-6)/3
        elif t < 13:
            ta,tb = 15,27
            mix = (t-9)/4
        else:
            ta,tb,mix = 27,27,0.
        elapsed = t-impact

        pa,aa = move_layer(ta,reds[ta][0],ground_bottom-reds[ta][1][1]-hop)
        pb,ab = move_layer(tb,reds[tb][0],ground_bottom-reds[tb][1][1]-hop)
        im = over(im,pa*(1-mix)+pb*mix,aa*(1-mix)+ab*mix)

        patch = np.array(Image.fromarray(np.clip(im,0,255).round().astype(np.uint8)).resize(

        # The source uses the same Lanczos downsample. Only this local region
        # changes; all unaffected appearance elsewhere stays source-identical.
        output[t,:Y1,X0:X1] = patch

    # Samples before first contact and after settling are source-identical.
    # This is the complete original 4-second timeline, not an appended clip.
    artifact_io.save(output, ROOT)
    print('"'Saved',output.shape,'red restitution scale=0.05; first impact=',impact/24)
    print('New red bounce height',outgoing*outgoing/(2*gravity),'pixels;',
          'flight duration',flight_duration/24,'seconds')


if __name__ == '__main__':
    main()
