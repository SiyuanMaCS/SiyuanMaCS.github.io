+
+The camera and all static objects remain as photographed.  The ball is sampled
+along its original rolling path, with a mass-ratio collision and a reversed,
+friction-decelerated path after contact. This also retains its actual reflection,
+contact shadow, perspective scaling, surface texture, and rolling animation.

+import json
+import numpy as np
+import cv2
+from PIL import Image
+import artifact_io

+def read_source():
+    small = np.load(ROOT / 'source_pixels.npz')['rgb']
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    raw = []
+    while True:
+        ok, bgr = cap.read()
+        if not ok:
+            break
+        raw.append(cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB))
+    cap.release()
+    assert len(raw) == len(small) == 96
+    return small, raw

+def sample_path(raw, position):
+    """Sub-frame sampling at native resolution prevents low-res double edges."""
+    lo = int(np.floor(position))
+    hi = min(lo + 1, len(raw) - 1)
+    fraction = position - lo
+    return (raw[lo].astype(np.float32) * (1.0 - fraction)
+            + raw[hi].astype(np.float32) * fraction)
+

+def rock_pins(image, dt):
+    """Small damped contact rock. No topple, scale, or material change.
+
+    Native-pixel displacement fields are limited to the photographed pins and
+    their floor reflections; the blue ball is explicitly kept out of the warp.
+    """
+    impulse = np.exp(-dt / 10.0) * np.sin(dt * 0.255)
+    if abs(impulse) < 0.0003:
+        return image
+    image = image.copy()
+    # The three original pin centers, from numerical RGB segmentation.
+    x0, x1, y0, y1 = 615, 666, 349, 397
+    crop = image[y0:y1, x0:x1].copy()
+    yy, xx = np.mgrid[y0:y1, x0:x1].astype(np.float32)
+    dx = np.zeros_like(xx)
+    dy = np.zeros_like(yy)
+    height = np.clip((393.5 - yy) / 37.5, 0.0, 1.0)
+    vertical_gate = np.clip((yy - 350.) / 5., 0., 1.)
+    for center, direction in [(628.5, -1.), (640., 0.), (652., 1.)]:
+        gate = np.exp(-((xx-center) / 5.0)**6) * vertical_gate
+        # A 25-fold increase in inertia converts the strike into a tiny rock.
+        dx += direction * 1.05 * impulse * height * gate
+        dy -= (1.60 if direction == 0. else 0.32) * impulse * height * gate
+    # Avoid displacing the ball, its highlights, or antialiased silhouette.
+    blue = crop[..., 2] - crop[..., 0]
+    ball_exclusion = np.clip((blue - 12.) / 23., 0., 1.)
+    ball_exclusion = cv2.GaussianBlur(ball_exclusion, (5, 5), .65)
+    dx *= (1. - ball_exclusion)
+    dy *= (1. - ball_exclusion)
+    mx = (xx - x0 - dx).astype(np.float32)
+    my = (yy - y0 - dy).astype(np.float32)
+    image[y0:y1, x0:x1] = cv2.remap(crop, mx, my, cv2.INTER_LINEAR,
+                                   borderMode=cv2.BORDER_REPLICATE)
+
+    # Upright pin reflections receive the matching, much softer displacement.
+    x0, x1, y0, y1 = 618, 663, 397, 438
+    crop = image[y0:y1, x0:x1].copy()
+    yy, xx = np.mgrid[y0:y1, x0:x1].astype(np.float32)
+    gate = np.exp(-((xx - 640.) / 14.)**6)
+    gate *= np.sin(np.clip((yy - 397.) / 41., 0., 1.) * np.pi)**2
+    blue_exclusion = np.clip((crop[..., 2]-crop[..., 0]-12.) / 23., 0., 1.)
+    gate *= 1. - cv2.GaussianBlur(blue_exclusion, (5, 5), .65)
+    mx = (xx - x0).astype(np.float32)
+    my = (yy - y0 - 0.48 * impulse * gate).astype(np.float32)
+    image[y0:y1, x0:x1] = cv2.remap(crop, mx, my, cv2.INTER_LINEAR,
+                                   borderMode=cv2.BORDER_REPLICATE)
+    return image

+def main():
