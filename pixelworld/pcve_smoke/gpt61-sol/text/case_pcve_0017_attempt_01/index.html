+"""Render the full source-timeline edit using source RGB and local rigid-body math.
+
+The intervention scales the star ball's normal restitution by 1/4, not its
+incoming speed.  A calibrated pinhole floor projection permits the normal and
+tangent wall velocities to be treated separately.  Source pixels provide the
+set, the other ball, and the star-ball's shaded rotating surface.
+"""
+from pathlib import Path
+import json
+import cv2
+import numpy as np
+from scipy.ndimage import map_coordinates, gaussian_filter
+from scipy.optimize import least_squares
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+
+def read_video():
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    frames = []
+    while True:
+        ok, image = cap.read()
+        if not ok:
+            break
+        frames.append(cv2.cvtColor(image, cv2.COLOR_BGR2RGB))
+    cap.release()
+    return np.asarray(frames)
+

+def circle_fit(image, background, guess):
+    """Robust fit of the upper, unoccluded source sphere's outer RGB edges."""
+    cx, cy, r = guess
+    scale = image.shape[1] / 256
+    cx, cy, r = np.array([cx, cy, r]) * scale
+    x0, x1 = int(cx-r-8), int(cx+r+9)
+    y0, y1 = int(cy-r-8), int(cy+r+9)
+    patch = image[y0:y1, x0:x1].astype(float)
+    bg = background[y0:y1, x0:x1].astype(float)
+    difference = np.max(np.abs(patch-bg), axis=2)
+    yy, xx = np.indices(difference.shape)
+    xx, yy = xx+x0, yy+y0
+    rad = np.hypot(xx-cx, yy-cy)
+    foreground = ((difference > 24) & (rad < r*1.10)).astype(np.uint8)
+    contours, _ = cv2.findContours(foreground, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_NONE)
+    points = []
+    for contour in contours:
+        p = contour[:, 0, :].astype(float) + [x0, y0]
+        dist = np.linalg.norm(p-[cx,cy], axis=1)
+        use = (dist > .87*r) & (dist < 1.07*r) & (p[:,1] < cy+.68*r)
+        points.extend(p[use])
+    if len(points) > 30:
+        points = np.asarray(points)
+        def error(p):
+            distances = np.linalg.norm(points-p[:2], axis=1)-p[2]
+            return np.r_[distances, (p-[cx,cy,r])*3]
+        fit = least_squares(error, [cx,cy,r], loss='soft_l1', f_scale=1.8,
+                            bounds=([cx-3,cy-3,r-3], [cx+3,cy+3,r+3])).x
+        return fit/scale
+    return np.array(guess)
+
+
+def texture(image, center, side=160):
+    """A sphere-local sampled RGB sprite; opacity is supplied analytically."""
+    cx, cy, r = center
+    scale = image.shape[1]/256
+    q = (np.arange(side)+.5)/side*2-1
+    yy, xx = np.meshgrid(q, q, indexing='ij')
+    coords = [(cy + yy*r)*scale-.5, (cx + xx*r)*scale-.5]
+    tex = np.stack([map_coordinates(image[...,k].astype(float), coords,
+                                    order=1, mode='nearest') for k in range(3)], -1)
+    # At the edge, remove accidental background pixels by extending the sphere
+    # inward a fraction of a pixel. Only the thin edge ring is modified.
+    radius = np.hypot(xx,yy)
+    edge = radius > .976
+    factor = np.minimum(1, .976/np.maximum(radius,1e-5))
+    edgecoords = [(cy+yy*r*factor)*scale-.5, (cx+xx*r*factor)*scale-.5]
+    inner = np.stack([map_coordinates(image[...,k].astype(float), edgecoords,
+                                    order=1, mode='nearest') for k in range(3)], -1)
+    tex[edge] = inner[edge]
+    return tex.astype(np.float32)
+
+
+def calibrate_motion():
+    # Source wall contact is within the interval 11--12 on the zero-based
+    # timeline. Circle sizes calibrate forward perspective growth.
+    horizon, focal, height, radius = 20.0, 110.0, 4.80, 1.0
+    impact = 11.35
+    r0 = (65.5-horizon)/(height-radius)
+    pos = np.array([(82.7-128)/r0, focal/r0], dtype=float)
+    incoming = np.array([-14.0/r0, 0.0])
+    # Post-wall velocity is fitted before the later ball/ball interaction.
+    factual_rebound = np.array([.245, -.180])
+    normal = factual_rebound-incoming
+    normal /= np.linalg.norm(normal)
+    e_original = -np.dot(factual_rebound,normal)/np.dot(incoming,normal)
+    e_new = e_original*.25
+    outgoing = incoming-(1+e_new)*np.dot(incoming,normal)*normal
+    return impact, pos, outgoing, e_original, (horizon,focal,height,radius)
+
+
+def trajectory():
+    impact, start, velocity, e, camera = calibrate_motion()
+    # Source sliding deceleration, in radius units / source-frame squared.
+    # Subframe integration prevents a visible single-frame contact jump.
+    friction = .0062
+    dt = 1/240
+    pos = start.copy()
+    vel = velocity.copy()
+    angular_distance = 0.
+    positions = []
+    # Soccer ball rests at its initial source position. Test all sphere pairs
+    # in floor coordinates; a weakened wall rebound passes to its left.
+    h,f,height,R = camera
+    soccer_R = .84
+    soccer_r = (91.0-h)*soccer_R/(height-soccer_R)
+    soccer = np.array([(102.5-128)*soccer_R/soccer_r,
+                       f*soccer_R/soccer_r])
+    min_gap = float('inf')
+    last = impact
+    for frame in range(12,96):
+        target = float(frame)
+        while last < target-1e-9:
+            step = min(dt,target-last)
+            speed = np.linalg.norm(vel)
+            if speed > 1e-8:
+                new_speed = max(0.,speed-friction*step)
+                mid_vel = vel*((speed+new_speed)/(2*speed))
+                pos += mid_vel*step
+                angular_distance += np.linalg.norm(mid_vel)*step
+                vel *= new_speed/speed
+            min_gap = min(min_gap, np.linalg.norm(pos-soccer)-(R+soccer_R))
+            last += step
+        r = f*R/pos[1]
+        center = np.array([128+f*pos[0]/pos[1], h+r*(height-R)/R,r])
+        positions.append((center.copy(), angular_distance))
+    if min_gap < 0:
+        raise RuntimeError('Calibrated trajectory unexpectedly contacts the soccer ball')
+    return positions, {'original_restitution_estimate':float(e),
+                        'edited_restitution':float(e*.25),
+                        'normal_velocity_before':float(np.dot(np.array([-14/((65.5-20)/3.8),0]),
+                                                (np.array([.245,-.18])-np.array([-14/((65.5-20)/3.8),0]))/
+                                                np.linalg.norm(np.array([.245,-.18])-np.array([-14/((65.5-20)/3.8),0])))),
+                        'minimum_soccer_surface_gap_world_units':float(min_gap),
+                        'final_star_center_xy_radius':positions[-1][0].tolist()}
+
+
+def paste_sphere(frame, tex, center, soccer_protection):
+    cx, cy, r = center
+    yy, xx = np.indices(frame.shape[:2],dtype=float)
+    # Soft contact shadow anchors the sphere to the unchanged table surface.
+    ground_y = cy+r
+    ell = ((xx-cx)/(r*.81))**2+((yy-(ground_y+2.0))/(r*.245))**2
+    opacity = .23*np.exp(-ell*1.65)
+    # The foreground soccer ball must retain its appearance where the two
+    # projected silhouettes overlap, despite separated 3D floor positions.
+    opacity *= 1-soccer_protection
+    work = frame.astype(float)*(1-opacity[...,None])
+    x0,x1=max(0,int(cx-r-2)),min(256,int(cx+r+3))
+    y0,y1=max(0,int(cy-r-2)),min(144,int(cy+r+3))
+    py,px=np.indices((y1-y0,x1-x0),dtype=float)
+    dx,dy=(px+x0+.5-cx)/r,(py+y0+.5-cy)/r
+    distance=np.hypot(dx,dy)
+    alpha=np.clip((1-distance)*r+.5,0,1)
+    # Analytic sphere coverage and bilinear source RGB, evaluated at 4x for
+    # excellent edge antialiasing at the requested small output resolution.
+    S=4
+    hy,hx=np.indices(((y1-y0)*S,(x1-x0)*S),dtype=float)
+    hdx=(hx/S+x0+.5/S-cx)/r
+    hdy=(hy/S+y0+.5/S-cy)/r
+    mask=(hdx**2+hdy**2<1).astype(float)
+    coords=[(hdy+1)*tex.shape[0]/2-.5,(hdx+1)*tex.shape[1]/2-.5]
+    colors=np.stack([map_coordinates(tex[...,k],coords,order=1,mode='nearest') for k in range(3)],-1)
+    alpha=mask.reshape(y1-y0,S,x1-x0,S).mean((1,3))
+    premult=(colors*mask[...,None]).reshape(y1-y0,S,x1-x0,S,3).mean((1,3))
+    protection=soccer_protection[y0:y1,x0:x1]
+    alpha*=1-protection
+    premult*=1-protection[...,None]
+    work[y0:y1,x0:x1]=work[y0:y1,x0:x1]*(1-alpha[...,None])+premult
+    return np.clip(np.rint(work),0,255).astype(np.uint8)
+
+
+def main():
+    source=np.load(ROOT/'source_pixels.npz')['rgb']
+    assert source.shape==(96,144,256,3)
+    high=read_video()
+    assert len(high)==96
+    bg_high=high[0].copy()
+    bg_high[240:450,1050:1280]=high[40,240:450,1050:1280]
+    guesses=[(238.0,65.5,11.95),(223.0,65.5,11.95),
+             (208.4,65.5,11.95),(194.0,65.5,11.95),
+             (179.6,65.5,11.95),(165.4,65.5,11.95),
+             (151.5,65.5,11.95),(137.7,65.5,11.95),
+             (124.0,65.5,11.95),(110.4,65.5,11.95),
+             (96.9,65.5,11.95),(84.9,65.5,12.0),
+             (85.3,65.0,12.3)]
+    centers=[circle_fit(high[i],bg_high,g) for i,g in enumerate(guesses)]
+    textures=[texture(high[i],centers[i]) for i in range(13)]
+    # Initial actor/background pixels are factual and unchanged until the
+    # first restitution-sensitive event. Subsequent source actors are removed
+    # with actual clean pixels from the start of the same source timeline.
+    frames=source.copy()
+    pristine=source[0].copy()
+    Y,X=np.indices((144,256),dtype=float)
+    # Projection occlusion by the stationary soccer ball. The tiny overlap is
+    # handled with a round visible silhouette, not a copied scene rectangle.
+    soccer_distance=np.hypot((X+.5-102.3)/15.4,(Y+.5-91.6)/15.0)
+    protect=np.clip((1-soccer_distance)*15+.5,0,1)
+    path, diagnostics=trajectory()
+    for frame, (center, rolled) in zip(range(12,96),path):
+        base=source[frame].copy()
+        base[49:144,64:134]=pristine[49:144,64:134]
+        # Reduced linear and rolling travel samples a short continuous run of
+        # the actual star texture rather than inventing a different design.
+        phase=max(0.,11.0-rolled/.98)
+        lo=int(np.floor(phase)); hi=min(12,lo+1);mix=phase-lo
+        tex=textures[lo]*(1-mix)+textures[hi]*mix
+        frames[frame]=paste_sphere(base,tex,center,protect)
+    artifact_io.save(frames,ROOT)
+    (ROOT/'generation_diagnostics.json').write_text(json.dumps(diagnostics,indent=2))
+    print(json.dumps(diagnostics,indent=2))
+    print('Saved',frames.shape,frames.dtype,'to',ROOT)
+
+
+if __name__=='__main__':
+    main()

def circle_fit(image, background, guess):
    \"\"\"Robust fit of the upper, unoccluded source sphere's outer RGB edges.\"\"\"
    cx, cy, r = guess
    scale = image.shape[1] / 256
    cx, cy, r = np.array([cx, cy, r]) * scale
    x0, x1 = int(cx-r-8), int(cx+r+9)
    y0, y1 = int(cy-r-8), int(cy+r+9)
    patch = image[y0:y1, x0:x1].astype(float)
    bg = background[y0:y1, x0:x1].astype(float)
    difference = np.max(np.abs(patch-bg), axis=2)
    yy, xx = np.indices(difference.shape)
    xx, yy = xx+x0, yy+y0
    rad = np.hypot(xx-cx, yy-cy)
    foreground = ((difference > 24) & (rad < r*1.10)).astype(np.uint8)
    contours, _ = cv2.findContours(foreground, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_NONE)
    points = []
    for contour in contours:
        p = contour[:, 0, :].astype(float) + [x0, y0]
        dist = np.linalg.norm(p-[cx,cy], axis=1)
        use = (dist > .87*r) & (dist < 1.07*r) & (p[:,1] < cy+.68*r)
        points.extend(p[use])
    if len(points) > 30:
        points = np.asarray(points)
        def error(p):
            distances = np.linalg.norm(points-p[:2], axis=1)-p[2]
            return np.r_[distances, (p-[cx,cy,r])*3]

                            bounds=([cx-3,cy-3,r-3], [cx+3,cy+3,r+3])).x
        return fit/scale
    return np.array(guess)


def texture(image, center, side=160):
    \"\"\"A sphere-local sampled RGB sprite; opacity is supplied analytically.\"\"\"
    cx, cy, r = center
    scale = image.shape[1]/256
    q = (np.arange(side)+.5)/side*2-1
    yy, xx = np.meshgrid(q, q, indexing='ij')
    coords = [(cy + yy*r)*scale-.5, (cx + xx*r)*scale-.5]
    tex = np.stack([map_coordinates(image[...,k].astype(float), coords,
                                    order=1, mode='nearest') for k in range(3)], -1)
    # At the edge, remove accidental background pixels by extending the sphere
    # inward a fraction of a pixel. Only the thin edge ring is modified.
    radius = np.hypot(xx,yy)
    edge = radius > .976
    factor = np.minimum(1, .976/np.maximum(radius,1e-5))
    edgecoords = [(cy+yy*r*factor)*scale-.5, (cx+xx*r*factor)*scale-.5]
    inner = np.stack([map_coordinates(image[...,k].astype(float), edgecoords,
                                    order=1, mode='nearest') for k in range(3)], -1)
    tex[edge] = inner[edge]
    return tex.astype(np.float32)


def calibrate_motion():
    # Source wall contact is within the interval 11--12 on the zero-based
    # timeline. Circle sizes calibrate forward perspective growth.
    horizon, focal, height, radius = 20.0, 110.0, 4.80, 1.0
    impact = 11.35
    r0 = (65.5-horizon)/(height-radius)
    pos = np.array([(82.7-128)/r0, focal/r0], dtype=float)
    incoming = np.array([-14.0/r0, 0.0])
    # Post-wall velocity is fitted before the later ball/ball interaction.
    factual_rebound = np.array([.245, -.180])
    normal = factual_rebound-incoming
    normal /= np.linalg.norm(normal)
    e_original = -np.dot(factual_rebound,normal)/np.dot(incoming,normal)
    e_new = e_original*.25
    outgoing = incoming-(1+e_new)*np.dot(incoming,normal)*normal
    return impact, pos, outgoing, e_original, (horizon,focal,height,radius)


def trajectory():

    # Source sliding deceleration, in radius units / source-frame squared.
    # Subframe integration prevents a visible single-frame contact jump.
    friction = .0062
    dt = 1/240
    pos = start.copy()
    vel = velocity.copy()
    angular_distance = 0.
    positions = []
    # Soccer ball rests at its initial source position. Test all sphere pairs
    # in floor coordinates; a weakened wall rebound passes to its left.
    h,f,height,R = camera
    soccer_R = .84
    soccer_r = (91.0-h)*soccer_R/(height-soccer_R)
    soccer = np.array([(102.5-128)*soccer_R/soccer_r,
                       f*soccer_R/soccer_r])
    min_gap = float('inf')
    last = impact
    for frame in range(12,96):
        target = float(frame)
        while last < target-1e-9:

            speed = np.linalg.norm(vel)
            if speed > 1e-8:

                mid_vel = vel*((speed+new_speed)/(2*speed))
                pos += mid_vel*step
                angular_distance += np.linalg.norm(mid_vel)*step
                vel *= new_speed/speed

            last += step

def paste_sphere(frame, tex, center, soccer_protection):
    cx, cy, r = center
    yy, xx = np.indices(frame.shape[:2],dtype=float)
    # Soft contact shadow anchors the sphere to the unchanged table surface.
    ground_y = cy+r
    ell = ((xx-cx)/(r*.81))**2+((yy-(ground_y+2.0))/(r*.245))**2
    opacity = .23*np.exp(-ell*1.65)
    # The foreground soccer ball must retain its appearance where the two
    # projected silhouettes overlap, despite separated 3D floor positions.
    opacity *= 1-soccer_protection
    work = frame.astype(float)*(1-opacity[...,None])
    x0,x1=max(0,int(cx-r-2)),min(256,int(cx+r+3))
    y0,y1=max(0,int(cy-r-2)),min(144,int(cy+r+3))
    py,px=np.indices((y1-y0,x1-x0),dtype=float)
    dx,dy=(px+x0+.5-cx)/r,(py+y0+.5-cy)/r
    distance=np.hypot(dx,dy)
    alpha=np.clip((1-distance)*r+.5,0,1)
    # Analytic sphere coverage and bilinear source RGB, evaluated at 4x for
    # excellent edge antialiasing at the requested small output resolution.
    S=4
    hy,hx=np.indices(((y1-y0)*S,(x1-x0)*S),dtype=float)
    hdx=(hx/S+x0+.5/S-cx)/r
    hdy=(hy/S+y0+.5/S-cy)/r
    mask=(hdx**2+hdy**2<1).astype(float)
    coords=[(hdy+1)*tex.shape[0]/2-.5,(hdx+1)*tex.shape[1]/2-.5]
    colors=np.stack([map_coordinates(tex[...,k],coords,order=1,mode='nearest') for k in range(3)],-1)
    alpha=mask.reshape(y1-y0,S,x1-x0,S).mean((1,3))
    premult=(colors*mask[...,None]).reshape(y1-y0,S,x1-x0,S,3).mean((1,3))
    protection=soccer_protection[y0:y1,x0:x1]
    alpha*=1-protection
    premult*=1-protection[...,None]
    work[y0:y1,x0:x1]=work[y0:y1,x0:x1]*(1-alpha[...,None])+premult
    return np.clip(np.rint(work),0,255).astype(np.uint8)
