+"""Source-pixel counterfactual: reduce the launch speed, not gravity or time.
+
+All geometry and textures are measured from the supplied RGB sequence.  The
+three motion phases are supported rolling, ballistic travel, and frictional
+rolling on the receiving surface.  No new scene renderer is used.
+"""
+from pathlib import Path
+import numpy as np
+from scipy import ndimage as ndi
+from scipy.optimize import brentq
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+
+
+def main():
+    src = np.load(ROOT / "source_pixels.npz")["rgb"]
+    rgb = src.astype(np.float64)
+    # The moving car is the only significant changing connected region.
+    # Clean source background: later observations behind its starting path,
+    # and early observations behind its final resting position.
+    background = np.median(rgb[25:], axis=0)
+    background[38:57, 175:208] = np.median(rgb[:6, 38:57, 175:208], axis=0)
+
+    xs = np.array([
+        67.814, 78.967, 89.649, 99.873, 110.170, 120.255,
+        130.062, 138.265, 145.329, 151.850, 157.784, 163.056,
+        167.775, 172.035, 175.900, 179.257, 182.184, 185.362,
+        187.580, 189.227, 190.736, 191.640, 192.135, 192.750])
+    ys = np.array([
+        30.356, 30.344, 30.246, 30.582, 33.170, 37.745,
+        44.812, 48.059, 47.987, 47.612, 47.270, 47.070,
+        46.962, 46.588, 46.500, 46.284, 46.118, 46.225,
+        46.247, 46.068, 45.874, 45.802, 45.843, 45.761])
+
+    # Source projection and acceleration fitted before/after contact.
+    air_x = np.polyfit(np.arange(3, 7), xs[3:7], 2)
+    air_y = np.polyfit(np.arange(3, 7), ys[3:7], 2)
+    plane = np.polyfit(xs[8:], ys[8:], 1)
+    ground_x = np.polyfit(np.arange(7, 24), xs[7:], 2)
+    takeoff = -air_y[1] / (2 * air_y[0])
+    edge_x = np.polyval(air_x, takeoff)
+    edge_y = np.polyval(air_y, takeoff)
+    edge_v = np.polyval(np.polyder(air_x), takeoff)
+    acceleration = 0.4645
+    initial_v = (edge_x - xs[0] + 0.5 * acceleration * takeoff**2) / takeoff
+    gravity_half = air_y[0]
+    projected_air_accel = 2 * air_x[0]
+    landing = brentq(
+        lambda t: np.polyval(air_y, t)
+        - np.polyval(plane, np.polyval(air_x, t)), 6, 7)
+    landing_v = np.polyval(np.polyder(ground_x), landing)
+    air_landing_v = np.polyval(np.polyder(air_x), landing)
+    ground_deceleration = -2 * ground_x[0]
+    source_stop = -ground_x[1] / (2 * ground_x[0])
+
+    cf_takeoff = brentq(
+        lambda t: xs[0] + 0.75 * initial_v * t
+        - 0.5 * acceleration * t**2 - edge_x, 0, 8)
+    cf_edge_v = 0.75 * initial_v - acceleration * cf_takeoff
+
+    def flight_x(u):
+        return edge_x + cf_edge_v * u + 0.5 * projected_air_accel * u**2
+
+    flight_duration = brentq(
+        lambda u: edge_y + gravity_half * u**2
+        - np.polyval(plane, flight_x(u)), 0, 8)
+    cf_landing = cf_takeoff + flight_duration
+    cf_landing_x = flight_x(flight_duration)
+    cf_landing_v = landing_v * (
+        cf_edge_v + projected_air_accel * flight_duration) / air_landing_v
+    cf_stop = cf_landing + cf_landing_v / ground_deceleration
+    cf_stop_x = cf_landing_x + cf_landing_v**2 / (2 * ground_deceleration)
+
+    # Track each original sprite independently from the static background.
+    anchors = []
+    mattes = []
+    sprites = []
+    for k in range(96):
+        cx, cy = xs[min(k, 23)], ys[min(k, 23)]
+        x0, y0 = int(cx) - 15, int(cy) - 10
+        patch = rgb[k, y0:y0+24, x0:x0+32]
+        base = background[y0:y0+24, x0:x0+32]
+        diff = np.max(np.abs(patch - base), axis=2)
+        mask = ndi.binary_fill_holes(ndi.binary_closing(diff > 9))
+        labels, count = ndi.label(mask)
+        sizes = np.bincount(labels.ravel())
+        sizes[0] = 0
+        mask = labels == sizes.argmax()
+        # Include subpixel antialiasing and fine original contact shadow.
+        core = ndi.binary_fill_holes(mask)
+        alpha = ndi.gaussian_filter(core.astype(float), 0.45)
+        alpha[alpha < 0.025] = 0
+        # Unmix source background from the softened silhouette.  Keep
+        # premultiplied colors for clean, dark-rim-free subpixel translation.
+        premult = patch - (1 - alpha[..., None]) * base
+        premult = np.clip(premult, 0, 255 * alpha[..., None])
+        remove = ndi.binary_dilation(core, iterations=2)
+        anchors.append((cx, cy, x0, y0))
+        mattes.append(remove)
+        sprites.append((premult, alpha))
+
+    def source_position(s):
+        s = float(np.clip(s, 0, 23))
+        return np.interp(s, np.arange(24), xs), np.interp(s, np.arange(24), ys)
+
+    def counterfactual(t):
+        if t <= cf_takeoff:
+            x = xs[0] + 0.75 * initial_v * t - 0.5 * acceleration * t*t
+            s = float(np.interp(x, xs[:4], np.arange(4)))
+            _, y = source_position(s)
+            return x, y, s
+        if t < cf_landing:
+            u = t - cf_takeoff
+            s = takeoff + u * (landing - takeoff) / flight_duration
+            return flight_x(u), edge_y + gravity_half*u*u, s
+        elapsed = min(t, cf_stop) - cf_landing
+        x = cf_landing_x + cf_landing_v*elapsed - 0.5*ground_deceleration*elapsed**2
+        s = landing + elapsed * (source_stop - landing) / (cf_stop - cf_landing)
+        if t >= cf_stop:
+            x, s = cf_stop_x, 23.0
+        sx, sy = source_position(s)
+        # Preserve the observed landing pitch/bounce, damped for less impact.
+        y = np.polyval(plane, x) + 0.75 * (sy - np.polyval(plane, sx))
+        return x, y, s
+
+    def stamp(k, cx, cy, bx, by, grid_y, grid_x):
+        ax, ay, x0, y0 = anchors[k]
+        premult, alpha = sprites[k]
+        # Coordinates in the original sprite, with translation only.
+        coords = [grid_y + by - cy + ay - y0,
+                  grid_x + bx - cx + ax - x0]
+        a = ndi.map_coordinates(alpha, coords, order=1, mode="constant", cval=0)
+        p = np.stack([ndi.map_coordinates(premult[..., c], coords,
+                                         order=1, mode="constant", cval=0)
+                      for c in range(3)], axis=-1)
+        return p, a
+
+    frames = []
+    for t in range(96):
+        if t == 0:
+            frames.append(src[0].copy())
+            continue
+        frame = rgb[t].copy()
+        _, _, x0, y0 = anchors[t]
+        removed = mattes[t]
+        area = frame[y0:y0+24, x0:x0+32]
+        area[removed] = background[y0:y0+24, x0:x0+32][removed]
+
+        x, y, phase = counterfactual(float(t))
+        lo = min(int(np.floor(phase)), 23)
+        hi = min(lo + 1, 23)
+        fraction = phase - lo if lo != hi else 0.0
+        bx, by = int(np.floor(x)) - 18, int(np.floor(y)) - 13
+        gy, gx = np.mgrid[0:29, 0:37].astype(float)
+        p0, a0 = stamp(lo, x, y, bx, by, gy, gx)
+        p1, a1 = stamp(hi, x, y, bx, by, gy, gx)
+        p = (1 - fraction) * p0 + fraction * p1
+        alpha = (1 - fraction) * a0 + fraction * a1
+        area = frame[by:by+29, bx:bx+37]
+        area[:] = p + (1 - alpha[..., None]) * area
+        frames.append(np.clip(np.rint(frame), 0, 255).astype(np.uint8))
+
+    frames = np.stack(frames)
+    artifact_io.save(frames, ROOT)
+    print(f"Saved {len(frames)} RGB frames, 256x144, 24fps.")
+    print(f"Initial speed ratio: 0.75; takeoff frame-time {cf_takeoff:.3f}; "
+          f"landing {cf_landing:.3f}; stop {cf_stop:.3f}, x={cf_stop_x:.3f}.")
+
+
+if __name__ == "__main__":
+    main()
