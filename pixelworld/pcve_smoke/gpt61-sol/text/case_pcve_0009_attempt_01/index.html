+"""Numerical source-video edit: maintain the third mallet at the red/white midpoint.
+The original pixels provide all geometry, material texture, and background.
+"""
+from pathlib import Path
+import numpy as np
+import cv2
+from scipy import ndimage as ndi
+from PIL import Image
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+SCALE = 5
+
+
+def component(mask):
+    labels, n = ndi.label(mask)
+    if not n:
+        return mask
+    sizes = np.bincount(labels.ravel()); sizes[0] = 0
+    return labels == sizes.argmax()
+
+
+def centroid(mask, offset=(0,0)):
+    y, x = np.where(mask)
+    return np.array([x.mean()+offset[0], y.mean()+offset[1]])
+
+
+def run():
+    factual = np.load(ROOT/'source_pixels.npz')['rgb']
+    cap = cv2.VideoCapture(str(ROOT/'source_video.mp4'))
+    source = []
+    for _ in range(len(factual)):
+        ok, bgr = cap.read()
+        if not ok:
+            raise RuntimeError('Incomplete source video')
+        source.append(cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB))
+    cap.release()
+    source = np.asarray(source)
+    n, h, w, _ = source.shape
+    blue_centers, red_centers, white_centers = [], [], []
+    for t, f in enumerate(source):
+        # Color segmentation on the original-resolution source avoids quantization
+        # jitter and includes the antialiased cylindrical mallet head.
+        x0,y0,x1,y1 = 550,240,880,425
+        crop = f[y0:y1,x0:x1].astype(np.float32)
+        r,g,b = crop.transpose(2,0,1)
+        bm = component((g-r>13)&(b-g>20))
+        rm = component((r-g>50)&(r-b>40))
+        blue_centers.append(centroid(bm,(x0,y0)))
+        red_centers.append(centroid(rm,(x0,y0)))
+
+        xp = 105.5-2.1*np.clip(t-30,0,18)+max(0,t-48)*.055
+        yp = 81+.84*np.clip(t-30,0,18)-max(0,t-48)*.022
+        xa,xb = int((xp-10)*5),int((xp+10)*5)
+        ya,yb = int((yp-7)*5),int((yp+7)*5)
+        c=f[ya:yb,xa:xb].astype(np.float32)
+        r,g,b=c.transpose(2,0,1)
+        wm=component((r>233)&(g>227)&(np.abs(b-r)<11))
+        white_centers.append(centroid(wm,(xa,ya)))
+    blue_centers=np.array(blue_centers)
+    red_centers=np.array(red_centers)
+    white_centers=np.array(white_centers)
+    # Suppress only segmentation jitter, not the observed trajectories.
+    for centers in (blue_centers,red_centers,white_centers):
+        centers[:]=ndi.median_filter(centers,size=(3,1),mode='nearest')
+    target=(red_centers+white_centers)/2
+
+    # Recover the exposed static tabletop from frames in which the mallets
+    # are elsewhere. Only a small swept region is needed for this operation.
+    x0,y0,x1,y1=660,240,890,400
+    stack=source[:,y0:y1,x0:x1].astype(np.float32)
+    yy,xx=np.mgrid[y0:y1,x0:x1]
+    available=np.ones(stack.shape[:3],bool)
+    for t in range(n):
+        for centers, left,right,top,bottom in [
+            (blue_centers,37,45,29,48),
+            (red_centers,38,43,30,41),
+            (white_centers,43,45,30,46)]:
+            cx,cy=centers[t]
+            unavailable=(xx>cx-left)&(xx<cx+right)&(yy>cy-top)&(yy<cy+bottom)
+            available[t]&=~unavailable
+    # Pick an actual source frame for each clean pixel, then use a robust
+    # median only where multiple observations exist.
+    import warnings
+    with warnings.catch_warnings():
+        warnings.simplefilter('ignore',RuntimeWarning)
+        stack[~available]=np.nan
+        bg=np.nanmedian(stack,axis=0)
+    unknown=np.isnan(bg[:,:,0])
+    bg=np.nan_to_num(bg,nan=0).astype(np.uint8)
+    if unknown.any():
+        bg=cv2.inpaint(bg,unknown.astype(np.uint8)*255,4,cv2.INPAINT_TELEA)
+    background=source[0].copy()
+    background[y0:y1,x0:x1]=bg
+
+    # A source-derived matte includes the mallet and its soft cast shadow.
+    sx0,sy0,sx1,sy1=800,255,880,345
+    first=source[0,sy0:sy1,sx0:sx1].astype(np.float32)
+    exposed=source[-1,sy0:sy1,sx0:sx1].astype(np.float32)
+    dif=np.max(np.abs(first-exposed),axis=2)
+    matte=component(dif>9)
+    matte=ndi.binary_fill_holes(matte)
+    matte=ndi.binary_dilation(matte,iterations=2)
+    matte=ndi.gaussian_filter(matte.astype(np.float32),sigma=.8)
+    material=first.copy()
+    r,g,b=first.transpose(2,0,1)
+    pigment=np.clip((g-r+5)/25,0,1)*np.clip((b-g-13)/20,0,1)
+    pigment=ndi.gaussian_filter(pigment,.45)
+    chroma=np.maximum(b-(r+g)/2,0)*pigment
+    # Shift just the blue/teal body pigment to the requested green, retaining
+    # highlights, contact shading, and neutral/lavender cast-shadow pixels.
+    material[:,:,0]=r-.10*chroma
+    material[:,:,1]=g+.68*chroma
+    material[:,:,2]=b-1.22*chroma
+    material=np.clip(material,0,255)
+    template_center=blue_centers[0]-[sx0,sy0]
+    out=[]
+    for t,f in enumerate(source):
+        canvas=f.astype(np.float32)
+        # Translate the same rigid source matte to locate and erase its old
+        # position; preserve neighboring red pigment during contact frames.
+        shift=blue_centers[t]-blue_centers[0]
+        M=np.array([[1,0,sx0+shift[0]],[0,1,sy0+shift[1]]],np.float32)
+        old=cv2.warpAffine(matte,M,(w,h),flags=cv2.INTER_LINEAR)
+        r,g,b=canvas.transpose(2,0,1)
+        protected=(r-g>35)&(r-b>28)
+        old[protected]=0
+        canvas=canvas*(1-old[:,:,None])+background.astype(np.float32)*old[:,:,None]
+
+        # The mallet's projected head center, rather than its shadow centroid,
+        # is the endpoint used for the exact midpoint relation.
+        dest=target[t]-template_center
+        M=np.array([[1,0,dest[0]],[0,1,dest[1]]],np.float32)
+        alpha=cv2.warpAffine(matte,M,(w,h),flags=cv2.INTER_LINEAR)
+        premult=cv2.warpAffine(material*matte[:,:,None],M,(w,h),flags=cv2.INTER_LINEAR)
+        canvas=canvas*(1-alpha[:,:,None])+premult
+        # Match the exact resize used by the provided source RGB tensor.
+        rgb=np.clip(np.rint(canvas),0,255).astype(np.uint8)
+        out.append(np.asarray(Image.fromarray(rgb).resize((256,144),Image.Resampling.LANCZOS)))
+    frames=np.asarray(out,dtype=np.uint8)
+    artifact_io.save(frames,ROOT)
+    np.savez_compressed(ROOT/'.tmp'/'tracking.npz',red=red_centers/5,white=white_centers/5,green=target/5)
+    print('Saved',frames.shape,'RGB frames at 24 fps')
+    print('Midpoint error:',np.max(np.abs(target-(red_centers+white_centers)/2)))
+    print('Unobserved background pixels repaired:',int(unknown.sum()))
+
+if __name__=='__main__':
+    run()

\"\"\"Numerical source-video edit: maintain the third mallet at the red/white midpoint.
The original pixels provide all geometry, material texture, and background.

from pathlib import Path
import numpy as np
import cv2
from scipy import ndimage as ndi
from PIL import Image
import artifact_io

ROOT = Path(__file__).resolve().parent
SCALE = 5


def component(mask):
    labels, n = ndi.label(mask)
    if not n:
        return mask
    sizes = np.bincount(labels.ravel()); sizes[0] = 0
    return labels == sizes.argmax()


def centroid(mask, offset=(0,0)):
    y, x = np.where(mask)
    return np.array([x.mean()+offset[0], y.mean()+offset[1]])


def run():
    factual = np.load(ROOT/'source_pixels.npz')['rgb']
    cap = cv2.VideoCapture(str(ROOT/'source_video.mp4'))
    source = []
    for _ in range(len(factual)):
        ok, bgr = cap.read()
        if not ok:
            raise RuntimeError('Incomplete source video')
        source.append(cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB))

    source = np.asarray(source)

    blue_centers, red_centers, white_centers = [], [], []
    for t, f in enumerate(source):

        # jitter and includes the antialiased cylindrical mallet head.

        r,g,b = crop.transpose(2,0,1)

        rm = component((r-g>50)&(r-b>40))
        blue_centers.append(centroid(bm,(x0,y0)))
        red_centers.append(centroid(rm,(x0,y0)))

        rgb=np.clip(np.rint(canvas),0,255).astype(np.uint8)
