+"""Reduce the tennis ball's contact friction, retaining the source camera/render.
+
+Only source pixels supply appearance. Ballistic motion is unaffected; the ground
+motion is estimated from the source and its friction impulse and Coulomb term
+are multiplied by 0.2. Linear (non-contact) damping is kept unchanged.
+"""
+from pathlib import Path
+import json
+import subprocess
+import shutil
+
+import cv2
+import numpy as np
+from PIL import Image
+from scipy.optimize import least_squares
+
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+
+
+def ball_location(image, expected=None):
+    """Find the tennis yellow using RGB arithmetic, without learned masks."""
+    if expected is not None:
+        ex, ey = expected
+        left, top = int(round(ex)) - 22, int(round(ey)) - 22
+        image = image[top:top + 45, left:left + 45]
+    else:
+        left = top = 0
+    r, g, b = image.astype(float).transpose(2, 0, 1)
+    yellow = ((r > 100) & (g > 95) & (b < 100) &
+              (g > 1.4 * b) & (r > 1.4 * b) & (abs(r - g) < 85))
+    yy, xx = np.where(yellow)
+    if expected is None:
+        return np.array([xx.mean(), yy.mean()])
+    # Row midpoints are less biased by the light-colored moving seam than a
+    # color centroid. Work at the source's native resolution for subpixel motion.
+    row_centers = []
+    for row in yellow:
+        indices = np.flatnonzero(row)
+        if len(indices) >= 16:
+            row_centers.append((indices.min() + indices.max()) / 2)
+    x = np.mean(row_centers) if row_centers else xx.mean()
+    y = (yy.min() + yy.max()) / 2
+    return np.array([x + left, y + top])
+
+
+def displacement(t, velocity, friction, damping):
+    """Analytic forward/backward ground trajectory with finite stopping time."""
+    stop = np.log1p(damping * velocity / friction) / damping
+    t = np.minimum(np.asarray(t), stop)
+    return ((velocity + friction / damping) *
+            (-np.expm1(-damping * t)) / damping - friction * t / damping)
+
+
+def generate():
+    source = np.load(ROOT / 'source_pixels.npz')['rgb']
+    info = json.loads((ROOT / 'source_info.json').read_text())
+    assert source.shape == (96, 144, 256, 3)
+    assert info['execution_frame_1based'] == 1
+    coarse = np.array([ball_location(frame) for frame in source])
+
+    # Read just a narrow ground strip at native resolution; no full-resolution
+    # video tensor is needed. The strip is clean in the first 24 source frames.
+    strip_top, strip_bottom = 430, 515
+    native_strips, locations = [], []
+    capture = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    for t in range(len(source)):
+        ok, bgr = capture.read()
+        if not ok:
+            raise RuntimeError('Incomplete source decode')
+        rgb = cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB)
+        locations.append(ball_location(rgb, coarse[t] * 5 + 2))
+        native_strips.append(rgb[strip_top:strip_bottom].copy())
+    capture.release()
+    strips = np.stack(native_strips)
+    native = np.array(locations)
+    observed = (native - 2) / 5
+    background = np.median(strips[:24], axis=0).astype(np.float32)
+
+    air_time = np.arange(28, dtype=float)
+    px = np.polyfit(air_time, observed[:28, 0], 2)
+    py = np.polyfit(air_time, observed[:28, 1], 2)
+    ground_y = float(np.median(observed[28:, 1]))
+    touchdown = float(max(np.roots(py - np.array([0., 0., ground_y]))))
+    times = np.arange(28, 96, dtype=float) - 28
+
+    def residual(params):
+        x0, v, a, k = params
+        return x0 + displacement(times, v, a, k) - observed[28:, 0]
+
+    fit = least_squares(residual, [observed[28, 0], 2.4, .033, .011],
+                        bounds=([150, 0, .00001, .000001], [155, 5, .5, .1]))
+    fitted_x, v28, a, k = fit.x
+    # Anchor the reconstruction to the actual first contact image instead of
+    # allowing the fitted centroid's small seam bias to create a position jump.
+    back_time = touchdown - 28
+    land_x = observed[28, 0] + displacement(back_time, v28, a, k)
+    land_v = (v28 + a / k) * np.exp(-k * back_time) - a / k
+    incoming_v = float(np.polyval(np.polyder(px), touchdown))
+    contact_ratio = .2
+    modified_v = incoming_v - contact_ratio * (incoming_v - land_v)
+    target_x = observed[:, 0].copy()
+    target_x[28:] = land_x + displacement(np.arange(28, 96) - touchdown,
+                                         modified_v, a * contact_ratio, k)
+
+    result = source.copy()
+    grid_y, grid_x = np.indices(background.shape[:2], dtype=np.float32)
+    for t in range(28, 96):
+        cx, cy = native[t]
+        cy -= strip_top
+        dx, dy = grid_x - cx, grid_y - cy
+        # Retain antialiased edges, seam, shading, and the broad soft ground
+        # shadow. A soft geometric support rejects irrelevant codec noise.
+        distance = np.hypot(dx, dy)
+        sphere = np.clip((18 - distance) / 5, 0, 1)
+        shadow_distance = np.sqrt(((dx + 5) / 43) ** 2 + ((dy - 10) / 12) ** 2)
+        shadow = np.clip((1.15 - shadow_distance) / .35, 0, 1)
+        support = np.maximum(sphere, shadow)
+        raw_delta = strips[t].astype(np.float32) - background
+        significance = np.clip((np.max(abs(raw_delta), axis=2) - 1.5) / 3, 0, 1)
+        # Preserve the entire opaque sphere and all significant shadow detail.
+        support *= np.maximum(significance, sphere)
+        delta = raw_delta * support[..., None]
+        translation = 5 * (target_x[t] - observed[t, 0])
+        moved = cv2.warpAffine(delta,
+                               np.array([[1, 0, translation], [0, 1, 0]], np.float32),
+                               (background.shape[1], background.shape[0]),
+                               flags=cv2.INTER_CUBIC,
+                               borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+        clean_strip = strips[t].astype(np.float32) - delta
+        edited_strip = np.clip(np.rint(clean_strip + moved), 0, 255).astype(np.uint8)
+
+        # Use the exact source downsampling filter. Resize a padded local strip
+        # and transfer only its difference to the supplied RGB frames, leaving
+        # every other source pixel untouched.
+        low_h = (strip_bottom - strip_top) // 5
+        low_original = np.array(Image.fromarray(strips[t]).resize(
+            (256, low_h), Image.Resampling.LANCZOS)).astype(np.int16)
+        low_edited = np.array(Image.fromarray(edited_strip).resize(
+            (256, low_h), Image.Resampling.LANCZOS)).astype(np.int16)
+        start = strip_top // 5
+        result[t, start:start + low_h] = np.clip(
+            source[t, start:start + low_h].astype(np.int16) + low_edited - low_original,
+            0, 255).astype(np.uint8)
+
+    # The required RGB artifacts are saved before any inspection/encoding.
+    artifact_io.save(result, ROOT)
+    report = {
+        'friction_ratio': contact_ratio,
+        'touchdown_frame_zero_based': touchdown,
+        'touchdown_seconds': touchdown / 24,
+        'incoming_velocity_px_per_frame': incoming_v,
+        'original_contact_velocity': float(land_v),
+        'reduced_friction_contact_velocity': float(modified_v),
+        'original_coulomb_deceleration': float(a),
+        'unchanged_linear_damping': float(k),
+        'ground_fit_rms_pixels': float(np.sqrt(np.mean(fit.fun ** 2))),
+        'source_x': observed[:, 0].tolist(),
+        'edited_x': target_x.tolist(),
+        'shape': list(result.shape),
+        'airborne_frames_identical': bool(np.array_equal(result[:28], source[:28])),
+    }
+    (ROOT / 'generation_report.json').write_text(json.dumps(report, indent=2))
+
+    # Also provide a standard shareable MP4, encoded from the same saved frames.
+    ffmpeg = shutil.which('ffmpeg')
+    if ffmpeg:
+        subprocess.run([ffmpeg, '-y', '-loglevel', 'error', '-framerate', '24',
+                        '-i', str(ROOT / 'frames' / '%02d.png'), '-c:v', 'libx264',
+                        '-preset', 'medium', '-crf', '15', '-pix_fmt', 'yuv420p',
+                        '-movflags', '+faststart', str(ROOT / 'edited_video.mp4')],
+                       check=True)
+    else:
+        writer = cv2.VideoWriter(str(ROOT / 'edited_video.mp4'),
+                                 cv2.VideoWriter_fourcc(*'mp4v'), 24, (256, 144))
+        for frame in result:
+            writer.write(cv2.cvtColor(frame, cv2.COLOR_RGB2BGR))
+        writer.release()
+    print(json.dumps({k: v for k, v in report.items()
+                      if k not in ('source_x', 'edited_x')}, indent=2))
+    print('Saved pixels.npz, frames/, index.html, and edited_video.mp4')
+
+
+if __name__ == '__main__':
+    generate()
