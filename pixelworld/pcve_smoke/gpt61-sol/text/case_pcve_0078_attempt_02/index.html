+"""Fourfold soccer-ball mass intervention, using only source RGB and geometry.
+
+The incoming projectile and the stationary ball retain their original histories.
+The glancing impact impulse is adjusted for a fourfold target inertia, and the
+original surface textures are transported along the resulting trajectories.
+"""
+from pathlib import Path
+import numpy as np
+from scipy import ndimage as ndi
+from scipy.optimize import least_squares
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+Y, X = np.indices((144, 256), dtype=float)
+
+
+def largest(mask):
+    labels, _ = ndi.label(mask)
+    sizes = np.bincount(labels.ravel())
+    sizes[0] = 0
+    return ndi.binary_fill_holes(labels == sizes.argmax())
+
+
+def track_soccer(source):
+    positions = []
+    for a in source:
+        m = ((a[...,1]-a[...,0] < 12) & (a[...,2]-a[...,1] > -14)
+             & (Y >= 69) & (Y < 94) & (X > 60) & (X < 140))
+        m = largest(m)
+        edge = m & ~ndi.binary_erosion(m)
+        ys, xs = np.where(edge)
+        fit = least_squares(lambda p: np.hypot(xs-p[0], ys-p[1])-p[2],
+                            [xs.mean(), ys.mean(), 10])
+        positions.append(fit.x[:2])
+    return np.array(positions)
+
+
+def track_projectile(source):
+    points = []
+    for t in range(12, 38):
+        a = source[t]
+        m = ((a[...,0]-a[...,1] > 15) & (a[...,0] > 125)
+             & (Y > 58) & (Y < 95) & (X > 130))
+        m = largest(m)
+        ys, xs = np.where(m)
+        points.append([xs.mean(), ys.mean()])
+    return np.array(points)
+
+
+def travel(t, v, k, friction, contact):
+    elapsed = np.maximum(np.asarray(t)-contact, 0)
+    stop = np.log1p(v*k/friction)/k
+    elapsed = np.minimum(elapsed, stop)
+    return (v+friction/k)/k*(-np.expm1(-k*elapsed))-friction/k*elapsed
+
+
+def local_warp(rgb, cx, cy, tx, ty):
+    """Bilinear translation of premultiplied foreground / shadow residuals."""
+    coords = np.array([Y-ty+cy, X-tx+cx])
+    return np.stack([ndi.map_coordinates(rgb[..., c], coords, order=1,
+                                       mode='constant', cval=0, prefilter=False)
+                     for c in range(3)], axis=-1)
+
+
+def main():
+    source_u8 = np.load(ROOT/'source_pixels.npz')['rgb']
+    source = source_u8.astype(float)
+    ball = track_soccer(source)
+    small = track_projectile(source)
+    cx, cy = ball[0]
+
+    # Recover the unobstructed floor from the original timeline. The left part
+    # is clear initially; the right part is clear after the factual ball rolls.
+    plate = source[0].copy()
+    plate[68:104, 96:] = source[95, 68:104, 96:]
+
+    # Fit the original ball's measured rolling recoil. Drag and floor friction
+    # are retained: mass does not alter gravity or rolling-friction acceleration.
+    ts = np.arange(13, 57, dtype=float)
+    fit = least_squares(lambda p: cx-travel(ts, *p)-ball[13:57,0],
+                        [2.85, .049, .003, 12.68],
+                        bounds=([1,.0001,.0001,12], [5,.5,.2,13]))
+    velocity, drag, friction, contact = fit.x
+
+    # Factual momentum exchange estimates M/m approximately 2.8. If M becomes
+    # 4M, projectile impulse changes by (M+m)/(M+m/4), while target velocity
+    # changes by one quarter of that impulse multiplier.
+    projectile_speed = (small[2,0]-small[1,0])
+    mass_ratio = velocity/projectile_speed  # m / M
+    impulse_ratio = (1+mass_ratio)/(1+mass_ratio/4)
+    recoil_ratio = impulse_ratio/4
+    heavy_velocity = velocity*recoil_ratio
+
+    # Ball silhouettes / materials remain source pixels; only inertia changes.
+    ball_residuals = []
+    for t in range(12, 22):
+        bx, by = ball[t]
+        r = np.hypot(X-bx, Y-by)
+        coverage = np.clip((11.05-r)/.65, 0, 1)
+        residual = (source[t]-plate)*coverage[..., None]
+        ball_residuals.append(residual)
+
+    # Carry the source's soft, light-direction-consistent soccer shadow with it.
+    radius = np.hypot(X-cx,Y-cy)
+    coverage0 = np.clip((11.05-radius)/.65, 0, 1)
+    support = ((X > cx-14)&(X < cx+23)&(Y > cy-12)&(Y < cy+19))
+    shadow = (source[0]-plate)*(1-coverage0[...,None])*support[...,None]
+    # Suppress insignificant codec variations outside the actual shadow.
+    magnitude = np.max(np.abs(shadow), axis=-1)
+    shadow *= np.clip((magnitude-2)/5,0,1)[...,None]
+
+    # Grounded projectile texture and its shadow can be reused as one residual.
+    # Airborne texture is isolated from the ground shadow for the first two
+    # post-impact samples.
+    projectile_residuals = []
+    projectile_centers = []
+    for t in range(14, 38):
+        px, py = small[t-12]
+        support = ((X >= px-5)&(X <= px+9)&(Y >= py-5)&(Y <= py+9))
+        residual = (source[t]-plate)*support[...,None]
+        # Patch edge feathering prevents transporting floor compression noise.
+        weight = np.clip(np.minimum.reduce([X-(px-5),px+9-X,
+                                            Y-(py-5),py+9-Y])/1.2,0,1)
+        residual *= weight[...,None]
+        projectile_residuals.append(residual)
+        projectile_centers.append([px,py])
+    projectile_centers = np.asarray(projectile_centers)
+
+    px0, py0 = projectile_centers[0]
+    pr = np.hypot(X-px0,Y-py0)
+    pa = np.clip((3.9-pr)/.8,0,1)
+    projectile_air = (source[14]-plate)*pa[...,None]
+
+    # The projectile horizontal speed increases, but its vertical impulse is
+    # more upward, hence it reaches the floor slightly later, not earlier.
+    impact_x = np.interp(contact,np.arange(12,38),small[:,0])
+    factual_vy_out = 10.2
+    incoming_vy = 17.2
+    heavier_vy = incoming_vy-(incoming_vy-factual_vy_out)*impulse_ratio
+    vertical_ratio = heavier_vy/factual_vy_out
+
+    result = source_u8.copy()
+    diagnostics = []
+    for t in range(13, 96):
+        image = source[t].copy()
+        image[68:104,60:] = plate[68:104,60:]
+        distance = float(travel(t,heavy_velocity,drag,friction,contact))
+        bx = cx-distance
+        # Choose surface rotation phase by traveled arc length, not output time.
+        factual_distances = cx-ball[12:22,0]
+        phase = float(np.interp(distance,factual_distances,np.arange(12,22)))
+        i = min(int(np.floor(phase)),20)
+        frac = phase-i
+        by = float(np.interp(phase,np.arange(96),ball[:,1]))
+        residual = np.zeros_like(image)
+        for index, weight in [(i,1-frac),(i+1,frac)]:
+            if weight:
+                bc = ball[index]
+                residual += weight*local_warp(ball_residuals[index-12],
+                                              bc[0],bc[1],bx,by)
+        image += local_warp(shadow,cx,cy,bx,by)
+        image += residual
+
+        if t <= 37:
+            original_x = np.interp(t,np.arange(12,38),small[:,0])
+            sx = impact_x+impulse_ratio*(original_x-impact_x)
+            # Scene projection puts ground heights very slightly uphill to the
+            # right. Recover this from the measured factual rolling positions.
+            texture_phase = float(np.interp(sx,small[:,0],np.arange(12,38)))
+            ground_y = float(np.interp(sx,small[2:,0],small[2:,1]))
+            vert_phase = contact+(t-contact)*vertical_ratio
+            sy = float(np.interp(vert_phase,np.arange(12,38),small[:,1]))
+            if vert_phase < 14.2:
+                # Airborne shadow expands and separates along the source light.
+                height = max(0,ground_y-sy)
+                shx, shy = sx+3.1+.47*height, ground_y+.9-.19*height
+                sigma_x, sigma_y = 2.55+.025*height, 1.15+.015*height
+                opacity = .35*np.exp(-height/30)
+                alpha = opacity*np.exp(-.5*((X-shx)/sigma_x)**2
+                                        -.5*((Y-shy)/sigma_y)**2)
+                image *= 1-alpha[...,None]
+                image += local_warp(projectile_air,px0,py0,sx,sy)
+            else:
+                sy = ground_y
+                texture_phase = max(14,texture_phase)
+                ind = min(int(np.floor(texture_phase)),36)
+                frac_p = min(1,texture_phase-ind)
+                for j,w in [(ind,1-frac_p),(ind+1,frac_p)]:
+                    if w:
+                        pc = projectile_centers[j-14]
+                        image += w*local_warp(projectile_residuals[j-14],
+                                              pc[0],pc[1],sx,sy)
+        else:
+            sx,sy = 270.0,85.0  # deflected projectile has left the camera view
+        result[t] = np.clip(image,0,255).round().astype(np.uint8)
+        diagnostics.append([t, bx, by, sx, sy, phase])
+
+    # Exact initial conditions, unchanged camera, no visual mass indicator.
+    artifact_io.save(result, ROOT)
+    np.savetxt(ROOT/'motion.csv',np.asarray(diagnostics),delimiter=',',
+               header='frame,ball_x,ball_y,projectile_x,projectile_y,texture_phase')
+    print('Generated',result.shape,'RGB frames; 24 fps.')
+    print('Inferred impulse multiplier:',round(float(impulse_ratio),4))
+    print('Soccer recoil velocity multiplier:',round(float(recoil_ratio),4))
+    print('Soccer final center:',diagnostics[-1][1:3])
+
+
+if __name__ == '__main__':
+    main()

\"\"\"Fourfold soccer-ball mass intervention, using only source RGB and geometry.

The incoming projectile and the stationary ball retain their original histories.
The glancing impact impulse is adjusted for a fourfold target inertia, and the
original surface textures are transported along the resulting trajectories.
\"\"\"
from pathlib import Path
import numpy as np
from scipy import ndimage as ndi
from scipy.optimize import least_squares
import artifact_io

ROOT = Path(__file__).resolve().parent
Y, X = np.indices((144, 256), dtype=float)


def largest(mask):
    labels, _ = ndi.label(mask)
    sizes = np.bincount(labels.ravel())
    sizes[0] = 0
    return ndi.binary_fill_holes(labels == sizes.argmax())


def track_soccer(source):
    positions = []
    for a in source:
        m = ((a[...,1]-a[...,0] < 12) & (a[...,2]-a[...,1] > -14)
             & (Y >= 69) & (Y < 94) & (X > 60) & (X < 140))
        m = largest(m)
        edge = m & ~ndi.binary_erosion(m)
        ys, xs = np.where(edge)
        fit = least_squares(lambda p: np.hypot(xs-p[0], ys-p[1])-p[2],
                            [xs.mean(), ys.mean(), 10])
        positions.append(fit.x[:2])
    return np.array(positions)


def track_projectile(source):
    points = []
    for t in range(12, 38):
        a = source[t]
        m = ((a[...,0]-a[...,1] > 15) & (a[...,0] > 125)
             & (Y > 58) & (Y < 95) & (X > 130))
        m = largest(m)
        ys, xs = np.where(m)
        points.append([xs.mean(), ys.mean()])
    return np.array(points)


def travel(t, v, k, friction, contact):
    elapsed = np.maximum(np.asarray(t)-contact, 0)
    stop = np.log1p(v*k/friction)/k
    elapsed = np.minimum(elapsed, stop)
    return (v+friction/k)/k*(-np.expm1(-k*elapsed))-friction/k*elapsed


def local_warp(rgb, cx, cy, tx, ty):
    \"\"\"Bilinear translation of premultiplied foreground / shadow residuals.\"\"\"
    coords = np.array([Y-ty+cy, X-tx+cx])
    return np.stack([ndi.map_coordinates(rgb[..., c], coords, order=1,
                                       mode='constant', cval=0, prefilter=False)
                     for c in range(3)], axis=-1)
