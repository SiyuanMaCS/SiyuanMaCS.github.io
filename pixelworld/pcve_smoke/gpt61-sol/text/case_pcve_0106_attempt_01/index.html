+"""Mass counterfactual, reconstructed from the supplied RGB timeline.
+
+Only local pixels, array resampling and a fitted two-body impulse model are used.
+The increased mass does not alter gravity/friction accelerations or marble size.
+It changes the impulse when the large and small marbles first make contact.
+"""
+from pathlib import Path
+import numpy as np
+from scipy.ndimage import map_coordinates, gaussian_filter
+from scipy.optimize import least_squares
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+
+def distance(u, speed, friction):
+    u = np.clip(u, 0, speed / friction)
+    return speed * u - .5 * friction * u * u
+
+
+def measure_small(source):
+    yy, xx = np.indices(source.shape[1:3])
+    centers = []
+    for im in source:
+        a = im.astype(float)
+        mask = ((a[:, :, 2] > a[:, :, 0] + 30) &
+                (a[:, :, 1] > a[:, :, 0] + 30) & (yy > 62) & (yy < 82))
+        centers.append([xx[mask].mean(), yy[mask].mean()])
+    return np.asarray(centers)
+
+
+def fit_dynamics(source):
+    # Upper silhouette centers measured numerically from the amber marble.
+    big_x = np.array([
+        224.3, 218.2, 211.8, 205.4, 199.3, 192.9, 186.9, 180.7,
+        174.6, 168.7, 163., 157.3, 151.8, 146.4, 142., 137.7,
+        133.5, 129.4, 125.7, 122.1, 118.8, 115.6, 112.6, 110.,
+        107.7, 105.6, 103.8, 102.2, 100.9, 99.8, 99., 98.5,
+        98.2, 98.2, 98.2, 98.2, 98.2, 98.2, 98.2, 98.2])
+    big_x = np.r_[big_x, np.full(56, 98.2)]
+    small = measure_small(source)
+    incoming = np.polyfit(np.arange(5, 14), big_x[5:14], 2)
+
+    def residual(p):
+        contact, speed_big, drag_big, speed_small, drag_small = p
+        t = np.arange(14, 40)
+        xb = np.polyval(incoming, contact) - distance(t-contact, speed_big, drag_big)
+        xs = small[0, 0] - distance(t-contact, speed_small, drag_small)
+        return np.r_[xb-big_x[14:40], xs-small[14:40, 0]]
+
+    result = least_squares(residual, [13.6, 4.5, .226, 9., .34],
+                           bounds=([13.1, 3, .1, 7, .2], [14., 6, .4, 11, .5]))
+    contact, vb, ab, vs, ass = result.x
+    vi = -np.polyval(np.polyder(incoming), contact)
+    # Momentum and relative separation speed identify the original mass ratio
+    # and restitution. Increase just the large body's mass by five.
+    original_ratio = vs / (vi-vb)
+    restitution = (vs-vb) / vi
+    ratio = 5 * original_ratio
+    vb_new = vi * (ratio-restitution) / (ratio+1)
+    vs_new = vi * (1+restitution) * ratio / (ratio+1)
+    t = np.arange(len(source))
+    xb = np.polyval(incoming, contact) - distance(t-contact, vb_new, ab)
+    xs = small[0, 0] - distance(t-contact, vs_new, ass)
+    print('Contact frame (zero based):', round(contact, 4))
+    print('Inferred mass ratio and restitution:', original_ratio, restitution)
+    print('Post-collision speeds, original:', vb, vs, '5x mass:', vb_new, vs_new)
+    print('Final centers:', xb[-1], xs[-1])
+    return big_x, small, xb, xs, contact
+
+
+def background_plate(source):
+    # All plate pixels come from intervals in which that region is unobstructed.
+    plate = np.median(source[40:], axis=0)
+    early = np.median(source[:12], axis=0)
+    # At rest the two marbles occupy these regions for most of the recording.
+    # There are no marbles here at the start, so use the early clean samples.
+    plate[53:98, 82:119] = early[53:98, 82:119]
+    plate[62:101, :29] = early[62:101, :29]
+    return plate
+
+
+def radius_big(x):
+    return 9.80-.0058*x, 9.60-.0080*x
+
+
+def y_big(x):
+    return 75.15-.0496*x
+
+
+def y_small(x):
+    return 75.45-.0254*x
+
+
+def layer_delta(image, plate, cx, cy, rx, ry, small=False):
+    """Capture the marble plus its original antialiasing, shadow and reflection.
+
+    An elliptical support retains the source sphere. A separate tapered support
+    carries the floor light changes without moving the background itself.
+    """
+    yy, xx = np.indices(plate.shape[:2], dtype=float)
+    nx = (xx-cx)/rx
+    ny = (yy-cy)/ry
+    radial = np.sqrt(nx*nx + ny*ny)
+    # Include the antialias ring and its subtle dark outline intact.
+    sphere = np.clip((1.18-radial)/.13, 0, 1)
+    if small:
+        fx = np.abs((xx-cx)/rx)
+        fy = (yy-cy)/ry
+        reflection = np.clip((1.30-fx)/.25, 0, 1)
+        reflection *= np.clip((fy-.70)/.16, 0, 1)
+        reflection *= np.clip((3.00-fy)/.30, 0, 1)
+        shadow = np.exp(-((xx-cx+.7*rx)/(1.55*rx))**4)
+        shadow *= np.exp(-((yy-cy-1.08*ry)/(.32*ry))**4)
+    else:
+        fx = np.abs((xx-cx-.10*rx)/rx)
+        fy = (yy-cy)/ry
+        reflection = np.clip((1.15-fx)/.23, 0, 1)
+        reflection *= np.clip((fy-.80)/.16, 0, 1)
+        reflection *= np.clip((2.18-fy)/.24, 0, 1)
+        shadow = np.exp(-((xx-cx+.85*rx)/(1.85*rx))**4)
+        shadow *= np.exp(-((yy-cy-1.00*ry)/(.25*ry))**4)
+    support = np.maximum(sphere, np.maximum(reflection, shadow))
+    delta = image.astype(float)-plate
+    # Compression/grain is not a moving object. Preserve only coherent residuals
+    # outside the actual sphere, leaving the source's fine interior texture exact.
+    noiseweight = np.clip((np.max(np.abs(delta), axis=2)-3)/9, 0, 1)
+    weight = support * np.maximum(noiseweight, sphere)
+    return delta*weight[:, :, None]
+
+
+def warp_delta(delta, oldx, oldy, oldrx, oldry, newx, newy, newrx, newry):
+    yy, xx = np.indices(delta.shape[:2], dtype=float)
+    sx = oldx+(xx-newx)*oldrx/newrx
+    sy = oldy+(yy-newy)*oldry/newry
+    coords = np.array([sy, sx])
+    return np.stack([map_coordinates(delta[:, :, c], coords, order=1,
+                                     mode='constant', cval=0, prefilter=False)
+                     for c in range(3)], axis=2)
+
+
+def generate():
+    source = np.load(ROOT/'source_pixels.npz')['rgb']
+    bx, small, target_b, target_s, contact = fit_dynamics(source)
+    plate = background_plate(source)
+    frames = source.copy()
+    first_changed = int(np.floor(contact))+1
+    # The factual pre-contact path is exactly preserved: equal acceleration and
+    # identical initial velocity despite the increased mass.
+    for t in range(first_changed, len(source)):
+        new_bx, new_sx = target_b[t], target_s[t]
+        # Match spatial travel to source texture samples to preserve rolling and
+        # world-position-dependent lighting. Beyond the original resting point,
+        # the nearest late moving sample supplies the original material pixels.
+        bank_big = np.arange(15, 34)
+        tb = int(bank_big[np.argmin(abs(bx[bank_big]-new_bx))])
+        if new_bx >= bx[14]:
+            tb = 14
+        bank_small = np.arange(14, 40)
+        ts = int(bank_small[np.argmin(abs(small[bank_small, 0]-new_sx))])
+
+        old_bx = bx[tb]
+        old_by = y_big(old_bx)
+        old_br = radius_big(old_bx)
+        new_br = radius_big(new_bx)
+        db = layer_delta(source[tb], plate, old_bx, old_by, *old_br)
+        # Tiny turquoise body is in front at contact; suppress contamination of
+        # the amber template by the neighboring marble in the contact sample.
+        if tb == 14:
+            yy, xx = np.indices(plate.shape[:2])
+            blue = (source[tb, :, :, 2].astype(float) > source[tb, :, :, 0]+25)
+            blue &= (abs(xx-small[tb, 0])<7) & (abs(yy-small[tb, 1])<7)
+            db[blue] = 0
+        db = warp_delta(db, old_bx, old_by, *old_br,
+                        new_bx, y_big(new_bx), *new_br)
+
+        old_sx, old_sy = small[ts]
+        sr = (5.78-.0032*old_sx, 5.33-.0022*old_sx)
+        nr = (5.78-.0032*new_sx, 5.33-.0022*new_sx)
+        ds = layer_delta(source[ts], plate, old_sx, old_sy, *sr, small=True)
+        ds = warp_delta(ds, old_sx, old_sy, *sr,
+                        new_sx, old_sy+y_small(new_sx)-y_small(old_sx), *nr)
+        frames[t] = np.clip(np.rint(plate+db+ds), 0, 255).astype(np.uint8)
+    artifact_io.save(frames, ROOT)
+    np.savez_compressed(ROOT/'motion_diagnostics.npz',
+                        big_original=bx, small_original=small,
+                        big_counterfactual=target_b, small_counterfactual=target_s)
+    print('Saved', frames.shape, 'uint8 RGB frames to', ROOT)
+
+
+if __name__ == '__main__':
+    generate()

Only local pixels, array resampling and a fitted two-body impulse model are used.

ROOT = Path(__file__).resolve().parent


def distance(u, speed, friction):
    u = np.clip(u, 0, speed / friction)
    return speed * u - .5 * friction * u * u


def measure_small(source):
    yy, xx = np.indices(source.shape[1:3])
    centers = []
    for im in source:
        a = im.astype(float)
        mask = ((a[:, :, 2] > a[:, :, 0] + 30) &
                (a[:, :, 1] > a[:, :, 0] + 30) & (yy > 62) & (yy < 82))
        centers.append([xx[mask].mean(), yy[mask].mean()])
    return np.asarray(centers)


def fit_dynamics(source):
    # Upper silhouette centers measured numerically from the amber marble.
    big_x = np.array([
        224.3, 218.2, 211.8, 205.4, 199.3, 192.9, 186.9, 180.7,
        174.6, 168.7, 163., 157.3, 151.8, 146.4, 142., 137.7,

        107.7, 105.6, 103.8, 102.2, 100.9, 99.8, 99., 98.5,
        98.2, 98.2, 98.2, 98.2, 98.2, 98.2, 98.2, 98.2])
    big_x = np.r_[big_x, np.full(56, 98.2)]
    small = measure_small(source)
    incoming = np.polyfit(np.arange(5, 14), big_x[5:14], 2)

    def residual(p):
        contact, speed_big, drag_big, speed_small, drag_small = p
        t = np.arange(14, 40)
        xb = np.polyval(incoming, contact) - distance(t-contact, speed_big, drag_big)
        xs = small[0, 0] - distance(t-contact, speed_small, drag_small)
        return np.r_[xb-big_x[14:40], xs-small[14:40, 0]]

    result = least_squares(residual, [13.6, 4.5, .226, 9., .34],
                           bounds=([13.1, 3, .1, 7, .2], [14., 6, .4, 11, .5]))
    contact, vb, ab, vs, ass = result.x
    vi = -np.polyval(np.polyder(incoming), contact)
    # Momentum and relative separation speed identify the original mass ratio
    # and restitution. Increase just the large body's mass by five.
    original_ratio = vs / (vi-vb)
    restitution = (vs-vb) / vi
    ratio = 5 * original_ratio
    vb_new = vi * (ratio-restitution) / (ratio+1)
    vs_new = vi * (1+restitution) * ratio / (ratio+1)
    t = np.arange(len(source))

def layer_delta(image, plate, cx, cy, rx, ry, small=False):
    \"\"\"Capture the marble plus its original antialiasing, shadow and reflection.

    An elliptical support retains the source sphere. A separate tapered support
    carries the floor light changes without moving the background itself.
    \"\"\"
    yy, xx = np.indices(plate.shape[:2], dtype=float)
    nx = (xx-cx)/rx
    ny = (yy-cy)/ry
    radial = np.sqrt(nx*nx + ny*ny)
    # Include the antialias ring and its subtle dark outline intact.
    sphere = np.clip((1.18-radial)/.13, 0, 1)
    if small:
        fx = np.abs((xx-cx)/rx)
        fy = (yy-cy)/ry
        reflection = np.clip((1.30-fx)/.25, 0, 1)
        reflection *= np.clip((fy-.70)/.16, 0, 1)
        reflection *= np.clip((3.00-fy)/.30, 0, 1)
        shadow = np.exp(-((xx-cx+.7*rx)/(1.55*rx))**4)
        shadow *= np.exp(-((yy-cy-1.08*ry)/(.32*ry))**4)
    else:
        fx = np.abs((xx-cx-.10*rx)/rx)
        fy = (yy-cy)/ry
        reflection = np.clip((1.15-fx)/.23, 0, 1)
        reflection *= np.clip((fy-.80)/.16, 0, 1)
        reflection *= np.clip((2.18-fy)/.24, 0, 1)
        shadow = np.exp(-((xx-cx+.85*rx)/(1.85*rx))**4)
        shadow *= np.exp(-((yy-cy-1.00*ry)/(.25*ry))**4)
    support = np.maximum(sphere, np.maximum(reflection, shadow))
    delta = image.astype(float)-plate
    # Compression/grain is not a moving object. Preserve only coherent residuals
    # outside the actual sphere, leaving the source's fine interior texture exact.
    noiseweight = np.clip((np.max(np.abs(delta), axis=2)-3)/9, 0, 1)
    weight = support * np.maximum(noiseweight, sphere)
    return delta*weight[:, :, None]


def warp_delta(delta, oldx, oldy, oldrx, oldry, newx, newy, newrx, newry):
    yy, xx = np.indices(delta.shape[:2], dtype=float)
    sx = oldx+(xx-newx)*oldrx/newrx
    sy = oldy+(yy-newy)*oldry/newry
    coords = np.array([sy, sx])
    return np.stack([map_coordinates(delta[:, :, c], coords, order=1,
                                     mode='constant', cval=0, prefilter=False)
                     for c in range(3)], axis=2)
