        alpha[(xx<108+d)&(yy<95+.14*d)]=0
        cube_layers.append((a-bg,alpha))

    def warp_layer(layer,dx=0,dy=0,scale=1,origin=(0,0)):
        residual,alpha=layer
        ox,oy=origin
        coords=[oy+(yy-oy-dy)/scale,ox+(xx-ox-dx)/scale]
        al=ndi.map_coordinates(alpha,coords,order=1,mode='constant',cval=0)
        # Premultiplied-alpha RGB avoids edges picking up unrelated floor.
        prem=residual*alpha[:,:,None]
        co=np.stack([ndi.map_coordinates(prem[:,:,k],coords,order=1,mode='constant',cval=0) for k in range(3)],axis=2)
        return co,al

    def apply_interpolated(frame,first,second,frac,trans1,trans2):
        co1,al1=warp_layer(first,**trans1)
        co2,al2=warp_layer(second,**trans2)
        al=al1*(1-frac)+al2*frac
        co=co1*(1-frac)+co2*frac
        # Residuals were measured against the common clean view.
        return frame*(1-al[:,:,None])+bg*al[:,:,None]+co

    out=[]
    impact=12/.7
    trajectory=[]
    for t in range(n):
        q=t*.7 if t<=impact else 12+.7*(t-impact)
        # Coulomb sliding: v -> .7 v implies duration -> .7 and range -> .49.
        elapsed=max(0,t-impact)
        factual_clock=min(95,12+elapsed/.7)
        displacement=.49*np.interp(factual_clock,np.arange(96),cd)
        ci=np.interp(displacement,cd[:21]+np.arange(21)*1e-7,np.arange(21)) if displacement>0 else 0
        ia=int(np.floor(ci));ib=min(21,ia+1);frac=ci-ia
        frame=bg.copy()
        da=displacement-cd[ia];db=displacement-cd[ib]
        frame=apply_interpolated(frame,cube_layers[ia],cube_layers[ib],frac,
            dict(dx=da,dy=.14*da),dict(dx=db,dy=.14*db))
        ia=int(np.floor(q));ib=min(n-1,ia+1);frac=q-ia
        center=centers[ia]*(1-frac)+centers[ib]*frac
        radius=radii[ia]*(1-frac)+radii[ib]*frac
        def trans(i):
            return dict(dx=center[0]-centers[i,0],dy=center[1]-centers[i,1],scale=radius/radii[i],origin=tuple(centers[i]))
        frame=apply_interpolated(frame,ball_layers[ia],ball_layers[ib],frac,trans(ia),trans(ib))
        if t==0:frame=src[0].copy()
        out.append(np.clip(np.rint(frame),0,255).astype(np.uint8))
        trajectory.append([t,q,center[0],center[1],displacement,ci])
    frames=np.stack(out)
    artifact_io.save(frames,ROOT)
    np.savetxt(ROOT/'motion_check.csv',np.array(trajectory),delimiter=',',header='frame,source_ball_clock,ball_x,ball_y,block_displacement,source_block_pose')
    print('Saved',frames.shape,frames.dtype,'initial velocity ratio 0.7; impact frame',impact,'block travel',trajectory[-1][4])

if __name__=='__main__':
    main()
