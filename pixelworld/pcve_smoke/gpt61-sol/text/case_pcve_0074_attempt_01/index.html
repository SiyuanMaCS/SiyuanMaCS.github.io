+def hull_sprite(image, background, bounds, mode):
+    """Extract a subpixel silhouette, filling the die's dark recessed pips."""
+    x0, y0, x1, y1 = bounds
+    ix0 = int(np.floor((x0-X0)*SCALE))

+    else:
+        mask = difference>35

+        if num:
+            counts = np.bincount(lab.ravel()); counts[0]=0
+            mask = lab == counts.argmax()

+    points = np.column_stack((x,y)).astype(np.int32)
+    hull = cv2.convexHull(points)
+    alpha = np.zeros(a.shape[:2], np.uint8)
+    cv2.fillConvexPoly(alpha, hull, 255)
+    # Source already has high-resolution antialiasing; a fraction-pixel soft
+    # boundary avoids halos without changing the die's white bevels.
+    alpha = cv2.GaussianBlur(alpha.astype(np.float32)/255, (3,3), .48)
+    yy,xx=np.where(alpha>.03)
+    xa,xb,ya,yb=xx.min(),xx.max()+1,yy.min(),yy.max()+1
+    alpha=alpha[ya:yb,xa:xb]

+    left=X0+(ix0+xa)/SCALE
+    right=X0+(ix0+xb)/SCALE
+    top=(iy0+ya)/SCALE
+    bottom=(iy0+yb)/SCALE
+    return dict(rgb=texture, alpha=alpha, left=left, right=right,
+                top=top, bottom=bottom)
+

+def warped_sprite(sprite, bottom, width_ratio=1., height_ratio=1., center=None):
+    """Place a source texture with an affine projection, retaining fine pips."""
+    if center is None:
+        center=(sprite['left']+sprite['right'])/2
+    width=(sprite['right']-sprite['left'])*width_ratio
+    height=(sprite['bottom']-sprite['top'])*height_ratio
+    left=center-width/2
+    top=bottom-height
+    mat=np.array([[width_ratio,0,(left-X0)*SCALE],
+                  [0,height_ratio,top*SCALE]], dtype=np.float32)
+    h,w=Y1*SCALE,(X1-X0)*SCALE

+    premul=cv2.warpAffine(sprite['rgb']*sprite['alpha'][...,None],mat,
+                         (w,h),flags=cv2.INTER_LINEAR)
+    return premul,alpha

    travel=(floor_y-observed[0])/(1-perspective*(floor_y-observed[0]))
    contact_original=(-velocity+np.sqrt(velocity**2+2*gravity*travel))/gravity
    restitution=gravity*(15.05-contact_original)/(2*(velocity+gravity*contact_original))
    v=3*velocity
    contact=(-v+np.sqrt(v*v+2*gravity*travel))/gravity
    incoming=v+gravity*contact
    restitution=float(np.clip(restitution,.48,.55))

    def red_position(time):
        if time<=contact:
            d=v*time+.5*gravity*time*time
        else:
            elapsed=time-contact
            rebound=incoming*restitution
            # Dissipative floor impacts, retaining the observed coefficient.
            while elapsed>2*rebound/gravity and rebound>.35:
                elapsed-=2*rebound/gravity
                rebound*=restitution
            if rebound<=.35:
                d=travel
            else:
                d=travel-rebound*elapsed+.5*gravity*elapsed*elapsed
        return float(observed[0]+d/(1+perspective*d))

    yy,xx=np.mgrid[0:Y1*SCALE,0:(X1-X0)*SCALE].astype(np.float32)
    xx=xx/SCALE+X0; yy=yy/SCALE
    # Shadows are fitted from visible, unoccluded floor pixels. Directional
    # light makes raised shadows shift back across the tabletop.
    surface=(background[...,1]>background[...,0]*1.06)& \\
            (background[...,2]>background[...,1]*.73)
    surface=cv2.GaussianBlur(surface.astype(np.float32),(5,5),.9)
    smallbg=np.asarray(Image.fromarray(background.astype(np.uint8)).resize(
                       (X1-X0,Y1),Image.Resampling.LANCZOS)).astype(float)
    ys,xs=np.mgrid[:Y1,X0:X1].astype(float)
    def fit_shadow(frame,box,guess,blocked):
        target=factual[frame,:Y1,X0:X1].astype(float)
        dark=np.clip((smallbg-target).mean(axis=-1)/(smallbg.mean(axis=-1)+1),-.05,.4)
        xlo,ylo,xhi,yhi=box
        valid=(xs>=xlo)&(xs<=xhi)&(ys>=ylo)&(ys<=yhi)&blocked
        def gauss(p):
            amp,cx,cy,sx,sy=p
            return amp*np.exp(-.5*((xs-cx)/sx)**2-.5*((ys-cy)/sy)**2)
        opt=least_squares(lambda p:(gauss(p)-dark)[valid],guess,
                          bounds=([.02,guess[1]-4,guess[2]-3,3,.8],
                                  [.4,guess[1]+4,guess[2]+3,16,4.5]),
                          loss='soft_l1',f_scale=.02)
        return opt.x
    red_visible=~((xs>117)&(xs<130)&(ys<93))
    red_shadow=fit_shadow(15,(104,86,146,98),[.16,123,92,8.5,2.1],red_visible)
    white_visible=(xs>139)|(ys<71)|(ys>85)
    white_shadow=fit_shadow(27,(128,77,153,86),[.10,136,82,7.,2.],white_visible)

    def shadow(parameters,height):
        amp,cx,cy,sx,sy=parameters
        cx-=.19*height;cy-=.72*height
        sx+=.025*height;sy+=.02*height
        amp*=np.exp(-height/45)
        return amp*np.exp(-.5*((xx-cx)/sx)**2-.5*((yy-cy)/sy)**2)*surface

    output=[]
    trajectory=[]
    for t in range(96):
        bottom=red_position(float(t))
        trajectory.append(bottom)
        if t==0:
            # At the intervention the position/orientation/history is unchanged.
            output.append(factual[0].copy())
            continue
        canvas=background.copy()
        red_height=max(0,floor_y-bottom)
        white_height=max(0,85.0-white_bottom[t])
        shade=1-(1-shadow(red_shadow,red_height))*(1-shadow(white_shadow,white_height))
        canvas*=1-shade[...,None]

        # White: original timing, projected geometry, materials and pips.
        if t in white_sprites:
            ws=white_sprites[t]
            wp,wa=warped_sprite(ws,ws['bottom'])
        else:
            target_bottom=white_bottom[t]+.05
            actual_height=white_bottom[t]-white_top[t]
            donor_height=white_donor['bottom']-white_donor['top']
            ratio=actual_height/donor_height
            # Perspective dimensions remain nearly fixed during small bounces.
            ratio=float(np.clip(ratio,.94,1.045))
            wp,wa=warped_sprite(white_donor,target_bottom,
                                width_ratio=ratio,height_ratio=ratio,
                                center=132.15)
            rgb=np.divide(wp,np.maximum(wa[...,None],1e-5))
            old_bot=red_bottom[min(t,27)]
            _,old_red=warped_sprite(red_shape,old_bot)
            # Dilate the red bevel to exclude its antialiased outline.
            occluded=cv2.dilate((old_red>.01).astype(np.uint8),
                                np.ones((5,5),np.uint8)).astype(bool)
            rgb[~occluded]=original[t][~occluded]
            wp=rgb*wa[...,None]
        canvas=canvas*(1-wa[...,None])+wp

        # Height-based source interpolation preserves projected lighting and
        # face geometry rather than using a newly drawn substitute for the die.
        knots=np.array([observed[0],36.2,46.6,58.6,71.8,86.,89.8,floor_y])
        k=int(np.clip(np.searchsorted(knots,bottom)-1,0,len(knots)-2))
        blend=float(np.clip((bottom-knots[k])/(knots[k+1]-knots[k]),0,1))
        premul=np.zeros_like(canvas);alpha=np.zeros(canvas.shape[:2],np.float32)
        for idx,weight in [(k,1-blend),(k+1,blend)]:
            if weight<1e-6:continue
            rs=red_sprites[idx]
            # The camera's foreshortening varies with height, not elapsed time.
            true_bottom=knots[idx]
            ratio=1+(true_bottom-bottom)*.0011
            p,a=warped_sprite(rs,bottom+.08,width_ratio=ratio,
                              height_ratio=ratio,center=123.25)
            premul+=weight*p;alpha+=weight*a
        canvas=canvas*(1-alpha[...,None])+premul

        patch=np.asarray(Image.fromarray(np.clip(canvas,0,255).astype(np.uint8)).resize(
                         (X1-X0,Y1),Image.Resampling.LANCZOS))
        frame=factual[t].copy()
        # Feather only the far margin, outside both dies and their shadows.
        feather=np.ones((Y1,X1-X0),np.float32)
        feather[:,:2]=np.array([0,.65]);feather[:,-2:]=np.array([.65,0])
        feather[-2:]=np.array([.65,0])[:,None]
        region=frame[:Y1,X0:X1].astype(float)
        frame[:Y1,X0:X1]=np.clip(region*(1-feather[...,None])+
                                        patch*feather[...,None],0,255).astype(np.uint8)
        output.append(frame)
    frames=np.asarray(output,dtype=np.uint8)
    artifact_io.save(frames,ROOT)
    print('Saved',frames.shape,'RGB frames at 24 fps')
    print('Original fitted v/g/perspective:',fit.x)
    print('Floor restitution:',restitution,'new impact frame:',contact)
    print('Red bottom pixels, first 36:',np.round(trajectory[:36],2).tolist())
    print('Shadow fits:',red_shadow,white_shadow)
    print('Initial frame exact:',np.array_equal(frames[0],factual[0]))


if __name__=='__main__':
