+"""Counterfactual rigid-body edit, using source RGB, geometry and impulse math.
+
+The scene is fixed-camera. Its three moving mallets are composited from their
+actual RGB surfaces, including soft contact shadows. A projective coordinate
+makes the measured near-constant world velocities linear. Only the first
+material's coefficient of restitution changes; later collisions are integrated.
+"""
+from pathlib import Path
+import json
+import cv2
+import numpy as np
+from PIL import Image
+from scipy.ndimage import binary_fill_holes
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+FPS = 24
+VANISH_X = 379.18
+WORLD_SCALE = 2.974e-5
+
+
+def world_x(x):
+    return (1 / (VANISH_X - 167.83983849) - 1 / (VANISH_X - x)) / WORLD_SCALE
+
+
+def screen_x(q):
+    return VANISH_X - 1 / (1 / (VANISH_X - 167.83983849) - WORLD_SCALE * q)
+
+
+def component(mask):
+    n, labs, stats, centers = cv2.connectedComponentsWithStats(mask.astype(np.uint8), 8)
+    k = 1 + np.argmax(stats[1:, 4])
+    return (labs == k).astype(np.uint8), centers[k]
+
+
+def expand(core, radius=7):
+    k = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (2 * radius + 1, 2 * radius + 1))
+    return cv2.dilate(core, k)
+
+
+def load_source():
+    video = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    full = []
+    while True:
+        ok, frame = video.read()
+        if not ok:
+            break
+        full.append(cv2.cvtColor(frame, cv2.COLOR_BGR2RGB))
+    video.release()
+    return np.asarray(full), np.load(ROOT / 'source_pixels.npz')['rgb']
+
+
+def object_cores(frame):
+    r, g, b = frame.astype(np.float32).transpose(2, 0, 1)
+    blue = (b > r * 1.3) & (b > g * 1.15) & (b > 75)
+    red = (r > g * 1.5) & (r > b * 1.4) & (r > 110)
+    blue[:240] = False
+    blue[365:] = False
+    blue[:, :650] = False
+    blue[:, 890:] = False
+    red[:285] = False
+    red[435:] = False
+    red[:, :475] = False
+    red[:, 755:] = False
+    bc, bp = component(blue)
+    rc, rp = component(red)
+    return bc, rc, bp / 5, rp / 5
+
+
+def silver_center(t):
+    # Measured image-space factual silver trajectory. Its material is unchanged.
+    xs = [105.772] * 30 + [105.14, 103.28, 101.39, 99.50, 97.65,
+          95.79, 93.83, 91.93, 89.82, 87.60, 85.59, 83.68, 81.57,
+          79.33, 77.07, 74.83, 72.62, 70.40, 68.20]
+    x = xs[t] if t < len(xs) else 67.8
+    return np.array([x, 81.161 + (105.772 - x) * .382])
+
+
+def make_background(full, cores, anchors):
+    """Temporal clean-plate recovery; never replace static scene geometry."""
+    h, w = full.shape[1:3]
+    # Accumulate only in the compact region that the mallets traverse.
+    x0, x1, y0, y1 = 290, 890, 235, 520
+    sums = np.zeros((y1 - y0, x1 - x0, 3), np.float64)
+    counts = np.zeros(sums.shape[:2], np.int32)
+    yy, xx = np.indices((h, w))
+    for t in range(0, 96, 2):
+        bad = np.zeros((h, w), np.uint8)
+        for core in cores[t]:
+            outer = expand(core, 9)
+            # Direction and extent of the source's very soft shadows.
+            shifted = cv2.warpAffine(outer, np.float32([[1, 0, 3], [0, 1, 6]]),
+                                    (w, h), flags=cv2.INTER_NEAREST)
+            bad |= outer | shifted
+        sx, sy = silver_center(t) * 5
+        scale = (VANISH_X - sx / 5) / (VANISH_X - 105.772)
+        silver_bad = ((xx - sx) / (47 * scale)) ** 2 + ((yy - sy) / (36 * scale)) ** 2 < 1
+        bad |= silver_bad.astype(np.uint8)
+        good = bad[y0:y1, x0:x1] == 0
+        sums += full[t, y0:y1, x0:x1].astype(np.float64) * good[..., None]
+        counts += good
+    plate = np.rint(sums / np.maximum(counts[..., None], 1)).astype(np.uint8)
+    # Rare pixels occupied for the entire factual timeline are reconstructed
+    # locally from the neighboring floor, not from any external image.
+    holes = (counts == 0).astype(np.uint8) * 255
+    if holes.any():

+    recovered = full[0].copy()
+    recovery_region = np.zeros((h, w), np.uint8)
+    for core in cores[0]:
+        outer = expand(core, 13)
+        recovery_region |= outer
+        recovery_region |= cv2.warpAffine(outer, np.float32([[1, 0, 3], [0, 1, 7]]),
+                                         (w, h), flags=cv2.INTER_NEAREST)
+    sx, sy = anchors[0, 2] * 5
+    recovery_region |= ((((xx - sx) / 48) ** 2 + ((yy - sy) / 39) ** 2) < 1).astype(np.uint8)

+    a = alpha[y0:y1, x0:x1, None]

+    print('Clean plate: permanently covered pixels:', int(holes.sum() / 255))
+    return recovered

+def sprite(full, background, t, anchor, kind, cores):
+    # Native-resolution source body, with premultiplied photometric residual.
+    cx, cy = anchor * 5
+    x0, x1 = int(cx) - 65, int(cx) + 66
+    y0, y1 = int(cy) - 55, int(cy) + 61
+    patch = full[t, y0:y1, x0:x1].astype(np.float32)
+    base = background[y0:y1, x0:x1].astype(np.float32)
+    if kind in (0, 1):
+        core = cores[t][kind][y0:y1, x0:x1]
+    else:
+        r, g, b = patch.transpose(2, 0, 1)
+        core, _ = component((np.abs(r - g) < 15) & (b - g < 15) & (g > 210))
+    core = binary_fill_holes(cv2.morphologyEx(core, cv2.MORPH_CLOSE,
+                                            np.ones((3, 3), np.uint8))).astype(np.uint8)
+    outer = expand(core, 11)
+    shadow = cv2.warpAffine(outer, np.float32([[1, 0, 3], [0, 1, 6]]),
+                            (outer.shape[1], outer.shape[0]), flags=cv2.INTER_NEAREST)
+    alpha = cv2.GaussianBlur(np.maximum(outer, shadow).astype(np.float32), (7, 7), 1.1)
+    residual = patch - base
+    # Avoid carrying codec noise into otherwise unchanged floor texture.
+    delta = np.max(np.abs(residual), axis=2)
+    confidence = np.clip((delta - 3) / 8, 0, 1)
+    residual *= (alpha * confidence)[..., None]
+    return {'pixels': residual, 'anchor': np.array([cx - x0, cy - y0]),
+            'screen': anchor.copy()}
+
+
+def composite(canvas, obj, target):
+    reference = obj['screen']
+    scale = (VANISH_X - target[0]) / (VANISH_X - reference[0])
+    arr = obj['pixels']
+    anchor = obj['anchor']
+    # Warp just the small sprite ROI rather than resampling the whole scene.
+    left = int(np.floor(target[0] * 5 - anchor[0] * scale)) - 2
+    top = int(np.floor(target[1] * 5 - anchor[1] * scale)) - 2
+    width = int(np.ceil(arr.shape[1] * scale)) + 5
+    height = int(np.ceil(arr.shape[0] * scale)) + 5
+    matrix = np.float32([[scale, 0, target[0] * 5 - anchor[0] * scale - left],
+                         [0, scale, target[1] * 5 - anchor[1] * scale - top]])
+    transformed = cv2.warpAffine(arr, matrix, (width, height), flags=cv2.INTER_LINEAR)
+    xa, xb = max(0, left), min(canvas.shape[1], left + width)
+    ya, yb = max(0, top), min(canvas.shape[0], top + height)
+    canvas[ya:yb, xa:xb] += transformed[ya - top:yb - top, xa - left:xb - left]
+
+
+def dynamics(initial, factual_at_14):
+    """Equal effective masses; restitution acts on contact-relative velocity."""
+    q = np.array([world_x(factual_at_14[0, 0]), world_x(initial[1, 0]), world_x(initial[2, 0])])
+    velocities = np.array([1.0, 0.0, 0.0])
+    restitution_original = .94
+    restitution_blue = .3 * restitution_original
+    restitutions = [restitution_blue, restitution_original]
+    positions = np.full((96, 3), np.nan)
+    drag = .002
+    substeps = 64
+    dt = 1 / substeps
+    contacts = []
+    for frame in range(14, 96):
+        positions[frame] = q
+        for sub in range(substeps):
+            velocities -= np.sign(velocities) * np.minimum(np.abs(velocities), drag * dt)
+            q += velocities * dt
+            for iteration in range(5):
+                for i in (0, 1):
+                    separation = 3.85
+                    overlap = separation - (q[i + 1] - q[i])
+                    if overlap > 0:
+                        q[i] -= overlap / 2
+                        q[i + 1] += overlap / 2
+                        relative = velocities[i] - velocities[i + 1]
+                        if relative > 0:
+                            impulse = (1 + restitutions[i]) * relative / 2
+                            velocities[i] -= impulse
+                            velocities[i + 1] += impulse
+                            if relative > .08:
+                                contacts.append([round(frame + sub * dt, 3), i, round(float(relative), 4)])
+                # The silver mallet reaches the same near rail as in the source.
+                wall_q = world_x(67.75)
+                if q[2] > wall_q:
+                    q[2] = wall_q
+                    if velocities[2] > 0:
+                        velocities[2] = 0
+    print('Collision times (frame, adjacent pair, relative speed):', contacts)
+    return positions, contacts, restitution_original, restitution_blue
+
+
+def main():
+    full, original = load_source()
+    assert original.shape == (96, 144, 256, 3)
+    cores, anchors = [], []
+    for t, f in enumerate(full):
+        bc, rc, bp, rp = object_cores(f)
+        cores.append((bc, rc))
+        anchors.append([bp, rp, silver_center(t)])
+    anchors = np.array(anchors)
+    anchors[0, 2] = [105.77203791, 81.16137441]
+    background = make_background(full, cores, anchors)
+    # Separated factual poses give unoccluded material and complete shadows.
+    blue = sprite(full, background, 22, anchors[22, 0], 0, cores)
+    red = sprite(full, background, 0, anchors[0, 1], 1, cores)
+    silver = sprite(full, background, 0, anchors[0, 2], 2, cores)
+    q, contacts, e0, eblue = dynamics(anchors[0], anchors[14])
+    frames = original.copy()
+    trajectories = []
+    for t in range(15, 96):
+        xs = screen_x(q[t])
+        blue_y = blue['screen'][1] + (blue['screen'][0] - xs[0]) * .36
+        red_y = red['screen'][1] + (red['screen'][0] - xs[1]) * .3784
+        silver_y = silver['screen'][1] + (silver['screen'][0] - xs[2]) * .382
+        targets = np.array([[xs[0], blue_y], [xs[1], red_y], [xs[2], silver_y]])
+        canvas = background.astype(np.float32)
+        # Correct back-to-front order in this oblique fixed camera.
+        for obj, target in zip((blue, red, silver), targets):
+            composite(canvas, obj, target)
+        native = np.clip(np.rint(canvas), 0, 255).astype(np.uint8)
+        frames[t] = np.array(Image.fromarray(native).resize((256, 144), Image.Resampling.LANCZOS))
+        trajectories.append({'frame_1based': t + 1, 'centers_xy': targets.round(4).tolist()})
+    artifact_io.save(frames, ROOT)
+    (ROOT / 'generation_details.json').write_text(json.dumps({
+        'source_history_preserved_through_frame_1based': 15,
+        'restitution_original_estimated': e0,
+        'restitution_blue_edited': eblue,
+        'collision_events_frame_0based': contacts,
+        'trajectories': trajectories,
+        'method': 'source RGB clean plate, native-resolution residual sprites, projective coordinates and contact impulses'
+    }, indent=2))
+    print('Saved', frames.shape, frames.dtype, 'to', ROOT)
+
+
+if __name__ == '__main__':
+    main()

        counts += good

    # Rare pixels occupied for the entire factual timeline are reconstructed
    # locally from the neighboring floor, not from any external image.
    holes = (counts == 0).astype(np.uint8) * 255
    if holes.any():
        plate = cv2.inpaint(plate, holes, 9, cv2.INPAINT_TELEA)
    recovered = full[0].copy()
    recovery_region = np.zeros((h, w), np.uint8)
    for core in cores[0]:
        outer = expand(core, 13)
        recovery_region |= outer
        recovery_region |= cv2.warpAffine(outer, np.float32([[1, 0, 3], [0, 1, 7]]),
                                         (w, h), flags=cv2.INTER_NEAREST)

    recovery_region |= ((((xx - sx) / 48) ** 2 + ((yy - sy) / 39) ** 2) < 1).astype(np.uint8)

    a = alpha[y0:y1, x0:x1, None]
    recovered[y0:y1, x0:x1] = np.rint(full[0, y0:y1, x0:x1] * (1 - a) + plate * a).astype(np.uint8)

def sprite(full, background, t, anchor, kind, cores):
    # Native-resolution source body, with premultiplied photometric residual.
    cx, cy = anchor * 5
    x0, x1 = int(cx) - 65, int(cx) + 66
    y0, y1 = int(cy) - 55, int(cy) + 61
    patch = full[t, y0:y1, x0:x1].astype(np.float32)
    base = background[y0:y1, x0:x1].astype(np.float32)
    if kind in (0, 1):
        core = cores[t][kind][y0:y1, x0:x1]
    else:
        r, g, b = patch.transpose(2, 0, 1)
        core, _ = component((np.abs(r - g) < 15) & (b - g < 15) & (g > 210))
    core = binary_fill_holes(cv2.morphologyEx(core, cv2.MORPH_CLOSE,
                                            np.ones((3, 3), np.uint8))).astype(np.uint8)
    outer = expand(core, 11)
    shadow = cv2.warpAffine(outer, np.float32([[1, 0, 3], [0, 1, 6]]),
                            (outer.shape[1], outer.shape[0]), flags=cv2.INTER_NEAREST)
    alpha = cv2.GaussianBlur(np.maximum(outer, shadow).astype(np.float32), (7, 7), 1.1)
    residual = patch - base
    # Avoid carrying codec noise into otherwise unchanged floor texture.
    delta = np.max(np.abs(residual), axis=2)
    confidence = np.clip((delta - 3) / 8, 0, 1)
    residual *= (alpha * confidence)[..., None]
    return {'pixels': residual, 'anchor': np.array([cx - x0, cy - y0]),
            'screen': anchor.copy()}


def composite(canvas, obj, target):
    reference = obj['screen']
    scale = (VANISH_X - target[0]) / (VANISH_X - reference[0])
    arr = obj['pixels']
    anchor = obj['anchor']
    # Warp just the small sprite ROI rather than resampling the whole scene.
    left = int(np.floor(target[0] * 5 - anchor[0] * scale)) - 2
    top = int(np.floor(target[1] * 5 - anchor[1] * scale)) - 2
    width = int(np.ceil(arr.shape[1] * scale)) + 5
    height = int(np.ceil(arr.shape[0] * scale)) + 5
    matrix = np.float32([[scale, 0, target[0] * 5 - anchor[0] * scale - left],
                         [0, scale, target[1] * 5 - anchor[1] * scale - top]])
    transformed = cv2.warpAffine(arr, matrix, (width, height), flags=cv2.INTER_LINEAR)
    xa, xb = max(0, left), min(canvas.shape[1], left + width)
    ya, yb = max(0, top), min(canvas.shape[0], top + height)
    canvas[ya:yb, xa:xb] += transformed[ya - top:yb - top, xa - left:xb - left]


def dynamics(initial, factual_at_14):
    \"\"\"Equal effective masses; restitution acts on contact-relative velocity.\"\"\"
    q = np.array([world_x(factual_at_14[0, 0]), world_x(initial[1, 0]), world_x(initial[2, 0])])
    velocities = np.array([1.0, 0.0, 0.0])
    restitution_original = .94
    restitution_blue = .3 * restitution_original
    restitutions = [restitution_blue, restitution_original]
    positions = np.full((96, 3), np.nan)
    drag = .002
    substeps = 64
    dt = 1 / substeps
    contacts = []
    for frame in range(14, 96):
        positions[frame] = q
        for sub in range(substeps):
            velocities -= np.sign(velocities) * np.minimum(np.abs(velocities), drag * dt)
            q += velocities * dt
            for iteration in range(5):
                for i in (0, 1):
                    separation = 3.85
                    overlap = separation - (q[i + 1] - q[i])
                    if overlap > 0:
                        q[i] -= overlap / 2
                        q[i + 1] += overlap / 2
                        relative = velocities[i] - velocities[i + 1]
                        if relative > 0:
                            impulse = (1 + restitutions[i]) * relative / 2
                            velocities[i] -= impulse
                            velocities[i + 1] += impulse
                            if relative > .08:
                                contacts.append([round(frame + sub * dt, 3), i, round(float(relative), 4)])
                # The silver mallet reaches the same near rail as in the source.

                    q[2] = wall_q
                    if velocities[2] > 0:
                        velocities[2] = 0
    print('Collision times (frame, adjacent pair, relative speed):', contacts)
    return positions, contacts, restitution_original, restitution_blue
