+"""Counterfactual low-mass bowling ball, rendered using source RGB cutouts.
+
+The mass change does not change free rolling velocity. At the rack, a ball
+with one fifth the inertia transfers a much smaller impulse: the two directly
+contacted pins tumble locally, rather than shooting into the gutters, while
+the ball loses its forward motion before ploughing through the centre pin.
+All image material is extracted from the supplied source video.
+"""
+from pathlib import Path
+import numpy as np
+import cv2
+from scipy import ndimage
+from PIL import Image
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+F = np.load(ROOT / 'source_pixels.npz')['rgb']
+H, W = 720, 1280
+
+
+def original_frames():
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    values = []
+    while True:
+        ok, a = cap.read()
+        if not ok:
+            break
+        values.append(a[:, :, ::-1].copy())
+    cap.release()
+    return values
+
+
+def composite(dst, rgb, alpha, matrix):
+    """Subpixel premultiplied-alpha cutout warp, without dark edge halos."""
+    a = cv2.warpAffine(alpha.astype(np.float32), matrix, (W, H),
+                       flags=cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)
+    p = cv2.warpAffine((rgb * alpha[..., None]).astype(np.float32), matrix,
+                      (W, H), flags=cv2.INTER_LINEAR,
+                      borderMode=cv2.BORDER_CONSTANT)
+    return dst * (1-a[..., None]) + p
+
+
+def pin_cutout(a, first=False):
+    # Segment the left pin against the dark back wall. Only numerical RGB
+    # predicates and component topology are used (no model or external mask).
+    box = a[352:391, 578:635].astype(np.float32)
+    r, g, b = box.transpose(2, 0, 1)
+    m = (box.mean(2) > 24) & ~((b > r*1.45) & (b > g*1.15) & (b > 35))
+    if first:
+        m[:, :45] = False
+    labels, n = ndimage.label(m)
+    sizes = np.bincount(labels.ravel()); sizes[0] = 0
+    mask = labels == sizes.argmax()
+    mask = ndimage.binary_fill_holes(ndimage.binary_closing(mask))
+    yy, xx = np.where(mask)
+    cx, cy = xx.mean()+578, yy.mean()+352
+    # Include the source's existing antialiased edge and recover premultiplied
+    # colour. The outside of the cutout was almost black in the original.
+    outer = ndimage.binary_dilation(mask)
+    alpha = np.where(mask, 1.0, np.minimum(box.mean(2)/75, .85)) * outer
+    rgba_rgb = box / np.maximum(alpha[..., None], .02)
+    rgba_rgb = np.clip(rgba_rgb, 0, 255)
+    return rgba_rgb, alpha.astype(np.float32), cx, cy
+
+
+def shadow(dst, x, y, rx, ry, strength):
+    yy, xx = np.mgrid[:H, :W]
+    a = strength*np.exp(-2*((xx-x)/rx)**2-2*((yy-y)/ry)**2)
+    return dst*(1-a[..., None])
+
+
+def main():
+    originals = original_frames()
+    source0 = originals[0].astype(np.float32)
+    # Clean plate: late-time RGB reveals the black rack wall and floor;
+    # early-time RGB reveals the initially ball-covered stretch of lane.
+    plate = source0.copy()
+    plate[353:397, 622:658] = np.median(
+        np.stack([originals[t][353:397, 622:658] for t in range(78, 96)]), axis=0)
+    plate[397:454, 620:661] = np.median(
+        np.stack([originals[t][397:454, 620:661] for t in range(78, 96)]), axis=0)
+    plate[487:542, 613:666] = originals[12][487:542, 613:666]
+
+    pins = {0: pin_cutout(originals[0], True)}
+    for t in range(38, 46):
+        pins[t] = pin_cutout(originals[t])
+    pin0, pa0, pcx, pcy = pins[0]
+
+    # Ball: source photograph at launch, including the genuine blue material,
+    # highlight and finger-hole texture. Cutout is a physical circle, not a
+    # resized ball: only perspective changes its projected size.
+    ball = source0[486:537, 615:666].copy()
+    yy, xx = np.mgrid[:51, :51]
+    bx0, by0, radius0 = 24.5, 24.9, 21.8
+    rr = np.hypot(xx-bx0, yy-by0)
+    ba = np.clip(radius0+.5-rr, 0, 1).astype(np.float32)
+    # Recover edge colour from the adjacent interior to avoid lane fringes.
+    bad = (ba > 0) & (ba < .95)
+    xi = np.clip(np.rint(bx0+(xx-bx0)*.91).astype(int), 0, 50)
+    yi = np.clip(np.rint(by0+(yy-by0)*.91).astype(int), 0, 50)
+    ball[bad] = ball[yi[bad], xi[bad]]
+
+    # Preserve the centre pin and its authentic reflection as a separate layer.
+    centre = source0[352:397, 635:645]
+    centre_bg = plate[352:397, 635:645]
+    cm = np.clip((centre.mean(2)-centre_bg.mean(2))/24, 0, 1)
+    cm = ndimage.maximum_filter(cm, 3)
+    centre_delta = source0[398:454, 634:646]-plate[398:454, 634:646]
+
+    result = F.copy()
+    for t in range(37, 96):
+        image = plate.copy()
+        dt = (t-37)/24
+        # A small damped rock of the centre pin. It is no longer driven all
+        # the way backward into the pit by the low-inertia ball.
+        rock = .028*np.exp(-dt*2.1)*np.sin(dt*10)
+        ca = np.array([[1, rock, 635-rock*44], [0, 1, 352]], np.float32)
+        image = composite(image, centre, cm, ca)
+        image[398:454, 634:646] += centre_delta
+
+        # The altered impulse is about 0.35 of the factual impulse. Gravity
+        # still topples the outer pins; their flight and lateral slide shrink.
+        phase = min(45., 37.+(t-37)*.64)
+        if phase < 38:
+            choices = [(0, 38-phase), (38, phase-37)]
+        else:
+            lo = int(np.floor(phase)); hi = min(45, lo+1)
+            choices = [(lo, 1-(phase-lo)), (hi, phase-lo)]
+        for side in [-1, 1]:
+            # Source's left pin is mirrored for the matching outer pin. Its
+            # fall orientation and material are preserved from actual pixels.
+            overlays = []
+            for key, blend in choices:
+                if blend <= 0:
+                    continue
+                rgb, alpha, cx, cy = pins[key]
+                # The base receives less sideways impulse; the centre of mass
+                # therefore remains within a few pixels of the rack.
+                newx = pcx+.35*(cx-pcx)
+                newy = cy
+                # Tiny dissipative settling, not an abrupt frozen collision.
+                if t > 52:
+                    newx -= .9*(1-np.exp(-(t-52)/18))
+                    newy += .22*np.exp(-(t-52)/9)*np.sin((t-52)*.7)
+                tx, ty = newx-cx+578, newy-cy+352
+                if side == -1:
+                    mat = np.array([[1, 0, tx], [0, 1, ty]], np.float32)
+                    shadowx = newx
+                else:
+                    mat = np.array([[-1, 0, 1279-tx], [0, 1, ty]], np.float32)
+                    shadowx = 1279-newx
+                # Ground reflection follows the falling pin and becomes
+                # compact as it lies down. The floor's RGB remains visible.
+                reflection = mat.copy()
+                reflection[1] = [0, -.62, 394+(.62*394)-.62*ty]
+                ar = alpha*.25
+                image = composite(image, rgb*.81, ar*blend, reflection)
+                image = shadow(image, shadowx, 392, 11, 2, .025*blend)
+                overlays.append((rgb, alpha*blend, mat))
+            # Interpolation of neighbouring photographed poses is done as
+            # weighted source-over for smooth motion at 24 fps.
+            for rgb, alpha, mat in overlays:
+                image = composite(image, rgb, alpha, mat)
+
+        # Post-contact velocity: small forward penetration during the impulse,
+        # then a mild rebound. The reduced mass is not confused with size.
+        # y is perspective projection of ball centre, radius proportional to
+        # reciprocal depth; a small rebound makes the ball slightly grow.
+        q = t-37
+        y = 77.18 - .40*(1-np.exp(-q/3.5)) + .94*(1-np.exp(-max(q-5, 0)/22))
+        x = 127.9-.16*(1-np.exp(-q/13))
+        rad = 1.80*(y-60.6)/(77.18-60.6)
+        s = rad*5/radius0
+        ballmat = np.array([[s, 0, x*5-s*bx0], [0, s, y*5-s*by0]], np.float32)
+        # Contact shadow and subdued blue reflection are tied to ground.
+        image = shadow(image, x*5, y*5+rad*4.3, rad*5.5, 2.0, .20)
+        refl = ballmat.copy()
+        refl[1] = [0, -s*.56, 2*(y*5+rad*5)-ballmat[1,2]-(1-.56)*y*5]
+        image = composite(image, ball*.78, ba*.16, refl)
+        image = composite(image, ball, ba, ballmat)
+
+        rendered = np.asarray(Image.fromarray(np.clip(image, 0, 255).astype(np.uint8))
+                              .resize((256, 144), Image.Resampling.LANCZOS))
+        # Outside the affected lane and rack, retain source pixels at the
+        # original time (including all texture and minor encoding variation).
+        result[t, 69:110, 111:145] = rendered[69:110, 111:145]
+    artifact_io.save(result, ROOT)
+    print('Saved', result.shape, result.dtype, 'at 24 fps')
+
+
+if __name__ == '__main__':
+    main()

The mass change does not change free rolling velocity. At the rack, a ball
with one fifth the inertia transfers a much smaller impulse: the two directly

import numpy as np
import cv2
from scipy import ndimage
from PIL import Image
import artifact_io

ROOT = Path(__file__).resolve().parent
F = np.load(ROOT / 'source_pixels.npz')['rgb']
H, W = 720, 1280


def original_frames():
    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
    values = []
    while True:
        ok, a = cap.read()
        if not ok:
            break
        values.append(a[:, :, ::-1].copy())
    cap.release()
    return values


def composite(dst, rgb, alpha, matrix):
    \"\"\"Subpixel premultiplied-alpha cutout warp, without dark edge halos.\"\"\"
    a = cv2.warpAffine(alpha.astype(np.float32), matrix, (W, H),
                       flags=cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)
    p = cv2.warpAffine((rgb * alpha[..., None]).astype(np.float32), matrix,
                      (W, H), flags=cv2.INTER_LINEAR,
                      borderMode=cv2.BORDER_CONSTANT)
    return dst * (1-a[..., None]) + p


def pin_cutout(a, first=False):
    # Segment the left pin against the dark back wall. Only numerical RGB
    # predicates and component topology are used (no model or external mask).
    box = a[352:391, 578:635].astype(np.float32)
    r, g, b = box.transpose(2, 0, 1)
    m = (box.mean(2) > 24) & ~((b > r*1.45) & (b > g*1.15) & (b > 35))
    if first:
        m[:, :45] = False
    labels, n = ndimage.label(m)
    sizes = np.bincount(labels.ravel()); sizes[0] = 0
    mask = labels == sizes.argmax()
    mask = ndimage.binary_fill_holes(ndimage.binary_closing(mask))
    yy, xx = np.where(mask)
    cx, cy = xx.mean()+578, yy.mean()+352
    # Include the source's existing antialiased edge and recover premultiplied
    # colour. The outside of the cutout was almost black in the original.
    outer = ndimage.binary_dilation(mask)
    alpha = np.where(mask, 1.0, np.minimum(box.mean(2)/75, .85)) * outer
    rgba_rgb = box / np.maximum(alpha[..., None], .02)
    rgba_rgb = np.clip(rgba_rgb, 0, 255)
    return rgba_rgb, alpha.astype(np.float32), cx, cy


def shadow(dst, x, y, rx, ry, strength):
    yy, xx = np.mgrid[:H, :W]
    a = strength*np.exp(-2*((xx-x)/rx)**2-2*((yy-y)/ry)**2)
    return dst*(1-a[..., None])

    source0 = originals[0].astype(np.float32)
    # Clean plate: late-time RGB reveals the black rack wall and floor;
    # early-time RGB reveals the initially ball-covered stretch of lane.

        np.stack([originals[t][353:397, 622:658] for t in range(78, 96)]), axis=0)
    plate[397:454, 620:661] = np.median(

    plate[487:542, 613:666] = originals[12][487:542, 613:666]

    # highlight and finger-hole texture. Cutout is a physical circle, not a

    centre = source0[352:397, 635:645]
