+"""Mass-only counterfactual, constructed from source RGB and local mechanics.
+
+The contact parameters are measured from the source trajectories. Gravity and
+the ball radii are unchanged. The new/old mass ratio inferred at first contact
+is .87; the edit multiplies the first mass by .25. No external assets are used.

+def dynamics():
+    # Coordinates and velocities are in image pixels / source frame. The
+    # floor-plane projection is nearly horizontal over this local interaction.
+    p = np.array([[78.9, 76.8], [66.35, 94.0]], dtype=float)
+    v = np.array([[-5.8, 15.7], [0., 0.]])
+    mass = np.array([.87 * .25, 1.])
+    radius = np.array([5.5, 5.1])
+    floor = np.array([93.8, 94.4])
+    dt = 1 / 160
+    contact = False
+    positions = np.zeros((96, 2, 2))
+    travel = np.zeros((96, 2))
+    positions[20] = p
+    distance = np.zeros(2)
+    events = []
+    for step in range(1, 75 * 160 + 1):
+        prev = p.copy()
+        v[0, 1] += 2.3 * dt
+        if contact:
+            # Source-fitted rolling resistance: the worn floor ball has
+            # additional velocity-dependent dissipation, not a changed mass.
+            v[0, 0] -= np.clip(v[0, 0], -.30 * dt, .30 * dt)
+            v[1, 0] *= np.exp(-.137 * dt)
+            v[1, 0] -= np.clip(v[1, 0], -.149 * dt, .149 * dt)
+        p += v * dt
+        for j in range(2):
+            if p[j, 1] > floor[j]:
+                p[j, 1] = floor[j]
+                if j == 0:
+                    v[j, 1] = -v[j, 1] * .46 if v[j, 1] > 1 else 0.
+                else:
+                    v[j, 1] = 0.
+        delta = p[0] - p[1]
+        separation = np.linalg.norm(delta)
+        normal = delta / separation
+        if separation < radius.sum() and not contact:
+            approach = np.dot(v[0] - v[1], normal)
+            if approach < 0:
+                impulse = -(1 + .628) * approach / (1 / mass[0] + 1 / mass[1])
+                v[0] += impulse * normal / mass[0]
+                v[1] -= impulse * normal / mass[1]
+                p[0] += normal * (radius.sum() - separation)
+                contact = True
+                events.append({'frame': 20 + step * dt, 'event': 'ball contact',
+                               'velocities': v.tolist()})
+        # The lighter ball reaches the lower left edge of the heavy table leg.
+        # This is a fixed, high-mass piece of the preserved source geometry.
+        if p[0, 0] > 158.3 and p[0, 1] > 85:
+            p[0, 0] = 158.3
+            if v[0, 0] > 0:
+                v[0, 0] *= -.4
+                events.append({'frame': 20 + step * dt, 'event': 'table leg contact'})
+        distance += np.abs(p[:, 0] - prev[:, 0])
+        if step % 160 == 0:
+            i = 20 + step // 160
+            positions[i] = p
+            travel[i] = distance
+    return positions, travel, events

+def sample_rgb(image, xx, yy):
+    return np.stack([map_coordinates(image[..., c], [yy, xx],
+                                     order=1, mode='nearest') for c in range(3)], -1)

+def ball_sprite(source, bg, index, center, radius):
+    # Residual RGB sprites retain the factual material, shading and tennis
+    # seam. Compositing residuals, rather than cutting rectangular patches,
+    # keeps the original soft subpixel silhouette against the destination.
+    u = np.linspace(-7.5, 7.5, 61)
+    yy, xx = np.meshgrid(u, u, indexing='ij')
+    cx, cy = center
+    color = sample_rgb(source[index], xx + cx, yy + cy)
+    ground = sample_rgb(bg, xx + cx, yy + cy)
+    mask = np.clip((radius + 1.05 - np.hypot(xx, yy)) / .75, 0, 1)
+    return (color - ground) * mask[..., None]
+
+
+def shadow_sprite(source, bg, index, center):
+    # Isolate the actual floor shadow; it remains on the floor during bounces.
+    u = np.arange(-10., 23.25, .25)
+    vv = np.arange(-4., 13.25, .25)
+    yy, xx = np.meshgrid(vv, u, indexing='ij')
+    cx, cy = center
+    diff = sample_rgb(source[index], xx + cx, yy + cy) - sample_rgb(bg, xx + cx, yy + cy)
+    grey = np.minimum(diff.mean(-1), 0)
+    exclusion = np.clip((np.hypot(xx, yy) - 5.5) / .8, 0, 1)
+    mask = exclusion * np.clip((yy - 2.5) / 1.5, 0, 1)
+    mask *= np.clip((12 - yy) / 3, 0, 1) * np.clip((22 - xx) / 4, 0, 1)
+    return gaussian_filter(grey * mask, .65)

+def paste_ball(canvas, sprite, center):
+    cx, cy = center
+    x0, x1 = max(0, int(cx)-9), min(256, int(cx)+10)
+    y0, y1 = max(0, int(cy)-9), min(144, int(cy)+10)
+    yy, xx = np.mgrid[y0:y1, x0:x1]
+    values = sample_rgb(sprite, (xx-cx+7.5)*4, (yy-cy+7.5)*4)
+    canvas[y0:y1, x0:x1] += values

+def paste_shadow(canvas, sprite, center, floor_y):
+    cx, cy = center
+    lift = max(0., floor_y-cy)
+    # Source light casts a short diagonal shadow to the lower right.
+    sx = cx + lift * .75
+    sy = floor_y + lift * .14
+    x0, x1 = max(0, int(sx)-11), min(256, int(sx)+25)
+    y0, y1 = max(0, int(sy)-5), min(144, int(sy)+14)
+    yy, xx = np.mgrid[y0:y1, x0:x1]
+    coords = [(yy-sy+4)*4, (xx-sx+10)*4]
+    sh = map_coordinates(sprite, coords, order=1, mode='constant', cval=0.)
+    canvas[y0:y1, x0:x1] += sh[..., None] * np.exp(-lift * .075)
+
+
+def interpolate_sprites(sprites, keys, phase):
+    phase = np.clip(phase, 0., len(keys)-1.)
+    a = int(phase)
+    b = min(a+1, len(keys)-1)
+    f = phase-a
+    return sprites[keys[a]]*(1-f) + sprites[keys[b]]*f
+

+def main():
+    source = np.load(ROOT/'source_pixels.npz')['rgb'].astype(np.float64)
+    bg = background(source)
+    positions, travel, events = dynamics()
+    new_sprites = {i: ball_sprite(source, bg, i, c, 5.5) for i,c in NEW_CENTERS.items()}
+    old_sprites = {i: ball_sprite(source, bg, i, c, 5.1) for i,c in OLD_CENTERS.items()}
+    new_keys = list(range(22,38))
+    old_keys = [0] + list(range(22,38))
+    new_shadow = shadow_sprite(source, bg, 38, NEW_CENTERS[38])
+    old_shadow = shadow_sprite(source, bg, 0, OLD_CENTERS[0])
+    frames = source.copy()
+    for i in range(21,96):
+        frame = source[i].copy()
+        frame[82:107, 10:136] = bg[82:107, 10:136]
+        pn, po = positions[i]
+        # Matching seam phases to travelled distance preserves rotation at
+        # the new speeds without rotating the illumination across the sphere.
+        phase_new = (travel[i,0] / 2.15) % (len(new_keys)-1)
+        phase_old = travel[i,1] / 5.0
+        sn = interpolate_sprites(new_sprites, new_keys, phase_new)
+        so = interpolate_sprites(old_sprites, old_keys, phase_old)
+        paste_shadow(frame, old_shadow, po, 94.4)
+        paste_shadow(frame, new_shadow, pn, 93.8)
+        paste_ball(frame, so, po)
+        paste_ball(frame, sn, pn)
+        frames[i] = np.clip(frame,0,255)
+    frames = np.rint(frames).astype(np.uint8)
+    assert frames.shape == (96,144,256,3)
+    assert np.array_equal(frames[:21], source[:21].astype(np.uint8))
+    artifact_io.save(frames, ROOT)
+    (ROOT/'motion_report.json').write_text(json.dumps({
+        'mass_multiplier': .25, 'events': events,
+        'final_centers': positions[95].tolist(),
+        'source_unchanged_through_frame_1based': 21,
+        'shape': list(frames.shape)
+    }, indent=2))
+    print('Saved complete counterfactual:', frames.shape)
+    print(json.dumps(events, indent=2))
+    print('Final centers:', positions[95])
+
+
+if __name__ == '__main__':
+    main()
