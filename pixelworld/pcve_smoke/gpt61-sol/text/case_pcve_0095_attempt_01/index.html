+def smoothstep(x):
+    x = np.clip(x, 0, 1)
+    return x*x*(3-2*x)

+def make_plate(refs):

+    # Reconstruct at the original resolution to keep the thin wood grain.
+    scale = refs[0].shape[1] / 256
+    x0, y0, x1, y1 = 0, 74, 63, 127

+           27:(23.1,88.4),30:(21.5,88.4),95:(21.4,88.4)}
+    # Only the initial marble, contact shadow and colored reflection need recovery.

+        for color, centers in [('blue',blue),('red',red)]:

+    count=masks.sum(0)

+    # A small lens between the spheres never appears in the source. Reconstruct
+    # each wood-grain scanline from its immediately exposed left/right neighbors.
+    # Neighboring observed texture is blended into this interpolation, not painted
+    # as a featureless circle.
+    for row in range(plate.shape[0]):
+        missing=np.flatnonzero(holes[row])
+        if not len(missing): continue

+            plate[row,lo:hi+1]=lv*(1-u)+rv*u

+    # horizontal strip, blended only over the unobserved lens.

from pathlib import Path
import numpy as np
import cv2
from scipy.ndimage import gaussian_filter, map_coordinates
from artifact_io import save

ROOT = Path(__file__).resolve().parent


def read_references():
    wanted = [0, 21, 22, 23, 24, 25, 26, 27, 30, 95]
    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
    refs = {}
    for t in range(96):
        ok, bgr = cap.read()
        if not ok:
            raise RuntimeError('Cannot decode the complete source')
        if t in wanted:
            refs[t] = cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB)
    cap.release()
    return refs


def smoothstep(x):
    x = np.clip(x, 0, 1)
    return x*x*(3-2*x)


def make_plate(refs):
    # The other times reveal most of the wood beneath the initial marble.
    # Reconstruct at the original resolution to keep the thin wood grain.
    scale = refs[0].shape[1] / 256
    x0, y0, x1, y1 = 0, 74, 63, 127
    ix0, iy0, ix1, iy1 = [int(v*scale) for v in (x0,y0,x1,y1)]
    yy, xx = np.mgrid[iy0:iy1,ix0:ix1].astype(float)
    xx = (xx+.5)/scale-.5; yy = (yy+.5)/scale-.5
    initial = refs[0][iy0:iy1,ix0:ix1].astype(float)
    # Centers in 256x144 coordinates, including objects partially outside view.
    blue = {0:(24.45,88.0),21:(24.45,88.0),22:(21.35,88.15),
            23:(14.2,88.55),24:(7.4,88.9),25:(1.2,89.2),
            26:(-4.4,89.5),27:(-9.5,89.8)}
    red = {21:(47.5,78.0),22:(39.0,88.25),23:(34.6,88.3),
           24:(30.7,88.3),25:(27.6,88.3),26:(25.0,88.4),
           27:(23.1,88.4),30:(21.5,88.4),95:(21.4,88.4)}
    # Only the initial marble, contact shadow and colored reflection need recovery.
    body = ((xx-24.5)/10.5)**2 + ((yy-88.6)/10.7)**2
    refl = ((xx-24.5)/13.8)**2 + ((yy-104.0)/17.0)**2
    need = (body < 1.0) | (refl < 1.0)
    stack, masks = [], []
    # Order prioritizes the frame with the clearest strip between the two spheres.
    order = [23,24,22,27,26,25,30,95,0,21]
    for t in order:
        im = refs[t][iy0:iy1,ix0:ix1].astype(float)
        good = np.ones(xx.shape,bool)
        for color, centers in [('blue',blue),('red',red)]:
            if t not in centers:
                continue
            cx,cy = centers[t]
            good &= (xx-cx)**2 + (yy-cy)**2 > 8.8**2
            # Exclude the localized contact shadow and reflection, not the table.
            good &= ((xx-cx)/9.0)**2 + ((yy-(cy+9.0))/3.6)**2 > 1
            good &= ((xx-(cx+0.8))/9.1)**2 + ((yy-(cy+15.0))/11.8)**2 > 1
        r,g,b = im.transpose(2,0,1)
        # Reject antialiased blue/red edges and colored reflection contamination.
        good &= (r-g > 14) & (r-g < 44) & (g-b > 12) & (g-b < 44)
        stack.append(im); masks.append(good)
    stack=np.stack(stack); masks=np.stack(masks)
    # Use median of revealed pixels; observations under either sphere are excluded.
    count=masks.sum(0)
    masked=np.where(masks[...,None],stack,np.nan)
    import warnings
    with warnings.catch_warnings():
        warnings.simplefilter('ignore',RuntimeWarning)
        recovered=np.nanmedian(masked,axis=0)
    holes=need & (count==0)
    plate=initial.copy()
    observed=need & ~holes
    plate[observed]=recovered[observed]
    # A small lens between the spheres never appears in the source. Reconstruct
    # each wood-grain scanline from its immediately exposed left/right neighbors.
    # Neighboring observed texture is blended into this interpolation, not painted
    # as a featureless circle.
    for row in range(plate.shape[0]):
        missing=np.flatnonzero(holes[row])
        if not len(missing): continue
        boundaries=np.r_[0,np.flatnonzero(np.diff(missing)>1)+1,len(missing)]
        for i,j in zip(boundaries[:-1],boundaries[1:]):
            lo,hi=int(missing[i]),int(missing[j-1])
            left=max(0,lo-1); right=min(plate.shape[1]-1,hi+1)
            lv=plate[row,max(0,left-2):left+1].mean(0)
            rv=plate[row,right:min(right+3,plate.shape[1])].mean(0)
            u=np.linspace(0,1,hi-lo+3)[1:-1,None]
            plate[row,lo:hi+1]=lv*(1-u)+rv*u
    # Recover tiny high-frequency grain variations from another part of the same
    # horizontal strip, blended only over the unobserved lens.
    detail=initial-gaussian_filter(initial,(1.1*scale,1.1*scale,0))
    donor=np.roll(detail, -int(19*scale),axis=1)
    plate += donor*gaussian_filter(holes.astype(float),.8*scale)[...,None]*.45
    # Smooth seam between the source texture and the restored footprint.
    alpha = np.maximum(smoothstep((1-body)*4),smoothstep((1-refl)*5))
    full=refs[0].astype(float)
    full[iy0:iy1,ix0:ix1]=initial*(1-alpha[...,None])+plate*alpha[...,None]
    small=cv2.resize(np.clip(full,0,255).astype(np.uint8),(256,144),interpolation=cv2.INTER_AREA)
    return small.astype(float),full

    assert source.shape==(96,144,256,3)
    refs=read_references()

    print('Counterfactual red center x:', [round(35.65-11.25*k+.3*k*k,2) for k in range(6)])

    main()
