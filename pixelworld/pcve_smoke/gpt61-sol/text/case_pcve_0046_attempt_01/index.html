+"""Counterfactual ramp motion, reconstructed only from the source RGB video.
+
+The intervention changes Coulomb friction, not the car's paint, geometry,
+lighting, camera, or ramp.  The incoming trajectory is measured from RGB
+registration.  A source-textured car is subsequently moved under the changed
+friction and held by static friction once its uphill velocity reaches zero.
+"""
+from pathlib import Path
+import json
+
+import cv2
+import numpy as np
+from PIL import Image
+from scipy.ndimage import (
+    binary_closing, binary_dilation, binary_fill_holes, gaussian_filter, label,
+)
+from scipy.optimize import minimize
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+
+
+def main():
+    src = np.load(ROOT / "source_pixels.npz")["rgb"]
+    spec = json.loads((ROOT / "source_info.json").read_text())
+    intervention = spec["execution_frame_1based"] - 1
+    capture = cv2.VideoCapture(str(ROOT / "source_video.mp4"))
+
+    def read_high(index):
+        capture.set(cv2.CAP_PROP_POS_FRAMES, index)
+        ok, image = capture.read()
+        if not ok:
+            raise RuntimeError(f"Cannot decode source frame {index}")
+        return cv2.cvtColor(image, cv2.COLOR_BGR2RGB)
+
+    anchor = read_high(intervention)
+    scale = anchor.shape[1] / src.shape[2]
+    # Static background samples after the factual car has left this part of
+    # the ramp.  Restrict reconstruction to the car region, so other pixels
+    # and the distant projected ground shadow are not changed.
+    x0, x1, y0, y1 = 350, 565, 350, 465
+    samples = [read_high(t)[y0:y1, x0:x1] for t in (30, 40, 50, 60, 70, 80, 90)]
+    clean_patch = np.median(samples, axis=0).astype(np.uint8)
+    background = anchor.copy()
+    background[y0:y1, x0:x1] = clean_patch
+
+    # Measured registration of neighboring car images, excluding background.
+    low_bg = np.array(Image.fromarray(background).resize((256, 144), Image.Resampling.LANCZOS))
+    velocities = []
+    for t in range(1, intervention + 1):
+        now = src[t].astype(np.float32)
+        previous = src[t - 1].astype(np.float32)
+        changed = np.max(np.abs(now - low_bg.astype(float)), axis=2) > 22
+        changed[:60] = False
+        changed[100:] = False
+        components, count = label(changed)
+        largest = max(range(1, count + 1), key=lambda k: np.sum(components == k))
+        silhouette = binary_fill_holes(components == largest)
+        yy, xx = np.where(silhouette)
+        colors = now[yy, xx]
+
+        def registration_loss(offset):
+            map_x = (xx + offset[0]).astype(np.float32)[None, :]
+            map_y = (yy + offset[1]).astype(np.float32)[None, :]
+            sampled = cv2.remap(previous, map_x, map_y, cv2.INTER_LINEAR)[0]
+            return float(np.minimum((sampled - colors) ** 2, 6000).mean())
+
+        fit = minimize(registration_loss, [15 - 0.9 * t, (15 - 0.9 * t) * 0.36],
+                       method="Nelder-Mead", options={"maxiter": 180, "xatol": 0.002})
+        velocities.append(fit.x)
+    velocities = np.array(velocities)
+    # Fit frame-interval speeds to obtain the instantaneous velocity at the
+    # intervention; screen displacement is nearly exactly linear in ramp
+    # distance over this short, orthographically viewed part of the scene.
+    intervals = np.arange(1, intervention + 1) - 0.5
+    fit_x = np.polyfit(intervals, velocities[:, 0], 1)
+    original_deceleration = float(-fit_x[0])
+    incoming_speed = float(np.polyval(fit_x, intervention))
+    direction_y = float(np.sum(velocities[:, 0] * velocities[:, 1]) /
+                        np.sum(velocities[:, 0] ** 2))
+
+    # The observed deceleration contains both downhill gravity and friction.
+    # The gravity component is unchanged by the intervention.  The ramp's
+    # low slope and the airborne gravity scale give about 0.64 px/frame^2
+    # downhill; the remaining measured braking is Coulomb friction.
+    downhill_gravity = 0.64
+    original_friction_deceleration = max(0.05, original_deceleration - downhill_gravity)
+    counterfactual_deceleration = downhill_gravity + 8 * original_friction_deceleration
+    stop_time = incoming_speed / counterfactual_deceleration
+
+    # Extract the car, including antialiased tire contacts and its local
+    # shadow, at native source resolution.  Hole filling retains similarly
+    # colored glazing and body pixels inside the silhouette.
+    anchor_patch = anchor[y0:y1, x0:x1]
+    difference = np.max(np.abs(anchor_patch.astype(float) - clean_patch.astype(float)), axis=2)
+    components, count = label(difference > 10)
+    largest = max(range(1, count + 1), key=lambda k: np.sum(components == k))
+    silhouette = binary_fill_holes(binary_closing(components == largest, iterations=2))
+    silhouette = binary_dilation(silhouette, iterations=3)
+    alpha_patch = gaussian_filter(silhouette.astype(np.float32), 0.8)
+    alpha = np.zeros(anchor.shape[:2], np.float32)
+    alpha[y0:y1, x0:x1] = alpha_patch
+    premultiplied = anchor.astype(np.float32) * alpha[..., None]
+    bg_float = background.astype(np.float32)
+
+    frames = src.copy()
+    trajectory = []
+    for t in range(intervention, len(frames)):
+        elapsed = min(float(t - intervention), stop_time)
+        distance = incoming_speed * elapsed - 0.5 * counterfactual_deceleration * elapsed ** 2
+        trajectory.append({"frame": t + 1, "ramp_distance_pixels": distance,
+                           "speed_pixels_per_frame": max(0.0, incoming_speed - counterfactual_deceleration * elapsed)})
+        if t == intervention:
+            # The physical intervention is instantaneous: position and
+            # appearance are continuous at the execution timestamp.
+            frames[t] = src[t]
+            continue
+        transform = np.array([[1, 0, -scale * distance],
+                              [0, 1, -scale * direction_y * distance]], np.float32)
+        moved_alpha = cv2.warpAffine(alpha, transform, (anchor.shape[1], anchor.shape[0]),
+                                     flags=cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)
+        moved_rgb = cv2.warpAffine(premultiplied, transform, (anchor.shape[1], anchor.shape[0]),
+                                   flags=cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)
+        image = np.clip(moved_rgb + bg_float * (1 - moved_alpha[..., None]), 0, 255).round().astype(np.uint8)
+        frames[t] = np.array(Image.fromarray(image).resize((256, 144), Image.Resampling.LANCZOS))
+
+    capture.release()
+    assert np.array_equal(frames[:intervention], src[:intervention])
+    assert frames.shape == (96, 144, 256, 3)
+    artifact_io.save(frames, ROOT)
+    # Also supply a conventional MP4 alongside the required RGB/player files.
+    writer = cv2.VideoWriter(str(ROOT / "edited_video.mp4"), cv2.VideoWriter_fourcc(*"mp4v"),
+                             24, (256, 144))
+    for image in frames:
+        writer.write(cv2.cvtColor(image, cv2.COLOR_RGB2BGR))
+    writer.release()
+    diagnostics = {
+        "intervention_frame": intervention + 1,
+        "measured_original_deceleration": original_deceleration,
+        "incoming_speed": incoming_speed,
+        "friction_multiplier": 8,
+        "changed_deceleration": counterfactual_deceleration,
+        "stopping_time_frames": stop_time,
+        "trajectory": trajectory,
+    }
+    (ROOT / "motion_diagnostics.json").write_text(json.dumps(diagnostics, indent=2))
+    print(json.dumps({k: v for k, v in diagnostics.items() if k != "trajectory"}, indent=2))
+    print("Saved 96 RGB frames, pixels.npz, index.html, and edited_video.mp4")
+
+
+if __name__ == "__main__":
+    main()
