+"""Counterfactual friction edit using source-pixel sprites and fitted dynamics.
+
+The impact is unchanged.  The soccer ball's dissipative ground-contact terms
+are multiplied by 0.4, with the initial post-impact linear velocity preserved.
+Pixel geometry is measured from the source; no external renderer is used.
+"""
+from pathlib import Path
+import numpy as np
+import cv2
+from PIL import Image
+from scipy import ndimage as ndi
+from scipy.optimize import least_squares, curve_fit
+from scipy.spatial import ConvexHull
+from scipy.integrate import solve_ivp
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+
+def load_source():
+    rgb = np.load(ROOT / 'source_pixels.npz')['rgb']
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    hi = []
+    for i in range(len(rgb)):
+        ok, frame = cap.read()
+        if not ok:
+            raise RuntimeError('Incomplete source video')
+        hi.append(frame[..., ::-1].copy())
+    cap.release()
+    return rgb, np.asarray(hi)
+
+
+def track_ball(rgb, hi):
+    centers = []
+    for small, frame in zip(rgb, hi):
+        a = small.astype(float)
+        white = (a[..., 0] > 150) & (a[..., 1] > 145) & (a[..., 2] > 145)
+        white[:69] = False
+        white[93:] = False
+        lab, _ = ndi.label(white)
+        count = np.bincount(lab.ravel())
+        count[0] = 0
+        y, x = np.where(lab == count.argmax())
+        cx = x.mean() * 5
+        start, end = int(cx - 65), int(cx + 65)
+        crop = frame[351:458, start:end].astype(float)
+        yy, xx = np.indices(crop.shape[:2])
+        mask = (crop[..., 0] > crop[..., 1] - 9) & (crop[..., 2] > crop[..., 1] - 9)
+        points = np.stack([xx[mask] + start, yy[mask] + 351], -1)
+        hull = points[ConvexHull(points).vertices]
+        result = least_squares(
+            lambda q: np.sqrt(((hull - q[:2]) ** 2).sum(-1)) - q[2],
+            [cx + 1, 405, 50], loss='soft_l1', f_scale=1,
+            bounds=([cx - 12, 396, 45], [cx + 12, 414, 55]))
+        centers.append(result.x / 5)
+    return np.asarray(centers)
+
+
+def ground_path(times, x0, velocity, quadratic_drag, contact_deceleration):
+    times = np.asarray(times, float)
+    if np.max(times) == 0:
+        return np.full_like(times, x0)
+    # Coulomb-like contact loss plus the speed-dependent loss measured in source.
+    solution = solve_ivp(
+        lambda t, state: [-max(state[1], 0),
+                          -quadratic_drag * max(state[1], 0) ** 2 - contact_deceleration],
+        [0, np.max(times)], [x0, velocity], t_eval=times,
+        rtol=1e-9, atol=1e-9)
+    return solution.y[0]
+
+
+def sprite(frame, clean, center, local_x, local_y):
+    """Separate the textured ball and its shadow from the source ground."""
+    cx, cy = center[:2] * 5
+    mx = (local_x + cx).astype(np.float32)
+    my = (local_y + cy).astype(np.float32)
+    src = cv2.remap(frame.astype(np.float32), mx, my, cv2.INTER_LINEAR,
+                    borderMode=cv2.BORDER_REFLECT_101)
+    bg = cv2.remap(clean.astype(np.float32), mx, my, cv2.INTER_LINEAR,
+                   borderMode=cv2.BORDER_REFLECT_101)
+    # Copy all source antialiasing at the spherical silhouette, rather than
+    # replacing the original pentagon pattern with a synthetic texture.
+    radius = np.sqrt(local_x ** 2 + local_y ** 2)
+    ball_alpha = np.clip((54.5 - radius) / 3.0, 0, 1)
+    # Ground shadows are transferred as illumination, not as a grass patch.
+    ratio = np.mean(src / np.maximum(bg, 12), axis=-1)
+    shade = np.clip((1 - ratio - .025) / .975, 0, .92)
+    shade = cv2.GaussianBlur(shade.astype(np.float32), (3, 3), .45)
+    shade *= (1 - ball_alpha)
+    return src, ball_alpha, shade
+
+
+def main():
+    rgb, hi = load_source()
+    track = track_ball(rgb, hi)
+    # The first motion is the impact during frames 13-14.  By frame 14 the
+    # projectile has separated. Its normal impulse and factual incoming path
+    # are retained. Infer the subsequent dissipative motion independently.
+    impact = 14
+    x0 = track[impact, 0]
+    sample_t = np.arange(15, 58, dtype=float) - impact
+    sample_x = track[15:58, 0]
+    fit, _ = curve_fit(
+        lambda t, v, d, a: ground_path(t, x0, v, d, a),
+        sample_t, sample_x, p0=[2.8, .02, .03],
+        bounds=([0, 0, 0], [10, 1, 1]))
+    v0, drag, deceleration = fit
+    t = np.arange(96 - impact, dtype=float)
+    new_x = ground_path(t, x0, v0, drag * .4, deceleration * .4)
+
+    # Static clean plate: each region comes from a source instant when it is
+    # unobstructed. The initial ball location is clear in the final frame.
+    clean = hi[0].copy()
+    clean[330:515, 540:765] = hi[-1, 330:515, 540:765]
+    size = 201
+    ly, lx = np.mgrid[-75:126, -75:126].astype(np.float32)
+    assert lx.shape == (size, size)
+    cache = {}
+    def get_sprite(j):
+        if j not in cache:
+            cache[j] = sprite(hi[j], clean, track[j], lx, ly)
+        return cache[j]
+
+    output = rgb.copy()
+    log = []
+    for frame_idx in range(impact + 1, 96):
+        n = frame_idx - impact
+        # Reduced contact torque delays spin-up while the ball slides farther.
+        # Source orientations are re-used along the friction-scaled clock.
+        phase = impact + .4 * n
+        lo = int(np.floor(phase))
+        frac = phase - lo
+        s0, a0, d0 = get_sprite(lo)
+        s1, a1, d1 = get_sprite(lo + 1)
+        texture = s0 * (1 - frac) + s1 * frac
+        alpha = a0  # source geometry remains a rigid sphere
+        shadow = d0 * (1 - frac) + d1 * frac
+
+        cx = new_x[n]
+        cy = track[impact, 1] + .011 * (x0 - cx)
+        image = hi[frame_idx].astype(np.float32)
+        # Remove the factual soccer ball and the entire factual cast shadow.
+        ox, oy = track[frame_idx, :2] * 5
+        left, top = int(ox - 75), int(oy - 75)
+        right, bottom = left + size, top + size
+        qy, qx = np.mgrid[top:bottom, left:right]
+        edge = np.minimum.reduce([qx - left, right - 1 - qx,
+                                  qy - top, bottom - 1 - qy])
+        erase = np.clip(edge / 10, 0, 1).astype(np.float32)[..., None]
+        image[top:bottom, left:right] = (
+            image[top:bottom, left:right] * (1 - erase) +
+            clean[top:bottom, left:right].astype(np.float32) * erase)
+
+        # Translate the counterfactual sphere and its source-derived shadow.
+        dst_left, dst_top = int(np.floor(cx * 5 - 75)), int(np.floor(cy * 5 - 75))
+        x1, x2 = max(0, dst_left), min(image.shape[1], dst_left + size)
+        y1, y2 = max(0, dst_top), min(image.shape[0], dst_top + size)
+        gy, gx = np.mgrid[y1:y2, x1:x2]
+        map_x = (gx - cx * 5 + 75).astype(np.float32)
+        map_y = (gy - cy * 5 + 75).astype(np.float32)
+        tex = cv2.remap(texture, map_x, map_y, cv2.INTER_LINEAR)
+        matte = cv2.remap(alpha, map_x, map_y, cv2.INTER_LINEAR)[..., None]
+        darkness = cv2.remap(shadow, map_x, map_y, cv2.INTER_LINEAR)[..., None]
+        region = image[y1:y2, x1:x2]
+        region *= 1 - darkness
+        region[:] = region * (1 - matte) + tex * matte
+        final = Image.fromarray(np.clip(image + .5, 0, 255).astype(np.uint8))
+        output[frame_idx] = np.asarray(final.resize((256, 144), Image.Resampling.LANCZOS))
+        log.append((frame_idx, cx, cy, phase))
+
+    # First mandatory submission is saved before any subsequent inspection.
+    artifact_io.save(output, ROOT)
+    np.savetxt(ROOT / 'motion.csv', np.array(log), delimiter=',',
+               header='frame_0based,ball_x,ball_y,source_rotation_phase', comments='')
+    print('Fitted source losses:', fit)
+    print('Counterfactual ball centers:', [(i, round(float(new_x[i-impact]), 3))
+                                         for i in [14, 24, 48, 72, 95]])
+    print('Saved', output.shape, output.dtype, 'to', ROOT)
+
+
+if __name__ == '__main__':
+    main()

import numpy as np

def load_source():

    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))

def track_ball(rgb, hi):
    centers = []
    for small, frame in zip(rgb, hi):
        a = small.astype(float)
        white = (a[..., 0] > 150) & (a[..., 1] > 145) & (a[..., 2] > 145)
        white[:69] = False
        white[93:] = False
        lab, _ = ndi.label(white)
        count = np.bincount(lab.ravel())
        count[0] = 0
        y, x = np.where(lab == count.argmax())
        cx = x.mean() * 5
        start, end = int(cx - 65), int(cx + 65)
        crop = frame[351:458, start:end].astype(float)
        yy, xx = np.indices(crop.shape[:2])
        mask = (crop[..., 0] > crop[..., 1] - 9) & (crop[..., 2] > crop[..., 1] - 9)
        points = np.stack([xx[mask] + start, yy[mask] + 351], -1)
        hull = points[ConvexHull(points).vertices]
        result = least_squares(
            lambda q: np.sqrt(((hull - q[:2]) ** 2).sum(-1)) - q[2],
            [cx + 1, 405, 50], loss='soft_l1', f_scale=1,
            bounds=([cx - 12, 396, 45], [cx + 12, 414, 55]))
        centers.append(result.x / 5)
    return np.asarray(centers)


def ground_path(times, x0, velocity, quadratic_drag, contact_deceleration):
    times = np.asarray(times, float)
    if np.max(times) == 0:
        return np.full_like(times, x0)
    # Coulomb-like contact loss plus the speed-dependent loss measured in source.
    solution = solve_ivp(
        lambda t, state: [-max(state[1], 0),
                          -quadratic_drag * max(state[1], 0) ** 2 - contact_deceleration],
        [0, np.max(times)], [x0, velocity], t_eval=times,
        rtol=1e-9, atol=1e-9)
    return solution.y[0]


def sprite(frame, clean, center, local_x, local_y):
    \"\"\"Separate the textured ball and its shadow from the source ground.\"\"\"
    cx, cy = center[:2] * 5
    mx = (local_x + cx).astype(np.float32)
    my = (local_y + cy).astype(np.float32)
    src = cv2.remap(frame.astype(np.float32), mx, my, cv2.INTER_LINEAR,
                    borderMode=cv2.BORDER_REFLECT_101)
    bg = cv2.remap(clean.astype(np.float32), mx, my, cv2.INTER_LINEAR,
                   borderMode=cv2.BORDER_REFLECT_101)
    # Copy all source antialiasing at the spherical silhouette, rather than
    # replacing the original pentagon pattern with a synthetic texture.
    radius = np.sqrt(local_x ** 2 + local_y ** 2)
    ball_alpha = np.clip((54.5 - radius) / 3.0, 0, 1)
    # Ground shadows are transferred as illumination, not as a grass patch.
    ratio = np.mean(src / np.maximum(bg, 12), axis=-1)
    shade = np.clip((1 - ratio - .025) / .975, 0, .92)
    shade = cv2.GaussianBlur(shade.astype(np.float32), (3, 3), .45)
    shade *= (1 - ball_alpha)
    return src, ball_alpha, shade


def main():

    track = track_ball(rgb, hi)
    # The first motion is the impact during frames 13-14.  By frame 14 the
    # projectile has separated. Its normal impulse and factual incoming path
    # are retained. Infer the subsequent dissipative motion independently.
    impact = 14
    x0 = track[impact, 0]
    sample_t = np.arange(15, 58, dtype=float) - impact
    sample_x = track[15:58, 0]
    fit, _ = curve_fit(
        lambda t, v, d, a: ground_path(t, x0, v, d, a),
        sample_t, sample_x, p0=[2.8, .02, .03],
        bounds=([0, 0, 0], [10, 1, 1]))
    v0, drag, deceleration = fit
    t = np.arange(96 - impact, dtype=float)
    new_x = ground_path(t, x0, v0, drag * .4, deceleration * .4)

    # Static clean plate: each region comes from a source instant when it is
    # unobstructed. The initial ball location is clear in the final frame.
    clean = hi[0].copy()
    clean[330:515, 540:765] = hi[-1, 330:515, 540:765]
    size = 201
    ly, lx = np.mgrid[-75:126, -75:126].astype(np.float32)
    assert lx.shape == (size, size)
    cache = {}
    def get_sprite(j):
        if j not in cache:
            cache[j] = sprite(hi[j], clean, track[j], lx, ly)
        return cache[j]

    output = rgb.copy()
    log = []
    for frame_idx in range(impact + 1, 96):
        n = frame_idx - impact
        # Reduced contact torque delays spin-up while the ball slides farther.
        # Source orientations are re-used along the friction-scaled clock.
        phase = impact + .4 * n
        lo = int(np.floor(phase))
        frac = phase - lo
        s0, a0, d0 = get_sprite(lo)
        s1, a1, d1 = get_sprite(lo + 1)
        texture = s0 * (1 - frac) + s1 * frac
        alpha = a0  # source geometry remains a rigid sphere
        shadow = d0 * (1 - frac) + d1 * frac

        cx = new_x[n]
        cy = track[impact, 1] + .011 * (x0 - cx)
        image = hi[frame_idx].astype(np.float32)
        # Remove the factual soccer ball and the entire factual cast shadow.

        left, top = int(ox - 75), int(oy - 75)

        qy, qx = np.mgrid[top:bottom, left:right]
        edge = np.minimum.reduce([qx - left, right - 1 - qx,

        image[top:bottom, left:right] = (

            clean[top:bottom, left:right].astype(np.float32) * erase)

        matte = cv2.remap(alpha, map_x, map_y, cv2.INTER_LINEAR)[..., None]
        darkness = cv2.remap(shadow, map_x, map_y, cv2.INTER_LINEAR)[..., None]
        region = image[y1:y2, x1:x2]
        region *= 1 - darkness
        region[:] = region * (1 - matte) + tex * matte
        final = Image.fromarray(np.clip(image + .5, 0, 255).astype(np.uint8))
        output[frame_idx] = np.asarray(final.resize((256, 144), Image.Resampling.LANCZOS))
        log.append((frame_idx, cx, cy, phase))

    # First mandatory submission is saved before any subsequent inspection.
    artifact_io.save(output, ROOT)

               header='frame_0based,ball_x,ball_y,source_rotation_phase', comments='')

                                         for i in [14, 24, 48, 72, 95]])
    print('Saved', output.shape, output.dtype, 'to', ROOT)


if __name__ == '__main__':
    main()
