+"""Counterfactual billiards edit, using only source pixels and numerical geometry.
+
+The scene is fixed-camera. A clean felt plate and source-derived ball sprites
+preserve the photographed rendering. The 1-D projective table coordinate is
+calibrated against the original shot, with equal-mass impulses and rolling drag.
+"""
+from pathlib import Path
+import numpy as np
+import cv2
+from scipy.ndimage import label, gaussian_filter
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+SOURCE = np.load(ROOT / 'source_pixels.npz')['rgb']
+N, H, W, _ = SOURCE.shape
+Y, X = np.mgrid[:H, :W]
+
+
+def clean_plate():
+    # Temporal background recovers every swept region except the cue's final
+    # resting place. There, choose only unobstructed green samples and smoothly
+    # interpolate the small, permanently occluded area.
+    a = SOURCE.astype(np.float32)
+    bg = np.median(a, axis=0)
+    region = (X >= 119) & (X <= 134) & (Y >= 64) & (Y <= 81)
+    green = ((a[..., 1]-a[..., 0]) > 48) & ((a[..., 1]-a[..., 2]) > 26) & (a[..., 1] > 138)
+    for y,x in zip(*np.where(region)):
+        vals = a[green[:,y,x],y,x]
+        if len(vals) >= 4:
+            bg[y,x] = np.median(vals,axis=0)
+        else:
+            # The cloth varies predominantly along the depth axis.
+            l = np.median(bg[y,113:119],axis=0)
+            r = np.median(bg[y,135:141],axis=0)
+            q = (x-116)/22
+            bg[y,x] = l*(1-q)+r*q
+    return bg
+

+def source_sprites():
+    cap = cv2.VideoCapture(str(ROOT/'source_video.mp4'))
+    _, bgr = cap.read()
+    first = cv2.cvtColor(bgr,cv2.COLOR_BGR2RGB).astype(np.float32)
+    cap.set(cv2.CAP_PROP_POS_FRAMES,30)
+    _, bgr = cap.read()
+    empty_cue = cv2.cvtColor(bgr,cv2.COLOR_BGR2RGB).astype(np.float32)
+    cap.release()
+    # Pixel-center mapping from original 1280x720 to the working 256x144.
+    # Source cue silhouette and its contact shadow, in full-resolution pixels.
+    x0,y0 = 395,435
+    cue = first[y0:515,x0:465].copy()
+    cbg = empty_cue[y0:515,x0:465]
+    yy,xx=np.mgrid[:80,:70]
+    cx,cy=429.5-x0,470-y0
+    ball=((xx-cx)/21)**2+((yy-cy)/18.6)**2
+    shadow=((xx-(cx-1))/25)**2+((yy-(cy+22))/15)**2
+    env=np.maximum(np.clip((1.15-ball)*5,0,1),np.clip((1.12-shadow)*4,0,1))
+    cue_delta=(cue-cbg)*env[...,None]
+    # Source target includes its specular highlight, tiny number patch and
+    # green reflected light. Fit a local bare-felt color from both sides.
+    x0b,y0b=606,320
+    obj=first[y0b:391,x0b:677].copy()
+    left=np.median(first[y0b:391,590:606],axis=1)
+    right=np.median(first[y0b:391,677:690],axis=1)
+    q=(np.arange(71)+606-598)/86
+    cloth=left[:,None,:]*(1-q[None,:,None])+right[:,None,:]*q[None,:,None]
+    yyb,xxb=np.mgrid[:71,:71]
+    bcx,bcy=640-x0b,350-y0b
+    bm=((xxb-bcx)/17)**2+((yyb-bcy)/16.8)**2
+    sm=((xxb-(bcx-2))/23)**2+((yyb-(bcy+20))/13)**2
+    benv=np.maximum(np.clip((1.12-bm)*6,0,1),np.clip((1.05-sm)*4,0,1))
+    black_delta=(obj-cloth)*benv[...,None]

+# Projective depth coordinate u along the source shot. Image scaling is
+# 1/(1+k*u), while the longitudinal position is u/(1+k*u).
+K=.00500016488
+X0=85.3152291
+INITIAL_SPEED=3.70566215
+DRAG=.0181224362
+BALL_RESTITUTION=.847007183
+CUSHION_RESTITUTION=.789768281
+BANK=145.506
+DIAMETER=8.0

+def to_u(x):
+    d=x-X0
+    return d/(1-K*d)

+def screen(u):
+    x=X0+u/(1+K*u)
+    # Ball centers follow the same projected aiming line, including perspective
+    # change in height above the felt.
+    y=93.65-.574*(x-85.40)
+    return np.array([x,y])

+def trajectories():
+    cue_u=to_u(85.40)
+    target_u=to_u(127.50)
+    # World-space midpoint, not the midpoint of perspective-distorted pixels.
+    positions=np.array([cue_u,(cue_u+target_u)/2,target_u],float)
+    velocities=np.array([INITIAL_SPEED,0.,0.])
+    path=np.empty((N,3))
+    dt=1/240
+    events=[]
+    for t in range(N):
+        path[t]=positions
+        for sub in range(240):
+            speed=np.abs(velocities)
+            move=np.maximum(speed-DRAG*dt/2,0)*np.sign(velocities)
+            positions+=move*dt
+            velocities=np.sign(velocities)*np.maximum(speed-DRAG*dt,0)
+            for i in range(2):
+                gap=positions[i+1]-positions[i]
+                closing=velocities[i]-velocities[i+1]
+                if gap < DIAMETER and closing > 0:
+                    correction=(DIAMETER-gap)/2
+                    positions[i]-=correction; positions[i+1]+=correction
+                    impulse=(1+BALL_RESTITUTION)*closing/2
+                    velocities[i]-=impulse;velocities[i+1]+=impulse
+                    events.append((round(t+(sub+1)*dt,3),f'{i} -> {i+1}'))
+            for i in range(3):
+                if positions[i] > BANK and velocities[i]>0:
+                    positions[i]=2*BANK-positions[i]
+                    velocities[i]*=-CUSHION_RESTITUTION
+                    events.append((round(t+(sub+1)*dt,3),f'{i} cushion'))
+    print('Physical events (zero-based frame time):',events)
+    print('Initial centers:',[screen(u).round(2).tolist() for u in path[0]])
+    print('Final centers:',[screen(u).round(2).tolist() for u in path[-1]])
+    return path,events
+

+def paste_delta(canvas,sprite,origin,center,scale):
+    """Subpixel source-sprite transformation with local supersampling.
+    Only the small ball footprint is touched, leaving unrelated pixels intact.
+    """
+    h,w=sprite.shape[:2]
+    ox,oy=origin
+    cx,cy=center
+    # Original sprite is five times the output dimensions.
+    s=scale/5
+    xlo=max(0,int(np.floor(cx-ox*s-2)))
+    ylo=max(0,int(np.floor(cy-oy*s-2)))
+    xhi=min(W,int(np.ceil(cx+(w-ox)*s+2)))
+    yhi=min(H,int(np.ceil(cy+(h-oy)*s+2)))
+    if xlo>=xhi or ylo>=yhi:return
+    SS=4
+    ys=ylo+(np.arange((yhi-ylo)*SS)+.5)/SS-.5
+    xs=xlo+(np.arange((xhi-xlo)*SS)+.5)/SS-.5
+    xx,yy=np.meshgrid(xs,ys)
+    mx=((xx-cx)/s+ox).astype(np.float32)
+    my=((yy-cy)/s+oy).astype(np.float32)
+    patch=cv2.remap(sprite,mx,my,cv2.INTER_LINEAR,borderMode=cv2.BORDER_CONSTANT)
+    patch=patch.reshape(yhi-ylo,SS,xhi-xlo,SS,3).mean((1,3))
+    canvas[ylo:yhi,xlo:xhi]+=patch
+
+
+def yellow_sprite():
+    # A polished solid yellow one-ball, with a white number medallion.
+    # Work at original-video pixel density; the silhouette and shadow mimic
+    # the source cue rather than using an unrelated pictorial overlay.
+    yy,xx=np.mgrid[:80,:70].astype(float)
+    cx,cy=34.5,35.
+    nx=(xx-cx)/19.2;ny=(yy-cy)/17.1
+    r2=nx*nx+ny*ny
+    nz=np.sqrt(np.maximum(1-r2,0))
+    coverage=np.clip((1-np.sqrt(r2))*9+.5,0,1)
+    light=np.clip(-.24*nx-.65*ny+.71*nz,0,1)
+    lit=.70+.30*light
+    color=np.array([231.,179.,14.])[None,None,:]*lit[...,None]
+    # Broad ambient reflection near the lower edge, source-style pale gloss.
+    spec=np.exp(-((nx+.25)**2/.027+(ny+.61)**2/.034))*65
+    color+=spec[...,None]*np.array([.38,.75,1.05])
+    color+=np.maximum(ny,0)[...,None]*np.array([-7.,1.,9.])
+    # Small white circular number spot facing the camera, not a stripe.
+    mx=(xx-(cx+.7))/7.2;my=(yy-(cy-.1))/6.2
+    med=np.clip((1-np.sqrt(mx*mx+my*my))*6+.5,0,1)*coverage
+    white=np.stack([215+18*light,213+19*light,200+26*light],axis=-1)
+    color=color*(1-med[...,None])+white*med[...,None]
+    # Rasterize the digit 1 with antialiased mathematical line segments.
+    def seg(ax,ay,bx,by,width):
+        vx=bx-ax;vy=by-ay
+        q=np.clip(((xx-ax)*vx+(yy-ay)*vy)/(vx*vx+vy*vy),0,1)
+        d=np.hypot(xx-ax-q*vx,yy-ay-q*vy)
+        return np.clip(width+.5-d,0,1)
+    ink=np.maximum.reduce([seg(cx-.3,cy-3,cx+1,cy-4, .65),
+                            seg(cx+1,cy-4,cx+1,cy+3, .7),
+                            seg(cx-.5,cy+3,cx+2.6,cy+3, .65)])*med
+    color=color*(1-ink[...,None])+np.array([38.,31.,19.])*ink[...,None]
+    # Bare-cloth baseline matching source color near the initial cue.
+    baseline=np.zeros_like(color)+np.array([110.,173.,138.])
+    shadow=.60*np.exp(-(((xx-(cx-1))/18)**2+((yy-(cy+24))/6.5)**2))
+    shadow+=.14*np.exp(-(((xx-cx)/25)**2+((yy-(cy+21))/13)**2))
+    shadow=np.clip(shadow,0,.72)*(1-coverage)
+    delta=(color-baseline)*coverage[...,None]-baseline*shadow[...,None]
+    # Feather an otherwise imperceptible broad shadow at the sprite boundary.
+    fade=np.minimum.reduce([np.clip(xx/4,0,1),np.clip((69-xx)/4,0,1),
+                             np.clip(yy/4,0,1),np.clip((79-yy)/4,0,1)])
+    delta*=fade[...,None]
+    return delta.astype(np.float32),(cx,cy)
+
+
+def generate():
+    bg=clean_plate()
+    cue,black=source_sprites()
+    yellow=yellow_sprite()
+    path,events=trajectories()
+    first_contact=next(t for t,e in events if e=='0 -> 1')
+    out=[]
+    targetscale=1/(1+K*path[0,2])
+    for t in range(N):
+        if t < first_contact:
+            # Unaffected motion before the first new impact remains exact.
+            canvas=SOURCE[t].astype(np.float32).copy()
+        else:
+            canvas=bg.copy()
+            # Farther spheres are composited first for correct depth ordering.
+            for i,spr,rel in [(2,black,1/targetscale),(0,cue,1.)]:
+                u=path[t,i]
+                paste_delta(canvas,spr[0],spr[1],screen(u),rel/(1+K*u))
+        u=path[t,1]
+        # The one-ball is inserted at time zero and thereafter moves freely;
+        # it is not re-anchored to a moving midpoint after the collision.
+        paste_delta(canvas,yellow[0],yellow[1],screen(u),1/(1+K*u))
+        out.append(np.rint(np.clip(canvas,0,255)).astype(np.uint8))
+    frames=np.stack(out)
+    artifact_io.save(frames,ROOT)
+    np.savez_compressed(ROOT/'motion_diagnostics.npz',depth_positions=path)
+    print('Saved',frames.shape,frames.dtype,'to',ROOT)
+
+
+if __name__=='__main__':
+    generate()

def clean_plate():
    # Temporal background recovers every swept region except the cue's final
    # resting place. There, choose only unobstructed green samples and smoothly
    # interpolate the small, permanently occluded area.
    a = SOURCE.astype(np.float32)
    bg = np.median(a, axis=0)
    region = (X >= 119) & (X <= 134) & (Y >= 64) & (Y <= 81)
    green = ((a[..., 1]-a[..., 0]) > 48) & ((a[..., 1]-a[..., 2]) > 26) & (a[..., 1] > 138)
    for y,x in zip(*np.where(region)):
        vals = a[green[:,y,x],y,x]
        if len(vals) >= 4:
            bg[y,x] = np.median(vals,axis=0)
        else:

            l = np.median(bg[y,113:119],axis=0)
            r = np.median(bg[y,135:141],axis=0)
            q = (x-116)/22
            bg[y,x] = l*(1-q)+r*q
    return bg

def screen(u):

    # Ball centers follow the same projected aiming line, including perspective
    # change in height above the felt.

def paste_delta(canvas,sprite,origin,center,scale):
    \"\"\"Subpixel source-sprite transformation with local supersampling.
    Only the small ball footprint is touched, leaving unrelated pixels intact.
    \"\"\"
    h,w=sprite.shape[:2]
    ox,oy=origin
    cx,cy=center
    # Original sprite is five times the output dimensions.
    s=scale/5
    xlo=max(0,int(np.floor(cx-ox*s-2)))
    ylo=max(0,int(np.floor(cy-oy*s-2)))
    xhi=min(W,int(np.ceil(cx+(w-ox)*s+2)))
    yhi=min(H,int(np.ceil(cy+(h-oy)*s+2)))
    if xlo>=xhi or ylo>=yhi:return
    SS=4
    ys=ylo+(np.arange((yhi-ylo)*SS)+.5)/SS-.5
    xs=xlo+(np.arange((xhi-xlo)*SS)+.5)/SS-.5
    xx,yy=np.meshgrid(xs,ys)
    mx=((xx-cx)/s+ox).astype(np.float32)
    my=((yy-cy)/s+oy).astype(np.float32)
    patch=cv2.remap(sprite,mx,my,cv2.INTER_LINEAR,borderMode=cv2.BORDER_CONSTANT)
    patch=patch.reshape(yhi-ylo,SS,xhi-xlo,SS,3).mean((1,3))
    canvas[ylo:yhi,xlo:xhi]+=patch

    # Work at original-video pixel density; the silhouette and shadow mimic
    # the source cue rather than using an unrelated pictorial overlay.
    yy,xx=np.mgrid[:80,:70].astype(float)
