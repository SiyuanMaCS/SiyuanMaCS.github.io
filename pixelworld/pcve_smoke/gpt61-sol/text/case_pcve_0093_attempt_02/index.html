+"""Remove the target sphere and reconstruct the unobstructed billiard shot.
+Only RGB source pixels, elementary projective geometry, and texture resampling
+are used.  The original camera, table, lighting, and room are retained.
+"""
+from pathlib import Path
+import numpy as np
+import cv2
+from scipy import ndimage
+from PIL import Image
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+
+def project(H, p):
+    q = H @ np.r_[p, 1.0]
+    return q[:2] / q[2]
+
+
+def fit_cloth(plate):
+    """Fit the very smooth felt under the two overlapping factual spheres."""
+    yy, xx = np.indices(plate.shape[:2])
+    region = (xx > 110) & (xx < 147) & (yy > 57) & (yy < 87)
+    excluded = (xx > 117) & (xx < 136) & (yy > 62) & (yy < 80)
+    valid = region & ~excluded
+    u, v = (xx - 127) / 15, (yy - 71) / 15
+    design = np.stack([u * 0 + 1, u, v, u*u, u*v, v*v], -1)
+    coefficients = np.linalg.lstsq(design[valid], plate[valid], rcond=None)[0]
+    field = design @ coefficients
+    # Replace only the coverage of the two balls and their shadows.
+    hole = (xx >= 119) & (xx <= 134) & (yy >= 64) & (yy <= 79)
+    plate[hole] = field[hole]
+    return plate, field
+
+
+def original_hd(index):
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    cap.set(cv2.CAP_PROP_POS_FRAMES, index)
+    success, bgr = cap.read()
+    cap.release()
+    if not success:
+        raise RuntimeError('Cannot decode source frame')
+    return bgr[:, :, ::-1].astype(np.float32)
+
+
+def cue_texture():
+    """Separate source sphere radiance and its soft/contact shadow at frame 9."""
+    original = original_hd(0)
+    shot = original_hd(8)
+    # Source-pixel contour fit, in the original 1280x720 image.
+    center = np.array([555.70777, 397.24605])
+    x0, y0, x1, y1 = 503, 369, 610, 443
+    bg = original[y0:y1, x0:x1]
+    fg = shot[y0:y1, x0:x1]
+    yy, xx = np.indices(fg.shape[:2], dtype=np.float32)
+    dx, dy = xx+x0-center[0], yy+y0-center[1]
+    rho = np.sqrt((dx/17.85)**2 + (dy/16.18)**2)
+    alpha = np.clip((1-rho)*17.0 + .5, 0, 1)
+    # Resolve the antialiased source edges against the known clean felt.
+    radiance = (fg - bg*(1-alpha[..., None])) / np.maximum(alpha[..., None], .08)
+    radiance = np.clip(radiance, 130, 255)
+    radiance[alpha == 0] = 0
+    # Premultiplied texture avoids dark edges when shrinking the sphere.
+    sphere = radiance * alpha[..., None]
+    shadow = np.clip((bg-fg)/np.maximum(bg, 1), 0, .96)
+    outside = rho > 1.025
+    shadow *= outside[..., None]
+    shadow *= ((dy > 11) & (dy < 42) & (np.abs(dx) < 47))[..., None]
+    shadow = ndimage.gaussian_filter(shadow, sigma=(.45, .45, 0))
+    return sphere, alpha, shadow, center - np.array([x0,y0])
+
+
+def trajectory():
+    # Rational projection fitted to the unaffected initial rolling motion.
+    p0, p1, p2 = 85.7825399, 5.53845925, .0212926242
+    X = lambda t: (p0+p1*t)/(1+p2*t)
+    Y = lambda t: -.5818614*X(t)+144.040055
+    S = lambda t: (1+p2*8)/(1+p2*t)
+    # Four felt corners yield the plane metric for a physical cushion rebound.
+    H = cv2.getPerspectiveTransform(
+        np.array([[0,0],[1,0],[1,2],[0,2]], np.float32),
+        np.array([[139,43],[200,54],[105,147],[-10,106]], np.float32))
+    inv = np.linalg.inv(H)
+    impact = 44.0
+    radius = 3.57
+    # Project the contact point instead of the elevated sphere center.
+    foot = lambda t: np.array([X(t), Y(t)+radius*S(t)*.93])
+    wall = project(inv, foot(impact))
+    velocity = (wall-project(inv,foot(impact-.1)))/.1
+    velocity[1] *= -1
+    velocity *= .74  # Cushion restitution, measured from the factual shot.
+    reference = project(inv, foot(8))
+    ref_denominator = H[2] @ np.r_[reference, 1.]
+    centers, scales = [], []
+    for t in range(96):
+        if t <= impact:
+            centers.append([X(t),Y(t)])
+            scales.append(S(t))
+        else:
+            dt = t-impact
+            # Gentle rolling loss, in the table plane rather than screen space.
+            travel = (1-np.exp(-.006*dt))/.006
+            world = wall + velocity*travel
+            position = project(H, world)
+            scale = ref_denominator/(H[2] @ np.r_[world, 1.])
+            centers.append([position[0], position[1]-radius*scale*.93])
+            scales.append(scale)
+    return np.array(centers), np.array(scales)
+
+
+def composite_ball(frame, center, scale, texture):
+    sphere, alpha, shadow, origin = texture
+    # Integrate five-by-five subpixels to match the original downsampling.
+    x,y = center
+    x0=max(0,int(np.floor(x-11*scale-2)))
+    x1=min(256,int(np.ceil(x+11*scale+2)))
+    y0=max(0,int(np.floor(y-7*scale-2)))
+    y1=min(144,int(np.ceil(y+11*scale+2)))
+    yy,xx=np.indices(((y1-y0)*5,(x1-x0)*5),dtype=np.float32)
+    # Original pixel positions use the center of each 5x5 RGB sampling cell.
+    world_x = x0+(xx+.5)/5-.5
+    world_y = y0+(yy+.5)/5-.5
+    mapx=((world_x-x)*5/scale+origin[0]).astype(np.float32)
+    mapy=((world_y-y)*5/scale+origin[1]).astype(np.float32)
+    remap=lambda a: cv2.remap(a.astype(np.float32),mapx,mapy,cv2.INTER_LINEAR,
+                              borderMode=cv2.BORDER_CONSTANT,borderValue=0)
+    A=remap(alpha)
+    C=remap(sphere)
+    D=remap(shadow)
+    base=np.asarray(Image.fromarray(frame[y0:y1,x0:x1]).resize(
+        ((x1-x0)*5,(y1-y0)*5),Image.Resampling.BICUBIC),dtype=np.float32)
+    composited=base*(1-D)*(1-A[...,None])+C
+    rgb=np.array(Image.fromarray(np.clip(composited,0,255).round().astype(np.uint8)).resize(
+        (x1-x0,y1-y0),Image.Resampling.LANCZOS))
+    # Preserve untouched RGB values beyond the texture rather than resampling
+    # a rectangular background patch.
+    coverage=np.max(D,axis=2)+A
+    cov=cv2.resize(coverage,(x1-x0,y1-y0),interpolation=cv2.INTER_AREA)
+    active=ndimage.binary_dilation(cov>.0005,iterations=1)
+    patch=frame[y0:y1,x0:x1]
+    patch[active]=rgb[active]
+
+
+def main():
+    source=np.load(ROOT/'source_pixels.npz')['rgb']
+    assert source.shape == (96,144,256,3)
+    median=np.median(source,axis=0).astype(np.float64)
+    plate,field=fit_cloth(median)
+    plate=np.clip(plate,0,255).round().astype(np.uint8)
+    texture=cue_texture()
+    centers,scales=trajectory()
+    yy,xx=np.indices((144,256))
+    # The original target and its complete cast shadow, including codec halo.
+    target=(xx>=121)&(xx<=135)&(yy>=64)&(yy<=79)
+    corridor=(xx>=115)&(xx<=179)&(yy>=40)&(yy<=83)
+    results=[]
+    for i in range(96):
+        frame=source[i].copy()
+        if i < 12:
+            # Initial cue motion is unaffected; remove just the target.
+            # The positive residual also retains any approaching cue halo.
+            delta=source[i].astype(np.int16)-source[0].astype(np.int16)
+            fixed=plate.astype(np.int16)+delta
+            frame[target]=np.clip(fixed[target],0,255).astype(np.uint8)
+        else:
+            # Erase factual ball positions, including the stopped cue ball and
+            # the target's travel/rebound. Recover room/cushion pixels temporally.
+            change=np.max(np.abs(source[i].astype(np.int16)-plate.astype(np.int16)),axis=2)>3
+            erase=ndimage.binary_dilation(change & corridor,iterations=2)&corridor
+            erase |= target
+            frame[erase]=plate[erase]
+            composite_ball(frame,centers[i],scales[i],texture)
+        results.append(frame)
+    frames=np.stack(results).astype(np.uint8)
+    artifact_io.save(frames, ROOT)
+    print('Saved',frames.shape,'RGB frames at 24 fps.')
+    print('Cue centers at frames 1,13,25,45,73,96:',centers[[0,12,24,44,72,95]].round(2))
+
+
+if __name__ == '__main__':
+    main()

\"\"\"Remove the target sphere and reconstruct the unobstructed billiard shot.
Only RGB source pixels, elementary projective geometry, and texture resampling
are used.  The original camera, table, lighting, and room are retained.
\"\"\"
from pathlib import Path
import numpy as np
import cv2
from scipy import ndimage
from PIL import Image
import artifact_io

ROOT = Path(__file__).resolve().parent


def project(H, p):
    q = H @ np.r_[p, 1.0]
    return q[:2] / q[2]


def fit_cloth(plate):
    \"\"\"Fit the very smooth felt under the two overlapping factual spheres.\"\"\"
    yy, xx = np.indices(plate.shape[:2])
    region = (xx > 110) & (xx < 147) & (yy > 57) & (yy < 87)
    excluded = (xx > 117) & (xx < 136) & (yy > 62) & (yy < 80)
    valid = region & ~excluded
    u, v = (xx - 127) / 15, (yy - 71) / 15
    design = np.stack([u * 0 + 1, u, v, u*u, u*v, v*v], -1)
    coefficients = np.linalg.lstsq(design[valid], plate[valid], rcond=None)[0]
    field = design @ coefficients
    # Replace only the coverage of the two balls and their shadows.
    hole = (xx >= 119) & (xx <= 134) & (yy >= 64) & (yy <= 79)
    plate[hole] = field[hole]
    return plate, field


def original_hd(index):
    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
    cap.set(cv2.CAP_PROP_POS_FRAMES, index)
    success, bgr = cap.read()
    cap.release()
    if not success:
        raise RuntimeError('Cannot decode source frame')
    return bgr[:, :, ::-1].astype(np.float32)


def cue_texture():
    \"\"\"Separate source sphere radiance and its soft/contact shadow at frame 9.\"\"\"
    original = original_hd(0)
    shot = original_hd(8)
    # Source-pixel contour fit, in the original 1280x720 image.
    center = np.array([555.70777, 397.24605])
    x0, y0, x1, y1 = 503, 369, 610, 443
    bg = original[y0:y1, x0:x1]
    fg = shot[y0:y1, x0:x1]
    yy, xx = np.indices(fg.shape[:2], dtype=np.float32)
    dx, dy = xx+x0-center[0], yy+y0-center[1]
    rho = np.sqrt((dx/17.85)**2 + (dy/16.18)**2)
    alpha = np.clip((1-rho)*17.0 + .5, 0, 1)
    # Resolve the antialiased source edges against the known clean felt.
    radiance = (fg - bg*(1-alpha[..., None])) / np.maximum(alpha[..., None], .08)
    radiance = np.clip(radiance, 130, 255)
    radiance[alpha == 0] = 0
    # Premultiplied texture avoids dark edges when shrinking the sphere.
    sphere = radiance * alpha[..., None]
    shadow = np.clip((bg-fg)/np.maximum(bg, 1), 0, .96)
    outside = rho > 1.025
    shadow *= outside[..., None]
    shadow *= ((dy > 11) & (dy < 42) & (np.abs(dx) < 47))[..., None]
    shadow = ndimage.gaussian_filter(shadow, sigma=(.45, .45, 0))
    return sphere, alpha, shadow, center - np.array([x0,y0])


def trajectory():
    # Rational projection fitted to the unaffected initial rolling motion.
    p0, p1, p2 = 85.7825399, 5.53845925, .0212926242
    X = lambda t: (p0+p1*t)/(1+p2*t)
    Y = lambda t: -.5818614*X(t)+144.040055
    S = lambda t: (1+p2*8)/(1+p2*t)
    # Four felt corners yield the plane metric for a physical cushion rebound.
    H = cv2.getPerspectiveTransform(
        np.array([[0,0],[1,0],[1,2],[0,2]], np.float32),
        np.array([[139,43],[200,54],[105,147],[-10,106]], np.float32))
    inv = np.linalg.inv(H)
    impact = 44.0
    radius = 3.57
    # Project the contact point instead of the elevated sphere center.
    foot = lambda t: np.array([X(t), Y(t)+radius*S(t)*.93])
    wall = project(inv, foot(impact))
    velocity = (wall-project(inv,foot(impact-.1)))/.1
    velocity[1] *= -1
    velocity *= .74  # Cushion restitution, measured from the factual shot.
    reference = project(inv, foot(8))
    ref_denominator = H[2] @ np.r_[reference, 1.]
    centers, scales = [], []
    for t in range(96):
        if t <= impact:
            centers.append([X(t),Y(t)])
            scales.append(S(t))
        else:
            dt = t-impact
            # Gentle rolling loss, in the table plane rather than screen space.
            travel = (1-np.exp(-.006*dt))/.006
            world = wall + velocity*travel
            position = project(H, world)
            scale = ref_denominator/(H[2] @ np.r_[world, 1.])
            centers.append([position[0], position[1]-radius*scale*.93])
            scales.append(scale)
    return np.array(centers), np.array(scales)


def composite_ball(frame, center, scale, texture):
    sphere, alpha, shadow, origin = texture
    # Integrate five-by-five subpixels to match the original downsampling.
    x,y = center
    x0=max(0,int(np.floor(x-11*scale-2)))
    x1=min(256,int(np.ceil(x+11*scale+2)))
    y0=max(0,int(np.floor(y-7*scale-2)))
    y1=min(144,int(np.ceil(y+11*scale+2)))
    yy,xx=np.indices(((y1-y0)*5,(x1-x0)*5),dtype=np.float32)
    # Original pixel positions use the center of each 5x5 RGB sampling cell.
    world_x = x0+(xx+.5)/5-.5
    world_y = y0+(yy+.5)/5-.5
    mapx=((world_x-x)*5/scale+origin[0]).astype(np.float32)
    mapy=((world_y-y)*5/scale+origin[1]).astype(np.float32)
    remap=lambda a: cv2.remap(a.astype(np.float32),mapx,mapy,cv2.INTER_LINEAR,
                              borderMode=cv2.BORDER_CONSTANT,borderValue=0)
    A=remap(alpha)
    C=remap(sphere)
    D=remap(shadow)
    base=np.asarray(Image.fromarray(frame[y0:y1,x0:x1]).resize(
        ((x1-x0)*5,(y1-y0)*5),Image.Resampling.BICUBIC),dtype=np.float32)
    composited=base*(1-D)*(1-A[...,None])+C
    rgb=np.array(Image.fromarray(np.clip(composited,0,255).round().astype(np.uint8)).resize(
        (x1-x0,y1-y0),Image.Resampling.LANCZOS))
    # Preserve untouched RGB values beyond the texture rather than resampling
    # a rectangular background patch.
    coverage=np.max(D,axis=2)+A
    cov=cv2.resize(coverage,(x1-x0,y1-y0),interpolation=cv2.INTER_AREA)
    active=ndimage.binary_dilation(cov>.0005,iterations=1)
    patch=frame[y0:y1,x0:x1]
    patch[active]=rgb[active]


def main():
    source=np.load(ROOT/'source_pixels.npz')['rgb']
    assert source.shape == (96,144,256,3)
    median=np.median(source,axis=0).astype(np.float64)
    plate,field=fit_cloth(median)
    plate=np.clip(plate,0,255).round().astype(np.uint8)
    texture=cue_texture()
    centers,scales=trajectory()
    yy,xx=np.indices((144,256))
    # The original target and its complete cast shadow, including codec halo.
    target=(xx>=121)&(xx<=135)&(yy>=64)&(yy<=79)
    corridor=(xx>=115)&(xx<=179)&(yy>=40)&(yy<=83)
    results=[]
    for i in range(96):
        frame=source[i].copy()
        if i < 12:
            # Initial cue motion is unaffected; remove just the target.
            # The positive residual also retains any approaching cue halo.
            delta=source[i].astype(np.int16)-source[0].astype(np.int16)
            fixed=plate.astype(np.int16)+delta
            frame[target]=np.clip(fixed[target],0,255).astype(np.uint8)
        else:
            # Erase factual ball positions, including the stopped cue ball and
            # the target's travel/rebound. Recover room/cushion pixels temporally.
            change=np.max(np.abs(source[i].astype(np.int16)-plate.astype(np.int16)),axis=2)>3
            erase=ndimage.binary_dilation(change & corridor,iterations=2)&corridor
            erase |= target
            frame[erase]=plate[erase]
            composite_ball(frame,centers[i],scales[i],texture)
        results.append(frame)
    frames=np.stack(results).astype(np.uint8)
    artifact_io.save(frames, ROOT)
    print('Saved',frames.shape,'RGB frames at 24 fps.')
    print('Cue centers at frames 1,13,25,45,73,96:',centers[[0,12,24,44,72,95]].round(2))


if __name__ == '__main__':
    main()
