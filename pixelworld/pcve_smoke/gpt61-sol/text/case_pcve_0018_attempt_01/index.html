+
+All appearance comes from the supplied recording.  The moving spheres are
+separated numerically, including their floor shadows, and are composited onto a
+clean temporal plate.  The launch is changed at t=0, not by adding new footage.

+import numpy as np
+from scipy.ndimage import map_coordinates, distance_transform_edt, gaussian_filter, binary_dilation

+def fit_tracks():

+        # the bottom of the red ball.

+def body_mask(p, extra=0):
+    cx,cy,r=p
+    # Coverage width agrees with the antialiased source edge at 256 pixels.
+    return np.clip(r+extra+.55-np.hypot(X-cx,Y-cy),0,1)

+def shadow_mask(p, kind):
+    cx,cy,r=p
+    # Soft, downward cast floor shadows from the scene's high light source.
+    ry=r*(1.24 if kind=='soccer' else .40)
+    sy=cy+r*(1.53 if kind=='soccer' else 1.25)
+    rx=r*(1.4 if kind=='soccer' else 1.20)
+    return np.exp(-2.1*(((X-cx+1)/rx)**2+((Y-sy)/ry)**2))

+def clean_plate():
+    masks=[]
+    for i in range(N):
+        m=body_mask(STAR[i],1.5)>0
+        m|=body_mask(SOCCER[i],2)>0
+        m|=shadow_mask(STAR[i],'star')>.025

+        if 9<=i<=14:m[54:91,62:99]=True

+    data=np.where(valid[...,None],SRC,np.nan)

+        plate=np.nanmedian(data,axis=0)
+    missing=~np.isfinite(plate[:,:,0])
+    # Only the continuously occluded patch of wood needs reconstruction.  Use
+    # smooth row interpolation from the supplied floor pixels, retaining grain.
+    for y in range(H):
+        good=np.where(~missing[y])[0]
+        if len(good):

+    plate=np.nan_to_num(plate,nan=210)
+    # Relax the small hidden patch with fixed, measured boundary RGB values.
+    for _ in range(160):
+        avg=(np.roll(plate,1,0)+np.roll(plate,-1,0)+np.roll(plate,1,1)+np.roll(plate,-1,1))*.25
+        plate[missing]=avg[missing]

+    # averaging; this avoids changing the static pink box and room details.
+    changed=np.any(np.asarray(masks),axis=0)
+    plate[~changed]=SRC[0][~changed]
+    return plate

+PLATE=clean_plate()

+def warp(arr,p,q):
+    scale=q[2]/p[2]
+    xx=(X-q[0])/scale+p[0]
+    yy=(Y-q[1])/scale+p[1]
+    if arr.ndim==2:
+        return map_coordinates(arr,[yy,xx],order=1,mode='constant',cval=0,prefilter=False)
+    return np.stack([map_coordinates(arr[:,:,c],[yy,xx],order=1,mode='constant',cval=0,prefilter=False) for c in range(arr.shape[2])],axis=-1)
+

+def layer(i,kind,p,q):
+    frame=SRC[i]
+    a=body_mask(p)
+    if kind=='star':
+        # The nearer soccer sphere must be rendered in front, not baked into
+        # the red sphere's lower cap when its counterfactual position changes.
+        covered=body_mask(SOCCER[i],.5)>.15
+        need=covered & (a>0)
+        good=(a>.15)&~covered
+        if need.any() and good.any():
+            inds=distance_transform_edt(~good,return_distances=False,return_indices=True)
+            frame=frame.copy();frame[need]=frame[inds[0][need],inds[1][need]]

+    prem=np.clip(prem,0,255*a[...,None])
+    return warp(prem,p,q),warp(a,p,q)
+

+def floor_shadow(i,kind,p,q):
+    mask=shadow_mask(p,kind)
+    # Do not include either sphere in the floor-shadow residual.
+    mask*=1-body_mask(STAR[i],1)
+    mask*=1-body_mask(SOCCER[i],1)
+    residual=np.minimum(SRC[i]-PLATE,0)*mask[...,None]
+    return warp(residual,p,q)
+
+
+def interp_track(track,s):
+    lo=int(np.floor(np.clip(s,0,N-1))); hi=min(lo+1,N-1)
+    w=np.clip(s-lo,0,1)
+    return track[lo]*(1-w)+track[hi]*w
+
+
+def perspective_scale_position(p,anchor,factor):
+    # Perspective-aware interpolation: inverse projected radius is linear in
+    # floor depth.  Both spheres keep their rigid spherical shapes.
+    u=p[:2]; a=anchor[:2];ra=anchor[2];r=p[2]
+    rr=1/((1-factor)/ra+factor/r)
+    pos=rr*((1-factor)*a/ra+factor*u/r)
+    return np.r_[pos,rr]
+

+def main():
+    # Measured source launch has gentle rolling deceleration.  Scale velocity,
+    # not acceleration.  Solve the source displacement for each edited time.
+    v=14.7983; dec=.13860
+    wall_s=11.55
+    wall_dist=v*wall_s-.5*dec*wall_s**2
+    nv=.75*v
+    wall_t=(nv-np.sqrt(nv*nv-2*dec*wall_dist))/dec
+    frames=[]
+    for t in range(N):
+        if t==0:
+            frames.append(SRC[0].astype(np.uint8));continue
+        if t<wall_t:
+            dist=nv*t-.5*dec*t*t
+            s=(v-np.sqrt(max(0,v*v-2*dec*dist)))/dec
+            target=interp_track(STAR,s)
+            targets=[target,SOCCER[0]]
+            indices=[s,0.0]
+        else:
+            # Reduced rebound momentum gives a slower wall-to-ball approach.
+            # The contact sequence remains source-supported, with a shorter
+            # roll-out for both the striking ball and the struck soccer ball.
+            s=wall_s+.75*(t-wall_t)
+            p=interp_track(STAR,s)
+            factor=.80
+            target=perspective_scale_position(p,STAR[12],factor)
+            # At the rebound itself, retain the wall-contact position exactly.
+            onset=np.clip((s-wall_s)/2.8,0,1)
+            target=(1-onset)*p+onset*target
+            sq=interp_track(SOCCER,s)
+            soccer=perspective_scale_position(sq,SOCCER[0],.72)
+            targets=[target,soccer]
+            indices=[s,s if s>=23 else 0.0]
+        out=PLATE.copy()
+        # Each appearance is interpolated *after* registering its sphere to the
+        # target position.  This gives smooth 75% launch motion, not dissolves
+        # between two different ball positions.
+        for kind,track,q,phase in zip(['star','soccer'],[STAR,SOCCER],targets,indices):
+            lo=int(np.clip(np.floor(phase),0,N-1));hi=min(lo+1,N-1);w=np.clip(phase-lo,0,1)
+            out+=(1-w)*floor_shadow(lo,kind,track[lo],q)+w*floor_shadow(hi,kind,track[hi],q)
+        for kind,track,q,phase in zip(['star','soccer'],[STAR,SOCCER],targets,indices):
+            lo=int(np.clip(np.floor(phase),0,N-1));hi=min(lo+1,N-1);w=np.clip(phase-lo,0,1)
+            rgb0,a0=layer(lo,kind,track[lo],q)
+            rgb1,a1=layer(hi,kind,track[hi],q)
+            a=(1-w)*a0+w*a1;rgb=(1-w)*rgb0+w*rgb1
+            out=out*(1-a[...,None])+rgb
+        frames.append(np.clip(np.rint(out),0,255).astype(np.uint8))
+    frames=np.asarray(frames,dtype=np.uint8)
+    artifact_io.save(frames,ROOT)
+    np.savez_compressed(ROOT/'motion_diagnostics.npz',star=STAR,soccer=SOCCER,wall_times=[wall_s,wall_t])
+    print(f'Saved {len(frames)} RGB frames, {W}x{H}, 24 fps; wall contact {wall_t/24:.3f}s')
+
+if __name__=='__main__':main()

def fit_tracks():

def body_mask(p, extra=0):
    cx,cy,r=p
    # Coverage width agrees with the antialiased source edge at 256 pixels.
    return np.clip(r+extra+.55-np.hypot(X-cx,Y-cy),0,1)


def shadow_mask(p, kind):
    cx,cy,r=p
    # Soft, downward cast floor shadows from the scene's high light source.
    ry=r*(1.24 if kind=='soccer' else .40)
    sy=cy+r*(1.53 if kind=='soccer' else 1.25)
    rx=r*(1.4 if kind=='soccer' else 1.20)
    return np.exp(-2.1*(((X-cx+1)/rx)**2+((Y-sy)/ry)**2))


def clean_plate():
    masks=[]
    for i in range(N):
        m=body_mask(STAR[i],1.5)>0
        m|=body_mask(SOCCER[i],2)>0
        m|=shadow_mask(STAR[i],'star')>.025
        m|=shadow_mask(SOCCER[i],'soccer')>.025
        # Compression fringes / reflection at the wall are also moving light.
        if 9<=i<=14:m[54:91,62:99]=True
        masks.append(m)
    valid=~np.asarray(masks)
    data=np.where(valid[...,None],SRC,np.nan)
    import warnings
    with warnings.catch_warnings():
        warnings.simplefilter('ignore',RuntimeWarning)
        plate=np.nanmedian(data,axis=0)
    missing=~np.isfinite(plate[:,:,0])
    # Only the continuously occluded patch of wood needs reconstruction.  Use
    # smooth row interpolation from the supplied floor pixels, retaining grain.
    for y in range(H):
        good=np.where(~missing[y])[0]
        if len(good):
            for c in range(3):
                plate[y,:,c]=np.interp(np.arange(W),good,plate[y,good,c])
    plate=np.nan_to_num(plate,nan=210)
    # Relax the small hidden patch with fixed, measured boundary RGB values.
    for _ in range(160):
        avg=(np.roll(plate,1,0)+np.roll(plate,-1,0)+np.roll(plate,1,1)+np.roll(plate,-1,1))*.25
        plate[missing]=avg[missing]
    # Unaffected background uses the exact first-frame plate instead of temporal
    # averaging; this avoids changing the static pink box and room details.
    changed=np.any(np.asarray(masks),axis=0)
    plate[~changed]=SRC[0][~changed]
    return plate

PLATE=clean_plate()


def warp(arr,p,q):
    scale=q[2]/p[2]
    xx=(X-q[0])/scale+p[0]
    yy=(Y-q[1])/scale+p[1]
    if arr.ndim==2:
        return map_coordinates(arr,[yy,xx],order=1,mode='constant',cval=0,prefilter=False)
    return np.stack([map_coordinates(arr[:,:,c],[yy,xx],order=1,mode='constant',cval=0,prefilter=False) for c in range(arr.shape[2])],axis=-1)


def layer(i,kind,p,q):
    frame=SRC[i]
    a=body_mask(p)
    if kind=='star':
        # The nearer soccer sphere must be rendered in front, not baked into
        # the red sphere's lower cap when its counterfactual position changes.
        covered=body_mask(SOCCER[i],.5)>.15
        need=covered & (a>0)
        good=(a>.15)&~covered
        if need.any() and good.any():
            inds=distance_transform_edt(~good,return_distances=False,return_indices=True)
            frame=frame.copy();frame[need]=frame[inds[0][need],inds[1][need]]
    prem=frame-(1-a[...,None])*PLATE
    prem=np.clip(prem,0,255*a[...,None])
    return warp(prem,p,q),warp(a,p,q)


def floor_shadow(i,kind,p,q):
    mask=shadow_mask(p,kind)
    # Do not include either sphere in the floor-shadow residual.
    mask*=1-body_mask(STAR[i],1)
    mask*=1-body_mask(SOCCER[i],1)
    residual=np.minimum(SRC[i]-PLATE,0)*mask[...,None]
    return warp(residual,p,q)


def interp_track(track,s):
    lo=int(np.floor(np.clip(s,0,N-1))); hi=min(lo+1,N-1)
    w=np.clip(s-lo,0,1)
    return track[lo]*(1-w)+track[hi]*w


def perspective_scale_position(p,anchor,factor):
    # Perspective-aware interpolation: inverse projected radius is linear in
    # floor depth.  Both spheres keep their rigid spherical shapes.
    u=p[:2]; a=anchor[:2];ra=anchor[2];r=p[2]
    rr=1/((1-factor)/ra+factor/r)
    pos=rr*((1-factor)*a/ra+factor*u/r)
    return np.r_[pos,rr]


def main():
    # Measured source launch has gentle rolling deceleration.  Scale velocity,
    # not acceleration.  Solve the source displacement for each edited time.
    v=14.7983; dec=.13860
    wall_s=11.55
    wall_dist=v*wall_s-.5*dec*wall_s**2
    nv=.75*v
    wall_t=(nv-np.sqrt(nv*nv-2*dec*wall_dist))/dec
    frames=[]
    for t in range(N):
        if t==0:
            frames.append(SRC[0].astype(np.uint8));continue
        if t<wall_t:
            dist=nv*t-.5*dec*t*t
            s=(v-np.sqrt(max(0,v*v-2*dec*dist)))/dec
            target=interp_track(STAR,s)
            targets=[target,SOCCER[0]]
            indices=[s,0.0]
        else:
            # Reduced rebound momentum gives a slower wall-to-ball approach.
            # The contact sequence remains source-supported, with a shorter
            # roll-out for both the striking ball and the struck soccer ball.
            s=wall_s+.75*(t-wall_t)
            p=interp_track(STAR,s)
            factor=.80
            target=perspective_scale_position(p,STAR[12],factor)
            # At the rebound itself, retain the wall-contact position exactly.
            onset=np.clip((s-wall_s)/2.8,0,1)
            target=(1-onset)*p+onset*target
            sq=interp_track(SOCCER,s)
            soccer=perspective_scale_position(sq,SOCCER[0],.72)
            targets=[target,soccer]
            indices=[s,s if s>=23 else 0.0]
        out=PLATE.copy()
        # Each appearance is interpolated *after* registering its sphere to the
        # target position.  This gives smooth 75% launch motion, not dissolves
        # between two different ball positions.
        for kind,track,q,phase in zip(['star','soccer'],[STAR,SOCCER],targets,indices):
            lo=int(np.clip(np.floor(phase),0,N-1));hi=min(lo+1,N-1);w=np.clip(phase-lo,0,1)
            out+=(1-w)*floor_shadow(lo,kind,track[lo],q)+w*floor_shadow(hi,kind,track[hi],q)
        for kind,track,q,phase in zip(['star','soccer'],[STAR,SOCCER],targets,indices):
            lo=int(np.clip(np.floor(phase),0,N-1));hi=min(lo+1,N-1);w=np.clip(phase-lo,0,1)
            rgb0,a0=layer(lo,kind,track[lo],q)
            rgb1,a1=layer(hi,kind,track[hi],q)
            a=(1-w)*a0+w*a1;rgb=(1-w)*rgb0+w*rgb1
            out=out*(1-a[...,None])+rgb
        frames.append(np.clip(np.rint(out),0,255).astype(np.uint8))
    frames=np.asarray(frames,dtype=np.uint8)
    artifact_io.save(frames,ROOT)

    print(f'Saved {len(frames)} RGB frames, {W}x{H}, 24 fps; wall contact {wall_t/24:.3f}s')

if __name__=='__main__':main()
