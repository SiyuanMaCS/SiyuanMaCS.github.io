+"""Remove the red mallet and reconstruct the resulting source-timeline impacts.
+
+All geometry, tracks, textures and mattes are derived from the local RGB video.
+No learned models or renderer are used.  Coordinates below are in source pixels.
+"""
+from pathlib import Path
+import warnings
+import numpy as np
+from scipy.ndimage import label, find_objects, binary_dilation, distance_transform_edt, map_coordinates
+from scipy.optimize import least_squares
+from scipy.sparse import lil_matrix
+from scipy.sparse.linalg import spsolve
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+Y, X = np.indices((144, 256))
+
+
+def get_tracks(src):
+    red, blue, white = [], [], []
+    region = (X > 90) & (X < 177) & (Y > 49) & (Y < 85)
+    for i, frame in enumerate(src):
+        r, g, b = frame.transpose(2, 0, 1)
+        for target, m in [(red, region & (r > 1.4*g) & (r > 1.4*b) & (r > 70)),
+                          (blue, region & (b > 1.5*r) & (b > 1.2*g) & (b > 60))]:
+            yy, xx = np.where(m)
+            target.append((xx.mean(), yy.mean()))
+        xp = 105 if i < 30 else 105 - 2.13*(i-30) if i < 49 else 68 + .05*(i-49)
+        yp = 81 if i < 30 else 81 + .82*(i-30) if i < 49 else 95.3
+        m = ((abs(X-xp) < 9) & (abs(Y-yp) < 5) & (r > 230) & (g > 228)
+             & (b > 222) & (abs(r-b) < 20) & (abs(g-b) < 23))
+        labs, _ = label(m)
+        slices = find_objects(labs)
+        k = max(range(len(slices)), key=lambda k: (labs[slices[k]] == k+1).sum())
+        yy, xx = np.where(labs == k+1)
+        white.append((xx.mean(), yy.mean()))
+    return np.array(red), np.array(blue), np.array(white)
+
+
+def object_masks(src, red, blue, white):
+    tight = np.zeros(src.shape[:3], bool)
+    wide = np.zeros_like(tight)
+    for i, frame in enumerate(src):
+        r, g, b = frame.transpose(2, 0, 1)
+        x, y = red[i]
+        core = (abs(X-x) < 8) & (abs(Y-y) < 5) & (r-g > 22) & (r-b > 22)
+        tight[i] |= binary_dilation(core)
+        tight[i] |= ((X-x-.4)/6.5)**2 + ((Y-y-3.8)/2.8)**2 < 1
+        x, y = blue[i]
+        core = (abs(X-x) < 8) & (abs(Y-y) < 5) & (b-r > 32) & (b-g > 8)
+        tight[i] |= binary_dilation(core)
+        tight[i] |= ((X-x-.8)/5)**2 + ((Y-y-4.2)/4.8)**2 < 1
+        x, y = white[i]
+        tight[i] |= (abs(X-x) < 9) & (Y > y-5) & (Y < y+6)
+        for (x, y), width, up, down in [(red[i], 9, 7, 9),
+                                        (blue[i], 9, 7, 12),
+                                        (white[i], 10, 7, 8)]:
+            wide[i] |= (abs(X-x) < width) & (Y > y-up) & (Y < y+down)
+    return tight, wide
+
+
+def restore_holes(plate):
+    """Source-texture exemplar plus a Dirichlet Poisson seam correction."""
+    missing = np.isnan(plate[:, :, 0])
+    labs, _ = label(missing)
+    for k, box in enumerate(find_objects(labs)):
+        hole = labs == k+1
+        ring = binary_dilation(hole, iterations=3) & ~missing
+        yy, xx = np.where(ring)
+        yh, xh = np.where(hole)
+        xm, ym = xx.mean(), yy.mean()
+        basis = np.stack([np.ones(len(xx)), (xx-xm)/10, (yy-ym)/10], 1)
+        best = None
+        for dy in range(-18, 19):
+            for dx in range(-30, 31):
+                if (dx == dy == 0 or min(xx+dx) < 0 or max(xx+dx) >= 256
+                    or min(yy+dy) < 0 or max(yy+dy) >= 144):
+                    continue
+                sample = plate[yy+dy, xx+dx]
+                inner = plate[yh+dy, xh+dx]
+                if not np.isfinite(sample).all() or not np.isfinite(inner).all():
+                    continue
+                diff = plate[yy, xx] - sample
+                coef = np.linalg.lstsq(basis, diff, rcond=None)[0]
+                residual = diff - basis@coef
+                score = (np.mean(residual**2) + .007*np.mean((basis@coef)**2)
+                         + .025*(dx*dx+dy*dy))
+                if best is None or score < best[0]:
+                    best = (score, dx, dy, coef)
+        _, dx, dy, coef = best
+        # A one-pixel expanded exemplar provides guidance gradients at the seam.
+        allpoints = binary_dilation(hole)
+        ye, xe = np.where(allpoints)
+        patch = plate.copy()
+        bas = np.stack([np.ones(len(xe)), (xe-xm)/10, (ye-ym)/10], 1)
+        patch[ye, xe] = plate[ye+dy, xe+dx] + bas@coef
+        lookup = {(int(y), int(x)): j for j, (y, x) in enumerate(zip(yh, xh))}
+        A = lil_matrix((len(yh), len(yh)))
+        rhs = np.zeros((len(yh), 3))
+        for j, (y, x) in enumerate(zip(yh, xh)):
+            A[j, j] = 4
+            for yn, xn in [(y-1, x), (y+1, x), (y, x-1), (y, x+1)]:
+                # Moderate the exemplar gradients in the unobserved small area.
+                rhs[j] += .75*(patch[y, x] - patch[yn, xn])
+                if (int(yn), int(xn)) in lookup:
+                    A[j, lookup[(int(yn), int(xn))]] = -1
+                else:
+                    rhs[j] += plate[yn, xn]
+        plate[yh, xh] = spsolve(A.tocsr(), rhs)
+    return np.clip(plate, 0, 255)
+
+
+def make_blue_layers(src, plate, center):
+    """Unpremultiply a source-pixel mallet matte; retain its soft floor shadow."""
+    delta = src[0] - plate
+    x, y = center
+    region = (abs(X-x) < 9) & (Y > y-7) & (Y < y+12)
+    # The blue-vs-floor chroma difference yields a subpixel edge matte.
+    chroma = ((src[0, :, :, 1]-src[0, :, :, 0]) -
+              (plate[:, :, 1]-plate[:, :, 0]))
+    alpha = np.clip(chroma/54., 0, 1)
+    alpha *= region & (Y < y+3.1)
+    alpha[alpha < .06] = 0
+    # Preserve the highlights of the originally rendered head and handle.
+    fg = np.zeros_like(src[0])
+    m = alpha > 0
+    fg[m] = (src[0][m] - (1-alpha[m, None])*plate[m]) / alpha[m, None]
+    fg = np.clip(fg, 0, 255)
+    # Source shadows are extracted independently, never carried as a floor tile.
+    shadow = np.zeros_like(src[0])
+    residual = delta - alpha[:, :, None]*(fg-plate)
+    shadow_region = region & (Y > y+1.3)
+    magnitude = np.maximum(-residual[:, :, :2].mean(2)-3., 0)
+    weight = np.clip(magnitude/7., 0, 1) * shadow_region
+    shadow = np.minimum(residual, 0) * weight[:, :, None]
+    # Remove isolated compression ripple at the edge of the extracted footprint.
+    taper = np.minimum(1, distance_transform_edt(region)/2)
+    shadow *= taper[:, :, None]
+    return alpha, fg*alpha[:, :, None], shadow
+
+
+def warp_layer(layer, origin, target, scale):
+    ox, oy = origin
+    tx, ty = target
+    coords = [(Y-ty)/scale+oy, (X-tx)/scale+ox]
+    if layer.ndim == 2:
+        return map_coordinates(layer, coords, order=1, mode='constant', cval=0)
+    return np.stack([map_coordinates(layer[:, :, c], coords, order=1,
+                                    mode='constant', cval=0) for c in range(3)], 2)
+
+
+def white_layer(src, plate, center, i):
+    x, y = center
+    region = (abs(X-x) < 10) & (Y > y-7) & (Y < y+8)
+    taper = np.minimum(1, distance_transform_edt(region)/2)
+    diff = (src[i] - plate) * taper[:, :, None]
+    diff[abs(diff).max(2) < 3] = 0
+    return diff
+
+
+def generate():
+    src = np.load(ROOT/'source_pixels.npz')['rgb'].astype(np.float64)
+    red, blue, white = get_tracks(src)
+    tight, wide = object_masks(src, red, blue, white)
+    with warnings.catch_warnings():
+        warnings.simplefilter('ignore', RuntimeWarning)
+        plate = np.nanmedian(np.where(~tight[:, :, :, None], src, np.nan), axis=0)
+    plate = restore_holes(plate)
+
+    # A perspective projection of constant plane velocity fits the pre-impact
+    # centers.  It predicts the blue mallet's unobstructed flight across the table.
+    tt = np.arange(14, dtype=float)
+    def projected(p, t):
+        return (p[0]+p[1]*t)/(1+p[2]*t)
+    params = least_squares(lambda p: projected(p, tt)-blue[:14, 0],
+                           [167.5, -2.45, -.0067]).x
+    slope, intercept = np.polyfit(blue[:14, 0], blue[:14, 1], 1)
+    # The source head-to-head contact with the white mallet occurred at about
+    # source time 29.5.  Match that contact distance, without the red mediator.
+    contact_x = np.interp(29.6, np.arange(96), red[:, 0])
+    contact_t = (params[0]-contact_x)/(contact_x*params[2]-params[1])
+    delay = contact_t - 29.6
+    alpha, premul, shadow = make_blue_layers(src, plate, blue[0])
+    whites = [white_layer(src, plate, white[i], i) for i in range(96)]
+    out = []
+    for i in range(96):
+        # Retain the original scene and its very small encoding/lighting changes.
+        erase = np.minimum(1, distance_transform_edt(wide[i])/1.6)
+        frame = src[i]*(1-erase[:, :, None]) + plate*erase[:, :, None]
+
+        # Re-time the white mallet only after the newly delayed direct collision.
+        wi = max(0., i-delay)
+        lo = min(95, int(np.floor(wi)))
+        hi = min(95, lo+1)
+        mix = wi-lo
+        frame += (1-mix)*whites[lo] + mix*whites[hi]
+
+        if i <= 13:
+            # Factual blue history before any possible contact is unchanged.
+            x, y = blue[i]
+            m = (abs(X-x) < 9) & (Y > y-7) & (Y < y+12)
+            taper = np.minimum(1, distance_transform_edt(m)/2)
+            frame += (src[i]-plate)*taper[:, :, None]
+        else:
+            if i <= contact_t:
+                bx = projected(params, float(i))
+                by = intercept+slope*bx
+            else:
+                # Reuse the source's decaying contact recoil on the now-blue body.
+                s = min(95., 29.6 + i-contact_t)
+                bx = np.interp(s, np.arange(96), red[:, 0])
+                by = intercept+slope*bx
+            target = (bx, by)
+            # Projective depth enlarges the mallet slightly as it approaches.
+            scale = (1-params[2]*0)/(1+params[2]*min(float(i), contact_t))
+            scale = 1 + .68*(scale-1)
+            a = warp_layer(alpha, blue[0], target, scale)
+            p = warp_layer(premul, blue[0], target, scale)
+            sh = warp_layer(shadow, blue[0], target, scale)
+            frame += sh
+            frame = frame*(1-a[:, :, None]) + p
+        out.append(np.clip(np.rint(frame), 0, 255).astype(np.uint8))
+    frames = np.stack(out)
+    artifact_io.save(frames, ROOT)
+    print(f'Saved {frames.shape}, uint8 RGB, 24 fps. Direct impact at frame {contact_t+1:.2f}; white delay {delay:.2f} frames.')
+    return frames
+
+
+if __name__ == '__main__':
+    generate()

from pathlib import Path
import warnings
import numpy as np
from scipy.ndimage import label, find_objects, binary_dilation, distance_transform_edt, map_coordinates
from scipy.optimize import least_squares
from scipy.sparse import lil_matrix
from scipy.sparse.linalg import spsolve
import artifact_io

def get_tracks(src):
    red, blue, white = [], [], []
    region = (X > 90) & (X < 177) & (Y > 49) & (Y < 85)
    for i, frame in enumerate(src):
        r, g, b = frame.transpose(2, 0, 1)
        for target, m in [(red, region & (r > 1.4*g) & (r > 1.4*b) & (r > 70)),
                          (blue, region & (b > 1.5*r) & (b > 1.2*g) & (b > 60))]:
            yy, xx = np.where(m)
            target.append((xx.mean(), yy.mean()))
        xp = 105 if i < 30 else 105 - 2.13*(i-30) if i < 49 else 68 + .05*(i-49)
        yp = 81 if i < 30 else 81 + .82*(i-30) if i < 49 else 95.3
        m = ((abs(X-xp) < 9) & (abs(Y-yp) < 5) & (r > 230) & (g > 228)
             & (b > 222) & (abs(r-b) < 20) & (abs(g-b) < 23))
        labs, _ = label(m)
        slices = find_objects(labs)
        k = max(range(len(slices)), key=lambda k: (labs[slices[k]] == k+1).sum())
        yy, xx = np.where(labs == k+1)
        white.append((xx.mean(), yy.mean()))
    return np.array(red), np.array(blue), np.array(white)


def object_masks(src, red, blue, white):
    tight = np.zeros(src.shape[:3], bool)
    wide = np.zeros_like(tight)
    for i, frame in enumerate(src):
        r, g, b = frame.transpose(2, 0, 1)
        x, y = red[i]
        core = (abs(X-x) < 8) & (abs(Y-y) < 5) & (r-g > 22) & (r-b > 22)
        tight[i] |= binary_dilation(core)
        tight[i] |= ((X-x-.4)/6.5)**2 + ((Y-y-3.8)/2.8)**2 < 1
        x, y = blue[i]
        core = (abs(X-x) < 8) & (abs(Y-y) < 5) & (b-r > 32) & (b-g > 8)
        tight[i] |= binary_dilation(core)
        tight[i] |= ((X-x-.8)/5)**2 + ((Y-y-4.2)/4.8)**2 < 1
        x, y = white[i]
        tight[i] |= (abs(X-x) < 9) & (Y > y-5) & (Y < y+6)
        for (x, y), width, up, down in [(red[i], 9, 7, 9),
                                        (blue[i], 9, 7, 12),
                                        (white[i], 10, 7, 8)]:
            wide[i] |= (abs(X-x) < width) & (Y > y-up) & (Y < y+down)
    return tight, wide


def restore_holes(plate):
    \"\"\"Source-texture exemplar plus a Dirichlet Poisson seam correction.\"\"\"
    missing = np.isnan(plate[:, :, 0])
    labs, _ = label(missing)
    for k, box in enumerate(find_objects(labs)):
        hole = labs == k+1
        ring = binary_dilation(hole, iterations=3) & ~missing
        yy, xx = np.where(ring)
        yh, xh = np.where(hole)
        xm, ym = xx.mean(), yy.mean()
        basis = np.stack([np.ones(len(xx)), (xx-xm)/10, (yy-ym)/10], 1)
        best = None
        for dy in range(-18, 19):
            for dx in range(-30, 31):
                if (dx == dy == 0 or min(xx+dx) < 0 or max(xx+dx) >= 256
                    or min(yy+dy) < 0 or max(yy+dy) >= 144):
                    continue
                sample = plate[yy+dy, xx+dx]
                inner = plate[yh+dy, xh+dx]
                if not np.isfinite(sample).all() or not np.isfinite(inner).all():
                    continue
                diff = plate[yy, xx] - sample
                coef = np.linalg.lstsq(basis, diff, rcond=None)[0]
                residual = diff - basis@coef
                score = (np.mean(residual**2) + .007*np.mean((basis@coef)**2)
                         + .025*(dx*dx+dy*dy))
                if best is None or score < best[0]:
                    best = (score, dx, dy, coef)
        _, dx, dy, coef = best
        # A one-pixel expanded exemplar provides guidance gradients at the seam.
        allpoints = binary_dilation(hole)
        ye, xe = np.where(allpoints)
        patch = plate.copy()
        bas = np.stack([np.ones(len(xe)), (xe-xm)/10, (ye-ym)/10], 1)
        patch[ye, xe] = plate[ye+dy, xe+dx] + bas@coef
        lookup = {(int(y), int(x)): j for j, (y, x) in enumerate(zip(yh, xh))}
        A = lil_matrix((len(yh), len(yh)))
        rhs = np.zeros((len(yh), 3))
        for j, (y, x) in enumerate(zip(yh, xh)):
            A[j, j] = 4
            for yn, xn in [(y-1, x), (y+1, x), (y, x-1), (y, x+1)]:
                # Moderate the exemplar gradients in the unobserved small area.
                rhs[j] += .75*(patch[y, x] - patch[yn, xn])
                if (int(yn), int(xn)) in lookup:
                    A[j, lookup[(int(yn), int(xn))]] = -1
                else:
                    rhs[j] += plate[yn, xn]
        plate[yh, xh] = spsolve(A.tocsr(), rhs)
    return np.clip(plate, 0, 255)


def make_blue_layers(src, plate, center):
    \"\"\"Unpremultiply a source-pixel mallet matte; retain its soft floor shadow.\"\"\"
    delta = src[0] - plate
    x, y = center
    region = (abs(X-x) < 9) & (Y > y-7) & (Y < y+12)
    # The blue-vs-floor chroma difference yields a subpixel edge matte.
    chroma = ((src[0, :, :, 1]-src[0, :, :, 0]) -
              (plate[:, :, 1]-plate[:, :, 0]))
    alpha = np.clip(chroma/54., 0, 1)
    alpha *= region & (Y < y+3.1)
    alpha[alpha < .06] = 0
    # Preserve the highlights of the originally rendered head and handle.
    fg = np.zeros_like(src[0])
    m = alpha > 0
    fg[m] = (src[0][m] - (1-alpha[m, None])*plate[m]) / alpha[m, None]
    fg = np.clip(fg, 0, 255)
    # Source shadows are extracted independently, never carried as a floor tile.
    shadow = np.zeros_like(src[0])
    residual = delta - alpha[:, :, None]*(fg-plate)
    shadow_region = region & (Y > y+1.3)
    magnitude = np.maximum(-residual[:, :, :2].mean(2)-3., 0)
    weight = np.clip(magnitude/7., 0, 1) * shadow_region
    shadow = np.minimum(residual, 0) * weight[:, :, None]
    # Remove isolated compression ripple at the edge of the extracted footprint.
    taper = np.minimum(1, distance_transform_edt(region)/2)
    shadow *= taper[:, :, None]
    return alpha, fg*alpha[:, :, None], shadow


def warp_layer(layer, origin, target, scale):
    ox, oy = origin
    tx, ty = target
    coords = [(Y-ty)/scale+oy, (X-tx)/scale+ox]
    if layer.ndim == 2:
        return map_coordinates(layer, coords, order=1, mode='constant', cval=0)
    return np.stack([map_coordinates(layer[:, :, c], coords, order=1,
                                    mode='constant', cval=0) for c in range(3)], 2)


def white_layer(src, plate, center, i):
    x, y = center
    region = (abs(X-x) < 10) & (Y > y-7) & (Y < y+8)
    taper = np.minimum(1, distance_transform_edt(region)/2)
    diff = (src[i] - plate) * taper[:, :, None]
    diff[abs(diff).max(2) < 3] = 0
    return diff


def generate():
    src = np.load(ROOT/'source_pixels.npz')['rgb'].astype(np.float64)
    red, blue, white = get_tracks(src)
    tight, wide = object_masks(src, red, blue, white)
    with warnings.catch_warnings():
        warnings.simplefilter('ignore', RuntimeWarning)
        plate = np.nanmedian(np.where(~tight[:, :, :, None], src, np.nan), axis=0)
    plate = restore_holes(plate)

    # A perspective projection of constant plane velocity fits the pre-impact
    # centers.  It predicts the blue mallet's unobstructed flight across the table.
    tt = np.arange(14, dtype=float)
    def projected(p, t):
        return (p[0]+p[1]*t)/(1+p[2]*t)
    params = least_squares(lambda p: projected(p, tt)-blue[:14, 0],
                           [167.5, -2.45, -.0067]).x
    slope, intercept = np.polyfit(blue[:14, 0], blue[:14, 1], 1)
    # The source head-to-head contact with the white mallet occurred at about
    # source time 29.5.  Match that contact distance, without the red mediator.
    contact_x = np.interp(29.6, np.arange(96), red[:, 0])
    contact_t = (params[0]-contact_x)/(contact_x*params[2]-params[1])
    delay = contact_t - 29.6
    alpha, premul, shadow = make_blue_layers(src, plate, blue[0])
    whites = [white_layer(src, plate, white[i], i) for i in range(96)]
    out = []
    for i in range(96):
        # Retain the original scene and its very small encoding/lighting changes.
        erase = np.minimum(1, distance_transform_edt(wide[i])/1.6)
        frame = src[i]*(1-erase[:, :, None]) + plate*erase[:, :, None]

        # Re-time the white mallet only after the newly delayed direct collision.
        wi = max(0., i-delay)
        lo = min(95, int(np.floor(wi)))
        hi = min(95, lo+1)
        mix = wi-lo
        frame += (1-mix)*whites[lo] + mix*whites[hi]

        if i <= 13:
            # Factual blue history before any possible contact is unchanged.
            x, y = blue[i]
            m = (abs(X-x) < 9) & (Y > y-7) & (Y < y+12)
            taper = np.minimum(1, distance_transform_edt(m)/2)
            frame += (src[i]-plate)*taper[:, :, None]
        else:
            if i <= contact_t:
                bx = projected(params, float(i))
                by = intercept+slope*bx
            else:
                # Reuse the source's decaying contact recoil on the now-blue body.
                s = min(95., 29.6 + i-contact_t)
                bx = np.interp(s, np.arange(96), red[:, 0])
                by = intercept+slope*bx
            target = (bx, by)
            # Projective depth enlarges the mallet slightly as it approaches.
            scale = (1-params[2]*0)/(1+params[2]*min(float(i), contact_t))
            scale = 1 + .68*(scale-1)

        out.append(np.clip(np.rint(frame), 0, 255).astype(np.uint8))
    frames = np.stack(out)
    artifact_io.save(frames, ROOT)
    print(f'Saved {frames.shape}, uint8 RGB, 24 fps. Direct impact at frame {contact_t+1:.2f}; white delay {delay:.2f} frames.')
    return frames
