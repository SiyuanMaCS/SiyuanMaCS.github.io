+
+No external scene assets or renderer.  The star's Coulomb deceleration is
+multiplied by five; restitution at the side cushion is unchanged.  The
+soccer ball stays in its initial state because the weakened rebound no
+longer reaches it.
+"""
+from pathlib import Path
+import numpy as np
+import cv2
+from PIL import Image
+from scipy.ndimage import map_coordinates, gaussian_filter, binary_fill_holes
+from artifact_io import save
+
+ROOT = Path(__file__).resolve().parent
+
+
+def read_native():
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    pictures = []
+    for _ in range(22):
+        ok, bgr = cap.read()
+        if not ok:
+            raise RuntimeError('Cannot read the complete initial source segment')
+        pictures.append(cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB))
+    cap.release()
+    return np.stack(pictures)

+def sample(image, x, y):
+    if image.ndim == 2:
+        return map_coordinates(image, [y, x], order=1, mode='nearest')
+    return np.stack([map_coordinates(image[..., k], [y, x], order=1,
+                                    mode='nearest') for k in range(3)], -1)
+
+

+    feather = np.minimum(np.minimum((xx-1075)/12., (yy-240)/12.),

+    poses = np.array([
+        [237.951,66.251,11.516], [223.210,66.084,11.373],
+        [208.614,65.881,11.227], [194.063,65.522,11.059],
+        [179.801,65.449,10.969], [165.682,65.255,10.935],
+        [151.620,65.351,10.909], [137.682,65.201,10.722],
+        [123.915,65.252,10.759], [110.329,65.210,10.723],
+        [96.790,65.587,10.857], [83.224,65.825,10.971]
+    ])
+    q = np.linspace(-1.075, 1.075, 151)
+    u, v = np.meshgrid(q, q)
+    radius_grid = np.sqrt(u*u+v*v)
+    textures = []
+    alphas = []

+        sy = (cy+v*radius)*5

+        back = sample(background, sx, sy)
+        # Extract the physical silhouette by source/background difference,
+        # constrain it to a disk, and fill the star-shaped pale interior.
+        diff = np.linalg.norm(tex-back, axis=2)
+        mask = binary_fill_holes((diff > 16) & (radius_grid < 1.025))
+        mask[radius_grid > 1.025] = False
+        # Guaranteed solid interior; lighting/material never changes here.

+                          (35-sy)/5).clip(0,1)

+            source_phase = 11-(cx-wall)/14

+        alpha_tex = alphas[i0]*(1-blend)+alphas[i1]*blend
+

+        scale = radius/r0
+        bx0=max(0,int((cx-23*scale)*5)-1)
+        bx1=min(1280,int((cx+25*scale)*5)+2)

+        ny=((py/5-cy)/scale-shy[0])/(shy[-1]-shy[0])*(len(shy)-1)
+        factors=sample(ratio,nx,ny)
+        image[by0:by1,bx0:bx1] *= factors
+

+        bx0=max(0,int((cx-1.08*radius)*5)-1)
+        bx1=min(1280,int((cx+1.08*radius)*5)+2)

+        by1=min(720,int((cy+1.08*radius)*5)+2)

+        ny=((py/5-cy)/radius-q[0])/(q[-1]-q[0])*150

+        alpha*=disc
+        local=image[by0:by1,bx0:bx1]
+        image[by0:by1,bx0:bx1]=color*alpha+local*(1-alpha)
+        rendered=np.asarray(Image.fromarray(image.clip(0,255).round().astype(np.uint8))
+                            .resize((256,144),Image.Resampling.LANCZOS))
+        frames.append(rendered)
+    output=np.stack(frames)
+    assert output.shape==(96,144,256,3) and output.dtype==np.uint8
+    assert np.array_equal(output[0],factual[0])
+    save(output,ROOT)
+    print('Saved 96 frames, 256x144 RGB, 24 fps.')
+    print('Cushion contact %.3f s; final center (%.3f, %.3f).'
+          %(hit/24,*centers[-1]))
+    print('First frame preserved; soccer ball and static scenery unaffected.')
+
+
+if __name__ == '__main__':
+    generate()
