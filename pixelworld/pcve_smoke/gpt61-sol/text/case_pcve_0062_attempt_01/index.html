+        masks = [(r-g > 38) & (g-b > 10) & (yy > 64) & (yy < 83),

+    if kind == 0:  # Red can, including its soft leftward contact shadow.
+        return (x >= np.floor(center-14)) & (x <= np.ceil(center+8)) & (y >= 62) & (y <= 84)
+    if kind == 1:  # Striped cup including lid, straw, and contact shadow.
+        return (x >= np.floor(center-12)) & (x <= np.ceil(center+9)) & (y >= 53) & (y <= 84)
+    return (x >= np.floor(center-9)) & (x <= np.ceil(center+9)) & (y >= 61) & (y <= 84)

+    for t in range(len(source)):
+        for kind in range(3):
+            obscured[t] |= boxmask(pos[t, kind], kind)

+    y, x = np.mgrid[:144, :256]

+    else:

+    support = labels == counts.argmax()

+    support = ndi.binary_fill_holes(support)
+    support |= bounds & (y >= 81) & (delta > 6)
+    alpha = support.astype(float)

+    alpha[edge] = edge_alpha[edge]

+        trajectories.append((red,bottle))

        dest[..., c] += residual + alpha*(old_bg-dest[..., c])
    return dest


def generate():
    source = np.load(ROOT/'source_pixels.npz')['rgb']
    pos = locations(source)
    plate, masks = clean_plate(source, pos)
    red_sprites = [sprite(source, plate, t, 0, pos[t,0]) for t in (0,1,2,3)]
    # Frame 24 exposes the complete white neck against an unobstructed wall.
    bottle_moving = sprite(source, plate, 24, 2, pos[24,2])
    bottle_still = sprite(source, plate, 0, 2, pos[0,2])

    # Can centers measured before its first factual impact.  Can and bottle
    # have ~10-pixel footprints.  The cup-free direct impact is at ~0.29s.
    red0 = pos[0,0]
    bottle0 = pos[0,2]
    v0 = float(np.polyfit(np.arange(4), pos[:4,0], 1)[0])
    # Source stopping curves consistently imply 0.23 pixels/frame"'^2.
    deceleration = 0.235
    separation = 10.25
    impact = (bottle0-separation-red0)/v0
    # Equal effective masses, near-inelastic tabletop contact.  The cup
    # was a comparable third mass, so removing it leaves a faster pair.
    pair_speed = v0/2
    frames = []
    trajectories = []
    for t in range(96):
        f = source[t].astype(float).copy()
        # Exact stationary scene from unobscured timeline observations.
        feather = ndi.gaussian_filter(masks[t].astype(float), .55)
        feather[masks[t]] = 1
        f = f*(1-feather[..., None]) + plate*feather[..., None]
        if t <= impact:
            red = red0 + v0*t
            bottle = bottle0
        else:
            tau = min(t-impact, pair_speed/deceleration)
            travel = pair_speed*tau - .5*deceleration*tau*tau
            bottle = bottle0+travel
            red = bottle-separation
        # Original appearance and location are exact for the unaffected
        # launch history, but the can continues through the removed cup.
        if t <= 3:
            sr = red_sprites[t]
            red = pos[t,0]
        else:
            sr = red_sprites[2 if t%8<4 else 3]
        f = composite(f, sr, red-sr['"'center'])
        sb = bottle_still if t <= impact else bottle_moving
        f = composite(f, sb, bottle-sb['center'])
        frames.append(np.clip(np.rint(f), 0, 255).astype(np.uint8))
        trajectories.append((red,bottle))
    result = np.stack(frames)
    assert result.shape == (96,144,256,3)
    artifact_io.save(result, ROOT)
    np.savez_compressed(ROOT/'edit_diagnostics.npz', centers=np.asarray(trajectories), source_centers=pos)
    print('Saved complete RGB video:',result.shape)
    print('Direct impact: frame %.2f (%.3fs); speed %.3f; friction %.3f' % (impact+1,impact/24,v0,deceleration))
    print('Counterfactual can/bottle centers at frames 1, 9, 17, 25, 96:',np.asarray(trajectories)[[0,8,16,24,95]])

if __name__ == '__main__':
    generate()
