+
+The camera and rink are fixed.  Preserve the measured incoming motion.  At
+contact apply the equal-mass normal collision impulse with restitution .95.
+Render the outgoing stones with source-derived, antialiased material sprites.

+def main():

+        # Fill the dark body but not the openings in the raised handles.
+        holes=binary_fill_holes(core)&~core
+        yy,_=np.indices(core.shape)
+        core |= holes & (yy>19)
+        alpha=gaussian_filter(core.astype(float),.65)

+        # red and blue painted circles remain visible underneath the stones.
+        ratio=np.clip(np.mean(tex/(bg+1e-6),axis=2),.6,1)
+        shadow=(1-ratio)*(1-alpha)

+        for w,p in sorted([(0,apos),(1,bpos)],key=lambda z:z[1][1]):

+        output[frame]=np.asarray(image.resize((256,144),Image.Resampling.LANCZOS))

    print('Last centres:',apos,bpos)
