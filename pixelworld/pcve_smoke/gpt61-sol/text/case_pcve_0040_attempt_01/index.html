+"""Counterfactual edit using only the supplied source RGB/video.
+
+The initial contact-phase wheel centers give the source's tangential slowing.
+Coulomb braking is multiplied by seven, with unilateral static contact at zero
+speed. This dissipates the car's initial momentum on the raised surface before
+it can reach the edge. Source-view sprites, not a new drawing, preserve its
+body, glazing, chrome wheels, perspective, illumination, and contact shadow.
+"""
+from pathlib import Path
+import cv2
+import numpy as np
+from PIL import Image
+from scipy import ndimage as ndi
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+W, H, N = 256, 144, 96
+
+
+def largest_component(mask):
+    labels, count = ndi.label(mask)
+    if not count:
+        return mask
+    sizes = np.bincount(labels.ravel())
+    sizes[0] = 0
+    return labels == sizes.argmax()
+
+
+def main():
+    factual = np.load(ROOT / 'source_pixels.npz')['rgb']
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    references = []
+    for _ in range(18):
+        ok, bgr = cap.read()
+        if not ok:
+            raise RuntimeError('Source video ended before the background samples.')
+        references.append(cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB))
+    sh, sw = references[0].shape[:2]
+    sx, sy = sw / W, sh / H
+    assert abs(sx - sy) < 1e-6
+    scale = sx
+
+    # This small patch is visible without the moving car in the majority of
+    # samples. Temporal median therefore reconstructs genuine source texture.
+    bx0, bx1, by0, by1 = 52, 211, 22, 57
+    ix0, ix1 = round(bx0 * scale), round(bx1 * scale)
+    iy0, iy1 = round(by0 * scale), round(by1 * scale)
+    background = np.median(np.stack([
+        f[iy0:iy1, ix0:ix1] for f in references[6:18]
+    ]), axis=0).astype(np.uint8)
+
+    # Screen-space axle midpoints measured numerically from source wheel hubs.
+    axle_x = np.array([67.8235, 79.2050, 90.0735])
+    initial_step = axle_x[1] - axle_x[0]
+    original_decel = initial_step - (axle_x[2] - axle_x[1])
+    v0 = initial_step + 0.5 * original_decel
+    decel = 7.0 * original_decel
+    stop_time = v0 / decel  # expressed in source-frame intervals
+
+    # Isolate the actual source car and its immediate contact shadow. Closing
+    # and hole filling retain dark windows even when they match the background.
+    sprites = []
+    for j in range(3):
+        cx = axle_x[j]
+        x0, x1 = round((cx - 13) * scale), round((cx + 13) * scale)
+        y0, y1 = round(24 * scale), round(37 * scale)
+        rgb = references[j][y0:y1, x0:x1].astype(np.float32)
+        bg = background[y0-iy0:y1-iy0, x0-ix0:x1-ix0].astype(np.float32)
+        contrast = np.abs(rgb - bg).max(axis=2)
+        silhouette = largest_component(ndi.binary_closing(contrast > 10,
+                                      iterations=2))
+        silhouette = ndi.binary_fill_holes(silhouette)
+        silhouette = ndi.binary_dilation(silhouette, iterations=2)
+        # Include the subtle connected penumbra beneath the wheels.
+        shadow = (contrast > 5) & ndi.binary_dilation(silhouette, iterations=3)
+        matte = ndi.binary_fill_holes(ndi.binary_closing(silhouette | shadow,
+                                                       iterations=1))
+        alpha = ndi.gaussian_filter(matte.astype(np.float32), 0.8)
+        sprites.append((rgb, alpha, x0, y0, cx))
+
+    # Measured factual car locations, only used to erase its factual trajectory.
+    factual_x = np.array([
+        67.82, 79.21, 90.07, 100.70, 111.0, 121.1, 130.7, 139.0,
+        145.8, 152.4, 158.2, 163.4, 168.7, 172.8, 176.9, 180.4,
+        183.5, 186.3, 188.5, 190.4, 191.6, 192.8, 193.3, 193.8,
+    ])
+    top = [24, 24, 24, 24, 27, 30, 37]
+    bottom = [37, 37, 37, 38, 41, 46, 56]
+
+    def place(rgb, alpha, x0, y0, delta):
+        # Warp premultiplied RGB and alpha separately for clean subpixel edges.
+        transform = np.float32([[1, 0, x0 + delta * scale], [0, 1, y0]])
+        a = cv2.warpAffine(alpha, transform, (sw, sh),
+                           flags=cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)
+        c = cv2.warpAffine(rgb * alpha[..., None], transform, (sw, sh),
+                           flags=cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)
+        return c, a

+    cap.set(cv2.CAP_PROP_POS_FRAMES, 1)

+            raise RuntimeError(f'Missing source frame {k}')

+        # Small clean-plate replacement removes body AND the original shadow;
+        # all objects outside its footprint retain their own source timeline.
+        old_x = factual_x[min(k, len(factual_x) - 1)]
+        old_top = top[k] if k < 7 else 38
+        old_bottom = bottom[k] if k < 7 else 55
+        x0, x1 = round((old_x - 14) * scale), round((old_x + 14) * scale)
+        y0, y1 = round(old_top * scale), round(old_bottom * scale)
+        scene[y0:y1, x0:x1] = background[y0-iy0:y1-iy0,
+                                                       x0-ix0:x1-ix0]
+
+        # Sevenfold Coulomb force. Integrating only until the stopping event
+        # prevents artificial reversal: the settled vehicle remains at rest.
+        tau = min(float(k), stop_time)
+        new_x = axle_x[0] + v0 * tau - 0.5 * decel * tau * tau
+        # Viewpoint changes are sampled by distance, not slowed source time.
+        # Thus the upper-track sprite never acquires the factual falling pose.
+        u = np.interp(new_x, axle_x, np.arange(3, dtype=float))
+        left = min(int(np.floor(u)), 1)
+        mix = float(u - left)
+        colors, alpha = [], []
+        for j in (left, left + 1):
+            rgb, a, px, py, reference_x = sprites[j]
+            c, m = place(rgb, a, px, py, new_x - reference_x)
+            colors.append(c)
+            alpha.append(m)
+        foreground = (1.0-mix) * colors[0] + mix * colors[1]
+        opacity = (1.0-mix) * alpha[0] + mix * alpha[1]
+        composite = scene * (1.0-opacity[..., None]) + foreground
+        full_rgb = np.clip(np.rint(composite), 0, 255).astype(np.uint8)
+        frames[k] = np.array(Image.fromarray(full_rgb).resize(
+            (W, H), Image.Resampling.LANCZOS))
+    cap.release()
+    assert frames.shape == (96, 144, 256, 3) and frames.dtype == np.uint8
+    artifact_io.save(frames, ROOT)
+    print(f'Saved {N} frames at 24 fps; sevenfold contact deceleration.')
+    print(f'Car stopping time: {stop_time / 24:.4f}s; '
+          f'initial/final axle x: {axle_x[0]:.3f}/{new_x:.3f} pixels.')
+
+
+if __name__ == '__main__':
+    main()

    assert abs(sx - sy) < 1e-6

    # This small patch is visible without the moving car in the majority of
    # samples. Temporal median therefore reconstructs genuine source texture.
    bx0, bx1, by0, by1 = 52, 211, 22, 57
    ix0, ix1 = round(bx0 * scale), round(bx1 * scale)
    iy0, iy1 = round(by0 * scale), round(by1 * scale)
    background = np.median(np.stack([
        f[iy0:iy1, ix0:ix1] for f in references[6:18]
    ]), axis=0).astype(np.uint8)

    # Screen-space axle midpoints measured numerically from source wheel hubs.
    axle_x = np.array([67.8235, 79.2050, 90.0735])
    initial_step = axle_x[1] - axle_x[0]
    original_decel = initial_step - (axle_x[2] - axle_x[1])
    v0 = initial_step + 0.5 * original_decel
    decel = 7.0 * original_decel
    stop_time = v0 / decel  # expressed in source-frame intervals

    # Isolate the actual source car and its immediate contact shadow. Closing
    # and hole filling retain dark windows even when they match the background.
    sprites = []
    for j in range(3):
        cx = axle_x[j]
        x0, x1 = round((cx - 13) * scale), round((cx + 13) * scale)
        y0, y1 = round(24 * scale), round(37 * scale)
        rgb = references[j][y0:y1, x0:x1].astype(np.float32)
        bg = background[y0-iy0:y1-iy0, x0-ix0:x1-ix0].astype(np.float32)
        contrast = np.abs(rgb - bg).max(axis=2)
        silhouette = largest_component(ndi.binary_closing(contrast > 10,
                                      iterations=2))
        silhouette = ndi.binary_fill_holes(silhouette)
        silhouette = ndi.binary_dilation(silhouette, iterations=2)
        # Include the subtle connected penumbra beneath the wheels.
        shadow = (contrast > 5) & ndi.binary_dilation(silhouette, iterations=3)
        matte = ndi.binary_fill_holes(ndi.binary_closing(silhouette | shadow,
                                                       iterations=1))
        alpha = ndi.gaussian_filter(matte.astype(np.float32), 0.8)
        sprites.append((rgb, alpha, x0, y0, cx))

    # Measured factual car locations, only used to erase its factual trajectory.
    factual_x = np.array([
        67.82, 79.21, 90.07, 100.70, 111.0, 121.1, 130.7, 139.0,
        145.8, 152.4, 158.2, 163.4, 168.7, 172.8, 176.9, 180.4,
        183.5, 186.3, 188.5, 190.4, 191.6, 192.8, 193.3, 193.8,
    ])
    top = [24, 24, 24, 24, 27, 30, 37]
    bottom = [37, 37, 37, 38, 41, 46, 56]

    def place(rgb, alpha, x0, y0, delta):
        # Warp premultiplied RGB and alpha separately for clean subpixel edges.
        transform = np.float32([[1, 0, x0 + delta * scale], [0, 1, y0]])
        a = cv2.warpAffine(alpha, transform, (sw, sh),
                           flags=cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)
        c = cv2.warpAffine(rgb * alpha[..., None], transform, (sw, sh),
                           flags=cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)
        return c, a

    frames = np.empty_like(factual)
    frames[0] = factual[0]  # physical intervention changes acceleration, not pose
    cap.set(cv2.CAP_PROP_POS_FRAMES, 1)
    for k in range(1, N):
        ok, bgr = cap.read()
        if not ok:
            raise RuntimeError(f'Missing source frame {k}')
        scene = cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB).astype(np.float32)
        # Small clean-plate replacement removes body AND the original shadow;
        # all objects outside its footprint retain their own source timeline.
        old_x = factual_x[min(k, len(factual_x) - 1)]
        old_top = top[k] if k < 7 else 38
        old_bottom = bottom[k] if k < 7 else 55
        x0, x1 = round((old_x - 14) * scale), round((old_x + 14) * scale)
        y0, y1 = round(old_top * scale), round(old_bottom * scale)
        scene[y0:y1, x0:x1] = background[y0-iy0:y1-iy0,
                                                       x0-ix0:x1-ix0]

        # Sevenfold Coulomb force. Integrating only until the stopping event
        # prevents artificial reversal: the settled vehicle remains at rest.
        tau = min(float(k), stop_time)
        new_x = axle_x[0] + v0 * tau - 0.5 * decel * tau * tau
        # Viewpoint changes are sampled by distance, not slowed source time.
        # Thus the upper-track sprite never acquires the factual falling pose.
        u = np.interp(new_x, axle_x, np.arange(3, dtype=float))
        left = min(int(np.floor(u)), 1)
        mix = float(u - left)
        colors, alpha = [], []
        for j in (left, left + 1):
            rgb, a, px, py, reference_x = sprites[j]
            c, m = place(rgb, a, px, py, new_x - reference_x)
            colors.append(c)
            alpha.append(m)
        foreground = (1.0-mix) * colors[0] + mix * colors[1]
        opacity = (1.0-mix) * alpha[0] + mix * alpha[1]
        composite = scene * (1.0-opacity[..., None]) + foreground
        full_rgb = np.clip(np.rint(composite), 0, 255).astype(np.uint8)
        frames[k] = np.array(Image.fromarray(full_rgb).resize(
            (W, H), Image.Resampling.LANCZOS))
    cap.release()
    assert frames.shape == (96, 144, 256, 3) and frames.dtype == np.uint8
    artifact_io.save(frames, ROOT)
    print(f'Saved {N} frames at 24 fps; sevenfold contact deceleration.')
    print(f'Car stopping time: {stop_time / 24:.4f}s; '
          f'initial/final axle x: {axle_x[0]:.3f}/{new_x:.3f} pixels.')


if __name__ == '__main__':
    main()
