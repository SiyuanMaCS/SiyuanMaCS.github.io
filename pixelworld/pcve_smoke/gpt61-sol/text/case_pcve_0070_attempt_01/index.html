+"""Source-pixel counterfactual: remove domino 3 and interrupt propagation.
+Only local source RGB, analytic cuboid geometry, and interpolated floor lighting.
+"""
+from pathlib import Path
+import numpy as np
+import cv2
+from scipy.interpolate import RBFInterpolator
+from scipy.ndimage import gaussian_filter
+from PIL import Image, ImageDraw
+import artifact_io
+
+ROOT=Path(__file__).resolve().parent
+S=np.load(ROOT/'source_pixels.npz')['rgb']
+H,W=144,256
+Y,X=np.mgrid[:H,:W]
+
+
+def smoothstep(x):
+    x=np.clip(x,0,1);return x*x*(3-2*x)
+
+
+def floor_surface():
+    # Recover diffuse green floor from unobstructed regions on this very timeline.
+    points=[];values=[]
+    for y in range(32,H,3):
+        for x in range(0,W,3):
+            if y<=65:
+                v=np.median(S[65:96,y,x],axis=0)
+            elif x<=49 or x>=203 or (151<=x<=169 and y<=132) or (82<=x<=94 and y<85) or (119<=x<=134 and y<83):
+                v=S[0,y,x]
+            else: continue
+            points.append((x/80,y/80));values.append(v)
+    rb=RBFInterpolator(np.array(points),np.array(values),neighbors=80,smoothing=.02,kernel='thin_plate_spline',degree=1)
+    pred=rb(np.c_[X.ravel()/80,Y.ravel()/80]).reshape(H,W,3)
+    # The original floor's minute color fluctuations are retained, not a flat fill.
+    detail=S[95].astype(float)-gaussian_filter(S[95].astype(float),(1.8,1.8,0))
+    detail=np.clip(detail,-1.1,1.1)
+    pred+=detail*.5
+    return np.clip(pred,0,255)
+
+FLOOR=floor_surface()
+
+
+def poly_mask(points, scale=3):
+    im=Image.new('L',(W*scale,H*scale))
+    ImageDraw.Draw(im).polygon([(round(x*scale),round(y*scale)) for x,y in points],fill=255)
+    return np.asarray(im.resize((W,H),Image.Resampling.LANCZOS),float)/255
+
+
+def erase_third(src):
+    a=src.astype(float).copy()
+    # Head and upright body: replacement from later exposed floor, preserving texture.
+    m=poly_mask([(139.5,32.8),(154.5,32.8),(154,101.8),(152,107),(138.5,107)])
+    a=a*(1-m[...,None])+FLOOR*m[...,None]
+    # Remove the third block's diffuse/contact shadow, including its penumbra.
+    # The shadow points towards the foreground and left, away from the window.
+    shadow=smoothstep((Y-82)/13)*np.exp(-.5*((X-(146-.38*(Y-101)))/(5.3+.24*np.maximum(Y-101,0)))**2)
+    shadow*=np.exp(-.5*((Y-105)/24)**2)
+    shadow*=1-smoothstep((X-153)/8)
+    # As floor lighting gets exposed, use the recovered local illumination. Do not
+    # overwrite the preceding domino, or the distant fourth domino.
+    shadow*=smoothstep((X-120)/7)
+    a=a*(1-shadow[...,None])+FLOOR*shadow[...,None]
+    return a
+
+# Static scene plate, with only dominos 1 and 2 removed as well. The fourth block,
+# its shadow, the room and the window stay exactly where the source placed them.
+PLATE=S[0].astype(float).copy()
+blend=smoothstep((X-49)/7)*(1-smoothstep((X-169)/9))*smoothstep((Y-30.3)/2)
+PLATE=PLATE*(1-blend[...,None])+FLOOR*blend[...,None]
+# Keep the original fourth domino's small left-side penumbra at the boundary.
+
+# Source face textures: use actual pixels for the dark fronts and gray sides.
+def texture(quad,size=(96,128)):
+    tw,th=size
+    M=cv2.getPerspectiveTransform(np.array(quad,np.float32),np.array([[0,0],[tw-1,0],[tw-1,th-1],[0,th-1]],np.float32))
+    return cv2.warpPerspective(S[0],M,(tw,th),flags=cv2.INTER_LINEAR).astype(float)
+
+TEX=[]
+TEX.append(texture([(67.5,40),(77.7,42.4),(69.0,98.5),(59.8,97.3)]))
+TEX.append(texture([(102.7,40),(110.1,40),(110.7,99.5),(104,99.5)]))
+SIDE=[]
+SIDE.append(texture([(77.8,43),(83.3,38),(74.7,94),(69.7,98.5)]))
+SIDE.append(texture([(111.7,40),(114.1,35.8),(114,93),(112,96)]))
+# Smooth tiny specular patches to prevent stretching source compression detail.
+SIDE=[gaussian_filter(q,(.8,1.5,0)) for q in SIDE]
+
+# An oblique, almost orthographic camera is recovered from source vertical and
+# floor edges. Rotation raises the trailing foot and pivots at the leading foot.
+V=np.array([-1.0,-62.0]); D=np.array([64.0,1.2])
+
+def vertices(i,theta):
+    pivot=np.array([69.5,100.0]) if i==0 else np.array([112.2,101.7])
+    thickness=.167 if i==0 else .139
+    depth=np.array([5.0,-6.0]) if i==0 else np.array([3.0,-5.2])
+    s,c=np.sin(theta),np.cos(theta)
+    lo=pivot-thickness*c*D+thickness*s*V
+    hi=lo+s*D+c*V
+    upper=pivot+s*D+c*V
+    return np.array([lo,pivot,upper,hi,lo+depth,pivot+depth,upper+depth,hi+depth])

+
+def shade_shadow(canvas,verts,theta,i):
+    # Sample a finite area-light shadow of the block on its green floor.
+    # Projection of height from the bright window points down-left.
+    soft=np.zeros((H,W),float)
+    for offx,offy in [(-5,-5),(0,-5),(5,-5),(-5,0),(0,0),(5,0),(-5,5),(0,5),(5,5)]:
+        points=[]
+        for k in [0,1,2,3,4,5,6,7]:
+            # Vertical coordinates of the rotated cuboid in units of its height.
+            z=[.167*np.sin(theta),0,np.cos(theta),np.cos(theta)+.167*np.sin(theta)][k%4]
+            # Put vertex on the ground, then project its height away from light.
+            p=verts[k]-z*V+np.array([-15+offx,26+offy])*z
+            points.append(p)
+        hull=cv2.convexHull(np.array(points,np.float32)).reshape(-1,2)
+        soft+=poly_mask(hull)/9
+    soft=gaussian_filter(soft,2.7)
+    contact=np.exp(-.5*((X-verts[1,0])/4.2)**2-.5*((Y-verts[1,1]-2.0)/2.8)**2)
+    sh=np.clip(.27*soft+.30*contact,0,.51)
+    sh*=smoothstep((Y-31)/3)
+    return canvas*(1-sh[...,None])
+
+
+def warp_face(canvas,tex,quad,mult=1.0):
+    h,w=tex.shape[:2]
+    q=np.array(quad,np.float32)
+    if abs(cv2.contourArea(q))<.06:return canvas
+    trans=cv2.getPerspectiveTransform(np.array([[0,0],[w-1,0],[w-1,h-1],[0,h-1]],np.float32),q)
+    # Supersampled texture warping/coverage for stable slender edges.
+    sc=3
+    big=cv2.warpPerspective(np.clip(tex*mult,0,255).astype(np.uint8),np.diag([sc,sc,1.0])@trans,(W*sc,H*sc),flags=cv2.INTER_LINEAR,borderMode=cv2.BORDER_REPLICATE)
+    col=cv2.resize(big,(W,H),interpolation=cv2.INTER_AREA).astype(float)
+    m=poly_mask(quad,sc)
+    return canvas*(1-m[...,None])+col*m[...,None]
+
+
+def draw_domino(a,i,theta):
+    v=vertices(i,theta)
+    # Gray side becomes the upper face during a fall. Its newly visible surface
+    # uses the same source material, with analytically changing diffuse lighting.
+    side=SIDE[i]
+    front=TEX[i]
+    grey=.78-.10*np.cos(theta)
+    # Far broad end and the very narrow top cap, behind front-facing surfaces.
+    cap=np.empty((32,32,3),float);cap[:]=[121,118,111]
+    a=warp_face(a,cap,v[[3,2,6,7]],.93)
+    a=warp_face(a,front,v[[3,2,1,0]],.94)
+    a=warp_face(a,side,v[[3,7,4,0]],grey)
+    # Fine bevel along the upper edge, consistent with the source's soft highlights.
+    return a
+
+
+def generate():
+    frames=[]
+    # Contact between source domino 2 and missing domino 3 starts around frame 31.
+    # Until that instant, preserve the unaffected original motion pixel-for-pixel.
+    for t in range(96):
+        if t<=29:
+            a=erase_third(S[t])
+        else:
+            dt=(t-29)/24
+            # Free fall under gravity after losing the third block's support.
+            # The second lies flat; first lands on its thickness and stays inclined.
+            th2=min(np.pi/2,.435+1.83*dt+2.05*dt*dt)
+            th1=min(1.371,.950+1.02*dt+.9*dt*dt)
+            # Very small damped rebound, no strike of the isolated fourth block.
+            if t>=40:
+                th2=np.pi/2-.021*np.exp(-(t-40)/3.3)*max(0,np.sin((t-40)*1.2))
+            if t>=38:
+                th1=1.371-.013*np.exp(-(t-38)/4.0)*np.sin((t-38)*.9)
+            v1=vertices(0,th1);v2=vertices(1,th2)
+            a=PLATE.copy()
+            a=shade_shadow(a,v2,th2,1)
+            a=shade_shadow(a,v1,th1,0)
+            a=draw_domino(a,1,th2)
+            a=draw_domino(a,0,th1)
+            # Continuous hand-off between measured source pixels and analytic
+            # free motion; early difference is localized to the two moving blocks.
+            if t<34:
+                original=erase_third(S[min(t,29)])
+                weight=smoothstep((t-29)/5)
+                # Mesh rendering is confined to the edited foreground/floor.
+                a=original*(1-weight)+a*weight
+        frames.append(np.clip(np.rint(a),0,255).astype(np.uint8))
+    return np.stack(frames)
+
+if __name__=='__main__':
+    frames=generate()
+    artifact_io.save(frames,ROOT)
+    print('Saved',frames.shape,'RGB frames at 24 fps')

def floor_surface():
