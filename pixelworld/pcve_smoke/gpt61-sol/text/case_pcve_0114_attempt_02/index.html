+"""Edit only the initial linear velocity of the source tennis ball.
+
+The camera projection and gravity are fitted from RGB-observed free flight;
+contact damping and rolling friction are fitted from the source ground motion.
+Sprites, shading, seams, shadows, and background all come from source pixels.
+"""
+from pathlib import Path
+import json
+import cv2
+import numpy as np
+from PIL import Image
+from scipy.ndimage import label
+from scipy.optimize import least_squares, curve_fit, brentq
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+
+def largest_component(mask):
+    labels, n = label(mask)
+    counts = np.bincount(labels.ravel())
+    counts[0] = 0
+    return labels == np.argmax(counts)
+
+
+def decode_source():
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    images = []
+    while True:
+        ok, bgr = cap.read()
+        if not ok:
+            break
+        images.append(cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB))
+    cap.release()
+    return np.stack(images)
+
+
+def analyze(source, background, small):
+    centers, radii = [], []
+    for i in range(len(source)):
+        r, g, b = small[i].astype(float).transpose(2, 0, 1)
+        m = largest_component((r > 100) & (g > 95) & (b < .65*g) & (g > .8*r))
+        y, x = np.where(m)
+        px, py = np.round((np.array([x.mean(), y.mean()]) + .4)*5).astype(int)
+        patch = source[i, py-20:py+21, px-20:px+21].astype(float)
+        bg = background[py-20:py+21, px-20:px+21].astype(float)
+        delta = np.max(abs(patch-bg), axis=-1)
+        m = largest_component(delta > 25)
+        ys, xs = np.where(m)
+        c = np.array([xs.mean()-20+px, ys.mean()-20+py])
+        # Robust circular silhouette estimate avoids including contact shadow.
+        theta = np.linspace(0, 2*np.pi, 80, endpoint=False)
+        rr = np.arange(5, 15, .1)
+        mx = 20+c[0]-px+np.cos(theta)[:, None]*rr
+        my = 20+c[1]-py+np.sin(theta)[:, None]*rr
+        ray_values = cv2.remap(delta.astype(np.float32), mx.astype(np.float32),
+                               my.astype(np.float32), cv2.INTER_LINEAR)
+        points = []
+        for th, values in zip(theta, ray_values):
+            indices = np.flatnonzero(values > 25)
+            if len(indices):
+                radius = rr[indices[-1]]
+                points.append([np.cos(th)*radius, np.sin(th)*radius])
+        points = np.array(points)
+        fit = least_squares(lambda p: np.linalg.norm(points-p[:2], axis=1)-p[2],
+                            [0, 0, 10.6], loss='soft_l1', f_scale=.3).x
+        centers.append(c+fit[:2])
+        radii.append(fit[2])
+    return np.array(centers), np.array(radii)
+
+
+def rolling(t, x0, velocity, k, friction):
+    t = np.asarray(t)
+    stop = np.log1p(velocity*k/friction)/k
+    t = np.minimum(np.maximum(t, 0), stop)
+    return x0+(velocity+friction/k)*(-np.expm1(-k*t))/k-friction*t/k
+
+
+def make_sprite(image, background, center, radius):
+    # Subpixel-centered source crop, with premultiplied matting to keep source
+    # background colors from being pasted onto the counterfactual trajectory.
+    axis = np.arange(-23, 24, dtype=np.float32)
+    xx, yy = np.meshgrid(axis, axis)
+    mapx = (xx+center[0]).astype(np.float32)
+    mapy = (yy+center[1]).astype(np.float32)
+    rgb = cv2.remap(image.astype(np.float32), mapx, mapy, cv2.INTER_LINEAR)
+    bg = cv2.remap(background.astype(np.float32), mapx, mapy, cv2.INTER_LINEAR)
+    distance = np.hypot(xx, yy)
+    alpha = np.clip((radius+.95-distance)/1.9, 0, 1).astype(np.float32)
+    premult = rgb-bg*(1-alpha[..., None])
+    premult = np.clip(premult, 0, 255*alpha[..., None])
+    # Contact shadow is a separate multiplicative source-derived layer.
+    shadow = np.clip(1-rgb.mean(-1)/(bg.mean(-1)+1e-5), 0, .7)
+    elliptic = np.sqrt(((xx-1)/16)**2+((yy-radius-1.6)/5.5)**2)
+    shadow *= np.clip((1.1-elliptic)/.3, 0, 1)
+    shadow *= np.clip((distance-radius+.8)/1.6, 0, 1)
+    return premult, alpha, shadow.astype(np.float32)
+
+
+def composite(canvas, sprite, center, scale, shadow_strength):
+    premult, alpha, shadow = sprite
+    cx, cy = center
+    ix, iy = int(round(cx)), int(round(cy))
+    extent = 31
+    x0, x1 = max(0, ix-extent), min(canvas.shape[1], ix+extent+1)
+    y0, y1 = max(0, iy-extent), min(canvas.shape[0], iy+extent+1)
+    yy, xx = np.mgrid[y0:y1, x0:x1].astype(np.float32)
+    mx = ((xx-cx)/scale+23).astype(np.float32)
+    my = ((yy-cy)/scale+23).astype(np.float32)
+    a = cv2.remap(alpha, mx, my, cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)
+    p = cv2.remap(premult, mx, my, cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)
+    sh = cv2.remap(shadow, mx, my, cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)
+    area = canvas[y0:y1, x0:x1]
+    area *= (1-shadow_strength*sh[..., None])
+    area[:] = area*(1-a[..., None])+p
+
+
+def main():
+    small = np.load(ROOT / 'source_pixels.npz')['rgb']
+    source = decode_source()
+    assert len(source) == 96
+    background = np.median(source[[0, 8, 16, 32, 56, 70]], axis=0).astype(np.uint8)
+    centers, radii = analyze(source, background, small)
+    measured = centers/5-.4
+    origin = measured[0]
+    tt = np.arange(22, dtype=float)
+
+    # A perspective camera maps ballistic motion to ratios of quadratics.
+    # Numerator/denominator linear coefficients are initial-velocity terms;
+    # their quadratic coefficients are gravity, not slowed-down time.
+    def flight(p, t, speed=1):
+        vx, ax, vy, ay, dz, az = p
+        t = np.asarray(t)
+        den = 1+speed*dz*t+az*t*t
+        return np.stack([(origin[0]+speed*vx*t+ax*t*t)/den,
+                         (origin[1]+speed*vy*t+ay*t*t)/den], axis=-1)
+
+    fit = least_squares(lambda p: (flight(p, tt)-measured[:22]).ravel(),
+                        [4.30, .015, -4.72, .22, -.0019, .000128]).x
+    floor_y = float(np.median(measured[30:, 1]))
+    contact = brentq(lambda t: flight(fit, t, .6)[1]-floor_y, 15, 35)
+    source_contact = brentq(lambda t: flight(fit, t)[1]-floor_y, 15, 35)
+    impact_x = flight(fit, contact, .6)[0]
+    deriv = lambda t, s: (flight(fit, t+.001, s)-flight(fit, t-.001, s))/.002
+
+    rt = np.arange(50, dtype=float)
+    rp, _ = curve_fit(rolling, rt, measured[28:78, 0],
+                      p0=[153, 2.4, .017, .027],
+                      bounds=([152, 1, .0001, .001], [154, 4, .2, .1]))
+    source_x0, source_v, drag, friction = rp
+    # Preserve the source impact's horizontal loss, rather than slowing gravity.
+    velocity = source_v * deriv(contact, .6)[0]/deriv(source_contact, 1)[0]
+    stop_time = np.log1p(velocity*drag/friction)/drag
+    source_distances = rolling(np.arange(70, dtype=float), *rp)-source_x0
+    sprites = [make_sprite(source[i], background, centers[i], radii[i])
+               for i in range(96)]
+    frames = np.empty_like(small)
+    motion = []
+    for i in range(96):
+        if i < contact:
+            position = flight(fit, float(i), .6)
+            # Same elapsed angular motion in free flight; only linear speed changed.
+            sprite_index = min(i, 21)
+            shadow_strength = 0
+            denom = 1+.6*fit[4]*i+fit[5]*i*i
+            radius = radii[0]/denom
+        else:
+            x = float(rolling(i-contact, impact_x, velocity, drag, friction))
+            position = np.array([x, floor_y])
+            distance = x-impact_x
+            # Source rolling orientation selected by traveled distance, not time.
+            phase = np.interp(distance, source_distances, np.arange(70))
+            sprite_index = min(95, 28+int(round(phase)))
+            shadow_strength = min(1, max(0, (i-contact)*2))
+            denom = 1+.6*fit[4]*contact+fit[5]*contact*contact
+            radius = radii[0]/denom
+        motion.append(position.tolist())
+        if i == 0:
+            frames[i] = small[i]
+            continue
+        canvas = source[i].astype(np.float32)
+        # Remove the factual ball and its contact shadow, without retiming the
+        # rest of the video. The large feather also cleans compression ringing.
+        cx, cy = centers[i]
+        ix, iy = np.round([cx, cy]).astype(int)
+        x0, x1 = max(0, ix-42), min(1280, ix+43)
+        y0, y1 = max(0, iy-42), min(720, iy+43)
+        yy, xx = np.mgrid[y0:y1, x0:x1]
+        removal = np.clip((40-np.hypot(xx-cx, yy-cy))/10, 0, 1)[..., None]
+        canvas[y0:y1, x0:x1] = (canvas[y0:y1, x0:x1]*(1-removal)
+                                +background[y0:y1, x0:x1]*removal)
+        center_hi = (position+.4)*5
+        scale = radius/radii[sprite_index]
+        composite(canvas, sprites[sprite_index], center_hi, scale, shadow_strength)
+        image = Image.fromarray(np.clip(canvas, 0, 255).round().astype(np.uint8))
+        frames[i] = np.array(image.resize((256, 144), Image.Resampling.LANCZOS))
+
+    artifact_io.save(frames, ROOT)
+    metadata = {
+        'initial_speed_multiplier': .6,
+        'flight_coefficients': fit.tolist(),
+        'initial_center': origin.tolist(),
+        'source_landing_frame_zero_based': float(source_contact),
+        'edited_landing_frame_zero_based': float(contact),
+        'edited_stop_frame_zero_based': float(contact+stop_time),
+        'rolling_drag': float(drag), 'rolling_friction': float(friction),
+        'edited_positions': motion,
+    }
+    (ROOT / 'motion_analysis.json').write_text(json.dumps(metadata, indent=2))
+    print('Saved 96 RGB frames at 256x144, 24 fps.')
+    print(f'Landing: frame {contact:.2f}, x={impact_x:.2f}; '
+          f'stop: frame {contact+stop_time:.2f}, x={motion[-1][0]:.2f}')
+
+
+if __name__ == '__main__':
+    main()

\"\"\"Edit only the initial linear velocity of the source tennis ball.

The camera projection and gravity are fitted from RGB-observed free flight;
contact damping and rolling friction are fitted from the source ground motion.
Sprites, shading, seams, shadows, and background all come from source pixels.

from pathlib import Path
import json
import cv2
import numpy as np
from PIL import Image
from scipy.ndimage import label
from scipy.optimize import least_squares, curve_fit, brentq
import artifact_io

ROOT = Path(__file__).resolve().parent


def largest_component(mask):
    labels, n = label(mask)
    counts = np.bincount(labels.ravel())
    counts[0] = 0
    return labels == np.argmax(counts)


def decode_source():
    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
    images = []
    while True:
        ok, bgr = cap.read()
        if not ok:
            break
        images.append(cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB))
    cap.release()
    return np.stack(images)


def analyze(source, background, small):
    centers, radii = [], []
    for i in range(len(source)):
        r, g, b = small[i].astype(float).transpose(2, 0, 1)
        m = largest_component((r > 100) & (g > 95) & (b < .65*g) & (g > .8*r))
        y, x = np.where(m)
        px, py = np.round((np.array([x.mean(), y.mean()]) + .4)*5).astype(int)
        patch = source[i, py-20:py+21, px-20:px+21].astype(float)
        bg = background[py-20:py+21, px-20:px+21].astype(float)
        delta = np.max(abs(patch-bg), axis=-1)
        m = largest_component(delta > 25)
        ys, xs = np.where(m)
        c = np.array([xs.mean()-20+px, ys.mean()-20+py])
        # Robust circular silhouette estimate avoids including contact shadow.
        theta = np.linspace(0, 2*np.pi, 80, endpoint=False)
        rr = np.arange(5, 15, .1)
        mx = 20+c[0]-px+np.cos(theta)[:, None]*rr
        my = 20+c[1]-py+np.sin(theta)[:, None]*rr
        ray_values = cv2.remap(delta.astype(np.float32), mx.astype(np.float32),
                               my.astype(np.float32), cv2.INTER_LINEAR)
        points = []
        for th, values in zip(theta, ray_values):
            indices = np.flatnonzero(values > 25)
            if len(indices):
                radius = rr[indices[-1]]
                points.append([np.cos(th)*radius, np.sin(th)*radius])
        points = np.array(points)
        fit = least_squares(lambda p: np.linalg.norm(points-p[:2], axis=1)-p[2],
                            [0, 0, 10.6], loss='soft_l1', f_scale=.3).x
        centers.append(c+fit[:2])
        radii.append(fit[2])
    return np.array(centers), np.array(radii)


def rolling(t, x0, velocity, k, friction):
    t = np.asarray(t)
    stop = np.log1p(velocity*k/friction)/k
    t = np.minimum(np.maximum(t, 0), stop)
    return x0+(velocity+friction/k)*(-np.expm1(-k*t))/k-friction*t/k


def make_sprite(image, background, center, radius):
    # Subpixel-centered source crop, with premultiplied matting to keep source
    # background colors from being pasted onto the counterfactual trajectory.
    axis = np.arange(-23, 24, dtype=np.float32)
    xx, yy = np.meshgrid(axis, axis)
    mapx, mapy = xx+center[0], yy+center[1]
    rgb = cv2.remap(image.astype(np.float32), mapx, mapy, cv2.INTER_LINEAR)
    bg = cv2.remap(background.astype(np.float32), mapx, mapy, cv2.INTER_LINEAR)
    distance = np.hypot(xx, yy)
    alpha = np.clip((radius+.95-distance)/1.9, 0, 1).astype(np.float32)
    premult = rgb-bg*(1-alpha[..., None])
    premult = np.clip(premult, 0, 255*alpha[..., None])
    # Contact shadow is a separate multiplicative source-derived layer.
    shadow = np.clip(1-rgb.mean(-1)/(bg.mean(-1)+1e-5), 0, .7)
    elliptic = np.sqrt(((xx-1)/16)**2+((yy-radius-1.6)/5.5)**2)
    shadow *= np.clip((1.1-elliptic)/.3, 0, 1)
    shadow *= np.clip((distance-radius+.8)/1.6, 0, 1)
    return premult, alpha, shadow.astype(np.float32)


def composite(canvas, sprite, center, scale, shadow_strength):
    premult, alpha, shadow = sprite
    cx, cy = center
    ix, iy = int(round(cx)), int(round(cy))
    extent = 31
    x0, x1 = max(0, ix-extent), min(canvas.shape[1], ix+extent+1)
    y0, y1 = max(0, iy-extent), min(canvas.shape[0], iy+extent+1)
    yy, xx = np.mgrid[y0:y1, x0:x1].astype(np.float32)
    mx, my = (xx-cx)/scale+23, (yy-cy)/scale+23
    a = cv2.remap(alpha, mx, my, cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)
    p = cv2.remap(premult, mx, my, cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)
    sh = cv2.remap(shadow, mx, my, cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)
    area = canvas[y0:y1, x0:x1]
    area *= (1-shadow_strength*sh[..., None])
    area[:] = area*(1-a[..., None])+p


def main():
    small = np.load(ROOT / 'source_pixels.npz')['rgb']
    source = decode_source()
    assert len(source) == 96
    background = np.median(source[[0, 8, 16, 32, 56, 70]], axis=0).astype(np.uint8)
    centers, radii = analyze(source, background, small)
    measured = centers/5-.4
    origin = measured[0]
    tt = np.arange(22, dtype=float)

    # A perspective camera maps ballistic motion to ratios of quadratics.
    # Numerator/denominator linear coefficients are initial-velocity terms;
    # their quadratic coefficients are gravity, not slowed-down time.
    def flight(p, t, speed=1):
        vx, ax, vy, ay, dz, az = p
        t = np.asarray(t)
        den = 1+speed*dz*t+az*t*t
        return np.stack([(origin[0]+speed*vx*t+ax*t*t)/den,
                         (origin[1]+speed*vy*t+ay*t*t)/den], axis=-1)

    fit = least_squares(lambda p: (flight(p, tt)-measured[:22]).ravel(),
                        [4.30, .015, -4.72, .22, -.0019, .000128]).x
    floor_y = float(np.median(measured[30:, 1]))
    contact = brentq(lambda t: flight(fit, t, .6)[1]-floor_y, 15, 35)
    source_contact = brentq(lambda t: flight(fit, t)[1]-floor_y, 15, 35)
    impact_x = flight(fit, contact, .6)[0]
    deriv = lambda t, s: (flight(fit, t+.001, s)-flight(fit, t-.001, s))/.002

    rt = np.arange(50, dtype=float)
    rp, _ = curve_fit(rolling, rt, measured[28:78, 0],
                      p0=[153, 2.4, .017, .027],
                      bounds=([152, 1, .0001, .001], [154, 4, .2, .1]))
    source_x0, source_v, drag, friction = rp
    # Preserve the source impact's horizontal loss, rather than slowing gravity.
    velocity = source_v * deriv(contact, .6)[0]/deriv(source_contact, 1)[0]
    stop_time = np.log1p(velocity*drag/friction)/drag
    source_distances = rolling(np.arange(70, dtype=float), *rp)-source_x0
    sprites = [make_sprite(source[i], background, centers[i], radii[i])
               for i in range(96)]
    frames = np.empty_like(small)
    motion = []
    for i in range(96):
        if i < contact:
            position = flight(fit, float(i), .6)
            # Same elapsed angular motion in free flight; only linear speed changed.
            sprite_index = min(i, 21)
            shadow_strength = 0
            denom = 1+.6*fit[4]*i+fit[5]*i*i
            radius = radii[0]/denom
        else:
            x = float(rolling(i-contact, impact_x, velocity, drag, friction))
            position = np.array([x, floor_y])
            distance = x-impact_x
            # Source rolling orientation selected by traveled distance, not time.
            phase = np.interp(distance, source_distances, np.arange(70))
            sprite_index = min(95, 28+int(round(phase)))
            shadow_strength = min(1, max(0, (i-contact)*2))
            denom = 1+.6*fit[4]*contact+fit[5]*contact*contact
            radius = radii[0]/denom
        motion.append(position.tolist())
        if i == 0:
            frames[i] = small[i]
            continue
        canvas = source[i].astype(np.float32)
        # Remove the factual ball and its contact shadow, without retiming the
        # rest of the video. The large feather also cleans compression ringing.
        cx, cy = centers[i]
        ix, iy = np.round([cx, cy]).astype(int)
        x0, x1 = max(0, ix-42), min(1280, ix+43)
        y0, y1 = max(0, iy-42), min(720, iy+43)
        yy, xx = np.mgrid[y0:y1, x0:x1]
        removal = np.clip((40-np.hypot(xx-cx, yy-cy))/10, 0, 1)[..., None]
        canvas[y0:y1, x0:x1] = (canvas[y0:y1, x0:x1]*(1-removal)
                                +background[y0:y1, x0:x1]*removal)
        center_hi = (position+.4)*5
        scale = radius/radii[sprite_index]
        composite(canvas, sprites[sprite_index], center_hi, scale, shadow_strength)
        image = Image.fromarray(np.clip(canvas, 0, 255).round().astype(np.uint8))
        frames[i] = np.array(image.resize((256, 144), Image.Resampling.LANCZOS))

    artifact_io.save(frames, ROOT)
    metadata = {
        'initial_speed_multiplier': .6,
        'flight_coefficients': fit.tolist(),
        'initial_center': origin.tolist(),
        'source_landing_frame_zero_based': float(source_contact),
        'edited_landing_frame_zero_based': float(contact),
        'edited_stop_frame_zero_based': float(contact+stop_time),
        'rolling_drag': float(drag), 'rolling_friction': float(friction),
        'edited_positions': motion,

    (ROOT / 'motion_analysis.json').write_text(json.dumps(metadata, indent=2))

          f'stop: frame {contact+stop_time:.2f}, x={motion[-1][0]:.2f}')

    main()
