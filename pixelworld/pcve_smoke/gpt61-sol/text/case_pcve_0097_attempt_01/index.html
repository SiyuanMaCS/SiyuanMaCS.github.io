+"""Source-pixel counterfactual: triple red contact friction starting at frame 18.
+
+The ramp path, sliding acceleration, landing height, and both ball appearances
+are measured from the supplied RGB sequence. Dynamics below are a small local
+numerical model, not a rendering or simulation engine. Source pixels supply all
+surface texture and illumination.
+"""
+from pathlib import Path
+import json
+import warnings
+import numpy as np
+from scipy.ndimage import label, map_coordinates
+from artifact_io import save
+
+ROOT = Path(__file__).resolve().parent
+SRC = np.load(ROOT / 'source_pixels.npz')['rgb']
+N, H, W, _ = SRC.shape
+YY, XX = np.mgrid[:H, :W]

+def measure():
+    red = []
+    blue = []
+    for i, frame in enumerate(SRC):
+        r, g, b = frame.astype(float).transpose(2, 0, 1)
+        lab, _ = label((r > 100) & (r > 2*g) & (r > 2*b))
+        area = np.bincount(lab.ravel()); area[0] = 0
+        y, x = np.where(lab == area.argmax())
+        red.append((x.mean() + .45, y.mean() - 1.65))
+        m = (b > 1.5*r) & (b > 1.2*g) & (b > 80)
+        y, x = np.where(m)
+        if i <= 24 and len(x):
+            blue.append((x.mean() + .30, y.mean() - .67))
+        elif i == 25:
+            blue.append((1.65, 89.0))
+        elif i == 26:
+            blue.append((-3.65, 89.3))
+        elif i == 27:
+            blue.append((-8.35, 89.5))
+        else:
+            blue.append((-20., 89.5))
+    return np.array(red), np.array(blue)
+
+
+RED, BLUE = measure()
+
+
+def radius(x, kind):
+    # Perspective magnification increases slightly towards the foot of the ramp.
+    if kind == 'red':
+        return float(np.interp(x, [21, 59, 93, 179], [7.95, 7.75, 7.60, 7.15]))
+    return 8.05
+
+
+def footprint(p, r):
+    x, y = p
+    sphere = (XX-x)**2 + (YY-y)**2 < (r+2.1)**2
+    shadow = ((XX-x+.5)/(r+3))**2 + ((YY-y-r-2.3)/4.7)**2 < 1
+    reflection = (abs(XX-x) < r+1.8) & (YY > y+r+1) & (YY < y+r+34)
+    return sphere | shadow | reflection
+
+
+def clean_background():
+    # Take a temporal median only among unobstructed observations of each pixel.
+    # The two balls separate during the collision, revealing even the floor
+    # that is hidden in both the first and last frame.
+    f = SRC.astype(np.float32)
+    for t in range(N):
+        covered = footprint(RED[t], radius(RED[t, 0], 'red'))
+        if t <= 27:
+            covered |= footprint(BLUE[t], 8.05)
+        f[t, covered] = np.nan
+    with warnings.catch_warnings():
+        warnings.simplefilter('ignore', RuntimeWarning)
+        bg = np.nanmedian(f, axis=0)
+    missing = np.isnan(bg[:, :, 0])
+    # Rare permanently hidden reflection pixels: interpolate from nearby floor.
+    for _ in range(12):
+        if not missing.any(): break
+        accum = np.zeros_like(bg); count = np.zeros((H, W), np.float32)
+        for dy, dx in [(-1,0),(1,0),(0,-1),(0,1)]:
+            v = np.roll(bg, (dy,dx), (0,1))
+            good = np.isfinite(v[:,:,0])
+            if dy == 1: good[0] = False
+            if dy == -1: good[-1] = False
+            if dx == 1: good[:,0] = False
+            if dx == -1: good[:,-1] = False
+            accum += np.nan_to_num(v) * good[:,:,None]
+            count += good
+        usable = missing & (count > 0)
+        bg[usable] = accum[usable] / count[usable,None]
+        missing = np.isnan(bg[:,:,0])
+    return np.nan_to_num(bg, nan=130.)
+
+
+BG = clean_background()

+def trajectories():
+    start = 17
+    fit = np.polyfit(np.arange(21), RED[:21,0], 2)
+    slope, intercept = np.polyfit(RED[:21,0], RED[:21,1], 1)
+    vx = 2*fit[0]*start + fit[1]
+    flat_decel = 2*np.polyfit(np.arange(23,30), RED[23:30,0], 2)[0]
+    factual_ramp_accel = -2*fit[0]
+    # Tangential contact force increases threefold. The normal projection is
+    # slightly smaller on the incline. Positive x acceleration brakes leftward
+    # motion; static friction holds only after speed has actually reached zero.
+    incline_normal_factor = .955
+    ramp_ax = -factual_ramp_accel + 2*flat_decel*incline_normal_factor
+    red_decel = 3*flat_decel
+    x, y = RED[start].copy()

+    by_ground = lambda xx: 89.17 - .046*xx
+    exit_x = RED[20,0]
+    vy_exit_original = slope*(2*fit[0]*20 + fit[1])
+    gravity = 2*(RED[21,1] - RED[20,1] - vy_exit_original)
+    gravity = float(np.clip(gravity, 7.5, 10.))
+    mode = 'ramp'; vy = 0.
+    path_r = RED.copy(); path_b = BLUE.copy()
+    step = 1/240
+    impact_time = None
+    stop_time = None
+    for frame in range(start+1, N):
+        for j in range(240):
+            now = frame-1+(j+1)*step
+            if mode == 'ramp':
+                old_x = x
+                new_v = min(0., vx+ramp_ax*step)
+                x += .5*(vx+new_v)*step
+                vx = new_v
+                y = slope*x+intercept
+                if x <= exit_x:
+                    x = exit_x
+                    y = slope*x+intercept
+                    vy = slope*vx
+                    mode = 'air'
+            elif mode == 'air':
+                x += vx*step
+                y += vy*step + .5*gravity*step*step
+                vy += gravity*step
+                if y >= ry_ground(x):
+                    y = ry_ground(x)
+                    vy = 0.
+                    mode = 'floor'
+            else:
+                new_v = min(0., vx + red_decel*step)
+                x += .5*(vx+new_v)*step
+                vx = new_v
+                y = ry_ground(x)
+                if vx == 0. and stop_time is None:
+                    stop_time = now
+            nbv = min(0., bv+flat_decel*step)
+            bx += .5*(bv+nbv)*step
+            bv = nbv
+            by = by_ground(bx)
+            # One-dimensional impulse along the line of centers. The source
+            # restitution is approximately .28; blue retains its original drag.
+            distance = radius(x, 'red') + 8.05
+            dy = y-by
+            if abs(dy) < distance:
+                dx_contact = np.sqrt(distance*distance-dy*dy)
+                if x-bx <= dx_contact and x > bx and vx < bv-1e-6:
+                    overlap = dx_contact-(x-bx)
+                    x += overlap*.5; bx -= overlap*.5
+                    v1, v2 = vx, bv
+                    e = .28
+                    vx = .5*((1-e)*v1+(1+e)*v2)
+                    bv = .5*((1+e)*v1+(1-e)*v2)
+                    if impact_time is None: impact_time = now
+            if mode == 'floor': y = ry_ground(x)
+        path_r[frame] = (x,y)
+        path_b[frame] = (bx,by)
+    details = dict(intervention_frame_1based=18,
+                   measured_original_flat_deceleration=float(flat_decel),
+                   new_red_flat_deceleration=float(red_decel),
+                   measured_original_ramp_acceleration=float(factual_ramp_accel),
+                   new_red_ramp_acceleration=float(ramp_ax),
+                   impact_frame_1based=None if impact_time is None else impact_time+1,
+                   red_stop_frame_1based=None if stop_time is None else stop_time+1,
+                   final_red=path_r[-1].tolist(), final_blue=path_b[-1].tolist())
+    return path_r, path_b, details
+

+PR, PB, DETAILS = trajectories()

+def sample(arr, sy, sx):
+    return np.stack([map_coordinates(arr[:,:,c], [sy,sx], order=1,
+                    mode='nearest', prefilter=False) for c in range(3)], axis=-1)
+

+def render_ball(canvas, p, kind, source_index, airborne=False):
+    origin = RED[source_index] if kind == 'red' else BLUE[source_index]
+    old_r = radius(origin[0], kind)
+    new_r = radius(p[0], kind)
+    scale = new_r/old_r
+    x0 = max(0, int(np.floor(p[0]-new_r-5)))
+    x1 = min(W, int(np.ceil(p[0]+new_r+5)))
+    y0 = max(0, int(np.floor(p[1]-new_r-3)))
+    y1 = min(H, int(np.ceil(p[1]+new_r+34)))
+    if x1 <= x0 or y1 <= y0: return
+    yy, xx = np.mgrid[y0:y1, x0:x1]
+    sx = origin[0] + (xx-p[0])/scale
+    sy = origin[1] + (yy-p[1])/scale
+    reference = sample(SRC[source_index].astype(np.float32), sy, sx)
+    reference_bg = sample(BG, sy, sx)
+    region = canvas[y0:y1,x0:x1]
+    d = np.sqrt((xx-p[0])**2+(yy-p[1])**2)
+    alpha = np.clip(new_r+.25-d, 0., 1.)
+    # Source edge residual preserves the original subpixel antialiasing; opaque
+    # sphere pixels keep their actual source lighting rather than floor texture.
+    support = np.clip((new_r+1.8-d)/1.2, 0., 1.)
+    sphere = (region*(1-alpha[:,:,None]) + reference*alpha[:,:,None]
+              +(reference-reference_bg)*(support-alpha)[:,:,None])
+    # A source-ratio shadow keeps local wood grain. Its soft red/blue reflection
+    # is moved with the ball too. Airborne shadow is handled separately below.
+    if not airborne:
+        vertical = yy-p[1]
+        sh = np.clip((vertical-(new_r-.7))/1.4, 0., 1.)
+        sh *= np.clip((new_r+3.1-abs(xx-p[0]))/3., 0., 1.)
+        sh *= np.clip((new_r+33-vertical)/9., 0., 1.)
+        ratio = np.clip((reference+4)/(reference_bg+4), .15, 1.65)
+        reflected = region*(1+(ratio-1)*sh[:,:,None])
+        sphere = reflected*(1-alpha[:,:,None]) + reference*alpha[:,:,None] \
+                 +(reference-reference_bg)*(support-alpha)[:,:,None]
+    region[:] = np.clip(sphere,0,255)
+
+
+def airborne_shadow(canvas, p):
+    # At the drop the source has a separated, blurred shadow on the lower floor.
+    gy = 89.62-.046*p[0]+8.3
+    height = max(0., (89.62-.046*p[0])-p[1])
+    spread = 4.8+.09*height
+    strength = .24*np.exp(-height/29.)
+    shadow = strength*np.exp(-.5*((XX-p[0]+.9)/spread)**2
+                            -.5*((YY-gy)/1.7)**2)
+    canvas *= (1-shadow[:,:,None])
+
+
+def generate():
+    out = SRC.copy()
+    # The instantaneous parameter switch at frame 18 does not teleport an object.
+    # Keep frames 1..18 byte-exact; the first changed displacement is frame 19.
+    for t in range(18,N):
+        frame = SRC[t].astype(np.float32)
+        remove = footprint(RED[t], radius(RED[t,0],'red'))
+        if t <= 27: remove |= footprint(BLUE[t],8.05)
+        frame[remove] = BG[remove]
+        red_pos = PR[t]; blue_pos = PB[t]
+        floor_y = 89.62-.046*red_pos[0]
+        is_air = red_pos[0] < RED[20,0]-.2 and red_pos[1] < floor_y-.35
+        if is_air:
+            template = 21
+            airborne_shadow(frame, red_pos)
+        elif red_pos[1] >= floor_y-.35:
+            template = 23
+        else:
+            template = int(np.argmin(abs(RED[17:21,0]-red_pos[0])))+17
+        # Blue is fully source-identical until it is affected by the impact.
+        if np.linalg.norm(blue_pos-BLUE[17]) < .12:
+            original_patch = footprint(BLUE[17],8.05)
+            frame[original_patch] = SRC[t,original_patch]
+        else:
+            render_ball(frame, blue_pos, 'blue',23)
+        render_ball(frame, red_pos, 'red',template, airborne=is_air)
+        out[t] = np.rint(np.clip(frame,0,255)).astype(np.uint8)
+    assert np.array_equal(out[:18], SRC[:18])
+    save(out, ROOT)
+    (ROOT/'motion_report.json').write_text(json.dumps(DETAILS,indent=2))
+    print(json.dumps(DETAILS,indent=2))
+    print('Saved', out.shape, out.dtype)
+
+
+if __name__ == '__main__':
+    generate()

The ramp path, sliding acceleration, landing height, and both ball appearances
are measured from the supplied RGB sequence. Dynamics below are a small local
numerical model, not a rendering or simulation engine. Source pixels supply all

import json
import warnings
import numpy as np
from scipy.ndimage import label, map_coordinates
from artifact_io import save

ROOT = Path(__file__).resolve().parent
SRC = np.load(ROOT / 'source_pixels.npz')['rgb']
N, H, W, _ = SRC.shape
YY, XX = np.mgrid[:H, :W]


def measure():
    red = []
    blue = []
    for i, frame in enumerate(SRC):
        r, g, b = frame.astype(float).transpose(2, 0, 1)
        lab, _ = label((r > 100) & (r > 2*g) & (r > 2*b))
        area = np.bincount(lab.ravel()); area[0] = 0
        y, x = np.where(lab == area.argmax())
        red.append((x.mean() + .45, y.mean() - 1.65))
        m = (b > 1.5*r) & (b > 1.2*g) & (b > 80)
        y, x = np.where(m)
        if i <= 24 and len(x):
            blue.append((x.mean() + .30, y.mean() - .67))
        elif i == 25:
            blue.append((1.65, 89.0))
        elif i == 26:
            blue.append((-3.65, 89.3))
        elif i == 27:
            blue.append((-8.35, 89.5))
        else:
            blue.append((-20., 89.5))
    return np.array(red), np.array(blue)


RED, BLUE = measure()


def radius(x, kind):
    # Perspective magnification increases slightly towards the foot of the ramp.
    if kind == 'red':
        return float(np.interp(x, [21, 59, 93, 179], [7.95, 7.75, 7.60, 7.15]))
    return 8.05


def footprint(p, r):
    x, y = p
    sphere = (XX-x)**2 + (YY-y)**2 < (r+2.1)**2
    shadow = ((XX-x+.5)/(r+3))**2 + ((YY-y-r-2.3)/4.7)**2 < 1
    reflection = (abs(XX-x) < r+1.8) & (YY > y+r+1) & (YY < y+r+34)
    return sphere | shadow | reflection


def clean_background():
    # Take a temporal median only among unobstructed observations of each pixel.
    # The two balls separate during the collision, revealing even the floor
    # that is hidden in both the first and last frame.
    f = SRC.astype(np.float32)
    for t in range(N):
        covered = footprint(RED[t], radius(RED[t, 0], 'red'))
        if t <= 27:
            covered |= footprint(BLUE[t], 8.05)
        f[t, covered] = np.nan
    with warnings.catch_warnings():
        warnings.simplefilter('ignore', RuntimeWarning)
        bg = np.nanmedian(f, axis=0)
    missing = np.isnan(bg[:, :, 0])
    # Rare permanently hidden reflection pixels: interpolate from nearby floor.
    for _ in range(12):
        if not missing.any(): break
        accum = np.zeros_like(bg); count = np.zeros((H, W), np.float32)
        for dy, dx in [(-1,0),(1,0),(0,-1),(0,1)]:
            v = np.roll(bg, (dy,dx), (0,1))
            good = np.isfinite(v[:,:,0])
            if dy == 1: good[0] = False
            if dy == -1: good[-1] = False
            if dx == 1: good[:,0] = False
            if dx == -1: good[:,-1] = False
            accum += np.nan_to_num(v) * good[:,:,None]
            count += good
        usable = missing & (count > 0)
        bg[usable] = accum[usable] / count[usable,None]
        missing = np.isnan(bg[:,:,0])
    return np.nan_to_num(bg, nan=130.)


BG = clean_background()


def trajectories():
    start = 17
    fit = np.polyfit(np.arange(21), RED[:21,0], 2)
    slope, intercept = np.polyfit(RED[:21,0], RED[:21,1], 1)
    vx = 2*fit[0]*start + fit[1]
    flat_decel = 2*np.polyfit(np.arange(23,30), RED[23:30,0], 2)[0]
    factual_ramp_accel = -2*fit[0]
    # Tangential contact force increases threefold. The normal projection is
    # slightly smaller on the incline. Positive x acceleration brakes leftward
    # motion; static friction holds only after speed has actually reached zero.
    incline_normal_factor = .955
    ramp_ax = -factual_ramp_accel + 2*flat_decel*incline_normal_factor
    red_decel = 3*flat_decel
    x, y = RED[start].copy()
    bx, by = BLUE[start].copy(); bv = 0.
    ry_ground = lambda xx: 89.62 - .046*xx
    by_ground = lambda xx: 89.17 - .046*xx
    exit_x = RED[20,0]
    vy_exit_original = slope*(2*fit[0]*20 + fit[1])
    gravity = 2*(RED[21,1] - RED[20,1] - vy_exit_original)
    gravity = float(np.clip(gravity, 7.5, 10.))
    mode = 'ramp'; vy = 0.
    path_r = RED.copy(); path_b = BLUE.copy()
    step = 1/240
    impact_time = None
    stop_time = None
    for frame in range(start+1, N):
        for j in range(240):
            now = frame-1+(j+1)*step
            if mode == 'ramp':
                old_x = x
                new_v = min(0., vx+ramp_ax*step)
                x += .5*(vx+new_v)*step
                vx = new_v
                y = slope*x+intercept
                if x <= exit_x:
                    x = exit_x
                    y = slope*x+intercept
                    vy = slope*vx
                    mode = 'air'
            elif mode == 'air':
                x += vx*step
                y += vy*step + .5*gravity*step*step
                vy += gravity*step
                if y >= ry_ground(x):
                    y = ry_ground(x)
                    vy = 0.
                    mode = 'floor'
            else:
                new_v = min(0., vx + red_decel*step)
                x += .5*(vx+new_v)*step
                vx = new_v
                y = ry_ground(x)
                if vx == 0. and stop_time is None:
                    stop_time = now
            nbv = min(0., bv+flat_decel*step)
            bx += .5*(bv+nbv)*step
            bv = nbv
            by = by_ground(bx)
            # One-dimensional impulse along the line of centers. The source
            # restitution is approximately .28; blue retains its original drag.
            distance = radius(x, 'red') + 8.05
            dy = y-by
            if abs(dy) < distance:

                    if impact_time is None: impact_time = now
            if mode == 'floor': y = ry_ground(x)

    return path_r, path_b, details

def sample(arr, sy, sx):
    return np.stack([map_coordinates(arr[:,:,c], [sy,sx], order=1,
                    mode='nearest', prefilter=False) for c in range(3)], axis=-1)


def render_ball(canvas, p, kind, source_index, airborne=False):
    origin = RED[source_index] if kind == 'red' else BLUE[source_index]
    old_r = radius(origin[0], kind)
    new_r = radius(p[0], kind)
    scale = new_r/old_r
    x0 = max(0, int(np.floor(p[0]-new_r-5)))
    x1 = min(W, int(np.ceil(p[0]+new_r+5)))
    y0 = max(0, int(np.floor(p[1]-new_r-3)))
    y1 = min(H, int(np.ceil(p[1]+new_r+34)))
    if x1 <= x0 or y1 <= y0: return
    yy, xx = np.mgrid[y0:y1, x0:x1]
    sx = origin[0] + (xx-p[0])/scale
    sy = origin[1] + (yy-p[1])/scale
    reference = sample(SRC[source_index].astype(np.float32), sy, sx)
    reference_bg = sample(BG, sy, sx)
    region = canvas[y0:y1,x0:x1]
    d = np.sqrt((xx-p[0])**2+(yy-p[1])**2)
    alpha = np.clip(new_r+.25-d, 0., 1.)
    # Source edge residual preserves the original subpixel antialiasing; opaque
    # sphere pixels keep their actual source lighting rather than floor texture.
    support = np.clip((new_r+1.8-d)/1.2, 0., 1.)
    sphere = (region*(1-alpha[:,:,None]) + reference*alpha[:,:,None]
              +(reference-reference_bg)*(support-alpha)[:,:,None])
    # A source-ratio shadow keeps local wood grain. Its soft red/blue reflection
    # is moved with the ball too. Airborne shadow is handled separately below.
    if not airborne:
        vertical = yy-p[1]
        sh = np.clip((vertical-(new_r-.7))/1.4, 0., 1.)
        sh *= np.clip((new_r+3.1-abs(xx-p[0]))/3., 0., 1.)
        sh *= np.clip((new_r+33-vertical)/9., 0., 1.)
        ratio = np.clip((reference+4)/(reference_bg+4), .15, 1.65)
        reflected = region*(1+(ratio-1)*sh[:,:,None])
        sphere = reflected*(1-alpha[:,:,None]) + reference*alpha[:,:,None] \\
                 +(reference-reference_bg)*(support-alpha)[:,:,None]
    region[:] = np.clip(sphere,0,255)


def airborne_shadow(canvas, p):
    # At the drop the source has a separated, blurred shadow on the lower floor.
    gy = 89.62-.046*p[0]+8.3
    height = max(0., (89.62-.046*p[0])-p[1])
    spread = 4.8+.09*height
    strength = .24*np.exp(-height/29.)
    shadow = strength*np.exp(-.5*((XX-p[0]+.9)/spread)**2
                            -.5*((YY-gy)/1.7)**2)
    canvas *= (1-shadow[:,:,None])


def generate():
    out = SRC.copy()
    # The instantaneous parameter switch at frame 18 does not teleport an object.
    # Keep frames 1..18 byte-exact; the first changed displacement is frame 19.
    for t in range(18,N):
        frame = SRC[t].astype(np.float32)
        remove = footprint(RED[t], radius(RED[t,0],'red'))
        if t <= 27: remove |= footprint(BLUE[t],8.05)
        frame[remove] = BG[remove]
        red_pos = PR[t]; blue_pos = PB[t]
        floor_y = 89.62-.046*red_pos[0]
        is_air = red_pos[0] < RED[20,0]-.2 and red_pos[1] < floor_y-.35
        if is_air:
            template = 21
            airborne_shadow(frame, red_pos)
        elif red_pos[1] >= floor_y-.35:
            template = 23
        else:
            template = int(np.argmin(abs(RED[17:21,0]-red_pos[0])))+17
        # Blue is fully source-identical until it is affected by the impact.
        if np.linalg.norm(blue_pos-BLUE[17]) < .12:
            original_patch = footprint(BLUE[17],8.05)
            frame[original_patch] = SRC[t,original_patch]
        else:
            render_ball(frame, blue_pos, 'blue',23)
        render_ball(frame, red_pos, 'red',template, airborne=is_air)
        out[t] = np.rint(np.clip(frame,0,255)).astype(np.uint8)
    assert np.array_equal(out[:18], SRC[:18])
    save(out, ROOT)
    (ROOT/'motion_report.json').write_text(json.dumps(DETAILS,indent=2))
    print(json.dumps(DETAILS,indent=2))
    print('Saved', out.shape, out.dtype)


if __name__ == '__main__':
    generate()
