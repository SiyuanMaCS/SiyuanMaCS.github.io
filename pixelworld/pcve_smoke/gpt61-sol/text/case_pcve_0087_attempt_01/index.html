+from pathlib import Path
+import numpy as np
+import cv2
+from PIL import Image
+import artifact_io

+ROOT = Path(__file__).resolve().parent
+FPS = 24
+NFRAMES = 96

+def read_source():
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    src = []
+    for i in range(13):
+        ok, bgr = cap.read()
+        if not ok:
+            raise RuntimeError('Could not decode source')
+        src.append(cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB))
+    cap.release()
+    return src
+

+def geometry():
+    # Numerically fitted intersections of the four felt edges.
+    corners = np.float32([[-14.71491761,105.114298],
+                          [99.705062,153.047915],
+                          [208.793546,50.267515],
+                          [137.982947,40.557849]])
+    f = 247.571156
+    K = np.array([[f,0,128],[0,f,72],[0,0,1.]])
+    h = cv2.getPerspectiveTransform(np.float32([[0,0],[0,1],[2,1],[2,0]]), corners)
+    a = np.linalg.inv(K) @ h
+    length = 2*np.linalg.norm(a[:,0])/np.linalg.norm(a[:,1])
+    h = cv2.getPerspectiveTransform(np.float32([[0,0],[0,1],[length,1],[length,0]]), corners)
+    a = np.linalg.inv(K) @ h
+    s = 1/np.linalg.norm(a[:,0])
+    r1, r2, t = a[:,0]*s, a[:,1]*s, a[:,2]*s
+    normal = np.cross(r1,r2)
+    # Both balls travel along the same longitudinal line in the factual shot.
+    def camera(p, rad):
+        return t + r1*p[0] + r2*p[1] - normal*rad
+    def project(p, rad):
+        q = camera(p,rad)
+        screen = K @ q
+        return screen[:2]/screen[2], q[2]
+    return length, project
+
+
+def background_and_sprites(src):
+    initial = src[0].astype(np.float32)
+    bg = initial.copy()
+    # The moving cue ball exposes its entire original footprint by frame 11.
+    bg[431:535,388:480] = src[10][431:535,388:480]
+    # The target's starting position is subsequently occluded by the stopped
+    # cue ball. Fit the smooth local felt from an unoccluded surrounding ring.
+    x0,x1,y0,y1 = 596,680,309,395
+    xx,yy = np.meshgrid(np.arange(x0,x1),np.arange(y0,y1))
+    X=(xx-638)/50.; Y=(yy-352)/50.
+    basis=np.stack([np.ones_like(X),X,Y,X*X,X*Y,Y*Y,
+                    X**3,X*X*Y,X*Y*Y,Y**3],axis=-1)
+    border=(xx<606)|(xx>669)|(yy<320)|(yy>383)
+    coeff=np.linalg.lstsq(basis[border], initial[y0:y1,x0:x1][border],rcond=None)[0]
+    fitted=basis@coeff
+    # Smooth patch boundary to avoid an artificial inpaint seam.
+    edge=np.minimum.reduce([xx-x0,x1-1-xx,yy-y0,y1-1-yy])
+    weight=np.clip(edge/6.,0,1)[...,None]
+    bg[y0:y1,x0:x1]=initial[y0:y1,x0:x1]*(1-weight)+fitted*weight
+
+    specs=[(388,431,480,535,np.array([85.7296,94.1803]),3.76),
+           (596,309,680,395,np.array([127.8468,69.9369]),2.74)]
+    sprites=[]
+    for x0,y0,x1,y1,center,radius in specs:
+        orig=initial[y0:y1,x0:x1]
+        clean=bg[y0:y1,x0:x1]
+        delta=orig-clean
+        yy,xx=np.mgrid[y0:y1,x0:x1]
+        dist=np.hypot(xx/5-center[0],yy/5-center[1])
+        # Approximate opaque core only used to make the transplanted sphere
+        # independent of the varying felt brightness. Pixel details remain
+        # those of the source, including highlights and contact shadows.
+        opaque=np.clip((radius-dist)/.30,0,1).astype(np.float32)
+        feather=np.minimum.reduce([xx-x0,x1-1-xx,yy-y0,y1-1-yy])
+        fade=np.clip(feather/5.,0,1).astype(np.float32)
+        delta*=fade[...,None]
+        sprites.append(dict(delta=delta, alpha=opaque*fade, clean=clean,
+                            center=center*5, origin=np.array([x0,y0])))
+    return bg,sprites
+

+def dynamics(length):
+    # Unit = felt width, time = one output frame. Perspective-correct
+    # source fit x(t)=.45716+.042928*t-.00011366*t^2.
+    p=np.array([[.45729115,.5300],[1.06747605,.5300]],dtype=float)
+    rad=np.array([.0278,.0260])
+    v=np.array([[4*.0429281857,0.],[0.,0.]])
+    friction=.000224
+    restitution=.86
+    cushion=.78
+    inset=.061
+    trajectory=[p.copy()]
+    events=[]
+    substeps=240
+    dt=1/substeps
+    for frame in range(1,NFRAMES):
+        for step in range(substeps):
+            speed=np.linalg.norm(v,axis=1)
+            v*=np.maximum(0,1-friction*dt/np.maximum(speed,1e-12))[:,None]
+            p+=v*dt
+            for i in range(2):
+                # Ball centers cannot pass through the cushions.
+                for axis,lo,hi in [(0,inset+rad[i],length-inset-rad[i]),
+                                   (1,inset+rad[i],1-inset-rad[i])]:
+                    if p[i,axis]<lo:
+                        p[i,axis]=lo+(lo-p[i,axis])
+                        v[i,axis]=abs(v[i,axis])*cushion
+                        events.append((round(frame-1+(step+1)*dt,4),'near cushion',i))
+                    elif p[i,axis]>hi:
+                        p[i,axis]=hi-(p[i,axis]-hi)
+                        v[i,axis]=-abs(v[i,axis])*cushion
+                        events.append((round(frame-1+(step+1)*dt,4),'far cushion',i))
+            separation=p[1]-p[0]
+            d=np.linalg.norm(separation)
+            if d<rad.sum():
+                n=separation/max(d,1e-12)
+                overlap=rad.sum()-d
+                p[0]-=n*(overlap/2+1e-9)
+                p[1]+=n*(overlap/2+1e-9)
+                approach=np.dot(v[1]-v[0],n)
+                if approach<0:
+                    impulse=-(1+restitution)*approach/2
+                    v[0]-=impulse*n
+                    v[1]+=impulse*n
+                    events.append((round(frame-1+(step+1)*dt,4),'ball contact'))
+        trajectory.append(p.copy())
+    return np.array(trajectory),rad,events
+
+
+def paste_sprite(out,bg,sprite,center,scale):
+    # Warp just a local bounding box, not the entire HD frame.
+    center=np.asarray(center)*5
+    origin=sprite['origin']; ref=sprite['center']
+    h,w=sprite['delta'].shape[:2]
+    tl=center+(origin-ref)*scale
+    br=center+(origin+np.array([w,h])-ref)*scale
+    x0=max(0,int(np.floor(tl[0]))-2); y0=max(0,int(np.floor(tl[1]))-2)
+    x1=min(out.shape[1],int(np.ceil(br[0]))+2);y1=min(out.shape[0],int(np.ceil(br[1]))+2)
+    if x1<=x0 or y1<=y0:return
+    yy,xx=np.mgrid[y0:y1,x0:x1]
+    mx=((xx-center[0])/scale+ref[0]-origin[0]).astype(np.float32)
+    my=((yy-center[1])/scale+ref[1]-origin[1]).astype(np.float32)
+    warp=lambda a:cv2.remap(a,mx,my,cv2.INTER_LINEAR,borderMode=cv2.BORDER_CONSTANT)
+    delta=warp(sprite['delta'])
+    opaque=warp(sprite['alpha'])[...,None]
+    clean=warp(sprite['clean'])
+    target=bg[y0:y1,x0:x1]
+    out[y0:y1,x0:x1]+=delta+opaque*(clean-target)
+
+
+def main():
+    src=read_source()
+    bg,sprites=background_and_sprites(src)
+    length,project=geometry()
+    path,rad,events=dynamics(length)
+    z0=[project(path[0,i],rad[i])[1] for i in range(2)]
+    # Source centers rather than abstract centroids keep first positions exact;
+    # perspective displacements are then relative to that calibrated anchor.
+    anchors=[project(path[0,i],rad[i])[0] for i in range(2)]
+    result=[]
+    original=np.load(ROOT/'source_pixels.npz')['rgb']
+    for frame in range(NFRAMES):
+        if frame==0:
+            result.append(original[0].copy())
+            continue
+        out=bg.copy()
+        current=[project(path[frame,i],rad[i]) for i in range(2)]
+        # Far-to-near ordering maintains physical occlusion at contacts.
+        for i in sorted(range(2),key=lambda i:current[i][1],reverse=True):
+            screen,z=current[i]
+            anchor=sprites[i]['center']/5
+            center=screen-anchors[i]+anchor
+            paste_sprite(out,bg,sprites[i],center,z0[i]/z)
+        image=Image.fromarray(np.clip(np.rint(out),0,255).astype(np.uint8))
+        result.append(np.array(image.resize((256,144),Image.Resampling.LANCZOS)))
+    result=np.stack(result)
+    artifact_io.save(result,ROOT)
+    np.savez_compressed(ROOT/'.tmp/motion_check.npz',positions=path,radii=rad)
+    print('Saved',result.shape,result.dtype)
+    print('Initial speed multiplier: 4.000')
+    print('Interaction events (zero-based fractional frame):',events)
+
+
+if __name__=='__main__':
+    main()

\"\"\"Counterfactual billiards edit, source-pixel rendering and calibrated mechanics.\"\"\"
from pathlib import Path
import numpy as np
import cv2
from PIL import Image
import artifact_io

ROOT = Path(__file__).resolve().parent
FPS = 24
NFRAMES = 96


def read_source():
    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
    src = []
    for i in range(13):
        ok, bgr = cap.read()
        if not ok:
            raise RuntimeError('Could not decode source')
        src.append(cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB))
    cap.release()
    return src


def geometry():
    # Numerically fitted intersections of the four felt edges.
    corners = np.float32([[-14.71491761,105.114298],
                          [99.705062,153.047915],
                          [208.793546,50.267515],
                          [137.982947,40.557849]])
    f = 247.571156
    K = np.array([[f,0,128],[0,f,72],[0,0,1.]])
    h = cv2.getPerspectiveTransform(np.float32([[0,0],[0,1],[2,1],[2,0]]), corners)
    a = np.linalg.inv(K) @ h
    length = 2*np.linalg.norm(a[:,0])/np.linalg.norm(a[:,1])
    h = cv2.getPerspectiveTransform(np.float32([[0,0],[0,1],[length,1],[length,0]]), corners)
    a = np.linalg.inv(K) @ h
    s = 1/np.linalg.norm(a[:,0])
    r1, r2, t = a[:,0]*s, a[:,1]*s, a[:,2]*s
    normal = np.cross(r1,r2)
    # Both balls travel along the same longitudinal line in the factual shot.
    def camera(p, rad):
        return t + r1*p[0] + r2*p[1] - normal*rad
    def project(p, rad):
        q = camera(p,rad)
        screen = K @ q
        return screen[:2]/screen[2], q[2]
    return length, project


def background_and_sprites(src):
    initial = src[0].astype(np.float32)
    bg = initial.copy()
    # The moving cue ball exposes its entire original footprint by frame 11.
    bg[431:535,388:480] = src[10][431:535,388:480]
    # The target's starting position is subsequently occluded by the stopped
    # cue ball. Fit the smooth local felt from an unoccluded surrounding ring.
    x0,x1,y0,y1 = 596,680,309,395
    xx,yy = np.meshgrid(np.arange(x0,x1),np.arange(y0,y1))
    X=(xx-638)/50.; Y=(yy-352)/50.
    basis=np.stack([np.ones_like(X),X,Y,X*X,X*Y,Y*Y,
                    X**3,X*X*Y,X*Y*Y,Y**3],axis=-1)
    border=(xx<606)|(xx>669)|(yy<320)|(yy>383)
    coeff=np.linalg.lstsq(basis[border], initial[y0:y1,x0:x1][border],rcond=None)[0]
    fitted=basis@coeff
    # Smooth patch boundary to avoid an artificial inpaint seam.
    edge=np.minimum.reduce([xx-x0,x1-1-xx,yy-y0,y1-1-yy])
    weight=np.clip(edge/6.,0,1)[...,None]
    bg[y0:y1,x0:x1]=initial[y0:y1,x0:x1]*(1-weight)+fitted*weight

    specs=[(388,431,480,535,np.array([85.7296,94.1803]),3.76),
           (596,309,680,395,np.array([127.8468,69.9369]),2.74)]
    sprites=[]
    for x0,y0,x1,y1,center,radius in specs:
        orig=initial[y0:y1,x0:x1]
        clean=bg[y0:y1,x0:x1]
        delta=orig-clean
        yy,xx=np.mgrid[y0:y1,x0:x1]
        dist=np.hypot(xx/5-center[0],yy/5-center[1])
        # Approximate opaque core only used to make the transplanted sphere
        # independent of the varying felt brightness. Pixel details remain
        # those of the source, including highlights and contact shadows.
        opaque=np.clip((radius-dist)/.30,0,1).astype(np.float32)
        feather=np.minimum.reduce([xx-x0,x1-1-xx,yy-y0,y1-1-yy])
        fade=np.clip(feather/5.,0,1).astype(np.float32)
        delta*=fade[...,None]
        sprites.append(dict(delta=delta, alpha=opaque*fade, clean=clean,
                            center=center*5, origin=np.array([x0,y0])))
    return bg,sprites


def dynamics(length):
    # Unit = felt width, time = one output frame. Perspective-correct
    # source fit x(t)=.45716+.042928*t-.00011366*t"'^2.
    p=np.array([[.45729115,.5300],[1.06747605,.5300]],dtype=float)
    rad=np.array([.0278,.0260])
    v=np.array([[4*.0429281857,0.],[0.,0.]])
    friction=.000224
    restitution=.86
    cushion=.78
    inset=.061
    trajectory=[p.copy()]
    events=[]
    substeps=240
    dt=1/substeps
    for frame in range(1,NFRAMES):
        for step in range(substeps):
            speed=np.linalg.norm(v,axis=1)
            v*=np.maximum(0,1-friction*dt/np.maximum(speed,1e-12))[:,None]
            p+=v*dt
            for i in range(2):
                # Ball centers cannot pass through the cushions.
                for axis,lo,hi in [(0,inset+rad[i],length-inset-rad[i]),
                                   (1,inset+rad[i],1-inset-rad[i])]:
                    if p[i,axis]<lo:
                        p[i,axis]=lo+(lo-p[i,axis])
                        v[i,axis]=abs(v[i,axis])*cushion
                        events.append((round(frame-1+(step+1)*dt,4),'"'near cushion',i))
                    elif p[i,axis]>hi:
                        p[i,axis]=hi-(p[i,axis]-hi)
                        v[i,axis]=-abs(v[i,axis])*cushion
                        events.append((round(frame-1+(step+1)*dt,4),'far cushion',i))
            separation=p[1]-p[0]
            d=np.linalg.norm(separation)
            if d<rad.sum():
                n=separation/max(d,1e-12)
                overlap=rad.sum()-d
                p[0]-=n*(overlap/2+1e-9)
                p[1]+=n*(overlap/2+1e-9)
                approach=np.dot(v[1]-v[0],n)
                if approach<0:
                    impulse=-(1+restitution)*approach/2
                    v[0]-=impulse*n
                    v[1]+=impulse*n
                    events.append((round(frame-1+(step+1)*dt,4),'ball contact'))
        trajectory.append(p.copy())
    return np.array(trajectory),rad,events


def paste_sprite(out,bg,sprite,center,scale):
    # Warp just a local bounding box, not the entire HD frame.
    center=np.asarray(center)*5
    origin=sprite['origin']; ref=sprite['center']
    h,w=sprite['delta'].shape[:2]
    tl=center+(origin-ref)*scale
    br=center+(origin+np.array([w,h])-ref)*scale
    x0=max(0,int(np.floor(tl[0]))-2); y0=max(0,int(np.floor(tl[1]))-2)
    x1=min(out.shape[1],int(np.ceil(br[0]))+2);y1=min(out.shape[0],int(np.ceil(br[1]))+2)
    if x1<=x0 or y1<=y0:return
    yy,xx=np.mgrid[y0:y1,x0:x1]
    mx=((xx-center[0])/scale+ref[0]-origin[0]).astype(np.float32)
    my=((yy-center[1])/scale+ref[1]-origin[1]).astype(np.float32)
    warp=lambda a:cv2.remap(a,mx,my,cv2.INTER_LINEAR,borderMode=cv2.BORDER_CONSTANT)
    delta=warp(sprite['delta'])
    opaque=warp(sprite['alpha'])[...,None]
    clean=warp(sprite['clean'])
    target=bg[y0:y1,x0:x1]
    out[y0:y1,x0:x1]+=delta+opaque*(clean-target)


def main():
    src=read_source()
    bg,sprites=background_and_sprites(src)
    length,project=geometry()
    path,rad,events=dynamics(length)
    z0=[project(path[0,i],rad[i])[1] for i in range(2)]
    # Source centers rather than abstract centroids keep first positions exact;
    # perspective displacements are then relative to that calibrated anchor.
    anchors=[project(path[0,i],rad[i])[0] for i in range(2)]
    result=[]
    original=np.load(ROOT/'source_pixels.npz')['rgb']
    for frame in range(NFRAMES):
        if frame==0:
            result.append(original[0].copy())
            continue
        out=bg.copy()
        current=[project(path[frame,i],rad[i]) for i in range(2)]
        # Far-to-near ordering maintains physical occlusion at contacts.
        for i in sorted(range(2),key=lambda i:current[i][1],reverse=True):
            screen,z=current[i]
            anchor=sprites[i]['center']/5
            center=screen-anchors[i]+anchor
            paste_sprite(out,bg,sprites[i],center,z0[i]/z)
        image=Image.fromarray(np.clip(np.rint(out),0,255).astype(np.uint8))
        result.append(np.array(image.resize((256,144),Image.Resampling.LANCZOS)))
    result=np.stack(result)
    artifact_io.save(result,ROOT)
    np.savez_compressed(ROOT/'.tmp/motion_check.npz',positions=path,radii=rad)
    print('Saved',result.shape,result.dtype)
    print('Initial speed multiplier: 4.000')
    print('Interaction events (zero-based fractional frame):',events)


if __name__=='__main__':
    main()
