+"""Tenfold cue-ball cloth friction, on the original four-second timeline.
+
+All appearance comes from source RGB. The moving sphere and its contact shadow
+are lifted from frame zero, then subpixel warped over the stationary source
+table. A projective camera model converts measured image positions into a
+constant-deceleration cloth trajectory. No external renderer is used.
+"""
+from pathlib import Path
+
+import cv2
+import numpy as np
+from PIL import Image
+from scipy.ndimage import gaussian_filter
+from scipy.optimize import least_squares
+
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+
+
+def load_source():
+    small = np.load(ROOT / "source_pixels.npz")["rgb"]
+    cap = cv2.VideoCapture(str(ROOT / "source_video.mp4"))
+    full = []
+    for _ in range(len(small)):
+        ok, bgr = cap.read()
+        if not ok:
+            raise RuntimeError("Incomplete source video")
+        full.append(cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB))
+    cap.release()
+    return small, np.asarray(full)
+
+
+def friction_fit(full, clean, patch, rect, anchor):
+    """Track only the first nine frames, before object-ball contamination.
+
+    The table's two long edges converge near (284,-21). Its far and
+    near edges give the same horizon for this nearly longitudinal shot.
+    q is forward displacement divided by the initial camera depth.
+    In these coordinates Coulomb cloth friction is a constant deceleration.
+    """
+    xlo, ylo, xhi, yhi = rect
+    ox, oy = anchor
+    tracks = []
+    for k in range(9):
+        t = k / 24
+        q0 = .45 * t - .025 * t * t
+        gx = (ox + 284.2 * q0) / (1 + q0)
+        gy = (oy - 20.8 * q0) / (1 + q0)
+        s0 = 1 / (1 + q0)
+        rx, ry = int((gx-11)*5), int((gy-12)*5)
+        width, height = 110, 125
+        target = (full[k, ry:ry+height, rx:rx+width].astype(np.float32)
+                  - clean[ry:ry+height, rx:rx+width])
+
+        def residual(p):
+            x, y, s = p
+            mat = np.array([[s, 0, 5*x-s*(5*ox-xlo)-rx],
+                            [0, s, 5*y-s*(5*oy-ylo)-ry]], np.float32)
+            warped = cv2.warpAffine(patch, mat, (width, height),
+                                    flags=cv2.INTER_LINEAR)
+            return ((warped-target)/100).ravel()
+
+        if k == 0:
+            tracks.append([ox, oy, 1.])
+        else:
+            fitted = least_squares(residual, [gx, gy, s0],
+                                   diff_step=.00015, max_nfev=70,
+                                   ftol=1e-7, xtol=1e-8)
+            tracks.append(fitted.x)
+    tracks = np.asarray(tracks)
+    times = np.arange(len(tracks)) / 24.
+
+    def fit_residual(p):
+        x, y, v, a = p
+        q = v*times - .5*a*times**2
+        return np.r_[(x+284.2*q)/(1+q)-tracks[:, 0],
+                     (y-20.8*q)/(1+q)-tracks[:, 1]]
+
+    fit = least_squares(fit_residual, [ox, oy, .45, .05],
+                        bounds=([ox-.15, oy-.15, .40, .025],
+                                [ox+.15, oy+.15, .50, .10]))
+    _, _, velocity, acceleration = fit.x
+    print("RGB cue tracks:", np.round(tracks, 5).tolist())
+    print("Factual depth-coordinate velocity, friction acceleration:",
+          velocity, acceleration)
+    return velocity, acceleration
+
+
+def generate():
+    source, full = load_source()
+    first = full[0].astype(np.float32)
+    # Frame 8 has pristine felt where the cue ball began. This replacement
+    # leaves the table, rails, pockets, and black ball exactly at frame zero.
+    xlo, ylo, xhi, yhi = 78*5, 85*5, 95*5, 105*5
+    rect = (xlo, ylo, xhi, yhi)
+    anchor = (85.8, 94.0)
+    raw_delta = (first[ylo:yhi, xlo:xhi]
+                 - full[8, ylo:yhi, xlo:xhi].astype(np.float32))
+    # Source-derived matte excludes remote codec residuals, retaining the
+    # full sphere, antialiased silhouette, and soft shadow on the cloth.
+    yy, xx = np.mgrid[ylo:yhi, xlo:xhi]
+    bx, by = 5*anchor[0], 5*anchor[1]
+    body = ((xx-bx)/24)**2 + ((yy-by)/22)**2
+    shadow = ((xx-bx)/33)**2 + ((yy-(by+21))/17)**2
+    matte = gaussian_filter(((body<1.45) | (shadow<1.6)).astype(float), 2.5)
+    delta = raw_delta * matte[..., None]
+    clean = first.copy()
+    clean[ylo:yhi, xlo:xhi] -= delta
+    v0, a0 = friction_fit(full, clean, delta, rect, anchor)
+    acceleration = 10*a0
+    stopping_time = v0 / acceleration
+    stopping_distance = v0*v0/(2*acceleration)
+
+    # Collision would begin at q~=0.24 on this shot. The counterfactual
+    # reaches less than that, so no impulse is transmitted to the black ball.
+    if stopping_distance >= .237:
+        raise RuntimeError("Friction fit no longer predicts a no-contact edit")
+    print("10x friction: stop time / normalized distance:",
+          stopping_time, stopping_distance)
+
+    frames = []
+    for i in range(96):
+        if i == 0:
+            frames.append(source[0].copy())
+            continue
+        t = min(i/24., stopping_time)
+        q = v0*t - .5*acceleration*t*t
+        x = (anchor[0] + 284.2*q)/(1+q)
+        y = (anchor[1] - 20.8*q)/(1+q)
+        scale = 1/(1+q)
+        transform = np.array([
+            [scale, 0, 5*x-scale*(5*anchor[0]-xlo)],
+            [0, scale, 5*y-scale*(5*anchor[1]-ylo)]], np.float32)
+        moved = cv2.warpAffine(delta, transform, (1280, 720),
+                               flags=cv2.INTER_LINEAR,
+                               borderMode=cv2.BORDER_CONSTANT)
+        rgb = np.clip(np.rint(clean + moved), 0, 255).astype(np.uint8)
+        # Same reconstruction filter that produced source_pixels.npz.
+        frame = np.asarray(Image.fromarray(rgb).resize(
+            (256, 144), Image.Resampling.LANCZOS))
+        frames.append(frame)
+    frames = np.asarray(frames, dtype=np.uint8)
+    artifact_io.save(frames, ROOT)
+    print("Saved", frames.shape, "RGB frames at 24 fps.")
+    print("Final projected cue center:", x, y,
+          "; black ball remains at its factual initial center.")
+
+
+if __name__ == "__main__":
+    generate()
