+"""Counterfactual fivefold marble mass, using source-only pixel mattes and
+an impulse model calibrated to the observed collision. No appearance change
+is made to either sphere when its inertial mass is changed.
+"""
+from pathlib import Path
+import json
+import numpy as np
+from scipy.ndimage import label, map_coordinates
+from scipy.optimize import brentq
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+
+def components(mask, minimum):
+    labs, count = label(mask)
+    candidates = []
+    for k in range(1, count + 1):
+        yy, xx = np.where(labs == k)
+        if len(xx) >= minimum:
+            candidates.append((len(xx), float(xx.mean()), float(yy.mean())))
+    return sorted(candidates, reverse=True)
+
+
+def measure_tracks(source):
+    yy, xx = np.indices(source.shape[1:3])
+    coarse = np.median(source, axis=0)
+    # The final resting places must not be mistaken for static scenery.
+    coarse[57:96, :110] = np.median(source[:12, 57:96, :110], axis=0)
+    tracks = [[], []]
+    for frame in source:
+        r, g, b = np.moveaxis(frame, -1, 0)
+        difference = np.max(abs(frame - coarse), axis=-1)
+        gold = (r > b*1.6) & (g > b*1.6) & (g > 70) & (yy < 79) & (difference > 20)
+        teal = (b > r*.8) & (g > 45) & (yy > 66) & (yy < 81) & (difference > 20)
+        for track, mask, minimum in zip(tracks, [gold, teal], [100, 30]):
+            c = components(mask, minimum)
+            track.append(c[0][1:] if c else track[-1])
+    tracks = [np.array(track) for track in tracks]
+    return tracks
+
+
+def plate(source, tracks):
+    yy, xx = np.indices(source.shape[1:3])
+    clean = source.copy()
+    for t in range(len(source)):
+        excluded = np.zeros(yy.shape, bool)
+        for kind, tr in enumerate(tracks):
+            x, y = tr[t]
+            # Mask all direct sphere pixels, soft shadows and table reflections.
+            width = 15 if kind == 0 else 11
+            excluded |= (abs(xx-x) < width) & (yy > y-12) & (yy < y+25)
+        clean[t, excluded] = np.nan
+    bg = np.nanmedian(clean, axis=0)
+    return np.where(np.isnan(bg), np.median(source, axis=0), bg)
+
+
+def trajectory(tracks):
+    big, small = tracks
+    # Recover the contact instant by extrapolating the initially stationary
+    # marble's early, nearly uniformly decelerated outgoing trajectory.
+    small_fit = np.polyfit(np.arange(15, 20), small[15:20, 0], 2)
+    contact = float(min(np.roots(small_fit - [0, 0, small[0, 0]])))
+    incoming_fit = np.polyfit(np.arange(7, 14), big[7:14, 0], 2)
+    outgoing_fit = np.polyfit(np.arange(15, 20), big[15:20, 0], 2)
+    u = float(np.polyval(np.polyder(incoming_fit), contact))
+    vb = float(np.polyval(np.polyder(outgoing_fit), contact))
+    vs = float(np.polyval(np.polyder(small_fit), contact))
+    ratio = (u-vb)/vs
+    restitution = (vs-vb)/u
+    new_ratio = 5*ratio
+    new_big_v = u*(1-restitution*new_ratio)/(1+new_ratio)
+    new_small_v = u*(1+restitution)/(1+new_ratio)
+    xb = float(np.polyval(outgoing_fit, contact))
+    xs = float(small[0, 0])
+
+    # The source directly measures speed-dependent rolling/sliding losses.
+    # Smooth outgoing fits supply that same mass-independent resistance at
+    # the new post-impact speeds, rather than slowing the whole source movie.
+    big_loss = np.polyfit(np.arange(15, 33), big[15:33, 0], 2)
+    small_loss = np.polyfit(np.arange(19, 40), small[19:40, 0], 3)
+    paths = []
+    for fit, velocity, x0 in [(big_loss, new_big_v, xb),
+                             (small_loss, new_small_v, xs)]:
+        velocity_fit = np.polyder(fit)
+        roots = np.roots(velocity_fit)
+        stop = min(float(r.real) for r in roots if abs(r.imag) < 1e-5 and r.real > 20)
+        start = brentq(lambda t: np.polyval(velocity_fit, t)-velocity, 16, stop)
+        tau = np.clip(np.arange(96)-contact, 0, stop-start)
+        x = x0+np.polyval(fit, start+tau)-np.polyval(fit, start)
+        paths.append(x)
+    report = dict(contact_frame_zero_based=contact, original_mass_ratio=ratio,
+                  edited_mass_ratio=new_ratio, restitution=restitution,
+                  incoming_speed=u, original_outgoing_speeds=[vb, vs],
+                  edited_outgoing_speeds=[new_big_v, new_small_v],
+                  final_centers=[float(p[-1]) for p in paths])
+    return paths, contact, report
+
+
+def smoothstep(v):
+    v = np.clip(v, 0, 1)
+    return v*v*(3-2*v)
+
+
+def texture(source, bg, track, kind, x):
+    # Choose the source pose at the edited spatial position; this preserves
+    # perspective, rolling highlights and the material's actual RGB texture.
+    t = int(np.argmin(abs(track[:40, 0]-x)))
+    cx, cy = track[t]
+    cx += .20 if kind == 0 else .05
+    cy += .20 if kind == 0 else .00
+    dy_slope = -.0480 if kind == 0 else -.0269
+    target_y = cy + dy_slope*(x-(cx-(.20 if kind == 0 else .05)))
+    dx, dy = x-(cx-(.20 if kind == 0 else .05)), target_y-cy
+    yy, xx = np.indices(bg.shape[:2], dtype=float)
+    if kind == 0:
+        rx = 9.1 - .005*(cx-130)
+        ry = 8.5 - .007*(cx-130)
+    else:
+        rx = 5.05 + .006*(131-cx)
+        ry = 4.75 + .0035*(131-cx)
+    # Soft analytic coverage approximates the downsampled spherical boundary.
+    radial = np.sqrt(((xx-cx)/rx)**2+((yy-cy)/ry)**2)
+    alpha = smoothstep((1.075-radial)*min(rx,ry)/1.2)
+    # Treat transparent reflections/shadows as irradiance deltas, while the
+    # sphere is an opaque foreground. This keeps poles correctly occluded.
+    reflected_radius = np.sqrt(((xx-cx)/(12 if kind == 0 else 8.5))**2 +
+                               ((yy-(cy+(13.5 if kind == 0 else 12)))/(8.5 if kind == 0 else 10.5))**2)
+    reflect_mask = smoothstep((1.1-reflected_radius)*5)
+    contact_mask = np.exp(-((xx-cx)/(12 if kind == 0 else 8))**4 -
+                          ((yy-(cy+ry))/(2.8))**4)
+    shadow = np.maximum(reflect_mask, contact_mask)*(1-alpha)
+    delta = source[t]-bg
+    # Tiny codec variations are not foreground and should not migrate.
+    strength = np.max(abs(delta), axis=-1)
+    shadow *= smoothstep((strength-3)/7)
+    premult = source[t]-(1-alpha[...,None])*bg
+    premult *= (alpha > 0)[...,None]
+    # Bound residual edge values to physically valid premultiplied colors.
+    premult = np.clip(premult, 0, 255*alpha[...,None])
+    shadow_delta = delta*shadow[...,None]
+    coords = np.array([yy-dy, xx-dx])
+    warp = lambda a: map_coordinates(a, coords, order=1, mode='constant', cval=0, prefilter=False)
+    a = warp(alpha)
+    p = np.stack([warp(premult[...,c]) for c in range(3)],axis=-1)
+    sh = np.stack([warp(shadow_delta[...,c]) for c in range(3)],axis=-1)
+    return a, p, sh
+
+
+def main():
+    source = np.load(ROOT/'source_pixels.npz')['rgb'].astype(np.float32)
+    spec = json.loads((ROOT/'source_info.json').read_text())
+    assert spec['execution_frame_1based'] == 1
+    tracks = measure_tracks(source)
+    bg = plate(source, tracks)
+    paths, contact, report = trajectory(tracks)
+    frames = source.astype(np.uint8).copy()
+    yy, xx = np.indices(bg.shape[:2])
+    affected_region = (yy >= 53) & (yy < 101) & (xx < 238)
+    for t in range(int(np.floor(contact))+1, 96):
+        f = source[t].copy()
+        f[affected_region] = bg[affected_region]
+        layers = [texture(source, bg, tracks[k], k, paths[k][t]) for k in range(2)]
+        for a,p,s in layers:
+            f += s
+        # Small marble is in the nearer depth lane.
+        for a,p,s in layers:
+            f = f*(1-a[...,None])+p
+        frames[t] = np.clip(np.rint(f),0,255).astype(np.uint8)
+    assert frames.shape == (96,144,256,3)
+    artifact_io.save(frames, ROOT)
+    (ROOT/'generation_notes.json').write_text(json.dumps(report,indent=2))
+    print(json.dumps(report,indent=2))
+
+
+if __name__ == '__main__':
+    main()

\"\"\"Counterfactual fivefold marble mass, using source-only pixel mattes and
an impulse model calibrated to the observed collision. No appearance change
is made to either sphere when its inertial mass is changed.
\"\"\"
from pathlib import Path
import json
import numpy as np
from scipy.ndimage import label, map_coordinates
from scipy.optimize import brentq
import artifact_io

ROOT = Path(__file__).resolve().parent


def components(mask, minimum):
    labs, count = label(mask)
    candidates = []
    for k in range(1, count + 1):
        yy, xx = np.where(labs == k)
        if len(xx) >= minimum:
            candidates.append((len(xx), float(xx.mean()), float(yy.mean())))
    return sorted(candidates, reverse=True)


def measure_tracks(source):
    yy, xx = np.indices(source.shape[1:3])
    coarse = np.median(source, axis=0)
    # The final resting places must not be mistaken for static scenery.
    coarse[57:96, :110] = np.median(source[:12, 57:96, :110], axis=0)
    tracks = [[], []]
    for frame in source:
        r, g, b = np.moveaxis(frame, -1, 0)
        difference = np.max(abs(frame - coarse), axis=-1)
        gold = (r > b*1.6) & (g > b*1.6) & (g > 70) & (yy < 79) & (difference > 20)
        teal = (b > r*.8) & (g > 45) & (yy > 66) & (yy < 81) & (difference > 20)
        for track, mask, minimum in zip(tracks, [gold, teal], [100, 30]):
            c = components(mask, minimum)
            track.append(c[0][1:] if c else track[-1])
    tracks = [np.array(track) for track in tracks]
    return tracks


def plate(source, tracks):
    yy, xx = np.indices(source.shape[1:3])
    clean = source.copy()
    for t in range(len(source)):
        excluded = np.zeros(yy.shape, bool)
        for kind, tr in enumerate(tracks):
            x, y = tr[t]
            # Mask all direct sphere pixels, soft shadows and table reflections.
            width = 15 if kind == 0 else 11
            excluded |= (abs(xx-x) < width) & (yy > y-12) & (yy < y+25)
        clean[t, excluded] = np.nan
    bg = np.nanmedian(clean, axis=0)
    return np.where(np.isnan(bg), np.median(source, axis=0), bg)


def trajectory(tracks):
    big, small = tracks
    # Recover the contact instant by extrapolating the initially stationary
    # marble's early, nearly uniformly decelerated outgoing trajectory.
    small_fit = np.polyfit(np.arange(15, 20), small[15:20, 0], 2)
    contact = float(min(np.roots(small_fit - [0, 0, small[0, 0]])))
    incoming_fit = np.polyfit(np.arange(7, 14), big[7:14, 0], 2)
    outgoing_fit = np.polyfit(np.arange(15, 20), big[15:20, 0], 2)
    u = float(np.polyval(np.polyder(incoming_fit), contact))
    vb = float(np.polyval(np.polyder(outgoing_fit), contact))
    vs = float(np.polyval(np.polyder(small_fit), contact))
    ratio = (u-vb)/vs
    restitution = (vs-vb)/u
    new_ratio = 5*ratio
    new_big_v = u*(1-restitution*new_ratio)/(1+new_ratio)
    new_small_v = u*(1+restitution)/(1+new_ratio)
    xb = float(np.polyval(outgoing_fit, contact))
    xs = float(small[0, 0])

    # The source directly measures speed-dependent rolling/sliding losses.
    # Smooth outgoing fits supply that same mass-independent resistance at
    # the new post-impact speeds, rather than slowing the whole source movie.
    big_loss = np.polyfit(np.arange(15, 33), big[15:33, 0], 2)
    small_loss = np.polyfit(np.arange(19, 40), small[19:40, 0], 3)
    paths = []
    for fit, velocity, x0 in [(big_loss, new_big_v, xb),
                             (small_loss, new_small_v, xs)]:
        velocity_fit = np.polyder(fit)
        roots = np.roots(velocity_fit)
        stop = min(float(r.real) for r in roots if abs(r.imag) < 1e-5 and r.real > 20)
        start = brentq(lambda t: np.polyval(velocity_fit, t)-velocity, 16, stop)
        tau = np.clip(np.arange(96)-contact, 0, stop-start)
        x = x0+np.polyval(fit, start+tau)-np.polyval(fit, start)
        paths.append(x)
    report = dict(contact_frame_zero_based=contact, original_mass_ratio=ratio,
                  edited_mass_ratio=new_ratio, restitution=restitution,
                  incoming_speed=u, original_outgoing_speeds=[vb, vs],
                  edited_outgoing_speeds=[new_big_v, new_small_v],
                  final_centers=[float(p[-1]) for p in paths])
    return paths, contact, report


def smoothstep(v):
    v = np.clip(v, 0, 1)
    return v*v*(3-2*v)


def texture(source, bg, track, kind, x):
    # Choose the source pose at the edited spatial position; this preserves
    # perspective, rolling highlights and the material's actual RGB texture.
    t = int(np.argmin(abs(track[:40, 0]-x)))
    cx, cy = track[t]
    cx += .20 if kind == 0 else .05
    cy += .20 if kind == 0 else .00
    dy_slope = -.0480 if kind == 0 else -.0269
    target_y = cy + dy_slope*(x-(cx-(.20 if kind == 0 else .05)))
    dx, dy = x-(cx-(.20 if kind == 0 else .05)), target_y-cy
    yy, xx = np.indices(bg.shape[:2], dtype=float)
    if kind == 0:
        rx = 9.1 - .005*(cx-130)
        ry = 8.5 - .007*(cx-130)
    else:
        rx = 5.05 + .006*(131-cx)
        ry = 4.75 + .0035*(131-cx)
    # Soft analytic coverage approximates the downsampled spherical boundary.
    radial = np.sqrt(((xx-cx)/rx)**2+((yy-cy)/ry)**2)
    alpha = smoothstep((1.075-radial)*min(rx,ry)/1.2)
    # Treat transparent reflections/shadows as irradiance deltas, while the
    # sphere is an opaque foreground. This keeps poles correctly occluded.
    reflected_radius = np.sqrt(((xx-cx)/(12 if kind == 0 else 8.5))**2 +
                               ((yy-(cy+(13.5 if kind == 0 else 12)))/(8.5 if kind == 0 else 10.5))**2)
    reflect_mask = smoothstep((1.1-reflected_radius)*5)
    contact_mask = np.exp(-((xx-cx)/(12 if kind == 0 else 8))**4 -
                          ((yy-(cy+ry))/(2.8))**4)
    shadow = np.maximum(reflect_mask, contact_mask)*(1-alpha)
    delta = source[t]-bg
    # Tiny codec variations are not foreground and should not migrate.
    strength = np.max(abs(delta), axis=-1)
    shadow *= smoothstep((strength-3)/7)
    premult = source[t]-(1-alpha[...,None])*bg
    premult *= (alpha > 0)[...,None]
    # Bound residual edge values to physically valid premultiplied colors.
    premult = np.clip(premult, 0, 255*alpha[...,None])
    shadow_delta = delta*shadow[...,None]
    coords = np.array([yy-dy, xx-dx])
    warp = lambda a: map_coordinates(a, coords, order=1, mode='constant', cval=0, prefilter=False)
    a = warp(alpha)
    p = np.stack([warp(premult[...,c]) for c in range(3)],axis=-1)
    sh = np.stack([warp(shadow_delta[...,c]) for c in range(3)],axis=-1)
    return a, p, sh


def main():
    source = np.load(ROOT/'source_pixels.npz')['rgb'].astype(np.float32)
    spec = json.loads((ROOT/'source_info.json').read_text())
    assert spec['execution_frame_1based'] == 1
    tracks = measure_tracks(source)
    bg = plate(source, tracks)
    paths, contact, report = trajectory(tracks)
    frames = source.astype(np.uint8).copy()
    yy, xx = np.indices(bg.shape[:2])
    affected_region = (yy >= 53) & (yy < 101) & (xx < 238)
    for t in range(int(np.floor(contact))+1, 96):
        f = source[t].copy()
        f[affected_region] = bg[affected_region]
        layers = [texture(source, bg, tracks[k], k, paths[k][t]) for k in range(2)]
        for a,p,s in layers:
            f += s
        # Small marble is in the nearer depth lane.
        for a,p,s in layers:
            f = f*(1-a[...,None])+p
        frames[t] = np.clip(np.rint(f),0,255).astype(np.uint8)
    assert frames.shape == (96,144,256,3)
    artifact_io.save(frames, ROOT)
    (ROOT/'generation_notes.json').write_text(json.dumps(report,indent=2))
    print(json.dumps(report,indent=2))


if __name__ == '__main__':
    main()
