+"""Remove the block at source frame 17 using only source RGB measurements.
+
+The collision has already happened when the intervention is made.  The sphere
+is separated from the block, so its acquired rolling velocity is retained.
+A temporally recovered clean plate removes the block and its lighting footprint;
+the original moving sphere is composited without changing its trajectory.
+"""
+from pathlib import Path
+import json
+import warnings
+import numpy as np
+from PIL import Image, ImageDraw
+from scipy.ndimage import (label, binary_dilation, gaussian_filter,
+                           distance_transform_edt, map_coordinates)
+from scipy.spatial import ConvexHull
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+Y, X = np.mgrid[:144, :256]
+
+
+def components(mask):
+    lab, _ = label(mask)
+    counts = np.bincount(lab.ravel())
+    counts[0] = 0
+    return lab == counts.argmax()
+
+
+def analyze(source):
+    """Find the sphere by its red chroma and block by top/front RGB faces."""
+    balls = []
+    blocks = []
+    valid = np.ones(source.shape[:3], bool)
+    for i, f in enumerate(source):
+        r, g, b = f.transpose(2, 0, 1)
+        red = components((r-g > 65) & (r > 130))
+        yy, xx = np.where(red)
+        cx = (xx.min()+xx.max())/2
+        cy = (yy.min()+yy.max())/2 + 1.25
+        rx = (xx.max()-xx.min())/2 + .8
+        ry = (yy.max()-yy.min())/2 + 2.0
+        balls.append((cx, cy, rx, ry))
+        # Color threshold omits the almost-black bottom of the sphere.
+        # This expanded oval excludes those pixels during plate reconstruction.
+        valid[i] &= (((X-cx)/(rx+1.5))**2 +
+                     ((Y-cy)/(ry+1.0))**2 > 1.06)
+        top = ((b/np.maximum(r, 1) > .59) &
+               (g/np.maximum(r, 1) > .72) & (r < 138) &
+               (X > 105) & (X < 250) & (Y > 50) & (Y < 78))
+        top = components(top)
+        yy, xx = np.where(top)
+        left, right, high = xx.min(), xx.max(), yy.min()
+        if i <= 12:
+            bounds = (108, 167, 54, 104)
+        elif i >= 35:
+            bounds = (180, 248, 65, 117)
+        else:
+            bounds = (left-3, right+6, high-2, 108+(i-16)*.55)
+        left, right, high, low = bounds
+        front = ((r < 107) & (g < 73) & (b < 62) &
+                 (X >= left) & (X <= right) & (Y >= high) & (Y < low+1))
+        front &= (((X-cx)/(rx+1))**2 + ((Y-cy)/(ry+1))**2 > 1)
+        points = np.array(np.where((top | front) & (X >= left) &
+                                  (X <= right))).T[:, ::-1]
+        hull = ConvexHull(points)
+        im = Image.new('L', (256, 144))
+        ImageDraw.Draw(im).polygon([tuple(p) for p in points[hull.vertices]], fill=255)
+        mask = binary_dilation(np.array(im) > 0, iterations=3)
+        blocks.append(mask)
+        valid[i] &= ~mask
+    return np.array(balls), np.array(blocks), valid
+
+
+def clean_plate(source, valid):
+    """Upper clear-floor samples reject both moving shadows and dark objects.
+
+    Average genuine RGB triples close to the brightest valid green sample;
+    green is used because it is low on the red sphere, unlike red itself.
+    """
+    green = source[:, :, :, 1]
+    score = np.where(valid, green, -999)
+    maximum = score.max(axis=0)
+    weights = valid & (green >= maximum[None]-2.5)
+    plate = (source*weights[..., None]).sum(0) / np.maximum(weights.sum(0)[..., None], 1)
+    missing = weights.sum(0) == 0
+    if missing.any():
+        index = distance_transform_edt(missing, return_distances=False, return_indices=True)
+        plate[missing] = plate[index[0][missing], index[1][missing]]
+    # The floor to the right of the block's initial footprint is already
+    # visible in frame 1. Keep that exact original texture and wall junction.
+    base = source[0].copy()
+    patch = np.zeros((144, 256), float)
+    patch[51:, 99:191] = 1
+    patch = gaussian_filter(patch, 2.0)
+    patch[54:135, 107:178] = 1
+    base = base*(1-patch[..., None]) + plate*patch[..., None]
+    return base
+
+
+def ball_shadow_template(source, ball):
+    """Transmission of the ball's shadow on unobstructed floor, from frame 5.
+
+    Later source pixels reveal the very same wood grain, making the ratio a
+    lighting-only layer rather than a pasted patch of floor texture.
+    """
+    f = source[4]
+    empty = source[95]
+    transmission = np.clip((f+1)/(empty+1), .13, 1)
+    transmission = gaussian_filter(transmission, (.55, .55, 0))
+    cx, cy, rx, ry = ball[4]
+    return transmission, (cx, cy, rx, ry)
+
+
+def generate():
+    source = np.load(ROOT/'source_pixels.npz')['rgb'].astype(np.float64)
+    balls, blocks, valid = analyze(source)
+    plate = clean_plate(source, valid)
+    transmission, reference = ball_shadow_template(source, balls)
+    refx, refy, refrx, refry = reference
+    result = source.astype(np.uint8).copy()
+    intervention = json.loads((ROOT/'source_info.json').read_text())['execution_frame_1based']-1
+
+    # Limit the edit to the part of the scene traversed by these two objects.
+    region = np.zeros((144, 256), float)
+    region[48:, 76:] = 1
+    region = gaussian_filter(region, 1.3)
+    region[52:, 80:] = 1
+    for i in range(intervention, len(source)):
+        cx, cy, rx, ry = balls[i]
+        # Match the original sphere's cast/contact shadow in projected units.
+        sx = refx + (X-cx)*(refrx/rx)
+        sy = refy + (Y-cy)*(refry/ry)
+        shadow = np.stack([map_coordinates(transmission[:, :, c], [sy, sx],
+                                          order=1, mode='constant', cval=1)
+                           for c in range(3)], axis=-1)
+        # Only use the shadow below the sphere, never the source sphere's
+        # body or any distant source features accidentally sampled by a warp.
+        footprint = np.exp(-((X-cx-.12*rx)/(1.75*rx))**8)
+        footprint *= np.clip((Y-(cy+.65*ry))/(.25*ry), 0, 1)
+        footprint *= np.clip(((cy+4.15*ry)-Y)/(.45*ry), 0, 1)
+        shadow = 1 + (shadow-1)*footprint[..., None]
+        background = plate*shadow
+        frame = source[i]*(1-region[..., None]) + background*region[..., None]
+        # Keep all original ball surface pixels, including dark grooves and
+        # the bottom rim; the ellipse provides a subpixel, antialiased border.
+        radius = np.sqrt(((X-cx)/rx)**2 + ((Y-cy)/ry)**2)
+        keep = np.clip((1.035-radius)*min(rx, ry)/.8, 0, 1)
+        # The brightest red edge sometimes lies just beyond the fitted oval.
+        r, g, b = source[i].transpose(2, 0, 1)
+        red_edge = ((r-g > 65) & (r > 130) & (radius < 1.16))
+        keep[red_edge] = 1
+        frame = frame*(1-keep[..., None]) + source[i]*keep[..., None]
+        result[i] = np.clip(np.rint(frame), 0, 255).astype(np.uint8)
+
+    assert np.array_equal(result[:intervention], source[:intervention].astype(np.uint8))
+    artifact_io.save(result, ROOT)
+    # Additional normal MP4 deliverable, generated from the same submitted RGB.
+    try:
+        import subprocess
+        proc = subprocess.Popen([
+            'ffmpeg', '-y', '-loglevel', 'error', '-f', 'rawvideo',
+            '-pixel_format', 'rgb24', '-video_size', '256x144', '-framerate', '24',
+            '-i', 'pipe:0', '-an', '-c:v', 'libx264', '-crf', '16',
+            '-pix_fmt', 'yuv420p', '-movflags', '+faststart', str(ROOT/'edited_video.mp4')
+        ], stdin=subprocess.PIPE)
+        proc.communicate(result.tobytes())
+        if proc.returncode:
+            raise RuntimeError('MP4 encoding failed')
+    except FileNotFoundError:
+        pass
+    print('Saved 96 RGB frames at 256x144, 24 fps. Frames 1â16 are exact source RGB.')
+
+
+if __name__ == '__main__':
+    generate()

def components(mask):
    lab, _ = label(mask)
    counts = np.bincount(lab.ravel())
    counts[0] = 0
    return lab == counts.argmax()


def analyze(source):
    \"\"\"Find the sphere by its red chroma and block by top/front RGB faces.\"\"\"
    balls = []
    blocks = []
    valid = np.ones(source.shape[:3], bool)
    for i, f in enumerate(source):
        r, g, b = f.transpose(2, 0, 1)
        red = components((r-g > 65) & (r > 130))
        yy, xx = np.where(red)
        cx = (xx.min()+xx.max())/2
        cy = (yy.min()+yy.max())/2 + 1.25
        rx = (xx.max()-xx.min())/2 + .8
        ry = (yy.max()-yy.min())/2 + 2.0
        balls.append((cx, cy, rx, ry))
        # Color threshold omits the almost-black bottom of the sphere.
        # This expanded oval excludes those pixels during plate reconstruction.
        valid[i] &= (((X-cx)/(rx+1.5))**2 +
                     ((Y-cy)/(ry+1.0))**2 > 1.06)
        top = ((b/np.maximum(r, 1) > .59) &
               (g/np.maximum(r, 1) > .72) & (r < 138) &
               (X > 105) & (X < 250) & (Y > 50) & (Y < 78))
        top = components(top)
        yy, xx = np.where(top)
        left, right, high = xx.min(), xx.max(), yy.min()
        if i <= 12:
            bounds = (108, 167, 54, 104)
        elif i >= 35:
            bounds = (180, 248, 65, 117)
        else:
            bounds = (left-3, right+6, high-2, 108+(i-16)*.55)
        left, right, high, low = bounds
        front = ((r < 107) & (g < 73) & (b < 62) &
                 (X >= left) & (X <= right) & (Y >= high) & (Y < low+1))
        front &= (((X-cx)/(rx+1))**2 + ((Y-cy)/(ry+1))**2 > 1)
        points = np.array(np.where((top | front) & (X >= left) &
                                  (X <= right))).T[:, ::-1]
        hull = ConvexHull(points)
        im = Image.new('L', (256, 144))
        ImageDraw.Draw(im).polygon([tuple(p) for p in points[hull.vertices]], fill=255)
        mask = binary_dilation(np.array(im) > 0, iterations=3)
        blocks.append(mask)
        valid[i] &= ~mask
    return np.array(balls), np.array(blocks), valid


def clean_plate(source, valid):
    \"\"\"Upper clear-floor samples reject both moving shadows and dark objects.

    Average genuine RGB triples close to the brightest valid green sample;
    green is used because it is low on the red sphere, unlike red itself.
    \"\"\"
    green = source[:, :, :, 1]
    score = np.where(valid, green, -999)
    maximum = score.max(axis=0)
    weights = valid & (green >= maximum[None]-2.5)
    plate = (source*weights[..., None]).sum(0) / np.maximum(weights.sum(0)[..., None], 1)
    missing = weights.sum(0) == 0
    if missing.any():
        index = distance_transform_edt(missing, return_distances=False, return_indices=True)
        plate[missing] = plate[index[0][missing], index[1][missing]]
    # The floor to the right of the block's initial footprint is already
    # visible in frame 1. Keep that exact original texture and wall junction.
    base = source[0].copy()
    patch = np.zeros((144, 256), float)
    patch[51:, 99:191] = 1
    patch = gaussian_filter(patch, 2.0)
    patch[54:135, 107:178] = 1
    base = base*(1-patch[..., None]) + plate*patch[..., None]
    return base


def ball_shadow_template(source, ball):
    \"\"\"Transmission of the ball's shadow on unobstructed floor, from frame 5.

    Later source pixels reveal the very same wood grain, making the ratio a
    lighting-only layer rather than a pasted patch of floor texture.
    \"\"\"
    f = source[4]
    empty = source[95]
    transmission = np.clip((f+1)/(empty+1), .13, 1)
    transmission = gaussian_filter(transmission, (.55, .55, 0))
    cx, cy, rx, ry = ball[4]
    return transmission, (cx, cy, rx, ry)


def generate():

    balls, blocks, valid = analyze(source)
    plate = clean_plate(source, valid)
    transmission, reference = ball_shadow_template(source, balls)
    refx, refy, refrx, refry = reference
    result = source.astype(np.uint8).copy()
    intervention = json.loads((ROOT/'source_info.json').read_text())['execution_frame_1based']-1

    # Limit the edit to the part of the scene traversed by these two objects.
    region = np.zeros((144, 256), float)
    region[48:, 76:] = 1
    region = gaussian_filter(region, 1.3)
    region[52:, 80:] = 1
    for i in range(intervention, len(source)):
        cx, cy, rx, ry = balls[i]
        # Match the original sphere's cast/contact shadow in projected units.
        sx = refx + (X-cx)*(refrx/rx)
        sy = refy + (Y-cy)*(refry/ry)
        shadow = np.stack([map_coordinates(transmission[:, :, c], [sy, sx],
                                          order=1, mode='constant', cval=1)
                           for c in range(3)], axis=-1)
        # Only use the shadow below the sphere, never the source sphere's
        # body or any distant source features accidentally sampled by a warp.
        footprint = np.exp(-((X-cx-.12*rx)/(1.75*rx))**8)
        footprint *= np.clip((Y-(cy+.65*ry))/(.25*ry), 0, 1)
        footprint *= np.clip(((cy+4.15*ry)-Y)/(.45*ry), 0, 1)
        shadow = 1 + (shadow-1)*footprint[..., None]
        background = plate*shadow
        frame = source[i]*(1-region[..., None]) + background*region[..., None]
        # Keep all original ball surface pixels, including dark grooves and
        # the bottom rim; the ellipse provides a subpixel, antialiased border.
        radius = np.sqrt(((X-cx)/rx)**2 + ((Y-cy)/ry)**2)
        keep = np.clip((1.035-radius)*min(rx, ry)/.8, 0, 1)
        # The brightest red edge sometimes lies just beyond the fitted oval.
        r, g, b = source[i].transpose(2, 0, 1)
        red_edge = ((r-g > 65) & (r > 130) & (radius < 1.16))
        keep[red_edge] = 1
        frame = frame*(1-keep[..., None]) + source[i]*keep[..., None]
        result[i] = np.clip(np.rint(frame), 0, 255).astype(np.uint8)

    assert np.array_equal(result[:intervention], source[:intervention].astype(np.uint8))
    artifact_io.save(result, ROOT)

            '-pix_fmt', 'yuv420p', '-movflags', '+faststart', str(ROOT/'edited_video.mp4')
        ], stdin=subprocess.PIPE)
        proc.communicate(result.tobytes())
        if proc.returncode:
            raise RuntimeError('MP4 encoding failed')
    except FileNotFoundError:
        pass
    print('Saved 96 RGB frames at 256x144, 24 fps. Frames 1â16 are exact source RGB.')


if __name__ == '__main__':
    generate()
