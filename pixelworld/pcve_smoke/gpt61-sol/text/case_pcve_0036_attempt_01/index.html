+No external renderer: objects and their specular reflections are source-pixel
+sprites. Floor deceleration is reduced by precisely the requested factor.
+"""

+import math
+import cv2
+import numpy as np
+from PIL import Image
+from scipy import ndimage
+import artifact_io

+ROOT = Path(__file__).resolve().parent
+FRICTION_RATIO = 0.125

+def read_original():
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    frames = []
+    while True:
+        ok, bgr = cap.read()
+        if not ok:
+            break
+        frames.append(bgr[:, :, ::-1].copy())
+    cap.release()
+    return np.stack(frames)

+def feather_mask(mask, sigma=0.5):
+    return ndimage.gaussian_filter(mask.astype(np.float32), sigma)
+
+
+def sprite(source, bg, box, center, reflection=False, exclude_blue=True):
+    x0, y0, x1, y1 = box
+    rgb = source[y0:y1, x0:x1].astype(np.float32)
+    floor = bg[y0:y1, x0:x1].astype(np.float32)
+    delta = rgb - floor
+    difference = np.max(np.abs(delta), axis=2)
+    if reflection:
+        # Reflected pin light is added to the existing polished surface.
+        light = np.maximum(delta, 0)
+        alpha = np.clip((light.max(2) - 8) / 20, 0, 1)
+        light *= alpha[:, :, None]
+        return (light, None, (x0 - center[0], y0 - center[1]))
+    mask = (difference > 18) & (rgb.mean(2) > 26)
+    if exclude_blue:
+        blue = (rgb[:, :, 2] > rgb[:, :, 0] * 1.28) & (rgb[:, :, 2] > rgb[:, :, 1] * 1.13)
+        mask &= ~blue
+    labs, n = ndimage.label(mask)
+    sizes = np.bincount(labs.ravel())
+    mask &= sizes[labs] >= 5
+    mask = ndimage.binary_fill_holes(mask)
+    # The source silhouette already includes antialiasing. A narrow coverage
+    # transition suppresses the very low-level codec noise around each pin.
+    alpha = feather_mask(mask, 0.38)
+    return (rgb * alpha[:, :, None], alpha, (x0 - center[0], y0 - center[1]))
+
+
+def place(image, spr, pos, matrix=None, mirror=False, additive=False):
+    rgb, alpha, offset = spr
+    if matrix is None:
+        matrix = np.eye(2, dtype=np.float32)
+    else:
+        matrix = np.asarray(matrix, dtype=np.float32)
+    if mirror:
+        matrix = matrix @ np.diag([-1., 1.]).astype(np.float32)
+    off = np.asarray(offset, dtype=np.float32)
+    h, w = rgb.shape[:2]
+    corners = np.array([[0, 0], [w, 0], [0, h], [w, h]], np.float32)
+    corners = (corners + off) @ matrix.T + pos
+    x0, y0 = np.floor(corners.min(0) - 2).astype(int)
+    x1, y1 = np.ceil(corners.max(0) + 2).astype(int)
+    x0, y0 = max(0, x0), max(0, y0)
+    x1, y1 = min(image.shape[1], x1), min(image.shape[0], y1)
+    if x1 <= x0 or y1 <= y0:
+        return
+    affine = np.column_stack([matrix, np.asarray(pos) + matrix @ off - [x0, y0]]).astype(np.float32)
+    color = cv2.warpAffine(rgb, affine, (x1-x0, y1-y0), flags=cv2.INTER_LINEAR)
+    target = image[y0:y1, x0:x1]
+    if additive or alpha is None:
+        target += color
+    else:
+        coverage = cv2.warpAffine(alpha, affine, (x1-x0, y1-y0), flags=cv2.INTER_LINEAR)
+        target *= 1-coverage[:, :, None]
+        target += color
+
+
+def ground_rotation(theta):
+    # Projection of an in-plane yaw on this low, almost axial camera.
+    c, s = math.cos(theta), math.sin(theta)
+    return np.array([[c, -s / .28], [.28*s, c]], np.float32)
+

\"\"\"Counterfactual bowling edit using source RGB mattes and low-friction motion.
No external renderer: objects and their specular reflections are source-pixel
sprites. Floor deceleration is reduced by precisely the requested factor.
\"\"\"
from pathlib import Path
import math
import cv2
import numpy as np
from PIL import Image
from scipy import ndimage
import artifact_io

ROOT = Path(__file__).resolve().parent
FRICTION_RATIO = 0.125


def read_original():
    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
    frames = []
    while True:
        ok, bgr = cap.read()
        if not ok:
            break
        frames.append(bgr[:, :, ::-1].copy())
    cap.release()
    return np.stack(frames)


def feather_mask(mask, sigma=0.5):
    return ndimage.gaussian_filter(mask.astype(np.float32), sigma)


def sprite(source, bg, box, center, reflection=False, exclude_blue=True):
    x0, y0, x1, y1 = box
    rgb = source[y0:y1, x0:x1].astype(np.float32)
    floor = bg[y0:y1, x0:x1].astype(np.float32)
    delta = rgb - floor
    difference = np.max(np.abs(delta), axis=2)
    if reflection:
        # Reflected pin light is added to the existing polished surface.
        light = np.maximum(delta, 0)
        alpha = np.clip((light.max(2) - 8) / 20, 0, 1)
        light *= alpha[:, :, None]
        return (light, None, (x0 - center[0], y0 - center[1]))
    mask = (difference > 18) & (rgb.mean(2) > 26)
    if exclude_blue:
        blue = (rgb[:, :, 2] > rgb[:, :, 0] * 1.28) & (rgb[:, :, 2] > rgb[:, :, 1] * 1.13)
        mask &= ~blue
    labs, n = ndimage.label(mask)
    sizes = np.bincount(labs.ravel())
    mask &= sizes[labs] >= 5
    mask = ndimage.binary_fill_holes(mask)
    # The source silhouette already includes antialiasing. A narrow coverage
    # transition suppresses the very low-level codec noise around each pin.
    alpha = feather_mask(mask, 0.38)
    return (rgb * alpha[:, :, None], alpha, (x0 - center[0], y0 - center[1]))


def place(image, spr, pos, matrix=None, mirror=False, additive=False):
    rgb, alpha, offset = spr
    if matrix is None:
        matrix = np.eye(2, dtype=np.float32)
    else:
        matrix = np.asarray(matrix, dtype=np.float32)
    if mirror:
        matrix = matrix @ np.diag([-1., 1.]).astype(np.float32)
    off = np.asarray(offset, dtype=np.float32)
    h, w = rgb.shape[:2]
    corners = np.array([[0, 0], [w, 0], [0, h], [w, h]], np.float32)
    corners = (corners + off) @ matrix.T + pos
    x0, y0 = np.floor(corners.min(0) - 2).astype(int)
    x1, y1 = np.ceil(corners.max(0) + 2).astype(int)
    x0, y0 = max(0, x0), max(0, y0)
    x1, y1 = min(image.shape[1], x1), min(image.shape[0], y1)
    if x1 <= x0 or y1 <= y0:
        return
    affine = np.column_stack([matrix, np.asarray(pos) + matrix @ off - [x0, y0]]).astype(np.float32)
    color = cv2.warpAffine(rgb, affine, (x1-x0, y1-y0), flags=cv2.INTER_LINEAR)
    target = image[y0:y1, x0:x1]
    if additive or alpha is None:
        target += color
    else:
        coverage = cv2.warpAffine(alpha, affine, (x1-x0, y1-y0), flags=cv2.INTER_LINEAR)
        target *= 1-coverage[:, :, None]
        target += color


def ground_rotation(theta):
    # Projection of an in-plane yaw on this low, almost axial camera.
    c, s = math.cos(theta), math.sin(theta)
    return np.array([[c, -s / .28], [.28*s, c]], np.float32)
