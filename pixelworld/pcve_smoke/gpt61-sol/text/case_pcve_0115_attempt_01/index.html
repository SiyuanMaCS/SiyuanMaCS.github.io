+"""Reconstruct the source tennis-ball flight and change only its initial velocity.
+
+The perspective projection of a constant-gravity ballistic trajectory has a
+quadratic numerator and denominator. Scaling the linear terms by 1.5 changes
+velocity, not gravity, the initial position, or the camera. Source RGB pixels
+supply both the static scene and the ball's illuminated surface.
+"""
+from pathlib import Path
+import json
+import numpy as np
+import cv2
+from scipy.ndimage import label
+from scipy.optimize import least_squares
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+
+def track_ball(frames, background):
+    positions = []
+    for frame in frames:
+        a = frame.astype(np.float64)
+        delta = a - background
+        weight = np.maximum(delta[..., 1] - delta[..., 2], 0)
+        weight[(weight < 15) | (a[..., 1] < a[..., 0] * 1.04)] = 0
+        components, _ = label(weight > 0)
+        sizes = np.bincount(components.ravel())
+        sizes[0] = 0
+        mask = components == np.argmax(sizes)
+        y, x = np.where(mask)
+        positions.append([np.average(x, weights=weight[mask]),
+                          np.average(y, weights=weight[mask])])
+    return np.array(positions)
+
+
+def project(coefficients, times, speed=1.0):
+    times = np.asarray(times)
+    a, b, c = coefficients[:3], coefficients[3:6], coefficients[6:]
+    denominator = 1 + speed * c[0] * times + c[1] * times**2
+    xy = np.stack([(a[0] + speed*a[1]*times + a[2]*times**2)/denominator,
+                   (b[0] + speed*b[1]*times + b[2]*times**2)/denominator], -1)
+    return xy, denominator
+
+
+def get_texture(image, clean, center, radius):
+    """Unmatte source ball pixels to a small premultiplied RGB sprite."""
+    hi_center = center * 5 + 2
+    ox, oy = np.floor(hi_center - 16).astype(int)
+    size = 34
+    patch = image[oy:oy+size, ox:ox+size].astype(np.float64)
+    background = clean[oy:oy+size, ox:ox+size].astype(np.float64)
+    yy, xx = np.mgrid[:size, :size]
+    distance = np.hypot(xx + ox-hi_center[0], yy + oy-hi_center[1])
+    alpha = np.clip((radius + .75-distance)/1.5, 0, 1)
+    premult = patch - background*(1-alpha[..., None])
+    premult = np.minimum(np.maximum(premult, 0), 255*alpha[..., None])
+    premult[alpha == 0] = 0
+    return premult.astype(np.float32), alpha.astype(np.float32), hi_center-np.array([ox,oy])
+
+
+def composite(frame, texture, center, scale):
+    color, alpha, reference = texture
+    x, y = center
+    pad = 5 * max(scale, 1)
+    x0, x1 = max(0, int(np.floor(x-pad))), min(256, int(np.ceil(x+pad))+1)
+    y0, y1 = max(0, int(np.floor(y-pad))), min(144, int(np.ceil(y+pad))+1)
+    if x0 >= x1 or y0 >= y1:
+        return
+    # Pixel-area integration on the same 5:1 source/downsample grid.
+    yy, xx = np.mgrid[:(y1-y0)*5, :(x1-x0)*5]
+    gx = x0 + (xx+.5)/5-.5
+    gy = y0 + (yy+.5)/5-.5
+    mx = ((gx-x)*5/scale+reference[0]).astype(np.float32)
+    my = ((gy-y)*5/scale+reference[1]).astype(np.float32)
+    c = cv2.remap(color, mx, my, cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)
+    a = cv2.remap(alpha, mx, my, cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)
+    h, w = y1-y0, x1-x0
+    c = c.reshape(h,5,w,5,3).mean((1,3))
+    a = a.reshape(h,5,w,5).mean((1,3))
+    region = frame[y0:y1, x0:x1]
+    region[:] = np.clip(np.rint(c + region*(1-a[...,None])), 0, 255).astype(np.uint8)
+
+
+def main():
+    source = np.load(ROOT/'source_pixels.npz')['rgb']
+    # First 64 frames never contain a stationary ball at one location, giving
+    # an unoccluded temporal estimate even at its eventual stopping position.
+    clean = np.median(source[:64], axis=0)
+    positions = track_ball(source, clean)
+    flight_times = np.arange(28, dtype=np.float64)
+    init = [28.4, 4.3, .015, 59.1, -4.7, .22, -.0021, .00014]
+    fit = least_squares(lambda q: (project(q, flight_times)[0]-positions[:28]).ravel(),
+                        init, max_nfev=3000, xtol=1e-12, ftol=1e-12, gtol=1e-12)
+    coefficients = fit.x
+    fitted, original_den = project(coefficients, flight_times)
+    changed, changed_den = project(coefficients, np.arange(96), 1.5)
+
+    # Only source video pixels are used for surface detail. Frame 40 provides
+    # clear background in the airborne region. Median samples remove codec
+    # flicker without synthesizing any new scene element.
+    cap = cv2.VideoCapture(str(ROOT/'source_video.mp4'))
+    high = {}
+    for t in range(41):
+        ok, bgr = cap.read()
+        if not ok:
+            raise RuntimeError('Source decode failed')
+        if t <= 27 or t == 40:
+            high[t] = cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB)
+    cap.release()
+    high_clean = np.median(np.stack([high[t] for t in [0,9,18,27,40]]),axis=0)
+    # The sphere is about 10.4 source-resolution pixels in radius. Perspective
+    # magnification follows the same depth denominator as its trajectory.
+    radius_at_start = 10.4
+    textures = {}
+    for t in range(1,21):
+        textures[t] = get_texture(high[t], high_clean, fitted[t],
+                                  radius_at_start/original_den[t])
+
+    result = source.copy()
+    yy, xx = np.mgrid[:144, :256]
+    for t in range(1,96):
+        # Remove factual ball and its small contact shadow, preserving every
+        # other moving/compression detail on the source timeline.
+        old_x, old_y = positions[t]
+        distance = np.hypot(xx-old_x, yy-old_y)
+        erase = np.clip((6-distance)/1.25, 0, 1)
+        result[t] = np.rint(result[t]*(1-erase[...,None]) + clean*erase[...,None]).astype(np.uint8)
+        # The new trajectory crosses the right image boundary while airborne;
+        # the ground interaction is consequently offscreen, not a reused roll.
+        x, y = changed[t]
+        if -5 < x < 261 and -5 < y < 149:
+            texture_t = t if t <= 20 else 3+(t-21)%18
+            scale = original_den[texture_t]/changed_den[t]
+            composite(result[t], textures[texture_t], changed[t], scale)
+    result[0] = source[0]  # Initial location and appearance stay unchanged.
+    artifact_io.save(result, ROOT)
+    diagnostics = {
+        'fit_rms_pixels': float(np.sqrt(np.mean((fitted-positions[:28])**2))),
+        'initial_speed_multiplier': 1.5,
+        'coefficients': coefficients.tolist(),
+        'counterfactual_centers_first_40': changed[:40].tolist(),
+        'shape': list(result.shape), 'fps': 24,
+    }
+    (ROOT/'generation_diagnostics.json').write_text(json.dumps(diagnostics,indent=2))
+    print('Saved 96 RGB frames at 256x144, 24 fps.')
+    print('Trajectory fit RMS:', diagnostics['fit_rms_pixels'])
+    print('Ball centers at frames 1, 13, 25, 36:', changed[[0,12,24,35]])
+
+if __name__ == '__main__':
+    main()

\"\"\"Reconstruct the source tennis-ball flight and change only its initial velocity.

The perspective projection of a constant-gravity ballistic trajectory has a
quadratic numerator and denominator. Scaling the linear terms by 1.5 changes
velocity, not gravity, the initial position, or the camera. Source RGB pixels
supply both the static scene and the ball's illuminated surface.
\"\"\"
from pathlib import Path
import json
import numpy as np
import cv2
from scipy.ndimage import label
from scipy.optimize import least_squares
import artifact_io

ROOT = Path(__file__).resolve().parent


def track_ball(frames, background):
    positions = []
    for frame in frames:
        a = frame.astype(np.float64)
        delta = a - background
        weight = np.maximum(delta[..., 1] - delta[..., 2], 0)
        weight[(weight < 15) | (a[..., 1] < a[..., 0] * 1.04)] = 0
        components, _ = label(weight > 0)
        sizes = np.bincount(components.ravel())
        sizes[0] = 0
        mask = components == np.argmax(sizes)
        y, x = np.where(mask)
        positions.append([np.average(x, weights=weight[mask]),
                          np.average(y, weights=weight[mask])])
    return np.array(positions)


def project(coefficients, times, speed=1.0):
    times = np.asarray(times)
    a, b, c = coefficients[:3], coefficients[3:6], coefficients[6:]
    denominator = 1 + speed * c[0] * times + c[1] * times**2
    xy = np.stack([(a[0] + speed*a[1]*times + a[2]*times**2)/denominator,
                   (b[0] + speed*b[1]*times + b[2]*times**2)/denominator], -1)
    return xy, denominator


def get_texture(image, clean, center, radius):
    \"\"\"Unmatte source ball pixels to a small premultiplied RGB sprite.\"\"\"
    hi_center = center * 5 + 2
    ox, oy = np.floor(hi_center - 16).astype(int)
    size = 34
    patch = image[oy:oy+size, ox:ox+size].astype(np.float64)
    background = clean[oy:oy+size, ox:ox+size].astype(np.float64)
    yy, xx = np.mgrid[:size, :size]
    distance = np.hypot(xx + ox-hi_center[0], yy + oy-hi_center[1])
    alpha = np.clip((radius + .75-distance)/1.5, 0, 1)
    premult = patch - background*(1-alpha[..., None])
    premult = np.minimum(np.maximum(premult, 0), 255*alpha[..., None])
    premult[alpha == 0] = 0
    return premult.astype(np.float32), alpha.astype(np.float32), hi_center-np.array([ox,oy])


def composite(frame, texture, center, scale):
    color, alpha, reference = texture
    x, y = center
    pad = 5 * max(scale, 1)
    x0, x1 = max(0, int(np.floor(x-pad))), min(256, int(np.ceil(x+pad))+1)
    y0, y1 = max(0, int(np.floor(y-pad))), min(144, int(np.ceil(y+pad))+1)
    if x0 >= x1 or y0 >= y1:
        return
    # Pixel-area integration on the same 5:1 source/downsample grid.
    yy, xx = np.mgrid[:(y1-y0)*5, :(x1-x0)*5]
    gx = x0 + (xx+.5)/5-.5
    gy = y0 + (yy+.5)/5-.5
    mx = ((gx-x)*5/scale+reference[0]).astype(np.float32)
    my = ((gy-y)*5/scale+reference[1]).astype(np.float32)
    c = cv2.remap(color, mx, my, cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)
    a = cv2.remap(alpha, mx, my, cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)
    h, w = y1-y0, x1-x0
    c = c.reshape(h,5,w,5,3).mean((1,3))
    a = a.reshape(h,5,w,5).mean((1,3))
    region = frame[y0:y1, x0:x1]
    region[:] = np.clip(np.rint(c + region*(1-a[...,None])), 0, 255).astype(np.uint8)


def main():
    source = np.load(ROOT/'source_pixels.npz')['rgb']
    # First 64 frames never contain a stationary ball at one location, giving
    # an unoccluded temporal estimate even at its eventual stopping position.
    clean = np.median(source[:64], axis=0)
    positions = track_ball(source, clean)
    flight_times = np.arange(28, dtype=np.float64)
    init = [28.4, 4.3, .015, 59.1, -4.7, .22, -.0021, .00014]
    fit = least_squares(lambda q: (project(q, flight_times)[0]-positions[:28]).ravel(),
                        init, max_nfev=3000, xtol=1e-12, ftol=1e-12, gtol=1e-12)
    coefficients = fit.x
    fitted, original_den = project(coefficients, flight_times)
    changed, changed_den = project(coefficients, np.arange(96), 1.5)

    # Only source video pixels are used for surface detail. Frame 40 provides
    # clear background in the airborne region. Median samples remove codec

    cap = cv2.VideoCapture(str(ROOT/'source_video.mp4'))

    for t in range(41):

    # The sphere is about 10.4 source-resolution pixels in radius. Perspective
    # magnification follows the same depth denominator as its trajectory.
    radius_at_start = 10.4
    textures = {}
    for t in range(1,21):

    result = source.copy()
    yy, xx = np.mgrid[:144, :256]
    for t in range(1,96):

        # other moving/compression detail on the source timeline.
        old_x, old_y = positions[t]
        distance = np.hypot(xx-old_x, yy-old_y)
        erase = np.clip((6-distance)/1.25, 0, 1)
        result[t] = np.rint(result[t]*(1-erase[...,None]) + clean*erase[...,None]).astype(np.uint8)
        # The new trajectory crosses the right image boundary while airborne;
        # the ground interaction is consequently offscreen, not a reused roll.
        x, y = changed[t]
        if -5 < x < 261 and -5 < y < 149:
            texture_t = t if t <= 20 else 3+(t-21)%18
            scale = original_den[texture_t]/changed_den[t]
            composite(result[t], textures[texture_t], changed[t], scale)
    result[0] = source[0]  # Initial location and appearance stay unchanged.
    artifact_io.save(result, ROOT)
    diagnostics = {
        'fit_rms_pixels': float(np.sqrt(np.mean((fitted-positions[:28])**2))),
        'initial_speed_multiplier': 1.5,
        'coefficients': coefficients.tolist(),
        'counterfactual_centers_first_40': changed[:40].tolist(),
        'shape': list(result.shape), 'fps': 24,
    }
    (ROOT/'generation_diagnostics.json').write_text(json.dumps(diagnostics,indent=2))
    print('Saved 96 RGB frames at 256x144, 24 fps.')
    print('Trajectory fit RMS:', diagnostics['fit_rms_pixels'])
    print('Ball centers at frames 1, 13, 25, 36:', changed[[0,12,24,35]])

if __name__ == '__main__':
    main()
