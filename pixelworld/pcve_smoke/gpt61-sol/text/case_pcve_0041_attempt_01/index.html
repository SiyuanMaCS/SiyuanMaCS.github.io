+ROOT = Path(__file__).resolve().parent

+def read_original():
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    keep = {}
+    for t in range(96):
+        ok, bgr = cap.read()
+        if not ok:
+            raise RuntimeError('Incomplete source video')
+        if t in (0, 1, 2, 7, 8, 9, 10, 23):
+            keep[t] = cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB)

+def component_matte(image, clean, bounds):
+    """Extract the connected car footprint, including its contact shadow."""
+    x0, y0, x1, y1 = bounds
+    a = image[y0:y1, x0:x1].astype(np.float32)
+    b = clean[y0:y1, x0:x1].astype(np.float32)
+    difference = np.max(np.abs(a-b), axis=2)
+    mask = difference > 22
+    mask = ndimage.binary_closing(mask, iterations=2)
+    labels, _ = ndimage.label(mask)
+    sizes = np.bincount(labels.ravel())
+    sizes[0] = 0
+    body = labels == sizes.argmax()
+    body = ndimage.binary_fill_holes(body)
+    body = ndimage.binary_dilation(body, iterations=2)
+    # Subpixel feathering at native resolution preserves the source edges.
+    alpha = ndimage.gaussian_filter(body.astype(np.float32), .65)
+    return a, alpha
+
+
+def main():

+    native = read_original()
+    clean = np.median(np.stack([native[i] for i in (0,1,2)]), axis=0).astype(np.uint8)
+    output = factual.copy()

+    # Measurements in the 256x144 projection.  The landed trajectory has

+    # High-friction impulse arrests slip. A short compliant contact interval
+    # distributes the impulse while the nose-down car rights itself.
+    friction_impulse = friction_changed * normal_impact_velocity
+    vx_after_impact = max(0.0, vx_air-friction_impulse)

    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
    keep = {}
    for t in range(96):
        ok, bgr = cap.read()
        if not ok:
            raise RuntimeError('Incomplete source video')
        if t in (0, 1, 2, 7, 8, 9, 10, 23):
            keep[t] = cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB)
    cap.release()
    return keep


def component_matte(image, clean, bounds):
    \"\"\"Extract the connected car footprint, including its contact shadow.\"\"\"
    x0, y0, x1, y1 = bounds
    a = image[y0:y1, x0:x1].astype(np.float32)
    b = clean[y0:y1, x0:x1].astype(np.float32)
    difference = np.max(np.abs(a-b), axis=2)
    mask = difference > 22
    mask = ndimage.binary_closing(mask, iterations=2)
    labels, _ = ndimage.label(mask)
    sizes = np.bincount(labels.ravel())
    sizes[0] = 0
    body = labels == sizes.argmax()
    body = ndimage.binary_fill_holes(body)
    body = ndimage.binary_dilation(body, iterations=2)
    # Subpixel feathering at native resolution preserves the source edges.
    alpha = ndimage.gaussian_filter(body.astype(np.float32), .65)
    return a, alpha


def main():
    factual = np.load(ROOT / 'source_pixels.npz')['rgb']
    native = read_original()
    clean = np.median(np.stack([native[i] for i in (0,1,2)]), axis=0).astype(np.uint8)
    output = factual.copy()
    S = 5.0
    # Measurements in the 256x144 projection.  The landed trajectory has
    # acceleration -0.43291 px/frame"'^2; the flight has gravity 2.05052.
    gravity = 2.05051810
    friction_original = .43290642 / gravity
    friction_changed = 7.0 * friction_original
    contact_t = 6.37869
    contact_x = 134.95680
    vx_air = 10.27132105
    normal_impact_velocity = 8.91653614
    # High-friction impulse arrests slip. A short compliant contact interval
    # distributes the impulse while the nose-down car rights itself.
    friction_impulse = friction_changed * normal_impact_velocity
    vx_after_impact = max(0.0, vx_air-friction_impulse)
    contact_duration = .44
    stop_x = contact_x + .5*(vx_air+vx_after_impact)*contact_duration
    slope, intercept = -.0460386436, 54.0250970
    stop_y = slope*stop_x + intercept
    centers = {

    # Restrict all reconstruction to the tabletop car corridor.
    # Background to its right is exposed in the first three factual frames.
    region = (600,190,1060,292)
    x0,y0,x1,y1 = region
    clean_roi = clean[y0:y1,x0:x1]
    h,w = clean_roi.shape[:2]
    sprites = {}
    for k in (7,8,9,10):
        rgb, alpha = component_matte(native[k], clean, region)
        sprites[k] = (rgb,alpha)
    for t in range(7,96):
        k = min(t,10)
        source_rgb,alpha = sprites[k]
        # Erase just the factual car and its shadow, not the whole scene.
        # Original centers after settling lie within this bounded region.
        if t <= 10:
            original = native[t][y0:y1,x0:x1].copy()
        else:
            # We retain the exact source RGB everywhere outside the edited
            # footprint; use the native plateau footprint to remove the car.
            original = native[23][y0:y1,x0:x1].copy()
        # Estimate factual footprint from the requested-size frame, which is
        # available on every source timestamp (including tiny codec changes).
        small_bg = np.asarray(Image.fromarray(clean).resize((256,144),Image.Resampling.LANCZOS))
        d = np.max(np.abs(factual[t].astype(np.int16)-small_bg.astype(np.int16)),2)
        m = np.zeros((144,256),np.uint8)
        # Only car corridor, so static unrelated toys are excluded.
        m[38:58,120:211] = (d[38:58,120:211] > 15).astype(np.uint8)
        l,n = ndimage.label(m)
        z = np.bincount(l.ravel()); z[0]=0
        m = ndimage.binary_fill_holes(l == z.argmax())
        m = ndimage.binary_dilation(m,iterations=2).astype(np.float32)
        erase = cv2.resize(m,(1280,720),interpolation=cv2.INTER_LINEAR)[y0:y1,x0:x1]
        # Begin with the actual frame'"'s background. The native first frames
        # provide the clean surface under both old and counterfactual cars.
        factual_native = np.asarray(Image.fromarray(factual[t]).resize((1280,720),Image.Resampling.BICUBIC)).astype(np.float32)
        base = factual_native[y0:y1,x0:x1].copy()
        base = base*(1-erase[...,None]) + clean_roi.astype(np.float32)*erase[...,None]
        # Grip causes a brief rear-wheel lift, then the chassis settles.
        pitch = {7:8.0,8:3.5,9:1.0}.get(t,0.0)
        cx,cy = centers[k]
        # Rotate about the forward tire contact, not the sprite center.
        pivot = ((cx+6.2)*S-x0,(cy+2.5)*S-y0)
        M = cv2.getRotationMatrix2D(pivot,-pitch,1.0)
        M[:,2] += [(stop_x-cx)*S,(stop_y-cy)*S]
        premul = source_rgb*alpha[...,None]
        warped_alpha = cv2.warpAffine(alpha,M,(w,h),flags=cv2.INTER_LINEAR)
        warped_rgb = cv2.warpAffine(premul,M,(w,h),flags=cv2.INTER_LINEAR)
        composite = base*(1-warped_alpha[...,None])+warped_rgb
        # Native-resolution compositing followed by the exact source filter.
        native_result = factual_native
        native_result[y0:y1,x0:x1] = composite
        rendered = np.asarray(Image.fromarray(np.uint8(np.clip(native_result,0,255))).resize((256,144),Image.Resampling.LANCZOS))
        # Restore unaffected pixels exactly; this also avoids re-resampling
        # the unmodified static scene outside the two car footprints.
        edit_native = np.zeros((720,1280),np.float32)
        edit_native[y0:y1,x0:x1] = np.maximum(erase,warped_alpha)
        edit = cv2.resize(edit_native,(256,144),interpolation=cv2.INTER_AREA)>0.001
        edit = ndimage.binary_dilation(edit,iterations=2)
        output[t][edit] = rendered[edit]
    assert np.array_equal(output[:7], factual[:7])
    artifact_io.save(output,ROOT)
    print(f'Saved {len(output)} frames, friction multiplier=7; stopping x={stop_x:.3f}, y={stop_y:.3f}')


if __name__ == '__main__':
    main()
