+"""Source-pixel counterfactual billiards edit, with a quarter-point one-ball.
+No external assets or renderer: numpy contact mechanics and RGB sprite warps.
+"""
+from pathlib import Path
+import numpy as np
+import cv2
+from scipy import ndimage
+from artifact_io import save
+
+ROOT = Path(__file__).resolve().parent
+SRC = np.load(ROOT / 'source_pixels.npz')['rgb']
+N, HH, WW, _ = SRC.shape
+YY, XX = np.indices((HH, WW))
+
+# Fixed camera table-plane calibration. The small vertical offset converts the
+# cloth homography to the projected centers of the spheres above the cloth.
+H = cv2.getPerspectiveTransform(np.float32([[0,0],[1,0],[0,2],[1,2]]),
+                               np.float32([[139,43],[202,52],[0,107],[116,142]]))
+H[1,2] -= 1.9
+HI = np.linalg.inv(H)
+def to_world(q):
+    z = HI @ np.r_[q, 1.]
+    return z[:2] / z[2]
+def to_screen(q):
+    z = H @ np.r_[q, 1.]
+    return z[:2] / z[2]
+
+CUE0 = np.array([85.9, 94.0])
+BLACK0 = np.array([127.9, 69.3])
+YELLOW0 = CUE0 + .25 * (BLACK0-CUE0)
+
+# Recover the original moving objects to remove them, not the rail/pockets.
+def largest_component(mask):
+    labels, count = ndimage.label(mask)
+    if not count:
+        return None
+    sizes = np.bincount(labels.ravel()); sizes[0] = 0
+    return np.where(labels == sizes.argmax())
+
+cue_track = []
+for image in SRC:
+    r,g,b = image.astype(float).transpose(2,0,1)
+    mask = (r>155)&(g>155)&(b>140)&(XX>76)&(XX<134)&(YY>64)&(YY<107)
+    y,x = largest_component(mask)
+    cue_track.append([x.mean()+.25, y.mean()+.35])
+cue_track = np.array(cue_track)
+median = np.median(SRC, axis=0).astype(float)
+black_track = []
+last = BLACK0.copy()
+for t, image in enumerate(SRC):
+    if t <= 12:
+        center = BLACK0.copy()
+    else:
+        # The large differences here are the dark target, including its shadow.
+        dif = np.max(abs(image.astype(float)-median),axis=2)>27
+        dif &= (YY>42)&(YY<76)&(XX>127)&(XX<175)
+        dif &= ((XX-last[0])**2+(YY-last[1])**2 < 8**2)
+        labels,count=ndimage.label(dif)
+        options=[]
+        for k in range(1,count+1):
+            y,x=np.where(labels==k)
+            if len(x)>=8:
+                options.append((len(x), np.array([x.mean(),y.mean()-.8])))
+        center=max(options,key=lambda z:z[0])[1] if options else last.copy()
+    black_track.append(center);last=center
+black_track=np.array(black_track)
+
+# Clean cloth is temporally observed almost everywhere. The final resting cue
+# covers its location for most frames, so reconstruct only that small smooth
+# cloth region from the unoccupied surrounding pixels.
+clean = median.copy()
+patch = (XX>=118)&(XX<=135)&(YY>=63)&(YY<=82)
+annulus=(XX>=112)&(XX<=140)&(YY>=59)&(YY<=89)&~patch
+cloth=(clean[:,:,1]-clean[:,:,0]>40)&(clean[:,:,1]-clean[:,:,2]>25)
+fitmask=annulus&cloth
+fx=(XX-126)/15.;fy=(YY-72)/15.
+features=np.stack([np.ones_like(fx),fx,fy,fx*fx,fx*fy,fy*fy],-1)
+coef=np.linalg.lstsq(features[fitmask],clean[fitmask],rcond=None)[0]
+clean[patch]=(features@coef)[patch]
+
+# Source sphere/shadow sprites, taken at original 1280x720 resolution.
+cap=cv2.VideoCapture(str(ROOT/'source_video.mp4'))
+ok, high=cap.read();cap.release()
+assert ok
+high=cv2.cvtColor(high,cv2.COLOR_BGR2RGB).astype(float)
+
+def sprite(center, radius, halfwidth=40):
+    cx,cy=center
+    x0=int(cx)-halfwidth;y0=int(cy)-halfwidth
+    y,x=np.indices((2*halfwidth+1,2*halfwidth+1))
+    dx=x+x0-cx;dy=y+y0-cy
+    pixels=high[y0:y0+2*halfwidth+1,x0:x0+2*halfwidth+1]
+    # Polynomial estimation of the cloth under the ball and its shadow.
+    valid=(np.hypot(dx,dy)>radius+8)&(pixels[:,:,1]-pixels[:,:,0]>35)
+    f=np.stack([np.ones_like(dx),dx/40,dy/40,(dx/40)**2,
+                dx*dy/1600,(dy/40)**2],-1)
+    coeff=np.linalg.lstsq(f[valid],pixels[valid],rcond=None)[0]
+    background=f@coeff
+    delta=pixels-background
+    # Restrict to ball and attached soft cloth shadow, removing codec ripples.
+    support=(dx/(radius+5))**2+((dy-6)/(radius+14))**2
+    window=np.clip((1.12-support)/.25,0,1)
+    delta*=window[:,:,None]
+    return {'delta':delta,'pixels':pixels,'bg':background,'dx':dx,'dy':dy,
+            'radius':radius,'center':np.array([cx-x0,cy-y0])}
+
+white = sprite((429.5,470.5),18.6,42)
+black = sprite((639.7,346.3),12.7,33)
+yellow = {k:(v.copy() if isinstance(v,np.ndarray) else v) for k,v in white.items()}
+dx,dy=white['dx'],white['dy'];rr=white['radius']
+radial=np.sqrt((dx/rr)**2+(dy/rr)**2)
+alpha=np.clip((1.015-radial)/.065,0,1)
+# Glossy saturated yellow resin, retaining original illumination and highlight.
+body=white['pixels']*np.array([1.02,.79,.095])
+highlight=np.exp(-((dx/rr+.25)**2+(dy/rr+.58)**2)/.045)*.65
+body=body*(1-highlight[:,:,None])+white['pixels']*highlight[:,:,None]
+# The numbered white circle sits on the front hemisphere.
+nx=dx/rr-.18;ny=dy/rr-.10
+roundel=np.clip((.36-np.sqrt(nx*nx+ny*ny))/.035,0,1)
+label=white['pixels']*.96
+body=body*(1-roundel[:,:,None])+label*roundel[:,:,None]
+# Tiny legible '1', built as analytic strokes rather than an external font.
+stem=(abs(nx-.015)<.027)&(ny>-.18)&(ny<.19)
+foot=(abs(nx)<.10)&(abs(ny-.19)<.025)
+capstroke=(ny<-.095)&(ny>-.18)&(abs(nx-(ny+.16)*.65)<.035)
+ink=(stem|foot|capstroke)&(roundel>.5)
+body[ink]=np.array([35.,31.,23.])
+yellow['delta']=white['delta']+(body-white['pixels'])*alpha[:,:,None]
+
+# Contact motion at 2880 integration steps/second. Equal-mass elastic
+# transfers, rolling drag, and the original far-cushion return are included.
+pos=np.array([to_world(CUE0),to_world(YELLOW0),to_world(BLACK0)])
+vel=np.zeros((3,2))
+direction=to_world(BLACK0)-to_world(CUE0);direction/=np.linalg.norm(direction)
+vel[0]=direction*.04065  # table units per source frame, fit to source cue
+ball_r=np.array([.036,.036,.033])
+trajectories=[]
+contacts=[]
+substeps=160;dt=1/substeps
+for t in range(N):
+    trajectories.append(pos.copy())
+    if t==N-1:break
+    for sub in range(substeps):
+        speed=np.linalg.norm(vel,axis=1)
+        drag=.00017
+        vel*=np.maximum(0,speed-drag*dt)[:,None]/np.maximum(speed[:,None],1e-12)
+        pos+=vel*dt
+        for i in range(3):
+            # The sphere-center calibration puts the rail contact at y=0.
+            if pos[i,1]<0 and vel[i,1]<0:
+                pos[i,1]=-pos[i,1];vel[i,1]*=-.78
+                contacts.append((round(t+sub*dt,3),'cushion',i))
+            if pos[i,1]>2-ball_r[i] and vel[i,1]>0:
+                pos[i,1]=2-ball_r[i];vel[i,1]*=-.78
+            for axis in [0]:
+                if pos[i,axis]<ball_r[i]:
+                    pos[i,axis]=ball_r[i];vel[i,axis]=abs(vel[i,axis])*.78
+                if pos[i,axis]>1-ball_r[i]:
+                    pos[i,axis]=1-ball_r[i];vel[i,axis]=-abs(vel[i,axis])*.78
+        for i,j in [(0,1),(0,2),(1,2)]:
+            delta=pos[j]-pos[i];distance=np.linalg.norm(delta)
+            if distance<ball_r[i]+ball_r[j]:
+                normal=delta/max(distance,1e-9)
+                rel=np.dot(vel[j]-vel[i],normal)
+                overlap=ball_r[i]+ball_r[j]-distance
+                pos[i]-=normal*overlap*.5;pos[j]+=normal*overlap*.5
+                if rel<0:
+                    impulse=-(1+.94)*rel*.5
+                    vel[i]-=impulse*normal;vel[j]+=impulse*normal
+                    contacts.append((round(t+sub*dt,3),'ball',i,j))
+trajectories=np.array(trajectories)
+
+# Warp supersampled source RGB differences to subpixel positions. The delta
+# includes the existing antialiasing, sphere lighting, and soft contact shadow.
+def paste(frame, spr, center, radius):
+    scale=radius/(spr['radius']/5)
+    x0=max(0,int(np.floor(center[0]-radius-4)))
+    x1=min(WW,int(np.ceil(center[0]+radius+5)))
+    y0=max(0,int(np.floor(center[1]-radius-4)))
+    y1=min(HH,int(np.ceil(center[1]+radius+7)))
+    if x0>=x1 or y0>=y1:return
+    # Pixel-average sample at 4x4 subpixel locations matches area downsampling.
+    gy,gx=np.indices(((y1-y0)*4,(x1-x0)*4),dtype=np.float32)
+    sx=(gx+.5)/4+x0-.5;sy=(gy+.5)/4+y0-.5
+    mx=(sx-center[0])*(5/scale)+spr['center'][0]
+    my=(sy-center[1])*(5/scale)+spr['center'][1]
+    delta=cv2.remap(spr['delta'].astype(np.float32),mx.astype(np.float32),my.astype(np.float32),cv2.INTER_LINEAR,
+                    borderMode=cv2.BORDER_CONSTANT,borderValue=0)
+    delta=delta.reshape(y1-y0,4,x1-x0,4,3).mean((1,3))
+    frame[y0:y1,x0:x1]+=delta
+
+frames=[]
+for t in range(N):
+    frame=SRC[t].astype(float).copy()
+    mask=np.zeros((HH,WW),bool)
+    for center,kind in [(cue_track[t],'cue'),(black_track[t],'black')]:
+        rad=(3.65+.044*(center[1]-94)) if kind=='cue' else (2.54+.031*(center[1]-69.3))
+        # Includes the lower contact shadow and native video filter support.
+        mask|=(((XX-center[0])/(rad+2.8))**2+
+               ((YY-center[1]-1.8)/(rad+4.1))**2<1)
+    frame[mask]=clean[mask]
+    centers=[to_screen(q) for q in trajectories[t]]
+    # Far-to-near ordering naturally produces correct depth occlusion.
+    for i in sorted(range(3),key=lambda i:centers[i][1]):
+        c=centers[i]
+        r=(3.70+.044*(c[1]-94)) if i<2 else (2.54+.031*(c[1]-69.3))
+        paste(frame,[white,yellow,black][i],c,max(1.2,r))
+    frames.append(np.clip(np.rint(frame),0,255).astype(np.uint8))
+frames=np.array(frames)
+save(frames,ROOT)
+print('Saved',frames.shape,frames.dtype)
+print('Initial screen positions (cue, one-ball, target):', [to_screen(q).round(3).tolist() for q in trajectories[0]])
+print('Contacts:',contacts)
+print('Final screen positions:',[to_screen(q).round(3).tolist() for q in trajectories[-1]])
