+"""Counterfactual restitution edit using only the supplied footage and numeric RGB.
+The unchanged fall is retained; fitted ballistic motion and Coulomb contact dynamics
+replace the bounce sequence. Ball material and floor are sampled from the footage.
+"""
+from pathlib import Path
+import numpy as np
+import cv2
+from PIL import Image
+from scipy.optimize import least_squares
+from scipy.ndimage import gaussian_filter1d
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+def read_source():
+    rgb = np.load(ROOT/'source_pixels.npz')['rgb']
+    cap = cv2.VideoCapture(str(ROOT/'source_video.mp4'))
+    hi=[]
+    while True:
+        ok,f=cap.read()
+        if not ok: break
+        hi.append(cv2.cvtColor(f,cv2.COLOR_BGR2RGB))
+    cap.release()
+    return rgb,np.array(hi)
+
+def track(rgb,hi,bg):
+    centers=[]; radii=[]
+    theta=np.arange(240)*2*np.pi/240
+    for i,f in enumerate(rgb):
+        a=f.astype(float);r,g,b=a.transpose(2,0,1)
+        mask=((r>1.8*g)&(r>1.8*b)&(r>90)&(g<85)).astype(np.uint8)
+        cs,_=cv2.findContours(mask,cv2.RETR_EXTERNAL,cv2.CHAIN_APPROX_SIMPLE)
+        x,y,w,h=cv2.boundingRect(max(cs,key=cv2.contourArea))
+        cx,cy=(np.array([x+(w-1)/2,y+(h-1)/2])+.5)*5-.5
+        r0=5*(11.55-.0055*((cy+.5)/5-.5))
+        rr=np.arange(r0-10,r0+10,.2)
+        mx=(cx+np.cos(theta[:,None])*rr).astype(np.float32)
+        my=(cy+np.sin(theta[:,None])*rr).astype(np.float32)
+        p=cv2.remap(hi[i].astype(np.float32),mx,my,cv2.INTER_LINEAR)
+        q=cv2.remap(bg.astype(np.float32),mx,my,cv2.INTER_LINEAR)
+        diff=(p[:,:,0]-2*p[:,:,1])-(q[:,:,0]-2*q[:,:,1])
+        diff=cv2.GaussianBlur(diff,(5,1),.5)
+        deriv=-np.diff(diff,axis=1)
+        idx=deriv.argmax(1); edge=rr[idx]+.1
+        pts=np.c_[cx+np.cos(theta)*edge,cy+np.sin(theta)*edge]
+        valid=(deriv.max(1)>2)&(np.abs(edge-r0)<8)
+        fit=least_squares(lambda v:np.linalg.norm(pts[valid]-v[:2],axis=1)-v[2],
+                          [cx,cy,r0],loss='soft_l1',f_scale=.7).x
+        centers.append((fit[:2]+.5)/5-.5); radii.append(fit[2]/5)
+    return np.array(centers),np.array(radii)
+
+def motion(c,r):
+    # Image-plane acceleration, collision time and velocity are measured from
+    # the first two unobstructed flight arcs, rather than assigned arbitrarily.
+    t=np.arange(12)
+    first=np.polyfit(t,c[:12,1],2)
+    second=np.polyfit(np.arange(13,32),c[13:32,1],2)
+    roots=np.roots(first-second)
+    tc=float(min(roots,key=lambda v:abs(v-12)).real)
+    ground0=float(np.polyval(first,tc))
+    vin=float(np.polyval(np.polyder(first),tc))
+    vout=float(np.polyval(np.polyder(second),tc))
+    e0=-vout/vin
+    g=2*second[0]
+    # Ground projection changes slightly as the ball moves into the foreground.
+    slope=.09
+    px=np.polyfit(t,c[:12,0],2)
+    xc=float(np.polyval(px,tc)); vx0=float(np.polyval(np.polyder(px),tc))
+    # The original first impact's tangential impulse and rotation imply a
+    # hollow-ball inertia ratio close to .53 and low kinetic friction.
+    k=.53
+    postx=least_squares(lambda p:xc+p[0]/p[1]*(1-np.exp(-p[1]*(np.arange(13,96)-tc)))-c[13:,0],
+                       [.368,.011],bounds=([0,.00001],[2,.08])).x
+    mu=(vx0-postx[0])/((1+e0)*vin)
+    rad=float(r[12]); omega0=-.060
+    x=xc; y=ground0; vx=vx0; vy=vin; omega=omega0; angle=0.
+    # Apply the same material's normal response scaled by precisely .4.
+    eff_e=e0*.4
+    normal=(1+eff_e)*vin
+    slip=vx-omega*rad
+    impulse=-np.clip(slip/(1+1/k),-mu*normal,mu*normal)
+    vx+=impulse; omega-=impulse/(k*rad); vy=-eff_e*vin
+    time=tc; grounded=False
+    positions=c.copy(); angles=np.zeros(96)
+    hits=[(tc,eff_e)]
+    for frame in range(int(np.floor(tc))+1,96):
+        end=float(frame)
+        while time < end-1e-9:
+            dt=min(1/240,end-time)
+            vx*=np.exp(-.01099*dt); omega*=np.exp(-.01099*dt)
+            if grounded:
+                # Static friction enforces rolling without an unnatural freeze.
+                slip=vx-omega*rad
+                j=-np.clip(slip/(1+1/k),-.030*g*dt,.030*g*dt)
+                vx+=j; omega-=j/(k*rad)
+                x+=vx*dt
+                y=ground0+slope*(x-xc); vy=slope*vx
+            else:
+                x+=vx*dt; y+=vy*dt+.5*g*dt*dt; vy+=g*dt
+                floor=ground0+slope*(x-xc)
+                floorv=slope*vx
+                if y>=floor and vy>floorv:
+                    vn=vy-floorv; y=floor
+                    # Later source flights indicate e~.83; all contacts use
+                    # the reduced restitution, including the small final hops.
+                    ee=.4*.83
+                    j=-np.clip((vx-omega*rad)/(1+1/k),-mu*(1+ee)*vn,mu*(1+ee)*vn)
+                    vx+=j; omega-=j/(k*rad)
+                    vy=slope*vx-ee*vn
+                    hits.append((time,ee))
+                    if ee*vn<.10:
+                        grounded=True; vy=slope*vx
+            angle+=omega*dt; time+=dt
+        positions[frame]=[x,y]; angles[frame]=angle
+    print('Fitted first contact:',round(tc,4),'original effective restitution:',round(e0,4),
+          'edited:',round(eff_e,4),'friction:',round(mu,4))
+    print('Edited contacts:',[(round(t,3),round(e,3)) for t,e in hits])
+    print('Edited centers [12,16,20,24,48,95]:',positions[[12,16,20,24,48,95]].round(3).tolist())
+    return positions,angles,tc,ground0,xc,slope
+
+def texture_spin(hi,c,r):
+    n=128
+    u,v=np.meshgrid((np.arange(n)+.5-n/2)/(n/2),(np.arange(n)+.5-n/2)/(n/2))
+    z=np.sqrt(np.maximum(0,1-u*u-v*v));good=u*u+v*v<.8**2
+    mats=np.zeros((good.sum()*2,3))
+    mats[:good.sum(),1]=z[good];mats[:good.sum(),2]=-v[good]
+    mats[good.sum():,0]=-z[good];mats[good.sum():,2]=u[good]
+    pics=[]
+    for i in range(96):
+        cx,cy=(c[i]+.5)*5-.5
+        p=cv2.remap(hi[i],(cx+5*r[i]*u).astype(np.float32),(cy+5*r[i]*v).astype(np.float32),cv2.INTER_LINEAR)
+        pics.append(cv2.cvtColor(p,cv2.COLOR_RGB2GRAY))
+    angular=[]
+    for i in range(95):
+        flow=cv2.calcOpticalFlowFarneback(pics[i],pics[i+1],None,.5,4,17,4,7,1.5,0)/(n/2)
+        values=np.r_[flow[:,:,0][good],flow[:,:,1][good]]
+        angular.append(np.linalg.lstsq(mats,values,rcond=None)[0][2])
+    angular=gaussian_filter1d(np.array(angular),.7)
+    phase=np.r_[0,np.cumsum(angular)]
+    return phase-phase[12]
+
+def shadow_template(hi,bg,c,r):
+    # Recover the actual soft contact shadow from a factual landing. Within
+    # the occluded footprint a fitted continuation supplies hidden shadow.
+    ref=48
+    cx,cy=(c[ref]+.5)*5-.5
+    yy,xx=np.mgrid[-45:170,-160:161]
+    mx=(cx+xx).astype(np.float32);my=(cy+yy).astype(np.float32)
+    f=cv2.remap(hi[ref].astype(np.float32),mx,my,cv2.INTER_LINEAR)
+    b=cv2.remap(bg.astype(np.float32),mx,my,cv2.INTER_LINEAR)
+    ratio=np.mean((b-f)/np.maximum(b,1),axis=-1)
+    ratio=cv2.GaussianBlur(ratio,(0,0),1.8)
+    # The RGB background is static; codec variation should not become shadow.
+    analytic=.32*np.exp(-.5*((xx-5)/36)**2-.5*((yy+5)/68)**2)
+    dist=np.sqrt(xx*xx+yy*yy)
+    outside=np.clip((dist-r[ref]*5-2)/7,0,1)
+    shadow=np.clip(ratio,0,.45)*outside+analytic*(1-outside)
+    taper=np.clip((160-np.abs(xx))/30,0,1)*np.clip((170-yy)/40,0,1)*np.clip((yy+45)/15,0,1)
+    shadow*=taper
+    # Restrict to the floor and smoothly fade a faint copied-codec halo.
+    shadow[my<335]=0
+    return shadow.astype(np.float32),cx,cy
+
+def render(hi,rgb,bg,c,r,p,angles,phase,tc,ground0,xc,slope):
+    output=rgb.copy()
+    sh,shx,shy=shadow_template(hi,bg,c,r)
+    rs=np.polyfit(c[13:32,1],r[13:32],1)
+    # Canonical sphere texture data from the current factual frame retains
+    # its detailed basketball seams and spatially varying illumination.
+    for i in range(13,96):
+        tx,ty=p[i]; cx,cy=(p[i]+.5)*5-.5
+        radius=float(np.polyval(rs,ty)*5)
+        canvas=bg.astype(np.float32).copy()
+        floor=ground0+slope*(tx-xc)
+        height=max(0,floor-ty)
+        floorhi=(floor+.5)*5-.5
+        # Move the original landing shadow with the edited ball's floor point.
+        left=int(cx-170);top=int(floorhi-50)
+        right=min(1280,left+350);bottom=min(720,top+230)
+        left=max(left,0);top=max(top,0)
+        yy,xx=np.mgrid[top:bottom,left:right]
+        spread=1+.015*height
+        mapx=((xx-cx)/spread+160).astype(np.float32)
+        mapy=((yy-floorhi-1.2*height)/spread+45).astype(np.float32)
+        opacity=cv2.remap(sh,mapx,mapy,cv2.INTER_LINEAR,borderMode=cv2.BORDER_CONSTANT)
+        opacity*=np.exp(-height/8.5)
+        canvas[top:bottom,left:right]*=(1-opacity[:,:,None])
+        # Recover an opaque sphere interior, then reconstruct its fractional
+        # outline at the native five-times-resolution before Lanczos sampling.
+        x0=max(0,int(cx-radius-4));x1=min(1280,int(cx+radius+5))
+        y0=max(0,int(cy-radius-4));y1=min(720,int(cy+radius+5))
+        yy,xx=np.mgrid[y0:y1,x0:x1]
+        u=(xx-cx)/radius;v=(yy-cy)/radius
+        delta=angles[i]-phase[i]
+        cs=np.cos(delta);sn=np.sin(delta)
+        su=cs*u+sn*v;sv=-sn*u+cs*v
+        d=np.sqrt(su*su+sv*sv)
+        # Clamp only the last native-resolution edge texel to prevent remnants
+        # of the old floor/wall bleeding into the relocated anti-aliased ball.
+        clamp=np.minimum(1,(1-1.4/(5*r[i]))/np.maximum(d,1e-8))
+        scx,scy=(c[i]+.5)*5-.5
+        mx=(scx+su*clamp*5*r[i]).astype(np.float32)
+        my=(scy+sv*clamp*5*r[i]).astype(np.float32)
+        tex=cv2.remap(hi[i].astype(np.float32),mx,my,cv2.INTER_LINEAR)
+        alpha=np.clip((radius-np.sqrt((xx-cx)**2+(yy-cy)**2))/.95+.5,0,1)
+        part=canvas[y0:y1,x0:x1]
+        part[:]=part*(1-alpha[:,:,None])+tex*alpha[:,:,None]
+        output[i]=np.array(Image.fromarray(np.clip(canvas,0,255).astype(np.uint8)).resize((256,144),Image.Resampling.LANCZOS))
+    return output
+
+def main():
+    rgb,hi=read_source()
+    # Only the airborne ball is removed. The clean static wooden floor,
+    # backdrop, camera and illumination remain those in the source.
+    bg=hi[0].copy()
+    bg[30:190,550:750]=hi[48,30:190,550:750]
+    c,r=track(rgb,hi,bg)
+    p,angles,tc,ground0,xc,slope=motion(c,r)
+    phase=texture_spin(hi,c,r)
+    frames=render(hi,rgb,bg,c,r,p,angles,phase,tc,ground0,xc,slope)
+    assert frames.shape==(96,144,256,3) and frames.dtype==np.uint8
+    assert np.array_equal(frames[:13],rgb[:13])
+    artifact_io.save(frames,ROOT)
+    print('Saved complete 96-frame counterfactual video.')
+
+if __name__=='__main__':
+    main()

def motion(c,r):
    # Image-plane acceleration, collision time and velocity are measured from
    # the first two unobstructed flight arcs, rather than assigned arbitrarily.
    t=np.arange(12)
    first=np.polyfit(t,c[:12,1],2)
    second=np.polyfit(np.arange(13,32),c[13:32,1],2)
    roots=np.roots(first-second)
    tc=float(min(roots,key=lambda v:abs(v-12)).real)
    ground0=float(np.polyval(first,tc))
    vin=float(np.polyval(np.polyder(first),tc))
    vout=float(np.polyval(np.polyder(second),tc))
    e0=-vout/vin
    g=2*second[0]
    # Ground projection changes slightly as the ball moves into the foreground.
    slope=.09
    px=np.polyfit(t,c[:12,0],2)
    xc=float(np.polyval(px,tc)); vx0=float(np.polyval(np.polyder(px),tc))
    # The original first impact's tangential impulse and rotation imply a
    # hollow-ball inertia ratio close to .53 and low kinetic friction.
    k=.53
    postx=least_squares(lambda p:xc+p[0]/p[1]*(1-np.exp(-p[1]*(np.arange(13,96)-tc)))-c[13:,0],
                       [.368,.011],bounds=([0,.00001],[2,.08])).x
    mu=(vx0-postx[0])/((1+e0)*vin)
    rad=float(r[12]); omega0=-.060
    x=xc; y=ground0; vx=vx0; vy=vin; omega=omega0; angle=0.
    # Apply the same material's normal response scaled by precisely .4.
    eff_e=e0*.4
    normal=(1+eff_e)*vin
    slip=vx-omega*rad
    impulse=-np.clip(slip/(1+1/k),-mu*normal,mu*normal)
    vx+=impulse; omega-=impulse/(k*rad); vy=-eff_e*vin
    time=tc; grounded=False
    positions=c.copy(); angles=np.zeros(96)
    hits=[(tc,eff_e)]
    for frame in range(int(np.floor(tc))+1,96):
        end=float(frame)
        while time < end-1e-9:
            dt=min(1/240,end-time)
            vx*=np.exp(-.01099*dt); omega*=np.exp(-.01099*dt)
            if grounded:
                # Static friction enforces rolling without an unnatural freeze.
                slip=vx-omega*rad
                j=-np.clip(slip/(1+1/k),-.030*g*dt,.030*g*dt)
                vx+=j; omega-=j/(k*rad)
                x+=vx*dt
                y=ground0+slope*(x-xc); vy=slope*vx
            else:
                x+=vx*dt; y+=vy*dt+.5*g*dt*dt; vy+=g*dt
                floor=ground0+slope*(x-xc)
                floorv=slope*vx
                if y>=floor and vy>floorv:
                    vn=vy-floorv; y=floor
                    # Later source flights indicate e~.83; all contacts use
                    # the reduced restitution, including the small final hops.
                    ee=.4*.83

                    vx+=j; omega-=j/(k*rad)
                    vy=slope*vx-ee*vn
                    hits.append((time,ee))
                    if ee*vn<.10:
                        grounded=True; vy=slope*vx
            angle+=omega*dt; time+=dt
        positions[frame]=[x,y]; angles[frame]=angle
    print('Fitted first contact:',round(tc,4),'original effective restitution:',round(e0,4),
          'edited:',round(eff_e,4),'friction:',round(mu,4))
    print('Edited contacts:',[(round(t,3),round(e,3)) for t,e in hits])
    print('Edited centers [12,16,20,24,48,95]:',positions[[12,16,20,24,48,95]].round(3).tolist())
    return positions,angles,tc,ground0,xc,slope

def shadow_template(hi,bg,c,r):

    # the occluded footprint a fitted continuation supplies hidden shadow.
    ref=48

        opacity=cv2.remap(sh,mapx,mapy,cv2.INTER_LINEAR,borderMode=cv2.BORDER_CONSTANT)
        opacity*=np.exp(-height/8.5)
        canvas[top:bottom,left:right]*=(1-opacity[:,:,None])
        # Recover an opaque sphere interior, then reconstruct its fractional
        # outline at the native five-times-resolution before Lanczos sampling.
        x0=max(0,int(cx-radius-4));x1=min(1280,int(cx+radius+5))
        y0=max(0,int(cy-radius-4));y1=min(720,int(cy+radius+5))
        yy,xx=np.mgrid[y0:y1,x0:x1]
        u=(xx-cx)/radius;v=(yy-cy)/radius
        delta=angles[i]-phase[i]
        cs=np.cos(delta);sn=np.sin(delta)
        su=cs*u+sn*v;sv=-sn*u+cs*v
        d=np.sqrt(su*su+sv*sv)
        # Clamp only the last native-resolution edge texel to prevent remnants
        # of the old floor/wall bleeding into the relocated anti-aliased ball.
        clamp=np.minimum(1,(1-1.4/(5*r[i]))/np.maximum(d,1e-8))
        scx,scy=(c[i]+.5)*5-.5
        mx=(scx+su*clamp*5*r[i]).astype(np.float32)
        my=(scy+sv*clamp*5*r[i]).astype(np.float32)
        tex=cv2.remap(hi[i].astype(np.float32),mx,my,cv2.INTER_LINEAR)
        alpha=np.clip((radius-np.sqrt((xx-cx)**2+(yy-cy)**2))/.95+.5,0,1)
        part=canvas[y0:y1,x0:x1]
        part[:]=part*(1-alpha[:,:,None])+tex*alpha[:,:,None]
        output[i]=np.array(Image.fromarray(np.clip(canvas,0,255).astype(np.uint8)).resize((256,144),Image.Resampling.LANCZOS))
    return output
