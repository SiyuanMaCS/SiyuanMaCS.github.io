+"""Edit the source billiards sequence with a five-times-heavier cue ball.
+
+Only measured source RGB, elementary projective geometry, and momentum /
+restitution equations are used. The intervention changes mass, not radius,
+initial velocity, the cloth, or the camera.
+"""
+from pathlib import Path
+import numpy as np
+import cv2
+from PIL import Image
+from scipy.optimize import least_squares
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+def read_video():
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    frames = []
+    while True:
+        ok, bgr = cap.read()
+        if not ok:
+            break
+        frames.append(cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB))
+    cap.release()
+    return np.asarray(frames, dtype=np.uint8)
+
+
+def measure(source):
+    """High-resolution color centroids; coordinates are on the output grid."""
+    h, w = source.shape[1:3]
+    yy, xx = np.mgrid[:h, :w]
+    cue, obj = [], []
+    for frame in source:
+        r, g, b = frame.astype(np.float32).transpose(2, 0, 1)
+        mask = ((r > 145) & (g > 145) & (b > 130) &
+                (xx > 380) & (xx < 650) & (yy > 320) & (yy < 515))
+        _, _, stats, centers = cv2.connectedComponentsWithStats(mask.astype(np.uint8))
+        j = np.argmax(stats[1:, 4]) + 1
+        cue.append(centers[j] / 5 - .4)
+        mask = ((g < 80) & (r < 90) & (b < 95) &
+                (xx > 600) & (xx < 875) &
+                (yy > -.7 * xx + 785) & (yy < -.5 * xx + 690))
+        _, _, stats, centers = cv2.connectedComponentsWithStats(mask.astype(np.uint8))
+        js = [j for j in range(1, len(stats)) if 50 < stats[j, 4] < 800
+              and stats[j, 2] < 40 and stats[j, 3] < 40]
+        j = max(js, key=lambda j: stats[j, 4])
+        obj.append(centers[j] / 5 - .4)
+    return np.asarray(cue), np.asarray(obj)
+
+
+def dynamics(cue, obj):
+    # The table's long edges intersect at about (282,-20). A projective
+    # coordinate s linearizes the rolling motion instead of treating screen
+    # displacement (which shrinks with distance) as physical speed.
+    vanish = 282.0
+    x0 = cue[0, 0]
+    to_world = lambda x: (x - x0) / (vanish - x)
+    samples = [np.arange(13), np.arange(17, 42), np.arange(49, 96)]
+    def residual(p):
+        drag, v0, intercept, vb, intercept_back, vr = p
+        a, b, c = samples
+        return np.r_[
+            to_world(cue[a, 0]) - (v0*a - .5*drag*a*a),
+            to_world(obj[b, 0]) - (intercept + vb*b - .5*drag*b*b),
+            to_world(obj[c, 0]) - (intercept_back + vr*c + .5*drag*c*c)]
+    fit = least_squares(residual, [.0001, .0188, .04, .017, 1.3, -.014]).x
+    drag, v0, c, vb, d, vr = fit
+    initial_object = to_world(np.mean(obj[:12, 0]) - .22)
+    contact_time = min(np.roots([-.5*drag, vb, c-initial_object]))
+    diameter = initial_object - (v0*contact_time - .5*drag*contact_time**2)
+    rail_time = min(np.roots([-drag, vb-vr, c-d]))
+    far_rail = c + vb*rail_time - .5*drag*rail_time**2
+    restitution = np.clip(2*(vb-drag*contact_time)/(v0-drag*contact_time)-1, .1, 1.)
+    rail_restitution = np.clip(-(vr+drag*rail_time)/(vb-drag*rail_time), .1, 1.)
+
+    # Coulomb rolling deceleration is independent of mass. The same initial
+    # velocity is therefore retained; mass enters every inter-ball impulse.
+    mass = np.array([5., 1.])
+    p = np.array([0., initial_object])
+    velocity = np.array([v0, 0.])
+    trajectory = []
+    events = []
+    steps = 240
+    dt = 1 / steps
+    for frame_index in range(96):
+        trajectory.append(p.copy())
+        if frame_index == 95:
+            break
+        for sub in range(steps):
+            next_v = np.sign(velocity) * np.maximum(0, np.abs(velocity)-drag*dt)
+            p += .5*(velocity+next_v)*dt
+            velocity = next_v
+            for j in range(2):
+                if p[j] > far_rail and velocity[j] > 0:
+                    p[j] = 2*far_rail - p[j]
+                    velocity[j] *= -rail_restitution
+                    events.append((round(frame_index+sub*dt, 3), 'cushion', j))
+            if p[1]-p[0] < diameter and velocity[0] > velocity[1]:
+                relative = velocity[0]-velocity[1]
+                impulse = (1+restitution)*relative/(1/mass[0]+1/mass[1])
+                velocity[0] -= impulse/mass[0]
+                velocity[1] += impulse/mass[1]
+                penetration = diameter-(p[1]-p[0])
+                p[0] -= penetration*mass[1]/mass.sum()
+                p[1] += penetration*mass[0]/mass.sum()
+                events.append((round(frame_index+sub*dt, 3), 'ball contact'))
+    trajectory = np.asarray(trajectory)
+    xs = (x0+vanish*trajectory)/(1+trajectory)
+    # Sphere-center height and light-weight centroid correction are both
+    # derived from the pre-impact cue track.
+    slope, intercept = np.polyfit(cue[:13, 0], cue[:13, 1], 1)
+    centers = np.stack([xs+.10, slope*(xs+.10)+intercept+.20], axis=-1)
+    print('Fitted rolling drag:', drag, 'ball restitution:', restitution,
+          'cushion restitution:', rail_restitution)
+    print('Counterfactual contacts:', events)
+    return centers, vanish
+
+
+def clean_plate(source):
+    ids = [0, 3, 6, 9, 12, 18, 25, 35, 55, 65, 80, 95]
+    plate = np.median(source[ids], axis=0).astype(np.float32)
+    # At the original collision location the stopped cue covers some pixels
+    # which used to be covered by the object ball. Select unobstructed cloth
+    # samples from the whole source, and interpolate only the tiny permanently
+    # occluded intersection. This prevents white/black ghosts in the plate.
+    x0, x1, y0, y1 = 590, 670, 315, 400
+    tiles = source[:, y0:y1, x0:x1].astype(np.float32)
+    cloth = np.array([114.5, 174.8, 140.8], np.float32)
+    scores = np.sum((tiles-cloth)**2, axis=-1)
+    best = scores.argmin(axis=0)
+    yy, xx = np.mgrid[:y1-y0, :x1-x0]
+    patch = tiles[best, yy, xx]
+    hidden = scores.min(axis=0) > 350
+    # Smooth constant cloth color is an accurate local model at this scale;
+    # preserve source grain from adjacent clean pixels for the hidden area.
+    texture = source[0, y0:y1, 555:555+x1-x0].astype(np.float32)
+    texture += cloth-np.median(texture.reshape(-1,3), axis=0)
+    patch[hidden] = texture[hidden]
+    plate[y0:y1, x0:x1] = patch
+    return plate
+
+
+def region_mask(dx, dy, radius, kind):
+    sphere = np.sqrt((dx/radius)**2+(dy/(radius*.94))**2)
+    # Cast shadow goes slightly left and towards the camera.
+    shadow = np.sqrt(((dx+.7*radius)/(radius*1.10))**2+
+                     ((dy-radius*.88)/(radius*.66))**2)
+    # Keep antialiasing and shadow penumbra sampled from the source.
+    return np.maximum(np.clip((1.28-sphere)/.23,0,1),
+                      np.clip((1.28-shadow)/.30,0,1)).astype(np.float32)
+
+
+def sprite(source, plate, frame_index, center, kind, vanish):
+    center_hi = (center+.4)*5
+    # Cue centroid is very close to its sphere center; dark ball centroid
+    # weights its bottom, so correct that bias when positioning a sprite.
+    radius = (3.98 if kind == 'cue' else 3.85)*(vanish-center[0])/(vanish-85.48)
+    half = 48
+    cx, cy = np.round(center_hi).astype(int)
+    x0, y0 = cx-half, cy-half
+    rgb = source[frame_index, y0:cy+half+1, x0:cx+half+1].astype(np.float32)
+    base = plate[y0:cy+half+1, x0:cx+half+1]
+    yy, xx = np.mgrid[y0:cy+half+1, x0:cx+half+1]
+    alpha = region_mask((xx-center_hi[0])/5, (yy-center_hi[1])/5, radius, kind)
+    delta = (rgb-base)*alpha[...,None]
+    return delta, np.array([center_hi[0]-x0, center_hi[1]-y0]), center[0]
+
+
+def remove_original(canvas, plate, center, kind, vanish):
+    high = (center+.4)*5
+    radius = (3.98 if kind == 'cue' else 3.85)*(vanish-center[0])/(vanish-85.48)
+    half = 48
+    cx,cy=np.round(high).astype(int)
+    x0,x1=max(0,cx-half),min(canvas.shape[1],cx+half+1)
+    y0,y1=max(0,cy-half),min(canvas.shape[0],cy+half+1)
+    yy,xx=np.mgrid[y0:y1,x0:x1]
+    # A little larger than the extracted sphere, to remove its complete
+    # antialiased outline and the soft source shadow.
+    weight=region_mask((xx-high[0])/5,(yy-high[1])/5,radius*1.16,kind)
+    canvas[y0:y1,x0:x1] = (canvas[y0:y1,x0:x1]*(1-weight[...,None])+
+                          plate[y0:y1,x0:x1]*weight[...,None])
+
+
+def place(canvas, item, target, vanish):
+    delta, anchor, original_x = item
+    scale = (vanish-target[0])/(vanish-original_x)
+    center=(target+.4)*5
+    hh,ww=delta.shape[:2]
+    half=int(np.ceil(max(hh,ww)*scale*.52))
+    cx,cy=np.round(center).astype(int)
+    x0,x1=max(0,cx-half),min(canvas.shape[1],cx+half+1)
+    y0,y1=max(0,cy-half),min(canvas.shape[0],cy+half+1)
+    yy,xx=np.mgrid[y0:y1,x0:x1]
+    mx=((xx-center[0])/scale+anchor[0]).astype(np.float32)
+    my=((yy-center[1])/scale+anchor[1]).astype(np.float32)
+    warped=cv2.remap(delta,mx,my,cv2.INTER_CUBIC,borderMode=cv2.BORDER_CONSTANT)
+    canvas[y0:y1,x0:x1] += warped
+
+
+def main():
+    source = read_video()
+    factual = np.load(ROOT / 'source_pixels.npz')['rgb']
+    assert len(source) == len(factual) == 96
+    cue, obj = measure(source)
+    targets, vanish = dynamics(cue, obj)
+    plate = clean_plate(source)
+    # Real source texture, highlights, the rolling number patch, and shadows.
+    cue_centers = cue + np.array([.10, .20])
+    obj_centers = obj + np.array([-.18, -.85])
+    cue_ids = [0, 3, 5, 8, 10]
+    object_ids = [0] + list(range(20,36)) + list(range(56,96))
+    cue_sprites = {i:sprite(source,plate,i,cue_centers[i],'cue',vanish) for i in cue_ids}
+    obj_sprites = {i:sprite(source,plate,i,obj_centers[i],'object',vanish) for i in object_ids}
+    frames=[]
+    for t in range(96):
+        # A mass edit is not a visible size or material edit. Nothing differs
+        # until the first impact, so retain the exact source RGB prefix.
+        if t <= 13:
+            frames.append(factual[t].copy())
+            continue
+        canvas = source[t].astype(np.float32)
+        remove_original(canvas,plate,cue_centers[t],'cue',vanish)
+        remove_original(canvas,plate,obj_centers[t],'object',vanish)
+        # Render farther objects before nearer ones for correct occlusion.
+        for j in [1,0]:
+            target=targets[t,j]
+            if j == 0:
+                i=min(cue_ids,key=lambda i:abs(cue_centers[i,0]-target[0]))
+                item=cue_sprites[i]
+            else:
+                outgoing=(targets[t,1,0]-targets[t-1,1,0]) >= 0
+                choices=([0]+list(range(20,36))) if outgoing else list(range(56,96))
+                i=min(choices,key=lambda i:abs(obj_centers[i,0]-target[0]))
+                item=obj_sprites[i]
+            place(canvas,item,target,vanish)
+        rgb=np.clip(np.rint(canvas),0,255).astype(np.uint8)
+        frames.append(np.asarray(Image.fromarray(rgb).resize((256,144),Image.Resampling.LANCZOS)))
+    frames=np.asarray(frames,dtype=np.uint8)
+    artifact_io.save(frames,ROOT)
+    print('Saved:',frames.shape,frames.dtype)
+
+if __name__ == '__main__':
+    main()

def dynamics(cue, obj):
    # The table's long edges intersect at about (282,-20). A projective
    # coordinate s linearizes the rolling motion instead of treating screen
    # displacement (which shrinks with distance) as physical speed.
    vanish = 282.0
    x0 = cue[0, 0]
    to_world = lambda x: (x - x0) / (vanish - x)
    samples = [np.arange(13), np.arange(17, 42), np.arange(49, 96)]
    def residual(p):
        drag, v0, intercept, vb, intercept_back, vr = p
        a, b, c = samples
        return np.r_[
            to_world(cue[a, 0]) - (v0*a - .5*drag*a*a),
            to_world(obj[b, 0]) - (intercept + vb*b - .5*drag*b*b),
            to_world(obj[c, 0]) - (intercept_back + vr*c + .5*drag*c*c)]
    fit = least_squares(residual, [.0001, .0188, .04, .017, 1.3, -.014]).x
    drag, v0, c, vb, d, vr = fit
    initial_object = to_world(np.mean(obj[:12, 0]) - .22)
    contact_time = min(np.roots([-.5*drag, vb, c-initial_object]))
    diameter = initial_object - (v0*contact_time - .5*drag*contact_time**2)
    rail_time = min(np.roots([-drag, vb-vr, c-d]))
    far_rail = c + vb*rail_time - .5*drag*rail_time**2
    restitution = np.clip(2*(vb-drag*contact_time)/(v0-drag*contact_time)-1, .1, 1.)
    rail_restitution = np.clip(-(vr+drag*rail_time)/(vb-drag*rail_time), .1, 1.)

    # Coulomb rolling deceleration is independent of mass. The same initial
    # velocity is therefore retained; mass enters every inter-ball impulse.
    mass = np.array([5., 1.])
    p = np.array([0., initial_object])
    velocity = np.array([v0, 0.])
    trajectory = []
    events = []
    steps = 240
    dt = 1 / steps
    for frame_index in range(96):
        trajectory.append(p.copy())
        if frame_index == 95:
            break
        for sub in range(steps):
            next_v = np.sign(velocity) * np.maximum(0, np.abs(velocity)-drag*dt)
            p += .5*(velocity+next_v)*dt
            velocity = next_v
            for j in range(2):
                if p[j] > far_rail and velocity[j] > 0:
                    p[j] = 2*far_rail - p[j]
                    velocity[j] *= -rail_restitution
                    events.append((round(frame_index+sub*dt, 3), 'cushion', j))
            if p[1]-p[0] < diameter and velocity[0] > velocity[1]:
                relative = velocity[0]-velocity[1]
                impulse = (1+restitution)*relative/(1/mass[0]+1/mass[1])
                velocity[0] -= impulse/mass[0]
                velocity[1] += impulse/mass[1]
                penetration = diameter-(p[1]-p[0])
                p[0] -= penetration*mass[1]/mass.sum()
                p[1] += penetration*mass[0]/mass.sum()
                events.append((round(frame_index+sub*dt, 3), 'ball contact'))
    trajectory = np.asarray(trajectory)
    xs = (x0+vanish*trajectory)/(1+trajectory)
    # Sphere-center height and light-weight centroid correction are both
    # derived from the pre-impact cue track.
    slope, intercept = np.polyfit(cue[:13, 0], cue[:13, 1], 1)
    centers = np.stack([xs+.10, slope*(xs+.10)+intercept+.20], axis=-1)
    print('Fitted rolling drag:', drag, 'ball restitution:', restitution,
          'cushion restitution:', rail_restitution)
    print('Counterfactual contacts:', events)
    return centers, vanish


def clean_plate(source):
    ids = [0, 3, 6, 9, 12, 18, 25, 35, 55, 65, 80, 95]

    # At the original collision location the stopped cue covers some pixels

    # occluded intersection. This prevents white/black ghosts in the plate.
    x0, x1, y0, y1 = 590, 670, 315, 400
    tiles = source[:, y0:y1, x0:x1].astype(np.float32)
    cloth = np.array([114.5, 174.8, 140.8], np.float32)
    scores = np.sum((tiles-cloth)**2, axis=-1)
    best = scores.argmin(axis=0)
    yy, xx = np.mgrid[:y1-y0, :x1-x0]
    patch = tiles[best, yy, xx]
    hidden = scores.min(axis=0) > 350
    # Smooth constant cloth color is an accurate local model at this scale;
    # preserve source grain from adjacent clean pixels for the hidden area.
    texture = source[0, y0:y1, 555:555+x1-x0].astype(np.float32)
    texture += cloth-np.median(texture.reshape(-1,3), axis=0)
    patch[hidden] = texture[hidden]
    plate[y0:y1, x0:x1] = patch
    return plate


def region_mask(dx, dy, radius, kind):
    sphere = np.sqrt((dx/radius)**2+(dy/(radius*.94))**2)
    # Cast shadow goes slightly left and towards the camera.
    shadow = np.sqrt(((dx+.7*radius)/(radius*1.10))**2+
                     ((dy-radius*.88)/(radius*.66))**2)
    # Keep antialiasing and shadow penumbra sampled from the source.
    return np.maximum(np.clip((1.28-sphere)/.23,0,1),
                      np.clip((1.28-shadow)/.30,0,1)).astype(np.float32)


def sprite(source, plate, frame_index, center, kind, vanish):
    center_hi = (center+.4)*5
    # Cue centroid is very close to its sphere center; dark ball centroid
    # weights its bottom, so correct that bias when positioning a sprite.
    radius = (3.98 if kind == 'cue' else 3.85)*(vanish-center[0])/(vanish-85.48)
    half = 48
    cx, cy = np.round(center_hi).astype(int)
    x0, y0 = cx-half, cy-half
    rgb = source[frame_index, y0:cy+half+1, x0:cx+half+1].astype(np.float32)
    base = plate[y0:cy+half+1, x0:cx+half+1]
    yy, xx = np.mgrid[y0:cy+half+1, x0:cx+half+1]
    alpha = region_mask((xx-center_hi[0])/5, (yy-center_hi[1])/5, radius, kind)
    delta = (rgb-base)*alpha[...,None]
    return delta, np.array([center_hi[0]-x0, center_hi[1]-y0]), center[0]


def remove_original(canvas, plate, center, kind, vanish):
    high = (center+.4)*5
    radius = (3.98 if kind == 'cue' else 3.85)*(vanish-center[0])/(vanish-85.48)
    half = 48
    cx,cy=np.round(high).astype(int)
    x0,x1=max(0,cx-half),min(canvas.shape[1],cx+half+1)
    y0,y1=max(0,cy-half),min(canvas.shape[0],cy+half+1)
    yy,xx=np.mgrid[y0:y1,x0:x1]
    # A little larger than the extracted sphere, to remove its complete
    # antialiased outline and the soft source shadow.
    weight=region_mask((xx-high[0])/5,(yy-high[1])/5,radius*1.16,kind)
    canvas[y0:y1,x0:x1] = (canvas[y0:y1,x0:x1]*(1-weight[...,None])+
                          plate[y0:y1,x0:x1]*weight[...,None])


def place(canvas, item, target, vanish):
    delta, anchor, original_x = item
    scale = (vanish-target[0])/(vanish-original_x)
    center=(target+.4)*5
    hh,ww=delta.shape[:2]
    half=int(np.ceil(max(hh,ww)*scale*.52))
    cx,cy=np.round(center).astype(int)
    x0,x1=max(0,cx-half),min(canvas.shape[1],cx+half+1)
    y0,y1=max(0,cy-half),min(canvas.shape[0],cy+half+1)
    yy,xx=np.mgrid[y0:y1,x0:x1]
    mx=((xx-center[0])/scale+anchor[0]).astype(np.float32)
    my=((yy-center[1])/scale+anchor[1]).astype(np.float32)
    warped=cv2.remap(delta,mx,my,cv2.INTER_CUBIC,borderMode=cv2.BORDER_CONSTANT)
    canvas[y0:y1,x0:x1] += warped


def main():
    source = read_video()
    factual = np.load(ROOT / 'source_pixels.npz')['rgb']
    assert len(source) == len(factual) == 96
    cue, obj = measure(source)
    targets, vanish = dynamics(cue, obj)
    plate = clean_plate(source)
    # Real source texture, highlights, the rolling number patch, and shadows.
    cue_centers = cue + np.array([.10, .20])
    obj_centers = obj + np.array([-.18, -.85])
    cue_ids = [0, 3, 5, 8, 10]
    object_ids = [0] + list(range(20,36)) + list(range(56,96))
    cue_sprites = {i:sprite(source,plate,i,cue_centers[i],'cue',vanish) for i in cue_ids}
    obj_sprites = {i:sprite(source,plate,i,obj_centers[i],'object',vanish) for i in object_ids}
    frames=[]
    for t in range(96):
        # A mass edit is not a visible size or material edit. Nothing differs
        # until the first impact, so retain the exact source RGB prefix.
        if t <= 13:
            frames.append(factual[t].copy())
            continue
        canvas = source[t].astype(np.float32)
        remove_original(canvas,plate,cue_centers[t],'cue',vanish)
        remove_original(canvas,plate,obj_centers[t],'object',vanish)
        # Render farther objects before nearer ones for correct occlusion.
        for j in [1,0]:
            target=targets[t,j]
            if j == 0:
                i=min(cue_ids,key=lambda i:abs(cue_centers[i,0]-target[0]))
                item=cue_sprites[i]
            else:
                outgoing=(targets[t,1,0]-targets[t-1,1,0]) >= 0
                choices=([0]+list(range(20,36))) if outgoing else list(range(56,96))
                i=min(choices,key=lambda i:abs(obj_centers[i,0]-target[0]))
                item=obj_sprites[i]
            place(canvas,item,target,vanish)
        rgb=np.clip(np.rint(canvas),0,255).astype(np.uint8)
        frames.append(np.asarray(Image.fromarray(rgb).resize((256,144),Image.Resampling.LANCZOS)))
    frames=np.asarray(frames,dtype=np.uint8)
    artifact_io.save(frames,ROOT)
    print('Saved:',frames.shape,frames.dtype)

if __name__ == '__main__':
    main()
