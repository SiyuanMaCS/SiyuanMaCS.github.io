+
+No mass marker is added: mass changes the response to impact, not geometry.
+The mass change is active from t=0; before contact it has no visible effect.
+"""
+from pathlib import Path
+import numpy as np
+from PIL import Image
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+

+def bilinear(image, x, y):
+    h, w = image.shape[:2]
+    x = np.clip(x, 0, w-1.001)
+    y = np.clip(y, 0, h-1.001)
+    ix, iy = np.floor(x).astype(int), np.floor(y).astype(int)
+    fx, fy = (x-ix)[..., None], (y-iy)[..., None]
+    return ((1-fy)*((1-fx)*image[iy, ix]+fx*image[iy, ix+1]) +
+            fy*((1-fx)*image[iy+1, ix]+fx*image[iy+1, ix+1]))
+
+
+def impact_motion(times):
+    # Height-normalized domino geometry, calibrated to the source's spacing
+    # and contact angles. The first two are constrained by mutual contact.
+    thickness, spacing, gravity = .16, .66, 18.0
+    masses = np.array([1., 1., 10.])
+    inertia = (1+thickness**2)/3.
+    def angles(q):
+        b = q + np.arcsin(np.clip(spacing*np.cos(q)-thickness, -.999, .999))
+        a = b + np.arcsin(np.clip(spacing*np.cos(b)-thickness, -.999, .999))
+        return np.array([a, b, q])
+    def derivative(q):
+        e = 1e-5
+        return (angles(q+e)-angles(q-e))/(2*e)
+    def effective_inertia(q):
+        return inertia*np.sum(masses*derivative(q)**2)
+    # Inelastic impulse sharing with tenfold inertia; retain the small
+    # resultant wobble, rather than treating the heavy block as nailed down.
+    r = derivative(0.)
+    v = 1.65*(r[0]**2+r[1]**2)/(r[0]**2+r[1]**2+masses[2])
+    q, time = 0., 0.
+    samples = []
+    dt = 1/2400.
+    for target in times:
+        if target <= 0:
+            samples.append(0.)
+            continue
+        while time < target-1e-10:
+            step = min(dt, target-time)
+            rr = derivative(q)
+            II = effective_inertia(q)
+            e = 1e-4
+            dII = (effective_inertia(q+e)-effective_inertia(q-e))/(2*e)
+            torque = np.sum(masses*gravity*.5*(np.sin(angles(q))-
+                                             thickness*np.cos(angles(q)))*rr)
+            acc = (torque-.5*dII*v*v)/II - 1.2*v
+            v += step*acc
+            q += step*v
+            if q < 0:
+                # The lower face returns to the ground; low restitution
+                # removes the impact energy and the train comes to rest.
+                q = 0.
+                v = max(0., -.06*v)
+                if v < .016:
+                    v = 0.
+            time += step
+        samples.append(q)
+    return np.array(samples)
