+Only source pixels supply the setting and the two stone appearances.

+import numpy as np
+import cv2

+import artifact_io

+ROOT=Path(__file__).resolve().parent

+def source_high():
+    cap=cv2.VideoCapture(str(ROOT/'source_video.mp4'))
+    ims=[]
+    while True:
+        ok,bgr=cap.read()
+        if not ok: break

+def centers(src):
+    tracks=[]
+    yy,xx=np.indices(src.shape[1:3])
+    for i,im in enumerate(src):
+        r,g,b=im.astype(float).transpose(2,0,1)
+        expected=min(47.5+2.03*i,122.5 if i<40 else 122.5-.155*(i-40))*5
+        masks=[(r-g>90)&(r-b>75)&(abs(xx-expected)<30)&(yy>300)&(yy<385),

+            k=1+np.argmax(st[1:,4])
+            ys,xs=np.where(l==k)
+            pair.extend([xs.mean()/5,ys.mean()/5])
+        tracks.append(pair)
+    return np.asarray(tracks)

+def make_sprite(im,bg,handle):
+    # Sprite is centered on the colored handle; source-ground pixels are
+    # removed through a matte. Gray body, highlights and handle remain native.
+    cx,cy=np.round(np.asarray(handle)*5).astype(int)

+    rgb=im[y0:y0+77,x0:x0+86].astype(np.float32)

+    colored=((rr-gg>75)&(rr-bb>65))|((rr-bb>65)&(gg-bb>55))

+    contours,_=cv2.findContours(raw,cv2.RETR_EXTERNAL,cv2.CHAIN_APPROX_SIMPLE)
+    solid=np.zeros(raw.shape,np.uint8)
+    cv2.drawContours(solid,[max(contours,key=cv2.contourArea)],-1,1,-1)
+    # Include original antialiased edge samples and feather only the perimeter.
+    a=cv2.GaussianBlur(cv2.dilate(solid,np.ones((3,3),np.uint8)).astype(float),(5,5),.75)
+    a=np.maximum(a,solid.astype(float))

+    yy,xx=np.indices(a.shape)

+    return premul.astype(np.float32),a.astype(np.float32),shade.astype(np.float32),np.array([x0,y0],float)-np.asarray(handle)*5

+def paste(dst,sprite,handle,scale=1.):
+    premul,a,shadow,offset=sprite
+    # All translations occur at native 1280x720 resolution before Lanczos.
+    where=np.asarray(handle)*5+offset*scale
+    x0,y0=np.floor(where).astype(int)
+    wh=(int(np.ceil(a.shape[1]*scale))+2,int(np.ceil(a.shape[0]*scale))+2)
+    mat=np.array([[scale,0,where[0]-x0],[0,scale,where[1]-y0]],np.float32)
+    af=cv2.warpAffine(a,mat,wh,flags=cv2.INTER_LINEAR)

+    region[:]=region*(1-sh[:,:,None])
+    region[:]=region*(1-af[:,:,None])+pf

+def trajectories(p):
+    # The measured incoming red path is slightly projective. Halving world
+    # speed halves its linear term and quarters its perspective-curvature term.
+    fit=np.polyfit(np.arange(38),p[:38,0],2)

+    yc=yfit[0];yv=2*yc*38+yfit[1];y0=p[38,2]
+    def incoming(t):

+        if t<=38:

+        else:

+    # The factual first contact records the projected equal-radius spacing.
+    spacing=p[38,2]-p[38,0]
+    lo,hi=38.,70.
+    for _ in range(50):
+        mid=(lo+hi)*.5
+        rr,yy=incoming(mid)
+        if yy-rr>spacing:lo=mid
+        else:hi=mid
+    hit=(lo+hi)/2
+    rh,yh=incoming(hit)
+    u1=rv+2*rc*hit;u2=yv+2*yc*(hit-38)
+    # Estimate restitution from the factual immediate separating motion.
+    v1=np.polyfit(np.arange(41,61),p[41:61,0],1)[0]
+    v2=np.polyfit(np.arange(41,61),p[41:61,2],1)[0]
+    original_rel=(fit[1]+2*fit[0]*38)-(yfit[1]+2*yc*38)
+    restitution=np.clip((v2-v1)/original_rel,.04,.15)
+    mean=(u1+u2)/2
+    out1=mean-restitution*(u1-u2)/2
+    out2=mean+restitution*(u1-u2)/2
+    drag=.0011
+    result=[]
+    for i in range(96):
+        if i<=hit:
+            rx,yx=incoming(float(i))
+        else:
+            dt=i-hit
+            def travel(v):
+                t=min(dt,abs(v)/drag)
+                return v*t-np.sign(v)*drag*t*t*.5
+            rx=rh+travel(out1);yx=yh+travel(out2)
+        # Both glide along the same rink centerline in perspective.
+        ry=p[0,1]+.061*(rx-r0)
+        yy=p[0,3]+.061*(yx-p[0,2])
+        if i<=38:yy=p[i,3]
+        result.append([rx,ry,yx,yy])
+    print(f'contact: frame {hit+1:.2f}, red x={rh:.2f}, yellow x={yh:.2f}; '
+          f'restitution={restitution:.3f}; post velocities={out1:.3f},{out2:.3f} px/frame')
+    return np.asarray(result)
+
+def main():

+        # footage with both stone silhouettes absent (no scene regeneration).
+        for k in [0,2]:

+            x0,y0=cx-48,cy-26
+            canvas[y0:y0+85,x0:x0+98]=bg[y0:y0+85,x0:x0+98]
+        rj=int(np.argmin(abs(p[:38,0]-rx)))
+        yj=i if i<38 else min(37,int(np.argmin(abs(p[:38,2]-yx))))
+        # Retain native yellow motion until the later counterfactual impact.
+        rs=reds[rj];ys=yellows[yj]

+        # observed appearance, less than one percent of its diameter.
+        yellow_scale=1+0.00032*(yx-p[yj,2])
+        # Depth sorting: yellow is very slightly nearer the fixed camera.
+        paste(canvas,rs,[rx,ry])

+        frames.append(frame)
+    frames=np.asarray(frames,dtype=np.uint8)
+    artifact_io.save(frames,ROOT)
+    np.save(ROOT/'.tmp/counterfactual_tracks.npy',motion)
+    print('Saved',frames.shape,'to',ROOT/'pixels.npz')
+
+if __name__=='__main__':main()

\"\"\"Counterfactual curling edit: halve red's initial velocity at source time zero.
Only source pixels supply the setting and the two stone appearances.
\"\"\"
from pathlib import Path
import numpy as np
import cv2
from PIL import Image
import artifact_io

ROOT=Path(__file__).resolve().parent

def source_high():
    cap=cv2.VideoCapture(str(ROOT/'source_video.mp4'))
    ims=[]
    while True:
        ok,bgr=cap.read()
        if not ok: break
        ims.append(cv2.cvtColor(bgr,cv2.COLOR_BGR2RGB))
    cap.release()
    return np.asarray(ims)

def centers(src):
    tracks=[]
    yy,xx=np.indices(src.shape[1:3])
    for i,im in enumerate(src):
        r,g,b=im.astype(float).transpose(2,0,1)
        expected=min(47.5+2.03*i,122.5 if i<40 else 122.5-.155*(i-40))*5
        masks=[(r-g>90)&(r-b>75)&(abs(xx-expected)<30)&(yy>300)&(yy<385),
               (r-b>70)&(g-b>60)&(yy>300)&(yy<410)]
        pair=[]
        for m in masks:
            n,l,st,c=cv2.connectedComponentsWithStats(m.astype(np.uint8))
            k=1+np.argmax(st[1:,4])
            ys,xs=np.where(l==k)
            pair.extend([xs.mean()/5,ys.mean()/5])
        tracks.append(pair)
    return np.asarray(tracks)

def make_sprite(im,bg,handle):
    # Sprite is centered on the colored handle; source-ground pixels are
    # removed through a matte. Gray body, highlights and handle remain native.
    cx,cy=np.round(np.asarray(handle)*5).astype(int)
    x0,y0=cx-43,cy-24
    rgb=im[y0:y0+77,x0:x0+86].astype(np.float32)
    back=bg[y0:y0+77,x0:x0+86].astype(np.float32)
    diff=back-rgb
    dark=diff.mean(axis=2)>15
    rr,gg,bb=rgb.transpose(2,0,1)
    colored=((rr-gg>75)&(rr-bb>65))|((rr-bb>65)&(gg-bb>55))
    raw=(dark|colored).astype(np.uint8)
    raw=cv2.morphologyEx(raw,cv2.MORPH_CLOSE,np.ones((3,3),np.uint8))
    contours,_=cv2.findContours(raw,cv2.RETR_EXTERNAL,cv2.CHAIN_APPROX_SIMPLE)
    solid=np.zeros(raw.shape,np.uint8)
    cv2.drawContours(solid,[max(contours,key=cv2.contourArea)],-1,1,-1)
    # Include original antialiased edge samples and feather only the perimeter.
    a=cv2.GaussianBlur(cv2.dilate(solid,np.ones((3,3),np.uint8)).astype(float),(5,5),.75)
    a=np.maximum(a,solid.astype(float))
    # Gentle shadow outside the opaque stone, taken from its original ice.
    yy,xx=np.indices(a.shape)
    shadow_zone=((xx-43)/37)**2+((yy-39)/21)**2<1
    shade=np.clip(diff.mean(axis=2)/np.maximum(back.mean(axis=2),1),0,.2)
    shade*=shadow_zone*(1-a)
    shade=cv2.GaussianBlur(shade,(5,5),1)
    # Premultiplied foreground accounts for source background at soft edges.
    premul=rgb-(1-a[:,:,None])*back
    premul*=a[:,:,None]>0.001
    return premul.astype(np.float32),a.astype(np.float32),shade.astype(np.float32),np.array([x0,y0],float)-np.asarray(handle)*5

def paste(dst,sprite,handle,scale=1.):
    premul,a,shadow,offset=sprite
    # All translations occur at native 1280x720 resolution before Lanczos.
    where=np.asarray(handle)*5+offset*scale
    x0,y0=np.floor(where).astype(int)
    wh=(int(np.ceil(a.shape[1]*scale))+2,int(np.ceil(a.shape[0]*scale))+2)
    mat=np.array([[scale,0,where[0]-x0],[0,scale,where[1]-y0]],np.float32)
    af=cv2.warpAffine(a,mat,wh,flags=cv2.INTER_LINEAR)
    pf=cv2.warpAffine(premul,mat,wh,flags=cv2.INTER_LINEAR)
    sh=cv2.warpAffine(shadow,mat,wh,flags=cv2.INTER_LINEAR)
    region=dst[y0:y0+wh[1],x0:x0+wh[0]]
    region[:]=region*(1-sh[:,:,None])
    region[:]=region*(1-af[:,:,None])+pf

def trajectories(p):
    # The measured incoming red path is slightly projective. Halving world
    # speed halves its linear term and quarters its perspective-curvature term.
    fit=np.polyfit(np.arange(38),p[:38,0],2)
    r0=p[0,0];rv=fit[1]*.5;rc=fit[0]*.25
    yfit=np.polyfit(np.arange(15,38),p[15:38,2],2)
    yc=yfit[0];yv=2*yc*38+yfit[1];y0=p[38,2]
    def incoming(t):
        red=r0+rv*t+rc*t*t
        if t<=38:
            yellow=np.interp(t,np.arange(39),p[:39,2])
        else:
            dt=t-38;yellow=y0+yv*dt+yc*dt*dt
        return red,yellow
    # The factual first contact records the projected equal-radius spacing.
    spacing=p[38,2]-p[38,0]
    lo,hi=38.,70.
    for _ in range(50):
        mid=(lo+hi)*.5
        rr,yy=incoming(mid)
        if yy-rr>spacing:lo=mid
        else:hi=mid
    hit=(lo+hi)/2
    rh,yh=incoming(hit)
    u1=rv+2*rc*hit;u2=yv+2*yc*(hit-38)
    # Estimate restitution from the factual immediate separating motion.
    v1=np.polyfit(np.arange(41,61),p[41:61,0],1)[0]
    v2=np.polyfit(np.arange(41,61),p[41:61,2],1)[0]
    original_rel=(fit[1]+2*fit[0]*38)-(yfit[1]+2*yc*38)
    restitution=np.clip((v2-v1)/original_rel,.04,.15)
    mean=(u1+u2)/2
    out1=mean-restitution*(u1-u2)/2
    out2=mean+restitution*(u1-u2)/2
    drag=.0011
    result=[]
    for i in range(96):
        if i<=hit:
            rx,yx=incoming(float(i))
        else:
            dt=i-hit
            def travel(v):
                t=min(dt,abs(v)/drag)
                return v*t-np.sign(v)*drag*t*t*.5
            rx=rh+travel(out1);yx=yh+travel(out2)
        # Both glide along the same rink centerline in perspective.
        ry=p[0,1]+.061*(rx-r0)
        yy=p[0,3]+.061*(yx-p[0,2])
        if i<=38:yy=p[i,3]
        result.append([rx,ry,yx,yy])
    print(f'contact: frame {hit+1:.2f}, red x={rh:.2f}, yellow x={yh:.2f}; '
          f'restitution={restitution:.3f}; post velocities={out1:.3f},{out2:.3f} px/frame')
    return np.asarray(result)

def main():
    src=source_high()
    low=np.load(ROOT/'source_pixels.npz')['rgb']
    p=centers(src)
    bg=np.median(src[:38],axis=0).astype(np.uint8)
    motion=trajectories(p)
    # Use the source stone appearance at corresponding rink positions, so
    # metal reflections and the small perspective scale changes stay native.
    reds=[make_sprite(src[j],bg,p[j,:2]) for j in range(38)]
    yellows=[make_sprite(src[j],bg,p[j,2:]) for j in range(38)]
    frames=[]
    for i,(rx,ry,yx,yy) in enumerate(motion):
        if i==0:
            frames.append(low[0].copy());continue
        canvas=src[i].astype(np.float32)
        # Erase only the two factual stones. The median here is static rink
        # footage with both stone silhouettes absent (no scene regeneration).
        for k in [0,2]:
            cx,cy=np.round(p[i,k:k+2]*5).astype(int)
            x0,y0=cx-48,cy-26
            canvas[y0:y0+85,x0:x0+98]=bg[y0:y0+85,x0:x0+98]
        rj=int(np.argmin(abs(p[:38,0]-rx)))
        yj=i if i<38 else min(37,int(np.argmin(abs(p[:38,2]-yx))))
        # Retain native yellow motion until the later counterfactual impact.
        rs=reds[rj];ys=yellows[yj]
        # Near-depth correction for the yellow going beyond its last clean
        # observed appearance, less than one percent of its diameter.
        yellow_scale=1+0.00032*(yx-p[yj,2])
        # Depth sorting: yellow is very slightly nearer the fixed camera.
        paste(canvas,rs,[rx,ry])
        paste(canvas,ys,[yx,yy],yellow_scale)
        frame=np.asarray(Image.fromarray(np.uint8(np.clip(canvas,0,255))).resize((256,144),Image.Resampling.LANCZOS))
        frames.append(frame)
    frames=np.asarray(frames,dtype=np.uint8)
    artifact_io.save(frames,ROOT)
    np.save(ROOT/'.tmp/counterfactual_tracks.npy',motion)
    print('Saved',frames.shape,'to',ROOT/'pixels.npz')

if __name__=='__main__':main()
