+"""Remove the red mallet at source frame 22, including its later collision.
+
+Reconstruct the revealed table from temporal observations of the same scene.
+The original source is retained everywhere outside the intervention footprint.
+"""
+from pathlib import Path
+
+import numpy as np
+from PIL import Image, ImageDraw
+
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+
+
+def soft_ellipse(x, y, cx, cy, rx, ry, feather=1.0):
+    distance = np.sqrt(((x - cx) / rx) ** 2 + ((y - cy) / ry) ** 2)
+    return np.clip((1.0 - distance) * min(rx, ry) / feather + 0.5, 0, 1)
+
+
+def main():
+    source = np.load(ROOT / 'source_pixels.npz')['rgb']
+    output = source.copy()
+    yy, xx = np.mgrid[:source.shape[1], :source.shape[2]]
+
+    # The view is locked off. The first six observations reveal the left
+    # half of the printed surface; the last observations reveal its right
+    # half. Their overlap is unobstructed, including the diagonal logo.
+    early = np.median(source[:6], axis=0)
+    late = np.median(source[70:], axis=0)
+    blend = np.clip((xx - 122.0) / 9.0, 0.0, 1.0)[..., None]
+    plate = early * (1.0 - blend) + late * blend
+
+    # The white mallet was stationary until the now-absent red mallet
+    # reached it. Keep its last pre-intervention appearance and reflection.
+    white_rest = soft_ellipse(xx, yy, 106.5, 84, 10.5, 15.0, 1.2)
+    plate = (plate * (1 - white_rest[..., None]) +
+             source[20].astype(float) * white_rest[..., None])
+
+    for i in range(21, len(source)):
+        frame = source[i].astype(float)
+        # Locate the red object by its local colour, including antialiased
+        # rim, cast shadow and faint reflection in the replacement footprint.
+        r, g, b = frame.transpose(2, 0, 1)
+        red = ((r > 1.30 * g) & (r > 1.10 * b) &
+               (xx >= 100) & (xx <= 143) & (yy >= 60) & (yy <= 85))
+        ys, xs = np.nonzero(red)
+        cx, cy = float(xs.mean()), float(ys.mean())
+        red_mask = soft_ellipse(xx, yy, cx + 0.2, cy + 4.5,
+                                9.7, 14.0, 1.25)
+
+        # Locate the white mallet in its factual trajectory. Remove the
+        # translated object/reflection and restore its stationary original.
+        white = ((r > 220) & (g > 220) & (b > 220) &
+                 (xx >= 61) & (xx <= 113) & (yy >= 75) & (yy <= 103))
+        wy, wx = np.nonzero(white)
+        if len(wx):
+            wx0, wy0 = float(wx.mean()), float(wy.mean())
+            white_moving = soft_ellipse(xx, yy, wx0, wy0 + 4.3,
+                                       10.0, 15.0, 1.25)
+        else:
+            white_moving = np.zeros_like(xx, dtype=float)
+        alpha = np.maximum.reduce([red_mask, white_moving, white_rest])
+        output[i] = np.clip(np.rint(frame * (1 - alpha[..., None]) +
+                                  plate * alpha[..., None]), 0, 255).astype(np.uint8)
+
+    # Save every complete candidate before any subsequent visual inspection.
+    artifact_io.save(output, ROOT)
+
+    # A focused, enlarged check of the intervention and downstream effect.
+    indices = [20, 21, 24, 29, 36, 50, 70, 95]
+    sheet = Image.new('RGB', (4 * 512, 2 * 310), (25, 28, 35))
+    draw = ImageDraw.Draw(sheet)
+    for k, index in enumerate(indices):
+        x, y = (k % 4) * 512, (k // 4) * 310
+        draw.text((x + 5, y + 4), f'Source frame {index + 1}', fill='white')
+        sheet.paste(Image.fromarray(output[index]).resize((512, 288)), (x, y + 22))
+    sheet.save(ROOT / '.tmp' / 'edit_check.png')
+
+
+if __name__ == '__main__':
+    main()
