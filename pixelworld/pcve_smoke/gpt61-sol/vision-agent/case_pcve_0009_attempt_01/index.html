+def track(images):
+    # Fixed-camera template tracking, with local scale refinement for perspective.
+    boxes = {'white': (497, 380, 570, 428),
+             'red': (674, 317, 734, 358),
+             'blue': (810, 266, 870, 306)}
+    searches = {'white': (290, 365, 585, 510),
+                'red': (520, 310, 745, 425),
+                'blue': (690, 260, 890, 365)}
+    offsets = {'white': 12., 'red': 9., 'blue': 10.}
+    result = {}
+    for name, (x, y, x2, y2) in boxes.items():
+        template = images[0, y:y2, x:x2]
+        sx, sy, sx2, sy2 = searches[name]
+        values = []
+        for im in images:
+            best = (-1., None)
+            for scale in np.arange(1., 1.241, .02):
+                t = cv2.resize(template, None, fx=scale, fy=scale,
+                               interpolation=cv2.INTER_LINEAR)

+                                             cv2.TM_CCOEFF_NORMED)
+                _, score, _, pos = cv2.minMaxLoc(response)
+                if score > best[0]:

+                                    pos[1] + sy + t.shape[0] - offsets[name] * scale,
+                                    scale))
+            values.append(best[1])
+        result[name] = np.asarray(values)
+    return result

+def footprint(shape, pos, name, feather=2.):
+    """Conservative source-pixel removal footprint including contact shadows."""
+    x, y, scale = pos
+    h, w = shape[:2]
+    mask = np.zeros((h, w), np.uint8)
+    rx = {'white': 42, 'red': 38, 'blue': 36}[name] * scale
+    # Rounded footprint includes stem and very faint table reflection.
+    cv2.ellipse(mask, (round(x), round(y - 7 * scale)),
+                (round(rx), round(32 * scale)), 0, 0, 360, 255, -1)
+    cv2.rectangle(mask, (round(x - 19 * scale), round(y - 39 * scale)),
+                  (round(x + 19 * scale), round(y)), 255, -1)
+    return cv2.GaussianBlur(mask, (0, 0), feather).astype(np.float32) / 255.

+def clean_table(images, tracks):
+    # Select actual unobstructed pixels, never temporal averages of mallets.
+    bg = images[0].copy()
+    missing = np.zeros(images.shape[1:3], np.uint8)
+    for name in tracks:
+        missing |= (footprint(bg.shape, tracks[name][0], name) > .002).astype(np.uint8)
+    for i in [95, 50, 35, 22, 12, 5, 65, 28, 17, 40]:
+        occupied = np.zeros_like(missing)
+        for name in tracks:
+            occupied |= (footprint(bg.shape, tracks[name][i], name) > .002).astype(np.uint8)
+        available = (missing != 0) & (occupied == 0)
+        bg[available] = images[i][available]
+        missing[available] = 0
+    # Only tiny permanently-occluded seams, if any, require spatial repair.
+    if missing.any():
+        bg = cv2.inpaint(bg, missing * 255, 3, cv2.INPAINT_TELEA)
+    return bg

+def green_sprite(images):
+    x0, y0 = 802, 256
+    crop = images[0, y0:330, x0:880].copy()
+    z = crop.astype(np.float32)
+    m = ((z[..., 1] - z[..., 0] > 8) &
+         (z[..., 2] - z[..., 0] > 25) &
+         (z[..., 2] - z[..., 1] > 3)).astype(np.uint8)
+    m[49:] = 0  # no reflection in the solid-body sprite
+    n, labels, stats, _ = cv2.connectedComponentsWithStats(m)
+    m = (labels == (1 + np.argmax(stats[1:, 4]))).astype(np.uint8) * 255
+    m = cv2.morphologyEx(m, cv2.MORPH_CLOSE, np.ones((3, 3), np.uint8))
+    contours, _ = cv2.findContours(m, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
+    cv2.drawContours(m, contours, -1, 255, -1)

+    alpha = cv2.GaussianBlur(m.astype(np.float32) / 255., (0, 0), .65)
+    hsv = cv2.cvtColor(crop, cv2.COLOR_RGB2HSV).astype(np.float32)

+    hsv[..., 0] = 58
+    hsv[..., 1] *= .91
+    hsv[..., 2] *= .97

+    return colored.astype(np.float32), alpha, np.array([839. - x0, 296. - y0])

+def draw_green(frame, sprite, alpha, anchor, position, scale):

+    # A soft contact shadow and subdued green reflection on the glossy table.
+    shadow = np.zeros(frame.shape[:2], np.float32)

+                (round(27 * scale), round(5 * scale)), 0, 0, 360, .14, -1)
+    shadow = cv2.GaussianBlur(shadow, (0, 0), 2.2 * scale)
+    frame *= 1. - shadow[..., None]

+                  - ((yy - y - 9 * scale) / (8 * scale)) ** 2) * .072
+    refl *= (yy > y + 3 * scale)
+    frame[:] = frame * (1 - refl[..., None]) + np.array([80., 172., 90.]) * refl[..., None]
+    mat = np.float32([[scale, 0, x - anchor[0] * scale],
+                      [0, scale, y - anchor[1] * scale]])
+    size = (frame.shape[1], frame.shape[0])
+    # Warp premultiplied colors to avoid colored interpolation fringes.
+    a = cv2.warpAffine(alpha, mat, size, flags=cv2.INTER_LINEAR)
+    p = cv2.warpAffine(sprite * alpha[..., None], mat, size, flags=cv2.INTER_LINEAR)
+    frame[:] = frame * (1 - a[..., None]) + p

+    x, y, scale = position
+    # A soft crop; table markings underneath are in exactly the same location.
+    mask = footprint(frame.shape, position, name, feather=1.5)
+    frame[:] = frame * (1 - mask[..., None]) + source.astype(np.float32) * mask[..., None]

+                                if tracks['white'][i, 0] < tracks['white'][0, 0] - 3)
+    advance = max(0, original_white_start - contact)
+    print('new chain contact', contact, 'white time advance', advance)
+    blue_reference = to_world(np.array([839., 296.]))

+                               footprint(canvas.shape, tracks['white'][i], 'white'))
+            canvas = canvas * (1 - erase[..., None]) + bg.astype(np.float32) * erase[..., None]
+            red_index = contact

+            paste_source_region(canvas, high[red_index], red, 'red')
+            paste_source_region(canvas, high[white_index], white, 'white')

+        p = to_image(midpoint)
+        denominator = plane[2] @ np.r_[midpoint, 1]

+        edited.append(result)

+    writer.release()
+    np.save(ROOT / '.tmp' / 'edit_positions.npy', np.asarray(positions))
+    print('Saved', edited.shape)

    x, y, scale = pos
    h, w = shape[:2]
    mask = np.zeros((h, w), np.uint8)
    rx = {'white': 42, 'red': 38, 'blue': 36}[name] * scale
    # Rounded footprint includes stem and very faint table reflection.
    cv2.ellipse(mask, (round(x), round(y - 7 * scale)),
                (round(rx), round(32 * scale)), 0, 0, 360, 255, -1)
    cv2.rectangle(mask, (round(x - 19 * scale), round(y - 39 * scale)),
                  (round(x + 19 * scale), round(y)), 255, -1)
    return cv2.GaussianBlur(mask, (0, 0), feather).astype(np.float32) / 255.


def clean_table(images, tracks):
    # Select actual unobstructed pixels, never temporal averages of mallets.
    bg = images[0].copy()
    missing = np.zeros(images.shape[1:3], np.uint8)
    for name in tracks:
        missing |= (footprint(bg.shape, tracks[name][0], name) > .002).astype(np.uint8)
    for i in [95, 50, 35, 22, 12, 5, 65, 28, 17, 40]:
        occupied = np.zeros_like(missing)
        for name in tracks:
            occupied |= (footprint(bg.shape, tracks[name][i], name) > .002).astype(np.uint8)
        available = (missing "'!= 0) & (occupied == 0)
        bg[available] = images[i][available]
        missing[available] = 0
    # Only tiny permanently-occluded seams, if any, require spatial repair.
    if missing.any():
        bg = cv2.inpaint(bg, missing * 255, 3, cv2.INPAINT_TELEA)
    return bg


def green_sprite(images):
    x0, y0 = 802, 256
    crop = images[0, y0:330, x0:880].copy()
    z = crop.astype(np.float32)
    m = ((z[..., 1] - z[..., 0] > 8) &
         (z[..., 2] - z[..., 0] > 25) &
         (z[..., 2] - z[..., 1] > 3)).astype(np.uint8)
    m[49:] = 0  # no reflection in the solid-body sprite
    n, labels, stats, _ = cv2.connectedComponentsWithStats(m)
    m = (labels == (1 + np.argmax(stats[1:, 4]))).astype(np.uint8) * 255
    m = cv2.morphologyEx(m, cv2.MORPH_CLOSE, np.ones((3, 3), np.uint8))
    contours, _ = cv2.findContours(m, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
    cv2.drawContours(m, contours, -1, 255, -1)
    m = cv2.dilate(m, cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (3, 3)))
    alpha = cv2.GaussianBlur(m.astype(np.float32) / 255., (0, 0), .65)
    hsv = cv2.cvtColor(crop, cv2.COLOR_RGB2HSV).astype(np.float32)
    # Preserve diffuse shading, gloss, and saturation, changing blue to green.
    hsv[..., 0] = 58
    hsv[..., 1] *= .91
    hsv[..., 2] *= .97
    colored = cv2.cvtColor(np.uint8(np.clip(hsv, 0, 255)), cv2.COLOR_HSV2RGB)
    return colored.astype(np.float32), alpha, np.array([839. - x0, 296. - y0])


def draw_green(frame, sprite, alpha, anchor, position, scale):
    x, y = position
    # A soft contact shadow and subdued green reflection on the glossy table.
    shadow = np.zeros(frame.shape[:2], np.float32)
    cv2.ellipse(shadow, (round(x + 3 * scale), round(y + 4 * scale)),
                (round(27 * scale), round(5 * scale)), 0, 0, 360, .14, -1)
    shadow = cv2.GaussianBlur(shadow, (0, 0), 2.2 * scale)
    frame *= 1. - shadow[..., None]
    yy, xx = np.indices(frame.shape[:2], dtype=np.float32)
    refl = np.exp(-((xx - x - 1.8 * scale) / (22 * scale)) ** 2
                  - ((yy - y - 9 * scale) / (8 * scale)) ** 2) * .072
    refl *= (yy > y + 3 * scale)
    frame[:] = frame * (1 - refl[..., None]) + np.array([80., 172., 90.]) * refl[..., None]
    mat = np.float32([[scale, 0, x - anchor[0] * scale],
                      [0, scale, y - anchor[1] * scale]])
    size = (frame.shape[1], frame.shape[0])
    # Warp premultiplied colors to avoid colored interpolation fringes.
    a = cv2.warpAffine(alpha, mat, size, flags=cv2.INTER_LINEAR)
    p = cv2.warpAffine(sprite * alpha[..., None], mat, size, flags=cv2.INTER_LINEAR)
    frame[:] = frame * (1 - a[..., None]) + p


def paste_source_region(frame, source, position, name):
    """Reuse the source object'"'s real material, shading, and local reflection.\"\"\"
    x, y, scale = position
    # A soft crop; table markings underneath are in exactly the same location.
    mask = footprint(frame.shape, position, name, feather=1.5)
    frame[:] = frame * (1 - mask[..., None]) + source.astype(np.float32) * mask[..., None]


def generate():
    original = np.load(ROOT / 'source_pixels.npz')['rgb']
    high = load_video()
    assert len(high) == 96
    tracks = track(high)
    bg = clean_table(high, tracks)
    sprite, alpha, anchor = green_sprite(high)
    # Plane coordinates allow a genuine tabletop midpoint, rather than a
    # perspective-biased midpoint between the tops of the handles.
    plane = cv2.getPerspectiveTransform(
        np.float32([[0, 0], [1, 0], [1, 2], [0, 2]]),
        np.float32([[40, 443], [610, 565], [1179, 278], [754, 224]]))
    inverse = np.linalg.inv(plane)
    def to_world(p):
        return cv2.perspectiveTransform(np.float32([[p[:2]]]), inverse)[0, 0]
    def to_image(p):
        return cv2.perspectiveTransform(np.float32([[p]]), plane)[0, 0]
    # The new intervening mallet intercepts red when the three bases meet.
    initial_white = to_world(tracks['white'][0])
    contact = next(i for i in range(14, 40)
                   if to_world(tracks['red'][i])[1] - initial_white[1] <= .23)
    # White receives the original impulse sooner, while red comes to rest at
    # this new contact point.  Its source rail rebound remains preserved.
    original_white_start = next(i for i in range(20, 40)
                                if tracks['white'][i, 0] < tracks['white'][0, 0] - 3)
    advance = max(0, original_white_start - contact)
    print('new chain contact', contact, 'white time advance', advance)
    blue_reference = to_world(np.array([839., 296.]))
    denominator_ref = plane[2] @ np.r_[blue_reference, 1]
    edited = []
    positions = []
    for i in range(96):
        canvas = high[i].astype(np.float32)
        if i >= contact:
            erase = np.maximum(footprint(canvas.shape, tracks['red'][i], 'red'),
                               footprint(canvas.shape, tracks['white'][i], 'white'))
            canvas = canvas * (1 - erase[..., None]) + bg.astype(np.float32) * erase[..., None]
            red_index = contact
            white_index = min(95, i + advance)
            red = tracks['red'][red_index]
            white = tracks['white'][white_index]
            paste_source_region(canvas, high[red_index], red, 'red')
            paste_source_region(canvas, high[white_index], white, 'white')
        else:
            red, white = tracks['red'][i], tracks['white'][i]
        midpoint = (to_world(red) + to_world(white)) * .5
        p = to_image(midpoint)
        denominator = plane[2] @ np.r_[midpoint, 1]
        scale = denominator_ref / denominator
        draw_green(canvas, sprite, alpha, anchor, p, scale)
        # Preserve supplied RGB exactly outside the composited source-pixel edit.
        delta = cv2.resize(canvas - high[i].astype(np.float32), (256, 144),
                           interpolation=cv2.INTER_AREA)
        result = np.uint8(np.clip(np.rint(original[i].astype(np.float32) + delta), 0, 255))
        edited.append(result)
        positions.append([*red[:2], *white[:2], *p])
    edited = np.asarray(edited, dtype=np.uint8)
    artifact_io.save(edited, ROOT)
    # Useful native-resolution export in addition to the required RGB artifacts.
    writer = cv2.VideoWriter(str(ROOT / 'edited_video.mp4'),
                             cv2.VideoWriter_fourcc(*'mp4v'), 24., (256, 144))
    for image in edited:
        writer.write(cv2.cvtColor(image, cv2.COLOR_RGB2BGR))
    writer.release()
    np.save(ROOT / '.tmp' / 'edit_positions.npy', np.asarray(positions))
    print('Saved', edited.shape)


if __name__ == '__main__':
    generate()
