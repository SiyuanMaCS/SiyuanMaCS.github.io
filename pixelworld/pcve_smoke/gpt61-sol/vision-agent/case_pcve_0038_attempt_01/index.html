+"""Source-pixel counterfactual: double launch velocity, not gravity or drag.
+
+The source's rolling and airborne coordinates are fitted in a perspective
+coordinate. All rendering uses source imagery and elementary image warping.
+"""
+from pathlib import Path
+import math
+
+import cv2
+import numpy as np
+from PIL import Image
+from scipy.ndimage import distance_transform_edt
+from scipy.optimize import brentq, least_squares
+
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+
+def read_source():
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    frames = []
+    while True:
+        ok, frame = cap.read()
+        if not ok:
+            break
+        frames.append(cv2.cvtColor(frame, cv2.COLOR_BGR2RGB))
+    cap.release()
+    return np.stack(frames)
+
+
+def sprite(source, background, index, box, polygon, anchor):
+    x0, y0, x1, y1 = box
+    fg = source[index, y0:y1, x0:x1].astype(np.float32)
+    bg = background[y0:y1, x0:x1].astype(np.float32)
+    diff = np.max(np.abs(fg - bg), axis=2)
+    mask = (diff > 13).astype(np.uint8)
+    mask = cv2.morphologyEx(mask, cv2.MORPH_CLOSE, np.ones((3, 3), np.uint8))
+    contours, _ = cv2.findContours(mask, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
+    mask[:] = 0
+    cv2.drawContours(mask, [max(contours, key=cv2.contourArea)], -1, 1, -1)
+    interior = cv2.erode(mask, np.ones((3, 3), np.uint8))
+    _, nearest = distance_transform_edt(1 - interior, return_indices=True)
+    nearby_color = fg[nearest[0], nearest[1]]
+    vector = nearby_color - bg
+    coverage = np.clip(np.sum((fg - bg) * vector, axis=2) /
+                       np.maximum(np.sum(vector * vector, axis=2), 1), 0, 1)
+    alpha = np.where(interior > 0, 1, coverage).astype(np.float32)
+    alpha *= cv2.dilate(mask, np.ones((3, 3), np.uint8))
+    silhouette = np.zeros(mask.shape, np.float32)
+    poly = np.array(polygon, np.int32) - [x0, y0]
+    cv2.fillPoly(silhouette, [poly.astype(np.int32)], 1)
+    silhouette = cv2.GaussianBlur(silhouette, (3, 3), .5)
+    alpha *= silhouette
+    # Unmix edge pixels from their original background before moving them.
+    color = np.clip(bg + (fg - bg) / np.maximum(alpha[..., None], .13), 0, 255)
+    return color.astype(np.float32), alpha, np.array(anchor) - [x0, y0]
+
+
+def composite_sprite(canvas, spr, x, y, scale, angle):
+    color, alpha, anchor = spr
+    theta = math.radians(angle)
+    co, si = math.cos(theta) * scale, math.sin(theta) * scale
+    matrix = np.array([[co, -si, x - co * anchor[0] + si * anchor[1]],
+                       [si, co, y - si * anchor[0] - co * anchor[1]]], np.float32)
+    premult = color * alpha[..., None]
+    h, w = canvas.shape[:2]
+    rgb = cv2.warpAffine(premult, matrix, (w, h), flags=cv2.INTER_LINEAR)
+    a = cv2.warpAffine(alpha, matrix, (w, h), flags=cv2.INTER_LINEAR)
+    return canvas * (1 - a[..., None]) + rgb
+
+
+def shadow(canvas, x, wheel_y, scale, opacity, clearance=0):
+    # Contact shadow at the actual supporting plane, softer during free fall.
+    h, w = canvas.shape[:2]
+    layer = np.zeros((h, w), np.float32)
+    center = (round(x + 3 * scale), round(wheel_y + 3 * scale))
+    radii = (max(1, round(47 * scale)), max(1, round((6 + clearance * .04) * scale)))
+    cv2.ellipse(layer, center, radii, -2.65, 0, 360, opacity, -1)
+    layer = cv2.GaussianBlur(layer, (0, 0), 2.3 + clearance * .065)
+    return canvas * (1 - layer[..., None])
+
+
+def main():
+    high = read_source()
+    original = np.load(ROOT / 'source_pixels.npz')['rgb']
+    # The camera and furniture are static. Temporal median removes every car
+    # location, including the starting position and the final resting position.
+    background = np.median(high[:18], axis=0).astype(np.uint8)
+    start = sprite(high, background, 0, (282, 120, 403, 176),
+                   [(283,149),(287,136),(294,130),(337,129),(349,130),
+                    (363,142),(381,144),(391,149),(395,157),(395,165),
+                    (387,167),(376,171),(369,167),(367,165),(315,167),
+                    (310,173),(298,175),(291,169),(287,165),(282,163)],
+                   (337.33,154.15))
+    flat = sprite(high, background, 20, (895,190,1008,255),
+                  [(906,224),(909,212),(916,208),(930,206),(961,207),
+                   (971,213),(981,218),(993,220),(998,228),(998,237),
+                   (993,242),(983,243),(977,238),(938,241),(934,245),
+                   (921,245),(917,241),(907,236)],
+                  (953.18,229.31))
+
+    # Projective fit to source car motion before landing.
+    times = np.arange(7, dtype=float)
+    xs = np.array([337.33,393.89,448.23,500.28,553.71,603.35,651.85])
+    M = np.stack([np.ones(7), times, -xs * times], axis=1)
+    pa, pb, pc = np.linalg.lstsq(M, xs, rcond=None)[0]
+    X = lambda q: (pa + pb * q) / (1 + pc * q)
+    inv_x = lambda x: (pa - x) / (pc * x - pb)
+    ys = np.array([154.15,154.2,152.57,154.93,168.04,190.52,223.28])
+    def residual(p):
+        return (p[0] + p[1]*times + p[2]*np.maximum(0,times-p[3])**2) / (1+pc*times) - ys
+    p = least_squares(residual, [154,1.5,5.5,2.1]).x
+    floor_y = lambda q: 273.427219 - .0462405513 * X(q)
+    airborne_y = lambda t: (p[0] + p[1]*2*t + p[2]*max(0,t-p[3]/2)**2)/(1+pc*2*t)
+    land = brentq(lambda t: airborne_y(t)-floor_y(2*t), 2, 8)
+    original_land = brentq(lambda t: (p[0]+p[1]*t+p[2]*(t-p[3])**2)/(1+pc*t)-floor_y(t), 3, 8)
+    # The same rolling deceleration inferred from the factual stopping path.
+    ground_x = np.array([693.49,727.87,759.73,788.61,815.37,839.04,860.37,
+                         880.27,897.46,913.33,926.63,936.44,945.75,953.18,
+                         957.99,961.08,962.26])
+    fit = np.polyfit(np.arange(7,24), inv_x(ground_x), 2)
+    drag = -2 * fit[0]
+    impact_loss = 1 - np.polyval(np.polyder(fit), original_land)
+    velocity = 2 - impact_loss
+    def q_motion(t):
+        if t <= land:
+            return 2*t
+        elapsed = min(t-land, velocity/drag)
+        return 2*land + velocity*elapsed - .5*drag*elapsed**2
+    edge = brentq(lambda t: X(q_motion(t))-1300, land, land+20)
+
+    result = []
+    for frame in range(96):
+        if frame == 0:
+            result.append(original[0].copy())
+            continue
+        t = float(frame)
+        q = q_motion(t)
+        x = X(q)
+        base = background.astype(np.float32).copy()
+        # All car motion is to the right; it passes in front of, not through,
+        # the vase (its path is several centimeters forward of the vase base).
+        if x < 1400:
+            scale = 1/(1+pc*q)
+            if t < land:
+                y = airborne_y(t)
+                flight = max(0, t-p[3]/2)
+                pitch = min(18, flight*4.1)
+                if x > 605:
+                    clearance = max(0, floor_y(q)-y)
+                    base = shadow(base, x, floor_y(q)+13*scale, scale,
+                                  .21*math.exp(-clearance/65), clearance)
+                elif q < p[3]:
+                    base = shadow(base, x, y+17*scale, scale, .11)
+                base = composite_sprite(base, start, x, y, scale, pitch)
+            else:
+                y = floor_y(q)
+                pitch = -1.2*math.exp(-(t-land)/.75)
+                flat_scale = (1+pc*inv_x(953.18))/(1+pc*q)
+                if t > edge:
+                    # Leaving the table starts a fresh ballistic fall. It is
+                    # subsequently offscreen; no in-view furniture is struck.
+                    dt = t-edge
+                    y += p[2]*dt*dt/(1+pc*q)
+                    pitch += min(55, 8*dt)
+                else:
+                    base = shadow(base, x, y+12*flat_scale, flat_scale, .23)
+                base = composite_sprite(base, flat, x, y, flat_scale, pitch)
+        rgb = np.clip(np.rint(base), 0, 255).astype(np.uint8)
+        result.append(np.array(Image.fromarray(rgb).resize((256,144), Image.Resampling.LANCZOS)))
+    frames = np.stack(result)
+    artifact_io.save(frames, ROOT)
+    print(f'Saved {len(frames)} frames; launch 2x, landing {land/24:.3f}s, table exit {edge/24:.3f}s')
+
+
+if __name__ == '__main__':
+    main()
