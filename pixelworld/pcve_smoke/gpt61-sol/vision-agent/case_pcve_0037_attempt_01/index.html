+"""Counterfactual toy-car motion, using only source pixels and analytic retiming.
+
+The support phase is slowed to .75 speed. Flight retains its gravity clock,
+with .75 horizontal velocity. On landing the same rolling resistance gives
+.75 stopping time and .75**2 rolling displacement. Sprites and static scene
+are sampled from the supplied video, not generated from an external model.

+import cv2
+import numpy as np
+from scipy import ndimage
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+def read_video():
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    frames = []
+    while True:
+        ok, bgr = cap.read()
+        if not ok:
+            break
+        frames.append(cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB))
+    cap.release()
+    return np.asarray(frames)
+
+

+    cy = np.array([154,154,153,155,168,190,223,238,237,235,234,233,232,
+                   231,230,229,229,228,227,227,226,226,226,226.], float)
+    sprites = []
+    # Align each sprite around its body, preserving source geometry, texture,
+    # antialiasing and its local contact shadow.
+    for k in range(24):

+        patch = high[k,y0:y0+104,x0:x0+160].astype(float)

+        if k <= 2:

+            mask[(np.arange(104)+y0) >= 185] = False
+        # Source codec noise isn't part of the movable object.
+        mask = ndimage.binary_closing(mask, iterations=2)
+        lab, count = ndimage.label(mask)
+        if count:

+        rgba = cv2.warpAffine(rgba,mat,(160,104),flags=cv2.INTER_LINEAR)
+        sprites.append(rgba)
+

+        out.append(frame)
