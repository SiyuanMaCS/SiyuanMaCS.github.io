+"""Counterfactual curling sequence, constructed solely from the supplied pixels.
+
+The new stone starts at the one-quarter point. Equal-mass impacts use the
+low restitution visible in the original red/yellow collision. Source pixels
+are retained until each original stone's trajectory is affected.
+"""
+from pathlib import Path
+import cv2
+import numpy as np
+from PIL import Image
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+
+def read_high():
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    images = []
+    while True:
+        ok, bgr = cap.read()
+        if not ok:
+            break
+        images.append(cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB))
+    cap.release()
+    return np.stack(images)

+def stone_sprite(high, blue=False, yellow=False):
+    if yellow:
+        origin = np.array([1024., 354.])
+        anchor = np.array([1068., 398.])
+        patch = high[0,354:434,1024:1112].copy()
+        outline = [[1055,374],[1073,373],[1084,378],[1091,385],
+                   [1093,395],[1091,403],[1086,409],[1076,412],
+                   [1060,412],[1049,408],[1043,401],[1041,393],
+                   [1043,383],[1048,378]]
+    else:
+        origin = np.array([198., 305.])
+        anchor = np.array([239., 346.])
+        patch = high[0,305:385,198:280].copy()
+        outline = [[226,322],[242,321],[258,328],[262,335],
+                   [263,343],[261,350],[255,355],[243,359],
+                   [230,359],[220,353],[215,348],[214,340],
+                   [215,335],[220,329]]
+    mask = np.zeros(patch.shape[:2], np.uint8)
+    poly = np.asarray(outline, np.int32) - origin.astype(np.int32)
+    cv2.fillPoly(mask, [poly], 255)
+    alpha = cv2.GaussianBlur(mask, (3,3), .6).astype(np.float32)/255.
+    # Remove the few rink-line pixels at the edge of the cutout.
+    yy, xx = np.indices(alpha.shape)
+    r,g,b = patch.astype(float).transpose(2,0,1)
+    edge_line = (yy + origin[1] > anchor[1]-4) & (r-g > 10) & (r-b > 3)
+    gray = (g+b)/2
+    patch[edge_line] = np.repeat(gray[edge_line,None], 3, axis=1).astype(np.uint8)
+    if blue:
+        hsv = cv2.cvtColor(patch, cv2.COLOR_RGB2HSV)
+        tint = (hsv[:,:,1] > 52) & ((hsv[:,:,0]<19)|(hsv[:,:,0]>165))
+        tint &= yy + origin[1] < anchor[1]-4
+        hsv[:,:,0][tint] = 107
+        patch = cv2.cvtColor(hsv, cv2.COLOR_HSV2RGB)
+    return patch.astype(np.float32), alpha, anchor-origin
+
+
+def factual_positions(high):
+    positions=[]
+    for t, image in enumerate(high):
+        image=image.astype(float)
+        r,g,b=image.transpose(2,0,1)
+        Y,X=np.indices(r.shape)
+        guess=47.8+1.98*t if t<38 else 122.8-.15*(t-38)
+        red=(r-g>40)&(r-b>30)&(abs(X-guess*5)<23)
+        red &= abs(Y-(66.4+(guess-47.8)*.0625)*5)<13
+        yellow=(r-b>65)&(g-b>55)&(Y>330)&(Y<410)&(X>580)
+        values=[]
+        for mask in (red,yellow):
+            y,x=np.where(mask)
+            values.append([x.mean()/5,y.mean()/5])
+        positions.append(values)
+    p=np.array(positions)
+    # Centers on the ice, accounting for handle elevation.
+    p[:,0,1]+=2.8
+    p[:,1,1]+=2.8
+    return p
+
+
+def simulate(factual):
+    starts=factual[0]
+    p=np.array([starts[0,0], .75*starts[0,0]+.25*starts[1,0],starts[1,0]])
+    v=np.array([np.mean(np.diff(factual[:16,0,0])),0.,np.mean(np.diff(factual[:20,1,0]))])
+    radius_sum=10.2
+    restitution=.09
+    substeps=160
+    dt=1/substeps
+    results=[]
+    first_hit=[None,None,None]
+    for t in range(96):
+        results.append(p.copy())
+        for s in range(substeps):
+            # Very slight sliding resistance, fitted to the late source drift.
+            v-=np.sign(v)*np.minimum(abs(v), .0009*dt)
+            p+=v*dt
+            for i in (0,1):
+                if p[i+1]-p[i]<radius_sum:
+                    d=(radius_sum-(p[i+1]-p[i]))/2
+                    p[i]-=d
+                    p[i+1]+=d
+                    if v[i]>v[i+1]:
+                        u,w=v[i],v[i+1]
+                        v[i]=((1-restitution)*u+(1+restitution)*w)/2
+                        v[i+1]=((1+restitution)*u+(1-restitution)*w)/2
+                        for j in (i,i+1):
+                            if first_hit[j] is None:
+                                first_hit[j]=t+(s+1)*dt
+    positions=np.array(results)
+    # The initial trajectory lies on the red line in the oblique camera.
+    slope=(starts[1,1]-starts[0,1])/(starts[1,0]-starts[0,0])
+    all_positions=np.stack([positions,starts[0,1]+slope*(positions-starts[0,0])],-1)
+    return all_positions,first_hit
+
+
+def blend_sprite(image, sprite, point):
+    rgb,alpha,anchor=sprite
+    # Small change in apparent size along the receding plane.
+    scale=1.+(point[0]-47.8)*.00017
+    if scale!=1:
+        rgb=cv2.resize(rgb,None,fx=scale,fy=scale,interpolation=cv2.INTER_LINEAR)
+        alpha=cv2.resize(alpha,(rgb.shape[1],rgb.shape[0]),interpolation=cv2.INTER_LINEAR)
+        anchor=anchor*scale
+    x,y=point*5-anchor
+    ix,iy=int(np.floor(x)),int(np.floor(y))
+    matrix=np.float32([[1,0,x-ix],[0,1,y-iy]])
+    rgb=cv2.warpAffine(rgb,matrix,(rgb.shape[1],rgb.shape[0]),flags=cv2.INTER_LINEAR)
+    alpha=cv2.warpAffine(alpha,matrix,(alpha.shape[1],alpha.shape[0]),flags=cv2.INTER_LINEAR)
+    h,w=alpha.shape
+    region=image[iy:iy+h,ix:ix+w]
+    region[:]=region*(1-alpha[:,:,None])+rgb*alpha[:,:,None]
+
+
+def add_shadow(image,point):
+    cx,cy=point*5
+    x0,x1=int(cx-47),int(cx+47)
+    y0,y1=int(cy-17),int(cy+40)
+    Y,X=np.mgrid[y0:y1,x0:x1]
+    # Contact and soft reflected darkness, akin to the original stones.
+    a=.09*np.exp(-((X-cx-2)/24)**2-((Y-cy-11)/10)**2)
+    a+=.055*np.exp(-((X-cx)/25)**2-((Y-cy-4)/4)**2)
+    image[y0:y1,x0:x1]*=1-a[:,:,None]
+
+
+def main():
+    source=np.load(ROOT/'source_pixels.npz')['rgb']
+    high=read_high()
+    clean=np.median(high[:30],axis=0).astype(np.float32)
+    factual=factual_positions(high)
+    positions,hits=simulate(factual)
+    sprites=[stone_sprite(high),stone_sprite(high,blue=True),stone_sprite(high,yellow=True)]
+    frames=[]
+    for t in range(96):
+        frame=high[t].astype(np.float32)
+        modified=[]
+        if t>hits[0]: modified.append((0,0))
+        if t>hits[2]: modified.append((2,1))
+        # Erase only original stones whose paths change. A smooth local mask
+        # restores the fixed ice, rings, line and illumination from clean frames.
+        erase=np.zeros(frame.shape[:2],np.float32)
+        for obj,src in modified:
+            cx,cy=factual[t,src]*5
+            x0,x1=int(cx-49),int(cx+49)
+            y0,y1=int(cy-42),int(cy+42)
+            erase[y0:y1,x0:x1]=1
+        if modified:
+            erase=cv2.GaussianBlur(erase,(11,11),2.)
+            frame=frame*(1-erase[:,:,None])+clean*erase[:,:,None]
+        objects=[1]+[o for o,s in modified]
+        for o in objects:
+            add_shadow(frame,positions[t,o])
+        for o in sorted(objects,key=lambda o:positions[t,o,1]):
+            blend_sprite(frame,sprites[o],positions[t,o])
+        reduced=cv2.resize(np.clip(frame,0,255).astype(np.uint8),(256,144),interpolation=cv2.INTER_AREA)
+        # Preserve the original RGB exactly outside intervention neighborhoods.
+        affected=erase.copy()
+        for o in objects:
+            cx,cy=positions[t,o]*5
+            x0,x1=int(cx-57),int(cx+57)
+            y0,y1=int(cy-45),int(cy+45)
+            affected[y0:y1,x0:x1]=1
+        lowmask=cv2.resize(affected,(256,144),interpolation=cv2.INTER_AREA)
+        output=source[t].copy()
+        output[lowmask>.001]=reduced[lowmask>.001]
+        frames.append(output)
+    frames=np.asarray(frames,dtype=np.uint8)
+    artifact_io.save(frames,ROOT)
+    print('Saved',frames.shape,'; first collisions (source frame offset):',hits)
+
+
+if __name__=='__main__':
+    main()
