import cv2
import numpy as np
from PIL import Image
import artifact_io

ROOT = Path(__file__).resolve().parent

def read_high():
    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
    result=[]
    while True:
        ok,frame=cap.read()
        if not ok: break
        result.append(cv2.cvtColor(frame,cv2.COLOR_BGR2RGB))
    cap.release()
    return np.asarray(result)

def polygon_alpha(shape, points):
    a=np.zeros(shape[:2],np.uint8)
    cv2.fillPoly(a,[np.round(points).astype(np.int32)],255)
    return cv2.GaussianBlur(a.astype(np.float32)/255,(3,3),0.55)

def over(image, rgb, alpha, matrix):
    # Premultiplied resampling avoids dark edges around a transformed sprite.
    h,w=image.shape[:2]
    aa=cv2.warpAffine(alpha,matrix,(w,h),flags=cv2.INTER_LINEAR)
    pp=cv2.warpAffine(rgb.astype(np.float32)*alpha[...,None],matrix,(w,h),flags=cv2.INTER_LINEAR)
    return image*(1-aa[...,None])+pp

def affine(pivot,angle=0,dx=0,dy=0):
    m=cv2.getRotationMatrix2D(tuple(pivot),-angle,1)
    m[:,2]+=[dx,dy]
    return m
