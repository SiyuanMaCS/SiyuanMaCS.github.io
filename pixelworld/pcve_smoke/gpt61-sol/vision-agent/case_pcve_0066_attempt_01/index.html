+"""Counterfactual domino mass edit using the source plate and rigid pixel warps.
+
+The third body's mass enters the inertia and gravitational torque explicitly.
+A nonpenetrating, damped four-body pivot model supplies the motion after the
+second/third impact.  The camera, materials, and lighting are sampled only
+from the supplied movie. No appearance change is made for a mass change.
+"""
+from pathlib import Path
+import numpy as np
+import cv2
+import av
+from PIL import Image
+import artifact_io
+
+ROOT=Path(__file__).resolve().parent
+
+def dynamics(n):
+    # Units are calibrated in projected source pixels and seconds.
+    height, width, gravity=66.,10.,700.
+    spacing=np.array([44.,39.,39.])
+    mass=np.array([1.,1.,10.,1.])
+    inertia=mass*(height**2+width**2)/3.
+    b=np.arcsin((spacing[1]-width)/height)
+    a=b+np.arcsin((spacing[0]*np.cos(b)-width)/height)
+    q=np.array([a,b,0.,0.]); initial=q.copy()
+    # The incoming rate matches the observed second-domino approach.
+    velocity=np.array([1.4,2.0,0.,0.])
+    dt=1./2400
+    trajectory=[q.copy()]
+    for k in range((n-1)*100):
+        torque=mass*gravity*(height*np.sin(q)-width*np.cos(q))/2.
+        velocity+=dt*(torque/inertia-.08*velocity)
+        q+=dt*velocity
+        for _ in range(8):
+            for i in range(4):
+                if q[i]<0:
+                    q[i]=0
+                    velocity[i]=max(0.,velocity[i])
+            for i in range(3):
+                aa,bb=q[i],q[i+1]
+                gap=spacing[i]*np.cos(bb)-width-height*np.sin(aa-bb)
+                if gap<0:
+                    j0=-height*np.cos(aa-bb)
+                    j1=-spacing[i]*np.sin(bb)+height*np.cos(aa-bb)
+                    denom=j0*j0/inertia[i]+j1*j1/inertia[i+1]
+                    correction=-gap/denom
+                    q[i]+=j0/inertia[i]*correction
+                    q[i+1]+=j1/inertia[i+1]*correction
+                    normal_velocity=j0*velocity[i]+j1*velocity[i+1]
+                    if normal_velocity<0:
+                        impulse=-1.03*normal_velocity/denom
+                        velocity[i]+=j0/inertia[i]*impulse
+                        velocity[i+1]+=j1/inertia[i+1]*impulse
+        if (k+1)%100==0:
+            trajectory.append(q.copy())
+    return np.array(trajectory)-initial
+
+
+def source_plate():
+    with av.open(str(ROOT/'source_video.mp4')) as movie:
+        for i, frame in enumerate(movie.decode(video=0)):
+            if i==30:

+    raise RuntimeError('Missing source impact frame')

+def make_layers(plate):
+    # High-resolution silhouettes: retain source textures and exact faces.
+    outlines=[
+        [(311,469),(350,425),(587,269),(612,306),(603,342),(337,503)],
+        [(521,488),(531,461),(673,187),(716,205),(711,239),(564,507)],

+    ]
+    layers=[]; erase=np.zeros(plate.shape[:2],np.uint8)
+    for outline in outlines:
+        mask=np.zeros(plate.shape[:2],np.uint8)
+        cv2.fillPoly(mask,[np.array(outline,np.int32)],255)
+        # Include the original antialiased boundary in the foreground.
+        mask=cv2.dilate(mask,np.ones((3,3),np.uint8))

+        # The original video and RGB sample have slightly different filters;
