+"""Counterfactual billiards: same geometry/velocity, fivefold cue mass.
+
+All imagery comes from source_video.mp4.  A projective coordinate along the
+observed near-central shot is inferred from its travel to the far cushion.
+Analytical rigid-body impulses and rolling resistance are integrated with
+sub-frame steps; masses are 5:1, including subsequent contacts/rebounds.
+"""
+from pathlib import Path
+import cv2
+import numpy as np
+from PIL import Image
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+
+def load_source():
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    full = []
+    while True:
+        ok, frame = cap.read()
+        if not ok:
+            break
+        full.append(cv2.cvtColor(frame, cv2.COLOR_BGR2RGB))
+    cap.release()
+    return np.asarray(full)
+
+
+def white_centers(full):
+    centers = []
+    for im in full[:13]:
+        r = im.astype(np.float32)
+        m = ((r[..., 0] > 140) & (r[..., 0] > .93*r[..., 1]) &
+             (r[..., 2] > .91*r[..., 1])).astype(np.uint8)
+        m[:230] = 0; m[510:] = 0; m[:, :350] = 0; m[:, 900:] = 0
+        n, labs, stats, cent = cv2.connectedComponentsWithStats(m)
+        ids = [j for j in range(1, n) if 100 < stats[j, 4] < 1800]
+        centers.append(cent[max(ids, key=lambda j: stats[j, 4])])
+    return np.asarray(centers)
+
+
+def make_sprite(frame, bg, center, halfsize, extent):
+    x, y = center
+    ix, iy = int(round(x)), int(round(y))
+    h = halfsize
+    a = frame[iy-h:iy+h+1, ix-h:ix+h+1].astype(np.float32)
+    b = bg[iy-h:iy+h+1, ix-h:ix+h+1].astype(np.float32)
+    yy, xx = np.mgrid[-h:h+1, -h:h+1]
+    xx = xx + ix-x; yy = yy + iy-y
+    # Keep the complete sphere and its soft contact shadow, but not the
+    # surrounding compression noise or other source objects.
+    rx, ry = extent
+    dist = np.sqrt((xx/rx)**2 + ((yy-4)/ry)**2)
+    support = np.clip((1.13-dist)/.16, 0, 1)
+    delta = (a-b) * support[...,None]
+    delta[np.max(np.abs(delta), axis=2) < 1.8] = 0
+    return delta, (x-(ix-h), y-(iy-h))
+
+
+def paste_delta(canvas, sprite, center, scale):
+    data, anchor = sprite
+    h, w = data.shape[:2]
+    cx, cy = center
+    left = int(np.floor(cx-anchor[0]*scale))-1
+    top = int(np.floor(cy-anchor[1]*scale))-1
+    right = int(np.ceil(cx+(w-anchor[0])*scale))+1
+    bottom = int(np.ceil(cy+(h-anchor[1])*scale))+1
+    yy, xx = np.mgrid[top:bottom, left:right].astype(np.float32)
+    mx = (xx-cx)/scale+anchor[0]
+    my = (yy-cy)/scale+anchor[1]
+    d = cv2.remap(data.astype(np.float32), mx.astype(np.float32), my.astype(np.float32), cv2.INTER_LINEAR,
+                  borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+    canvas[top:bottom,left:right] += d
+
+
+def main():
+    full = load_source()
+    small = np.load(ROOT/'source_pixels.npz')['rgb']
+    assert full.shape[0] == 96
+    wc = white_centers(full)
+    x0, y0 = wc[0]
+    # Distances are normalized to table-width units.  The rational projection
+    # is fit using the stationary target and source far-cushion contact.
+    target_s, rail_s = .585, 1.553
+    xb, yb = 639.5, 351.5
+    xr = 852.0
+    a, c = np.linalg.solve([[target_s, -target_s*xb],
+                           [rail_s, -rail_s*xr]], [xb-x0, xr-x0])
+    def project(s):
+        return (x0+a*s)/(1+c*s)
+    def unproject(x):
+        return (x-x0)/(a-c*x)
+    co = np.polyfit(np.arange(13)/24, unproject(wc[:,0]), 2)
+    initial_speed = co[1]
+    drag = -2*co[0]
+    wslope = np.polyfit(wc[:,0], wc[:,1], 1)[0]
+    bslope = -.58816
+
+    # Background: source temporal median removes moving objects.  The cue
+    # stops in the factual clip, so recover its empty felt from frame zero.
+    bg = np.median(full, axis=0).astype(np.uint8)
+    yy, xx = np.mgrid[:720,:1280]
+    occupied = ((xx-630)/29)**2 + ((yy-360)/36)**2 < 1
+    black_initial = ((xx-xb)/24)**2 + ((yy-yb-4)/29)**2 < 1
+    take = occupied & ~black_initial
+    bg[take] = full[0][take]
+    hole = (occupied & black_initial).astype(np.uint8)*255
+    bg = cv2.inpaint(bg, hole, 4, cv2.INPAINT_TELEA)
+
+    cue_sprite = make_sprite(full[0], bg, (x0,y0), 48, (24,34))
+    black_bank = []
+    for i in [0] + list(range(16,42)) + list(range(51,83)):
+        t = i/24
+        if i == 0:
+            cx, cy = xb,yb
+        else:
+            # Source trajectory polynomials, only used to locate its texture.
+            ss = (-.07185*t*t+.90306*t+.10066 if i < 45
+                  else .050285*t*t-.698279*t+2.678263)
+            px = project(ss)
+            py = yb+bslope*(px-xb)
+            ix,iy = int(round(px)),int(round(py))
+            crop = full[i,iy-21:iy+22,ix-21:ix+22]
+            yy1,xx1 = np.mgrid[-21:22,-21:22]
+            dark = ((crop[...,0]<75)&(crop[...,1]<85)&
+                    ((xx1/18)**2+(yy1/18)**2<1)).astype(np.uint8)
+            n, lab, stats, cent = cv2.connectedComponentsWithStats(dark)
+            j = 1+np.argmax(stats[1:,4]); st=stats[j]
+            cx = ix-21+st[0]+(st[2]-1)/2
+            cy = iy-21+st[1]+(st[3]-1)/2
+        ss = unproject(cx)
+        ratio = (1+c*target_s)/(1+c*ss)
+        spr = make_sprite(full[i],bg,(cx,cy),35,(19*ratio,27*ratio))
+        black_bank.append((ss, i<45, spr))
+
+    # Preserve initial velocities, not initial momentum, on changing mass.
+    # The mass change alone has no visual consequence before the first impact.
+    positions = np.zeros((96,2))
+    velocities = np.zeros((96,2))
+    s = np.array([0.,target_s]); v=np.array([initial_speed,0.])
+    masses=np.array([5.,1.]); radii=np.array([.040,.036])
+    dt=1/(24*240)
+    events=[]
+    for f in range(96):
+        positions[f]=s; velocities[f]=v
+        if f==95: break
+        for sub in range(240):
+            time=(f+sub/240)/24
+            s += v*dt
+            # Rolling resistance is a force proportional to normal load,
+            # hence does not make the heavier ball accelerate more slowly.
+            decel=np.where(v>=0,drag,.106)
+            v = np.sign(v)*np.maximum(np.abs(v)-decel*dt,0.)
+            for k in range(2):
+                if s[k]>rail_s:
+                    s[k]=2*rail_s-s[k]
+                    v[k]=-abs(v[k])*.80
+                    events.append(('cushion',round(time,4),k))
+            gap=s[1]-s[0]; n=np.sign(gap)
+            if abs(gap)<radii.sum() and (v[1]-v[0])*n<0:
+                impulse=-(1+.80)*(v[1]-v[0])*n/(1/masses[0]+1/masses[1])
+                v[0]-=n*impulse/masses[0]
+                v[1]+=n*impulse/masses[1]
+                correction=radii.sum()-abs(gap)
+                s[0]-=n*correction/6; s[1]+=n*correction*5/6
+                events.append(('balls',round(time,4),v.copy().tolist()))
+
+    frames=[]
+    for f in range(96):
+        if f<=12:
+            frames.append(small[f].copy())
+            continue
+        canvas=bg.astype(np.float32).copy()
+        # Draw distant ball first for correct perspective occlusion.
+        for k in np.argsort(positions[f])[::-1]:
+            ss=positions[f,k]
+            cx=project(ss)
+            if k==0:
+                cy=y0+wslope*(cx-x0)
+                sprite=cue_sprite
+                scale=1/(1+c*ss)
+            else:
+                cy=yb+bslope*(cx-xb)
+                positive=velocities[f,k]>=0
+                options=[p for p in black_bank if p[1]==positive or p[0]<.59]
+                ref=min(options,key=lambda p:abs(p[0]-ss))
+                sprite=ref[2]
+                scale=(1+c*ref[0])/(1+c*ss)
+            paste_delta(canvas,sprite,(cx,cy),scale)
+        im=Image.fromarray(np.clip(canvas,0,255).astype(np.uint8))
+        frames.append(np.array(im.resize((256,144),Image.Resampling.LANCZOS)))
+    frames=np.asarray(frames,dtype=np.uint8)
+    artifact_io.save(frames,ROOT)
+    print('Saved',frames.shape,'mass ratio 5:1; events:',events)
+    # Conventional video alongside the lossless player/tensor artifacts.
+    out=cv2.VideoWriter(str(ROOT/'edited_video.mp4'),cv2.VideoWriter_fourcc(*'mp4v'),24,(256,144))
+    for im in frames: out.write(cv2.cvtColor(im,cv2.COLOR_RGB2BGR))
+    out.release()
+
+if __name__=='__main__':
+    main()

\"\"\"Counterfactual billiards: same geometry/velocity, fivefold cue mass.

All imagery comes from source_video.mp4.  A projective coordinate along the
observed near-central shot is inferred from its travel to the far cushion.
Analytical rigid-body impulses and rolling resistance are integrated with
sub-frame steps; masses are 5:1, including subsequent contacts/rebounds.
\"\"\"
from pathlib import Path
import cv2
import numpy as np
from PIL import Image
import artifact_io

ROOT = Path(__file__).resolve().parent


def load_source():
    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
    full = []
    while True:
        ok, frame = cap.read()
        if not ok:
            break
        full.append(cv2.cvtColor(frame, cv2.COLOR_BGR2RGB))
    cap.release()
    return np.asarray(full)


def white_centers(full):
    centers = []
    for im in full[:13]:
        r = im.astype(np.float32)
        m = ((r[..., 0] > 140) & (r[..., 0] > .93*r[..., 1]) &
             (r[..., 2] > .91*r[..., 1])).astype(np.uint8)
        m[:230] = 0; m[510:] = 0; m[:, :350] = 0; m[:, 900:] = 0
        n, labs, stats, cent = cv2.connectedComponentsWithStats(m)
        ids = [j for j in range(1, n) if 100 < stats[j, 4] < 1800]
        centers.append(cent[max(ids, key=lambda j: stats[j, 4])])
    return np.asarray(centers)


def make_sprite(frame, bg, center, halfsize, extent):
    x, y = center
    ix, iy = int(round(x)), int(round(y))
    h = halfsize
    a = frame[iy-h:iy+h+1, ix-h:ix+h+1].astype(np.float32)
    b = bg[iy-h:iy+h+1, ix-h:ix+h+1].astype(np.float32)
    yy, xx = np.mgrid[-h:h+1, -h:h+1]
    xx = xx + ix-x; yy = yy + iy-y
    # Keep the complete sphere and its soft contact shadow, but not the
    # surrounding compression noise or other source objects.
    rx, ry = extent
    dist = np.sqrt((xx/rx)**2 + ((yy-4)/ry)**2)
    support = np.clip((1.13-dist)/.16, 0, 1)
    delta = (a-b) * support[...,None]
    delta[np.max(np.abs(delta), axis=2) < 1.8] = 0
    return delta, (x-(ix-h), y-(iy-h))


def paste_delta(canvas, sprite, center, scale):
    data, anchor = sprite
    h, w = data.shape[:2]
    cx, cy = center
    left = int(np.floor(cx-anchor[0]*scale))-1
    top = int(np.floor(cy-anchor[1]*scale))-1
    right = int(np.ceil(cx+(w-anchor[0])*scale))+1
    bottom = int(np.ceil(cy+(h-anchor[1])*scale))+1
    yy, xx = np.mgrid[top:bottom, left:right].astype(np.float32)
    mx = (xx-cx)/scale+anchor[0]
    my = (yy-cy)/scale+anchor[1]
    d = cv2.remap(data, mx, my, cv2.INTER_LINEAR,
                  borderMode=cv2.BORDER_CONSTANT, borderValue=0)
    canvas[top:bottom,left:right] += d


def main():
    full = load_source()
    small = np.load(ROOT/'source_pixels.npz')['rgb']
    assert full.shape[0] == 96
    wc = white_centers(full)
    x0, y0 = wc[0]
    # Distances are normalized to table-width units.  The rational projection
    # is fit using the stationary target and source far-cushion contact.
    target_s, rail_s = .585, 1.553
    xb, yb = 639.5, 351.5
    xr = 852.0
    a, c = np.linalg.solve([[target_s, -target_s*xb],
                           [rail_s, -rail_s*xr]], [xb-x0, xr-x0])
    def project(s):
        return (x0+a*s)/(1+c*s)
    def unproject(x):
        return (x-x0)/(a-c*x)
    co = np.polyfit(np.arange(13)/24, unproject(wc[:,0]), 2)
    initial_speed = co[1]
    drag = -2*co[0]
    wslope = np.polyfit(wc[:,0], wc[:,1], 1)[0]
    bslope = -.58816

    # Background: source temporal median removes moving objects.  The cue
    # stops in the factual clip, so recover its empty felt from frame zero.
    bg = np.median(full, axis=0).astype(np.uint8)
    yy, xx = np.mgrid[:720,:1280]
    occupied = ((xx-630)/29)**2 + ((yy-360)/36)**2 < 1
    black_initial = ((xx-xb)/24)**2 + ((yy-yb-4)/29)**2 < 1
    take = occupied & ~black_initial
    bg[take] = full[0][take]
    hole = (occupied & black_initial).astype(np.uint8)*255
    bg = cv2.inpaint(bg, hole, 4, cv2.INPAINT_TELEA)

    cue_sprite = make_sprite(full[0], bg, (x0,y0), 48, (24,34))
    black_bank = []
    for i in [0] + list(range(16,42)) + list(range(51,83)):
        t = i/24
        if i == 0:
            cx, cy = xb,yb
        else:

            ss = (-.07185*t*t+.90306*t+.10066 if i < 45
                  else .050285*t*t-.698279*t+2.678263)

    # The mass change alone has no visual consequence before the first impact.
    positions = np.zeros((96,2))

    events=[]

            # hence does not make the heavier ball accelerate more slowly.
            decel=np.where(v>=0,drag,.106)
            v = np.sign(v)*np.maximum(np.abs(v)-decel*dt,0.)

            gap=s[1]-s[0]; n=np.sign(gap)

                impulse=-(1+.80)*(v[1]-v[0])*n/(1/masses[0]+1/masses[1])
                v[0]-=n*impulse/masses[0]
                v[1]+=n*impulse/masses[1]
                correction=radii.sum()-abs(gap)
                s[0]-=n*correction/6; s[1]+=n*correction*5/6
                events.append(('balls',round(time,4),v.copy().tolist()))

    frames=[]
    for f in range(96):
        if f<=12:
            frames.append(small[f].copy())
            continue

        for k in np.argsort(positions[f])[::-1]:
            ss=positions[f,k]
            cx=project(ss)
            if k==0:
                cy=y0+wslope*(cx-x0)
                sprite=cue_sprite
                scale=1/(1+c*ss)

                options=[p for p in black_bank if p[1]==positive or p[0]<.59]
                ref=min(options,key=lambda p:abs(p[0]-ss))
                sprite=ref[2]
                scale=(1+c*ref[0])/(1+c*ss)

        im=Image.fromarray(np.clip(canvas,0,255).astype(np.uint8))
        frames.append(np.array(im.resize((256,144),Image.Resampling.LANCZOS)))
    frames=np.asarray(frames,dtype=np.uint8)

    # Conventional video alongside the lossless player/tensor artifacts.
    out=cv2.VideoWriter(str(ROOT/'edited_video.mp4'),cv2.VideoWriter_fourcc(*'mp4v'),24,(256,144))
    for im in frames: out.write(cv2.cvtColor(im,cv2.COLOR_RGB2BGR))
    out.release()

if __name__=='__main__':
    main()
