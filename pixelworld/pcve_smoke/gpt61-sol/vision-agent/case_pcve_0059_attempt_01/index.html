+"""Pixel-based edit: change only mass, integrate the subsequent contact chain.
+
+The early collision (can -> cup) is unaffected. The second collision has a
+fivefold carton mass, with Coulomb sliding friction and low restitution.
+Textures, illumination and the stationary scene are taken from this video.
+"""
+from pathlib import Path
+import cv2
+import numpy as np
+from PIL import Image, ImageDraw, ImageFilter
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+
+def read_video():
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    frames = []
+    while True:
+        ok, bgr = cap.read()
+        if not ok:
+            break
+        frames.append(cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB))
+    cap.release()
+    return np.asarray(frames)
+
+
+def trajectory():
+    # Coordinates are in the source's 256-wide camera projection. Velocities
+    # are pixels / frame; source friction is about 0.26 pixels / frame^2.
+    x = np.array([140.96, 156.10, 170.60], dtype=float)
+    v = np.array([8.10, 8.95, 0.0], dtype=float)
+    mass = np.array([1., 1., 5.])
+    radius = np.array([4.70, 5.65, 4.10])
+    result = np.zeros((96, 3))
+    result[:10] = x
+    dt = 1 / 256
+    for t in range(10, 96):
+        for step in range(256):
+            v = np.sign(v) * np.maximum(np.abs(v) - .26 * dt, 0.)
+            x += v * dt
+            # Repeated short steps resolve the can/cup/carton contact chain.
+            for a, b in [(0, 1), (1, 2)]:
+                overlap = radius[a] + radius[b] - (x[b] - x[a])
+                if overlap > 0:
+                    x[a] -= overlap * mass[b] / (mass[a] + mass[b])
+                    x[b] += overlap * mass[a] / (mass[a] + mass[b])
+                    if v[a] > v[b]:
+                        impulse = 1.04 * (v[a] - v[b]) / (1/mass[a] + 1/mass[b])
+                        v[a] -= impulse / mass[a]
+                        v[b] += impulse / mass[b]
+        result[t] = x
+    return result
+
+
+def make_sprite(rgb, polygon, extra_polygons=()):
+    mask = Image.new('L', (1280, 720))
+    draw = ImageDraw.Draw(mask)
+    draw.polygon(polygon, fill=255)
+    for poly in extra_polygons:
+        draw.polygon(poly, fill=255)
+    # Match the mild pixel-edge softness in the original render/encode.
+    mask = mask.filter(ImageFilter.GaussianBlur(.65))
+    alpha = np.asarray(mask).astype(np.float32) / 255
+    return rgb.astype(np.float32) * alpha[..., None], alpha
+
+
+def place(canvas, sprite, dx, angle, pivot):
+    color, alpha = sprite
+    mat = cv2.getRotationMatrix2D(pivot, angle, 1.0)
+    mat[0, 2] += dx
+    a = cv2.warpAffine(alpha, mat, (1280, 720), flags=cv2.INTER_LINEAR)
+    c = cv2.warpAffine(color, mat, (1280, 720), flags=cv2.INTER_LINEAR)
+    canvas *= 1 - a[..., None]
+    canvas += c
+
+
+def main():
+    source = np.load(ROOT / 'source_pixels.npz')['rgb']
+    high = read_video()
+    reference = high[9]
+    # This region is empty at frame 35. Reuse its actual tabletop/chair pixels
+    # rather than painting or estimating the background beneath the objects.
+    background = reference.copy()
+    background[265:450, 635:970] = high[34, 265:450, 635:970]
+
+    can = make_sprite(reference, [
+        (688, 325), (719, 325), (724, 333), (727, 343),
+        (727, 401), (725, 407), (718, 410), (688, 410),
+        (682, 405), (681, 341), (683, 332)])
+    cup = make_sprite(reference, [
+        (751, 311), (760, 310), (801, 310), (809, 312),
+        (810, 316), (806, 320), (802, 381), (800, 406),
+        (796, 410), (762, 410), (758, 406), (755, 377),
+        (752, 322), (749, 318), (749, 313)], [
+        [(778, 312), (778, 300), (779, 285), (781, 282),
+         (784, 284), (782, 307), (782, 312)]])
+    milk = make_sprite(reference, [
+        (834, 322), (849, 322), (851, 324), (851, 330),
+        (867, 330), (872, 334), (873, 343), (871, 407),
+        (868, 412), (837, 412), (833, 407), (832, 388),
+        (830, 346), (831, 334), (834, 330)])
+    sprites = [can, cup, milk]
+    start = np.array([140.96, 156.10, 170.60])
+    pivots = [(704.8, 408.), (780.5, 408.), (853., 411.)]
+    motion = trajectory()
+    frames = source.copy()
+    yy, xx = np.indices((185, 335))
+    xx = xx + 635
+    yy = yy + 265
+    for t in range(10, 96):
+        canvas = background.astype(np.float32).copy()
+        # Short contact shadows follow all three objects. The unchanged soft
+        # cast lighting is already present in the tabletop background.
+        shadow = np.zeros((185, 335), np.float32)
+        for j in range(3):
+            cx = motion[t, j] * 5
+            rx = [24., 25., 20.][j]
+            cy = [412., 413., 414.][j]
+            shadow += .115 * np.exp(-.5*((xx-cx)/rx)**2 - .5*((yy-cy)/3.2)**2)
+            shadow += .035 * np.exp(-.5*((xx-(cx+8))/ (rx*1.3))**2 - .5*((yy-(cy+2))/5.)**2)
+        canvas[265:450, 635:970] *= 1 - np.minimum(shadow, .20)[..., None]
+        # A small damped upright wobble, not a change of shape or scale.
+        elapsed = max(t - 9.55, 0.)
+        wave = np.sin(elapsed * 1.8) * np.exp(-elapsed / 2.8)
+        angles = [.30 * wave, -1.45 * wave, -.22 * wave]
+        for j in range(3):
+            place(canvas, sprites[j], (motion[t, j]-start[j])*5,
+                  angles[j], pivots[j])
+        rendered = np.asarray(Image.fromarray(np.uint8(np.clip(canvas, 0, 255))).resize(
+            (256, 144), Image.Resampling.LANCZOS))
+        # Outside the edited interaction region preserve the source exactly.
+        frames[t, 52:91, 125:256] = rendered[52:91, 125:256]
+    artifact_io.save(frames, ROOT)
+    print('Saved 96 RGB frames. Final projected positions:', motion[-1])
+
+
+if __name__ == '__main__':
+    main()
