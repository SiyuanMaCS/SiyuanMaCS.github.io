+"""Triple the moving volleyball's friction from the first simulation frame.
+
+Uses source-only pixel reprojection: fit a perspective path on the floor,
+apply a tripled tangential deceleration, and reproject the observed rotating
+ball. No scene renderer, model weights, or external assets are used.
+"""
+from pathlib import Path
+import numpy as np
+import cv2
+from PIL import Image
+from scipy.ndimage import gaussian_filter
+from scipy.spatial.transform import Rotation
+import artifact_io
+
+CASE = Path(__file__).resolve().parent
+
+# Observed source sphere centers/radii, in the original video's pixel plane.
+POSES = np.array([
+    [404.5,605.5,60.5],[428,573,56.5],[448,542,53.5],
+    [463,516,51.5],[479,490,49.5],[493.5,466,47.5],
+    [508.5,446,45.5],[520,426,43.5],[532,408,42],
+    [541,393,40.5],[550,379,39.5],[556,368,38.5],
+    [561,360,37.5],[566,352,37],[570,345,36.5],
+    [574,340,36.5],[576,336,36],[578,333,36],
+    [579,332,36],[579,332,36]], dtype=float)
+
+
+def load_high_resolution():
+    cap = cv2.VideoCapture(str(CASE / 'source_video.mp4'))
+    frames = []
+    while True:
+        ok, frame = cap.read()
+        if not ok:
+            break
+        frames.append(cv2.cvtColor(frame, cv2.COLOR_BGR2RGB))
+    cap.release()
+    return np.asarray(frames)
+
+
+def make_clean_room(high):
+    """Recover the empty floor from temporally unobstructed source pixels."""
+    H, W = high.shape[1:3]
+    yy, xx = np.mgrid[:H, :W]
+    # Initial ball and its soft shadow have moved out of this patch by frame 25.
+    patch = np.zeros((H, W), np.float32)
+    patch[485:710, 205:525] = 1
+    patch = gaussian_filter(patch, 10)
+    room = high[0].astype(np.float32) * (1-patch[...,None])
+    room += high[25].astype(np.float32) * patch[...,None]
+    # Restrict background replacement to the source motion/shadow corridor.
+    corridor = np.zeros((H,W), np.float32)
+    for cx, cy, r in POSES:
+        d = ((xx-(cx-.65*r))/(2.1*r))**2 + ((yy-(cy+.13*r))/(1.6*r))**2
+        corridor = np.maximum(corridor, (d<1).astype(np.float32))
+    corridor = gaussian_filter(corridor, 5)
+    return room, corridor
+
+
+def illumination(normals):
+    light = np.array([.38,-.66,.65])
+    light /= np.linalg.norm(light)
+    return .61 + .39*np.maximum(normals @ light, 0)
+
+
+def ball_image(high, phase, center, radius):
+    """Source sphere reprojection between observed rotation phases."""
+    cx,cy = center
+    left, top = int(np.floor(cx-radius-3)), int(np.floor(cy-radius-3))
+    right, bottom = int(np.ceil(cx+radius+3)), int(np.ceil(cy+radius+3))
+    yy,xx = np.mgrid[top:bottom, left:right]
+    x,y = (xx-cx)/radius,(yy-cy)/radius
+    d = np.sqrt(x*x+y*y)
+    alpha = np.clip((1-d)*radius+.5,0,1).astype(np.float32)
+    z = np.sqrt(np.maximum(1-x*x-y*y,0))
+    normals = np.stack([x,y,z],axis=-1)
+    # View-space spin measured by matching source panel boundaries on the sphere.
+    omega = np.array([.80593923,.16644358,.38847351])
+    nearest = int(round(phase))
+    colors = np.zeros((*x.shape,3),np.float64)
+    weights = np.zeros(x.shape,np.float64)
+    for j in range(max(0,nearest-2), min(7,nearest+3)):
+        spin = Rotation.from_rotvec(omega*(phase-j)).as_matrix()
+        mapped = normals @ spin
+        sx,sy,sr = POSES[j]
+        mapx=(sx+sr*mapped[...,0]).astype(np.float32)
+        mapy=(sy+sr*mapped[...,1]).astype(np.float32)
+        samples = cv2.remap(high[j].astype(np.float32),mapx,mapy,
+                            cv2.INTER_LINEAR,borderMode=cv2.BORDER_REFLECT)
+        # Choose mostly the closest observed view; adjacent views fill limbs
+        # exposed by the additional fractional rotation.
+        w = np.maximum(mapped[...,2]-.02,0)**5
+        w *= np.exp(-5.5*abs(phase-j))
+        samples *= (illumination(normals)/np.maximum(illumination(mapped),.4))[...,None]
+        colors += samples*w[...,None]
+        weights += w
+    missing=weights<1e-8
+    if missing.any():
+        sx,sy,sr = POSES[nearest]
+        samples=cv2.remap(high[nearest].astype(np.float32),
+                          (sx+sr*x).astype(np.float32),(sy+sr*y).astype(np.float32),
+                          cv2.INTER_LINEAR)
+        colors[missing]=samples[missing]
+        weights[missing]=1
+    colors/=weights[...,None]
+    return (left,top,right,bottom),colors.clip(0,255),alpha
+
+
+def render():
+    factual=np.load(CASE/'source_pixels.npz')['rgb']
+    high=load_high_resolution()
+    room,corridor=make_clean_room(high)
+    H,W=high.shape[1:3]
+    yy,xx=np.mgrid[:H,:W]
+
+    # Perspective coordinate q = 0 initially, q = 1 at the factual rest point.
+    # The projected sphere's radius and floor path share the same depth factor.
+    y0=POSES[0,1]; yh=-60.0; yf=POSES[-1,1]
+    depth=(y0-yh)/(yf-yh)-1
+    x0=POSES[0,0]; xf=POSES[-1,0]
+    xh=(xf*(1+depth)-x0)/depth
+    observed_q=((y0-yh)/(POSES[:,1]-yh)-1)/depth
+    # Initial world-plane velocity is preserved. Estimate it over the first
+    # bare-floor samples, before the factual ball reaches the rug.
+    ts=np.arange(5)/24
+    v0=np.dot(ts,observed_q[:5])/np.dot(ts,ts)
+    original_drag=v0*v0/2
+    friction_drag=3*original_drag
+    stop_time=v0/friction_drag
+
+    # Extract the actual source shadow as an attenuation map on clean floor.
+    # Keep its direction, softness, color, and contact structure unchanged.
+    initial_ratio=(high[0].astype(np.float32)+.1)/(room+.1)
+    initial_ratio=np.clip(initial_ratio,.32,1)
+    initial_ratio=gaussian_filter(initial_ratio,(2.2,2.2,0))
+    r0=POSES[0,2]
+    shadow_window=np.exp(-(((xx-(x0-.65*r0))/(1.9*r0))**4+
+                            ((yy-(y0+.27*r0))/(1.22*r0))**4))
+    # Pixels occluded by the sphere are not a shadow sample.
+    sphere_dist=np.sqrt((xx-x0)**2+(yy-y0)**2)
+    outside=np.clip((sphere_dist-r0+1)/5,0,1)
+    attenuation=(1-initial_ratio)*shadow_window[...,None]*outside[...,None]
+    attenuation=attenuation.astype(np.float32)
+
+    output=[]
+    stationary=None
+    final_center=None
+    for i in range(96):
+        if i==0:
+            output.append(factual[0].copy())
+            continue
+        t=min(i/24,stop_time)
+        q=v0*t-.5*friction_drag*t*t
+        q=max(0,q)
+        denominator=1+depth*q
+        cx=xh+(x0-xh)/denominator
+        cy=yh+(y0-yh)/denominator
+        r=r0/denominator
+        phase=float(np.interp(q,observed_q,np.arange(len(POSES))))
+        # At rest the object's position, orientation, and shadow stay fixed.
+        if stationary is None or t<stop_time:
+            scene=room.copy()
+            scale=r/r0
+            sx=(x0+(xx-cx)/scale).astype(np.float32)
+            sy=(y0+(yy-cy)/scale).astype(np.float32)
+            shifted=cv2.remap(attenuation,sx,sy,cv2.INTER_LINEAR,
+                              borderMode=cv2.BORDER_CONSTANT,borderValue=0)
+            scene*=1-shifted
+            # Small tight contact occlusion, beneath the sphere's silhouette.
+            contact=np.exp(-(((xx-cx)/(r*.61))**2+
+                              ((yy-(cy+r*.87))/(r*.15))**2)*2.0)
+            scene*=1-.15*contact[...,None]
+            bounds,ball,alpha=ball_image(high,phase,(cx,cy),r)
+            left,top,right,bottom=bounds
+            patch=scene[top:bottom,left:right]
+            scene[top:bottom,left:right]=patch*(1-alpha[...,None])+ball*alpha[...,None]
+            stationary=scene
+            final_center=(cx,cy,r,phase)
+        else:
+            scene=stationary
+        # Preserve factual background pixels outside the physical edit region.
+        edited=high[i].astype(np.float32)*(1-corridor[...,None])+scene*corridor[...,None]
+        frame=np.asarray(Image.fromarray(np.rint(edited).clip(0,255).astype(np.uint8))
+                          .resize((256,144),Image.Resampling.LANCZOS))
+        output.append(frame)
+    output=np.asarray(output,np.uint8)
+    artifact_io.save(output,CASE)
+    print(f'Saved {len(output)} frames. Tripled friction: stop at {stop_time:.3f}s; '
+          f'center/radius/rotation phase={final_center}; path fraction={q:.6f}.')
+    # Convenience video in addition to the required lossless RGB artifacts.
+    writer=cv2.VideoWriter(str(CASE/'edited_video.mp4'),
+                           cv2.VideoWriter_fourcc(*'mp4v'),24,(256,144))
+    for f in output:
+        writer.write(cv2.cvtColor(f,cv2.COLOR_RGB2BGR))
+    writer.release()
+
+
+if __name__=='__main__':
+    render()

from pathlib import Path
import numpy as np
import cv2
from PIL import Image
from scipy.ndimage import gaussian_filter
from scipy.spatial.transform import Rotation
import artifact_io

CASE = Path(__file__).resolve().parent

# Observed source sphere centers/radii, in the original video's pixel plane.
POSES = np.array([
    [404.5,605.5,60.5],[428,573,56.5],[448,542,53.5],
    [463,516,51.5],[479,490,49.5],[493.5,466,47.5],
    [508.5,446,45.5],[520,426,43.5],[532,408,42],
    [541,393,40.5],[550,379,39.5],[556,368,38.5],

def load_high_resolution():
    cap = cv2.VideoCapture(str(CASE / 'source_video.mp4'))
    frames = []
    while True:
        ok, frame = cap.read()
        if not ok:
            break
        frames.append(cv2.cvtColor(frame, cv2.COLOR_BGR2RGB))
    cap.release()
    return np.asarray(frames)


def make_clean_room(high):
    \"\"\"Recover the empty floor from temporally unobstructed source pixels.\"\"\"
    H, W = high.shape[1:3]
    yy, xx = np.mgrid[:H, :W]
    # Initial ball and its soft shadow have moved out of this patch by frame 25.
    patch = np.zeros((H, W), np.float32)
    patch[485:710, 205:525] = 1
    patch = gaussian_filter(patch, 10)
    room = high[0].astype(np.float32) * (1-patch[...,None])
    room += high[25].astype(np.float32) * patch[...,None]
    # Restrict background replacement to the source motion/shadow corridor.
    corridor = np.zeros((H,W), np.float32)
    for cx, cy, r in POSES:
        d = ((xx-(cx-.65*r))/(2.1*r))**2 + ((yy-(cy+.13*r))/(1.6*r))**2
        corridor = np.maximum(corridor, (d<1).astype(np.float32))
    corridor = gaussian_filter(corridor, 5)
    return room, corridor


def illumination(normals):
    light = np.array([.38,-.66,.65])
    light /= np.linalg.norm(light)
    return .61 + .39*np.maximum(normals @ light, 0)


def ball_image(high, phase, center, radius):
    \"\"\"Source sphere reprojection between observed rotation phases.\"\"\"
    cx,cy = center
    left, top = int(np.floor(cx-radius-3)), int(np.floor(cy-radius-3))
    right, bottom = int(np.ceil(cx+radius+3)), int(np.ceil(cy+radius+3))
    yy,xx = np.mgrid[top:bottom, left:right]
    x,y = (xx-cx)/radius,(yy-cy)/radius
    d = np.sqrt(x*x+y*y)
    alpha = np.clip((1-d)*radius+.5,0,1).astype(np.float32)
    z = np.sqrt(np.maximum(1-x*x-y*y,0))
    normals = np.stack([x,y,z],axis=-1)
    # View-space spin measured by matching source panel boundaries on the sphere.
    omega = np.array([.80593923,.16644358,.38847351])
    nearest = int(round(phase))
    colors = np.zeros((*x.shape,3),np.float64)
    weights = np.zeros(x.shape,np.float64)
    for j in range(max(0,nearest-2), min(7,nearest+3)):
        spin = Rotation.from_rotvec(omega*(phase-j)).as_matrix()
        mapped = normals @ spin
        sx,sy,sr = POSES[j]
        mapx=(sx+sr*mapped[...,0]).astype(np.float32)
        mapy=(sy+sr*mapped[...,1]).astype(np.float32)
        samples = cv2.remap(high[j].astype(np.float32),mapx,mapy,
                            cv2.INTER_LINEAR,borderMode=cv2.BORDER_REFLECT)
        # Choose mostly the closest observed view; adjacent views fill limbs
        # exposed by the additional fractional rotation.
        w = np.maximum(mapped[...,2]-.02,0)**5
        w *= np.exp(-5.5*abs(phase-j))
        samples *= (illumination(normals)/np.maximum(illumination(mapped),.4))[...,None]
        colors += samples*w[...,None]
        weights += w
    missing=weights<1e-8
    if missing.any():
        sx,sy,sr = POSES[nearest]
        samples=cv2.remap(high[nearest].astype(np.float32),
                          (sx+sr*x).astype(np.float32),(sy+sr*y).astype(np.float32),
                          cv2.INTER_LINEAR)
        colors[missing]=samples[missing]
        weights[missing]=1
    colors/=weights[...,None]
    return (left,top,right,bottom),colors.clip(0,255),alpha


def render():
    factual=np.load(CASE/'source_pixels.npz')['rgb']
    high=load_high_resolution()
    room,corridor=make_clean_room(high)
    H,W=high.shape[1:3]
    yy,xx=np.mgrid[:H,:W]

    # Perspective coordinate q = 0 initially, q = 1 at the factual rest point.
    # The projected sphere's radius and floor path share the same depth factor.
    y0=POSES[0,1]; yh=-60.0; yf=POSES[-1,1]
    depth=(y0-yh)/(yf-yh)-1
    x0=POSES[0,0]; xf=POSES[-1,0]
    xh=(xf*(1+depth)-x0)/depth
    observed_q=((y0-yh)/(POSES[:,1]-yh)-1)/depth
    # Initial world-plane velocity is preserved. Estimate it over the first
    # bare-floor samples, before the factual ball reaches the rug.
    ts=np.arange(5)/24
    v0=np.dot(ts,observed_q[:5])/np.dot(ts,ts)
    original_drag=v0*v0/2
    friction_drag=3*original_drag
    stop_time=v0/friction_drag

    # Extract the actual source shadow as an attenuation map on clean floor.
    # Keep its direction, softness, color, and contact structure unchanged.
    initial_ratio=(high[0].astype(np.float32)+.1)/(room+.1)
    initial_ratio=np.clip(initial_ratio,.32,1)
    initial_ratio=gaussian_filter(initial_ratio,(2.2,2.2,0))
    r0=POSES[0,2]
    shadow_window=np.exp(-(((xx-(x0-.65*r0))/(1.9*r0))**4+
                            ((yy-(y0+.27*r0))/(1.22*r0))**4))
    # Pixels occluded by the sphere are not a shadow sample.
    sphere_dist=np.sqrt((xx-x0)**2+(yy-y0)**2)
    outside=np.clip((sphere_dist-r0+1)/5,0,1)
    attenuation=(1-initial_ratio)*shadow_window[...,None]*outside[...,None]
    attenuation=attenuation.astype(np.float32)

    output=[]
    stationary=None
    final_center=None
    for i in range(96):
        if i==0:
            output.append(factual[0].copy())
            continue
        t=min(i/24,stop_time)
        q=v0*t-.5*friction_drag*t*t
        q=max(0,q)
        denominator=1+depth*q
        cx=xh+(x0-xh)/denominator
        cy=yh+(y0-yh)/denominator
        r=r0/denominator
        phase=float(np.interp(q,observed_q,np.arange(len(POSES))))
        # At rest the object's position, orientation, and shadow stay fixed.
        if stationary is None or t<stop_time:
            scene=room.copy()
            scale=r/r0
            sx=(x0+(xx-cx)/scale).astype(np.float32)
            sy=(y0+(yy-cy)/scale).astype(np.float32)
            shifted=cv2.remap(attenuation,sx,sy,cv2.INTER_LINEAR,
                              borderMode=cv2.BORDER_CONSTANT,borderValue=0)
            scene*=1-shifted
            # Small tight contact occlusion, beneath the sphere's silhouette.
            contact=np.exp(-(((xx-cx)/(r*.61))**2+
                              ((yy-(cy+r*.87))/(r*.15))**2)*2.0)
            scene*=1-.15*contact[...,None]
            bounds,ball,alpha=ball_image(high,phase,(cx,cy),r)
            left,top,right,bottom=bounds
            patch=scene[top:bottom,left:right]
            scene[top:bottom,left:right]=patch*(1-alpha[...,None])+ball*alpha[...,None]
            stationary=scene
            final_center=(cx,cy,r,phase)
        else:
            scene=stationary
        # Preserve factual background pixels outside the physical edit region.
        edited=high[i].astype(np.float32)*(1-corridor[...,None])+scene*corridor[...,None]
        frame=np.asarray(Image.fromarray(np.rint(edited).clip(0,255).astype(np.uint8))
                          .resize((256,144),Image.Resampling.LANCZOS))
        output.append(frame)
    output=np.asarray(output,np.uint8)
    artifact_io.save(output,CASE)
    print(f'Saved {len(output)} frames. Tripled friction: stop at {stop_time:.3f}s; '
          f'center/radius/rotation phase={final_center}; path fraction={q:.6f}.')
    # Convenience video in addition to the required lossless RGB artifacts.
    writer=cv2.VideoWriter(str(CASE/'edited_video.mp4'),
                           cv2.VideoWriter_fourcc(*'mp4v'),24,(256,144))
    for f in output:
        writer.write(cv2.cvtColor(f,cv2.COLOR_RGB2BGR))
    writer.release()


if __name__=='__main__':
    render()
