+"""Counterfactual from the source pixels: 0.75 initial star-ball velocity.
+
+The fixed camera permits reuse of actual photographic object appearances.
+A constant rolling-deceleration fit is used before and after the wall rebound;
+with the reduced initial velocity the rebound stops short of the second ball.
+"""
+from pathlib import Path
+import cv2
+import numpy as np
+from PIL import Image
+import artifact_io
+
+ROOT=Path(__file__).resolve().parent
+
+def read_video():
+    cap=cv2.VideoCapture(str(ROOT/'source_video.mp4'))
+    result=[]
+    while True:
+        ok,bgr=cap.read()
+        if not ok:break
+        result.append(cv2.cvtColor(bgr,cv2.COLOR_BGR2RGB))
+    cap.release()
+    return np.asarray(result)
+
+def smooth_mask(mask):
+    mask=mask.astype(np.uint8)
+    mask=cv2.morphologyEx(mask,cv2.MORPH_CLOSE,np.ones((3,3),np.uint8))
+    contours,_=cv2.findContours(mask,cv2.RETR_EXTERNAL,cv2.CHAIN_APPROX_SIMPLE)
+    filled=np.zeros_like(mask)
+    if contours:
+        cv2.drawContours(filled,[max(contours,key=cv2.contourArea)],-1,255,-1)
+    return cv2.GaussianBlur(filled.astype(np.float32)/255,(3,3),.65)
+
+def generate():
+    original=np.load(ROOT/'source_pixels.npz')['rgb']
+    src=read_video()
+    H,W=src.shape[1:3]
+    yy,xx=np.indices((H,W),dtype=np.float32)
+    # Pixel locations/radii measured on the original fixed-camera trajectory.
+    tracks=np.array([
+        [1190,330,57],[1117,331,56],[1045,332,56],
+        [973,331,56],[901,330,55],[829,330,55],
+        [758,331,56],[688,330,56],[620,329,56],
+        [552,329,55],[485,330,55],[418,329,55],
+        [422,324,56],[443,330,59],[460,343,61],
+        [467,352,64],[474,361,66],[479,369,68],
+        [484,377,70], [490,385,72],[494,391,74],
+        [499,399,76],[505,405,78]],dtype=np.float32)
+
+    # Frame zero provides a pristine scene everywhere except the star ball's
+    # initial location. Later source pixels expose that small background area.
+    bg=src[0].astype(np.float32).copy()
+    hole=((xx-1190)**2/(110**2)+(yy-353)**2/(112**2))
+    feather=np.clip((1.2-hole)/.25,0,1)
+    bg=bg*(1-feather[...,None])+src[12].astype(np.float32)*feather[...,None]
+
+    # Keep the front, unaffected soccer ball as an occluding foreground layer.
+    br,bgch,bb=src[0].astype(np.float32).transpose(2,0,1)
+    front=((br-bb<48)&(br-bgch<32)&((xx-502)**2+(yy-461)**2<76**2))
+    front=smooth_mask(front)
+
+    sprites=[]
+    for k,(cx,cy,rad) in enumerate(tracks):
+        image=src[k].astype(np.float32)
+        rr,gg,bl=image.transpose(2,0,1)
+        disk=(xx-cx)**2+(yy-cy)**2
+        # Nonlearned color segmentation supplies the original antialiased
+        # silhouette, including the pale stars and partially occluded edges.
+        obj=((rr-gg>19)&(bl-gg>-29)&(rr>145)&(disk<(rad+5)**2))
+        alpha=smooth_mask(obj)
+        # Recover the soft moving contact shadow as an additive illumination
+        # change, rather than transplanting floorboard pixels with the sprite.
+        diff=image-bg
+        extent=np.exp(-((xx-cx)/(rad*1.55))**6-((yy-cy-rad*.78)/(rad*.66))**6)
+        extent*=np.clip((yy-cy-rad*.22)/(rad*.40),0,1)
+        extent*=(1-alpha)*(1-front)
+        negative=np.minimum(np.mean(diff,axis=2),0)
+        darkness=np.maximum(np.max(np.abs(diff),axis=2)-5,0)
+        shadow=diff*np.clip(darkness/10,0,1)[...,None]*extent[...,None]
+        # Only negative changes belong to ground contact shadows.
+        shadow*=np.clip(-negative/4,0,1)[...,None]
+        sprites.append((image*alpha[...,None],alpha,shadow))
+
+    # Before impact x(t)=a*t^2+b*t+c. The observed rolling deceleration
+    # stays fixed while only the initial velocity is multiplied by 0.75.
+    a,b,c=.356393606,-73.9308192,1188.00824
+    wall_source=11.0
+    wall_x=a*wall_source**2+b*wall_source+c
+    roots=np.roots([a,.75*b,c-wall_x])
+    wall_time=float(min(r.real for r in roots if abs(r.imag)<1e-8 and r.real>0))
+    rebound_ratio=(.75*b+2*a*wall_time)/(b+2*a*wall_source)
+    # In ground-plane coordinates the original post-rebound travel follows
+    # 0.9824*u-0.02966*u^2. Its stopping distance scales with speed squared.
+    stop_original=16.5628
+    stop_new=stop_original*rebound_ratio
+
+    output=[]
+    source_times=[]
+    for n in range(96):
+        t=float(n)
+        if t<=wall_time:
+            delta=.75*b*t+a*t*t
+            s=(-b-np.sqrt(b*b+4*a*delta))/(2*a)
+        else:
+            dt=min(t-wall_time,stop_new)
+            normalized_distance=2*rebound_ratio*dt/stop_original-(dt/stop_original)**2
+            u=stop_original*(1-np.sqrt(max(0,1-normalized_distance)))
+            s=wall_source+u
+        s=float(np.clip(s,0,len(tracks)-1.001))
+        source_times.append(s)
+        lo=int(np.floor(s));hi=lo+1;mix=s-lo
+        target=tracks[lo]*(1-mix)+tracks[hi]*mix
+        color=np.zeros_like(bg)
+        alpha=np.zeros((H,W),np.float32)
+        shadows=np.zeros_like(bg)
+        for k,weight in [(lo,1-mix),(hi,mix)]:
+            if weight<1e-6:continue
+            cx,cy,rad=tracks[k]
+            scale=target[2]/rad
+            matrix=np.array([[scale,0,target[0]-scale*cx],
+                             [0,scale,target[1]-scale*cy]],np.float32)
+            p,mask,shadow=sprites[k]
+            color+=weight*cv2.warpAffine(p,matrix,(W,H),flags=cv2.INTER_LINEAR)
+            alpha+=weight*cv2.warpAffine(mask,matrix,(W,H),flags=cv2.INTER_LINEAR)
+            shadows+=weight*cv2.warpAffine(shadow,matrix,(W,H),flags=cv2.INTER_LINEAR)
+        frame=(bg+shadows)*(1-alpha[...,None])+color
+        frame=frame*(1-front[...,None])+bg*front[...,None]
+        small=cv2.resize(np.clip(frame,0,255).astype(np.uint8),(256,144),interpolation=cv2.INTER_AREA)
+        if n==0:small=original[0].copy()
+        output.append(small)
+    frames=np.array(output,dtype=np.uint8)
+    artifact_io.save(frames,ROOT)
+    # A standard MP4 accompanies the required lossless RGB output and player.
+    writer=cv2.VideoWriter(str(ROOT/'edited_video.mp4'),cv2.VideoWriter_fourcc(*'mp4v'),24,(256,144))
+    for frame in frames:writer.write(cv2.cvtColor(frame,cv2.COLOR_RGB2BGR))
+    writer.release()
+    print('Saved',frames.shape,'wall time',round(wall_time/24,3),
+          'rebound velocity ratio',round(rebound_ratio,3),
+          'settled source pose',round(source_times[-1],3))
+
+if __name__=='__main__':generate()

\"\"\"Counterfactual from the source pixels: 0.75 initial star-ball velocity.

The fixed camera permits reuse of actual photographic object appearances.
A constant rolling-deceleration fit is used before and after the wall rebound;
with the reduced initial velocity the rebound stops short of the second ball.
\"\"\"
from pathlib import Path
import cv2
import numpy as np
from PIL import Image
import artifact_io

ROOT=Path(__file__).resolve().parent

def read_video():
    cap=cv2.VideoCapture(str(ROOT/'source_video.mp4'))
    result=[]
    while True:
        ok,bgr=cap.read()
        if not ok:break
        result.append(cv2.cvtColor(bgr,cv2.COLOR_BGR2RGB))
    cap.release()
    return np.asarray(result)

def smooth_mask(mask):
    mask=mask.astype(np.uint8)
    mask=cv2.morphologyEx(mask,cv2.MORPH_CLOSE,np.ones((3,3),np.uint8))
    contours,_=cv2.findContours(mask,cv2.RETR_EXTERNAL,cv2.CHAIN_APPROX_SIMPLE)
    filled=np.zeros_like(mask)
    if contours:
        cv2.drawContours(filled,[max(contours,key=cv2.contourArea)],-1,255,-1)
    return cv2.GaussianBlur(filled.astype(np.float32)/255,(3,3),.65)

def generate():
    original=np.load(ROOT/'source_pixels.npz')['rgb']
    src=read_video()
    H,W=src.shape[1:3]
    yy,xx=np.indices((H,W),dtype=np.float32)
    # Pixel locations/radii measured on the original fixed-camera trajectory.
    tracks=np.array([
        [1190,330,57],[1117,331,56],[1045,332,56],
        [973,331,56],[901,330,55],[829,330,55],
        [758,331,56],[688,330,56],[620,329,56],
        [552,329,55],[485,330,55],[418,329,55],
        [422,324,56],[443,330,59],[460,343,61],
        [467,352,64],[474,361,66],[479,369,68],
        [484,377,70], [490,385,72],[494,391,74],
        [499,399,76],[505,405,78]],dtype=np.float32)

    # Frame zero provides a pristine scene everywhere except the star ball's
    # initial location. Later source pixels expose that small background area.
    bg=src[0].astype(np.float32).copy()
    hole=((xx-1190)**2/(110**2)+(yy-353)**2/(112**2))
    feather=np.clip((1.2-hole)/.25,0,1)
    bg=bg*(1-feather[...,None])+src[12].astype(np.float32)*feather[...,None]

    # Keep the front, unaffected soccer ball as an occluding foreground layer.
    br,bgch,bb=src[0].astype(np.float32).transpose(2,0,1)
    front=((br-bb<48)&(br-bgch<32)&((xx-502)**2+(yy-461)**2<76**2))
    front=smooth_mask(front)

    sprites=[]
    for k,(cx,cy,rad) in enumerate(tracks):
        image=src[k].astype(np.float32)
        rr,gg,bl=image.transpose(2,0,1)
        disk=(xx-cx)**2+(yy-cy)**2
        # Nonlearned color segmentation supplies the original antialiased
        # silhouette, including the pale stars and partially occluded edges.
        obj=((rr-gg>19)&(bl-gg>-29)&(rr>145)&(disk<(rad+5)**2))
        alpha=smooth_mask(obj)
        # Recover the soft moving contact shadow as an additive illumination
        # change, rather than transplanting floorboard pixels with the sprite.
        diff=image-bg
        extent=np.exp(-((xx-cx)/(rad*1.55))**6-((yy-cy-rad*.78)/(rad*.66))**6)
        extent*=np.clip((yy-cy-rad*.22)/(rad*.40),0,1)
        extent*=(1-alpha)*(1-front)
        negative=np.minimum(np.mean(diff,axis=2),0)
        darkness=np.maximum(np.max(np.abs(diff),axis=2)-5,0)
        shadow=diff*np.clip(darkness/10,0,1)[...,None]*extent[...,None]
        # Only negative changes belong to ground contact shadows.
        shadow*=np.clip(-negative/4,0,1)[...,None]
        sprites.append((image*alpha[...,None],alpha,shadow))

    # Before impact x(t)=a*t"'^2+b*t+c. The observed rolling deceleration
    # stays fixed while only the initial velocity is multiplied by 0.75.
    a,b,c=.356393606,-73.9308192,1188.00824
    wall_source=11.0
    wall_x=a*wall_source**2+b*wall_source+c
    roots=np.roots([a,.75*b,c-wall_x])
    wall_time=float(min(r.real for r in roots if abs(r.imag)<1e-8 and r.real>0))
    rebound_ratio=(.75*b+2*a*wall_time)/(b+2*a*wall_source)
    # In ground-plane coordinates the original post-rebound travel follows
    # 0.9824*u-0.02966*u''^2. Its stopping distance scales with speed squared.
    stop_original=16.5628
    stop_new=stop_original*rebound_ratio

    output=[]
    source_times=[]
    for n in range(96):
        t=float(n)
        if t<=wall_time:
            delta=.75*b*t+a*t*t
            s=(-b-np.sqrt(b*b+4*a*delta))/(2*a)
        else:
            dt=min(t-wall_time,stop_new)
            normalized_distance=2*rebound_ratio*dt/stop_original-(dt/stop_original)**2
            u=stop_original*(1-np.sqrt(max(0,1-normalized_distance)))
            s=wall_source+u
        s=float(np.clip(s,0,len(tracks)-1.001))
        source_times.append(s)
        lo=int(np.floor(s));hi=lo+1;mix=s-lo
        target=tracks[lo]*(1-mix)+tracks[hi]*mix
        color=np.zeros_like(bg)
        alpha=np.zeros((H,W),np.float32)
        shadows=np.zeros_like(bg)
        for k,weight in [(lo,1-mix),(hi,mix)]:
            if weight<1e-6:continue
            cx,cy,rad=tracks[k]
            scale=target[2]/rad
            matrix=np.array([[scale,0,target[0]-scale*cx],
                             [0,scale,target[1]-scale*cy]],np.float32)
            p,mask,shadow=sprites[k]
            color+=weight*cv2.warpAffine(p,matrix,(W,H),flags=cv2.INTER_LINEAR)
            alpha+=weight*cv2.warpAffine(mask,matrix,(W,H),flags=cv2.INTER_LINEAR)
            shadows+=weight*cv2.warpAffine(shadow,matrix,(W,H),flags=cv2.INTER_LINEAR)
        frame=(bg+shadows)*(1-alpha[...,None])+color
        frame=frame*(1-front[...,None])+bg*front[...,None]
        small=cv2.resize(np.clip(frame,0,255).astype(np.uint8),(256,144),interpolation=cv2.INTER_AREA)
        if n==0:small=original[0].copy()
        output.append(small)
    frames=np.array(output,dtype=np.uint8)
    artifact_io.save(frames,ROOT)
    # A standard MP4 accompanies the required lossless RGB output and player.
    writer=cv2.VideoWriter(str(ROOT/'"'edited_video.mp4'),cv2.VideoWriter_fourcc(*'mp4v'),24,(256,144))
    for frame in frames:writer.write(cv2.cvtColor(frame,cv2.COLOR_RGB2BGR))
    writer.release()
    print('Saved',frames.shape,'wall time',round(wall_time/24,3),
          'rebound velocity ratio',round(rebound_ratio,3),
          'settled source pose',round(source_times[-1],3))

if __name__=='__main__':generate()
