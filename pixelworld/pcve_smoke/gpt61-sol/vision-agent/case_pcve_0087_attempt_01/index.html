+"""Counterfactual edit, using source pixels and a calibrated billiard model.
+
+The projected balls in the factual shot share one straight line. The parameter
+along that line is made projectively linear before measuring speed/friction.
+Only initial cue velocity changes; ball masses, drag and cushions stay fixed.
+"""
+from pathlib import Path
+import cv2
+import numpy as np
+from PIL import Image
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+N, FPS = 96, 24
+
+# Projective coordinate recovered from the motion and its constant rolling
+# deceleration. Coordinate 1 is the initial object-ball position, 0 its far
+# cushion contact. The opposite cushion has coordinate 2.
+VP = 285.89965255
+XB = 127.93806713
+Q0 = 1.0 / (VP - 170.0)
+QS = Q0 - 1.0 / (VP - XB)
+
+def screen_x(s):
+    return VP - 1.0 / (Q0 - QS * s)
+
+def parameter(x):
+    return (Q0 - 1.0 / (VP - x)) / QS
+
+
+def simulate():
+    p = np.array([parameter(85.79769610), 1.0], dtype=np.float64)
+    # Original initial longitudinal velocity -0.9659254 per second.
+    v = np.array([-0.96592541 * 4.0, 0.0], dtype=np.float64)
+    friction = 0.1137
+    contact_distance = 0.081
+    ball_restitution = 0.90
+    cushion_restitution = 0.78
+    dt = 1.0 / (FPS * 200)
+    result = []
+    events = []
+    for frame in range(N):
+        result.append(p.copy())
+        if frame == N - 1:
+            break
+        for sub in range(200):
+            # Integrate the rolling resistance analytically over the substep.
+            old = v.copy()
+            v = np.sign(v) * np.maximum(np.abs(v) - friction * dt, 0.0)
+            p += 0.5 * (old + v) * dt
+            for j in range(2):
+                if p[j] < 0.0:
+                    p[j] = -p[j]
+                    v[j] = abs(v[j]) * cushion_restitution
+                    events.append((round((frame + sub/200)/FPS, 4), j, 'far cushion'))
+                if p[j] > 2.0:
+                    p[j] = 4.0 - p[j]
+                    v[j] = -abs(v[j]) * cushion_restitution
+                    events.append((round((frame + sub/200)/FPS, 4), j, 'near cushion'))
+            delta = p[1] - p[0]
+            dist = abs(delta)
+            if dist < contact_distance:
+                normal = 1.0 if delta >= 0.0 else -1.0
+                approach = (v[1] - v[0]) * normal
+                overlap = contact_distance - dist
+                p[0] -= 0.5 * overlap * normal
+                p[1] += 0.5 * overlap * normal
+                if approach < 0.0:
+                    impulse = -(1.0 + ball_restitution) * approach * 0.5
+                    v[0] -= impulse * normal
+                    v[1] += impulse * normal
+                    events.append((round((frame + sub/200)/FPS, 4), 'ball collision'))
+    return np.array(result), events
+
+
+def first_full_frame():
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    ok, frame = cap.read()
+    cap.release()
+    if not ok:
+        raise RuntimeError('Unable to read source video')
+    return cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
+
+
+def clean_plate(source):
+    # Both edit regions are entirely on uninterrupted green felt. Remove only
+    # the two starting balls and their contact/cast shadows, at native scale.
+    mask = np.zeros(source.shape[:2], np.uint8)
+    for cx, cy, radius, shadow_x, shadow_y, sx, sy in [
+        (429, 473, 25, 423, 499, 35, 22),
+        (640, 351, 19, 635, 372, 27, 17),
+    ]:
+        cv2.circle(mask, (cx, cy), radius, 255, -1)
+        cv2.ellipse(mask, (shadow_x, shadow_y), (sx, sy), 0, 0, 360, 255, -1)
+    # Native texture and shading are recovered locally; no scene synthesis.
+    plate = cv2.inpaint(source, mask, 7, cv2.INPAINT_TELEA)
+    return plate
+
+
+def sprite(source, plate, cx, cy, radius):
+    x0, y0 = int(cx) - 52, int(cy) - 52
+    raw = source[y0:y0+105, x0:x0+105].astype(np.float32)
+    bg = plate[y0:y0+105, x0:x0+105].astype(np.float32)
+    y, x = np.mgrid[y0:y0+105, x0:x0+105]
+    distance = np.hypot(x - cx, y - cy)
+    alpha = np.clip((radius + 0.65 - distance) / 1.30, 0.0, 1.0).astype(np.float32)
+    # Unpremultiply native antialiasing against the original felt. Keep the
+    # exact photographed illumination, reflections and black ball's number.
+    premul = (raw - bg * (1.0 - alpha[..., None]))
+    premul[alpha == 0] = 0.0
+    premul = np.clip(premul, 0, 255 * alpha[..., None])
+    # Native shadow, separated so balls correctly occlude each other's shadows.
+    ratio = np.clip(raw / np.maximum(bg, 1), 0.25, 1.0)
+    outer = distance > radius + 1.0
+    shadow_area = (((x - cx + 5) / (radius*1.9))**2 +
+                   ((y - cy - radius*1.18) / (radius*1.15))**2) < 1.0
+    strength = np.where(outer & shadow_area, 1.0 - ratio.mean(2), 0.0)
+    # Suppress tiny codec differences outside the cast shadow.
+    strength = np.where(strength > 0.025, strength, 0.0).astype(np.float32)
+    return {'center': (cx,cy), 'radius': radius, 'origin': (x0,y0),
+            'alpha': alpha, 'premul': premul, 'shadow': strength}
+
+
+def warp_patch(array, sprite, x, y, scale, shape):
+    cx,cy = sprite['center']; x0,y0 = sprite['origin']
+    matrix = np.array([[scale, 0, x + scale*(x0-cx)],
+                       [0, scale, y + scale*(y0-cy)]], np.float32)
+    return cv2.warpAffine(array, matrix, (shape[1],shape[0]),
+                          flags=cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)
+
+
+def generate():
+    src = np.load(ROOT/'source_pixels.npz')['rgb']
+    full = first_full_frame()
+    plate = clean_plate(full)
+    balls = [sprite(full,plate,428.98848052,473.06202295,20.08969259),
+             sprite(full,plate,639.69033566,351.12280732,15.02458877)]
+    positions, events = simulate()
+    frames = []
+    for i in range(N):
+        if i == 0:
+            frames.append(src[0].copy())
+            continue
+        image = plate.astype(np.float32).copy()
+        states = []
+        for j,s in enumerate(positions[i]):
+            sp = balls[j]
+            sx = screen_x(s)
+            initial_x = sp['center'][0]/5
+            initial_y = sp['center'][1]/5
+            # The source's ball-center loci are coincident to within 0.1 px.
+            sy = initial_y - 0.584*(sx - initial_x)
+            scale = (VP - sx) / (VP - initial_x)
+            states.append((sx*5, sy*5, scale, sp))
+        for x,y,scale,sp in states:
+            shadow = warp_patch(sp['shadow'],sp,x,y,scale,image.shape)
+            image *= 1.0 - shadow[...,None]
+        # Render from far to near, respecting perspective overlap at impacts.
+        for x,y,scale,sp in sorted(states, key=lambda state: state[1]):
+            alpha = warp_patch(sp['alpha'],sp,x,y,scale,image.shape)
+            color = warp_patch(sp['premul'],sp,x,y,scale,image.shape)
+            image = image*(1.0-alpha[...,None]) + color
+        rgb = np.clip(image,0,255).round().astype(np.uint8)
+        rgb = np.array(Image.fromarray(rgb).resize((256,144), Image.Resampling.LANCZOS))
+        frames.append(rgb)
+    frames = np.asarray(frames,np.uint8)
+    artifact_io.save(frames,ROOT)
+    print('Saved', frames.shape, 'RGB frames at', FPS, 'fps')
+    print('Physical events:',events)
+
+if __name__ == '__main__':
+    generate()

\"\"\"Counterfactual edit, using source pixels and a calibrated billiard model.

The projected balls in the factual shot share one straight line. The parameter
along that line is made projectively linear before measuring speed/friction.
Only initial cue velocity changes; ball masses, drag and cushions stay fixed.
\"\"\"
from pathlib import Path
import cv2
import numpy as np
from PIL import Image
import artifact_io

ROOT = Path(__file__).resolve().parent
N, FPS = 96, 24

# Projective coordinate recovered from the motion and its constant rolling
# deceleration. Coordinate 1 is the initial object-ball position, 0 its far
# cushion contact. The opposite cushion has coordinate 2.
VP = 285.89965255
XB = 127.93806713
Q0 = 1.0 / (VP - 170.0)
QS = Q0 - 1.0 / (VP - XB)

def screen_x(s):
    return VP - 1.0 / (Q0 - QS * s)

def parameter(x):
    return (Q0 - 1.0 / (VP - x)) / QS


def simulate():
    p = np.array([parameter(85.79769610), 1.0], dtype=np.float64)
    # Original initial longitudinal velocity -0.9659254 per second.
    v = np.array([-0.96592541 * 4.0, 0.0], dtype=np.float64)
    friction = 0.1137
    contact_distance = 0.081
    ball_restitution = 0.90
    cushion_restitution = 0.78
    dt = 1.0 / (FPS * 200)
    result = []
    events = []
    for frame in range(N):
        result.append(p.copy())
        if frame == N - 1:
            break
        for sub in range(200):
            # Integrate the rolling resistance analytically over the substep.
            old = v.copy()
            v = np.sign(v) * np.maximum(np.abs(v) - friction * dt, 0.0)
            p += 0.5 * (old + v) * dt
            for j in range(2):
                if p[j] < 0.0:
                    p[j] = -p[j]
                    v[j] = abs(v[j]) * cushion_restitution
                    events.append((round((frame + sub/200)/FPS, 4), j, 'far cushion'))
                if p[j] > 2.0:
                    p[j] = 4.0 - p[j]
                    v[j] = -abs(v[j]) * cushion_restitution
                    events.append((round((frame + sub/200)/FPS, 4), j, 'near cushion'))
            delta = p[1] - p[0]
            dist = abs(delta)
            if dist < contact_distance:
                normal = 1.0 if delta >= 0.0 else -1.0
                approach = (v[1] - v[0]) * normal
                overlap = contact_distance - dist
                p[0] -= 0.5 * overlap * normal
                p[1] += 0.5 * overlap * normal
                if approach < 0.0:
                    impulse = -(1.0 + ball_restitution) * approach * 0.5
                    v[0] -= impulse * normal
                    v[1] += impulse * normal
                    events.append((round((frame + sub/200)/FPS, 4), 'ball collision'))
    return np.array(result), events


def first_full_frame():
    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
    ok, frame = cap.read()
    cap.release()
    if not ok:
        raise RuntimeError('Unable to read source video')
    return cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)


def clean_plate(source):
    # Both edit regions are entirely on uninterrupted green felt. Remove only
    # the two starting balls and their contact/cast shadows, at native scale.
    mask = np.zeros(source.shape[:2], np.uint8)
    for cx, cy, radius, shadow_x, shadow_y, sx, sy in [
        (429, 473, 25, 423, 499, 35, 22),
        (640, 351, 19, 635, 372, 27, 17),
    ]:
        cv2.circle(mask, (cx, cy), radius, 255, -1)
        cv2.ellipse(mask, (shadow_x, shadow_y), (sx, sy), 0, 0, 360, 255, -1)
    # Native texture and shading are recovered locally; no scene synthesis.
    plate = cv2.inpaint(source, mask, 7, cv2.INPAINT_TELEA)
    return plate


def sprite(source, plate, cx, cy, radius):
    x0, y0 = int(cx) - 52, int(cy) - 52
    raw = source[y0:y0+105, x0:x0+105].astype(np.float32)
    bg = plate[y0:y0+105, x0:x0+105].astype(np.float32)
    y, x = np.mgrid[y0:y0+105, x0:x0+105]
    distance = np.hypot(x - cx, y - cy)
    alpha = np.clip((radius + 0.65 - distance) / 1.30, 0.0, 1.0).astype(np.float32)
    # Unpremultiply native antialiasing against the original felt. Keep the
    # exact photographed illumination, reflections and black ball's number.
    premul = (raw - bg * (1.0 - alpha[..., None]))
    premul[alpha == 0] = 0.0
    premul = np.clip(premul, 0, 255 * alpha[..., None])
    # Native shadow, separated so balls correctly occlude each other's shadows.
    ratio = np.clip(raw / np.maximum(bg, 1), 0.25, 1.0)
    outer = distance > radius + 1.0
    shadow_area = (((x - cx + 5) / (radius*1.9))**2 +
                   ((y - cy - radius*1.18) / (radius*1.15))**2) < 1.0
    strength = np.where(outer & shadow_area, 1.0 - ratio.mean(2), 0.0)
    # Suppress tiny codec differences outside the cast shadow.
    strength = np.where(strength > 0.025, strength, 0.0).astype(np.float32)
    return {'center': (cx,cy), 'radius': radius, 'origin': (x0,y0),
            'alpha': alpha, 'premul': premul, 'shadow': strength}


def warp_patch(array, sprite, x, y, scale, shape):
    cx,cy = sprite['center']; x0,y0 = sprite['origin']
    matrix = np.array([[scale, 0, x + scale*(x0-cx)],
                       [0, scale, y + scale*(y0-cy)]], np.float32)
    return cv2.warpAffine(array, matrix, (shape[1],shape[0]),
                          flags=cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)


def generate():
    src = np.load(ROOT/'source_pixels.npz')['rgb']
    full = first_full_frame()
    plate = clean_plate(full)
    balls = [sprite(full,plate,428.98848052,473.06202295,20.08969259),
             sprite(full,plate,639.69033566,351.12280732,15.02458877)]
    positions, events = simulate()
    frames = []
    for i in range(N):
        if i == 0:
            frames.append(src[0].copy())
            continue
        image = plate.astype(np.float32).copy()
        states = []
        for j,s in enumerate(positions[i]):
            sp = balls[j]
            sx = screen_x(s)
            initial_x = sp['center'][0]/5
            initial_y = sp['center'][1]/5
            # The source's ball-center loci are coincident to within 0.1 px.
            sy = initial_y - 0.584*(sx - initial_x)
            scale = (VP - sx) / (VP - initial_x)
            states.append((sx*5, sy*5, scale, sp))
        for x,y,scale,sp in states:
            shadow = warp_patch(sp['shadow'],sp,x,y,scale,image.shape)
            image *= 1.0 - shadow[...,None]
        # Render from far to near, respecting perspective overlap at impacts.
        for x,y,scale,sp in sorted(states, key=lambda state: state[1]):
            alpha = warp_patch(sp['alpha'],sp,x,y,scale,image.shape)
            color = warp_patch(sp['premul'],sp,x,y,scale,image.shape)
            image = image*(1.0-alpha[...,None]) + color
        rgb = np.clip(image,0,255).round().astype(np.uint8)
        rgb = np.array(Image.fromarray(rgb).resize((256,144), Image.Resampling.LANCZOS))
        frames.append(rgb)
    frames = np.asarray(frames,np.uint8)
    artifact_io.save(frames,ROOT)
    print('Saved', frames.shape, 'RGB frames at', FPS, 'fps')
    print('Physical events:',events)

if __name__ == '__main__':
    generate()
