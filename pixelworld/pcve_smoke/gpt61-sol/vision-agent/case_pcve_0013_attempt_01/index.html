\"\"\"Source-pixel counterfactual: fivefold block-floor Coulomb friction.\"\"\"
from pathlib import Path
import cv2
import numpy as np
from PIL import Image
from scipy.ndimage import label, gaussian_filter
import artifact_io

ROOT=Path(__file__).resolve().parent


def load_source():
    cap=cv2.VideoCapture(str(ROOT/'source_video.mp4'))
    src=[]
    while True:
        ok,bgr=cap.read()
        if not ok: break
        src.append(cv2.cvtColor(bgr,cv2.COLOR_BGR2RGB))
    cap.release()
    return np.stack(src)


def circle_measure(im, low, i):
    p=low.astype(float)
    m=(p[:,:,0]>p[:,:,1]*1.65)&(p[:,:,0]>p[:,:,2]*2.5)&(p[:,:,0]>125)
    labs,_=label(m);bc=np.bincount(labs.ravel());bc[0]=0
    y,x=np.where(labs==bc.argmax());cx=x.mean()+1.1;cy=y.mean()+1.3
    r=15+.15*min(i,12)+.043*max(i-12,0)
    yy,xx=np.mgrid[:720,:1280]
    p=im.astype(float)
    m=((p[:,:,0]>p[:,:,1]*1.8)&(p[:,:,0]>p[:,:,2]*2.5)&(p[:,:,0]>90)&(abs(xx-cx*5)<(r+3)*5)&(abs(yy-cy*5)<(r+3)*5)).astype('uint8')*255
    m=cv2.morphologyEx(m,cv2.MORPH_CLOSE,np.ones((5,5),dtype='uint8'))
    c=max(cv2.findContours(m,cv2.RETR_EXTERNAL,cv2.CHAIN_APPROX_NONE)[0],key=cv2.contourArea).reshape(-1,2)
    c=c[c[:,1]<(cy+.2*r)*5]
    if 11<=i<=14:c=c[c[:,0]<(cx+.4*r)*5]
    X,Y=c.T
    v=np.linalg.lstsq(np.column_stack((2*X,2*Y,np.ones_like(X))),X**2+Y**2,rcond=None)[0]
    return np.r_[v[:2],np.sqrt(v[2]+v[0]**2+v[1]**2)]


def alpha_poly(vertices,shape=(720,1280)):
    a=np.zeros(shape,np.uint8)
    cv2.fillPoly(a,[np.round(vertices).astype(np.int32)],255)
    return cv2.GaussianBlur(a.astype(np.float32)/255,(3,3),.5)


def generate():
    src=load_source();low=np.load(ROOT/'source_pixels.npz')['rgb']
    n,h,w,_=src.shape
    yy,xx=np.mgrid[:h,:w]
    circles=np.array([circle_measure(src[i],low[i],i) for i in range(n)])
    # Track the source block to estimate its original deceleration.
    gray=[cv2.cvtColor(im,cv2.COLOR_RGB2GRAY) for im in low]
    m=np.zeros((144,256),np.uint8);m[65:95,114:156]=255
    p=cv2.goodFeaturesToTrack(gray[0],50,.015,3,mask=m);q=p.copy()

        q,st,err=cv2.calcOpticalFlowPyrLK(gray[i-1],gray[i],q,None,winSize=(11,11),maxLevel=2)
        displacement.append(tuple(np.median(q-p,axis=0).ravel()))
    displacement=np.array(displacement)
    # A finite constant-deceleration slide, with acceleration multiplied by five.
    # Source fit gives ~7.2 px/frame and ~.30 px/frame"'^2 after impact.
    v0=7.27;a0=.303;friction_multiplier=5.0
    a=a0*friction_multiplier
    stop=v0/a;distance=v0*stop-.5*a*stop*stop

    A=[547,304];B=[638,274];C=[817,304];D=[749,346]
    E=[748,515];F=[551,478];G=[820,477]
    vertices=np.array([A,B,C,D,E,F,G],np.float32)
    faces=[[0,1,2,3],[0,3,4,5],[3,2,6,4]]
    silhouette=vertices[[0,1,2,6,4,5]]

    # moving bodies and their shadow neighborhoods. No synthetic floor texture.
    bg=src[0].copy()

    valid_frames=[]

        cx,cy,r=circles[i]

        ball=((xx-cx)**2+(yy-cy)**2 < (r+9)**2)

        dx,dy=displacement[i]*5

    allowed=((xx>520)&(xx<930)&(yy>375)&(yy<575))

    # A slightly wider front and shorter right face are the modest yaw left

    deformation=np.array([[-8,0],[-4,0],[-5,0],[6,0],[6,0],[-8,0],[-5,0]],np.float32)
    source_face_masks=[alpha_poly(vertices[face]) for face in faces]

    # much earlier stopped block. Recoil is small; residual spin dissipates.
    contact_index=31

        canvas*=1-sh[:,:,None]

        if i<=contact_index:

        else:

            # Tiny return motion after contact with the high-friction block.
            target=contact_circle.copy()
            target[0]-=5.2*(1-np.exp(-dt/7))

        ball_alpha=np.zeros((h,w),np.float32)
        for index,weight in [(idx0,1-blend),(idx1,blend)]:

            d=np.sqrt((xx-cx)**2+(yy-cy)**2)

        cx,cy,r=target

        canvas*=1-bs[:,:,None]

        # edge, exactly as in the original collision.
        for face,mask in zip(faces,source_face_masks):

            canvas=canvas*(1-am[:,:,None])+warp

        frames.append(np.array(image.resize((256,144),Image.Resampling.LANCZOS)))

    writer=cv2.VideoWriter(str(ROOT/'edited_video.mp4'),cv2.VideoWriter_fourcc(*'mp4v'),24,(256,144))

    writer.release()
    Image.fromarray(bg).resize((640,360)).save(ROOT/'.tmp/background.png')
    print('Saved',output.shape,'block stop frame',12+stop,'slide pixels',distance)
