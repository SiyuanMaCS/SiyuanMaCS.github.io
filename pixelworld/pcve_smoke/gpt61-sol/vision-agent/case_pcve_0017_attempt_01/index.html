+"""Counterfactual restitution edit, using source pixels and analytic rigid motion.
+
+No learned resources are used. The cabinet impact is reconstructed in floor-plane
+coordinates; only its normal impulse is changed, retaining tangential velocity.
+"""
+from pathlib import Path
+import numpy as np
+import cv2
+from PIL import Image
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent

+def rotation(axis, angle):
+    axis=np.asarray(axis,dtype=float);axis/=np.linalg.norm(axis)
+    x,y,z=axis
+    K=np.array([[0,-z,y],[z,0,-x],[-y,x,0]])
+    return np.eye(3)+np.sin(angle)*K+(1-np.cos(angle))*(K@K)
+
+def load_texture():
+    # Use the last wholly unoccluded post-impact source silhouette as material.
+    cap=cv2.VideoCapture(str(ROOT/'source_video.mp4'))

+def star_on_sphere(normals, direction, angular_size, phase):
+    """Small five-point print on a sphere's previously unseen hemisphere."""
+    d=np.array(direction,dtype=float);d/=np.linalg.norm(d)
+    tangent=np.cross(d,[0,1,0]);tangent/=np.linalg.norm(tangent)
+    bitangent=np.cross(d,tangent)
+    u=normals@tangent;v=normals@bitangent;dot=normals@d
+    ang=np.arctan2(v,u)-phase
+    # Exact radial outline of the ten alternating vertices.
+    verts=np.stack([np.cos(np.arange(10)*np.pi/5),np.sin(np.arange(10)*np.pi/5)],axis=1)
+    verts*=np.where(np.arange(10)%2==0,angular_size,angular_size*.44)[:,None]
+    uv=np.stack([u*np.cos(phase)+v*np.sin(phase),-u*np.sin(phase)+v*np.cos(phase)],axis=-1)
+    inside=np.zeros(u.shape,dtype=bool)
+    for j in range(10):
+        a=verts[j];b=verts[(j+1)%10]
+        crossing=((a[1]>uv[...,1])!=(b[1]>uv[...,1])) & (uv[...,0] < (b[0]-a[0])*(uv[...,1]-a[1])/(b[1]-a[1]+1e-12)+a[0])
+        inside^=crossing
+    return inside & (dot>.85)

+    print('Saved',frames.shape,'normal restitution:',round(old_e,3),'->',round(new_e,3),'final star:',np.round(states[-1][1:],2))

No learned resources are used. The cabinet impact is reconstructed in floor-plane
coordinates; only its normal impulse is changed, retaining tangential velocity.
\"\"\"
from pathlib import Path
import numpy as np
import cv2
from PIL import Image
import artifact_io

ROOT = Path(__file__).resolve().parent

def rotation(axis, angle):
    axis=np.asarray(axis,dtype=float);axis/=np.linalg.norm(axis)
    x,y,z=axis
    K=np.array([[0,-z,y],[z,0,-x],[-y,x,0]])
    return np.eye(3)+np.sin(angle)*K+(1-np.cos(angle))*(K@K)

def load_texture():
    # Use the last wholly unoccluded post-impact source silhouette as material.
    cap=cv2.VideoCapture(str(ROOT/'source_video.mp4'))
    for i in range(13):
        ok,im=cap.read()
        if not ok: raise RuntimeError('Source decode failed')
    cap.release()
    return cv2.cvtColor(im,cv2.COLOR_BGR2RGB).astype(np.float32)

def star_on_sphere(normals, direction, angular_size, phase):
    \"\"\"Small five-point print on a sphere's previously unseen hemisphere.\"\"\"
    d=np.array(direction,dtype=float);d/=np.linalg.norm(d)
    tangent=np.cross(d,[0,1,0]);tangent/=np.linalg.norm(tangent)
    bitangent=np.cross(d,tangent)
    u=normals@tangent;v=normals@bitangent;dot=normals@d
    ang=np.arctan2(v,u)-phase
    # Exact radial outline of the ten alternating vertices.
    verts=np.stack([np.cos(np.arange(10)*np.pi/5),np.sin(np.arange(10)*np.pi/5)],axis=1)
    verts*=np.where(np.arange(10)%2==0,angular_size,angular_size*.44)[:,None]
    uv=np.stack([u*np.cos(phase)+v*np.sin(phase),-u*np.sin(phase)+v*np.cos(phase)],axis=-1)
    inside=np.zeros(u.shape,dtype=bool)

        inside''^=crossing
    return inside & (dot>.85)
