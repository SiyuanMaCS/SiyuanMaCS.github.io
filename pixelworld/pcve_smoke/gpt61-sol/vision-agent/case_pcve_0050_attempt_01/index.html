+"""Counterfactual curling shot: double red's initial velocity at time zero.
+
+The fixed rink is recovered by temporal median from the supplied source. Stone
+sprites are cut from that same source, and translated/scaled with the perspective
+of the rink. Trajectories use an equal-mass, weakly restitutive head-on impact.
+No extra scene assets or external renderer are used.
+"""
+from pathlib import Path
+import cv2
+import numpy as np
+from PIL import Image, ImageDraw
+from scipy.ndimage import gaussian_filter
+from scipy.optimize import least_squares
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+
+def load_video():
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    frames = []
+    while True:
+        ok, bgr = cap.read()
+        if not ok:
+            break
+        frames.append(cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB))
+    cap.release()
+    return np.asarray(frames)
+
+
+def track_stones(high):
+    """Color-handle positions, gated to exclude the red paint of the house."""
+    coords = []
+    yy, xx = np.indices(high.shape[1:3])
+    for t, frame in enumerate(high):
+        a = frame.astype(float)
+        r, g, b = a.transpose(2, 0, 1)
+        row = []
+        for j in range(2):
+            xp = (47.5 + 1.98*t if t <= 38 else 123-.15*(t-38)) if j == 0 else (
+                214.4-2.14*t if t <= 38 else 133+.16*(t-38))
+            if j == 0:
+                mask = (r-g > 60) & (r-b > 65) & (r > 140)
+            else:
+                mask = (r-b > 60) & (g-b > 55) & (g > 140)
+            mask &= (abs(xx/5-xp) < 4) & (abs(yy/5-(.0613*xp+63.6)) < 1.9)
+            y, x = np.where(mask)
+            row.extend([x.mean()/5, y.mean()/5])
+        coords.append(row)
+    return np.asarray(coords)
+
+
+def motion_model(coords):
+    # Fit perspective and constant ice drag to the uncollided source motion.
+    # x = 128 + z/(1-k*z), with z in convenient ground-plane units.
+    t = np.arange(38, dtype=float)
+    valid = np.ones((38, 2), dtype=bool)
+    valid[:8, 1] = False  # the bright reflection affects the initial handle centroid
+    def predicted(p):
+        k, zr, zy, vr, vy, drag = p
+        z = np.array([zr+vr*t-.5*drag*t*t,
+                      zy+vy*t+.5*drag*t*t]).T
+        return 128+z/(1-k*z)
+    p = least_squares(lambda p: (predicted(p)-coords[:38, [0, 2]])[valid],
+                      [.0003, -83, 83, 2.06, -2.08, .001]).x
+    k, zr, zy, vr, vy, drag = p
+    drag = max(.001, drag)
+    # The intervention changes only red's initial linear speed.
+    # Contact spacing calibrated on the source's frame-39 touching pair.
+    def unproject(x):
+        return (x-128)/(1+k*(x-128))
+    # Keep exact initial positions, compensating the tiny fit residual.
+    zr = unproject(coords[0, 0])
+    zy = unproject(coords[0, 2])
+    spacing = unproject(coords[38, 2])-unproject(coords[38, 0])
+    vr *= 2
+    contact = float(np.roots([drag, -(vr-vy), zy-zr-spacing])[1])
+    r_contact = zr+vr*contact-.5*drag*contact**2
+    y_contact = zy+vy*contact+.5*drag*contact**2
+    ur = vr-drag*contact
+    uy = vy+drag*contact
+    restitution = .095  # rebound in the source is ~9.5% of approach velocity
+    mean = .5*(ur+uy)
+    r_after = mean-.5*restitution*(ur-uy)
+    y_after = mean+.5*restitution*(ur-uy)
+    out = []
+    for t in range(96):
+        if t <= contact:
+            rz = zr+vr*t-.5*drag*t*t
+            yz = zy+vy*t+.5*drag*t*t
+        else:
+            dt = t-contact
+            rz = r_contact+r_after*dt-.5*drag*dt*dt
+            yz = y_contact+y_after*dt-.5*drag*dt*dt
+        xs = 128+np.array([rz, yz])/(1-k*np.array([rz, yz]))
+        # Preserve yellow's original trajectory until contact.
+        if t <= contact:
+            xs[1] = coords[t, 2]
+        out.append(xs)
+    return np.asarray(out), k, contact
+
+
+def sprite(high0, background, j, center):
+    # The contour follows the granite body, excluding the underlying painted
+    # line, while keeping the colored handle and its small metal supports.
+    if j == 0:
+        polygon = [(222,325),(227,322),(237,322),(241,323),(245,321),
+                   (248,325),(254,328),(258,330),(261,333),(263,338),
+                   (263,345),(260,350),(256,353),(250,356),(242,358),
+                   (232,357),(225,355),(219,352),(216,348),(214,343),
+                   (215,337),(218,332),(221,329)]
+    else:
+        polygon = [(1055,375),(1057,373),(1069,372),(1076,373),
+                   (1081,375),(1083,378),(1088,381),(1092,384),
+                   (1095,389),(1096,395),(1095,401),(1091,406),
+                   (1084,409),(1077,411),(1067,411),(1058,409),
+                   (1051,406),(1046,402),(1044,398),(1044,391),
+                   (1047,384),(1051,380)]
+    x, y = center
+    x0 = int(x)-45
+    y0 = int(y)-28
+    w, h = 96, 86
+    mask = Image.new('L', (w*4, h*4))
+    d = ImageDraw.Draw(mask)
+    d.polygon([((px-x0)*4, (py-y0)*4) for px, py in polygon], fill=255)
+    alpha = np.array(mask.resize((w,h), Image.Resampling.LANCZOS)).astype(float)/255
+    alpha = gaussian_filter(alpha, .35)
+    a = high0[y0:y0+h, x0:x0+w].astype(float)
+    bg = background[y0:y0+h, x0:x0+w].astype(float)
+    # Approximate matte decontamination at the antialiased silhouette.
+    fg = np.clip((a-bg*(1-alpha[:,:,None]))/np.maximum(alpha[:,:,None], .015),0,255)
+    premult = fg*alpha[:,:,None]
+    return premult.astype(np.float32), alpha.astype(np.float32), (x0-x,y0-y)
+
+
+def paste_sprite(canvas, spr, at, scales):
+    premult, alpha, offset = spr
+    sx, sy = scales
+    x, y = at
+    h,w = alpha.shape
+    # Place a faint grounded contact shadow before the stone.
+    cx=x+1.5*sx
+    cy=y+24*sy
+    rx=26*sx
+    ry=7*sy
+    x0=max(0,int(cx-rx*2.5));x1=min(canvas.shape[1],int(cx+rx*2.5)+1)
+    y0=max(0,int(cy-ry*2.5));y1=min(canvas.shape[0],int(cy+ry*2.5)+1)
+    if x0<x1 and y0<y1:
+        yy,xx=np.mgrid[y0:y1,x0:x1]
+        shadow=.14*np.exp(-.5*((xx-cx)/rx)**2-.5*((yy-cy)/ry)**2)
+        canvas[y0:y1,x0:x1]*=1-shadow[:,:,None]
+    ox=x+offset[0]*sx
+    oy=y+offset[1]*sy
+    left=max(0,int(np.floor(ox))-1)
+    top=max(0,int(np.floor(oy))-1)
+    right=min(canvas.shape[1],int(np.ceil(ox+w*sx))+1)
+    bottom=min(canvas.shape[0],int(np.ceil(oy+h*sy))+1)
+    if right<=left or bottom<=top:
+        return
+    matrix=np.array([[sx,0,ox-left],[0,sy,oy-top]],dtype=np.float32)
+    ap=cv2.warpAffine(alpha,matrix,(right-left,bottom-top),flags=cv2.INTER_LINEAR)
+    rgb=cv2.warpAffine(premult,matrix,(right-left,bottom-top),flags=cv2.INTER_LINEAR)
+    patch=canvas[top:bottom,left:right]
+    patch*=1-ap[:,:,None]
+    patch+=rgb
+
+
+def main():
+    high=load_video()
+    factual=np.load(ROOT/'source_pixels.npz')['rgb']
+    background=np.median(high[:30],axis=0).astype(np.uint8)
+    coords=track_stones(high)
+    paths,k,contact=motion_model(coords)
+    sprites=[sprite(high[0],background,j,coords[0,2*j:2*j+2]*5) for j in range(2)]
+    frames=[]
+    for t in range(96):
+        if t==0:
+            frames.append(factual[0].copy())
+            continue
+        # The background is stationary; original yellow pixels are retained
+        # without modification until it participates in the earlier collision.
+        if t <= contact:
+            canvas=high[t].astype(np.float32)
+            x,y=coords[t,:2]*5
+            x0=max(0,int(x)-45);x1=min(1280,int(x)+49)
+            y0=max(0,int(y)-28);y1=min(720,int(y)+58)
+            canvas[y0:y1,x0:x1]=background[y0:y1,x0:x1]
+            indices=[0]
+        else:
+            canvas=background.astype(np.float32)
+            indices=[0,1]
+        for j in indices:
+            x=paths[t,j]
+            y=.06135*x+63.55
+            if j==1 and t<=contact:
+                y=coords[t,3]
+            ratio=(1+k*(x-128))/(1+k*(coords[0,2*j]-128))
+            paste_sprite(canvas,sprites[j],(x*5,y*5),(ratio,ratio))
+        image=Image.fromarray(np.clip(np.rint(canvas),0,255).astype(np.uint8))
+        frames.append(np.asarray(image.resize((256,144),Image.Resampling.LANCZOS)))
+    frames=np.asarray(frames,dtype=np.uint8)
+    artifact_io.save(frames,ROOT)
+    np.save(ROOT/'.tmp/counterfactual_positions.npy',paths)
+    print('Saved 96 frames. Earlier contact: %.3f s (frame %.2f); final x: %s' % (
+          contact/24,contact+1,np.round(paths[-1],2)))
+
+
+if __name__=='__main__':
+    main()

The fixed rink is recovered by temporal median from the supplied source. Stone

def track_stones(high):
    \"\"\"Color-handle positions, gated to exclude the red paint of the house.\"\"\"
    coords = []
    yy, xx = np.indices(high.shape[1:3])
    for t, frame in enumerate(high):
        a = frame.astype(float)
        r, g, b = a.transpose(2, 0, 1)
        row = []
        for j in range(2):
            xp = (47.5 + 1.98*t if t <= 38 else 123-.15*(t-38)) if j == 0 else (
                214.4-2.14*t if t <= 38 else 133+.16*(t-38))
            if j == 0:
                mask = (r-g > 60) & (r-b > 65) & (r > 140)
            else:
                mask = (r-b > 60) & (g-b > 55) & (g > 140)

def motion_model(coords):
    # Fit perspective and constant ice drag to the uncollided source motion.
    # x = 128 + z/(1-k*z), with z in convenient ground-plane units.
    t = np.arange(38, dtype=float)
    valid = np.ones((38, 2), dtype=bool)
    valid[:8, 1] = False  # the bright reflection affects the initial handle centroid
    def predicted(p):
        k, zr, zy, vr, vy, drag = p
        z = np.array([zr+vr*t-.5*drag*t*t,
                      zy+vy*t+.5*drag*t*t]).T
        return 128+z/(1-k*z)
    p = least_squares(lambda p: (predicted(p)-coords[:38, [0, 2]])[valid],
                      [.0003, -83, 83, 2.06, -2.08, .001]).x
    k, zr, zy, vr, vy, drag = p
    drag = max(.001, drag)
    # The intervention changes only red's initial linear speed.
    # Contact spacing calibrated on the source's frame-39 touching pair.
    def unproject(x):
        return (x-128)/(1+k*(x-128))
    # Keep exact initial positions, compensating the tiny fit residual.
    zr = unproject(coords[0, 0])
    zy = unproject(coords[0, 2])
    spacing = unproject(coords[38, 2])-unproject(coords[38, 0])
    vr *= 2
    contact = float(np.roots([drag, -(vr-vy), zy-zr-spacing])[1])
    r_contact = zr+vr*contact-.5*drag*contact**2
    y_contact = zy+vy*contact+.5*drag*contact**2
    ur = vr-drag*contact
    uy = vy+drag*contact
    restitution = .095  # rebound in the source is ~9.5% of approach velocity
    mean = .5*(ur+uy)
    r_after = mean-.5*restitution*(ur-uy)
    y_after = mean+.5*restitution*(ur-uy)
    out = []
    for t in range(96):
        if t <= contact:
            rz = zr+vr*t-.5*drag*t*t
            yz = zy+vy*t+.5*drag*t*t
        else:
            dt = t-contact
            rz = r_contact+r_after*dt-.5*drag*dt*dt
            yz = y_contact+y_after*dt-.5*drag*dt*dt
        xs = 128+np.array([rz, yz])/(1-k*np.array([rz, yz]))
        # Preserve yellow's original trajectory until contact.
        if t <= contact:
            xs[1] = coords[t, 2]
        out.append(xs)
    return np.asarray(out), k, contact

def paste_sprite(canvas, spr, at, scales):
    premult, alpha, offset = spr
    sx, sy = scales
    x, y = at
    h,w = alpha.shape
    # Place a faint grounded contact shadow before the stone.
    cx=x+1.5*sx
    cy=y+24*sy
    rx=26*sx
    ry=7*sy
    x0=max(0,int(cx-rx*2.5));x1=min(canvas.shape[1],int(cx+rx*2.5)+1)
    y0=max(0,int(cy-ry*2.5));y1=min(canvas.shape[0],int(cy+ry*2.5)+1)
    if x0<x1 and y0<y1:
        yy,xx=np.mgrid[y0:y1,x0:x1]
        shadow=.14*np.exp(-.5*((xx-cx)/rx)**2-.5*((yy-cy)/ry)**2)
        canvas[y0:y1,x0:x1]*=1-shadow[:,:,None]
    ox=x+offset[0]*sx
    oy=y+offset[1]*sy
    left=max(0,int(np.floor(ox))-1)
    top=max(0,int(np.floor(oy))-1)
    right=min(canvas.shape[1],int(np.ceil(ox+w*sx))+1)
    bottom=min(canvas.shape[0],int(np.ceil(oy+h*sy))+1)
    if right<=left or bottom<=top:
        return
    matrix=np.array([[sx,0,ox-left],[0,sy,oy-top]],dtype=np.float32)
    ap=cv2.warpAffine(alpha,matrix,(right-left,bottom-top),flags=cv2.INTER_LINEAR)
    rgb=cv2.warpAffine(premult,matrix,(right-left,bottom-top),flags=cv2.INTER_LINEAR)
    patch=canvas[top:bottom,left:right]
    patch*=1-ap[:,:,None]
    patch+=rgb
