+"""Counterfactual initial-velocity edit, using only the supplied scene pixels.
+
+The initial velocity points uphill.  Keeping gravity, friction and the ramp
+fixed while multiplying that velocity by .6 makes the car turn around on the
+ramp, rather than reach the elevated end.  A two-contact height profile then
+lets it roll rear-first off the low end and onto the pavement.
+"""
+from pathlib import Path
+import math
+
+import cv2
+import numpy as np
+from PIL import Image, ImageDraw
+from scipy import ndimage
+
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+SCALE = 5
+
+
+def read_high_resolution():
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    images = []
+    for _ in range(18):
+        ok, image = cap.read()
+        if not ok:
+            raise RuntimeError('Incomplete source video')
+        images.append(cv2.cvtColor(image, cv2.COLOR_BGR2RGB))
+    cap.release()
+    return np.stack(images)
+
+
+def polygon_mask(points, shape):
+    image = Image.new('L', (shape[1], shape[0]))
+    ImageDraw.Draw(image).polygon(points, fill=255)
+    return np.asarray(image).astype(np.float32) / 255
+
+
+def rolling_pose(x):
+    # Source wheel-midpoint coordinates and the unchanged contact incline.
+    x0, y0 = 173.6, 112.9
+    slope = .36
+    ground_axle = 121.8
+    half_wheelbase = 10.
+    front = min(y0 + slope * (x - half_wheelbase - x0), ground_axle)
+    rear = min(y0 + slope * (x + half_wheelbase - x0), ground_axle)
+    ramp_angle = math.atan(slope)
+    angle = math.atan2(rear - front, 2 * half_wheelbase)
+    # Preserve the photographed pose on the incline, and level it progressively
+    # as the rear and then the front wheels cross the bottom edge.
+    source_wheel_angle = math.atan2(33, 100)
+    rotation = source_wheel_angle * (angle / ramp_angle - 1.)
+    return (front + rear) / 2., rotation, angle / ramp_angle
+
+
+def motion():
+    # Fit the initial uphill source acceleration from its first 10 positions.
+    # Units are output pixels and source-frame intervals.  Uphill slowdown is
+    # .91 px/frame^2; separate its gravitational and rolling-resistance parts.
+    initial_v = -.6 * 14.65
+    gravity = .73
+    resistance = .18
+    x, v = 173.6, initial_v
+    positions = [x]
+    states = [v]
+    # Small-step integration resolves the velocity reversal without a hold.
+    for _ in range(95):
+        for _ in range(100):
+            _, _, slope_fraction = rolling_pose(x)
+            g = gravity * slope_fraction
+            friction = resistance
+            if abs(v) < .002 and abs(g) <= friction:
+                v = 0.
+                continue
+            direction = math.copysign(1., v if abs(v) > .002 else g)
+            acceleration = g - direction * friction
+            vn = v + acceleration * .01
+            if g < friction and v * vn < 0:
+                vn = 0.
+            x += (v + vn) * .005
+            v = vn
+        positions.append(x)
+        states.append(v)
+    return np.asarray(positions), np.asarray(states)
+
+
+def generate():
+    source = np.load(ROOT / 'source_pixels.npz')['rgb']
+    high = read_high_resolution()
+    high_bg = np.median(high, axis=0).astype(np.float32)
+    low_bg = np.median(source[:18], axis=0).astype(np.float32)
+    h, w = high.shape[1:3]
+
+    # Tight hand-traced photographic silhouette: no generated replacement car.
+    silhouette = [
+        (777,541),(777,531),(781,520),(786,514),(798,510),
+        (812,509),(827,512),(833,515),(839,521),(844,526),
+        (851,525),(861,524),(867,522),(866,518),(860,513),
+        (860,510),(868,512),(878,516),(890,523),(902,529),
+        (914,534),(929,541),(937,547),(941,554),(943,563),
+        (942,572),(939,583),(936,591),(930,597),(924,600),
+        (916,599),(907,594),(896,589),(884,585),(872,581),
+        (860,577),(848,573),(837,569),(826,565),(815,561),
+        (804,557),(792,552),(783,547)
+    ]
+    alpha = polygon_mask(silhouette, (h,w))
+    alpha = cv2.GaussianBlur(alpha, (5,5), .55)
+    # The nearer rail must stay in front of the wheels when appropriate.
+    beam = [(60.6,76.3),(198.0,124.8),(194.7,133.6),(57.1,85.0)]
+    rail = polygon_mask([(x*SCALE,y*SCALE) for x,y in beam], (h,w))
+    rail = cv2.GaussianBlur(rail, (3,3), .4)
+
+    # The source matte is cropped to make subpixel warps inexpensive.
+    xlo, ylo, xhi, yhi = 750, 495, 960, 610
+    texture = high[0,ylo:yhi,xlo:xhi].astype(np.float32)
+    matte = alpha[ylo:yhi,xlo:xhi]
+    ch,cw = texture.shape[:2]
+    yy,xx = np.mgrid[:ch,:cw].astype(np.float32)
+    pivot = np.array([173.6*SCALE-xlo,112.9*SCALE-ylo])
+    wheels = [(818-xlo,549-ylo,11.), (917-xlo,580-ylo,13.)]
+    positions, velocities = motion()
+    frames = []
+
+    for i, x in enumerate(positions):
+        if i == 0:
+            frames.append(source[0].copy())
+            continue
+        if x - 24 > 256:
+            frames.append(low_bg.astype(np.uint8))
+            continue
+        y, rotation, ramp_fraction = rolling_pose(float(x))
+        sprite = texture.copy()
+        # Rotate just the photographed hub interiors, retaining the tire and
+        # the car's shape. Their rotation reverses with the translation.
+        path = (x-positions[0]) * math.sqrt(1+.36**2)
+        for wx,wy, radius in wheels:
+            theta = path / (3.35)  # rolling angle, radians
+            c,s = math.cos(theta),math.sin(theta)
+            dx,dy = xx-wx,yy-wy
+            mx = wx+c*dx+s*dy
+            my = wy-s*dx+c*dy
+            hub = cv2.remap(texture,mx,my,cv2.INTER_LINEAR,
+                            borderMode=cv2.BORDER_REPLICATE)
+            disk = np.clip((radius-np.sqrt(dx*dx+dy*dy))/1.2,0,1)
+            sprite = sprite*(1-disk[...,None])+hub*disk[...,None]
+
+        c,s = math.cos(rotation),math.sin(rotation)
+        linear = np.array([[c,-s],[s,c]],dtype=np.float32)
+        translation = np.array([x*SCALE,y*SCALE])-linear@pivot
+        affine = np.column_stack((linear,translation)).astype(np.float32)
+        transformed_alpha = cv2.warpAffine(matte,affine,(w,h),
+                                          flags=cv2.INTER_LINEAR)
+        foreground = cv2.warpAffine(sprite*matte[...,None],affine,(w,h),
+                                   flags=cv2.INTER_LINEAR)
+
+        # Soft contact shadow travels with the vehicle; the large stationary
+        # shadow of the ramp itself is already present in the background.
+        shadow = np.zeros((h,w),np.float32)
+        contact_y = y + 3.4
+        shadow_angle = math.degrees(math.atan(.36)*ramp_fraction)
+        cv2.ellipse(shadow,(round(x*SCALE),round(contact_y*SCALE)),
+                    (round(18*SCALE),round((1.1+1.0*(1-ramp_fraction))*SCALE)),
+                    shadow_angle,0,360,.24,-1,cv2.LINE_AA)
+        shadow = cv2.GaussianBlur(shadow,(15,15),2.2)
+        rendered = high_bg * (1-shadow[...,None])
+        rendered = rendered*(1-transformed_alpha[...,None])+foreground
+        rendered = rendered*(1-rail[...,None])+high_bg*rail[...,None]
+        # Apply only the moving-object delta to the original 256x144 scene.
+        # Everything outside that local edit remains the supplied pixels.
+        delta = cv2.resize(rendered-high_bg,(256,144),interpolation=cv2.INTER_AREA)
+        frame = np.clip(np.rint(low_bg+delta),0,255).astype(np.uint8)
+        frames.append(frame)
+
+    frames = np.stack(frames)
+    artifact_io.save(frames,ROOT)
+    np.savez_compressed(ROOT / '.tmp' / 'trajectory.npz',
+                        x=positions,v=velocities)
+    print('Saved',frames.shape,'RGB frames at 24 fps.')
+    print('Uphill turn at frame',positions.argmin(),
+          '; minimum x',round(positions.min(),2),
+          '; exit screen at frame',next((i for i,x in enumerate(positions)
+                                      if x-24>256),None))
+
+
+if __name__ == '__main__':
+    generate()
