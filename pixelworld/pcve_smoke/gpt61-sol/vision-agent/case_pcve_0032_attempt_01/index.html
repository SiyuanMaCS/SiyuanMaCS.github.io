+No simulation/rendering model is used: sphere pixels and floor lighting are reprojected.
+"""
+from pathlib import Path
+import json
+import cv2
+import numpy as np
+from PIL import Image
+from scipy.optimize import least_squares
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+def read_original():
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    hi = []
+    while True:
+        ok, image = cap.read()
+        if not ok:
+            break
+        hi.append(cv2.cvtColor(image, cv2.COLOR_BGR2RGB))
+    cap.release()
+    return np.asarray(hi)

+def track(hi, small):
+    # Color locates the object; its outer edge determines subpixel center and radius.
+    centers, radii = [], []
+    Y, X = np.mgrid[:144, :256]
+    for i, f in enumerate(small):
+        r, g, b = f.astype(float).transpose(2, 0, 1)
+        estimate = 128 + .88 * min(i, 12) + .35 * max(0, min(i-12, 50)) + .14 * max(0, i-62)
+        mask = (r-g > 42) & (r > 98) & (np.abs(X-estimate) < 12) & (Y < 92)
+        yy, xx = np.where(mask)
+        cx, cy = estimate, yy.min() + 10.7
+        for _ in range(4):
+            mask = (r-g > 36) & (r > 90) & ((X-cx)**2 + (Y-cy)**2 < 11.7**2)
+            yy, xx = np.where(mask)
+            cx, cy = (xx.min()+xx.max())/2, (yy.min()+yy.max())/2
+        hx, hy = (cx+.5)*5, (cy+.5)*5
+        ix, iy = int(hx), int(hy)
+        patch = hi[i, iy-65:iy+66, ix-65:ix+66]
+        edge = cv2.Canny(patch, 35, 65)
+        ey, ex = np.where(edge > 0)
+        ex, ey = ex+ix-65, ey+iy-65
+        d = np.hypot(ex-hx, ey-hy)
+        keep = (d > 52) & (d < 60)
+        ex, ey = ex[keep], ey[keep]
+        fit = least_squares(lambda p: np.hypot(ex-p[0], ey-p[1])-p[2],
+                            [hx, hy, 56], loss='soft_l1', f_scale=.6)
+        centers.append(fit.x[:2]/5-.5)
+        radii.append(fit.x[2]/5)
+    return np.asarray(centers), np.asarray(radii)
+
+def paste_ball(canvas, original, src_center, src_radius, target_center, target_radius):
+    # Work at source resolution to retain original leather texture and seam detail.
+    cx, cy = (src_center+.5)*5
+    tx, ty = (target_center+.5)*5
+    rr = target_radius*5
+    x0, x1 = max(0, int(tx-rr-5)), min(1280, int(tx+rr+6))
+    y0, y1 = max(0, int(ty-rr-5)), min(720, int(ty+rr+6))
+    yy, xx = np.mgrid[y0:y1, x0:x1]
+    scale = src_radius/target_radius
+    mx = (cx+(xx-tx)*scale).astype(np.float32)
+    my = (cy+(yy-ty)*scale).astype(np.float32)
+    sphere = cv2.remap(original, mx, my, cv2.INTER_LINEAR).astype(np.float32)
+    # Edge is a one-high-resolution-pixel antialias, before Lanczos reduction.
+    alpha = np.clip((rr+.65-np.hypot(xx-tx, yy-ty))/1.3, 0, 1)[..., None]
+    canvas[y0:y1, x0:x1] = canvas[y0:y1, x0:x1]*(1-alpha)+sphere*alpha
+
+def main():
+    small = np.load(ROOT / 'source_pixels.npz')['rgb']
+    hi = read_original()
+    centers, radii = track(hi, small)
+    before = np.polyfit(np.arange(13,32), centers[13:32,1], 2)
+    after = np.polyfit(np.arange(33,48), centers[33:48,1], 2)
+    hit = float(min(t for t in np.roots(before-after) if t > 20))
+    ground_y = float(np.polyval(before, hit))
+    g = float(2*before[0])
+    velocity_in = float(np.polyval(np.polyder(before), hit))
+    original_e = float(-np.polyval(np.polyder(after), hit)/velocity_in)
+    new_e = original_e*.1
+    launch = velocity_in*new_e

+    hit_x = float(np.polyval(xpoly, hit))

+    # Continuous floor contact brings rolling resistance into play earlier.
+    rolling_drag = .018
+    contact_radius = float(np.median(radii[48:]))
+    out = small.copy()

+    initx, inity = (centers[0]+.5)*5

+    clean = clean.astype(np.float32)
+

+    sj = 48
+    sx, sy = (centers[sj]+.5)*5
+    shadow = hi[sj].astype(np.float32)-clean

+    sxwindow = np.clip((140-np.abs(X-sx))/30,0,1)

+    shadow *= (radial*sxwindow*sywindow)[...,None]
+
+    trajectory = []
+    for i in range(32,96):
+        dt = i-hit
+        x = hit_x + vx*(1-np.exp(-rolling_drag*dt))/rolling_drag
+        floor = ground_y + .01*(x-hit_x)
+        # Analytic impact flights. At 10% restitution the first apex is < 0.3 px.
+        flight = dt
+        v = launch
+        h = 0.
+        for _ in range(8):
+            duration = 2*v/g
+            if flight < duration:
+                h = v*flight-.5*g*flight**2
+                break
+            flight -= duration
+            v *= new_e
+            if v < .002:
+                h = 0.
+                break
+        y = floor-h
+        trajectory.append([i, float(x), float(y)])
+        canvas = hi[i].astype(np.float32)
+        ox, oy = (centers[i]+.5)*5
+        # Remove only the original moving object and its floor lighting footprint.
+        ball_remove = np.clip((radii[i]*5+9-np.hypot(X-ox,Y-oy))/5,0,1)
+        floor_remove = np.clip((165-np.abs(X-ox))/30,0,1)
+        floor_remove *= np.clip((Y-385)/30,0,1)*np.clip((595-Y)/35,0,1)
+        remove = np.maximum(ball_remove,floor_remove)[...,None]
+        canvas = canvas*(1-remove)+clean*remove

+        # A floor-attached reflection: its very small first bounce changes only contact shade.
+        shift_y = (floor-centers[sj,1])*5
+        moved_shadow = cv2.warpAffine(shadow, np.float32([[1,0,tx-sx],[0,1,shift_y]]),
+                                      (1280,720),flags=cv2.INTER_LINEAR)

+        paste_ball(canvas,hi[i],centers[i],radii[i],np.array([x,y]),contact_radius)

+                            .resize((256,144),Image.Resampling.LANCZOS))
+    artifact_io.save(out, ROOT)
+    (ROOT / '.tmp' / 'motion.json').write_text(json.dumps({
+        'execution_frame':20,'next_impact_frame_1based':hit+1,
+        'original_restitution_estimate':original_e,'new_restitution':new_e,
+        'gravity_px_per_frame_squared':g,'trajectory':trajectory},indent=2))
+    # A conventional video is convenient in addition to the required RGB artifacts.
+    writer = cv2.VideoWriter(str(ROOT/'edited_video.mp4'),cv2.VideoWriter_fourcc(*'mp4v'),24,(256,144))
+    for f in out:
+        writer.write(cv2.cvtColor(f,cv2.COLOR_RGB2BGR))
+    writer.release()
+    print('Saved complete 96-frame edit. Restitution:',original_e,'->',new_e)
+
+if __name__ == '__main__':
+    main()

        cx, cy = estimate, yy.min() + 10.7
        for _ in range(4):
            mask = (r-g > 36) & (r > 90) & ((X-cx)**2 + (Y-cy)**2 < 11.7**2)
            yy, xx = np.where(mask)
            cx, cy = (xx.min()+xx.max())/2, (yy.min()+yy.max())/2
        hx, hy = (cx+.5)*5, (cy+.5)*5
        ix, iy = int(hx), int(hy)
        patch = hi[i, iy-65:iy+66, ix-65:ix+66]
        edge = cv2.Canny(patch, 35, 65)
        ey, ex = np.where(edge > 0)
        ex, ey = ex+ix-65, ey+iy-65
        d = np.hypot(ex-hx, ey-hy)
        keep = (d > 52) & (d < 60)
        ex, ey = ex[keep], ey[keep]
        fit = least_squares(lambda p: np.hypot(ex-p[0], ey-p[1])-p[2],
                            [hx, hy, 56], loss='soft_l1', f_scale=.6)
        centers.append(fit.x[:2]/5-.5)
        radii.append(fit.x[2]/5)
    return np.asarray(centers), np.asarray(radii)

def paste_ball(canvas, original, src_center, src_radius, target_center, target_radius):
    # Work at source resolution to retain original leather texture and seam detail.
    cx, cy = (src_center+.5)*5
    tx, ty = (target_center+.5)*5
    rr = target_radius*5
    x0, x1 = max(0, int(tx-rr-5)), min(1280, int(tx+rr+6))
    y0, y1 = max(0, int(ty-rr-5)), min(720, int(ty+rr+6))
    yy, xx = np.mgrid[y0:y1, x0:x1]
    scale = src_radius/target_radius
    mx = (cx+(xx-tx)*scale).astype(np.float32)
    my = (cy+(yy-ty)*scale).astype(np.float32)
    sphere = cv2.remap(original, mx, my, cv2.INTER_LINEAR).astype(np.float32)
    # Edge is a one-high-resolution-pixel antialias, before Lanczos reduction.
    alpha = np.clip((rr+.65-np.hypot(xx-tx, yy-ty))/1.3, 0, 1)[..., None]
    canvas[y0:y1, x0:x1] = canvas[y0:y1, x0:x1]*(1-alpha)+sphere*alpha

def main():
    small = np.load(ROOT / 'source_pixels.npz')['rgb']
    hi = read_original()
    centers, radii = track(hi, small)
    before = np.polyfit(np.arange(13,32), centers[13:32,1], 2)
    after = np.polyfit(np.arange(33,48), centers[33:48,1], 2)
    hit = float(min(t for t in np.roots(before-after) if t > 20))
    ground_y = float(np.polyval(before, hit))
    g = float(2*before[0])
    velocity_in = float(np.polyval(np.polyder(before), hit))
    original_e = float(-np.polyval(np.polyder(after), hit)/velocity_in)
    new_e = original_e*.1
    launch = velocity_in*new_e
    xpoly = np.polyfit(np.arange(23,32), centers[23:32,0], 1)
    hit_x = float(np.polyval(xpoly, hit))
    vx = float(xpoly[0])
    # Continuous floor contact brings rolling resistance into play earlier.
    rolling_drag = .018
    contact_radius = float(np.median(radii[48:]))
    out = small.copy()
    clean = hi[0].copy()
    Y, X = np.mgrid[:720, :1280]
    initx, inity = (centers[0]+.5)*5
    old_first_ball = np.hypot(X-initx,Y-inity) < 65
    clean[old_first_ball] = hi[-1][old_first_ball]
    clean = clean.astype(np.float32)

    # Source floor reflection and contact shadow from a near-contact frame.
    sj = 48
    sx, sy = (centers[sj]+.5)*5
    shadow = hi[sj].astype(np.float32)-clean
    ball_dist = np.hypot(X-sx,Y-sy)
    radial = np.clip((ball_dist-(radii[sj]*5+.7))/2,0,1)
    sxwindow = np.clip((140-np.abs(X-sx))/30,0,1)
    sywindow = np.clip((Y-sy-27)/15,0,1)*np.clip((sy+175-Y)/40,0,1)
    shadow *= (radial*sxwindow*sywindow)[...,None]

    trajectory = []
    for i in range(32,96):
        dt = i-hit
        x = hit_x + vx*(1-np.exp(-rolling_drag*dt))/rolling_drag
        floor = ground_y + .01*(x-hit_x)
        # Analytic impact flights. At 10% restitution the first apex is < 0.3 px.
        flight = dt
        v = launch
        h = 0.
        for _ in range(8):
            duration = 2*v/g
            if flight < duration:
                h = v*flight-.5*g*flight**2
                break
            flight -= duration
            v *= new_e
            if v < .002:
                h = 0.
                break
        y = floor-h
        trajectory.append([i, float(x), float(y)])
        canvas = hi[i].astype(np.float32)
        ox, oy = (centers[i]+.5)*5
        # Remove only the original moving object and its floor lighting footprint.
        ball_remove = np.clip((radii[i]*5+9-np.hypot(X-ox,Y-oy))/5,0,1)
        floor_remove = np.clip((165-np.abs(X-ox))/30,0,1)
        floor_remove *= np.clip((Y-385)/30,0,1)*np.clip((595-Y)/35,0,1)
        remove = np.maximum(ball_remove,floor_remove)[...,None]
        canvas = canvas*(1-remove)+clean*remove
        tx, ty = (np.array([x,y])+.5)*5
        # A floor-attached reflection: its very small first bounce changes only contact shade.
        shift_y = (floor-centers[sj,1])*5
        moved_shadow = cv2.warpAffine(shadow, np.float32([[1,0,tx-sx],[0,1,shift_y]]),
                                      (1280,720),flags=cv2.INTER_LINEAR)
        canvas += moved_shadow
        paste_ball(canvas,hi[i],centers[i],radii[i],np.array([x,y]),contact_radius)
        out[i] = np.asarray(Image.fromarray(np.clip(np.rint(canvas),0,255).astype(np.uint8))
                            .resize((256,144),Image.Resampling.LANCZOS))
    artifact_io.save(out, ROOT)
    (ROOT / '.tmp' / 'motion.json').write_text(json.dumps({
        'execution_frame':20,'next_impact_frame_1based':hit+1,
        'original_restitution_estimate':original_e,'new_restitution':new_e,
        'gravity_px_per_frame_squared':g,'trajectory':trajectory},indent=2))
    # A conventional video is convenient in addition to the required RGB artifacts.
    writer = cv2.VideoWriter(str(ROOT/'edited_video.mp4'),cv2.VideoWriter_fourcc(*'mp4v'),24,(256,144))
    for f in out:
        writer.write(cv2.cvtColor(f,cv2.COLOR_RGB2BGR))
    writer.release()
    print('Saved complete 96-frame edit. Restitution:',original_e,'->',new_e)

if __name__ == '__main__':
    main()
