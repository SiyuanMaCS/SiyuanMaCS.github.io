+"""Counterfactual friction edit, using source imagery and analytic rigid motion.
+
+All positions and accelerations are in the source's 256-pixel projection.
+The frictional part of the measured uphill deceleration is multiplied by .2;
+the component due to gravity, free-flight gravity and the scenery are unchanged.
+"""
+from pathlib import Path
+import math
+import cv2
+import numpy as np
+from PIL import Image
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+SCALE = 5
+
+
+def read_original():
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    frames = []
+    while True:
+        ok, bgr = cap.read()
+        if not ok:
+            break
+        frames.append(cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB))
+    cap.release()
+    return np.asarray(frames)
+
+
+def matte(image, points):
+    # Hand-traced image geometry, not an external segmentation model.
+    alpha = np.zeros(image.shape[:2], np.uint8)
+    cv2.fillPoly(alpha, [np.array(points, np.int32)], 255)
+    alpha = cv2.GaussianBlur(alpha, (3, 3), .45).astype(np.float32) / 255
+    return image.astype(np.float32) * alpha[..., None], alpha
+
+
+def warp_sprite(rgb, alpha, center, angle, target, target_angle):
+    # Rotation in screen coordinates (clockwise angles are positive).
+    phi = math.radians(target_angle - angle)
+    co, si = math.cos(phi), math.sin(phi)
+    mat = np.array([[co, -si, 0], [si, co, 0]], np.float32)
+    old = np.array(center) * SCALE
+    new = np.array(target) * SCALE
+    mat[:, 2] = new - mat[:, :2] @ old
+    size = (1280, 720)
+    color = cv2.warpAffine(rgb, mat, size, flags=cv2.INTER_LINEAR)
+    a = cv2.warpAffine(alpha, mat, size, flags=cv2.INTER_LINEAR)
+    return color, a
+
+
+def generate():
+    factual = np.load(ROOT / 'source_pixels.npz')['rgb']
+    source = read_original()
+    # The camera and all fixtures are static. The late source reveals the ramp;
+    # the early source reveals the ground covered by the stopped factual car.
+    background = source[-1].copy()
+    background[470:670, :246] = source[0, 470:670, :246]
+    background = background.astype(np.float32)
+
+    ramp_outline = [
+        (763,538),(780,537),(781,522),(788,512),(803,508),(825,510),
+        (835,514),(843,524),(852,525),(871,522),(870,519),(865,515),
+        (889,518),(918,529),(932,535),(938,538),(943,540),(949,546),
+        (950,552),(944,558),(946,562),(943,584),(938,590),(930,594),
+        (919,595),(909,593),(897,590),(876,580),(839,566),(825,562),
+        (814,562),(802,556),(793,551),(779,549),(765,543)
+    ]
+    level_outline = [
+        (180,364),(185,352),(195,345),(218,343),(234,340),(244,336),
+        (263,330),(283,327),(302,331),(320,334),(338,330),(350,327),
+        (352,330),(345,336),(342,340),(352,345),(356,365),(351,373),
+        (335,380),(325,381),(317,384),(306,385),(296,381),(288,374),
+        (271,377),(238,377),(222,377),(216,383),(204,383),(194,379),
+        (187,373),(180,370)
+    ]
+    slope_rgb, slope_alpha = matte(source[0], ramp_outline)
+    level_rgb, level_alpha = matte(source[13], level_outline)
+    center0 = (173.4, 109.46)
+    level_center = (55.14, 71.4)
+    slope_angle = 20.0
+    level_angle = 0.5
+
+    # Observed early source trajectory: x ~= 173.4 - 358*t + 275*t*t.
+    # Separate gravitational and contact-friction forces before the edit.
+    slope = .363
+    gravity = 1250.0
+    original_deceleration = 550.0
+    gravity_along_x = gravity * slope / (1 + slope*slope)
+    original_friction = original_deceleration - gravity_along_x
+    acceleration = gravity_along_x + .2 * original_friction
+    speed0 = -358.0
+    edge_center_x = 62.0
+    release = (-speed0 - math.sqrt(speed0**2 - 2*acceleration*(center0[0]-edge_center_x))) / acceleration
+    launch_speed = speed0 + acceleration * release
+    edge_y = center0[1] + slope * (edge_center_x - center0[0])
+    launch_vy = -65.0
+    # Contact at the front tires on the pavement, with a brief pitch recovery.
+    ground_center_y = 118.8
+    fall_duration = (-launch_vy + math.sqrt(launch_vy**2 + 2*gravity*(ground_center_y-edge_y))) / gravity
+    touchdown = release + fall_duration
+    land_x = edge_center_x + launch_speed * fall_duration
+
+    yy, xx = np.indices((720, 1280), dtype=np.float32)
+    out = []
+    for i in range(96):
+        t = i / 24
+        if i == 0:
+            out.append(factual[0].copy())
+            continue
+        frame = background.copy()
+        if t < release:
+            cx = center0[0] + speed0*t + .5*acceleration*t*t
+            cy = center0[1] + slope*(cx-center0[0])
+            u = np.clip((83-cx)/(83-edge_center_x), 0, 1)
+            angle = 20 - 9*u*u
+            mix = .38*u*u
+            air_height = 0
+        elif t < touchdown:
+            dt = t-release
+            cx = edge_center_x + launch_speed*dt
+            cy = edge_y + launch_vy*dt + .5*gravity*dt*dt
+            u = dt/fall_duration
+            angle = 11 - 50*(u*u*(3-2*u))
+            mix = .38 + .62*min(1, dt/.11)
+            air_height = max(0, ground_center_y-cy)
+        else:
+            dt = t-touchdown
+            # Lower contact friction leaves almost all horizontal momentum.
+            ground_deceleration = .2*330
+            v = max(0, -launch_speed-ground_deceleration*dt)
+            travel_time = min(dt, -launch_speed/ground_deceleration)
+            cx = land_x + launch_speed*travel_time + .5*ground_deceleration*travel_time**2
+            cy = ground_center_y - .9*math.exp(-dt*22)*math.sin(dt*55)
+            angle = -39*max(0, 1-dt/.09)**2
+            mix = 1
+            air_height = 0
+
+        # Avoid unnecessary raster work after the object has left the camera.
+        if cx > -25:
+            if t >= release:
+                # Project the moving shadow onto the undisturbed pavement.
+                shadow_x = (cx+5)*SCALE
+                shadow_y = 126.5*SCALE
+                rx = (17 + .055*air_height)*SCALE
+                ry = (2.2 + .02*air_height)*SCALE
+                opacity = .29*math.exp(-air_height/60)
+                gauss = np.exp(-2*((xx-shadow_x)/rx)**2 - 2*((yy-shadow_y)/ry)**2)
+                frame *= 1-opacity*gauss[...,None]
+
+            rgb0, a0 = warp_sprite(slope_rgb, slope_alpha, center0,
+                                  slope_angle, (cx,cy), angle)
+            rgb1, a1 = warp_sprite(level_rgb, level_alpha, level_center,
+                                  level_angle, (cx,cy), angle)
+            rgb = (1-mix)*rgb0 + mix*rgb1
+            alpha = (1-mix)*a0 + mix*a1
+            frame = frame*(1-alpha[...,None]) + rgb
+        image = Image.fromarray(np.clip(frame,0,255).astype(np.uint8))
+        out.append(np.asarray(image.resize((256,144), Image.Resampling.LANCZOS)))
+    frames = np.asarray(out, dtype=np.uint8)
+    artifact_io.save(frames, ROOT)
+    print(f'Saved {frames.shape}: friction ratio=.2, release={release:.3f}s, touchdown={touchdown:.3f}s')
+
+
+if __name__ == '__main__':
+    generate()
