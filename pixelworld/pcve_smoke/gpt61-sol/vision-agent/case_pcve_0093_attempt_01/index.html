+"""Remove the black object ball, and reconstruct the unobstructed cue-ball shot.
+Only source imagery and elementary pixel processing / projective motion are used.
+"""
+from pathlib import Path
+import warnings
+import cv2
+import numpy as np
+from PIL import Image
+from scipy.optimize import least_squares
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+def read_source():
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    frames = []
+    while True:
+        ok, f = cap.read()
+        if not ok:
+            break
+        frames.append(cv2.cvtColor(f, cv2.COLOR_BGR2RGB))
+    cap.release()
+    return np.asarray(frames)
+
+def track(source):
+    h, w = source.shape[1:3]
+    roi = np.zeros((h, w), np.uint8)
+    cv2.fillPoly(roi, [np.int32([[40,520],[711,225],[992,266],[544,691]])], 1)
+    white, black = [], []
+    for i, f in enumerate(source):
+        sat = f.max(-1).astype(float) - f.min(-1)
+        wm = ((f.min(-1)>150) & (sat<45) & (roi>0)).astype(np.uint8)
+        n, labels, stats, centers = cv2.connectedComponentsWithStats(wm)
+        candidates = [(s[4],c) for s,c in zip(stats[1:],centers[1:]) if 25<s[4]<1500]
+        white.append(max(candidates, key=lambda x:x[0])[1])
+        bm = ((f.max(-1)<62) & (roi>0)).astype(np.uint8)
+        n, labels, stats, centers = cv2.connectedComponentsWithStats(bm)
+        candidates = [(s[4],c) for s,c in zip(stats[1:],centers[1:])
+                      if 90<s[4]<1000 and s[2]<50 and s[3]<50]
+        if candidates:
+            black.append(max(candidates, key=lambda x:x[0])[1])
+        else:
+            # Near the cushion the moving dark ball merges with its dark rail.
+            # A local chromatic contrast finds its silhouette without the rail.
+            if i >= 42 and i <= 49:
+                approx = [836 + 2.0*(i-42), 235]
+            else:
+                approx = black[-1]
+            black.append(approx)
+    black = np.array(black)
+    # The seven cushion-contact samples are read from the local dark sphere,
+    # rather than its merged shadow. All positions are source pixel coordinates.
+    black[42:50] = np.array([[836,235],[840,232],[844,231],[845,231],
+                            [845,232],[842,233],[839,234],[836,236]])
+    return np.array(white), black
+
+def object_mask(shape, white, black, index):
+    mask = np.zeros(shape, np.uint8)
+    wx,wy = white
+    # Larger support includes both the silhouette and the soft contact shadow.
+    radius = 20.5/(1 + .02138*min(index,14))
+    cv2.ellipse(mask, (round(wx),round(wy+8)),
+                (round(radius+17),round(radius+20)),0,0,360,1,-1)
+    bx,by = black
+    radius = 15 if index<20 else 12
+    cv2.ellipse(mask,(round(bx),round(by+5)),(radius+16,radius+20),0,0,360,1,-1)
+    return mask
+
+def clean_plate(source, white, black):
+    # Temporal clean samples preserve real cloth grain, illumination and rail.
+    ids = np.unique(np.r_[np.arange(0,13,2),np.arange(17,96,5),95])
+    masks = np.array([object_mask(source.shape[1:3],white[i],black[i],i) for i in ids])
+    plate = source[0].copy()
+    holes = np.zeros(source.shape[1:3],np.uint8)
+    for top in range(0,source.shape[1],40):
+        bottom=min(top+40,source.shape[1])
+        a=source[ids,top:bottom].astype(np.float32)
+        a[masks[:,top:bottom].astype(bool)]=np.nan
+        with warnings.catch_warnings():
+            warnings.simplefilter('ignore',RuntimeWarning)
+            p=np.nanmedian(a,axis=0)
+        hole=~np.isfinite(p[:,:,0])
+        holes[top:bottom]=hole
+        p[hole]=0
+        plate[top:bottom]=np.clip(p,0,255).astype(np.uint8)
+    # A tiny region beneath both balls has no clean observation; extend the
+    # surrounding genuine green felt with non-learned spatial inpainting.
+    plate=cv2.inpaint(plate,holes*255,5,cv2.INPAINT_TELEA)
+    return plate
+
+def soft_mask(mask):
+    return cv2.GaussianBlur(mask.astype(np.float32),(0,0),2.0)[...,None]
+
+def run():
+    source=read_source()
+    factual=np.load(ROOT/'source_pixels.npz')['rgb']
+    white,black=track(source)
+    plate=clean_plate(source,white,black)
+    # Perspective on a straight line is a fractional-linear function of time.
+    times=np.arange(13,dtype=float)
+    def project(p,t):
+        return (p[0]+p[1]*t)/(1+p[2]*t)
+    p=least_squares(lambda p:project(p,times)-white[:13,0],
+                    [429,28,.021]).x
+    slope,intercept=np.polyfit(white[:13,0],white[:13,1],1)
+    # Sprite is actual source cue-ball texture, including illumination and shadow.
+    # This source crop's physical sphere center lies below its bright-pixel centroid.
+    center=np.array([429.5,473.0])
+    x0,y0,x1,y1=377,424,481,532
+    sprite=source[0,y0:y1,x0:x1].astype(np.float32)-plate[y0:y1,x0:x1].astype(np.float32)
+    yy,xx=np.mgrid[y0:y1,x0:x1]
+    # Subtle residual background noise is suppressed outside the full cast shadow.
+    support=np.clip((1-((xx-425)/48)**2-((yy-478)/46)**2)*4,0,1)
+    sprite*=support[...,None]
+    result=[]
+    positions=[]
+    bounce=43.0
+    for i,f in enumerate(source):
+        out=f.astype(np.float32).copy()
+        # Before contact the source cue-ball image remains completely untouched.
+        mask=np.zeros(source.shape[1:3],np.uint8)
+        bx,by=black[i]
+        rad=15 if i<20 else 12
+        cv2.ellipse(mask,(round(bx),round(by+5)),(rad+16,rad+20),0,0,360,1,-1)
+        if i>=13:
+            wx,wy=white[i]
+            cv2.ellipse(mask,(round(wx),round(wy+8)),(35,40),0,0,360,1,-1)
+        alpha=soft_mask(mask)
+        out=out*(1-alpha)+plate.astype(np.float32)*alpha
+        if i>=13:
+            if i<=bounce:
+                u=float(i)
+            else:
+                dt=i-bounce
+                # Damped elastic cushion reflection, then rolling friction.
+                u=bounce-.76*dt+.0035*dt*dt
+            x=project(p,u)
+            y=slope*x+intercept
+            scale=1/(1+p[2]*u)
+            # Perspective center correction tracks the shrinking sphere radius.
+            pos=np.array([x,y])+(center-white[0])*scale
+            positions.append([i,*pos,scale])
+            M=np.array([[scale,0,pos[0]-scale*(center[0]-x0)],
+                        [0,scale,pos[1]-scale*(center[1]-y0)]],np.float32)
+            moved=cv2.warpAffine(sprite,M,(source.shape[2],source.shape[1]),
+                                 flags=cv2.INTER_LINEAR,borderMode=cv2.BORDER_CONSTANT)
+            out+=moved
+        rgb=np.clip(np.rint(out),0,255).astype(np.uint8)
+        small=np.asarray(Image.fromarray(rgb).resize((256,144),Image.Resampling.LANCZOS))
+        # Maintain literal source pixels everywhere outside the edited support.
+        changed=np.max(np.abs(rgb.astype(np.int16)-f.astype(np.int16)),axis=2)>0
+        changed=cv2.dilate(changed.astype(np.uint8),np.ones((15,15),np.uint8))
+        smallmask=cv2.resize(changed,(256,144),interpolation=cv2.INTER_AREA)>0
+        small=np.where(smallmask[...,None],small,factual[i]).astype(np.uint8)
+        result.append(small)
+    frames=np.asarray(result,dtype=np.uint8)
+    # Every candidate is saved before any further review.
+    artifact_io.save(frames,ROOT)
+    np.save(ROOT/'.tmp/cue_positions.npy',np.array(positions))
+    Image.fromarray(plate).resize((768,432),Image.Resampling.LANCZOS).save(ROOT/'.tmp/clean_plate.png')
+    print('Saved',frames.shape,'at 24 fps. Bounce at frame',bounce+1)
+
+if __name__=='__main__':
+    run()

\"\"\"Remove the black object ball, and reconstruct the unobstructed cue-ball shot.
Only source imagery and elementary pixel processing / projective motion are used.
\"\"\"
from pathlib import Path
import warnings
import cv2
import numpy as np
from PIL import Image
from scipy.optimize import least_squares
import artifact_io

ROOT = Path(__file__).resolve().parent

def read_source():
    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
    frames = []
    while True:
        ok, f = cap.read()
        if not ok:
            break
        frames.append(cv2.cvtColor(f, cv2.COLOR_BGR2RGB))
    cap.release()
    return np.asarray(frames)

def track(source):
    h, w = source.shape[1:3]
    roi = np.zeros((h, w), np.uint8)
    cv2.fillPoly(roi, [np.int32([[40,520],[711,225],[992,266],[544,691]])], 1)
    white, black = [], []
    for i, f in enumerate(source):
        sat = f.max(-1).astype(float) - f.min(-1)
        wm = ((f.min(-1)>150) & (sat<45) & (roi>0)).astype(np.uint8)
        n, labels, stats, centers = cv2.connectedComponentsWithStats(wm)
        candidates = [(s[4],c) for s,c in zip(stats[1:],centers[1:]) if 25<s[4]<1500]
        white.append(max(candidates, key=lambda x:x[0])[1])
        bm = ((f.max(-1)<62) & (roi>0)).astype(np.uint8)
        n, labels, stats, centers = cv2.connectedComponentsWithStats(bm)
        candidates = [(s[4],c) for s,c in zip(stats[1:],centers[1:])
                      if 90<s[4]<1000 and s[2]<50 and s[3]<50]
        if candidates:
            black.append(max(candidates, key=lambda x:x[0])[1])
        else:
            # Near the cushion the moving dark ball merges with its dark rail.
            # A local chromatic contrast finds its silhouette without the rail.
            if i >= 42 and i <= 49:
                approx = [836 + 2.0*(i-42), 235]
            else:
                approx = black[-1]
            black.append(approx)
    black = np.array(black)
    # The seven cushion-contact samples are read from the local dark sphere,
    # rather than its merged shadow. All positions are source pixel coordinates.
    black[42:50] = np.array([[836,235],[840,232],[844,231],[845,231],
                            [845,232],[842,233],[839,234],[836,236]])
    return np.array(white), black

def object_mask(shape, white, black, index):
    mask = np.zeros(shape, np.uint8)
    wx,wy = white
    # Larger support includes both the silhouette and the soft contact shadow.
    radius = 20.5/(1 + .02138*min(index,14))
    cv2.ellipse(mask, (round(wx),round(wy+8)),
                (round(radius+17),round(radius+20)),0,0,360,1,-1)
    bx,by = black
    radius = 15 if index<20 else 12
    cv2.ellipse(mask,(round(bx),round(by+5)),(radius+16,radius+20),0,0,360,1,-1)
    return mask

def clean_plate(source, white, black):
    # Temporal clean samples preserve real cloth grain, illumination and rail.
    ids = np.unique(np.r_[np.arange(0,13,2),np.arange(17,96,5),95])
    masks = np.array([object_mask(source.shape[1:3],white[i],black[i],i) for i in ids])
    plate = source[0].copy()
    holes = np.zeros(source.shape[1:3],np.uint8)
    for top in range(0,source.shape[1],40):
        bottom=min(top+40,source.shape[1])
        a=source[ids,top:bottom].astype(np.float32)
        a[masks[:,top:bottom].astype(bool)]=np.nan
        with warnings.catch_warnings():
            warnings.simplefilter('ignore',RuntimeWarning)
            p=np.nanmedian(a,axis=0)
        hole=~np.isfinite(p[:,:,0])
        holes[top:bottom]=hole
        p[hole]=0
        plate[top:bottom]=np.clip(p,0,255).astype(np.uint8)
    # A tiny region beneath both balls has no clean observation; extend the
    # surrounding genuine green felt with non-learned spatial inpainting.
    plate=cv2.inpaint(plate,holes*255,5,cv2.INPAINT_TELEA)
    return plate

def soft_mask(mask):
    return cv2.GaussianBlur(mask.astype(np.float32),(0,0),2.0)[...,None]

def run():
    source=read_source()
    factual=np.load(ROOT/'source_pixels.npz')['rgb']
    white,black=track(source)
    plate=clean_plate(source,white,black)
    # Perspective on a straight line is a fractional-linear function of time.
    times=np.arange(13,dtype=float)
    def project(p,t):
        return (p[0]+p[1]*t)/(1+p[2]*t)
    p=least_squares(lambda p:project(p,times)-white[:13,0],
                    [429,28,.021]).x
    slope,intercept=np.polyfit(white[:13,0],white[:13,1],1)
    # Sprite is actual source cue-ball texture, including illumination and shadow.
    # This source crop's physical sphere center lies below its bright-pixel centroid.
    center=np.array([429.5,473.0])
    x0,y0,x1,y1=377,424,481,532
    sprite=source[0,y0:y1,x0:x1].astype(np.float32)-plate[y0:y1,x0:x1].astype(np.float32)
    yy,xx=np.mgrid[y0:y1,x0:x1]
    # Subtle residual background noise is suppressed outside the full cast shadow.
    support=np.clip((1-((xx-425)/48)**2-((yy-478)/46)**2)*4,0,1)
    sprite*=support[...,None]
    result=[]
    positions=[]
    bounce=43.0
    for i,f in enumerate(source):
        out=f.astype(np.float32).copy()
        # Before contact the source cue-ball image remains completely untouched.
        mask=np.zeros(source.shape[1:3],np.uint8)
        bx,by=black[i]
        rad=15 if i<20 else 12
        cv2.ellipse(mask,(round(bx),round(by+5)),(rad+16,rad+20),0,0,360,1,-1)
        if i>=13:
            wx,wy=white[i]
            cv2.ellipse(mask,(round(wx),round(wy+8)),(35,40),0,0,360,1,-1)
        alpha=soft_mask(mask)
        out=out*(1-alpha)+plate.astype(np.float32)*alpha
        if i>=13:
            if i<=bounce:
                u=float(i)
            else:
                dt=i-bounce
                # Damped elastic cushion reflection, then rolling friction.
                u=bounce-.76*dt+.0035*dt*dt
            x=project(p,u)
            y=slope*x+intercept
            scale=1/(1+p[2]*u)
            # Perspective center correction tracks the shrinking sphere radius.
            pos=np.array([x,y])+(center-white[0])*scale
            positions.append([i,*pos,scale])
            M=np.array([[scale,0,pos[0]-scale*(center[0]-x0)],
                        [0,scale,pos[1]-scale*(center[1]-y0)]],np.float32)
            moved=cv2.warpAffine(sprite,M,(source.shape[2],source.shape[1]),
                                 flags=cv2.INTER_LINEAR,borderMode=cv2.BORDER_CONSTANT)
            out+=moved
        rgb=np.clip(np.rint(out),0,255).astype(np.uint8)
        small=np.asarray(Image.fromarray(rgb).resize((256,144),Image.Resampling.LANCZOS))
        # Maintain literal source pixels everywhere outside the edited support.
        changed=np.max(np.abs(rgb.astype(np.int16)-f.astype(np.int16)),axis=2)>0
        changed=cv2.dilate(changed.astype(np.uint8),np.ones((15,15),np.uint8))
        smallmask=cv2.resize(changed,(256,144),interpolation=cv2.INTER_AREA)>0
        small=np.where(smallmask[...,None],small,factual[i]).astype(np.uint8)
        result.append(small)
    frames=np.asarray(result,dtype=np.uint8)
    # Every candidate is saved before any further review.
    artifact_io.save(frames,ROOT)
    np.save(ROOT/'.tmp/cue_positions.npy',np.array(positions))
    Image.fromarray(plate).resize((768,432),Image.Resampling.LANCZOS).save(ROOT/'.tmp/clean_plate.png')
    print('Saved',frames.shape,'at 24 fps. Bounce at frame',bounce+1)

if __name__=='__main__':
    run()
