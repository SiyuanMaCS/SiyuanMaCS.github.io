+def project(vertices):

+    depth=D+np.cos(ALPHA)*Y-np.sin(ALPHA)*Z
+    return np.stack((CX+400*X/depth,CY+400*(-np.sin(ALPHA)*Y-np.cos(ALPHA)*Z)/depth),axis=1)

+def world(o,angle):
+    x,y=(XA,YA) if o==0 else (XB,YB)
+    t=np.deg2rad(angle)
+    return np.array([[x+u*np.cos(t)+z*np.sin(t),y+dep,-u*np.sin(t)+z*np.cos(t)]

+def poly_mask(points,scale=1):

+    cv2.fillConvexPoly(m,np.round(np.array(points)*scale).astype(np.int32),255)
+    return m.astype(float)/255

+def harmonic_fill(a,mask):
+    """Interpolation of a smooth tabletop through permanently covered base pixels."""
+    points=np.argwhere(mask);n=len(points)
+    lookup=np.full(mask.shape,-1,int);lookup[mask]=np.arange(n)
+    mat=lil_matrix((n,n));b=np.zeros((n,3))
+    for i,(y,x) in enumerate(points):
+        count=0
+        for yy,xx in [(y-1,x),(y+1,x),(y,x-1),(y,x+1)]:
+            if 0<=yy<H and 0<=xx<W:
+                count+=1
+                if mask[yy,xx]:mat[i,lookup[yy,xx]]=-1
+                else:b[i]+=a[yy,xx]
+        mat[i,i]=count
+    out=a.astype(float).copy()
+    out[mask]=spsolve(mat.tocsr(),b)
+    return out

+def overlap(a,b):
+    for poly in (a,b):
+        edges=np.roll(poly,-1,axis=0)-poly
+        for edge in edges:
+            axis=np.array([-edge[1],edge[0]])
+            aa=a@axis;bb=b@axis
+            if aa.max()<=bb.min()+1e-6 or bb.max()<=aa.min()+1e-6:return False
+    return True

+def limit_angle(b):

+def pose(t):

from pathlib import Path
import numpy as np
import cv2
from scipy.ndimage import gaussian_filter
from scipy.sparse import lil_matrix
from scipy.sparse.linalg import spsolve
from artifact_io import save

ROOT=Path(__file__).resolve().parent
S=4
W,H=256,144
P=np.array([6.56768244,138.11079155,98.52427041,13.52870632,.14827956,.76344532,-1.13034291,-.13818723,-.40115976,-.21266882,10.93936287,59.13693909,29.9997072,78.37052706,76.33536513])
D,CX,CY,ALPHA,T,WIDTH,XA,YA,XB,YB=P[:10]
ALPHA=np.deg2rad(ALPHA)

def project(vertices):
    X,Y,Z=np.asarray(vertices).T
    depth=D+np.cos(ALPHA)*Y-np.sin(ALPHA)*Z
    return np.stack((CX+400*X/depth,CY+400*(-np.sin(ALPHA)*Y-np.cos(ALPHA)*Z)/depth),axis=1)

def world(o,angle):
    x,y=(XA,YA) if o==0 else (XB,YB)
    t=np.deg2rad(angle)
    return np.array([[x+u*np.cos(t)+z*np.sin(t),y+dep,-u*np.sin(t)+z*np.cos(t)]
        for dep in [0,WIDTH] for u,z in [(-T,0),(0,0),(0,1),(-T,1)]])

def poly_mask(points,scale=1):
    m=np.zeros((H*scale,W*scale),np.uint8)
    cv2.fillConvexPoly(m,np.round(np.array(points)*scale).astype(np.int32),255)
    return m.astype(float)/255

def harmonic_fill(a,mask):
    \"\"\"Interpolation of a smooth tabletop through permanently covered base pixels.\"\"\"
    points=np.argwhere(mask);n=len(points)
    lookup=np.full(mask.shape,-1,int);lookup[mask]=np.arange(n)
    mat=lil_matrix((n,n));b=np.zeros((n,3))
    for i,(y,x) in enumerate(points):
        count=0
        for yy,xx in [(y-1,x),(y+1,x),(y,x-1),(y,x+1)]:
            if 0<=yy<H and 0<=xx<W:
                count+=1
                if mask[yy,xx]:mat[i,lookup[yy,xx]]=-1
                else:b[i]+=a[yy,xx]
        mat[i,i]=count
    out=a.astype(float).copy()
    out[mask]=spsolve(mat.tocsr(),b)
    return out

def make_plate(frames):
    data=frames.astype(float)
    green=(data[:,:,:,1]-data[:,:,:,0]>16)&(data[:,:,:,1]-data[:,:,:,2]>25)
    # Select revealed tabletop, never a pip or a gray domino face.
    score=np.where(green,data[:,:,:,1],-1000)
    best=np.argmax(score,axis=0)
    mosaic=data[best,np.arange(H)[:,None],np.arange(W)[None,:]]
    missing=~green.any(axis=0)
    missing[:32]=False
    # The bases remain occluded in the factual video; remove their shadows too.
    yy,xx=np.mgrid[:H,:W]
    for cx,cy,rx,ry in [(69,107,14,15),(110,108,15,15),(146,106,15,15)]:
        missing|=((xx-cx)/rx)**2+((yy-cy)/ry)**2<1
    missing &= (xx>47)&(xx<171)&(yy>32)
    mosaic=harmonic_fill(mosaic,missing)
    # Suppress codec/selection differences on this essentially untextured table.
    mosaic=gaussian_filter(mosaic,(1.35,1.35,0))
    plate=data[0].copy()
    area=((xx>47)&(xx<171)&(yy>32)).astype(float)
    area=cv2.GaussianBlur(area,(0,0),2.3)
    plate=plate*(1-area[:,:,None])+mosaic*area[:,:,None]
    # Restore everything outside the first three original objects.
    plate[:31]=data[0,:31]
    plate[:,176:]=data[0,:,176:]
    return plate,mosaic

def rectify(image,quad):
    size=(96,192)
    target=np.array([[0,191],[95,191],[95,0],[0,0]],np.float32)
    mat=cv2.getPerspectiveTransform(np.asarray(quad,np.float32),target)
    tex=cv2.warpPerspective(image.astype(np.float32),mat,size,flags=cv2.INTER_LINEAR)
    return tex

def textures(frames):
    ans=[]
    for o in range(2):
        row=[]
        for k,indices in enumerate(FACES):
            if o==0:
                tex=rectify(frames[31],ANCHORS[o][indices])
            elif k==2:
                # Standing undotted narrow face is fully unobstructed.
                q=np.array([[102.6,101.8],[112.4,101.8],[111.5,39.1],[101.4,39.1]])
                tex=rectify(frames[0],q)
            else:
                # Newly exposed reverse face: sample the factual fallen reverse.
                if k==0:q=np.array([[112.0,93.0],[171.2,79.0],[168.0,69.2],[119.7,80.0]])
                else:q=np.array([[171.2,79.0],[174.4,87.0],[172.0,78.3],[168.0,69.2]])
                tex=rectify(frames[55],q)
                # Keep a clean diffuse texture rather than edge contamination.
                color=np.median(tex[30:160,15:80],axis=(0,1))
                yy,xx=np.mgrid[:192,:96]
                tex=np.broadcast_to(color,(192,96,3)).copy()
                tex+=(yy[:,:,None]/191-.5)*1.8
                if k==0:tex*=.94
            # Avoid dragging antialiased table pixels into the body face.
            tex=gaussian_filter(tex,(.8,.8,0))
            tex=np.clip(tex,20,113)
            row.append(tex)
        ans.append(row)
    return ans

def overlap(a,b):
    for poly in (a,b):
        edges=np.roll(poly,-1,axis=0)-poly
        for edge in edges:
            axis=np.array([-edge[1],edge[0]])
            aa=a@axis;bb=b@axis
            if aa.max()<=bb.min()+1e-6 or bb.max()<=aa.min()+1e-6:return False
    return True

def limit_angle(b):
    pb=world(1,b)[:4][:,[0,2]]
    lo,hi=0.,90.
    for _ in range(30):
        mid=(lo+hi)*.5
        if overlap(world(0,mid)[:4][:,[0,2]],pb):hi=mid
        else:lo=mid
    return lo

def pose(t):
    dt=t-31
    thb=30+2.05*dt+.215*dt*dt
    if thb>=90:
        impact=(-2.05+np.sqrt(2.05**2+4*.215*60))/(2*.215)
        tau=dt-impact
        thb=90-1.2*np.exp(-tau/2.0)*max(np.sin(tau*1.4),0)
        if tau>9:thb=90.
    tha=min(59.14+2.9*dt+.16*dt*dt,limit_angle(thb))
    return tha,thb

def shadow(plate,angles):
    yy,xx=np.mgrid[:H*S,:W*S]
    opacity=np.zeros((H*S,W*S),float)
    contact=np.zeros_like(opacity)
    for o,angle in enumerate(angles):
        v=world(o,angle)
        cast=v.copy();cast[:,0]-=.32*v[:,2];cast[:,1]-=1.05*v[:,2];cast[:,2]=0
        hull=cv2.convexHull(project(cast).astype(np.float32))[:,0,:]
        shade=poly_mask(hull,S)
        opacity+=cv2.GaussianBlur(shade,(0,0),2.1*S)*.24
        # Ambient contact extends under the body, not into the empty gap.
        near=v.copy();near[:,2]=0
        foot=cv2.convexHull(project(near).astype(np.float32))[:,0,:]
        flat=poly_mask(foot,S)
        contact+=cv2.GaussianBlur(flat,(0,0),1.15*S)*.21
    op=np.clip(opacity+contact,0,.47)
    high=cv2.resize(plate.astype(np.float32),(W*S,H*S),interpolation=cv2.INTER_CUBIC)
    return high*(1-op[:,:,None])

def render(plate,tex,t):
    angles=pose(t)
    canvas=shadow(plate,angles)
    # The first domino is in front and finally rests on the second.
    for o in (1,0):
        v=project(world(o,angles[o]))
        observed=project(world(o,59.13693909 if o==0 else 29.9997072))
        correction=ANCHORS[o]-observed
        blend=max(0,1-(t-31)/11)
        v+=correction*blend
        for k,indices in enumerate(FACES):
            quad=v[indices]*S
            texture=tex[o][k].copy()
            # Surface illumination changes as a face turns toward overhead light.
            if k==0 and o==0:texture*=1+.07*min((t-31)/12,1)
            target=np.array([[0,191],[95,191],[95,0],[0,0]],np.float32)
            mat=cv2.getPerspectiveTransform(target,quad.astype(np.float32))
            warped=cv2.warpPerspective(texture.astype(np.float32),mat,(W*S,H*S),flags=cv2.INTER_LINEAR,borderMode=cv2.BORDER_REPLICATE)
            mask=poly_mask(v[indices],S)
            canvas=canvas*(1-mask[:,:,None])+warped*mask[:,:,None]
    # Subpixel edges and original depth of field.
    canvas=cv2.GaussianBlur(canvas,(0,0),.36*S)
    return cv2.resize(canvas,(W,H),interpolation=cv2.INTER_AREA)

def generate():
    source=np.load(ROOT/'source_pixels.npz')['rgb']
    plate,mosaic=make_plate(source)
    tex=textures(source)
    yy,xx=np.mgrid[:H,:W]
    # Local patch removes the third body and its local cast/contact shadow.
    patch=(((xx>=139)&(xx<=155)&(yy>=32)&(yy<=107)) | (((xx-142)/17)**2+((yy-108)/22)**2<1)).astype(float)
    patch=cv2.GaussianBlur(patch,(0,0),1.7)
    result=[]
    for t in range(96):
        if t<31:
            frame=source[t].astype(float)*(1-patch[:,:,None])+mosaic*patch[:,:,None]
            # Keep the still-factual moving first two dominoes in front of the patch.
            rgb=source[t]
            fg=((rgb[:,:,1]<69)&(rgb[:,:,0]<70)&(yy>=35)&(yy<=103)&(xx>=48)&(xx<142)).astype(np.uint8)
            count,labels,stats,centers=cv2.connectedComponentsWithStats(fg)
            keep=np.zeros((H,W),np.uint8)
            for label in range(1,count):
                if stats[label,cv2.CC_STAT_AREA]>35 and centers[label,0]<139:
                    comp=(labels==label).astype(np.uint8)
                    co,_=cv2.findContours(comp,cv2.RETR_EXTERNAL,cv2.CHAIN_APPROX_SIMPLE)
                    cv2.fillConvexPoly(keep,cv2.convexHull(max(co,key=cv2.contourArea)),255)
            keep=cv2.dilate(keep,np.ones((3,3),np.uint8))
            keep=keep.astype(float)/255
            frame=frame*(1-keep[:,:,None])+source[t]*keep[:,:,None]
        else:frame=render(plate,tex,t)
        result.append(np.clip(np.round(frame),0,255).astype(np.uint8))
    result=np.stack(result)
    save(result,ROOT)
    # Convenience encoded video alongside the required RGB artifacts.
    writer=cv2.VideoWriter(str(ROOT/'edited_video.mp4'),cv2.VideoWriter_fourcc(*'mp4v'),24,(W,H))
    for frame in result:writer.write(cv2.cvtColor(frame,cv2.COLOR_RGB2BGR))
    writer.release()
    print('Saved',result.shape,'at 24 fps')

if __name__=='__main__':generate()
