+def ball_circles(source):
+    tracks=[]
+    for image in source:
+        a=image.astype(float)
+        mask=(a[:,:,0]>1.75*a[:,:,1])&(a[:,:,0]-a[:,:,1]>60)&(a[:,:,0]>1.8*a[:,:,2])

+        cx,cy=fit[:2]

+        tracks.append((cx,cy,radius))

+def free_trajectory(tracks):
+    t=np.arange(12,dtype=float)
+    observed=tracks[:12]
+    # Projected straight world-space motion is a rational curve under perspective.
+    def residual(p):
+        k,a,b,c,d,r,e=p
+        den=1-k*t
+        return np.r_[(a+b*t)/den-observed[:,0],(c+d*t)/den-observed[:,1],
+                     .75*((r+e*t)/den-observed[:,2])]
+    p=least_squares(residual,[.001,15.4,7,56.2,1.4,15,.15]).x
+    k,a,b,c,d,r,e=p
+    def at(i):
+        if i<12:
+            return tracks[i]
+        den=1-k*i
+        return np.array([(a+b*i)/den,(c+d*i)/den,(r+e*i)/den])
+    return at
+
+def sphere_texture(image, background, circle):
+    # Recover a soft RGB sphere sprite, excluding its floor shadow.
+    cx,cy,r=circle
+    radius=r-.25
+    h,w=image.shape[:2]
+    yy,xx=np.mgrid[:h,:w]
+    distance=np.sqrt((xx-cx)**2+(yy-cy)**2)
+    alpha=np.clip((radius+.65-distance)/1.3,0,1)
+    # Recover surface color from the antialiased edge rather than copy floor pixels.
+    surface=(image.astype(float)-background*(1-alpha[:,:,None]))/np.maximum(alpha[:,:,None],.22)
+    # A dark contact edge needs a slightly soft transition, as in the factual render.
+    return np.clip(surface,0,255).astype('float32'),alpha.astype('float32')
+
+def composite_ball(bg, texture, src_circle, target_circle):
+    image=bg.astype('float32').copy()
+    cx,cy,r=target_circle
+    if cx-r>bg.shape[1]+2 or cx+r<-2:
+        return bg.copy()
+    sx,sy,sr=src_circle
+    yy,xx=np.mgrid[:bg.shape[0],:bg.shape[1]]
+    # Floor contact and elongated soft cast shadow in the original light direction.
+    broad=np.exp(-.5*(((xx-cx-.36*r)/(.99*r))**2+((yy-cy-.96*r)/(.22*r))**2))
+    contact=np.exp(-.5*(((xx-cx)/(.57*r))**2+((yy-cy-.94*r)/(.075*r))**2))
+    darkness=1-.25*broad-.34*contact
+    image*=darkness[:,:,None].astype('float32')
+    u=(sx+(xx-cx)*sr/r).astype('float32');v=(sy+(yy-cy)*sr/r).astype('float32')
+    rgb,alpha=texture
+    rgb=cv2.remap(rgb,u,v,cv2.INTER_LINEAR,borderMode=cv2.BORDER_REPLICATE)
+    alpha=cv2.remap(alpha,u,v,cv2.INTER_LINEAR,borderMode=cv2.BORDER_CONSTANT)
+    image=image*(1-alpha[:,:,None])+rgb*alpha[:,:,None]
+    return np.clip(image,0,255).astype('uint8')
+
+def main():
+    source=np.load(ROOT/'source_pixels.npz')['rgb']
+    tracks=ball_circles(source)
+    bg=room_plate(source,tracks)
+    at=free_trajectory(tracks)
+    textures=[sphere_texture(im,bg,tracks[i]) for i,im in enumerate(source)]
+    output=[]
+    h,w=bg.shape[:2];yy,xx=np.mgrid[:h,:w]
+    for i in range(len(source)):
+        circle=at(i)
+        if i<12:
+            # Ball history is unaffected until the missing block's first contact.
+            # Copy its original pixels and shadow, stopping before the block.
+            cx,cy,r=circle
+            mask=((xx-cx)**2+(yy-cy)**2<(r+2)**2)
+            mask|=(((xx-cx-5)/(r+14))**2+((yy-cy-r+1)/8)**2<1)
+            mask&=(xx<107)
+            frame=bg.copy();frame[mask]=source[i][mask]
+        else:
+            # Source later contains a clean view of a complete rotation of the ball.
+            # Retiming this cycle preserves the source ball's texture and lighting.
+            # Eight slow-spin samples per output sample approximate free rolling.
+            j=18+((i-12)*8)%52
+            frame=composite_ball(bg,textures[j],tracks[j],circle)
+        output.append(frame)
+    frames=np.array(output,dtype='uint8')
+    artifact_io.save(frames,ROOT)
+    Image.fromarray(bg).resize((768,432)).save(ROOT/'.tmp/room_plate.png')
+    print('Saved',frames.shape,'RGB frames at 24 fps')
+
+if __name__=='__main__':
+    main()

\"\"\"Remove the block at time zero; reconstruct the room and continue free rolling.
Only source pixels and local image/geometry calculations are used.
\"\"\"
from pathlib import Path
import warnings
import numpy as np
import cv2
from scipy.optimize import least_squares
from scipy.ndimage import gaussian_filter
from PIL import Image
import artifact_io

ROOT = Path(__file__).resolve().parent

def ball_circles(source):
    tracks=[]
    for image in source:
        a=image.astype(float)
        mask=(a[:,:,0]>1.75*a[:,:,1])&(a[:,:,0]-a[:,:,1]>60)&(a[:,:,0]>1.8*a[:,:,2])
        contours,_=cv2.findContours(mask.astype('uint8'),cv2.RETR_EXTERNAL,cv2.CHAIN_APPROX_NONE)
        points=cv2.convexHull(max(contours,key=cv2.contourArea)).reshape(-1,2).astype(float)
        fit=np.linalg.lstsq(np.c_[2*points,np.ones(len(points))],(points**2).sum(1),rcond=None)[0]
        cx,cy=fit[:2]
        radius=np.sqrt(fit[2]+cx*cx+cy*cy)+.6
        tracks.append((cx,cy,radius))
    return np.array(tracks)

def room_plate(source, tracks):
    h,w=source.shape[1:3]
    y,x=np.mgrid[:h,:w]
    keys=[0,10,12,14,15,17,20,23,26,29,33,95]
    left=[108,108,109,118,121,134,152,166,177,181,181,181]
    right=[167,167,168,177,180,193,212,226,239,244,245,245]
    top=[55,55,55,56,57,59,61,63,65,66,66,66]
    bottom=[105,105,105,106,107,110,112,114,117,118,118,118]
    available=[]
    for i,image in enumerate(source):
        cx,cy,r=tracks[i]
        bad=((x-cx)**2+(y-cy)**2<=(r+3)**2)
        bad |= ((x-cx-6)/(r+16))**2+((y-cy-r+2)/10)**2<1
        l=np.interp(i,keys,left);rr=np.interp(i,keys,right)
        t=np.interp(i,keys,top);b=np.interp(i,keys,bottom)
        margin=8 if i<15 else 15
        bad|=(x>=l-3)&(x<=rr+margin)&(y>=t-1)&(y<=b+17)
        pixels=image.astype(float);pixels[bad]=np.nan
        available.append(pixels)
    with warnings.catch_warnings():
        warnings.simplefilter('ignore', RuntimeWarning)
        bg=np.nanmedian(np.array(available),axis=0)
    holes=~np.isfinite(bg[:,:,0])
    # Small footprint never observed: complete with classical local image inpainting.
    bg=np.nan_to_num(bg).astype('uint8')
    bg=cv2.inpaint(bg,holes.astype('uint8')*255,3,cv2.INPAINT_TELEA)
    # Keep static wall at its exact source appearance (no intervention affects it).
    wall=(x>=69)&(y<=43+(x-69)*.156)
    bg[wall]=source[0][wall]
    return bg

def free_trajectory(tracks):
    t=np.arange(12,dtype=float)
    observed=tracks[:12]
    # Projected straight world-space motion is a rational curve under perspective.
    def residual(p):
        k,a,b,c,d,r,e=p
        den=1-k*t
        return np.r_[(a+b*t)/den-observed[:,0],(c+d*t)/den-observed[:,1],
                     .75*((r+e*t)/den-observed[:,2])]
    p=least_squares(residual,[.001,15.4,7,56.2,1.4,15,.15]).x
    k,a,b,c,d,r,e=p
    def at(i):
        if i<12:
            return tracks[i]
        den=1-k*i
        return np.array([(a+b*i)/den,(c+d*i)/den,(r+e*i)/den])
    return at

def sphere_texture(image, background, circle):
    # Recover a soft RGB sphere sprite, excluding its floor shadow.
    cx,cy,r=circle
    radius=r-.25
    h,w=image.shape[:2]
    yy,xx=np.mgrid[:h,:w]
    distance=np.sqrt((xx-cx)**2+(yy-cy)**2)
    alpha=np.clip((radius+.65-distance)/1.3,0,1)
    # Recover surface color from the antialiased edge rather than copy floor pixels.
    surface=(image.astype(float)-background*(1-alpha[:,:,None]))/np.maximum(alpha[:,:,None],.22)
    # A dark contact edge needs a slightly soft transition, as in the factual render.
    return np.clip(surface,0,255).astype('float32'),alpha.astype('float32')

def composite_ball(bg, texture, src_circle, target_circle):
    image=bg.astype('float32').copy()
    cx,cy,r=target_circle
    if cx-r>bg.shape[1]+2 or cx+r<-2:
        return bg.copy()
    sx,sy,sr=src_circle
    yy,xx=np.mgrid[:bg.shape[0],:bg.shape[1]]
    # Floor contact and elongated soft cast shadow in the original light direction.
    broad=np.exp(-.5*(((xx-cx-.36*r)/(.99*r))**2+((yy-cy-.96*r)/(.22*r))**2))
    contact=np.exp(-.5*(((xx-cx)/(.57*r))**2+((yy-cy-.94*r)/(.075*r))**2))
    darkness=1-.25*broad-.34*contact
    image*=darkness[:,:,None].astype('float32')
    u=(sx+(xx-cx)*sr/r).astype('float32');v=(sy+(yy-cy)*sr/r).astype('float32')
    rgb,alpha=texture
    rgb=cv2.remap(rgb,u,v,cv2.INTER_LINEAR,borderMode=cv2.BORDER_REPLICATE)
    alpha=cv2.remap(alpha,u,v,cv2.INTER_LINEAR,borderMode=cv2.BORDER_CONSTANT)
    image=image*(1-alpha[:,:,None])+rgb*alpha[:,:,None]
    return np.clip(image,0,255).astype('uint8')

def main():
    source=np.load(ROOT/'source_pixels.npz')['rgb']
    tracks=ball_circles(source)
    bg=room_plate(source,tracks)
    at=free_trajectory(tracks)
    textures=[sphere_texture(im,bg,tracks[i]) for i,im in enumerate(source)]
    output=[]
    h,w=bg.shape[:2];yy,xx=np.mgrid[:h,:w]
    for i in range(len(source)):
        circle=at(i)
        if i<12:
            # Ball history is unaffected until the missing block's first contact.
            # Copy its original pixels and shadow, stopping before the block.
            cx,cy,r=circle
            mask=((xx-cx)**2+(yy-cy)**2<(r+2)**2)
            mask|=(((xx-cx-5)/(r+14))**2+((yy-cy-r+1)/8)**2<1)
            mask&=(xx<107)
            frame=bg.copy();frame[mask]=source[i][mask]
        else:
            # Source later contains a clean view of a complete rotation of the ball.
            # Retiming this cycle preserves the source ball's texture and lighting.
            # Eight slow-spin samples per output sample approximate free rolling.
            j=18+((i-12)*8)%52
            frame=composite_ball(bg,textures[j],tracks[j],circle)
        output.append(frame)
    frames=np.array(output,dtype='uint8')
    artifact_io.save(frames,ROOT)
    Image.fromarray(bg).resize((768,432)).save(ROOT/'.tmp/room_plate.png')
    print('Saved',frames.shape,'RGB frames at 24 fps')

if __name__=='__main__':
    main()
