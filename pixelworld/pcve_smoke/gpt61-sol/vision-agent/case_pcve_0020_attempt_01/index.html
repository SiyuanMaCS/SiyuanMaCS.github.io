                m|=((X-(x-1.5))/(r*1.2))**2+((Y-(y+r+1.2))/4.2)**2<1
            a[m]=np.nan
            stack.append(a)
        with __import__('warnings').catch_warnings():
            __import__('warnings').simplefilter('ignore')
            clean=np.nanmedian(np.array(stack),axis=0)
        missing=np.isnan(clean[:,:,0]).astype(np.uint8)*255
        clean=np.nan_to_num(clean).astype(np.uint8)
        clean=cv2.inpaint(clean,missing,8,cv2.INPAINT_TELEA)
        # feather only outside original object removal bounds
        if x0==69:
            mask=((X-100.5)/17.5)**2+((Y-92)/18.5)**2<1
            mask|=((X-99)/19)**2+((Y-107)/5)**2<1
        else:
            mask=((X-237.8)/13.2)**2+((Y-66)/13.2)**2<1
            mask|=((X-237)/15)**2+((Y-78)/4.5)**2<1
        alpha=cv2.GaussianBlur(mask.astype(np.float32),(7,7),1.2)[:,:,None]
        region=base[y0*S:y1*S,x0*S:x1*S].astype(float)
        base[y0*S:y1*S,x0*S:x1*S]=np.clip(region*(1-alpha)+clean*alpha,0,255).astype(np.uint8)
    return base

N=180
U,V=np.meshgrid(np.linspace(-1.06,1.06,N),np.linspace(-1.06,1.06,N))
A=np.clip((1-np.sqrt(U*U+V*V))*N/2+0.7,0,1).astype(np.float32)

def sample(raw,c):
    x,y,r=c
    return cv2.remap(raw,((x+r*U)*S).astype('float32'),((y+r*V)*S).astype('float32'),cv2.INTER_LINEAR)

def alpha_paste(canvas,tex,center,radius,alpha=A,angle=0):
    x,y=center
    left=int(np.floor((x-radius*1.06)*S));top=int(np.floor((y-radius*1.06)*S))
    right=int(np.ceil((x+radius*1.06)*S));bottom=int(np.ceil((y+radius*1.06)*S))
    w,h=right-left,bottom-top
    # remapping maintains smooth, subpixel placement as the bodies slow down.
    xx,yy=np.meshgrid((np.arange(left,right)/S-x)/radius,(np.arange(top,bottom)/S-y)/radius)
    ca,sa=np.cos(angle),np.sin(angle)
    mx=((ca*xx+sa*yy)/2.12+.5)*(N-1)
    my=((-sa*xx+ca*yy)/2.12+.5)*(N-1)
    color=cv2.remap(tex,mx.astype('float32'),my.astype('float32'),cv2.INTER_LINEAR)
    aa=np.clip((1-np.sqrt(xx*xx+yy*yy))*radius*S+.7,0,1)[:,:,None]
    view=canvas[top:bottom,left:right]
    view[:]=view*(1-aa)+color*aa


def shadow(canvas,x,y,r,strength=.19):
    Y,X=np.meshgrid(np.arange(canvas.shape[0])/S,np.arange(canvas.shape[1])/S,indexing='ij')
    # Soft, warm floor-contact occlusion, matching the incident room lighting.
    d=((X-(x-1.0))/(r*.78))**2+((Y-(y+r+.8))/(r*.16+1.0))**2
    soft=np.exp(-d*1.8)*strength
    contact=np.exp(-((X-x)/(r*.40))**2*2-((Y-y-r)/.7)**2*2)*.16
    shade=np.clip(soft+contact,0,.4)[:,:,None]
    canvas*=1-shade


def main():
    raw=source_frames()
    original=np.load(ROOT/'source_pixels.npz')['rgb']
    clean=background(raw)
    ball0=np.array([100.5,92.0,14.1])
    football_tex=sample(raw[22],ball0)
    # Reconstruct the hidden part of the pink sphere from an earlier unoccluded
    # photograph of the very same ball. Its visible surface at impact is exact.
    pink0=pink_early[22].copy()
    pinktex=sample(raw[22],pink0)
    earlier=sample(raw[14],pink_early[14])
    gx=pink0[0]+pink0[2]*U;gy=pink0[1]+pink0[2]*V
    occluded=np.clip((14.4-np.sqrt((gx-100.5)**2+(gy-92.0)**2))/1.5+.5,0,1)
    pinktex=(pinktex*(1-occluded[:,:,None])+earlier*occluded[:,:,None]).astype(np.uint8)

    # Linear impulse: original mass ratio 1:1, edited ratio 1:10.
    # Restitution and rolling drag are unchanged. The heavy ball receives only
    # 2/11 of the equal-mass collision velocity and quickly settles.
    m_pink,m_football=1.,10.
    restitution=.68
    incident_speed=20.
    v_p=(m_pink-restitution*m_football)/(m_pink+m_football)*incident_speed
    v_f=(1+restitution)*m_pink/(m_pink+m_football)*incident_speed
    drag=5.5
    def displacement(speed,t):
        duration=min(t,abs(speed)/drag)
        return np.sign(speed)*(abs(speed)*duration-.5*drag*duration**2)

    out=original.copy()
    for i in range(23,96):
        t=(i-22)/FPS
        dp=displacement(v_p,t)
        df=displacement(v_f,t)
        pink=pink0.copy();pink[1]+=dp;pink[0]+=.27*dp
        pink[2]+=dp*.105
        football=ball0.copy();football[1]+=df;football[0]+=.08*df
        football[2]+=df*.18
        canvas=clean.astype(np.float32)
        shadow(canvas,*pink,strength=.18)
        shadow(canvas,*football,strength=.20)
        # Front-to-back occlusion follows depth throughout the rebound.
        alpha_paste(canvas,pinktex,pink[:2],pink[2],angle=-dp*.018)
        alpha_paste(canvas,football_tex,football[:2],football[2],angle=df*.016)
        out[i]=np.asarray(Image.fromarray(np.clip(canvas,0,255).astype(np.uint8)).resize((256,144),Image.Resampling.LANCZOS))
    artifact_io.save(out,ROOT)
    # Standard mp4 for convenient external viewing, in addition to RGB artifacts.
    writer=cv2.VideoWriter(str(ROOT/'edited_video.mp4'),cv2.VideoWriter_fourcc(*'mp4v'),24,(256,144))
    for f in out: writer.write(cv2.cvtColor(f,cv2.COLOR_RGB2BGR))
    writer.release()
    Image.fromarray(clean).resize((512,288),Image.Resampling.LANCZOS).save(ROOT/'clean_plate.png')
    print('Saved 96 RGB frames; mass ratio 10:1, football drift %.3f pixels, pink recoil %.3f pixels.'%(displacement(v_f,4),displacement(v_p,4)))
