    contact_mid = ((r0+vr*contact_t)+(y0+vy*contact_t))/2

    # Work in a small source-resolution region; everything else stays byte-identical.
    left, top, right, bottom = 97, 60, 160, 90
    crop = (slice(top*5, bottom*5), slice(left*5, right*5))
    frames = source.copy()
    for t in range(96):
        incoming_mid = (r0+y0+(vr+vy)*t)/2
        if t < contact_t:
            # Before contact the red and yellow stones are untouched.
            mx = incoming_mid
            canvas = hi[0][crop].astype(np.float32).copy()
            # Start with the exact unedited frame under the new center stone.
            # Only a central patch is committed, so the approaching stones remain original.
            draw_stone(canvas, blue, (mx-left, 76.1 + .064*(mx-130.65)-top))
            out = np.array(Image.fromarray(np.clip(canvas, 0, 255).astype(np.uint8)).resize((right-left, bottom-top), Image.Resampling.LANCZOS))
            bx0, bx1 = int(np.floor(mx))-8, int(np.floor(mx))+10
            frames[t, top:bottom, bx0:bx1] = out[:, bx0-left:bx1-left]
        else:
            # Equal opposing impacts leave the middle stone in the center.
            # The incoming pair reverse earlier than in the unedited two-stone shot.
            dt = t-contact_t
            gap = contact_gap + .195*dt - .00022*dt*dt
            mx = contact_mid
            my = 76.1 + .064*(mx-130.65)
            positions = [(mx-gap, my-.064*gap), (mx, my), (mx+gap, my+.064*gap)]
            canvas = hi[0][crop].astype(np.float32).copy()
            # Rear-to-front order agrees with the oblique view down the rink.
            for sprite, p in zip((red, blue, yellow), positions):
                draw_stone(canvas, sprite, (p[0]-left, p[1]-top))
            out = np.array(Image.fromarray(np.clip(canvas, 0, 255).astype(np.uint8)).resize((right-left, bottom-top), Image.Resampling.LANCZOS))
            frames[t, top:bottom, left:right] = out
    artifact_io.save(frames, ROOT)


if __name__ == '__main__':
    main()
