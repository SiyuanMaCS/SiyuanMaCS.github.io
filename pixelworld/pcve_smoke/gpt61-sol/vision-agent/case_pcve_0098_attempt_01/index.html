    yy,xx=np.mgrid[:patch.shape[0],:patch.shape[1]]
    lx,ly=x-left,y-top
    dist=np.sqrt(((xx-lx)/rx)**2+((yy-ly)/ry)**2)
    alpha=np.clip((1-dist)*min(rx,ry)+.45,0,1)
    alpha=gaussian_filter(alpha,.35)
    # Capture the photographed contact shadow and faint tinted reflection,
    # but not the wood texture, by storing multiplicative lighting ratios.
    px,py=origin
    back=plate[top-py:top-py+patch.shape[0],left-px:left-px+patch.shape[1]].astype(np.float32)
    lighting=np.clip((patch+3)/(back+3),.42,1.16)
    smask=np.exp(-.5*((xx-lx)/(rx*.72))**2)
    smask*=np.clip((yy-(ly+ry-1))/2,0,1)
    smask*=np.clip((ly+ry+45-yy)/20,0,1)
    smask*=np.clip((yy-(ly+ry-4))/5,0,1)
    # No spurious color ratios outside the real source shadow footprint.
    lighting=1+(lighting-1)*smask[:,:,None]
    return patch,alpha,lighting,(lx,ly)


def paste_sprite(canvas,sprite,center):
    rgb,alpha,lighting,local=sprite
    # Subpixel translation at the source video's 5x sampling resolution.
    dx,dy=np.asarray(center)-local
    mat=np.array([[1,0,dx],[0,1,dy]],np.float32)
    h,w=canvas.shape[:2]
    acol=cv2.warpAffine(alpha.astype(np.float32),mat,(w,h),flags=cv2.INTER_LINEAR)
    premult=cv2.warpAffine((rgb*alpha[:,:,None]).astype(np.float32),mat,(w,h),flags=cv2.INTER_LINEAR)
    shade=cv2.warpAffine((lighting-1).astype(np.float32),mat,(w,h),flags=cv2.INTER_LINEAR)
    canvas*=1+shade
    canvas[:]=canvas*(1-acol[:,:,None])+premult


def generate():
    source=np.load(ROOT/'source_pixels.npz')['rgb']
    cap=cv2.VideoCapture(str(ROOT/'source_video.mp4'))
    full=[]
    while True:
        ok,bgr=cap.read()
        if not ok:break
        full.append(cv2.cvtColor(bgr,cv2.COLOR_BGR2RGB))
    cap.release()

    # Camera is fixed. Work at the 1280x720 original raster for clean edges.
    ox,oy=0,350
    ex,ey=330,555
    crops=[f[oy:ey,ox:ex].copy() for f in full]
    centers=[]
    blue0=np.array([124.179,442.50])
    for i,f in enumerate(full):
        objs=[]
        if i<=27:
            bc=detect(f,'blue',(0,398,185,489))
            if bc is not None:
                if i<=21:bc=blue0.copy()
                objs.append((bc,42.5))
        if i>=21:
            rc=detect(f,'red',(0,330 if i==21 else 398,300,488))
            if rc is not None: objs.append((rc,44.0))
        centers.append(objs)
    bg=clean_plate(crops,centers,(ox,oy))
    # Separate sphere surfaces retain source illumination and material.
    blue=make_sprite(full[0],blue0,(41.1,40.5),bg,(ox,oy),True)
    red_center=np.array([173.5,440.5])
    red=make_sprite(full[23],red_center,(42.5,41.5),bg,(ox,oy))

    # In source pixel units, the approach speed is 11.8 px/frame. Equal-mass
    # outbound speeds imply e~.24; subsequent stopping implies a~.63.
    m_red,m_blue=1.0,40.0
    e=.24
    incoming=-11.8
    vr=(m_red-e*m_blue)/(m_red+m_blue)*incoming
    vb=(1+e)*m_red/(m_red+m_blue)*incoming
    resistance=.63
    contact_time=21.66
    red_contact=41.00
    blue_contact=(blue0[0]+.5)/5-.5
    def displacement(v,t):
        t=min(max(t,0),abs(v)/resistance)
        return v*t-np.sign(v)*resistance*t*t/2

    out=source.copy()
    trajectories=[]
    for i in range(22,96):
        dt=i-contact_time
        bx=blue_contact+displacement(vb,dt)
        rx=red_contact+displacement(vr,dt)
        by=(blue0[1]+.5)/5-.5
        ry=87.65 + .06*min(dt,5)
        # Ground plane projection is slightly diagonal in the source.
        by += -.048*(bx-blue_contact)
        ry += -.035*(rx-red_contact)
        # A static plate is restricted to the object's affected footprints;
        # all ramp, yellow-marble, table and room pixels outside stay factual.
        base=full[i].astype(np.float32)
        affected=np.zeros((720,1280),np.float32)
        # Actual source spheres and ground-lighting footprints to erase.
        yy,xx=np.mgrid[oy:ey,ox:ex]
        for c,rad in centers[i]:
            disk=np.sqrt((xx-c[0])**2+(yy-c[1])**2)
            mask=np.clip((rad+7-disk)/3,0,1)
            sh=np.sqrt(((xx-c[0])/(rad+21))**2+((yy-c[1]-rad-14)/25)**2)
            mask=np.maximum(mask,np.clip((1.05-sh)*12,0,1))
            affected[oy:ey,ox:ex]=np.maximum(affected[oy:ey,ox:ex],mask)
        # Also clean destinations so old blue shadow does not double up.
        for cx,cy,rad in [(bx*5+2,by*5+2,42.5),(rx*5+2,ry*5+2,44.)]:
            disk=np.sqrt((xx-cx)**2+(yy-cy)**2)
            mask=np.clip((rad+6-disk)/3,0,1)
            sh=np.sqrt(((xx-cx)/(rad+22))**2+((yy-cy-rad-13)/25)**2)
            mask=np.maximum(mask,np.clip((1.05-sh)*12,0,1))
            affected[oy:ey,ox:ex]=np.maximum(affected[oy:ey,ox:ex],mask)
        # Yellow remains in the exact factual pixels, including its shadow.
        affected[:399]=0
        a=affected[oy:ey,ox:ex,None]
        base[oy:ey,ox:ex]=base[oy:ey,ox:ex]*(1-a)+bg*a
        paste_sprite(base,blue,np.array([bx*5+2,by*5+2]))
        paste_sprite(base,red,np.array([rx*5+2,ry*5+2]))
        out[i]=cv2.resize(np.rint(base).clip(0,255).astype(np.uint8),(256,144),interpolation=cv2.INTER_AREA)
        trajectories.append((i,rx,ry,bx,by))
    artifact_io.save(out,ROOT)
    print('Saved 96 frames, 256x144 at 24 fps.')
    print('Impulse model: red vx %.3f; blue vx %.3f px/frame; final positions '%(vr,vb),trajectories[-1])
