from pathlib import Path
import numpy as np
import cv2
from PIL import Image, ImageDraw
import artifact_io

ROOT=Path(__file__).resolve().parent

def load_source():
    cap=cv2.VideoCapture(str(ROOT/'source_video.mp4'))
    result=[]
    for _ in range(96):
        ok,f=cap.read()
        if not ok: raise RuntimeError('Incomplete source')
        result.append(cv2.cvtColor(f,cv2.COLOR_BGR2RGB))
    cap.release()
    return np.asarray(result)

# Silhouettes measured directly from the source image, unchanged by the mass edit.
POLYS=[[(837,266),(843,265),(848,269),(850,278),(850,283),(860,284),(867,288),(866,298),(858,302),(830,302),(814,298),(810,292),(812,286),(827,282),(836,282)],[(694,319),(700,316),(707,318),(712,322),(714,332),(724,333),(731,336),(735,340),(734,350),(729,354),(709,357),(688,355),(676,351),(674,339),(680,335),(692,333)],[(519,389),(521,383),(529,380),(536,382),(541,388),(542,396),(553,398),(563,402),(568,407),(568,417),(561,423),(550,426),(524,426),(506,422),(496,417),(496,405),(504,400),(518,397)]]

def project(p):
    return cv2.perspectiveTransform(np.asarray(p,np.float32).reshape(1,-1,2),H)[0]

def track(hi):
    yy,xx=np.mgrid[:720,:1280];out=[];prevw=BASE[2]-[2,9]
    for f in hi:
        r,g,b=f.astype(float).transpose(2,0,1);points=[]
        for j in range(3):
            if j==0: m=(b-r>40)&(b-g>10)&(r<125)&(xx>580)&(xx<890)&(yy>255)&(yy<450)
            elif j==1: m=(r-g>45)&(r-b>30)&(xx>470)&(xx<750)&(yy>280)&(yy<450)
            else: m=(r>244)&(g>244)&(b>244)&((xx-prevw[0])**2+(yy-prevw[1])**2<45**2)
            n,l,st,c=cv2.connectedComponentsWithStats(m.astype(np.uint8))
            ids=[k for k in range(1,n) if st[k,4]>5]
            k=max(ids,key=lambda k:st[k,4]);points.append(c[k])
            if j==2:prevw=c[k]
        out.append(points)
    return np.asarray(out)

def make_background(hi,tr):
    masks=[]
    for j,p in enumerate(POLYS):
        mask=np.zeros((720,1280),np.uint8);cv2.fillPoly(mask,[np.array(p,np.int32)],255)
        mask=cv2.dilate(mask,np.ones((9,9),np.uint8))
        # Preserve the original broad lighting, excluding just the moving reflections.
        cx,cy=BASE[j]
        cv2.ellipse(mask,(int(cx),int(cy+22)),(24+j*5,28),0,0,360,255,-1)
        masks.append(mask)
    bg=hi[0].copy();valid=np.zeros((720,1280),bool)
    for i in range(96):
        occupied=np.zeros((720,1280),np.uint8)
        for j,mask in enumerate(masks):
            d=tr[i,j]-tr[0,j]
            moved=cv2.warpAffine(mask,np.float32([[1,0,d[0]],[0,1,d[1]]]),(1280,720),flags=cv2.INTER_NEAREST)
            occupied=cv2.max(occupied,moved)
        good=(occupied==0)&~valid
        bg[good]=hi[i][good];valid[good]=True
    holes=np.uint8(~valid)*255
    bg=cv2.inpaint(bg,holes,5,cv2.INPAINT_TELEA)
    return bg,holes

def simulation():
    # The intervention changes inertia only, not geometry or the initial velocity.
    p0=cv2.perspectiveTransform(BASE.astype(np.float32)[None],np.linalg.inv(H))[0]
    y=p0[:,1].astype(float);x=np.ones(3)*.6
    v=np.array([-.787,0.,0.]);m=np.array([10.,1.,1.]);radius=.068
    output=[np.column_stack((x,y.copy()))]
    dt=1/(24*100)

                    closing=(v[b]-v[a])*normal
                    if closing>0:
                        impulse=1.96*closing/(1/m[a]+1/m[b])
                        v[a]+=impulse*normal/m[a];v[b]-=impulse*normal/m[b]
        output.append(np.column_stack((x,y.copy())))
    return np.array(output)

def composite(frame,rgb,mask,M):
    h,w=frame.shape[:2]
    # Crop the affine destination for efficiency, retaining sub-pixel filtering.
    corners=np.float32([[0,0],[rgb.shape[1],0],[0,rgb.shape[0]],[rgb.shape[1],rgb.shape[0]]])
    dest=corners@M[:,:2].T+M[:,2]
    lo=np.maximum(np.floor(dest.min(0)).astype(int)-3,[0,0]);hi=np.minimum(np.ceil(dest.max(0)).astype(int)+3,[w,h])
    if np.any(hi<=lo):return
    mm=M.copy();mm[:,2]-=lo
    sz=tuple((hi-lo).tolist())
    aa=cv2.warpAffine(mask,mm,sz,flags=cv2.INTER_LINEAR)
    col=cv2.warpAffine(rgb*mask[...,None],mm,sz,flags=cv2.INTER_LINEAR)
    sl=np.s_[lo[1]:hi[1],lo[0]:hi[0]]
    frame[sl]=frame[sl]*(1-aa[...,None])+col

            jac=project([positions[i,j],positions[i,j]+[.01,0]])
            scale=np.linalg.norm(jac[1]-jac[0])/np.linalg.norm(jac0[1]-jac0[0])

    writer.release()
