+"""Remove the initiating domino and its lighting footprint.
+
+With no initiating body, the other three dominoes retain their initial
+equilibrium poses for the complete four-second source timeline.  All scene
+reconstruction uses only pixels from the supplied video.
+"""
+from pathlib import Path
+
+import numpy as np
+from scipy.ndimage import gaussian_filter, gaussian_filter1d
+from PIL import Image
+
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+
+
+def smoothstep(a, b, x):
+    t = np.clip((x - a) / (b - a), 0, 1)
+    return t * t * (3 - 2 * t)
+
+
+def reconstruct():
+    source = np.load(ROOT / 'source_pixels.npz')['rgb']
+    initial = source[0].astype(np.float64)
+    # The initial camera and the three untriggered dominoes remain unchanged.
+    image = initial.copy()
+    h, w = initial.shape[:2]
+    yy, xx = np.indices((h, w), dtype=np.float64)
+
+    # After the original dominoes have fallen, the upper tabletop behind the
+    # first one is visible. Recover it directly, without repainting the room.
+    clear = np.median(source[70:96].astype(np.float64), axis=0)
+    upper_weight = (smoothstep(51, 57, xx)
+                    * (1 - smoothstep(85, 93, xx))
+                    * smoothstep(29, 33, yy)
+                    * (1 - smoothstep(65, 73, yy)))
+    image = image * (1 - upper_weight[..., None]) + clear * upper_weight[..., None]
+
+    # Below that area, the fallen original body hides the tabletop. Its
+    # unoccluded neighboring pixels provide the local lighting gradient.
+    # Row-wise interpolation respects the vertical color/illumination change
+    # and retains the approach to the second domino's independent shadow.
+    left = gaussian_filter1d(np.mean(initial[:, 42:49], axis=1), 1.1, axis=0)
+    right = gaussian_filter1d(np.mean(initial[:, 94:99], axis=1), 1.1, axis=0)
+    horizontal = np.clip((xx - 46) / (96 - 46), 0, 1)
+    surface = left[:, None, :] * (1 - horizontal[..., None]) + right[:, None, :] * horizontal[..., None]
+
+    # A wide, soft footprint includes the contact shadow and its long soft
+    # reflection. Its edge is separate from the next domino's contact patch.
+    lower_weight = (smoothstep(46, 57, xx)
+                    * (1 - smoothstep(83, 98, xx))
+                    * smoothstep(64, 72, yy))
+    image = image * (1 - lower_weight[..., None]) + surface * lower_weight[..., None]
+
+    # Exact opaque silhouette clearance: especially the diagonal foot and
+    # antialiasing at the lower-left edge. The seamless row-field continues
+    # past the removed body's former base.
+    hard = ((yy >= 69) & (yy <= 102) & (xx >= 57) & (xx <= 80))
+    image[hard] = surface[hard]
+
+    image = np.clip(np.rint(image), 0, 255).astype(np.uint8)
+    return np.repeat(image[None], 96, axis=0)
+
+
+def main():
+    frames = reconstruct()
+    artifact_io.save(frames, ROOT)
+    # Also provide a conventional video, in addition to the required RGB
+    # tensor, frame PNGs, and self-contained HTML player.
+    import cv2
+    out = cv2.VideoWriter(str(ROOT / 'edited_video.mp4'),
+                          cv2.VideoWriter_fourcc(*'mp4v'), 24, (256, 144))
+    for frame in frames:
+        out.write(cv2.cvtColor(frame, cv2.COLOR_RGB2BGR))
+    out.release()
+    Image.fromarray(frames[0]).resize((1024, 576)).save(ROOT / '.tmp' / 'edited_preview.png')
+
+
+if __name__ == '__main__':
+    main()
