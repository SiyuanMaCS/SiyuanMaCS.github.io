+def block_layer(v,bg,j):

    la=np.arccos(np.clip(-local[:,:,1],-1,1))
    lo=np.arctan2(local[:,:,0],local[:,:,2])
    mx=((lo+np.pi)/(2*np.pi)*atlas.shape[1]-.5).astype(np.float32)
    my=(la/np.pi*atlas.shape[0]-.5).astype(np.float32)
    color=cv2.remap(atlas,mx,my,cv2.INTER_LINEAR,borderMode=cv2.BORDER_WRAP)*light(normal)[:,:,None]
    alpha=np.clip((1-np.sqrt(rr))*min(rx,ry)+.5,0,1)
    out[ya:yb,xa:xb]=out[ya:yb,xa:xb]*(1-alpha[:,:,None])+color*alpha[:,:,None]


def block_layer(v,bg,j):
    p=polygon(j)
    m=np.zeros(bg.shape[:2],np.uint8)
    cv2.fillPoly(m,[np.round(p).astype(np.int32)],255)
    # Native edges include source motion blur. Feather by less than one final pixel.
    alpha=gaussian_filter(m.astype(np.float32)/255,.75)
    # Native shadow contrast is detached from floor texture before translation.
    cx=float(p[:,0].mean());bottom=float(p[:,1].max())
    yy,xx=np.indices(m.shape)
    proximity=np.exp(-.5*((xx-cx-5)/180)**2-.5*((yy-bottom+8)/38)**2)
    ratio=np.minimum(1,v[j].astype(np.float32)/np.maximum(bg,1))
    dark=np.mean(1-ratio,axis=2)
    shadow=np.clip(dark,0,.40)*proximity*(1-alpha)
    # Sphere shadow/body, close to the cube at contact, must not travel with it.
    shadow[:,:int(p[:,0].min()-12)]=0
    return alpha,shadow


def generate():
    v=load_video()
    original=np.load(ROOT/'source_pixels.npz')['rgb']
    bc,br=ball_geometry(v)
    bg=make_plate(v,bc,br)
    atlas=make_atlas(v,bc,br)
    # Integrate a 1.8x restitution impulse with unchanged sliding friction.
    # With the fitted native contact e~0.75, impulse changes by this factor.
    e=.75
    impulse=(1+1.8*e)/(1+e)
    block_polys=np.asarray([polygon(i) for i in range(96)])
    block_center=block_polys.mean(axis=1)
    origin=block_center[12]
    alphas={};shadows={}
    results=[]
    for i in range(96):
        if i<=12:
            results.append(original[i].copy());continue
        elapsed=i-12
        # Same Coulomb drag: scaled speed extends stopping distance quadratically.
        s=min(95,12+elapsed/impulse)
        k=int(s);r=s-k
        center=(1-r)*block_center[k]+r*block_center[min(k+1,95)]
        target=origin+(center-origin)*impulse**2
        # The extra angular impulse turns the sliding cube more rapidly as well.
        j=min(38,int(round(12+elapsed*impulse)))
        p=polygon(j);oldcenter=p.mean(axis=0)
        scale=1+.00018*(target[0]-oldcenter[0])
        scale=max(.98,min(1.13,scale))
        M=np.float32([[scale,0,target[0]-scale*oldcenter[0]],
                      [0,scale,target[1]-scale*oldcenter[1]]])
        out=bg.copy()
        if target[0]-180*scale<1280:
            if j not in alphas:
                alphas[j],shadows[j]=block_layer(v,bg,j)
            shadow=cv2.warpAffine(shadows[j],M,(1280,720),flags=cv2.INTER_LINEAR)
            out*=1-shadow[:,:,None]
        # Finite slip relaxation after the more elastic impact, then backward roll.
        dx=-4.7*elapsed-5.3*3*(1-np.exp(-elapsed/3))
        cx=500+dx;cy=366.5+.18*dx
        rx=84+.031*dx;ry=83.5+.030*dx
        phi=-.18-.075*elapsed+.115*3*(1-np.exp(-elapsed/3))
        render_ball(out,atlas,cx,cy,rx,ry,phi)
        if target[0]-180*scale<1280:
            alpha=cv2.warpAffine(alphas[j],M,(1280,720),flags=cv2.INTER_LINEAR)
            color=cv2.warpAffine(v[j].astype(np.float32),M,(1280,720),flags=cv2.INTER_LINEAR)
            out=out*(1-alpha[:,:,None])+color*alpha[:,:,None]
        small=cv2.resize(np.clip(out,0,255).astype(np.uint8),(256,144),interpolation=cv2.INTER_AREA)
        results.append(small)
    frames=np.asarray(results,np.uint8)
    artifact_io.save(frames,ROOT)
    # Also write an ordinary MP4 for convenient playback.
    writer=cv2.VideoWriter(str(ROOT/'edited_video.mp4'),cv2.VideoWriter_fourcc(*'mp4v'),24,(256,144))
    for f in frames:writer.write(cv2.cvtColor(f,cv2.COLOR_RGB2BGR))
    writer.release()
    print('Saved 96 RGB frames, 256x144, 24 fps:', ROOT/'pixels.npz')

if __name__=='__main__':generate()
