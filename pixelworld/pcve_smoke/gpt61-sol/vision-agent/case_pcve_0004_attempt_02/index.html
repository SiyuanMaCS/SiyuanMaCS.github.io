        frame = np.clip(source[0].astype(np.float32)+small_delta,0,255)
        output.append(np.rint(frame).astype(np.uint8))
    frames = np.stack(output)
    artifact_io.save(frames, ROOT)
    print(f'Saved {len(frames)} RGB frames; blue stop at {stop_time:.3f}s.')
