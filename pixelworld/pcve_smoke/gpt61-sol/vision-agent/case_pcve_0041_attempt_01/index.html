+"""Counterfactual friction edit, using only measured source motion and pixels.
+
+The car is still in free flight at the intervention. Its contact with the
+tabletop occurs near source frame 7; increased friction then dissipates its
+horizontal velocity seven times as quickly. The room and other objects are
+unaltered. No new appearance assets are used.
+"""
+from pathlib import Path
+import numpy as np
+import cv2
+from PIL import Image
+from scipy.ndimage import binary_fill_holes
+from scipy.optimize import curve_fit
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+
+
+def read_original():
+    cap = cv2.VideoCapture(str(ROOT / "source_video.mp4"))
+    frames = []
+    for i in range(24):
+        ok, bgr = cap.read()
+        if not ok:
+            raise RuntimeError("Cannot decode source video")
+        frames.append(cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB))
+    cap.release()
+    return np.asarray(frames)
+
+
+def sprite(image, clean, rect):
+    """Extract car and its original soft contact shadow at source resolution."""
+    x0, y0, x1, y1 = rect
+    foreground = image[y0:y1, x0:x1].astype(np.float32)
+    background = clean[y0:y1, x0:x1].astype(np.float32)
+    difference = np.max(np.abs(foreground - background), axis=2)
+    mask = (difference > 9).astype(np.uint8)
+    # Smooth compression noise, but retain all enclosed body/window pixels.
+    mask = cv2.morphologyEx(mask, cv2.MORPH_CLOSE, np.ones((3, 3), np.uint8))
+    count, labels, stats, _ = cv2.connectedComponentsWithStats(mask)
+    index = 1 + np.argmax(stats[1:, cv2.CC_STAT_AREA])
+    mask = binary_fill_holes(labels == index).astype(np.float32)
+    # Car edges remain source pixels, with a narrow antialias transition.
+    expanded = cv2.dilate(mask, np.ones((3, 3), np.uint8))
+    alpha = cv2.GaussianBlur(expanded, (3, 3), .55)
+    # A residual is preferable to a rectangular transplant: the receiving
+    # tabletop grain remains visible through the source contact shadow.
+    residual = (foreground - background) * alpha[:, :, None]
+    return residual, (x0, y0)
+
+
+def main():
+    source = np.load(ROOT / "source_pixels.npz")["rgb"]
+    hires = read_original()
+    clean_high = np.median(hires, axis=0).astype(np.uint8)
+    clean_low = np.median(source[:24], axis=0).astype(np.uint8)
+
+    # Recover the factual Coulomb-like deceleration from the red body, after
+    # landing, independently of the static red objects elsewhere in the room.
+    positions = []
+    for frame in source[6:26]:
+        a = frame.astype(float)
+        r, g, b = a.transpose(2, 0, 1)
+        difference = np.max(abs(a - source[0].astype(float)), axis=2)
+        mask = (r-g > 30) & (r-b > 30) & (difference > 20)
+        mask[:30] = False
+        mask[53:] = False
+        mask[:, :110] = False
+        mask[:, 210:] = False
+        y, x = np.where(mask)
+        weights = (r-g-30)[mask]
+        positions.append(np.sum(x*weights)/np.sum(weights))
+
+    def trajectory(t, p, v, deceleration):
+        elapsed = np.minimum(t, v/deceleration)
+        return p + v*elapsed - .5*deceleration*elapsed**2
+
+    parameters, _ = curve_fit(
+        trajectory, np.arange(len(positions)), positions,
+        p0=[130, 8, .45], bounds=([120, 5, .1], [140, 15, 2]))
+    initial_velocity = parameters[1]
+    friction_deceleration = parameters[2]*7
+    start = positions[0]
+    stop_time = initial_velocity/friction_deceleration
+
+    # The early source frames include the actual descending pitch. For the
+    # brief arrest phase, use source landing poses and their genuine shadows.
+    sprites = {}
+    for i in [7, 8, 9]:
+        center = positions[i-6]*5
+        rect = (int(center)-65, 206, int(center)+65, 278)
+        sprites[i] = sprite(hires[i], clean_high, rect)
+
+    frames = source.copy()
+    for i in range(7, 96):
+        t = min(float(i-6), stop_time)
+        new_position = start + initial_velocity*t - .5*friction_deceleration*t*t
+        pose = min(i, 9)
+        old_position = positions[pose-6]
+        dx = (new_position-old_position)*5
+        # The projected tabletop travel line rises slightly toward the right.
+        # Compensating for it keeps wheels planted when translating leftward.
+        dy = -.057*dx
+        residual, (x0, y0) = sprites[pose]
+        matrix = np.array([[1, 0, x0+dx], [0, 1, y0+dy]], np.float32)
+        moved = cv2.warpAffine(residual, matrix, (1280, 720),
+                               flags=cv2.INTER_LINEAR,
+                               borderMode=cv2.BORDER_CONSTANT)
+        scene = np.clip(clean_high.astype(np.float32)+moved, 0, 255).astype(np.uint8)
+        rendered = np.asarray(Image.fromarray(scene).resize((256,144), Image.Resampling.LANCZOS))
+        # Preserve the original timeline's fine compression variation in all
+        # unaffected areas. Only the car's swept region needs clean backing.
+        out = source[i].copy()
+        out[39:58, 117:211] = clean_low[39:58, 117:211]
+        # Blend the high-resolution reconstruction only around the new car.
+        left = int(np.floor(new_position-15))
+        right = int(np.ceil(new_position+16))
+        out[39:58, left:right] = rendered[39:58, left:right]
+        frames[i] = out
+
+    # Frames 1--5 (and the unaffected free-flight phase) are exact source RGB.
+    assert np.array_equal(frames[:5], source[:5])
+    artifact_io.save(frames, ROOT)
+    print(f"Factual deceleration: {parameters[2]:.3f} px/frameÂ²")
+    print(f"Modified deceleration: {friction_deceleration:.3f} px/frameÂ²")
+    print(f"Car stops at x={new_position:.2f}, frame {7+int(np.ceil(stop_time))}")
+
+
+if __name__ == "__main__":
+    main()
