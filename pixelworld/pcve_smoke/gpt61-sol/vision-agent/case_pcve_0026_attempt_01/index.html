+"""Counterfactual rolling volleyball, reconstructed from the supplied video.
+
+The floor path is fitted in perspective coordinates.  A piecewise rolling
+resistance model accounts for the harder floor and the high-resistance rug.
+Only the initial velocity changes (by 0.6); the surface resistances do not.
+The source sphere imagery is advected to intermediate physical path samples,
+so its texture, shading, contact shadow, and the unchanged room are retained.
+"""
+from pathlib import Path
+import cv2
+import numpy as np
+from scipy.optimize import least_squares
+from PIL import Image
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+FPS = 24
+
+def load_video():
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    images = []
+    while True:
+        ok, image = cap.read()
+        if not ok:
+            break
+        images.append(cv2.cvtColor(image, cv2.COLOR_BGR2RGB))
+    cap.release()
+    return np.asarray(images)
+
+
+def measure_sphere(images):
+    """Fit circles to the highly distinctive yellow/blue volleyball panels."""
+    values = []
+    h, w = images.shape[1:3]
+    scale = w / 256
+    yy, xx = np.indices((h, w))
+    for image in images[:20]:
+        r, g, b = image.astype(float).transpose(2, 0, 1)
+        mask = (((b-r > 25) & (b-g > 5)) |
+                ((r-b > 55) & (g-b > 40) & (g > 85)))
+        mask &= (yy > 55*scale) & (xx > 64*scale) & (xx < 125*scale)
+        n, labels, stats, centers = cv2.connectedComponentsWithStats(mask.astype(np.uint8))
+        k = 1 + np.argmax(stats[1:, cv2.CC_STAT_AREA])
+        cy, cx = np.where(labels == k)
+        mask &= ((xx-cx.mean())**2+(yy-cy.mean())**2 < (13*scale)**2)
+        y, x = np.where(mask)
+        hull = cv2.convexHull(np.column_stack([x,y]).astype(np.int32)).squeeze()
+        def residual(p):
+            return np.hypot(hull[:,0]-p[0], hull[:,1]-p[1])-p[2]
+        p0 = [(x.min()+x.max())/2, (y.min()+y.max())/2, (y.max()-y.min())/2]
+        fit = least_squares(residual, p0, loss='soft_l1', f_scale=1)
+        values.append(fit.x/scale)
+    return np.asarray(values)
+
+
+def fit_motion(observed):
+    """q is distance divided by the original stopping distance.
+
+    Projected position = start + delta * (1+depth)*q/(1+depth*q).
+    The rug transition and resistance are shared by factual and edited paths.
+    """
+    t = np.arange(20)/FPS
+    start, end = observed[0,:2], observed[-1,:2]
+
+    def factual_q(times, v, qe, a):
+        ve = np.sqrt(max(v*v-2*a*qe, 1e-9))
+        te = 2*qe/(v+ve)
+        arug = ve*ve/(2*(1-qe))
+        dt = np.clip(times-te, 0, ve/arug)
+        return np.where(times < te, v*times-a*times*times/2,
+                        qe+ve*dt-arug*dt*dt/2)
+
+    def residual(params):
+        v, qe, a, depth, radius = params
+        q = factual_q(t, v, qe, a)
+        fraction = (1+depth)*q/(1+depth*q)
+        centers = start+(end-start)*fraction[:,None]
+        radii = radius/(1+depth*q)
+        return np.r_[(centers-observed[:,:2]).ravel(), .5*(radii-observed[:,2])]
+
+    fit = least_squares(residual, [1.85,.54,.01,.65,11.4],
+                        bounds=([1,.2,0,.1,10],[4,.85,2,1.5,13]))
+    v, qe, a, depth, radius = fit.x
+    ve = np.sqrt(v*v-2*a*qe)
+    arug = ve*ve/(2*(1-qe))
+    original_transition = 2*qe/(v+ve)
+    times = np.arange(96)/FPS
+    vc = .6*v
+    # If reduced kinetic energy cannot reach the rug, stop on the hard floor.
+    if vc*vc <= 2*a*qe:
+        tt = np.minimum(times, vc/a)
+        q = vc*tt-a*tt*tt/2
+    else:
+        vec = np.sqrt(vc*vc-2*a*qe)
+        tec = 2*qe/(vc+vec)
+        dt = np.clip(times-tec, 0, vec/arug)
+        q = np.where(times < tec, vc*times-a*times*times/2,
+                     qe+vec*dt-arug*dt*dt/2)
+    # Invert the original path-distance law to obtain exact rolling texture
+    # phases. This is not a uniform playback-time slowdown: the edited ball
+    # stops sooner in space, with its original floor/rug resistance intact.
+    on_rug = q >= qe
+    if a > 1e-7:
+        ts = 2*q/(v+np.sqrt(np.maximum(v*v-2*a*q, 0)))
+    else:
+        ts = q/v
+    d = np.maximum(q-qe, 0)
+    rug_time = 2*d/(ve+np.sqrt(np.maximum(ve*ve-2*arug*d, 0)))
+    ts = np.where(on_rug, original_transition+rug_time, ts)
+    print('Original speed, rug transition, hard-floor resistance, perspective:',fit.x)
+    print('Rug resistance:', arug, 'edited final distance fraction:',q[-1])
+    print('Edited stopping pixel:',start+(end-start)*(1+depth)*q[-1]/(1+depth*q[-1]))
+    print('Source rolling phase at edited stop:',ts[-1]*FPS)
+    return ts*FPS
+
+
+def interpolate_crop(a, b, fraction, flow_ab, flow_ba):
+    """Bidirectional dense pixel advection, without any learned models."""
+    h,w = a.shape[:2]
+    yy,xx = np.indices((h,w), dtype=np.float32)
+    def inverse_warp(image, flow, amount):
+        # Invert x_destination = x_source + amount * flow(x_source).
+        mx, my = xx.copy(), yy.copy()
+        for _ in range(5):
+            fx = cv2.remap(flow[:,:,0],mx,my,cv2.INTER_LINEAR,borderMode=cv2.BORDER_REPLICATE)
+            fy = cv2.remap(flow[:,:,1],mx,my,cv2.INTER_LINEAR,borderMode=cv2.BORDER_REPLICATE)
+            mx = xx-amount*fx
+            my = yy-amount*fy
+        return cv2.remap(image,mx,my,cv2.INTER_LINEAR,borderMode=cv2.BORDER_REPLICATE).astype(np.float32)
+    i0 = inverse_warp(a,flow_ab,fraction)
+    i1 = inverse_warp(b,flow_ba,1-fraction)
+    return np.clip(i0*(1-fraction)+i1*fraction,0,255).astype(np.uint8)
+
+
+def main():
+    rgb = np.load(ROOT/'source_pixels.npz')['rgb']
+    high = load_video()
+    positions = measure_sphere(high)
+    phases = fit_motion(positions)
+    # Crop contains the entire moving ball and its shadow throughout this edit.
+    # Outside this crop all source imagery is unaltered.
+    H,W = high.shape[1:3]
+    x0,x1 = round(W*50/256),round(W*132/256)
+    y0,y1 = round(H*48/144),H
+    cache = {}
+    output = []
+    for index, phase in enumerate(phases):
+        if index == 0:
+            output.append(rgb[0].copy())
+            continue
+        j = int(np.floor(phase)); f = float(phase-j)
+        a, b = high[j], high[j+1]
+        if j not in cache:
+            ga = cv2.cvtColor(a[y0:y1,x0:x1],cv2.COLOR_RGB2GRAY)
+            gb = cv2.cvtColor(b[y0:y1,x0:x1],cv2.COLOR_RGB2GRAY)
+            args = (.5,5,41,5,7,1.5,0)
+            ab = cv2.calcOpticalFlowFarneback(ga,gb,None,*args)
+            ba = cv2.calcOpticalFlowFarneback(gb,ga,None,*args)
+            cache[j] = (ab,ba)
+        frame = np.clip(a.astype(float)*(1-f)+b.astype(float)*f,0,255).astype(np.uint8)
+        frame[y0:y1,x0:x1] = interpolate_crop(a[y0:y1,x0:x1], b[y0:y1,x0:x1],f,*cache[j])
+        image = Image.fromarray(frame).resize((256,144),Image.Resampling.LANCZOS)
+        output.append(np.asarray(image))
+    frames = np.asarray(output,dtype=np.uint8)
+    artifact_io.save(frames, ROOT)
+    # A conventional shareable MP4 is provided in addition to required RGB data.
+    video = cv2.VideoWriter(str(ROOT/'edited_video.mp4'),cv2.VideoWriter_fourcc(*'mp4v'),24,(256,144))
+    for frame in frames:
+        video.write(cv2.cvtColor(frame,cv2.COLOR_RGB2BGR))
+    video.release()
+    print('Saved',frames.shape,'RGB frames and edited_video.mp4')
+
+if __name__ == '__main__':
+    main()

\"\"\"Counterfactual rolling volleyball, reconstructed from the supplied video.

The floor path is fitted in perspective coordinates.  A piecewise rolling
resistance model accounts for the harder floor and the high-resistance rug.
Only the initial velocity changes (by 0.6); the surface resistances do not.
The source sphere imagery is advected to intermediate physical path samples,
so its texture, shading, contact shadow, and the unchanged room are retained.
\"\"\"
from pathlib import Path
import cv2
import numpy as np
from scipy.optimize import least_squares
from PIL import Image
import artifact_io

ROOT = Path(__file__).resolve().parent
FPS = 24

def load_video():
    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
    images = []
    while True:
        ok, image = cap.read()
        if not ok:
            break
        images.append(cv2.cvtColor(image, cv2.COLOR_BGR2RGB))
    cap.release()
    return np.asarray(images)


def measure_sphere(images):
    \"\"\"Fit circles to the highly distinctive yellow/blue volleyball panels.\"\"\"
    values = []
    h, w = images.shape[1:3]
    scale = w / 256
    yy, xx = np.indices((h, w))
    for image in images[:20]:
        r, g, b = image.astype(float).transpose(2, 0, 1)
        mask = (((b-r > 25) & (b-g > 5)) |
                ((r-b > 55) & (g-b > 40) & (g > 85)))
        mask &= (yy > 55*scale) & (xx > 64*scale) & (xx < 125*scale)
        n, labels, stats, centers = cv2.connectedComponentsWithStats(mask.astype(np.uint8))
        k = 1 + np.argmax(stats[1:, cv2.CC_STAT_AREA])
        cy, cx = np.where(labels == k)
        mask &= ((xx-cx.mean())**2+(yy-cy.mean())**2 < (13*scale)**2)
        y, x = np.where(mask)
        hull = cv2.convexHull(np.column_stack([x,y]).astype(np.int32)).squeeze()
        def residual(p):
            return np.hypot(hull[:,0]-p[0], hull[:,1]-p[1])-p[2]
        p0 = [(x.min()+x.max())/2, (y.min()+y.max())/2, (y.max()-y.min())/2]
        fit = least_squares(residual, p0, loss='soft_l1', f_scale=1)
        values.append(fit.x/scale)
    return np.asarray(values)


def fit_motion(observed):
    \"\"\"q is distance divided by the original stopping distance.

    Projected position = start + delta * (1+depth)*q/(1+depth*q).
    The rug transition and resistance are shared by factual and edited paths.
    \"\"\"
    t = np.arange(20)/FPS
    start, end = observed[0,:2], observed[-1,:2]

    def factual_q(times, v, qe, a):
        ve = np.sqrt(max(v*v-2*a*qe, 1e-9))
        te = 2*qe/(v+ve)
        arug = ve*ve/(2*(1-qe))
        dt = np.clip(times-te, 0, ve/arug)
        return np.where(times < te, v*times-a*times*times/2,
                        qe+ve*dt-arug*dt*dt/2)

    def residual(params):
        v, qe, a, depth, radius = params
        q = factual_q(t, v, qe, a)
        fraction = (1+depth)*q/(1+depth*q)
        centers = start+(end-start)*fraction[:,None]
        radii = radius/(1+depth*q)
        return np.r_[(centers-observed[:,:2]).ravel(), .5*(radii-observed[:,2])]

    fit = least_squares(residual, [1.85,.54,.01,.65,11.4],
                        bounds=([1,.2,0,.1,10],[4,.85,2,1.5,13]))
    v, qe, a, depth, radius = fit.x
    ve = np.sqrt(v*v-2*a*qe)
    arug = ve*ve/(2*(1-qe))
    original_transition = 2*qe/(v+ve)
    times = np.arange(96)/FPS
    vc = .6*v
    # If reduced kinetic energy cannot reach the rug, stop on the hard floor.
    if vc*vc <= 2*a*qe:
        tt = np.minimum(times, vc/a)
        q = vc*tt-a*tt*tt/2
    else:
        vec = np.sqrt(vc*vc-2*a*qe)
        tec = 2*qe/(vc+vec)
        dt = np.clip(times-tec, 0, vec/arug)
        q = np.where(times < tec, vc*times-a*times*times/2,
                     qe+vec*dt-arug*dt*dt/2)
    # Invert the original path-distance law to obtain exact rolling texture
    # phases. This is not a uniform playback-time slowdown: the edited ball
    # stops sooner in space, with its original floor/rug resistance intact.
    on_rug = q >= qe
    if a > 1e-7:
        ts = 2*q/(v+np.sqrt(np.maximum(v*v-2*a*q, 0)))
    else:
        ts = q/v
    d = np.maximum(q-qe, 0)
    rug_time = 2*d/(ve+np.sqrt(np.maximum(ve*ve-2*arug*d, 0)))
    ts = np.where(on_rug, original_transition+rug_time, ts)
    print('Original speed, rug transition, hard-floor resistance, perspective:',fit.x)
    print('Rug resistance:', arug, 'edited final distance fraction:',q[-1])
    print('Edited stopping pixel:',start+(end-start)*(1+depth)*q[-1]/(1+depth*q[-1]))
    print('Source rolling phase at edited stop:',ts[-1]*FPS)
    return ts*FPS


def interpolate_crop(a, b, fraction, flow_ab, flow_ba):
    \"\"\"Bidirectional dense pixel advection, without any learned models.\"\"\"
    h,w = a.shape[:2]
    yy,xx = np.indices((h,w), dtype=np.float32)
    def inverse_warp(image, flow, amount):
        # Invert x_destination = x_source + amount * flow(x_source).
        mx, my = xx.copy(), yy.copy()
        for _ in range(5):
            fx = cv2.remap(flow[:,:,0],mx,my,cv2.INTER_LINEAR,borderMode=cv2.BORDER_REPLICATE)
            fy = cv2.remap(flow[:,:,1],mx,my,cv2.INTER_LINEAR,borderMode=cv2.BORDER_REPLICATE)
            mx = xx-amount*fx
            my = yy-amount*fy
        return cv2.remap(image,mx,my,cv2.INTER_LINEAR,borderMode=cv2.BORDER_REPLICATE).astype(np.float32)
    i0 = inverse_warp(a,flow_ab,fraction)
    i1 = inverse_warp(b,flow_ba,1-fraction)
    return np.clip(i0*(1-fraction)+i1*fraction,0,255).astype(np.uint8)


def main():
    rgb = np.load(ROOT/'source_pixels.npz')['rgb']
    high = load_video()
    positions = measure_sphere(high)
    phases = fit_motion(positions)
    # Crop contains the entire moving ball and its shadow throughout this edit.
    # Outside this crop all source imagery is unaltered.
    H,W = high.shape[1:3]
    x0,x1 = round(W*50/256),round(W*132/256)
    y0,y1 = round(H*48/144),H
    cache = {}
    output = []
    for index, phase in enumerate(phases):
        if index == 0:
            output.append(rgb[0].copy())
            continue
        j = int(np.floor(phase)); f = float(phase-j)
        a, b = high[j], high[j+1]
        if j not in cache:
            ga = cv2.cvtColor(a[y0:y1,x0:x1],cv2.COLOR_RGB2GRAY)
            gb = cv2.cvtColor(b[y0:y1,x0:x1],cv2.COLOR_RGB2GRAY)
            args = (.5,5,41,5,7,1.5,0)
            ab = cv2.calcOpticalFlowFarneback(ga,gb,None,*args)
            ba = cv2.calcOpticalFlowFarneback(gb,ga,None,*args)
            cache[j] = (ab,ba)
        frame = np.clip(a.astype(float)*(1-f)+b.astype(float)*f,0,255).astype(np.uint8)
        frame[y0:y1,x0:x1] = interpolate_crop(a[y0:y1,x0:x1], b[y0:y1,x0:x1],f,*cache[j])
        image = Image.fromarray(frame).resize((256,144),Image.Resampling.LANCZOS)
        output.append(np.asarray(image))
    frames = np.asarray(output,dtype=np.uint8)
    artifact_io.save(frames, ROOT)
    # A conventional shareable MP4 is provided in addition to required RGB data.
    video = cv2.VideoWriter(str(ROOT/'edited_video.mp4'),cv2.VideoWriter_fourcc(*'mp4v'),24,(256,144))
    for frame in frames:
        video.write(cv2.cvtColor(frame,cv2.COLOR_RGB2BGR))
    video.release()
    print('Saved',frames.shape,'RGB frames and edited_video.mp4')

if __name__ == '__main__':
    main()
