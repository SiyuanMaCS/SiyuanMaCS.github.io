+"""Counterfactual dice drop, using only the source's pixels and ballistic geometry."""
+from pathlib import Path
+import cv2
+import numpy as np
+from PIL import Image
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+
+def read_source():
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    images = []
+    while True:
+        ok, bgr = cap.read()
+        if not ok:
+            break
+        images.append(cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB))
+    cap.release()
+    return np.asarray(images)
+
+
+def extract_red(frame):
+    r, g, b = np.moveaxis(frame.astype(float), -1, 0)
+    mask = ((r > 1.22*g) & (r > 1.16*b) & (r > 145) & (g > 65)).astype('uint8')
+    mask[:, :575] = 0
+    mask[:, 650:] = 0
+    mask[480:] = 0
+    _, labels, stats, _ = cv2.connectedComponentsWithStats(mask)
+    mask = (labels == 1 + np.argmax(stats[1:, 4])).astype('uint8')
+    contours, _ = cv2.findContours(mask, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
+    cv2.drawContours(mask, contours, -1, 1, -1)
+    # Include the antialiasing rim, but not a rectangle of the old background.
+    mask = cv2.dilate(mask, np.ones((3, 3), np.uint8))
+    mask = cv2.GaussianBlur(mask.astype(float), (3, 3), 0.55)
+    ys, xs = np.nonzero(mask > 0.01)
+    bounds = (xs.min()-1, ys.min()-1, xs.max()+2, ys.max()+2)
+    x0,y0,x1,y1 = bounds
+    rgba = np.dstack([frame[y0:y1,x0:x1], mask[y0:y1,x0:x1]*255]).astype('uint8')
+    return rgba, np.array([(x0+x1-1)/2, (y0+y1-1)/2])
+
+
+def paste_sprite(canvas, rgba, center, angle=0):
+    h,w = rgba.shape[:2]
+    # Premultiplied-alpha warping prevents dark outlines on subpixel motion.
+    alpha = rgba[:,:,3].astype(float)/255
+    premul = rgba[:,:,:3].astype(float)*alpha[:,:,None]
+    matrix = cv2.getRotationMatrix2D(((w-1)/2, (h-1)/2), angle, 1)
+    matrix[:,2] += np.asarray(center) - np.array([(w-1)/2, (h-1)/2])
+    h_out,w_out = canvas.shape[:2]
+    a = cv2.warpAffine(alpha, matrix, (w_out,h_out), flags=cv2.INTER_LINEAR)
+    p = cv2.warpAffine(premul, matrix, (w_out,h_out), flags=cv2.INTER_LINEAR)
+    canvas *= 1-a[:,:,None]
+    canvas += p
+
+
+def main():
+    source = read_source()
+    factual = np.load(ROOT/'source_pixels.npz')['rgb']
+    H,W = source.shape[1:3]
+    scale = W/256
+
+    # The first image supplies a genuinely empty landing area; a later frame
+    # supplies the tiny region concealed by the initially airborne dice.
+    background = source[0].copy()
+    background[:165, 570:715] = source[40,:165,570:715]
+
+    red_sprites = [extract_red(source[i]) for i in range(29)]
+    # White is unchanged. A clean source silhouette is used only to restore
+    # the few white pixels previously hidden by the slower factual red die.
+    white_frame = source[6]
+    white_mask = np.zeros((H,W), np.uint8)
+    white_poly = np.array([[662,285],[681,291],[695,302],[696,342],
+                           [687,350],[664,358],[643,347],[628,336],
+                           [627,306],[634,298]],np.int32)
+    cv2.fillPoly(white_mask,[white_poly],255)
+    white_mask = cv2.GaussianBlur(white_mask,(3,3),0.55)
+    white_rgba = np.dstack([white_frame[280:364,620:703],
+                            white_mask[280:364,620:703]])
+    white_template_center = np.array([661.,321.5])
+
+    # Fit the original free-fall, before the first table contact.
+    fitted_y = np.array([red_sprites[i][1][1]/scale for i in range(6)])
+    qa,qv,qy = np.polyfit(np.arange(6),fitted_y,2)
+    gravity = 2*qa
+    initial_velocity = 3*qv
+    floor_y = red_sprites[28][1][1]/scale
+    hit = (-initial_velocity+np.sqrt(initial_velocity**2+2*gravity*(floor_y-qy)))/gravity
+    # Original post-impact parabola gives restitution, with unchanged gravity.
+    rebound_fit = np.polyfit(np.arange(6,15),
+                             [red_sprites[i][1][1]/scale for i in range(6,15)],2)
+    original_hit = (-qv+np.sqrt(qv*qv+2*gravity*(floor_y-qy)))/gravity
+    original_incoming = qv+gravity*original_hit
+    original_outgoing = -(2*rebound_fit[0]*original_hit+rebound_fit[1])
+    restitution = np.clip(original_outgoing/original_incoming,0.44,0.52)
+    impact_speed = initial_velocity+gravity*hit
+    bounce_starts = []
+    start, speed = hit, impact_speed*restitution
+    while speed > 0.6:
+        bounce_starts.append((start,speed))
+        start += 2*speed/gravity
+        speed *= restitution
+    end = start
+
+    yy,xx = np.mgrid[:H,:W]
+    frames=[]
+    trajectory=[]
+    for i in range(96):
+        t=float(i)
+        if t < hit:
+            red_y=qy+initial_velocity*t+0.5*gravity*t*t
+        else:
+            red_y=floor_y
+            for st,sp in bounce_starts:
+                dt=t-st
+                if 0 <= dt <= 2*sp/gravity:
+                    red_y=floor_y-sp*dt+0.5*gravity*dt*dt
+                    break
+        trajectory.append(red_y)
+        # Keep the original video and its white die wherever possible.
+        canvas=source[i].astype(float)
+        original_rgba,original_center=red_sprites[min(i,28)]
+        rh,rw=original_rgba.shape[:2]
+        # Remove original red and its soft, nearby table shadow using the
+        # photographed empty table. The enlarged outline includes its rim.
+        removal=np.zeros((H,W),np.float32)
+        alpha=original_rgba[:,:,3].astype(float)/255
+        left=int(round(original_center[0]-(rw-1)/2))
+        top=int(round(original_center[1]-(rh-1)/2))
+        removal[top:top+rh,left:left+rw]=alpha
+        removal=cv2.dilate(removal,np.ones((7,7),np.uint8))
+        shadow_zone=((xx-618)/75)**2+((yy-466)/27)**2
+        removal=np.maximum(removal,np.clip((1.18-shadow_zone)*5,0,1))
+        removal=cv2.GaussianBlur(removal,(5,5),0.7)
+        canvas=canvas*(1-removal[:,:,None])+background*removal[:,:,None]
+
+        # If factual red overlapped white, reconstruct only that erased part.
+        r,g,b = np.moveaxis(source[i].astype(float),-1,0)
+        white_pixels=((r>185)&(g>183)&(b>170)).astype('uint8')
+        white_pixels[:,:625]=0
+        white_pixels[:,710:]=0
+        white_pixels[470:]=0
+        _,labs,stats,_=cv2.connectedComponentsWithStats(white_pixels)
+        j=1+np.argmax(stats[1:,4])
+        wx,wy,ww,wh=stats[j,:4]
+        if 7 <= i <= 13:
+            target_y=wy+34
+            tmp=background.astype(float)
+            template_local=np.array([(white_rgba.shape[1]-1)/2,(white_rgba.shape[0]-1)/2])
+            target=template_local+np.array([0.,target_y-white_template_center[1]])+np.array([620.,280.])
+            paste_sprite(tmp,white_rgba,target)
+            affected=removal*(xx>=624)*(yy>=wy-4)*(yy<=wy+wh+6)
+            canvas=canvas*(1-affected[:,:,None])+tmp*affected[:,:,None]
+
+        height=max(0.,floor_y-red_y)
+        red_x=red_sprites[0][1][0]/scale+0.65*(1-np.exp(-t/4))
+        # Contact shadow stays on the tabletop as the die moves vertically.
+        sx=(6.8+height*0.035)*scale
+        sy=(1.8+height*0.027)*scale
+        opacity=0.21*np.exp(-height/42)
+        shadow=np.exp(-0.5*(((xx-red_x*scale)/sx)**2+
+                               ((yy-(floor_y+6.6)*scale)/sy)**2))*opacity
+        canvas *= 1-shadow[:,:,None]
+
+        # Angular velocity is not multiplied: the edit concerns translation.
+        # Source silhouettes retain the original five/three pip faces.
+        if i==0:
+            sprite=red_sprites[0][0]
+        elif i < hit:
+            sprite=red_sprites[min(i,5)][0]
+        elif i < end:
+            sprite=red_sprites[6+min(i%9,8)][0]
+        else:
+            sprite=red_sprites[28][0]
+        paste_sprite(canvas,sprite,(red_x*scale,red_y*scale))
+        out=np.asarray(Image.fromarray(np.clip(canvas,0,255).astype('uint8')).resize(
+            (256,144),Image.Resampling.LANCZOS))
+        if i==0:
+            out=factual[0].copy()
+        frames.append(out)
+
+    frames=np.asarray(frames,dtype=np.uint8)
+    artifact_io.save(frames,ROOT)
+    print('Saved',frames.shape,'initial vertical speed',qv,'->',initial_velocity,
+          'first contact frame',hit,'restitution',restitution,'settles at',end)
+    print('Red center trajectory:',np.round(trajectory[:45],2))
+
+
+if __name__=='__main__':
+    main()
