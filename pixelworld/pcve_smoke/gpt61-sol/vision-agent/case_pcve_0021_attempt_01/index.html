+def sphere_grid(n):
+    y,x=np.mgrid[:n,:n].astype(float)
+    x=(x+.5-n/2)/(n/2);y=-(y+.5-n/2)/(n/2)
+    z=np.sqrt(np.maximum(0,1-x*x-y*y))
+    return np.stack((x,y,z),axis=-1),x*x+y*y<.975**2

+def trajectory(at,bt):
+    contact=22
+    pa=world(at[contact],RA);pb=world(bt[contact],RB)
+    # Contact normal is the center-to-center direction on the floor.
+    n=(pb-pa)/np.linalg.norm(pb-pa)
+    ts=np.arange(15,22)/24
+    history=np.array([world(at[i],RA) for i in range(15,22)])
+    vel=np.polyfit(ts,history,1)[0]
+    speed=np.dot(vel,n)
+    mass_a,mass_b=10.,1.
+    restitution=.25
+    impulse=(1+restitution)*speed/(1/mass_a+1/mass_b)
+    va=vel-impulse/mass_a*n
+    vb=impulse/mass_b*n
+    # Ordinary Coulomb rolling loss: mass does not multiply the deceleration.
+    decel=1.8
+    paths=[]
+    for start,v in [(pa,va),(pb,vb)]:
+        speed=np.linalg.norm(v);direction=v/speed
+        time=np.maximum(0,(np.arange(96)-contact)/24)
+        t=np.minimum(time,speed/decel)
+        travel=speed*t-.5*decel*t*t
+        paths.append((start+travel[:,None]*direction,travel,direction))

        quality[hidden]=0
        take=quality>score
        tex[take]=(rgb/shade(q)[:,:,None])[take];score[take]=quality[take]
    # Smooth base pigment in slivers that no provided view observes.
    tex=np.clip(tex,0,255)
    return tex.astype(np.float32)


def world(tr,R):
    cx,cy,_=tr
    z=F*(1-R)/(cy-HORIZON)
    return np.array([(cx-128)*z/F,z])


def project(pos,R):
    x,z=pos
    return np.array([128+F*x/z,HORIZON+F*(1-R)/z,F*R/z])


def trajectory(at,bt):
    contact=22
    pa=world(at[contact],RA);pb=world(bt[contact],RB)
    # Contact normal is the center-to-center direction on the floor.
    n=(pb-pa)/np.linalg.norm(pb-pa)
    ts=np.arange(15,22)/24
    history=np.array([world(at[i],RA) for i in range(15,22)])
    vel=np.polyfit(ts,history,1)[0]
    speed=np.dot(vel,n)
    mass_a,mass_b=10.,1.
    restitution=.25
    impulse=(1+restitution)*speed/(1/mass_a+1/mass_b)
    va=vel-impulse/mass_a*n
    vb=impulse/mass_b*n
    # Ordinary Coulomb rolling loss: mass does not multiply the deceleration.
    decel=1.8
    paths=[]
    for start,v in [(pa,va),(pb,vb)]:
        speed=np.linalg.norm(v);direction=v/speed
        time=np.maximum(0,(np.arange(96)-contact)/24)
        t=np.minimum(time,speed/decel)
        travel=speed*t-.5*decel*t*t
        paths.append((start+travel[:,None]*direction,travel,direction))
    print('incoming velocity',vel,'mass ratio',mass_a/mass_b,'post-impulse',va,vb)
    return paths


def draw_shadow(bg,tr):
    cx,cy,r=tr*5
    y0=max(0,int(cy+r-r*.7-5));y1=min(bg.shape[0],int(cy+r+r*.75+15))
    x0=max(0,int(cx-2*r-10));x1=min(bg.shape[1],int(cx+2*r+20))
    if y1<=y0 or x1<=x0:return
    yy,xx=np.mgrid[y0:y1,x0:x1]
    diffuse=np.exp(-2.2*(((xx-cx-5)/(1.35*r))**2+((yy-cy-r-4)/(.39*r))**2))
    contact=np.exp(-2.4*(((xx-cx)/(.8*r))**2+((yy-cy-r+2)/(.14*r+1))**2))
    alpha=.105*diffuse+.25*contact
    bg[y0:y1,x0:x1]*=(1-alpha[:,:,None]*np.array([.70,.83,1.]))


def render_ball(bg,tr,tex,R):
    cx,cy,r=tr*5
    if r<=0:return
    x0=max(0,int(cx-r-2));x1=min(1280,int(cx+r+3))
    y0=max(0,int(cy-r-2));y1=min(720,int(cy+r+3))
    if x1<=x0 or y1<=y0:return
    yy,xx=np.mgrid[y0:y1,x0:x1]
    x=(xx-cx)/r;y=-(yy-cy)/r
    d=np.sqrt(x*x+y*y)
    z=np.sqrt(np.maximum(0,1-d*d));p=np.stack((x,y,z),axis=-1)
    q=p@R
    lon=np.arctan2(q[:,:,0],q[:,:,2]);lat=np.arcsin(np.clip(q[:,:,1],-1,1))
    mx=((lon/(2*np.pi)+.5)*tex.shape[1]-.5).astype('float32')
    my=((.5-lat/np.pi)*tex.shape[0]-.5).astype('float32')
    rgb=cv2.remap(tex,mx,my,cv2.INTER_LINEAR,borderMode=cv2.BORDER_WRAP)
    rgb*=shade(p)[:,:,None]
    alpha=np.clip((1-d)*r+.5,0,1)
    bg[y0:y1,x0:x1]=bg[y0:y1,x0:x1]*(1-alpha[:,:,None])+rgb*alpha[:,:,None]


def main():
    original=np.load(ROOT/'source_pixels.npz')['rgb']
    src=decode()
    at,bt=tracks(src)
    np.savez(ROOT/'.tmp/motion_tracks.npz',star=at,soccer=bt)
    bg=recover_background(src,at,bt)
    Image.fromarray(bg).resize((256,144),Image.Resampling.LANCZOS).save(ROOT/'.tmp/background.png')
    poses=soccer_poses(src,bt)
    balltex=make_soccer_atlas(src,bt,poses)
    startex=make_star_atlas(src,at,bt)
    Image.fromarray(np.uint8(np.clip(balltex,0,255))).save(ROOT/'.tmp/soccer_atlas.png')
    Image.fromarray(np.uint8(np.clip(startex,0,255))).save(ROOT/'.tmp/star_atlas.png')
    paths=trajectory(at,bt)
    out=original.copy()
    # Same velocity and appearance at intervention; a mass change is invisible
    # until the finite-mass collision. Preserve the source exactly up to contact.
    for i in range(23,96):
        scene=bg.astype(float).copy()
        (ap,ad,adir),(bp,bd,bdir)=paths
        atr=project(ap[i],RA);btr=project(bp[i],RB)
        # Calibrated display scale keeps physical and factual silhouette matched.
        atr[2]*=at[22,2]/project(ap[22],RA)[2]
        btr[2]*=bt[22,2]/project(bp[22],RB)[2]
        # Actual captured rolling-axis direction; total angular displacement
        # follows the longer counterfactual path rather than a time freeze.
        aR=rot([ad[i]/RA,0,-.20*ad[i]/RA])
        # Use the observed initial roll orientation, continued geometrically.
        observed=cv2.Rodrigues(poses[30])[0].ravel()
        unit=observed/max(1e-8,np.linalg.norm(observed))
        if unit[0]<0:unit=-unit
        bR=rot(unit*(bd[i]/RB)*.93)
        draw_shadow(scene,atr);draw_shadow(scene,btr)
        render_ball(scene,atr,startex,aR)
        render_ball(scene,btr,balltex,bR)
        frame=Image.fromarray(np.uint8(np.clip(scene,0,255))).resize((256,144),Image.Resampling.LANCZOS)
        out[i]=np.array(frame)
    artifact_io.save(out,ROOT)
    # Convenient conventional encoding alongside the required lossless artifact.
    writer=cv2.VideoWriter(str(ROOT/'edited_video.mp4'),cv2.VideoWriter_fourcc(*'mp4v'),24,(256,144))
    for frame in out:writer.write(cv2.cvtColor(frame,cv2.COLOR_RGB2BGR))
    writer.release()
    print('Saved',out.shape,'at 24 fps')

if __name__=='__main__':main()
