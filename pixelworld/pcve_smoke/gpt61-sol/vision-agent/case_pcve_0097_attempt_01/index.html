        bg=cv2.inpaint(bg,missing.astype(np.uint8)*255,2,cv2.INPAINT_TELEA)
    return bg

def get_hi(cap,index):
    cap.set(cv2.CAP_PROP_POS_FRAMES,index)
    ok,a=cap.read()
    if not ok: raise RuntimeError('Cannot read source video')
    return cv2.cvtColor(a,cv2.COLOR_BGR2RGB)

def sprite(hi,approx,color):
    \"\"\"High-resolution photographic sphere matte, independent of its shadow.\"\"\"
    h,w=hi.shape[:2]; scale=w/256
    x,y=approx*scale
    yy,xx=np.mgrid[:h,:w]
    roi=(xx-x)**2+(yy-y)**2<(9.6*scale)**2
    a=hi.astype(float)
    if color=='red':
        chroma=(a[:,:,0]>1.4*a[:,:,1])&(a[:,:,0]>1.4*a[:,:,2])&(a[:,:,0]>100)&((a[:,:,1]-a[:,:,2])<50)
    else:
        chroma=(a[:,:,2]>1.12*a[:,:,0])&(a[:,:,2]>1.03*a[:,:,1])&(a[:,:,2]>65)
    mask=(roi&chroma).astype(np.uint8)
    contours,_=cv2.findContours(mask,cv2.RETR_EXTERNAL,cv2.CHAIN_APPROX_NONE)
    contour=max(contours,key=cv2.contourArea)
    (cx,cy),r=cv2.minEnclosingCircle(contour)
    # The hue mask reaches the limb except for subpixel filtering.
    r+=0.3
    half=int(np.ceil(r+3)); x0=int(np.floor(cx))-half; y0=int(np.floor(cy))-half
    patch=hi[y0:y0+2*half+1,x0:x0+2*half+1].astype(np.float32)
    py,px=np.mgrid[:patch.shape[0],:patch.shape[1]]
    radius=np.hypot(px+x0-cx,py+y0-cy)
    alpha=np.clip((r-radius+0.4)/0.85,0,1).astype(np.float32)
    # Bright environmental shading and fine ball texture remain photographic.
    return patch,alpha,np.array([cx-x0,cy-y0]),r/scale

def paste_ball(frame,item,pos,radius):
    patch,alpha,c,r=item
    sc=radius/(r*5.)
    # Source video is five times the requested resolution.
    mat=np.array([[sc,0,pos[0]-sc*c[0]],[0,sc,pos[1]-sc*c[1]]],np.float32)
    premult=patch*alpha[:,:,None]
    rgb=cv2.warpAffine(premult,mat,(256,144),flags=cv2.INTER_LINEAR,borderMode=cv2.BORDER_CONSTANT)
    al=cv2.warpAffine(alpha,mat,(256,144),flags=cv2.INTER_LINEAR,borderMode=cv2.BORDER_CONSTANT)
    return frame*(1-al[:,:,None])+rgb

def contact_shadow(frame,pos,radius,air_height=0):
    yy,xx=np.mgrid[:144,:256]
    x,y=pos
    contact=np.exp(-(((xx-x-0.7)/(radius*.64))**2+((yy-y-radius*.91)/(radius*.26))**2)*2)
    diffuse=np.exp(-(((xx-x-1)/(radius*.93))**2+((yy-y-radius*.93)/(radius*.40))**2)*2)
    opacity=(.43*contact+.13*diffuse)*np.exp(-air_height/6)
    return frame*(1-opacity[:,:,None])

def simulate(red,blue):
    # Coordinates use source frames as time units. Positive x is upslope.
    ids=np.arange(6,21)
    poly=np.polyfit(ids,red[ids,0],2)
    factual_a=2*poly[0]
    drag=float(np.polyfit(np.arange(23,30),red[23:30,0],2)[0]*2)
    drag=np.clip(drag,.50,.72)
    ramp_a=factual_a+2*drag
    line=np.polyfit(red[8:21,0],red[8:21,1],1)
    slope,intercept=line
    # The center passes the low lip at this point; fall height is measured
    # from the observed sphere path immediately past the edge.
    lip=57.0
    ground_y=87.8
    g=4.8
    restitution=.30
    p=red[17].copy(); v=float(2*poly[0]*17+poly[1])
    bp=blue[17].copy(); bv=0.
    mode='ramp'; vy=0.; collided=False
    states={17:(p.copy(),bp.copy(),mode,0.)}
    dt=1/120
    for i in range(18,96):
        for _ in range(120):
            if mode=='ramp':
                nv=min(0.,v+ramp_a*dt)
                nx=p[0]+(v+nv)*.5*dt
                p[0]=nx; p[1]=slope*nx+intercept; v=nv
                if nx<=lip:
                    mode='air';vy=slope*v
            elif mode=='air':
                p[0]+=v*dt
                p[1]+=vy*dt+.5*g*dt*dt;vy+=g*dt
                if p[1]>=ground_y:
                    p[1]=ground_y;mode='table';vy=0
            else:
                nv=min(0.,v+3*drag*dt)
                p[0]+=.5*(v+nv)*dt;v=nv
            # Two equal spheres; matching the source's mild inelastic impact.
            if mode=='table' and not collided and p[0]-bp[0]<=16.0:
                p[0]=bp[0]+16.
                incoming=v
                v=(1-restitution)*.5*incoming
                bv=(1+restitution)*.5*incoming
                collided=True
            if collided:
                nbv=min(0.,bv+drag*dt)
                dx=(bv+nbv)*.5*dt
                bp[0]+=dx;bp[1]-=.047*dx;bv=nbv
        states[i]=(p.copy(),bp.copy(),mode,max(0.,ground_y-p[1]) if mode=='air' else 0.)
    print('inferred baseline drag:',drag,'ramp acceleration:',factual_a,'counterfactual:',ramp_a)
    print('final red/blue:',p,bp)
    return states

def generate():
    src=np.load(ROOT/'source_pixels.npz')['rgb']
    red=np.array([center(f,'red') for f in src]);blue=np.array([center(f,'blue') for f in src])
    bg=build_background(src,red,blue)
    cap=cv2.VideoCapture(str(ROOT/'source_video.mp4'))
    red_sprites={j:sprite(get_hi(cap,j),red[j],'red') for j in [17,18,19,20,21,22,24]}
    blue_sprite=sprite(get_hi(cap,0),blue[0],'blue')
    cap.release()
    states=simulate(red,blue)
    out=src.copy()
    for i in range(18,96):
        rp,bp,mode,height=states[i]
        f=src[i].astype(np.float32)
        remove=footprint(*red[i])|footprint(*blue[i],r=9.8)
        # Feather replacement only at the outside of removed ball footprints.
        f[remove]=bg[remove]
        if mode=='ramp':
            j=min([17,18,19,20],key=lambda k:abs(red[k,0]-rp[0]))
            radius=np.interp(rp[0],red[17:21,0][::-1],[red_sprites[k][3] for k in range(17,21)][::-1])
        elif mode=='air':
            j=21;radius=7.9
        else:
            j=22;radius=8.0
        f=contact_shadow(f,rp,radius,height)
        f=paste_ball(f,red_sprites[j],rp,radius)
        # The blue ball is affected only at the later, weaker collision.
        if np.linalg.norm(bp-blue[0])<.15:
            mask=footprint(*blue[0],r=9.8)
            f[mask]=src[0,mask]
        else:
            f=contact_shadow(f,bp,8.0)
            f=paste_ball(f,blue_sprite,bp,8.0)
        out[i]=np.clip(np.rint(f),0,255).astype(np.uint8)
    # Frame 18 is the instantaneous change of coefficient: position continuous.
    assert np.array_equal(out[:18],src[:18])
    artifact_io.save(out,ROOT)
    Image.fromarray(bg).save(ROOT/'.tmp/background.png')
    sheet=Image.new('RGB',(4*512,3*310),(25,28,35));draw=ImageDraw.Draw(sheet)
    for k,i in enumerate([16,17,18,19,20,21,22,23,24,29,40,95]):
        x=(k%4)*512;y=(k//4)*310
        sheet.paste(Image.fromarray(out[i]).resize((512,288)),(x,y+20));draw.text((x+4,y+3),f'frame {i+1}',fill='white')
    sheet.save(ROOT/'.tmp/candidate_detail.png')

if __name__=='__main__':generate()
