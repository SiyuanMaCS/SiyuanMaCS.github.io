+def sprite(rgb, mask, anchor):
+    yy,xx=np.where(mask>0.001)

+def paste(canvas,spr,at,scale):

+def gaussian_shadow(canvas,x,y,rx,ry,strength):

\"\"\"Source-pixel counterfactual for a tenfold increase in rolling-ball mass.
No external assets: static plate and textured rigid bodies are sampled from the
supplied clip, and the postimpact trajectory uses a 1D impulse/friction model.

from pathlib import Path
import cv2

ROOT = Path(__file__).resolve().parent

def read_source():
    cap = cv2.VideoCapture(str(ROOT/'source_video.mp4'))
    a=[]
    while True:
        ok,bgr=cap.read()
        if not ok: break
        a.append(cv2.cvtColor(bgr,cv2.COLOR_BGR2RGB))
    cap.release()
    return np.array(a)

def ball_geometry(a, low, frame):
    m=(low[:,:,0].astype(float)>low[:,:,1]*1.6)&(low[:,:,1].astype(float)>low[:,:,2]*1.65)&(low[:,:,0]>130)
    n,l,s,c=cv2.connectedComponentsWithStats(m.astype(np.uint8))
    x,y=c[1+s[1:,cv2.CC_STAT_AREA].argmax()]*5
    Y,X=np.mgrid[:a.shape[0],:a.shape[1]]

def plate(source,geometries):
    h,w=source.shape[1:3];Y,X=np.mgrid[:h,:w]
    best=np.full((h,w),-1.,np.float32);out=source[0].copy()
    anchors=np.array([[0,686,405,143,132],[12,686,405,143,132],[13,700,414,155,145],[16,791,416,146,139],[20,884,439,148,143],[24,963,451,151,143],[28,1015,452,152,134],[32,1045,455,158,140],[95,1045,455,158,140]])
    for i,a in enumerate(source):
        x,y,aa,bb,ang=geometries[i]
        bad=(X-x)**2/(aa/2+15)**2+(Y-y)**2/(bb/2+15)**2<1
        cx,cy,rx,ry=[np.interp(i,anchors[:,0],anchors[:,j]) for j in range(1,5)]
        bad |= (abs(X-cx)<rx+15)&(abs(Y-cy)<ry+15)
        lum=cv2.cvtColor(a,cv2.COLOR_RGB2GRAY).astype(np.float32);lum[bad]=-1
        take=lum>best;out[take]=a[take];best[take]=lum[take]
    # Outside the moving-body corridor keep the original plate byte-identical.
    region=np.zeros((h,w),np.float32);region[200:675,:1250]=1
    region=cv2.GaussianBlur(region,(0,0),15)
    out=source[0]*(1-region[:,:,None])+out*region[:,:,None]
    return out.astype(np.float32)


def sprite(rgb, mask, anchor):
    yy,xx=np.where(mask>0.001)
    x0,x1=max(0,xx.min()-3),min(rgb.shape[1],xx.max()+4)
    y0,y1=max(0,yy.min()-3),min(rgb.shape[0],yy.max()+4)
    return (rgb[y0:y1,x0:x1].astype(np.float32),mask[y0:y1,x0:x1],(anchor[0]-x0,anchor[1]-y0))


def paste(canvas,spr,at,scale):
    rgb,alpha,anchor=spr
    matrix=np.array([[scale,0,at[0]-scale*anchor[0]],[0,scale,at[1]-scale*anchor[1]]],np.float32)
    # Premultiplied resampling avoids dark fringes around round silhouettes.
    size=(canvas.shape[1],canvas.shape[0])
    a=cv2.warpAffine(alpha,matrix,size,flags=cv2.INTER_LINEAR)
    p=cv2.warpAffine(rgb*alpha[:,:,None],matrix,size,flags=cv2.INTER_LINEAR)
    canvas*=1-a[:,:,None];canvas+=p


def gaussian_shadow(canvas,x,y,rx,ry,strength):
    h,w=canvas.shape[:2]
    x0,x1=max(0,int(x-rx*3)),min(w,int(x+rx*3)+1)
    y0,y1=max(0,int(y-ry*3)),min(h,int(y+ry*3)+1)
    if x1<=x0 or y1<=y0:return
    yy,xx=np.mgrid[y0:y1,x0:x1]
    a=strength*np.exp(-.5*((xx-x)/rx)**2-.5*((yy-y)/ry)**2)
    canvas[y0:y1,x0:x1]*=(1-a[:,:,None])


def generate():
    source=read_source();low=np.load(ROOT/'source_pixels.npz')['rgb']
    H,W=source.shape[1:3]
    geo=[ball_geometry(a,low[i],i) for i,a in enumerate(source)]
    bg=plate(source,geo)
    ball_sprites={}
    for i in list(range(1,11))+list(range(15,96)):
        x,y,aa,bb,ang=geo[i]
        mask=np.zeros((H,W),np.uint8)
        cv2.ellipse(mask,((x,y),(aa,bb),ang),255,-1,lineType=cv2.LINE_AA)
        ball_sprites[i]=sprite(source[i],mask.astype(np.float32)/255,(x,y))
    box_sprites={}
    for i,points in POLYS.items():
        p=np.array(points,np.int32);mask=np.zeros((H,W),np.uint8)
        cv2.fillPoly(mask,[p],255,lineType=cv2.LINE_AA)
        anchor=(p[:,0].min(),p[:,1].max())
        box_sprites[i]=(sprite(source[i],mask.astype(np.float32)/255,anchor),anchor)

    # At the same contact time, m_ball/m_crate changes from about 1 to 10.
    # Restitution is estimated from the supplied clip's two separation speeds.
    mass=10.;restitution=.82;incoming=35.
    vb=(mass-restitution)/(mass+1)*incoming
    vc=(1+restitution)*mass/(mass+1)*incoming
    # Coulomb sliding deceleration is independent of crate mass; the rolling
    # ball retains most of its momentum rather than slowing to factual speed.
    ball_drag=.045;crate_drag=1.55
    contact=12
    result=[]
    for frame in range(96):
        if frame<=contact:
            result.append(low[frame].copy());continue
        t=frame-contact
        ball_dx=vb*t-.5*ball_drag*t*t
        crate_t=min(t,vc/crate_drag)
        crate_dx=vc*crate_t-.5*crate_drag*crate_t*crate_t
        canvas=bg.copy()
        # Ground-plane projection: the objects travel diagonally rightwards
        # and slightly towards the camera, as in the factual trajectory.
        ball_x=492+ball_dx
        radius=83*(1+ball_dx*.00036)
        ball_y=368+.215*ball_dx
        box_phase=min(32,12+int(round(t*1.85)))
        choices=np.array(list(POLYS))
        box_index=int(choices[np.argmin(abs(choices-box_phase))])
        box_s,anchor=box_sprites[box_index]
        box_scale=(1+crate_dx*.00027)/(1+max(0,anchor[0]-546)*.00027)
        # Change in visible crate yaw is retained from its original impact.
        box_x=546+crate_dx
        box_bottom=517+.168*crate_dx
        box_width=280*(1+crate_dx*.00027)
        gaussian_shadow(canvas,box_x+box_width*.52+14,box_bottom-13,box_width*.39,19*box_scale,.22)
        gaussian_shadow(canvas,box_x+box_width*.50,box_bottom-5,box_width*.37,7*box_scale,.30)
        gaussian_shadow(canvas,ball_x+radius*.35,ball_y+radius+8,radius*.88,radius*.22,.26)
        gaussian_shadow(canvas,ball_x+2,ball_y+radius-2,radius*.65,radius*.060,.42)
        # Advance the source's rolling texture by distance, not by the old
        # slowed trajectory. A full source rotation spans about 150px.
        phase=15+ball_dx/3.35
        texture_idx=15+int(phase-15)%81
        spr=ball_sprites[texture_idx]
        sr=np.mean(geo[texture_idx][2:4])/2
        paste(canvas,spr,(ball_x,ball_y),radius/sr)
        paste(canvas,box_s,(box_x,box_bottom),box_scale)
        rgb=np.clip(canvas,0,255).astype(np.uint8)
        result.append(np.array(Image.fromarray(rgb).resize((256,144),Image.Resampling.LANCZOS)))
    frames=np.array(result,dtype=np.uint8)
    artifact_io.save(frames,ROOT)
    # Ordinary MP4 is an additional convenient preview; pixels.npz remains the
    # authoritative lossless artifact written by artifact_io.save.
    writer=cv2.VideoWriter(str(ROOT/'edited_video.mp4'),cv2.VideoWriter_fourcc(*'mp4v'),24,(256,144))
    for f in frames:writer.write(cv2.cvtColor(f,cv2.COLOR_RGB2BGR))
    writer.release()
    print('Saved',frames.shape,'RGB frames and edited_video.mp4')
