+"""Counterfactual increased car/contact friction, using source-pixel sprites.
+
+The measured uphill motion supplies initial velocity and the original sum of
+incline gravity and friction.  Gravity itself is unchanged; only the contact
+resistance is multiplied by four.  Static friction then holds the car at rest.
+No external rendering, assets, or learned models are used.
+"""
+from pathlib import Path
+import cv2
+import numpy as np
+from PIL import Image
+from scipy.ndimage import binary_fill_holes, gaussian_filter
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+
+def load_high_resolution():
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    result = []
+    for _ in range(24):
+        ok, frame = cap.read()
+        if not ok:
+            raise RuntimeError('Incomplete source video')
+        result.append(cv2.cvtColor(frame, cv2.COLOR_BGR2RGB))
+    cap.release()
+    return np.asarray(result)
+
+
+def main():
+    low = np.load(ROOT / 'source_pixels.npz')['rgb']
+    src = load_high_resolution()
+    h, w = src.shape[1:3]
+    scale = w / 256.0
+    # Ground/ramp are static. The original initial position is revealed in
+    # later source frames. Feather only on already unobstructed background.
+    yy, xx = np.mgrid[:h, :w]
+    edge_distance = np.minimum.reduce([xx-742, 1007-xx, yy-480, 625-yy])
+    patch = np.clip(edge_distance / 15.0, 0, 1)[..., None]
+    clean = src[0].astype(np.float32) * (1-patch) + src[23] * patch
+
+    # Near-side front wheel centres measured from the allowed source frames.
+    cx = np.array([164.2,149.6,135.8,122.8,110.8,99.6,89.4,80.2,71.8,64.2,57.6])
+    cy = np.array([109.6,104.4,99.4,94.8,90.4,86.4,82.8,79.4,76.4,73.6,71.2])
+    t = np.arange(len(cx))/24.0
+    fit = np.polyfit(t, cx, 2)
+    original_acceleration = 2*fit[0]
+    v0 = -fit[1]
+    slope = np.polyfit(cx[:9], cy[:9], 1)[0]
+    # Effective projected gravitational acceleration estimated from the
+    # airborne source segment; this is not multiplied with the friction.
+    projected_g = 1150.0
+    gravity_along_x = projected_g * slope / (1+slope*slope)
+    old_friction_x = max(0.0, original_acceleration-gravity_along_x)
+    new_friction_x = 4.0 * old_friction_x
+    deceleration = gravity_along_x + new_friction_x
+    stop_time = v0 / deceleration
+    assert new_friction_x > gravity_along_x  # held by static contact friction
+
+    sprites = []
+    for k in range(9):
+        ix = int(round(cx[k]*scale-65))
+        iy = int(round(cy[k]*scale-55))
+        sw, sh = 240, 130
+        rgb = src[k, iy:iy+sh, ix:ix+sw].astype(np.float32)
+        base = clean[iy:iy+sh, ix:ix+sw]
+        changed = (np.max(np.abs(rgb-base), axis=2)>20).astype(np.uint8)
+        changed = cv2.morphologyEx(changed, cv2.MORPH_CLOSE, np.ones((5,5),np.uint8))
+        count, labels, stats, centers = cv2.connectedComponentsWithStats(changed)
+        ids = np.flatnonzero(stats[:,4]>120)
+        ids = ids[ids!=0]
+        mask = binary_fill_holes(np.isin(labels,ids)).astype(np.uint8)
+        # Similar-colour spokes and windows are interior car pixels even
+        # when their colour happens to resemble background pixels.
+        front = (int(round(cx[k]*scale-ix)), int(round(cy[k]*scale-iy)))
+        cv2.circle(mask, front, 18, 1, -1)
+        cv2.circle(mask, (front[0]+94,front[1]+34), 20, 1, -1)
+        sy,sx=np.mgrid[iy:iy+sh,ix:ix+sw]
+        # Preserve the near rail's occlusion; it remains stationary.
+        mask[sy > 0.359*sx + 271.2] = 0
+        alpha=gaussian_filter(mask.astype(np.float32), .45)
+        sprites.append((rgb,alpha,ix,iy))
+
+    result=[]
+    for i in range(96):
+        if i==0:
+            result.append(low[0].copy())
+            continue
+        time=min(i/24.0,stop_time)
+        target_x=cx[0]-v0*time+.5*deceleration*time*time
+        target_y=cy[0]+slope*(target_x-cx[0])
+        # Pixel sprites sample the same pose while the new trajectory moves
+        # through it more slowly, then stops before the ramp's upper edge.
+        j=int(np.searchsorted(-cx,-target_x)-1)
+        j=max(0,min(j,7))
+        f=float(np.clip((cx[j]-target_x)/(cx[j]-cx[j+1]),0,1))
+        layers=[]
+        alphas=[]
+        for k in [j,j+1]:
+            rgb,alpha,ix,iy=sprites[k]
+            dx=(target_x-cx[k])*scale
+            dy=(target_y-cy[k])*scale
+            transform=np.array([[1,0,ix+dx],[0,1,iy+dy]],np.float32)
+            a=cv2.warpAffine(alpha,transform,(w,h),flags=cv2.INTER_LINEAR)
+            col=cv2.warpAffine(rgb*alpha[...,None],transform,(w,h),flags=cv2.INTER_LINEAR)
+            alphas.append(a)
+            layers.append(col)
+        a=(1-f)*alphas[0]+f*alphas[1]
+        col=(1-f)*layers[0]+f*layers[1]
+        output=clean*(1-a[...,None])+col
+        # Restore the near-side wood face after compositing, so the car is
+        # consistently behind the rail and its contact never floats above it.
+        rail=np.zeros((h,w),np.uint8)
+        cv2.fillPoly(rail,[np.array([(308,381),(991,626),(979,671),(295,431)],np.int32)],1)
+        output[rail.astype(bool)]=clean[rail.astype(bool)]
+        output=np.uint8(np.clip(output,0,255))
+        frame=np.asarray(Image.fromarray(output).resize((256,144),Image.Resampling.LANCZOS))
+        result.append(frame)
+    frames=np.stack(result).astype(np.uint8)
+    artifact_io.save(frames, ROOT)
+    print(f'Saved {len(frames)} frames. Friction x4: stop at {stop_time:.3f}s, front wheel x={target_x:.2f}.')
+
+
+if __name__=='__main__':
+    main()

    projected_g = 1150.0
    gravity_along_x = projected_g * slope / (1+slope*slope)
    old_friction_x = max(0.0, original_acceleration-gravity_along_x)
    new_friction_x = 4.0 * old_friction_x
    deceleration = gravity_along_x + new_friction_x
    stop_time = v0 / deceleration
    assert new_friction_x > gravity_along_x  # held by static contact friction

    sprites = []
    for k in range(9):
        ix = int(round(cx[k]*scale-65))
        iy = int(round(cy[k]*scale-55))
        sw, sh = 240, 130
        rgb = src[k, iy:iy+sh, ix:ix+sw].astype(np.float32)
        base = clean[iy:iy+sh, ix:ix+sw]
        changed = (np.max(np.abs(rgb-base), axis=2)>20).astype(np.uint8)
        changed = cv2.morphologyEx(changed, cv2.MORPH_CLOSE, np.ones((5,5),np.uint8))
        count, labels, stats, centers = cv2.connectedComponentsWithStats(changed)
        ids = np.flatnonzero(stats[:,4]>120)
        ids = ids[ids"'!=0]
        mask = binary_fill_holes(np.isin(labels,ids)).astype(np.uint8)
        # Similar-colour spokes and windows are interior car pixels even
        # when their colour happens to resemble background pixels.
        front = (int(round(cx[k]*scale-ix)), int(round(cy[k]*scale-iy)))
        cv2.circle(mask, front, 18, 1, -1)
        cv2.circle(mask, (front[0]+94,front[1]+34), 20, 1, -1)
        sy,sx=np.mgrid[iy:iy+sh,ix:ix+sw]
        # Preserve the near rail'"'s occlusion; it remains stationary.
        mask[sy > 0.359*sx + 271.2] = 0
        alpha=gaussian_filter(mask.astype(np.float32), .45)
        sprites.append((rgb,alpha,ix,iy))

    result=[]
    for i in range(96):
        if i==0:
            result.append(low[0].copy())
            continue
        time=min(i/24.0,stop_time)
        target_x=cx[0]-v0*time+.5*deceleration*time*time
        target_y=cy[0]+slope*(target_x-cx[0])
        # Pixel sprites sample the same pose while the new trajectory moves
        # through it more slowly, then stops before the ramp's upper edge.
        j=int(np.searchsorted(-cx,-target_x)-1)
        j=max(0,min(j,7))
        f=float(np.clip((cx[j]-target_x)/(cx[j]-cx[j+1]),0,1))
        layers=[]
        alphas=[]
        for k in [j,j+1]:
            rgb,alpha,ix,iy=sprites[k]
            dx=(target_x-cx[k])*scale
            dy=(target_y-cy[k])*scale
            transform=np.array([[1,0,ix+dx],[0,1,iy+dy]],np.float32)
            a=cv2.warpAffine(alpha,transform,(w,h),flags=cv2.INTER_LINEAR)
            col=cv2.warpAffine(rgb*alpha[...,None],transform,(w,h),flags=cv2.INTER_LINEAR)
            alphas.append(a)
            layers.append(col)
        a=(1-f)*alphas[0]+f*alphas[1]
        col=(1-f)*layers[0]+f*layers[1]
        output=clean*(1-a[...,None])+col
        # Restore the near-side wood face after compositing, so the car is
        # consistently behind the rail and its contact never floats above it.
        rail=np.zeros((h,w),np.uint8)
        cv2.fillPoly(rail,[np.array([(308,381),(991,626),(979,671),(295,431)],np.int32)],1)
        output[rail.astype(bool)]=clean[rail.astype(bool)]
        output=np.uint8(np.clip(output,0,255))
        frame=np.asarray(Image.fromarray(output).resize((256,144),Image.Resampling.LANCZOS))
        result.append(frame)
    frames=np.stack(result).astype(np.uint8)
    artifact_io.save(frames, ROOT)
    print(f'Saved {len(frames)} frames. Friction x4: stop at {stop_time:.3f}s, front wheel x={target_x:.2f}.')


if __name__=='__main__':
    main()
