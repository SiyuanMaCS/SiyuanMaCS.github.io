Planar rigid-disk impulses are integrated numerically.  The camera, table and
mallet textures are taken from the source, not synthesized by another model.

from pathlib import Path
import cv2
import numpy as np
from PIL import Image, ImageDraw

ROOT=Path(__file__).resolve().parent

def read_source():
    low=np.load(ROOT/'source_pixels.npz')['rgb']
    cap=cv2.VideoCapture(str(ROOT/'source_video.mp4'))
    high=[]
    while True:
        ok,im=cap.read()
        if not ok: break
        high.append(cv2.cvtColor(im,cv2.COLOR_BGR2RGB))

def track(src):
    # Source color segmentation, restricted to the playable surface.
    yy,xx=np.mgrid[:720,:1280]
    roi=(xx>300)&(xx<910)&(yy>250)&(yy<510)
    paths=[]
    for color in ['blue','red']:
        pts=[]
        for im in src:
            r,g,b=im.astype(np.float32).transpose(2,0,1)
            if color=='blue': m=roi&(b>r*1.45)&(b>g*1.1)&(b>70)
            else: m=roi&(r>g*1.35)&(r>b*1.22)&(r>150)
            _,lab,stats,centers=cv2.connectedComponentsWithStats(m.astype(np.uint8))
            k=1+np.argmax(stats[1:,4]);pts.append(centers[k])
        paths.append(np.asarray(pts))
    # White is tracked with its first-frame local texture.
    template=cv2.cvtColor(src[0,380:429,492:574],cv2.COLOR_RGB2GRAY)
    prev=(492,380);pts=[]
    for im in src:
        x,y=prev; x0=max(250,x-20);y0=max(350,y-20)
        x1=min(600,x+102);y1=min(540,y+69)
        score=cv2.matchTemplate(cv2.cvtColor(im[y0:y1,x0:x1],cv2.COLOR_RGB2GRAY),template,cv2.TM_CCOEFF_NORMED)
        _,_,_,(tx,ty)=cv2.minMaxLoc(score)
        prev=(x0+tx,y0+ty);pts.append([prev[0]+41,prev[1]+27])
    paths.append(np.array(pts,float))
    return np.stack(paths,axis=1)

def object_region(shape, center, scale=1., reflection=True):
    mask=np.zeros(shape[:2],np.uint8)
    x,y=center; rx=33*scale
    # Contains actual handle, base, the faint contact shadow, and reflection.
    pts=np.round([[x-rx,y-27*scale],[x+rx,y-27*scale],
                  [x+rx+4*scale,y+20*scale],[x-rx-2*scale,y+20*scale]]).astype(np.int32)
    cv2.fillConvexPoly(mask,pts,255)
    if reflection:
        cv2.ellipse(mask,(round(x+3*scale),round(y+28*scale)),
                    (round(32*scale),round(34*scale)),0,0,360,255,-1)
    return mask

def clean_plate(src,paths):
    # Recover exposed table pixels over time; conceal only pixels that were
    # covered in every frame with classical local image inpainting.
    ids=[0,3,6,9,12,14,16,18,20,23,26,29,32,36,42,48,55,65,80,95]
    x0,x1,y0,y1=285,905,248,540
    data=src[ids,y0:y1,x0:x1].astype(np.float32)
    valid=[]
    for j,i in enumerate(ids):
        mask=np.zeros((720,1280),np.uint8)
        for k in range(3):
            c=paths[i,k];sc=(33 if k==0 else 36 if k==1 else 42)/(33 if k==0 else 36 if k==1 else 42)
            # Perspective growth toward near end.
            sc=1+(paths[0,k,0]-c[0])*0.00052
            mask=np.maximum(mask,object_region(mask.shape,c,sc))
        valid.append(mask[y0:y1,x0:x1]==0)
    valid=np.array(valid)
    # Correct insignificant codec/light offsets using unoccluded corner pixels.
    base=src[0,y0:y1,x0:x1].astype(np.float32)
    for j in range(len(ids)):
        ref=valid[j].copy();ref[:40]=False;ref[130:]=False
        off=np.median((data[j]-base)[ref],axis=0)
        data[j]-=off
    data[~valid]=np.nan
    import warnings
    with warnings.catch_warnings():
        warnings.simplefilter('ignore',RuntimeWarning)
        patch=np.nanmedian(data,axis=0)
    missing=~np.isfinite(patch[:,:,0])
    patch=np.uint8(np.nan_to_num(np.clip(patch,0,255),nan=0))
    patch=cv2.inpaint(patch,missing.astype(np.uint8)*255,4,cv2.INPAINT_TELEA)
    plate=src[0].copy();plate[y0:y1,x0:x1]=patch
    return plate

def sprite(src,plate,center,color):
    x,y=np.round(center).astype(int)
    x0,x1=x-43,x+49; y0,y1=y-31,y+65
    im=src[0,y0:y1,x0:x1].astype(np.float32)
    back=plate[y0:y1,x0:x1].astype(np.float32)
    yy,xx=np.mgrid[y0:y1,x0:x1]
    r,g,b=im.transpose(2,0,1)
    if color=='blue': m=(b>r*1.42)&(b>g*1.08)&(b>70)
    elif color=='red':m=(r>g*1.28)&(r>b*1.17)&(r>150)
    else:m=(r>190)&(g>195)&(b>195)&(np.maximum.reduce([r,g,b])-np.minimum.reduce([r,g,b])<32)&((im-back).mean(2)>12)
    _,lab,stats,_=cv2.connectedComponentsWithStats(m.astype(np.uint8))
    k=1+np.argmax(stats[1:,4]);m=(lab==k).astype(np.uint8)
    m=cv2.morphologyEx(m,cv2.MORPH_CLOSE,np.ones((3,3),np.uint8))
    # The hue mask excludes the shaded base edge. Include its one-pixel fringe.
    alpha=cv2.GaussianBlur(cv2.dilate(m,np.ones((3,3),np.uint8)).astype(np.float32),(3,3),.65)
    alpha=np.maximum(alpha,m)
    # Use original RGB for the body and additive source reflection residual.
    body=im.copy()
    refl=(im-back)*(1-alpha[:,:,None])
    envelope=np.exp(-((xx-(x+1))/24)**2-((yy-(y+27))/27)**2)
    envelope*=((yy>y+10)&(yy<y+60))
    # Suppress background logo texture residuals; only low contrast reflected light.
    refl=cv2.GaussianBlur(refl,(5,5),1.0)*envelope[:,:,None]
    refl=np.clip(refl,-14,25)
    return body,alpha,refl,np.array([x-x0,y-y0],float)

def simulate(paths,H):
    p=paths[0].copy();p[:,1]+=np.array([4,5,4])
    init=cv2.perspectiveTransform(p.astype(np.float32).reshape(-1,1,2),H).reshape(-1,2).astype(float)
    ref=paths[14].copy();ref[:,1]+=np.array([4,5,4])
    start=cv2.perspectiveTransform(ref.astype(np.float32).reshape(-1,1,2),H).reshape(-1,2).astype(float)
    # Radius from actual first contact in the input, not from the outer handle.
    radius=np.linalg.norm(start[0]-init[1])*.5
    radius=np.array([radius,radius,radius*1.03])
    samples=[]
    for i in range(15):
        q=paths[i].copy();q[:,1]+=np.array([4,5,4])
        samples.append(cv2.perspectiveTransform(q.astype(np.float32).reshape(-1,1,2),H).reshape(-1,2))
    pos=samples[-1].astype(float).copy()
    # Extract the incoming world-space speed before the source impulse.
    pp=[]
    for i in range(2,13):
        q=paths[i,0]+[0,4]
        pp.append(cv2.perspectiveTransform(q.astype(np.float32).reshape(1,1,2),H)[0,0])
    velocity=np.polyfit(np.arange(2,13)/24,np.array(pp),1)[0]
    vel=np.zeros((3,2),float);vel[0]=velocity
    # Original nearly elastic mallet interaction (about .90) becomes .27.
    e_original=.90;e_blue=e_original*.30
    # Frame 14 is the measured onset of the first disk contact.
    n=pos[0]-pos[1];n/=np.linalg.norm(n)
    closing=np.dot(vel[0]-vel[1],n)
    if closing<0:
        impulse=-(1+e_blue)*closing/2
        vel[0]+=impulse*n;vel[1]-=impulse*n
    events=[(14/24,'blue-red',e_blue)]
    for fi in range(15,96):
        dt=1/(24*100)
        for sub in range(100):
            # Small kinetic drag of the air-hockey surface, unchanged.
            speeds=np.linalg.norm(vel,axis=1)
            vel*=np.maximum(0,1-.009*dt/np.maximum(speeds,1e-9))[:,None]
            pos+=vel*dt
            for a,b in [(0,1),(0,2),(1,2)]:
                diff=pos[a]-pos[b];dist=np.linalg.norm(diff);R=radius[a]+radius[b]
                if dist<R:
                    norm=diff/max(dist,1e-12)
                    pos[a]+=norm*(R-dist)/2;pos[b]-=norm*(R-dist)/2
                    speed=np.dot(vel[a]-vel[b],norm)
                    if speed<0:
                        e=e_blue if a==0 or b==0 else e_original
                        j=-(1+e)*speed/2
                        vel[a]+=j*norm;vel[b]-=j*norm
                        if -speed>.04:events.append(((fi-1)/24+sub*dt,f'{a}-{b}',e))
            for k in range(3):
                for axis,bounds in [(0,(radius[k],1-radius[k])),(1,(radius[k]*.87,2-radius[k]))]:
                    lo,hi=bounds
                    if pos[k,axis]<lo:
                        pos[k,axis]=lo
                        if vel[k,axis]<0:vel[k,axis]*=-(.10*.3 if k==0 else .10)
                    elif pos[k,axis]>hi:
                        pos[k,axis]=hi
                        if vel[k,axis]>0:vel[k,axis]*=-(.10*.3 if k==0 else .10)
        samples.append(pos.copy())
    print('collision events',events)
    return np.array(samples),init

def paste_sprite(canvas,sp,xy,scale):
    body,alpha,refl,origin=sp
    h,w=alpha.shape; origin=origin*scale
    shape=(max(1,round(w*scale)),max(1,round(h*scale)))
    b=cv2.resize(body,shape,interpolation=cv2.INTER_LINEAR)
    a=cv2.resize(alpha,shape,interpolation=cv2.INTER_LINEAR)
    r=cv2.resize(refl,shape,interpolation=cv2.INTER_LINEAR)
    x,y=np.array(xy)-origin; xi,yi=int(np.floor(x)),int(np.floor(y))
    frac=(x-xi,y-yi)
    mat=np.float32([[1,0,frac[0]],[0,1,frac[1]]])
    b=cv2.warpAffine(b,mat,shape,flags=cv2.INTER_LINEAR)
    a=cv2.warpAffine(a,mat,shape,flags=cv2.INTER_LINEAR)
    r=cv2.warpAffine(r,mat,shape,flags=cv2.INTER_LINEAR)
    # Mild compact contact shadow, separate from surface reflection.
    sh=np.zeros(canvas.shape[:2],np.float32)
    cv2.ellipse(sh,(round(xy[0]+3*scale),round(xy[1]+12*scale)),(round(28*scale),round(5*scale)),0,0,360,.08,-1)
    sh=cv2.GaussianBlur(sh,(9,9),2)
    canvas*=1-sh[:,:,None]
    dst=canvas[yi:yi+shape[1],xi:xi+shape[0]]
    dst[:]=np.clip(dst+r,0,255)*(1-a[:,:,None])+b*a[:,:,None]

def main():
    low,src=read_source(); paths=track(src)
    plate=clean_plate(src,paths)
    H=cv2.getPerspectiveTransform(np.float32([[62,435],[603,560],[1150,272],[751,226]]),np.float32([[0,0],[1,0],[1,2],[0,2]]))
    inv=np.linalg.inv(H)
    sim,init=simulate(paths,H)
    sprites=[sprite(src,plate,paths[0,k],c) for k,c in enumerate(['blue','red','white'])]
    ground=cv2.perspectiveTransform(sim.astype(np.float32).reshape(-1,1,2),inv).reshape(96,3,2)
    original_ground=paths[0]+np.array([[0,4],[0,5],[0,4]])
    out=low.copy()
    for i in range(15,96):
        # Only the table area is edited; retain the exact source scene elsewhere.
        im=src[i].astype(np.float32)
        im[248:540,285:905]=plate[248:540,285:905]
        for k in np.argsort(ground[i,:,1]):
            # Perspective-correct size and vertical handle projection.
            old_den=(inv[2,:2]@init[k]+inv[2,2])
            new_den=(inv[2,:2]@sim[i,k]+inv[2,2])
            scale=old_den/new_den
            anchor=ground[i,k]-np.array([0,[4,5,4][k]*scale])
            paste_sprite(im,sprites[k],anchor,scale)
        out[i]=cv2.resize(np.uint8(np.clip(im,0,255)),(256,144),interpolation=cv2.INTER_AREA)
    artifact_io.save(out,ROOT)
    # An ordinary mp4 is supplied in addition to the contract's RGB artifacts.
    writer=cv2.VideoWriter(str(ROOT/'edited_video.mp4'),cv2.VideoWriter_fourcc(*'mp4v'),24,(256,144))
    for frame in out:writer.write(cv2.cvtColor(frame,cv2.COLOR_RGB2BGR))
    writer.release()
    Image.fromarray(plate).save(ROOT/'.tmp/clean_plate.png')
    np.save(ROOT/'.tmp/trajectory.npy',ground)
    print('Saved 96 RGB frames at 256x144, 24 fps.')

if __name__=='__main__':main()
