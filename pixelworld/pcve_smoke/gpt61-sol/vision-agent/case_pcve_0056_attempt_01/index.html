+"""Source-pixel counterfactual: add a third curling stone and resolve contacts.
+
+All imagery is extracted from the supplied video. The ice is reconstructed
+temporally; the stone mattes and blue recolor are local pixel operations.
+"""
+from pathlib import Path
+import numpy as np
+import cv2
+from PIL import Image
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+
+def load_source():
+    capture = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    video = []
+    while True:
+        ok, frame = capture.read()
+        if not ok:
+            break
+        video.append(cv2.cvtColor(frame, cv2.COLOR_BGR2RGB))
+    capture.release()
+    return np.array(video)
+
+
+def color_tracks(video):
+    tracks = []
+    for color in ['red', 'yellow']:
+        track = []
+        for im in video:
+            h = cv2.cvtColor(im, cv2.COLOR_RGB2HSV)
+            if color == 'red':
+                m = ((h[..., 0] < 10) | (h[..., 0] > 170)) & (h[..., 1] > 135) & (h[..., 2] > 85)
+            else:
+                m = (h[..., 0] > 20) & (h[..., 0] < 42) & (h[..., 1] > 100) & (h[..., 2] > 100)
+            m[:310] = False
+            m[400:] = False
+            n, labels, stats, centers = cv2.connectedComponentsWithStats(m.astype(np.uint8))
+            component = 1 + np.argmax(stats[1:, 4])
+            track.append(centers[component])
+        tracks.append(np.array(track))
+    return tracks
+

+def sprite(video, background, kind):
+    # Native 1280-pixel capture lets the source's miniature handles and metal
+    # shell survive the final antialiased reduction to 256 pixels.
+    x, y = (239, 350) if kind == 'red' else (1068, 402)
+    patch = video[0, y - 40:y + 25, x - 45:x + 45].astype(np.float32)
+    ice = background[y - 40:y + 25, x - 45:x + 45].astype(np.float32)
+    hsv = cv2.cvtColor(patch.astype(np.uint8), cv2.COLOR_RGB2HSV)
+    diff = patch - ice
+    yy, xx = np.mgrid[:65, :90]
+    dx, dy = xx - 45, yy - 40
+    mask = ((np.mean(-diff, axis=2) > 22) & (hsv[..., 1] < 85)) | (hsv[..., 1] > 98)
+    mask &= (dx > -29) & (dx < 29) & (dy > -31) & (dy < 11)
+    count, labels, stats, centers = cv2.connectedComponentsWithStats(mask.astype(np.uint8))
+    component = 1 + np.argmax(stats[1:, 4])
+    matte = (labels == component).astype(np.uint8) * 255
+    # Include the near-white highlights and the top bar, which cannot be
+    # separated from light ice by brightness threshold alone.
+    cv2.ellipse(matte, (45, 23), (22, 9), 0, 0, 360, 255, -1)
+    cv2.rectangle(matte, (31, 13), (59, 22), 255, -1)
+    matte = cv2.morphologyEx(matte, cv2.MORPH_CLOSE, np.ones((3, 3), np.uint8))
+    contours, _ = cv2.findContours(matte, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
+    cv2.drawContours(matte, contours, -1, 255, cv2.FILLED)
+    alpha = cv2.GaussianBlur(matte.astype(np.float32) / 255, (3, 3), .55)
+    # Prevent ice-colored fringes when a matte is laid on a painted ring.
+    rgb = np.clip((patch - (1 - alpha[..., None]) * ice) / np.maximum(alpha[..., None], .08), 0, 255)
+    # Native source contact-shadow ratios, decoupled from the red centerline.
+    ratio = np.clip(np.mean((ice - patch) / np.maximum(ice, 20), axis=2), 0, .25)
+    region = ((dx + 3) / 37) ** 2 + ((dy - 5) / 15) ** 2 < 1
+    shadow = cv2.GaussianBlur((ratio * (1 - alpha) * region).astype(np.float32), (7, 7), 1.8)
+    return rgb, alpha, shadow, (x, y)
+
+
+def blue_sprite(red):
+    rgb, alpha, shadow, origin = red
+    rgb = rgb.copy()
+    h = cv2.cvtColor(np.clip(rgb, 0, 255).astype(np.uint8), cv2.COLOR_RGB2HSV)
+    yy, xx = np.mgrid[:65, :90]
+    colored = ((h[..., 0] < 12) | (h[..., 0] > 166)) & (h[..., 1] > 60) & (yy < 32) & (alpha > .3)
+    # Preserve the specular values and the steel shell; change only red enamel.
+    h[..., 0][colored] = 106
+    h[..., 1][colored] = np.minimum(245, np.maximum(155, h[..., 1][colored].astype(int)))
+    blue = cv2.cvtColor(h, cv2.COLOR_HSV2RGB).astype(np.float32)
+    rgb[colored] = blue[colored]
+    return rgb, alpha.copy(), shadow.copy(), origin
+
+
+def paste_stone(canvas, asset, x, y, scale):
+    rgb, alpha, shadow, origin = asset
+    # Warp premultiplied color, preserving its subpixel edge coverage.
+    matrix = np.array([[scale, 0, x - 45 * scale], [0, scale, y - 40 * scale]], np.float32)
+    size = (canvas.shape[1], canvas.shape[0])
+    a = cv2.warpAffine(alpha, matrix, size, flags=cv2.INTER_LINEAR)
+    s = cv2.warpAffine(shadow, matrix, size, flags=cv2.INTER_LINEAR)
+    c = cv2.warpAffine(rgb * alpha[..., None], matrix, size, flags=cv2.INTER_LINEAR)
+    canvas *= 1 - s[..., None]
+    canvas *= 1 - a[..., None]
+    canvas += c
+
+
+def trajectory(video, tracks, background):
+    # Convert screen x to a weak-perspective coordinate along the centerline.
+    # The slight perspective also accounts for the different apparent radii.
+    x0, x1, perspective = 239., 1068., .047
+    span = x1 - x0
+
+    def world(x):
+        d = np.asarray(x) - x0
+        return d / (span * (1 - perspective) + perspective * d)
+
+    def screen(q):
+        return x0 + span * (1 - perspective) * q / (1 - perspective * q)
+
+    # Enamel centroid offsets relative to the solid stone centers.
+    red_x = tracks[0][:38, 0] + (x0 - tracks[0][0, 0])
+    yellow_x = tracks[1][:38, 0] + (x1 - tracks[1][0, 0])
+    time = np.arange(38)
+    fits = [np.polyfit(time, world(red_x), 2), np.polyfit(time, world(yellow_x), 2)]
+    fits[0][-1] = 0
+    fits[1][-1] = 1
+    # Exactly a quarter of the visible segment from yellow toward red.
+    q = np.array([0., float(world(.25 * x0 + .75 * x1)), 1.])
+    v = np.array([fits[0][1], 0., fits[1][1]])
+    free = np.array([False, True, False])
+    diam = .0600
+    restitution = .095  # Observed low-speed rebound in the factual two-stone contact.
+    friction = .000007
+    history = []
+    events = []
+    substeps = 80
+    for frame in range(96):
+        history.append(q.copy())
+        for substep in range(substeps):
+            t = frame + (substep + .5) / substeps
+            for body, poly in [(0, fits[0]), (2, fits[1])]:
+                if not free[body]:
+                    v[body] = 2 * poly[0] * t + poly[1]
+            q += v / substeps
+            for i, j in [(0, 1), (1, 2)]:
+                gap = q[j] - q[i]
+                if gap < diam:
+                    correction = (diam - gap) / 2
+                    q[i] -= correction
+                    q[j] += correction
+                    if v[i] > v[j]:
+                        if not free[i] or not free[j]:
+                            events.append((round(t / 24, 3), i, j))
+                        impulse = (1 + restitution) * (v[i] - v[j]) / 2
+                        v[i] -= impulse
+                        v[j] += impulse
+                        free[i] = free[j] = True
+            # Coulomb sliding loss, calibrated to the source's slow rebound.
+            for i in range(3):
+                if free[i]:
+                    v[i] = np.sign(v[i]) * max(0., abs(v[i]) - friction / substeps)
+    print('First contacts (seconds, body IDs):', events)
+    history = np.array(history)
+    return screen(history), history
+
+
+def main():
+    video = load_source()
+    assert video.shape == (96, 720, 1280, 3)
+    # Both original stones move: a temporal median removes them without
+    # inventing any ice texture, reflections, target markings or lighting.
+    background = np.median(video[:37], axis=0).astype(np.uint8)
+    tracks = color_tracks(video)
+    red = sprite(video, background, 'red')
+    yellow = sprite(video, background, 'yellow')
+    blue = blue_sprite(red)
+    positions, world_positions = trajectory(video, tracks, background)
+    frames = []
+    for frame in range(96):
+        canvas = background.astype(np.float32)
+        for body, asset in enumerate([red, blue, yellow]):
+            x = positions[frame, body]
+            # Project the same plane-space centerline used by both source stones.
+            y = 350 + (402 - 350) * (x - 239) / (1068 - 239)
+            q = world_positions[frame, body]
+            ref_q = 1. if body == 2 else 0.
+            scale = (1 - .047 * ref_q) / (1 - .047 * q)
+            paste_stone(canvas, asset, x, y, scale)
+        rgb = np.uint8(np.clip(canvas + .5, 0, 255))
+        small = np.array(Image.fromarray(rgb).resize((256, 144), Image.Resampling.LANCZOS))
+        frames.append(small)
+    frames = np.stack(frames)
+    artifact_io.save(frames, ROOT)
+    print('Saved:', frames.shape, frames.dtype)
+
+
+if __name__ == '__main__':
+    main()
