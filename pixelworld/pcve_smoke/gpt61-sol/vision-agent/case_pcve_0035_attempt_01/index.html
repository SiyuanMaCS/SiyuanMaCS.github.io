+"""Bowling-ball friction counterfactual, reconstructed from source observations.
+
+The camera is fixed. A lane-plane projective coordinate makes ball translation
+linear in physical distance. Fit constant deceleration before pin contact,
+multiply that deceleration by five, and reproject the new travel distance.
+Source appearance is sampled at the matching rolling distance, so the ball,
+its contact shadow and its floor reflection stay registered. The counterfactual
+never reaches the pins and therefore never samples a pin-impact image.
+"""
+from pathlib import Path
+import cv2
+import numpy as np
+from PIL import Image
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+FPS = 24
+
+
+def load_source():
+    pixels = np.load(ROOT / 'source_pixels.npz')['rgb']
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    high = []
+    # Enough clean pre-impact observations to estimate rolling resistance.
+    for i in range(36):
+        ok, bgr = cap.read()
+        if not ok:
+            raise RuntimeError('Incomplete source video')
+        high.append(cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB))
+    cap.release()
+    return pixels, np.asarray(high)
+
+
+def lane_horizon(frame):
+    # Parallel lane edges meet at the horizon of the lane plane.
+    edge_fits = []
+    for left in (True, False):
+        points = []
+        for y in range(85, 144):
+            a = frame[y].astype(float)
+            x = np.arange(256)
+            expected = 160.5 - .574 * y if left else 107.5 + .345 * y
+            select = ((np.abs(x - expected) < 2) & (a[:, 1] > 130)
+                      & (a[:, 2] > 130) & (a[:, 0] < 125))
+            if np.any(select):
+                points.append((y, x[select].mean()))
+        points = np.asarray(points)
+        edge_fits.append(np.polyfit(points[:, 0], points[:, 1], 1))
+    a, b = edge_fits
+    return (b[1] - a[1]) / (a[0] - b[0])
+
+
+def ball_centers(high):
+    centers = []
+    for frame in high:
+        a = frame[370:550, 605:665].astype(float)
+        blue = ((a[:, :, 2] - a[:, :, 0] > 45)
+                & (a[:, :, 2] - a[:, :, 1] > 10)
+                & (a[:, :, 1] - a[:, :, 0] > 30))
+        yy, xx = np.where(blue)
+        centers.append(((xx.mean() + 605) / 5, (yy.mean() + 370) / 5))
+    return np.asarray(centers)
+
+
+def generate():
+    original, high = load_source()
+    centers = ball_centers(high)
+    horizon = lane_horizon(original[0])
+    t_obs = np.arange(len(high)) / FPS
+    # 1 / (image_y - horizon) is affine in world-space lane depth.
+    inverse_depth = 1 / (centers[:, 1] - horizon)
+    c, velocity, quadratic = np.linalg.lstsq(
+        np.c_[np.ones_like(t_obs), t_obs, t_obs ** 2],
+        inverse_depth, rcond=None)[0]
+    original_deceleration = max(-2 * quadratic, 1e-6)
+    deceleration = 5 * original_deceleration
+    stop_time = velocity / deceleration
+    t = np.arange(96) / FPS
+    moving_t = np.minimum(t, stop_time)
+    depth = inverse_depth[0] + velocity * moving_t - .5 * deceleration * moving_t ** 2
+    target_y = horizon + 1 / depth
+    # All samples are pre-contact; standing pins and their reflections remain.
+    source_times = np.interp(target_y, centers[::-1, 1], np.arange(len(high))[::-1])
+    frames = []
+    for n, f in enumerate(source_times):
+        if n == 0:
+            frames.append(original[0].copy())
+            continue
+        lo = int(np.floor(f))
+        hi = min(lo + 1, len(high) - 1)
+        alpha = f - lo
+        # High-resolution interpolation followed by the same reconstruction
+        # filter as the provided source keeps fractional travel smooth, including
+        # the diffuse floor reflection. A stationary camera is not retimed.
+        image = np.rint((1 - alpha) * high[lo].astype(np.float32)
+                        + alpha * high[hi].astype(np.float32)).astype(np.uint8)
+        frame = np.asarray(Image.fromarray(image).resize((256, 144), Image.Resampling.LANCZOS)).copy()
+        # The alley, fixtures, and other lanes have no counterfactual changes.
+        # Use original time-indexed pixels outside the causal central-lane area.
+        frame[:, :116] = original[n, :, :116]
+        frame[:, 141:] = original[n, :, 141:]
+        frame[:68, :] = original[n, :68, :]
+        frame[125:, :] = original[n, 125:, :]
+        frames.append(frame)
+    frames = np.asarray(frames, dtype=np.uint8)
+    artifact_io.save(frames, ROOT)
+    print(f'Horizon: {horizon:.3f}; friction: 5x; stop time: {stop_time:.3f}s; '
+          f'stopped ball y: {target_y[-1]:.3f}; matched source frame: {source_times[-1]:.3f}')
+
+
+if __name__ == '__main__':
+    generate()
