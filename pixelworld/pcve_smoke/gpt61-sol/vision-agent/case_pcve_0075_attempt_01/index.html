+"""Counterfactual die drop, reconstructed from source pixels and a fitted ballistic path.
+
+No change to the camera, table, or red die. The fitted initial velocity alone
+is multiplied by four; gravity, perspective and bounce restitution are retained.
+"""
+from pathlib import Path
+import cv2
+import numpy as np
+from PIL import Image
+from scipy.ndimage import gaussian_filter
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+
+def component_hull(mask):
+    n, labels, stats, centers = cv2.connectedComponentsWithStats(mask.astype(np.uint8))
+    j = 1 + np.argmax(stats[1:, cv2.CC_STAT_AREA])
+    points = np.argwhere(labels == j)[:, ::-1].astype(np.int32)
+    hull = cv2.convexHull(points)
+    result = np.zeros(mask.shape, np.uint8)
+    cv2.fillConvexPoly(result, hull, 255)
+    return result
+
+
+def white_mask(frame, y):
+    f = frame.astype(np.float32)
+    mask = ((f[..., 0] > 165) & (f[..., 1] > 160) & (f[..., 2] > 145)
+            & (np.abs(f[..., 0] - f[..., 1]) < 25)
+            & (f[..., 1] - f[..., 2] < 30))
+    mask[:, :626] = False
+    mask[:, 703:] = False
+    mask[:max(0, int(y)-43)] = False
+    mask[int(y)+44:] = False
+    return component_hull(mask)
+
+
+def red_mask(frame):
+    f = frame.astype(np.float32)
+    mask = ((f[..., 0] > 130) & (f[..., 0]-f[..., 1] > 30)
+            & (f[..., 0]-f[..., 2] > 30))
+    mask[:, :565] = False
+    mask[:, 648:] = False
+    mask[475:] = False
+    return component_hull(mask)
+
+
+def generate():
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    original = []
+    while True:
+        ok, bgr = cap.read()
+        if not ok:
+            break
+        original.append(cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB))
+    cap.release()
+    original = np.stack(original)
+    small = np.load(ROOT / 'source_pixels.npz')['rgb']
+    H, W = original.shape[1:3]
+    # Disjoint temporal plates expose the entire central background without a die.
+    plate = np.median(original[:7], axis=0).astype(np.uint8)
+    plate[:280, 565:740] = np.median(original[40:], axis=0).astype(np.uint8)[:280, 565:740]
+    plate[280:, 565:740] = np.median(original[:3], axis=0).astype(np.uint8)[280:, 565:740]
+
+    ys = [39.25,64.82,100.61,144.83,196.99,256.43,322.57,
+          393.48,358.49,331.,311.47,300.98,300.,307.31,323.21,
+          347.17,378.51,382.03,369.17,364.6,368.43,379.86,
+          390.45,383.93,385.42,391.54,390.10,391.52,392.05,392.45]
+    ys += [392.5]*(96-len(ys))
+
+    # Sprite masks come only from unobscured source poses. Perspective and
+    # source lighting change naturally with projected height.
+    sprites = []
+    for k in [0,1,2,3,4,5,6]:
+        mask = white_mask(original[k], ys[k])
+        yy, xx = np.nonzero(mask)
+        x0,x1,y0,y1 = xx.min()-3,xx.max()+4,max(0,yy.min()-3),yy.max()+4
+        alpha = cv2.dilate(mask, np.ones((3,3),np.uint8))
+        alpha = gaussian_filter(alpha.astype(np.float32)/255, .45)
+        rgb = original[k,y0:y1,x0:x1].astype(np.float32)
+        aa = alpha[y0:y1,x0:x1]
+        center = ((xx.min()+xx.max())/2-x0, (yy.min()+yy.max())/2-y0)
+        sprites.append((ys[k],rgb,aa,center))
+
+    # Perspective-aware fit to the first seven factual frames:
+    # screen_y = y0 + u/(1+k*u), u = v*t + g*t*t/2.
+    y0 = 39.0233700
+    factual_v = 504.790395
+    half_g = 2944.08283
+    perspective = .000303713382
+    floor_y = 392.5
+    floor_u = (floor_y-y0)/(1-perspective*(floor_y-y0))
+    velocity = 4*factual_v
+    gravity = 2*half_g
+    impact = float(np.roots([half_g,velocity,-floor_u])[1])
+    incoming = velocity + gravity*impact
+    restitution = .51921647
+    segments = []
+    start = impact
+    rebound = incoming*restitution
+    while rebound > 18:
+        end = start + 2*rebound/gravity
+        segments.append((start,end,rebound))
+        start = end
+        rebound *= restitution
+    settle_time = start
+
+    grid_y,grid_x = np.mgrid[:H,:W]
+    output = []
+    for i in range(96):
+        t = i/24
+        if i == 0:
+            output.append(small[0].copy())
+            continue
+        if t < impact:
+            u = velocity*t+half_g*t*t
+        else:
+            u = floor_u
+            for ta,tb,vb in segments:
+                if ta <= t < tb:
+                    dt = t-ta
+                    u = floor_u-vb*dt+half_g*dt*dt
+                    break
+        y = y0+u/(1+perspective*u)
+        x = 664.0 - 1.5*(u/floor_u)
+        frame = original[i].astype(np.float32).copy()
+        red = red_mask(original[i])
+        protect = cv2.dilate(red,np.ones((3,3),np.uint8)) > 0
+        old = white_mask(original[i],ys[i])
+        erase = cv2.dilate(old,np.ones((7,7),np.uint8)) > 0
+        # Remove the old die's contact shadow as well as its visible silhouette.
+        erase |= (((grid_x-679)/49)**2+((grid_y-432)/22)**2 < 1)
+        if ys[i] > 290:
+            erase |= (((grid_x-679)/45)**2+((grid_y-(ys[i]+34))/17)**2 < 1)
+        erase &= ~protect
+        frame[erase] = plate[erase]
+
+        # Soft directional contact shadow anchored to the unchanged table plane.
+        height = max(0.,floor_u-u)
+        rx = 28 + .03*height
+        ry = 8.5 + .011*height
+        sx = 675+.035*height
+        sy = 429+.012*height
+        opacity = .27*np.exp(-height/100)
+        shadow = opacity*np.exp(-.5*((grid_x-sx)/rx)**2-.5*((grid_y-sy)/ry)**2)
+        shadow[protect] = 0
+        frame *= (1-shadow[...,None])
+
+        target = min(sprites, key=lambda s:abs(s[0]-y))
+        source_y,rgb,alpha,center = target
+        # Dice have a nearly constant apparent size; this is the weak depth
+        # dependence measured on the original downward flight.
+        scale = (1-.00018*(y-y0))/(1-.00018*(source_y-y0))
+        M = np.array([[scale,0,x-scale*center[0]],
+                      [0,scale,y-scale*center[1]]],np.float32)
+        a = cv2.warpAffine(alpha,M,(W,H),flags=cv2.INTER_LINEAR)
+        # Premultiplied interpolation avoids seams around the rounded cube.
+        rgba = cv2.warpAffine(rgb*alpha[...,None],M,(W,H),flags=cv2.INTER_LINEAR)
+        frame = frame*(1-a[...,None])+rgba
+        # The pink cube is in the foreground whenever their projections overlap.
+        frame[protect] = original[i][protect]
+        frame = np.clip(frame,0,255).astype(np.uint8)
+        low = np.array(Image.fromarray(frame).resize((256,144),Image.Resampling.LANCZOS))
+        # Outside the edited region retain the supplied RGB pixels exactly.
+        low[:,:112] = small[i,:,:112]
+        low[:,150:] = small[i,:,150:]
+        output.append(low)
+    frames = np.stack(output)
+    artifact_io.save(frames, ROOT)
+    print(f'Saved {len(frames)} frames. First contact {impact:.3f}s; settled {settle_time:.3f}s.')
+
+if __name__ == '__main__':
+    generate()

\"\"\"Counterfactual die drop, reconstructed from source pixels and a fitted ballistic path.

No change to the camera, table, or red die. The fitted initial velocity alone
is multiplied by four; gravity, perspective and bounce restitution are retained.
\"\"\"
from pathlib import Path
import cv2
import numpy as np
from PIL import Image
from scipy.ndimage import gaussian_filter
import artifact_io

ROOT = Path(__file__).resolve().parent


def component_hull(mask):
    n, labels, stats, centers = cv2.connectedComponentsWithStats(mask.astype(np.uint8))
    j = 1 + np.argmax(stats[1:, cv2.CC_STAT_AREA])
    points = np.argwhere(labels == j)[:, ::-1].astype(np.int32)
    hull = cv2.convexHull(points)
    result = np.zeros(mask.shape, np.uint8)
    cv2.fillConvexPoly(result, hull, 255)
    return result


def white_mask(frame, y):
    f = frame.astype(np.float32)
    mask = ((f[..., 0] > 165) & (f[..., 1] > 160) & (f[..., 2] > 145)
            & (np.abs(f[..., 0] - f[..., 1]) < 25)
            & (f[..., 1] - f[..., 2] < 30))
    mask[:, :626] = False
    mask[:, 703:] = False
    mask[:max(0, int(y)-43)] = False
    mask[int(y)+44:] = False
    return component_hull(mask)


def red_mask(frame):
    f = frame.astype(np.float32)
    mask = ((f[..., 0] > 130) & (f[..., 0]-f[..., 1] > 30)
            & (f[..., 0]-f[..., 2] > 30))
    mask[:, :565] = False
    mask[:, 648:] = False
    mask[475:] = False
    return component_hull(mask)


def generate():
    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
    original = []
    while True:
        ok, bgr = cap.read()
        if not ok:
            break
        original.append(cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB))
    cap.release()
    original = np.stack(original)
    small = np.load(ROOT / 'source_pixels.npz')['rgb']
    H, W = original.shape[1:3]
    # Disjoint temporal plates expose the entire central background without a die.
    plate = np.median(original[:7], axis=0).astype(np.uint8)
    plate[:280, 565:740] = np.median(original[40:], axis=0).astype(np.uint8)[:280, 565:740]
    plate[280:, 565:740] = np.median(original[:3], axis=0).astype(np.uint8)[280:, 565:740]

    ys = [39.25,64.82,100.61,144.83,196.99,256.43,322.57,
          393.48,358.49,331.,311.47,300.98,300.,307.31,323.21,
          347.17,378.51,382.03,369.17,364.6,368.43,379.86,
          390.45,383.93,385.42,391.54,390.10,391.52,392.05,392.45]
    ys += [392.5]*(96-len(ys))

    # Sprite masks come only from unobscured source poses. Perspective and
    # source lighting change naturally with projected height.
    sprites = []
    for k in [0,1,2,3,4,5,6]:
        mask = white_mask(original[k], ys[k])
        yy, xx = np.nonzero(mask)
        x0,x1,y0,y1 = xx.min()-3,xx.max()+4,max(0,yy.min()-3),yy.max()+4
        alpha = cv2.dilate(mask, np.ones((3,3),np.uint8))
        alpha = gaussian_filter(alpha.astype(np.float32)/255, .45)
        rgb = original[k,y0:y1,x0:x1].astype(np.float32)
        aa = alpha[y0:y1,x0:x1]
        center = ((xx.min()+xx.max())/2-x0, (yy.min()+yy.max())/2-y0)
        sprites.append((ys[k],rgb,aa,center))

    # Perspective-aware fit to the first seven factual frames:
    # screen_y = y0 + u/(1+k*u), u = v*t + g*t*t/2.
    y0 = 39.0233700
    factual_v = 504.790395
    half_g = 2944.08283
    perspective = .000303713382
    floor_y = 392.5
    floor_u = (floor_y-y0)/(1-perspective*(floor_y-y0))
    velocity = 4*factual_v
    gravity = 2*half_g
    impact = float(np.roots([half_g,velocity,-floor_u])[1])
    incoming = velocity + gravity*impact
    restitution = .51921647
    segments = []
    start = impact
    rebound = incoming*restitution
    while rebound > 18:
        end = start + 2*rebound/gravity
        segments.append((start,end,rebound))
        start = end
        rebound *= restitution
    settle_time = start

    grid_y,grid_x = np.mgrid[:H,:W]
    output = []
    for i in range(96):
        t = i/24
        if i == 0:
            output.append(small[0].copy())
            continue
        if t < impact:
            u = velocity*t+half_g*t*t
        else:
            u = floor_u
            for ta,tb,vb in segments:
                if ta <= t < tb:
                    dt = t-ta
                    u = floor_u-vb*dt+half_g*dt*dt
                    break
        y = y0+u/(1+perspective*u)
        x = 664.0 - 1.5*(u/floor_u)
        frame = original[i].astype(np.float32).copy()
        red = red_mask(original[i])
        protect = cv2.dilate(red,np.ones((3,3),np.uint8)) > 0
        old = white_mask(original[i],ys[i])
        erase = cv2.dilate(old,np.ones((7,7),np.uint8)) > 0
        # Remove the old die's contact shadow as well as its visible silhouette.
        erase |= (((grid_x-679)/49)**2+((grid_y-432)/22)**2 < 1)
        if ys[i] > 290:
            erase |= (((grid_x-679)/45)**2+((grid_y-(ys[i]+34))/17)**2 < 1)
        erase &= ~protect
        frame[erase] = plate[erase]

        # Soft directional contact shadow anchored to the unchanged table plane.
        height = max(0.,floor_u-u)
        rx = 28 + .03*height
        ry = 8.5 + .011*height
        sx = 675+.035*height
        sy = 429+.012*height
        opacity = .27*np.exp(-height/100)
        shadow = opacity*np.exp(-.5*((grid_x-sx)/rx)**2-.5*((grid_y-sy)/ry)**2)
        shadow[protect] = 0
        frame *= (1-shadow[...,None])

        target = min(sprites, key=lambda s:abs(s[0]-y))
        source_y,rgb,alpha,center = target
        # Dice have a nearly constant apparent size; this is the weak depth
        # dependence measured on the original downward flight.
        scale = (1-.00018*(y-y0))/(1-.00018*(source_y-y0))
        M = np.array([[scale,0,x-scale*center[0]],
                      [0,scale,y-scale*center[1]]],np.float32)
        a = cv2.warpAffine(alpha,M,(W,H),flags=cv2.INTER_LINEAR)
        # Premultiplied interpolation avoids seams around the rounded cube.
        rgba = cv2.warpAffine(rgb*alpha[...,None],M,(W,H),flags=cv2.INTER_LINEAR)
        frame = frame*(1-a[...,None])+rgba
        # The pink cube is in the foreground whenever their projections overlap.
        frame[protect] = original[i][protect]
        frame = np.clip(frame,0,255).astype(np.uint8)
        low = np.array(Image.fromarray(frame).resize((256,144),Image.Resampling.LANCZOS))
        # Outside the edited region retain the supplied RGB pixels exactly.
        low[:,:112] = small[i,:,:112]
        low[:,150:] = small[i,:,150:]
        output.append(low)
    frames = np.stack(output)
    artifact_io.save(frames, ROOT)
    print(f'Saved {len(frames)} frames. First contact {impact:.3f}s; settled {settle_time:.3f}s.')

if __name__ == '__main__':
    generate()
