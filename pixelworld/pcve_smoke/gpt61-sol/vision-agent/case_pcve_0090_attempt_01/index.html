+from pathlib import Path
+import math
+import cv2
+import numpy as np
+from PIL import Image
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+N = 96
+# Projective coordinate along the shot line.  The denominator is measured
+# from the source's shrinking ball diameter and constant-speed trajectory.
+P0 = np.array([429.5, 473.0])
+P1 = np.array([639.5, 350.5])
+K = .34
+
+def project(s):
+    u = 1.34*s/(1+.34*s)
+    return P0 + u*(P1-P0)
+
+def scale(s):
+    return 1/(1+.34*s)
+
+def components(mask, x0, y0, minimum):
+    n, lab, stats, centers = cv2.connectedComponentsWithStats(mask.astype(np.uint8), 8)
+    out=[]
+    for j in range(1,n):
+        if stats[j,4] >= minimum:
+            x,y=centers[j]+[x0,y0]
+            out.append((x,y,stats[j,4],stats[j,0]+x0,stats[j,1]+y0,stats[j,2],stats[j,3]))
+    return out
+
+def read_source():
+    cap=cv2.VideoCapture(str(ROOT/'source_video.mp4'))
+    samples={}; tracks=[]; patches=[]
+    for i in range(N):
+        ok, bgr=cap.read()
+        if not ok: raise RuntimeError('Incomplete source')
+        rgb=cv2.cvtColor(bgr,cv2.COLOR_BGR2RGB)
+        if i in (0,20,40): samples[i]=rgb.copy()
+        c=rgb[330:505,390:665].astype(np.float32)
+        mask=(c[:,:,0]>140)&(c[:,:,1]>155)&(np.abs(c[:,:,0]-c[:,:,1])<22)&(np.abs(c[:,:,1]-c[:,:,2])<25)
+        white=max(components(mask,390,330,300),key=lambda v:v[2])
+        c=rgb[200:385,620:885]
+        mask=(c[:,:,0]<70)&(c[:,:,1]<70)&(c[:,:,2]<70)
+        candidates=components(mask,620,200,70)
+        candidates=[v for v in candidates if not (v[0]<750 and v[1]<225)]
+        black=max(candidates,key=lambda v:v[2])
+        # The center of the dark ball's bounding box is stable despite its
+        # rotating white number patch.
+        bc=np.array([black[3]+(black[5]-1)/2,black[4]+(black[6]-1)/2])
+        if i<13: bc=P1.copy()
+        wc=np.array(white[:2])+[.6,2.7]
+        tracks.append((wc,bc))
+        patch=cv2.getRectSubPix(rgb.astype(np.float32),(81,81),tuple(bc.astype(float)))
+        patches.append((patch,bc))
+    cap.release()
+    return samples,tracks,patches

+def feather_replace(base, other, center, radius):
+    yy,xx=np.mgrid[:base.shape[0],:base.shape[1]]
+    d=np.sqrt(((xx-center[0])/radius[0])**2+((yy-center[1])/radius[1])**2)
+    a=np.clip((1-d)*5,0,1)
+    return (base*(1-a[:,:,None])+other*a[:,:,None]).astype(np.float32)
+
+def simulate():
+    # Equal-mass billiard impacts, no kinematic repositioning after insertion.
+    # One coordinate unit is the initial cue-to-target separation.
+    pos=np.array([0.,.25,1.])
+    vel=np.array([.0662,0.,0.])
+    diameter=.143
+    dt=1/160
+    states=[]
+    for f in range(N):
+        states.append(pos.copy())
+        for sub in range(160):
+            t=f+sub*dt
+            pos+=vel*dt
+            for j in range(3):
+                for l in range(j+1,3):
+                    delta=pos[l]-pos[j]
+                    normal=1 if delta>=0 else -1
+                    if abs(delta)<diameter and (vel[j]-vel[l])*normal>0:
+                        overlap=diameter-abs(delta)
+                        pos[j]-=normal*overlap/2
+                        pos[l]+=normal*overlap/2
+                        impulse=.9875*(vel[j]-vel[l])
+                        vel[j]-=impulse
+                        vel[l]+=impulse
+            for j in range(3):
+                if pos[j]>3.012:
+                    pos[j]=6.024-pos[j]
+                    vel[j]=-abs(vel[j])*.78
+                # Slow residual motion settles; the main shot initially slides
+                # almost uniformly, then changes to a decelerating roll.
+                decel=.00015 if abs(vel[j])<.01 else (.00050 if vel[j]<0 else .00002)
+                vel[j]=np.sign(vel[j])*max(0,abs(vel[j])-decel*dt)
+    return np.array(states)
+
+def add_patch(canvas, residual, center, factor):
+    # Subpixel affine transport of a photographed ball and its contact shadow.
+    h,w=residual.shape[:2]
+    x0=max(0,int(center[0]-w*factor/2-3)); x1=min(canvas.shape[1],int(center[0]+w*factor/2+4))
+    y0=max(0,int(center[1]-h*factor/2-3)); y1=min(canvas.shape[0],int(center[1]+h*factor/2+4))
+    mat=np.array([[factor,0,center[0]-x0-factor*(w-1)/2],
+                  [0,factor,center[1]-y0-factor*(h-1)/2]],np.float32)
+    warped=cv2.warpAffine(residual,mat,(x1-x0,y1-y0),flags=cv2.INTER_LINEAR,borderMode=cv2.BORDER_CONSTANT)
+    canvas[y0:y1,x0:x1]+=warped
+
+def soften_support(shape, rx, ry, cy=0):
+    yy,xx=np.mgrid[:shape[0],:shape[1]]
+    d=np.sqrt(((xx-(shape[1]-1)/2)/rx)**2+((yy-(shape[0]-1)/2-cy)/ry)**2)
+    return np.clip((1-d)*6,0,1).astype(np.float32)
+
+def main():
+    samples,tracks,black_patches=read_source()
+    source=np.load(ROOT/'source_pixels.npz')['rgb']
+    # Real cloth pixels from later source frames create an unobstructed plate.
+    plate=samples[0].astype(np.float32)
+    plate=feather_replace(plate,samples[20].astype(np.float32),P0,(52,57))
+    plate=feather_replace(plate,samples[40].astype(np.float32),P1,(47,51))
+    white=cv2.getRectSubPix(samples[0].astype(np.float32),(81,81),tuple(P0))
+    white_bg=cv2.getRectSubPix(plate,(81,81),tuple(P0))
+    white_res=(white-white_bg)*soften_support((81,81),35,40,4)[:,:,None]
+    # Keep the source's illumination, specular highlight, and contact shadow.
+    # Only the ball material becomes the solid yellow number-one material.
+    yy,xx=np.mgrid[:81,:81]; x=(xx-40)/20.0; y=(yy-40)/19.6
+    alpha=np.clip((1-np.sqrt(x*x+y*y))*19+.3,0,1).astype(np.float32)
+    material=white.copy()
+    # warm gold lacquer, retaining a neutral specular reflection
+    tint=np.array([1.025,.80,.085],np.float32)
+    gold=material*tint
+    highlight=np.clip((material.mean(axis=2)-222)/18,0,.45)
+    gold=gold*(1-highlight[:,:,None])+material*highlight[:,:,None]
+    yellow=white+alpha[:,:,None]*(gold-white)
+    # White inset with a black 1, drawn in the sphere's upper-front face.
+    lx=(x+.24)/.28; ly=(y+.34)/.245
+    label=np.clip((1-np.sqrt(lx*lx+ly*ly))*12+.3,0,1)*alpha
+    yellow=yellow*(1-label[:,:,None])+np.array([236,237,224])*label[:,:,None]
+    # A tiny antialiased printed numeral: stem, serif, and baseline.
+    num=np.zeros((81,81),np.uint8)
+    cv2.line(num,(35,31),(35,36),255,1,cv2.LINE_AA)
+    cv2.line(num,(33,32),(35,31),255,1,cv2.LINE_AA)
+    cv2.line(num,(33,36),(37,36),255,1,cv2.LINE_AA)
+    na=num.astype(float)/255*label
+    yellow=yellow*(1-na[:,:,None])+np.array([34,31,20])*na[:,:,None]
+    yellow_res=(yellow-white_bg)*soften_support((81,81),35,40,4)[:,:,None]
+    clean_small=np.asarray(Image.fromarray(np.clip(plate,0,255).astype(np.uint8)).resize((256,144),Image.Resampling.LANCZOS))
+    paths=simulate()
+    output=[]
+    Y,X=np.mgrid[:144,:256]
+    for i,state in enumerate(paths):
+        canvas=plate.copy()
+        add_patch(canvas,white_res,project(state[0]),scale(state[0]))
+        add_patch(canvas,yellow_res.astype(np.float32),project(state[1]),scale(state[1]))
+        # Select a genuine source texture at corresponding rolling time.
+        ti=0 if state[2]<=1.0001 else min(95,max(15,i+2))
+        tex,tc=black_patches[ti]
+        if ti in (13,14):tex,tc=black_patches[15]
+        bg=cv2.getRectSubPix(plate,(81,81),tuple(tc.astype(float)))
+        tx=(tc[0]-P0[0])/(P1[0]-P0[0])
+        ts=tx/(1.34-.34*tx)
+        src_scale=scale(ts)
+        residual=(tex-bg)*soften_support((81,81),28*src_scale/.746,32*src_scale/.746,3)[:,:,None]
+        add_patch(canvas,residual,project(state[2]),scale(state[2])/src_scale)
+        rendered=np.asarray(Image.fromarray(np.clip(canvas,0,255).astype(np.uint8)).resize((256,144),Image.Resampling.LANCZOS))
+        # Preserve source pixels exactly outside edited balls and their shadows.
+        edit=np.zeros((144,256),float)
+        for p,r in [(tracks[i][0],(33*scale(min(.91, i*.065)),39*scale(min(.91,i*.065)))),
+                    (tracks[i][1],(28,32))]+[(project(s),(34*scale(s),40*scale(s))) for s in state]:
+            cx,cy=(p-2)/5
+            d=np.sqrt(((X-cx)/(r[0]/5))**2+((Y-cy-0.4)/(r[1]/5))**2)
+            edit=np.maximum(edit,np.clip((1-d)*4,0,1))
+        out=source[i].astype(float)*(1-edit[:,:,None])+rendered.astype(float)*edit[:,:,None]
+        output.append(np.clip(np.rint(out),0,255).astype(np.uint8))
+    frames=np.stack(output)
+    artifact_io.save(frames,ROOT)
+    # Supplementary readily playable movie, with the required tensor/player
+    # being the lossless canonical artifacts.
+    writer=cv2.VideoWriter(str(ROOT/'edited_video.mp4'),cv2.VideoWriter_fourcc(*'mp4v'),24,(256,144))
+    for fr in frames: writer.write(cv2.cvtColor(fr,cv2.COLOR_RGB2BGR))
+    writer.release()
+    np.save(ROOT/'.tmp'/'counterfactual_positions.npy',paths)
+    print('Saved',frames.shape,'RGB frames; final ball coordinates:',paths[-1])
+
+if __name__=='__main__':
+    main()

from pathlib import Path
import math
import cv2
import numpy as np
from PIL import Image
import artifact_io

ROOT = Path(__file__).resolve().parent
N = 96
# Projective coordinate along the shot line.  The denominator is measured
# from the source's shrinking ball diameter and constant-speed trajectory.
P0 = np.array([429.5, 473.0])
P1 = np.array([639.5, 350.5])
K = .34

def project(s):
    u = 1.34*s/(1+.34*s)
    return P0 + u*(P1-P0)

def scale(s):
    return 1/(1+.34*s)

def components(mask, x0, y0, minimum):
    n, lab, stats, centers = cv2.connectedComponentsWithStats(mask.astype(np.uint8), 8)
    out=[]
    for j in range(1,n):
        if stats[j,4] >= minimum:
            x,y=centers[j]+[x0,y0]
            out.append((x,y,stats[j,4],stats[j,0]+x0,stats[j,1]+y0,stats[j,2],stats[j,3]))
    return out

def read_source():
    cap=cv2.VideoCapture(str(ROOT/'source_video.mp4'))
    samples={}; tracks=[]; patches=[]
    for i in range(N):
        ok, bgr=cap.read()
        if not ok: raise RuntimeError('Incomplete source')
        rgb=cv2.cvtColor(bgr,cv2.COLOR_BGR2RGB)
        if i in (0,20,40): samples[i]=rgb.copy()
        c=rgb[330:505,390:665].astype(np.float32)
        mask=(c[:,:,0]>140)&(c[:,:,1]>155)&(np.abs(c[:,:,0]-c[:,:,1])<22)&(np.abs(c[:,:,1]-c[:,:,2])<25)
        white=max(components(mask,390,330,300),key=lambda v:v[2])
        c=rgb[200:385,620:885]
        mask=(c[:,:,0]<70)&(c[:,:,1]<70)&(c[:,:,2]<70)
        candidates=components(mask,620,200,70)
        candidates=[v for v in candidates if not (v[0]<750 and v[1]<225)]
        black=max(candidates,key=lambda v:v[2])
        # The center of the dark ball's bounding box is stable despite its
        # rotating white number patch.
        bc=np.array([black[3]+(black[5]-1)/2,black[4]+(black[6]-1)/2])
        if i<13: bc=P1.copy()
        wc=np.array(white[:2])+[.6,2.7]
        tracks.append((wc,bc))
        patch=cv2.getRectSubPix(rgb.astype(np.float32),(81,81),tuple(bc.astype(float)))
        patches.append((patch,bc))
    cap.release()
    return samples,tracks,patches

def feather_replace(base, other, center, radius):
    yy,xx=np.mgrid[:base.shape[0],:base.shape[1]]
    d=np.sqrt(((xx-center[0])/radius[0])**2+((yy-center[1])/radius[1])**2)
    a=np.clip((1-d)*5,0,1)
    return base*(1-a[:,:,None])+other*a[:,:,None]

def simulate():
    # Equal-mass billiard impacts, no kinematic repositioning after insertion.
    # One coordinate unit is the initial cue-to-target separation.
    pos=np.array([0.,.25,1.])
    vel=np.array([.0662,0.,0.])
    diameter=.143
    dt=1/160
    states=[]
    for f in range(N):
        states.append(pos.copy())
        for sub in range(160):
            t=f+sub*dt
            pos+=vel*dt
            for j in range(3):
                for l in range(j+1,3):
                    delta=pos[l]-pos[j]
                    normal=1 if delta>=0 else -1
                    if abs(delta)<diameter and (vel[j]-vel[l])*normal>0:
                        overlap=diameter-abs(delta)
                        pos[j]-=normal*overlap/2
                        pos[l]+=normal*overlap/2
                        impulse=.9875*(vel[j]-vel[l])
                        vel[j]-=impulse
                        vel[l]+=impulse
            for j in range(3):
                if pos[j]>3.012:
                    pos[j]=6.024-pos[j]
                    vel[j]=-abs(vel[j])*.78
                # Slow residual motion settles; the main shot initially slides
                # almost uniformly, then changes to a decelerating roll.
                decel=.00015 if abs(vel[j])<.01 else (.00050 if vel[j]<0 else .00002)
                vel[j]=np.sign(vel[j])*max(0,abs(vel[j])-decel*dt)
    return np.array(states)

def add_patch(canvas, residual, center, factor):
    # Subpixel affine transport of a photographed ball and its contact shadow.
    h,w=residual.shape[:2]
    x0=max(0,int(center[0]-w*factor/2-3)); x1=min(canvas.shape[1],int(center[0]+w*factor/2+4))
    y0=max(0,int(center[1]-h*factor/2-3)); y1=min(canvas.shape[0],int(center[1]+h*factor/2+4))
    mat=np.array([[factor,0,center[0]-x0-factor*(w-1)/2],
                  [0,factor,center[1]-y0-factor*(h-1)/2]],np.float32)
    warped=cv2.warpAffine(residual,mat,(x1-x0,y1-y0),flags=cv2.INTER_LINEAR,borderMode=cv2.BORDER_CONSTANT)
    canvas[y0:y1,x0:x1]+=warped

def soften_support(shape, rx, ry, cy=0):
    yy,xx=np.mgrid[:shape[0],:shape[1]]
    d=np.sqrt(((xx-(shape[1]-1)/2)/rx)**2+((yy-(shape[0]-1)/2-cy)/ry)**2)
    return np.clip((1-d)*6,0,1).astype(np.float32)

    source=np.load(ROOT/'source_pixels.npz')['rgb']
    # Real cloth pixels from later source frames create an unobstructed plate.
    plate=samples[0].astype(np.float32)
    plate=feather_replace(plate,samples[20].astype(np.float32),P0,(52,57))

    # White inset with a black 1, drawn in the sphere's upper-front face.
