    below=np.clip((gy-cy-17)/7,0,1)
    ratio=np.clip(patch/np.maximum(ground,1),.52,1.16)
    shadow=(ratio-1)*shadow_region[...,None]*below[...,None]*(1-alpha[...,None])
    return patch,alpha,shadow,(x0,y0),center

def decay(t,x,v,k):
    return x+v*(-np.expm1(-k*t))/k

    high=read_video()

    # This identifies the weakly elastic, damped head-on stone collision.

        u.append(np.polyfit(np.arange(25,38),p[25:38,j,0]/5,1)[0])
        fit,_=curve_fit(decay,np.arange(39,96)-38,p[39:96,j,0]/5,
                        p0=[p[38,j,0]/5,[-.2,.2][j],.008])
        w.append(fit[1]); k.append(fit[2])
    original_mass_ratio=(u[0]-w[0])/(w[1]-u[1])
    e=np.clip((w[1]-w[0])/(u[0]-u[1]),0,1)
    m=original_mass_ratio*.2
    v=np.array([((1-e*m)*u[0]+m*(1+e)*u[1])/(1+m),
                ((m-e)*u[1]+(1+e)*u[0])/(1+m)])
    # Camera projection of the same straight, planar line of travel.
    slope=[np.polyfit(p[15:38,j,0],p[15:38,j,1],1)[0] for j in range(2)]
    sprites=[make_sprite(high[37],bg,p[37,j],j) for j in range(2)]
    frames=low.copy()
    trajectories=[]
    for i in range(39,96):
        t=i-38
        dest=[]
        for j in range(2):
            dx=v[j]*(-np.expm1(-k[j]*t))/k[j]*5
            dest.append(p[38,j]+np.array([dx,slope[j]*dx]))
        trajectories.append(dest)
        # All lighting and markings outside the footprints remain source pixels.
        canvas=bg.astype(float).copy()
        # First the semitransparent contact reflections.
        for j in range(2):
            patch,a,shadow,origin,reference=sprites[j]
            delta=dest[j]-reference
            M=np.float32([[1,0,origin[0]+delta[0]],[0,1,origin[1]+delta[1]]])
            field=cv2.warpAffine(shadow.astype(np.float32),M,(1280,720),flags=cv2.INTER_LINEAR)
            canvas*=1+field
        # Then the original source stone pixels, without rescaling them.
        for j in range(2):
            patch,a,shadow,origin,reference=sprites[j]
            delta=dest[j]-reference
            M=np.float32([[1,0,origin[0]+delta[0]],[0,1,origin[1]+delta[1]]])
            premult=cv2.warpAffine((patch*a[...,None]).astype(np.float32),M,(1280,720),flags=cv2.INTER_LINEAR)
            aa=cv2.warpAffine(a.astype(np.float32),M,(1280,720),flags=cv2.INTER_LINEAR)
            canvas=canvas*(1-aa[...,None])+premult
        rendered=np.array(Image.fromarray(np.clip(canvas,0,255).round().astype(np.uint8)).resize((256,144),Image.Resampling.LANCZOS))
        roi=np.zeros((144,256),bool)
        for point in list(p[i])+dest:

            x0=max(0,int(np.floor(x-9)));x1=min(256,int(np.ceil(x+10)))
            y0=max(0,int(np.floor(y-5)));y1=min(144,int(np.ceil(y+13)))
            roi[y0:y1,x0:x1]=True
        frames[i,roi]=rendered[roi]
    artifact_io.save(frames,ROOT)
    # A conventional MP4 is supplied in addition to the required RGB artifacts.
    writer=cv2.VideoWriter(str(ROOT/'edited_video.mp4'),cv2.VideoWriter_fourcc(*'mp4v'),24,(256,144))
    for f in frames:writer.write(cv2.cvtColor(f,cv2.COLOR_RGB2BGR))
    writer.release()
    print('Saved 96 frames. Inferred original yellow/red mass ratio:',original_mass_ratio)
    print('Restitution:',e,'Incoming pixel velocities:',u,'New velocities:',v)

if __name__=='__main__':main()
