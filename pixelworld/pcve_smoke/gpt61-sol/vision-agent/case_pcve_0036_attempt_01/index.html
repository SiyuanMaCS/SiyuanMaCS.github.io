+"""Counterfactual edit: 1/8 of the original pin-floor friction.
+
+The source is a locked-camera view. Its pin silhouettes, shading and reflected
+light are used as moving pixel layers. A low-friction displacement is integrated
+from the instant of ball contact, with no changes to the pre-contact scene.
+No external renderer or model is used.
+"""
+from pathlib import Path
+import cv2
+import numpy as np
+from PIL import Image, ImageDraw
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+
+def load_video():
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    frames = []
+    while True:
+        ok, frame = cap.read()
+        if not ok:
+            break
+        frames.append(cv2.cvtColor(frame, cv2.COLOR_BGR2RGB))
+    cap.release()
+    return np.stack(frames)
+
+
+def keep_components(mask, minimum=5):
+    num, labels, stats, _ = cv2.connectedComponentsWithStats(mask.astype(np.uint8))
+    result = np.zeros_like(mask, dtype=np.uint8)
+    for k in range(1, num):
+        if stats[k, cv2.CC_STAT_AREA] >= minimum:
+            result[labels == k] = 1
+    return result
+
+
+def pin_layer(frame, plate, which, f):
+    """An antialiased matte, from the white pins' actual source pixels."""
+    rgb = frame.astype(np.float32)
+    delta = rgb - plate.astype(np.float32)
+    r, g, b = rgb.transpose(2,0,1)
+    mask = (delta[:,:,0] > 17) & (r > 40) & (g < 1.68*(r+1)) & (b-g < 26)
+    region = np.zeros(mask.shape, np.uint8)
+    if which == 0:
+        region[353:395, 565:634] = 1
+    elif which == 1:
+        region[353:395, 646:715] = 1
+    else:
+        region[354:393, 632:655] = 1
+        # The ball's blue pixels are not included in this white-pin layer.
+    mask = keep_components(mask & (region>0), 5)
+    # Include softly shaded edges without pulling in the blue bowling ball.
+    a = cv2.GaussianBlur(cv2.dilate(mask, np.ones((3,3),np.uint8)), (3,3), .52)
+    # cv2 uint8 blur would quantize the alpha; compute on float instead.
+    a = cv2.GaussianBlur(cv2.dilate(mask, np.ones((3,3),np.uint8)).astype(np.float32), (3,3), .48)
+    a *= (region>0)
+    # A one-pixel surround from the source gives matched edge antialiasing.
+    return rgb, a
+
+
+def reflection_mask(frame, plate, which):
+    rgb = frame.astype(np.float32)
+    diff = rgb-plate.astype(np.float32)
+    r,g,b = rgb.transpose(2,0,1)
+    mask = (diff[:,:,0]>9) & (r>45) & (g<1.75*(r+1)) & (b-g<25)
+    reg = np.zeros(mask.shape, np.uint8)
+    if which==0:
+        reg[394:440,565:633]=1
+    elif which==1:
+        reg[394:440,647:716]=1
+    else:
+        reg[394:440,633:647]=1
+    m=keep_components(mask & (reg>0), 7)
+    a=cv2.GaussianBlur(cv2.dilate(m,np.ones((3,3),np.uint8)).astype(np.float32),(5,5),.7)
+    return a*(reg>0)
+
+
+def warp_layer(image, alpha, matrix, size):
+    # Premultiplied alpha interpolation keeps the exact source silhouette.
+    wa = cv2.warpAffine(alpha, matrix, size, flags=cv2.INTER_LINEAR)
+    wc = cv2.warpAffine(image*alpha[:,:,None], matrix, size, flags=cv2.INTER_LINEAR)
+    return wc,wa
+
+
+def make_video():
+    src = load_video()
+    factual = np.load(ROOT/'source_pixels.npz')['rgb']
+    # Everything but a tiny area surrounding the pins is copied byte-for-byte.
+    result = factual.copy()
+    H,W=src.shape[1:3]
+    plate=src[0].copy()
+    # This end-of-shot patch supplies the revealed empty back wall and floor;
+    # the original first frame supplies the side areas before any pins land there.
+    plate[350:445,615:664] = src[86,350:445,615:664]
+    size=(W,H)
+
+    # Reference centers, measured on the source pin silhouettes. The bodies
+    # preserve source angular motion, while their new COMs retain more speed.
+    left_x=[628,624,620,616,612,609,606,603,600,598,596,594,592,590,589]
+    right_x=[651,656,660,663,667,671,675,678,682,685,688,690,692,694,695]
+    center_y=[371,371,371,371,372,373,373,372,372,372,371,372,373,374,375]
+
+    for f in range(37,96):
+        original=src[f].astype(np.float32)
+        clean=original.copy()
+        # Remove the factual pin set, but preserve the ball and its reflection.
+        originals=[pin_layer(src[f],plate,k,f) for k in range(3)]
+        total=np.maximum.reduce([p[1] for p in originals])
+        clean=clean*(1-total[:,:,None])+plate.astype(float)*total[:,:,None]
+        rm=[reflection_mask(src[f],plate,k) for k in range(3)]
+        refl=np.maximum.reduce(rm)
+        clean=clean*(1-refl[:,:,None])+plate.astype(float)*refl[:,:,None]
+
+        for k in (0,1):
+            n=f-37
+            # The first few collision frames retain the contact history. After
+            # that, a Coulomb-drag correction accumulates as (mu-mu/8)*t^2/2.
+            pose=min(f,47)
+            idx=pose-37
+            source_x=(left_x if k==0 else right_x)[idx]
+            sign=-1 if k==0 else 1
+            displacement=.115*n*n
+            if f<=47:
+                new_x=source_x+sign*displacement
+            else:
+                # Sliding speed at frame 47 is not suddenly clamped to rest.
+                new_x=(left_x if k==0 else right_x)[10]+sign*(11.5+3.6*(f-47)-.024*(f-47)**2)
+            # Falling into the gutters occludes the outgoing pins; low friction
+            # makes them clear the pin deck instead of stopping at its edges.
+            edge=578 if k==0 else 699
+            distance=max(0,(edge-new_x) if k==0 else (new_x-edge))
+            drop=0.40*distance
+            opacity=np.clip(1-distance/18,0,1)
+            if opacity<=0:
+                continue
+            scale=1-.007*distance
+            rgb,a=pin_layer(src[pose],plate,k,pose)
+            matrix=np.array([[scale,0,new_x-scale*source_x], [0,scale,385*(1-scale)+drop]],dtype=np.float32)
+            wc,wa=warp_layer(rgb,a,matrix,size)
+            # The far lip of the gutter partly hides pins below lane height.
+            wa*=opacity;wc*=opacity
+            clean=clean*(1-wa[:,:,None])+wc
+            # The floor reflection follows the same lateral motion. It vanishes
+            # at the gutter because the gutter is dark rather than polished wood.
+            ref_a=reflection_mask(src[pose],plate,k)
+            ref_delta=src[pose].astype(np.float32)-plate.astype(np.float32)
+            ref_move=np.array([[1,0,new_x-source_x],[0,1,0]],np.float32)
+            ref_w=cv2.warpAffine(ref_delta*ref_a[:,:,None],ref_move,size)
+            fade=float(np.clip(1-distance/5,0,1))
+            clean+=ref_w*fade
+
+        # The central pin transfers most of its energy straight down-lane.
+        # Lower drag allows it to slide farther into the dark pit while the
+        # source ball's unaffected axial flight remains intact.
+        n=f-37
+        if f<52:
+            pose=min(f,49)
+            rgb,a=pin_layer(src[pose],plate,2,pose)
+            scale=1-.013*n
+            dy=-.20*n-.012*n*n
+            matrix=np.array([[scale,0,640*(1-scale)],[0,scale,372*(1-scale)+dy]],np.float32)
+            wc,wa=warp_layer(rgb,a,matrix,size)
+            fade=float(np.clip((52-f)/5,0,1))
+            clean=clean*(1-wa[:,:,None]*fade)+wc*fade
+            if f<43:
+                ra=reflection_mask(src[pose],plate,2)
+                refl_delta=src[pose].astype(np.float32)-plate.astype(np.float32)
+                m=np.array([[scale,0,640*(1-scale)],[0,scale,410*(1-scale)-dy]],np.float32)
+                wr=cv2.warpAffine(refl_delta*ra[:,:,None],m,size)
+                clean+=wr*max(0,1-n/7)
+
+        edit=np.asarray(Image.fromarray(np.clip(clean,0,255).astype(np.uint8)).resize((256,144),Image.Resampling.LANCZOS))
+        # Restrict the output edit to the only affected image region.
+        result[f,69:90,108:146]=edit[69:90,108:146]
+    return result
+
+
+if __name__=='__main__':
+    frames=make_video()
+    artifact_io.save(frames, ROOT)
+    # A closer view is useful for checking small pins in a distant scene.
+    sh=Image.new('RGB',(960,480),(25,28,35)); d=ImageDraw.Draw(sh)
+    for j,f in enumerate([36,38,40,42,44,46,48,50,52,56,68,95]):
+        patch=Image.fromarray(frames[f]).crop((108,68,146,90)).resize((240,132),Image.Resampling.NEAREST)
+        x=(j%4)*240;y=(j//4)*160
+        sh.paste(patch,(x,y));d.text((x+3,y+135),str(f),fill='white')
+    sh.save(ROOT/'.tmp'/'candidate_detail.png')

    num, labels, stats, _ = cv2.connectedComponentsWithStats(mask.astype(np.uint8))

def warp_layer(image, alpha, matrix, size):
    # Premultiplied alpha interpolation keeps the exact source silhouette.
    wa = cv2.warpAffine(alpha, matrix, size, flags=cv2.INTER_LINEAR)
    wc = cv2.warpAffine(image*alpha[:,:,None], matrix, size, flags=cv2.INTER_LINEAR)
    return wc,wa
