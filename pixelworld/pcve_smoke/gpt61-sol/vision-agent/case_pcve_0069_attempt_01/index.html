                if theta>=np.pi/2:
                    theta=np.pi/2;hit=i;break
        else:
            t=(i-hit)/24
            theta=np.pi/2-.028*np.exp(-t*15)*max(0,np.sin(t*39))
        angles.append(theta)

    def shadow(theta):
        p,uv,project,_=geom(par,theta)
        light=np.array([-.19,.9,1.])
        floor=p-p[:,2,None]*light[None,:]
        outline=project(floor)/5
        m=np.zeros((h,w),np.float32)
        cv2.fillConvexPoly(m,cv2.convexHull(outline.astype(np.float32)).round().astype(np.int32),1)
        soft=gaussian_filter(m,2.7)
        tight=gaussian_filter(m,.7)
        return np.clip(.19*soft+.13*tight,0,.42)
    # Preserve the original first shadow at the transition, then smoothly adapt it.
    initial_shadow=shadow(th18)
    residual=np.clip(1-np.mean(early[18]/np.maximum(base,1),axis=-1),0,.65)
    residual*=first
    residual[(yy<97)&(xx<104)]=0
    frames=[]
    for i in range(96):
        if i<19:
            frame=early[i]
        else:
            th=angles[i]
            sh=shadow(th)
            ramp=min(1,(i-18)/7)
            sh=sh*ramp+residual*(1-ramp)
            frame=base*(1-sh[:,:,None])
            canvas=cv2.resize(frame,(1280,720),interpolation=cv2.INTER_CUBIC)
            p,uv,project,axes=geom(par,th)
            R,V,F=axes
            az,el,d,f,T,W,_=par
            origin=((339-640)*d/f)*R+((505-360)*d/f)*V+d*F
            # Visibility is evaluated using the perspective camera, not a screen rotation.
            normal=np.array([np.cos(th),0,-np.sin(th)])
            front_gain=1+.12*min(1,max(0,(th-th18)/(np.pi/2-th18)))
            if np.dot(normal,origin+np.mean(p[[0,2,5,4]],axis=0))<0:
                render_face(canvas,tex_dots,uv[[0,4,5,2]])
            else:
                render_face(canvas,tex_back,uv[[1,7,6,3]],.86+.14*min(1,max(0,(th-.65)/.92)))
            render_face(canvas,tex_top,uv[[2,3,6,5]],1-.25*(th-th18)/(np.pi/2-th18))
            render_face(canvas,tex_front,uv[[0,1,3,2]],front_gain)
            frame=cv2.resize(canvas,(w,h),interpolation=cv2.INTER_AREA)
        frames.append(np.clip(frame,0,255).round().astype(np.uint8))
    frames=np.stack(frames)
    artifact_io.save(frames,ROOT)
    # Optional conventional playback copy, in addition to the required RGB artifacts.
    writer=cv2.VideoWriter(str(ROOT/'edited_video.mp4'),cv2.VideoWriter_fourcc(*'mp4v'),24,(w,h))
    for frame in frames:writer.write(cv2.cvtColor(frame,cv2.COLOR_RGB2BGR))
    writer.release()
