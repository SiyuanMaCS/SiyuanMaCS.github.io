+"""Counterfactual initial-velocity edit, using measured source pixels only.
+
+The ascending motion is fitted to a constant-acceleration trajectory.  Only
+its initial velocity is multiplied by .95; the ramp's acceleration is kept.
+Source silhouettes supply the surface detail and pose of the small rigid body.
+"""
+from pathlib import Path
+import numpy as np
+import cv2
+from PIL import Image
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+def source_high():
+    cap=cv2.VideoCapture(str(ROOT/'source_video.mp4'))
+    images=[]
+    while True:
+        ok,b=cap.read()
+        if not ok: break
+        images.append(cv2.cvtColor(b,cv2.COLOR_BGR2RGB))
+    cap.release()
+    return np.asarray(images)
+
+
+def component_mask(image, bg, frame):
+    diff=np.max(np.abs(image.astype(np.int16)-bg.astype(np.int16)),axis=2)
+    mask=(diff>22).astype(np.uint8)
+    mask[:300]=0
+    n,lab,stats,ct=cv2.connectedComponentsWithStats(mask)
+    large=[k for k in range(1,n) if stats[k,4]>3000]
+    k=min(large,key=lambda k:stats[k,1])
+    obj=(lab==k).astype(np.uint8)
+    # Before the landing, separate the projected ground shadow from the body.
+    if frame==18:
+        poly=np.array([[83,595],[95,562],[127,530],[169,486],[211,477],
+                       [229,493],[246,492],[253,515],[244,540],[221,554],
+                       [200,560],[183,583],[150,599],[113,618],[91,617]])
+        limit=np.zeros_like(obj);cv2.fillPoly(limit,[poly],1);obj*=limit
+    elif frame==19:
+        poly=np.array([[54,608],[60,577],[91,556],[129,534],[173,506],
+                       [201,500],[208,517],[231,519],[239,537],[226,560],
+                       [197,577],[160,589],[122,611],[89,622],[63,624]])
+        limit=np.zeros_like(obj);cv2.fillPoly(limit,[poly],1);obj*=limit
+    elif frame>=20:
+        # Grounded silhouette: do not absorb the shadow into the color layer.
+        obj[625:]=0
+    obj=cv2.morphologyEx(obj,cv2.MORPH_CLOSE,np.ones((3,3),np.uint8))
+    contours,_=cv2.findContours(obj,cv2.RETR_EXTERNAL,cv2.CHAIN_APPROX_SIMPLE)
+    cv2.drawContours(obj,contours,-1,1,cv2.FILLED)
+    # Small dilation fills codec fringes.  Soft edges are handled at native res.
+    obj=cv2.dilate(obj,np.ones((3,3),np.uint8))
+    alpha=cv2.GaussianBlur(obj.astype(np.float32),(3,3),.55)
+    ys,xs=np.nonzero(obj)
+    return alpha, (float(xs.mean()),float(ys.mean())), (int(xs.min()),int(ys.min()),int(xs.max()),int(ys.max()))
+
+
+def main():
+    high=source_high(); H,W=high.shape[1:3]
+    low=np.load(ROOT/'source_pixels.npz')['rgb']
+    # The set is stationary.  These two source instants expose every pixel of
+    # the stage.  The seam lies on unchanging scenery and is identical in both.
+    bg=high[-1].copy();bg[:,:270]=high[0,:,:270]
+    masks=[];centers=[];bounds=[]
+    for i in range(22):
+        a,c,b=component_mask(high[i],bg,i)
+        masks.append(a);centers.append(c);bounds.append(b)
+    centers=np.array(centers)
+
+    # Fit a fixed patch in the car body. Wheel rotation does not bias velocity.
+    templ=high[0,528:557,815:862]
+    points=[]
+    for i in range(11):
+        corr=cv2.matchTemplate(high[i],templ,cv2.TM_CCOEFF_NORMED)
+        points.append(cv2.minMaxLoc(corr)[3])
+    points=np.array(points,float)
+    acc2,v0,x0=np.polyfit(np.arange(11),points[:,0],2)
+    slope=np.polyfit(points[:10,0],points[:10,1],1)[0]
+    departure=10.30
+    distance=v0*departure+acc2*departure**2
+    new_departure=float(np.min(np.roots([acc2,.95*v0,-distance])))
+    old_v=v0+2*acc2*departure
+    new_v=.95*v0+2*acc2*new_departure
+    dv=new_v-old_v
+
+    # Foreground side of the fixed timber and support. This maintains the
+    # original depth ordering when the car's back passes behind the pedestal.
+    occ=np.zeros((H,W),np.uint8)
+    cv2.fillPoly(occ,[np.array([[297,386],[987,632],[976,674],[280,430]])],1)
+    cv2.fillPoly(occ,[np.array([[250,432],[277,412],[355,424],[376,469],
+                              [371,646],[352,659],[249,654]])],1)
+    occ=cv2.GaussianBlur(occ.astype(np.float32),(3,3),.45)
+
+    # Pixel-derived shadow transmissions.  They are modulated, not pasted
+    # opaque, so the unchanging asphalt texture is preserved.
+    shadows=[]
+    for i in range(22):
+        ratio=np.mean((high[i].astype(float)+2)/(bg.astype(float)+2),axis=2)
+        sh=np.clip(1-ratio,0,.72)
+        sh[masks[i]>.05]=0
+        sh[:570]=0
+        sh[:,350:]=0
+        sh[occ>.1]=0
+        sh[sh<.06]=0
+        shadows.append(cv2.GaussianBlur(sh.astype(np.float32),(5,5),.7))
+
+    def warp(a,dx,dy):
+        return cv2.warpAffine(a,np.array([[1,0,dx],[0,1,dy]],np.float32),
+                             (W,H),flags=cv2.INTER_LINEAR,borderMode=cv2.BORDER_CONSTANT)
+
+    def layers_at(q,dx,dy):
+        q=float(np.clip(q,0,21));i=int(q);j=min(i+1,21);r=q-i
+        # Align both integer-time exemplars to a single interpolated position
+        # before blending. This prevents two separate outlines/ghost cars.
+        c=(1-r)*centers[i]+r*centers[j]
+        prem=np.zeros((H,W,3),np.float32);alpha=np.zeros((H,W),np.float32)
+        sh=np.zeros((H,W),np.float32)
+        for k,weight in [(i,1-r),(j,r)]:
+            if weight<1e-6:continue
+            tx=c[0]-centers[k,0]+dx;ty=c[1]-centers[k,1]+dy
+            ak=warp(masks[k],tx,ty)
+            col=warp(high[k].astype(np.float32)*masks[k][...,None],tx,ty)
+            prem+=weight*col;alpha+=weight*ak
+            # Shadow stays on the pavement, rather than following vertical lift.
+            sh+=weight*warp(shadows[k],tx,0)
+        alpha*=1-occ;prem*=1-occ[...,None]
+        canvas=bg.astype(np.float32)*(1-sh[...,None])
+        return canvas*(1-alpha[...,None])+prem
+
+    # The smaller exit speed adds a downward and rightward displacement to
+    # the observed free-flight. Near the pedestal, contact deflects the rear
+    # outwards; the toy lands immediately beside its left face, not inside it.
+    e_land=7.45
+    impact_time=new_departure+e_land
+    contact_start=4.5
+    final_dx=17.0
+    frames=[]
+    for t in range(96):
+        if t==0:
+            frames.append(low[0]);continue
+        if t<=new_departure:
+            disp=.95*v0*t+acc2*t*t
+            q=float(np.min(np.roots([acc2,v0,-disp])))
+            frame=layers_at(q,0,0)
+        else:
+            e=t-new_departure
+            if e<e_land:
+                q=departure+e
+                dx=dv*e
+                dy=slope*dv*e
+                if e>contact_start:
+                    p=np.clip((e-contact_start)/(e_land+2-contact_start),0,1)
+                    s=p*p*(3-2*p)
+                    dx=dx*(1-s)+final_dx*s
+                frame=layers_at(q,dx,dy)
+            elif t<impact_time+2.2:
+                # Front-wheel impact followed by the rear wheel settling.
+                p=(t-impact_time)/2.2
+                q=(departure+e_land)*(1-p)+21*p
+                dx0=dv*e_land
+                cp=np.clip((e_land-contact_start)/(e_land+2-contact_start),0,1)
+                dx0=dx0*(1-cp*cp*(3-2*cp))+final_dx*cp*cp*(3-2*cp)
+                dx=dx0*(1-p)+final_dx*p
+                i=int(q);r=q-i;j=min(i+1,21)
+                bottom=(1-r)*bounds[i][3]+r*bounds[j][3]
+                # Maintain the front tire at its contact height as pitch settles.
+                dy=625-bottom
+                frame=layers_at(q,dx,dy)
+            else:
+                frame=layers_at(21,final_dx,0)
+        small=Image.fromarray(np.clip(frame,0,255).astype(np.uint8)).resize((256,144),Image.Resampling.LANCZOS)
+        frames.append(np.asarray(small))
+    frames=np.asarray(frames,np.uint8)
+    artifact_io.save(frames,ROOT)
+    print('Fit: v0=%.4f, acceleration=%.4f, ramp departure %.3f -> %.3f frames, exit velocity %.3f -> %.3f' %
+          (v0,2*acc2,departure,new_departure,old_v,new_v))
+    print('Saved',frames.shape)
+
+if __name__=='__main__': main()

\"\"\"Counterfactual initial-velocity edit, using measured source pixels only.

The ascending motion is fitted to a constant-acceleration trajectory.  Only
its initial velocity is multiplied by .95; the ramp's acceleration is kept.
Source silhouettes supply the surface detail and pose of the small rigid body.
\"\"\"
from pathlib import Path
import numpy as np
import cv2
from PIL import Image
import artifact_io

ROOT = Path(__file__).resolve().parent

def source_high():
    cap=cv2.VideoCapture(str(ROOT/'source_video.mp4'))
    images=[]
    while True:
        ok,b=cap.read()
        if not ok: break
        images.append(cv2.cvtColor(b,cv2.COLOR_BGR2RGB))
    cap.release()
    return np.asarray(images)


def component_mask(image, bg, frame):
    diff=np.max(np.abs(image.astype(np.int16)-bg.astype(np.int16)),axis=2)
    mask=(diff>22).astype(np.uint8)
    mask[:300]=0
    n,lab,stats,ct=cv2.connectedComponentsWithStats(mask)
    large=[k for k in range(1,n) if stats[k,4]>3000]
    k=min(large,key=lambda k:stats[k,1])
    obj=(lab==k).astype(np.uint8)
    # Before the landing, separate the projected ground shadow from the body.
    if frame==18:
        poly=np.array([[83,595],[95,562],[127,530],[169,486],[211,477],
                       [229,493],[246,492],[253,515],[244,540],[221,554],
                       [200,560],[183,583],[150,599],[113,618],[91,617]])
        limit=np.zeros_like(obj);cv2.fillPoly(limit,[poly],1);obj*=limit
    elif frame==19:
        poly=np.array([[54,608],[60,577],[91,556],[129,534],[173,506],
                       [201,500],[208,517],[231,519],[239,537],[226,560],
                       [197,577],[160,589],[122,611],[89,622],[63,624]])
        limit=np.zeros_like(obj);cv2.fillPoly(limit,[poly],1);obj*=limit
    elif frame>=20:
        # Grounded silhouette: do not absorb the shadow into the color layer.
        obj[625:]=0
    obj=cv2.morphologyEx(obj,cv2.MORPH_CLOSE,np.ones((3,3),np.uint8))
    contours,_=cv2.findContours(obj,cv2.RETR_EXTERNAL,cv2.CHAIN_APPROX_SIMPLE)
    cv2.drawContours(obj,contours,-1,1,cv2.FILLED)
    # Small dilation fills codec fringes.  Soft edges are handled at native res.
    obj=cv2.dilate(obj,np.ones((3,3),np.uint8))
    alpha=cv2.GaussianBlur(obj.astype(np.float32),(3,3),.55)
    ys,xs=np.nonzero(obj)
    return alpha, (float(xs.mean()),float(ys.mean())), (int(xs.min()),int(ys.min()),int(xs.max()),int(ys.max()))


def main():
    high=source_high(); H,W=high.shape[1:3]
    low=np.load(ROOT/'source_pixels.npz')['rgb']
    # The set is stationary.  These two source instants expose every pixel of
    # the stage.  The seam lies on unchanging scenery and is identical in both.
    bg=high[-1].copy();bg[:,:270]=high[0,:,:270]
    masks=[];centers=[];bounds=[]
    for i in range(22):
        a,c,b=component_mask(high[i],bg,i)
        masks.append(a);centers.append(c);bounds.append(b)
    centers=np.array(centers)

    # Fit a fixed patch in the car body. Wheel rotation does not bias velocity.
    templ=high[0,528:557,815:862]
    points=[]
    for i in range(11):
        corr=cv2.matchTemplate(high[i],templ,cv2.TM_CCOEFF_NORMED)
        points.append(cv2.minMaxLoc(corr)[3])
    points=np.array(points,float)
    acc2,v0,x0=np.polyfit(np.arange(11),points[:,0],2)
    slope=np.polyfit(points[:10,0],points[:10,1],1)[0]
    departure=10.30
    distance=v0*departure+acc2*departure**2
    new_departure=float(np.min(np.roots([acc2,.95*v0,-distance])))
    old_v=v0+2*acc2*departure
    new_v=.95*v0+2*acc2*new_departure
    dv=new_v-old_v

    # Foreground side of the fixed timber and support. This maintains the
    # original depth ordering when the car's back passes behind the pedestal.
    occ=np.zeros((H,W),np.uint8)
    cv2.fillPoly(occ,[np.array([[297,386],[987,632],[976,674],[280,430]])],1)
    cv2.fillPoly(occ,[np.array([[250,432],[277,412],[355,424],[376,469],
                              [371,646],[352,659],[249,654]])],1)
    occ=cv2.GaussianBlur(occ.astype(np.float32),(3,3),.45)

    # Pixel-derived shadow transmissions.  They are modulated, not pasted
    # opaque, so the unchanging asphalt texture is preserved.
    shadows=[]
    for i in range(22):
        ratio=np.mean((high[i].astype(float)+2)/(bg.astype(float)+2),axis=2)
        sh=np.clip(1-ratio,0,.72)
        sh[masks[i]>.05]=0
        sh[:570]=0
        sh[:,350:]=0
        sh[occ>.1]=0
        sh[sh<.06]=0
        shadows.append(cv2.GaussianBlur(sh.astype(np.float32),(5,5),.7))

    def warp(a,dx,dy):
        return cv2.warpAffine(a,np.array([[1,0,dx],[0,1,dy]],np.float32),
                             (W,H),flags=cv2.INTER_LINEAR,borderMode=cv2.BORDER_CONSTANT)

    def layers_at(q,dx,dy):
        q=float(np.clip(q,0,21));i=int(q);j=min(i+1,21);r=q-i
        # Align both integer-time exemplars to a single interpolated position
        # before blending. This prevents two separate outlines/ghost cars.
        c=(1-r)*centers[i]+r*centers[j]
        prem=np.zeros((H,W,3),np.float32);alpha=np.zeros((H,W),np.float32)
        sh=np.zeros((H,W),np.float32)
        for k,weight in [(i,1-r),(j,r)]:
            if weight<1e-6:continue
            tx=c[0]-centers[k,0]+dx;ty=c[1]-centers[k,1]+dy
            ak=warp(masks[k],tx,ty)
            col=warp(high[k].astype(np.float32)*masks[k][...,None],tx,ty)
            prem+=weight*col;alpha+=weight*ak
            # Shadow stays on the pavement, rather than following vertical lift.
            sh+=weight*warp(shadows[k],tx,0)
        alpha*=1-occ;prem*=1-occ[...,None]
        canvas=bg.astype(np.float32)*(1-sh[...,None])
        return canvas*(1-alpha[...,None])+prem

    # The smaller exit speed adds a downward and rightward displacement to
    # the observed free-flight. Near the pedestal, contact deflects the rear
    # outwards; the toy lands immediately beside its left face, not inside it.
    e_land=7.45
    impact_time=new_departure+e_land
    contact_start=4.5
    final_dx=17.0
    frames=[]
    for t in range(96):
        if t==0:
            frames.append(low[0]);continue
        if t<=new_departure:
            disp=.95*v0*t+acc2*t*t
            q=float(np.min(np.roots([acc2,v0,-disp])))
            frame=layers_at(q,0,0)
        else:
            e=t-new_departure
            if e<e_land:
                q=departure+e
                dx=dv*e
                dy=slope*dv*e
                if e>contact_start:
                    p=np.clip((e-contact_start)/(e_land+2-contact_start),0,1)
                    s=p*p*(3-2*p)
                    dx=dx*(1-s)+final_dx*s
                frame=layers_at(q,dx,dy)
            elif t<impact_time+2.2:
                # Front-wheel impact followed by the rear wheel settling.
                p=(t-impact_time)/2.2
                q=(departure+e_land)*(1-p)+21*p
                dx0=dv*e_land
                cp=np.clip((e_land-contact_start)/(e_land+2-contact_start),0,1)
                dx0=dx0*(1-cp*cp*(3-2*cp))+final_dx*cp*cp*(3-2*cp)
                dx=dx0*(1-p)+final_dx*p
                i=int(q);r=q-i;j=min(i+1,21)
                bottom=(1-r)*bounds[i][3]+r*bounds[j][3]
                # Maintain the front tire at its contact height as pitch settles.
                dy=625-bottom
                frame=layers_at(q,dx,dy)
            else:
                frame=layers_at(21,final_dx,0)
        small=Image.fromarray(np.clip(frame,0,255).astype(np.uint8)).resize((256,144),Image.Resampling.LANCZOS)
        frames.append(np.asarray(small))
    frames=np.asarray(frames,np.uint8)
    artifact_io.save(frames,ROOT)
    print('Fit: v0=%.4f, acceleration=%.4f, ramp departure %.3f -> %.3f frames, exit velocity %.3f -> %.3f' %
          (v0,2*acc2,departure,new_departure,old_v,new_v))
    print('Saved',frames.shape)

if __name__=='__main__': main()
