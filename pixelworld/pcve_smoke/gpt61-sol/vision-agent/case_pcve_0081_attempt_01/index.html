+"""Counterfactual mass edit, using source-image mattes and damped rigid motion.
+
+A mass change does not change radius, initial state or gravitational acceleration.
+The visible collision is therefore the first place the image diverges. The
+source horizontal impulse implies an apple/ball mass ratio near 1/3. Doubling
+that ratio changes the ball impulse by 1.6 and the apple recoil impulse by .8.
+No external renderer or assets are used.
+"""
+from pathlib import Path
+import cv2
+import numpy as np
+from PIL import Image
+from scipy.ndimage import gaussian_filter
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+
+def load_source():
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    images = []
+    while True:
+        ok, frame = cap.read()
+        if not ok:
+            break
+        images.append(cv2.cvtColor(frame, cv2.COLOR_BGR2RGB))
+    cap.release()
+    return np.array(images), np.load(ROOT / 'source_pixels.npz')['rgb']
+
+
+def feather_region(x, y, cx, cy, rx, ry, width=2):
+    r = np.sqrt(((x-cx)/rx)**2 + ((y-cy)/ry)**2)
+    return np.clip((1-r)*min(rx, ry)/width, 0, 1)
+
+
+def patch_bounds(cx, cy, halfx, halfy, shape):
+    return (max(0, int(np.floor(cx-halfx))), max(0, int(np.floor(cy-halfy))),
+            min(shape[1], int(np.ceil(cx+halfx))), min(shape[0], int(np.ceil(cy+halfy))))
+
+
+def solid_component(mask):
+    n, labels, stats, _ = cv2.connectedComponentsWithStats(mask.astype(np.uint8))
+    if n < 2:
+        return np.zeros(mask.shape, np.float32)
+    idx = 1 + np.argmax(stats[1:, 4])
+    comp = (labels == idx).astype(np.uint8)
+    contours, _ = cv2.findContours(comp, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
+    out = np.zeros_like(comp)
+    cv2.drawContours(out, contours, -1, 1, cv2.FILLED)
+    out = cv2.morphologyEx(out, cv2.MORPH_CLOSE, np.ones((3,3), np.uint8))
+    return gaussian_filter(out.astype(np.float32), .5)
+
+
+def translate_layer(dst, data, alpha, bounds, dx, dy, multiply=False):
+    x0, y0, x1, y1 = bounds
+    # Warp premultiplied color, so resampled edges cannot carry grass halos.
+    h, w = dst.shape[:2]
+    tx0 = max(0, int(np.floor(x0+dx))-2)
+    ty0 = max(0, int(np.floor(y0+dy))-2)
+    tx1 = min(w, int(np.ceil(x1+dx))+2)
+    ty1 = min(h, int(np.ceil(y1+dy))+2)
+    if tx1 <= tx0 or ty1 <= ty0:
+        return
+    size = (tx1-tx0, ty1-ty0)
+    mat = np.array([[1,0,x0+dx-tx0], [0,1,y0+dy-ty0]], np.float32)
+    aa = cv2.warpAffine(alpha, mat, size, flags=cv2.INTER_LINEAR)
+    roi = dst[ty0:ty1,tx0:tx1]
+    if multiply:
+        roi *= (1-aa[:,:,None])
+    else:
+        premul = data.astype(np.float32)*alpha[:,:,None]
+        warped = cv2.warpAffine(premul, mat, size, flags=cv2.INTER_LINEAR)
+        roi[:] = roi*(1-aa[:,:,None]) + warped
+
+
+def get_ball_tracks(hi):
+    xs = []
+    ys = []
+    for im in hi:
+        m = ((im[:,:,0]>120)&(im[:,:,1]>120)&(im[:,:,2]>120)).astype(np.uint8)
+        m[:330] = 0
+        m[475:] = 0
+        n, labels, stats, _ = cv2.connectedComponentsWithStats(m)
+        j = 1 + np.argmax(stats[1:,4])
+        x,y,w,h,_ = stats[j]
+        xs.append((x+w/2)/5)
+        ys.append((y+h/2)/5)
+    return np.array(xs), np.array(ys)
+
+
+def get_apple_tracks(hi, bg):
+    result = {}
+    for k in range(9,39):
+        im = hi[k].astype(np.float32)
+        diff = abs(im-bg).max(axis=2)
+        mask = (diff>30)&(im[:,:,0]>im[:,:,1]+20)&(im[:,:,0]>115)
+        mask[:,:620]=False
+        n, labels, stats, _ = cv2.connectedComponentsWithStats(mask.astype(np.uint8))
+        if n < 2:
+            continue
+        j = 1+np.argmax(stats[1:,4])
+        x,y,w,h,area = stats[j]
+        if area < 90:
+            continue
+        result[k] = ((x+w/2)/5, (y+h/2)/5, w/2, h/2)
+    return result
+
+
+def extract_ball(hi, bg, bx, by, k):
+    # The contact-frame sprite is taken from immediately before impact; at
+    # this instant the ball has barely turned and the apple overlaps its shadow.
+    if k == 13:
+        k = 12
+    cx,cy = bx[k]*5, by[k]*5
+    bounds = patch_bounds(cx, cy, 68, 64, bg.shape)
+    x0,y0,x1,y1 = bounds
+    im = hi[k,y0:y1,x0:x1].astype(np.float32)
+    back = bg[y0:y1,x0:x1]
+    yy,xx = np.mgrid[y0:y1,x0:x1]
+    near = (((xx-cx)/53.5)**2+((yy-cy)/53.5)**2)<1
+    # The ball is gray/white or blue-black; grass is green. Include dark facets
+    # and fill the interior, retaining the actual polygonal silhouette.
+    non_grass = ((im[:,:,1]-im[:,:,0])<14) & ((im[:,:,1]-im[:,:,2])<18)
+    change = abs(im-back).max(2)>13
+    surface = solid_component(near & non_grass & change)
+    # Extend shadow bounding box separately so it doesn't clip behind the ball.
+    sb = patch_bounds(cx+28,cy-1,84,58,bg.shape)
+    sx0,sy0,sx1,sy1 = sb
+    src = hi[k,sy0:sy1,sx0:sx1].astype(np.float32)
+    bb = bg[sy0:sy1,sx0:sx1]
+    sy,sx = np.mgrid[sy0:sy1,sx0:sx1]
+    dark = np.clip(1-(src.mean(2)+1)/(bb.mean(2)+1),0,.65)
+    support = feather_region(sx,sy,cx+32,cy-1,60,47,3)
+    ball_area = (((sx-cx)/54)**2+((sy-cy)/54)**2)<1
+    dark[ball_area] = 0
+    dark *= support
+    dark[dark<.055] = 0
+    dark = gaussian_filter(dark,.55)
+    return im, surface, bounds, dark, sb, cx, cy
+
+
+def extract_apple(hi,bg,tracks,k):
+    cx,cy,rx,ry = tracks[k]
+    cx*=5; cy*=5
+    bounds=patch_bounds(cx,cy,rx+8,ry+8,bg.shape)
+    x0,y0,x1,y1=bounds
+    im=hi[k,y0:y1,x0:x1].astype(np.float32)
+    bb=bg[y0:y1,x0:x1]
+    yy,xx=np.mgrid[y0:y1,x0:x1]
+    near=(((xx-cx)/(rx+3))**2+((yy-cy)/(ry+3))**2)<1
+    colored=(im[:,:,0]>im[:,:,1]+9)&(im[:,:,0]>108)
+    alpha=solid_component(near & colored & (abs(im-bb).max(2)>12))
+    sb=patch_bounds(cx+10,cy+4,rx+28,ry+18,bg.shape)
+    sx0,sy0,sx1,sy1=sb
+    src=hi[k,sy0:sy1,sx0:sx1].astype(np.float32)
+    back=bg[sy0:sy1,sx0:sx1]
+    sy,sx=np.mgrid[sy0:sy1,sx0:sx1]
+    dark=np.clip(1-(src.mean(2)+1)/(back.mean(2)+1),0,.6)
+    dark*=feather_region(sx,sy,cx+10,cy+3,rx+23,ry+12,3)
+    dark[(((sx-cx)/(rx+3))**2+((sy-cy)/(ry+3))**2)<1]=0
+    dark[dark<.06]=0
+    dark=gaussian_filter(dark,.6)
+    return im,alpha,bounds,dark,sb,cx,cy
+
+
+def erase_ball(dst,bg,cx,cy):
+    bounds=patch_bounds(cx+20,cy,92,65,bg.shape)
+    x0,y0,x1,y1=bounds
+    yy,xx=np.mgrid[y0:y1,x0:x1]
+    a=np.maximum(feather_region(xx,yy,cx,cy,58,58,3),
+                 feather_region(xx,yy,cx+35,cy-1,65,53,4))
+    dst[y0:y1,x0:x1]=dst[y0:y1,x0:x1]*(1-a[:,:,None])+bg[y0:y1,x0:x1]*a[:,:,None]
+
+
+def erase_apple(dst,bg,track):
+    cx,cy,rx,ry=track;cx*=5;cy*=5
+    bounds=patch_bounds(cx+8,cy+4,rx+40,ry+25,bg.shape)
+    x0,y0,x1,y1=bounds
+    yy,xx=np.mgrid[y0:y1,x0:x1]
+    a=np.maximum(feather_region(xx,yy,cx,cy,rx+6,ry+6,2),
+                 feather_region(xx,yy,cx+13,cy+5,rx+30,ry+18,3))
+    dst[y0:y1,x0:x1]=dst[y0:y1,x0:x1]*(1-a[:,:,None])+bg[y0:y1,x0:x1]*a[:,:,None]
+
+
+def generate():
+    hi, low=load_source()
+    bg=np.median(hi[:8],axis=0).astype(np.float32)
+    # The ball's starting footprint is uncovered later. Recover its precise
+    # stationary grass texture rather than inpainting or generating new grass.
+    bg[325:480,530:770]=np.median(hi[80:96,325:480,530:770],axis=0)
+    bx,by=get_ball_tracks(hi)
+    apples=get_apple_tracks(hi,bg)
+    bcache={}
+    acache={}
+    out=low.copy()
+    impulse=1.6
+    recoil=.8
+    # Fit parameters in px/frame from factual free roll (linear drag + rolling
+    # resistance). Static friction stops the slow tail. With greater impulse,
+    # the ball rolls both farther and for longer; acceleration is not doubled.
+    damping=.0428364662
+    resistance=.00732421789
+    v0=2.5636641*impulse
+    threshold=.25
+    stop=np.log((v0+resistance/damping)/(threshold+resistance/damping))/damping
+    new_ball=[]
+    new_apple=[]
+    for k in range(96):
+        if k < 13:
+            new_ball.append(bx[k]);new_apple.append(None)
+            continue
+        if k == 13:
+            cbx=bx[0]+impulse*(bx[13]-bx[0])
+        else:
+            t=min(k-14,stop)
+            f=(1-np.exp(-damping*t))/damping
+            cbx=bx[0]+impulse*(bx[14]-bx[0])-v0*f+resistance/damping*(t-f)
+        new_ball.append(cbx)
+        if k == 13:
+            ax0,ay0,_,_=apples[12]
+            ax,ay,_,_=apples[13]
+            cax=ax0+recoil*(ax-ax0)
+            # Less upward recoil means it descends slightly more at impact.
+            cay=ay+1.65
+        else:
+            n=k-14
+            factual_x=145.096585+176.186890*(1-np.exp(-.0417388721*n))
+            cax=apples[12][0]+recoil*(factual_x-apples[12][0])
+            cay=85.6-.8*(1-np.exp(-n*.06))
+        new_apple.append((cax,cay))
+        dst=hi[k].astype(np.float32).copy()
+        erase_ball(dst,bg,bx[k]*5,by[k]*5)
+        if k in apples:
+            erase_apple(dst,bg,apples[k])
+        # Rotation donor advances with the fractional progress through the
+        # longer counterfactual roll, including the initial sliding regime.
+        source_equivalent=bx[0]+(cbx-bx[0])/impulse
+        ref=int(np.argmin(abs(bx-source_equivalent)))
+        if k==13: ref=12
+        if ref not in bcache:
+            bcache[ref]=extract_ball(hi,bg,bx,by,ref)
+        data,alpha,bounds,shadow,sb,cx,cy=bcache[ref]
+        dx=cbx*5-cx
+        dy=by[ref]*5-cy
+        translate_layer(dst,None,shadow,sb,dx,dy,True)
+        translate_layer(dst,data,alpha,bounds,dx,dy)
+        if cax < 262:
+            valid=[j for j in apples if 14<=j<=36]
+            aref=13 if k==13 else min(valid,key=lambda j:abs(apples[j][0]-cax))
+            if aref not in acache:
+                acache[aref]=extract_apple(hi,bg,apples,aref)
+            data,alpha,bounds,shadow,sb,cx,cy=acache[aref]
+            dx=cax*5-cx;dy=cay*5-cy
+            # At contact keep the projected ground shadow vertically fixed.
+            sdy=0 if k==13 else dy
+            translate_layer(dst,None,shadow,sb,dx,sdy,True)
+            translate_layer(dst,data,alpha,bounds,dx,dy)
+        rgb=np.clip(dst+.5,0,255).astype(np.uint8)
+        out[k]=np.array(Image.fromarray(rgb).resize((256,144),Image.Resampling.LANCZOS))
+    artifact_io.save(out,ROOT)
+    np.savez_compressed(ROOT/'.tmp'/'counterfactual_tracks.npz',ball=np.array(new_ball),apple=np.array([p if p else (np.nan,np.nan) for p in new_apple]))
+    print('Saved:',out.shape,'ball final x:',round(new_ball[-1],2),'stop frame:',round(stop+14,1))
+
+
+if __name__=='__main__':
+    generate()

\"\"\"Counterfactual mass edit, using source-image mattes and damped rigid motion.

A mass change does not change radius, initial state or gravitational acceleration.
The visible collision is therefore the first place the image diverges. The
source horizontal impulse implies an apple/ball mass ratio near 1/3. Doubling
that ratio changes the ball impulse by 1.6 and the apple recoil impulse by .8.

from pathlib import Path
import cv2
import numpy as np
from PIL import Image
from scipy.ndimage import gaussian_filter
import artifact_io

ROOT = Path(__file__).resolve().parent


def load_source():
    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
    images = []
    while True:
        ok, frame = cap.read()
        if not ok:
            break
        images.append(cv2.cvtColor(frame, cv2.COLOR_BGR2RGB))
    cap.release()
    return np.array(images), np.load(ROOT / 'source_pixels.npz')['rgb']


def feather_region(x, y, cx, cy, rx, ry, width=2):
    r = np.sqrt(((x-cx)/rx)**2 + ((y-cy)/ry)**2)
    return np.clip((1-r)*min(rx, ry)/width, 0, 1)


def patch_bounds(cx, cy, halfx, halfy, shape):
    return (max(0, int(np.floor(cx-halfx))), max(0, int(np.floor(cy-halfy))),
            min(shape[1], int(np.ceil(cx+halfx))), min(shape[0], int(np.ceil(cy+halfy))))


def solid_component(mask):
    n, labels, stats, _ = cv2.connectedComponentsWithStats(mask.astype(np.uint8))
    if n < 2:
        return np.zeros(mask.shape, np.float32)
    idx = 1 + np.argmax(stats[1:, 4])
    comp = (labels == idx).astype(np.uint8)
    contours, _ = cv2.findContours(comp, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
    out = np.zeros_like(comp)
    cv2.drawContours(out, contours, -1, 1, cv2.FILLED)
    out = cv2.morphologyEx(out, cv2.MORPH_CLOSE, np.ones((3,3), np.uint8))
    return gaussian_filter(out.astype(np.float32), .5)


def translate_layer(dst, data, alpha, bounds, dx, dy, multiply=False):
    x0, y0, x1, y1 = bounds
    # Warp premultiplied color, so resampled edges cannot carry grass halos.
    h, w = dst.shape[:2]
    tx0 = max(0, int(np.floor(x0+dx))-2)
    ty0 = max(0, int(np.floor(y0+dy))-2)
    tx1 = min(w, int(np.ceil(x1+dx))+2)
    ty1 = min(h, int(np.ceil(y1+dy))+2)
    if tx1 <= tx0 or ty1 <= ty0:
        return
    size = (tx1-tx0, ty1-ty0)
    mat = np.array([[1,0,x0+dx-tx0], [0,1,y0+dy-ty0]], np.float32)
    aa = cv2.warpAffine(alpha, mat, size, flags=cv2.INTER_LINEAR)
    roi = dst[ty0:ty1,tx0:tx1]
    if multiply:
        roi *= (1-aa[:,:,None])
    else:
        premul = data.astype(np.float32)*alpha[:,:,None]
        warped = cv2.warpAffine(premul, mat, size, flags=cv2.INTER_LINEAR)
        roi[:] = roi*(1-aa[:,:,None]) + warped


def get_ball_tracks(hi):
    xs = []
    ys = []
    for im in hi:
        m = ((im[:,:,0]>120)&(im[:,:,1]>120)&(im[:,:,2]>120)).astype(np.uint8)
        m[:330] = 0
        m[475:] = 0
        n, labels, stats, _ = cv2.connectedComponentsWithStats(m)
        j = 1 + np.argmax(stats[1:,4])
        x,y,w,h,_ = stats[j]
        xs.append((x+w/2)/5)
        ys.append((y+h/2)/5)
    return np.array(xs), np.array(ys)


def get_apple_tracks(hi, bg):
    result = {}
    for k in range(9,39):
        im = hi[k].astype(np.float32)
        diff = abs(im-bg).max(axis=2)
        mask = (diff>30)&(im[:,:,0]>im[:,:,1]+20)&(im[:,:,0]>115)
        mask[:,:620]=False
        n, labels, stats, _ = cv2.connectedComponentsWithStats(mask.astype(np.uint8))
        if n < 2:
            continue
        j = 1+np.argmax(stats[1:,4])
        x,y,w,h,area = stats[j]
        if area < 90:
            continue
        result[k] = ((x+w/2)/5, (y+h/2)/5, w/2, h/2)
    return result


def extract_ball(hi, bg, bx, by, k):
    # The contact-frame sprite is taken from immediately before impact; at
    # this instant the ball has barely turned and the apple overlaps its shadow.
    if k == 13:
        k = 12
    cx,cy = bx[k]*5, by[k]*5
    bounds = patch_bounds(cx, cy, 68, 64, bg.shape)
    x0,y0,x1,y1 = bounds
    im = hi[k,y0:y1,x0:x1].astype(np.float32)
    back = bg[y0:y1,x0:x1]
    yy,xx = np.mgrid[y0:y1,x0:x1]
    near = (((xx-cx)/53.5)**2+((yy-cy)/53.5)**2)<1
    # The ball is gray/white or blue-black; grass is green. Include dark facets
    # and fill the interior, retaining the actual polygonal silhouette.
    non_grass = ((im[:,:,1]-im[:,:,0])<14) & ((im[:,:,1]-im[:,:,2])<18)
    change = abs(im-back).max(2)>13
    surface = solid_component(near & non_grass & change)
    # Extend shadow bounding box separately so it doesn't clip behind the ball.
    sb = patch_bounds(cx+28,cy-1,84,58,bg.shape)
    sx0,sy0,sx1,sy1 = sb
    src = hi[k,sy0:sy1,sx0:sx1].astype(np.float32)
    bb = bg[sy0:sy1,sx0:sx1]
    sy,sx = np.mgrid[sy0:sy1,sx0:sx1]
    dark = np.clip(1-(src.mean(2)+1)/(bb.mean(2)+1),0,.65)
    support = feather_region(sx,sy,cx+32,cy-1,60,47,3)
    ball_area = (((sx-cx)/54)**2+((sy-cy)/54)**2)<1
    dark[ball_area] = 0
    dark *= support
    dark[dark<.055] = 0
    dark = gaussian_filter(dark,.55)
    return im, surface, bounds, dark, sb, cx, cy


def extract_apple(hi,bg,tracks,k):
    cx,cy,rx,ry = tracks[k]
    cx*=5; cy*=5
    bounds=patch_bounds(cx,cy,rx+8,ry+8,bg.shape)
    x0,y0,x1,y1=bounds
    im=hi[k,y0:y1,x0:x1].astype(np.float32)
    bb=bg[y0:y1,x0:x1]
    yy,xx=np.mgrid[y0:y1,x0:x1]
    near=(((xx-cx)/(rx+3))**2+((yy-cy)/(ry+3))**2)<1
    colored=(im[:,:,0]>im[:,:,1]+9)&(im[:,:,0]>108)
    alpha=solid_component(near & colored & (abs(im-bb).max(2)>12))
    sb=patch_bounds(cx+10,cy+4,rx+28,ry+18,bg.shape)
    sx0,sy0,sx1,sy1=sb
    src=hi[k,sy0:sy1,sx0:sx1].astype(np.float32)
    back=bg[sy0:sy1,sx0:sx1]
    sy,sx=np.mgrid[sy0:sy1,sx0:sx1]
    dark=np.clip(1-(src.mean(2)+1)/(back.mean(2)+1),0,.6)
    dark*=feather_region(sx,sy,cx+10,cy+3,rx+23,ry+12,3)
    dark[(((sx-cx)/(rx+3))**2+((sy-cy)/(ry+3))**2)<1]=0
    dark[dark<.06]=0
    dark=gaussian_filter(dark,.6)
    return im,alpha,bounds,dark,sb,cx,cy


def erase_ball(dst,bg,cx,cy):
    bounds=patch_bounds(cx+20,cy,92,65,bg.shape)
    x0,y0,x1,y1=bounds
    yy,xx=np.mgrid[y0:y1,x0:x1]
    a=np.maximum(feather_region(xx,yy,cx,cy,58,58,3),
                 feather_region(xx,yy,cx+35,cy-1,65,53,4))
    dst[y0:y1,x0:x1]=dst[y0:y1,x0:x1]*(1-a[:,:,None])+bg[y0:y1,x0:x1]*a[:,:,None]

    yy,xx=np.mgrid[y0:y1,x0:x1]

def generate():
    hi, low=load_source()
    bg=np.median(hi[:8],axis=0).astype(np.float32)
    # The ball's starting footprint is uncovered later. Recover its precise
    # stationary grass texture rather than inpainting or generating new grass.
    bg[325:480,530:770]=np.median(hi[80:96,325:480,530:770],axis=0)
    bx,by=get_ball_tracks(hi)
    apples=get_apple_tracks(hi,bg)
    bcache={}
    acache={}
    out=low.copy()
    impulse=1.6
    recoil=.8
    # Fit parameters in px/frame from factual free roll (linear drag + rolling
    # resistance). Static friction stops the slow tail. With greater impulse,
    # the ball rolls both farther and for longer; acceleration is not doubled.
    damping=.0428364662
    resistance=.00732421789
    v0=2.5636641*impulse
    threshold=.25
    stop=np.log((v0+resistance/damping)/(threshold+resistance/damping))/damping
    new_ball=[]
    new_apple=[]
    for k in range(96):
        if k < 13:
            new_ball.append(bx[k]);new_apple.append(None)
            continue
        if k == 13:
            cbx=bx[0]+impulse*(bx[13]-bx[0])
        else:
            t=min(k-14,stop)
            f=(1-np.exp(-damping*t))/damping
            cbx=bx[0]+impulse*(bx[14]-bx[0])-v0*f+resistance/damping*(t-f)
        new_ball.append(cbx)
        if k == 13:
            ax0,ay0,_,_=apples[12]
            ax,ay,_,_=apples[13]
            cax=ax0+recoil*(ax-ax0)
            # Less upward recoil means it descends slightly more at impact.
            cay=ay+1.65
        else:
            n=k-14
            factual_x=145.096585+176.186890*(1-np.exp(-.0417388721*n))
            cax=apples[12][0]+recoil*(factual_x-apples[12][0])
            cay=85.6-.8*(1-np.exp(-n*.06))
        new_apple.append((cax,cay))
        dst=hi[k].astype(np.float32).copy()
        erase_ball(dst,bg,bx[k]*5,by[k]*5)
        if k in apples:
            erase_apple(dst,bg,apples[k])
        # Rotation donor advances with the fractional progress through the
        # longer counterfactual roll, including the initial sliding regime.
        source_equivalent=bx[0]+(cbx-bx[0])/impulse
        ref=int(np.argmin(abs(bx-source_equivalent)))
        if k==13: ref=12
        if ref not in bcache:
            bcache[ref]=extract_ball(hi,bg,bx,by,ref)
        data,alpha,bounds,shadow,sb,cx,cy=bcache[ref]
        dx=cbx*5-cx
        dy=by[ref]*5-cy
        translate_layer(dst,None,shadow,sb,dx,dy,True)
        translate_layer(dst,data,alpha,bounds,dx,dy)
        if cax < 262:
            valid=[j for j in apples if 14<=j<=36]
            aref=13 if k==13 else min(valid,key=lambda j:abs(apples[j][0]-cax))
            if aref not in acache:
                acache[aref]=extract_apple(hi,bg,apples,aref)
            data,alpha,bounds,shadow,sb,cx,cy=acache[aref]
            dx=cax*5-cx;dy=cay*5-cy
            # At contact keep the projected ground shadow vertically fixed.
            sdy=0 if k==13 else dy
            translate_layer(dst,None,shadow,sb,dx,sdy,True)
            translate_layer(dst,data,alpha,bounds,dx,dy)
        rgb=np.clip(dst+.5,0,255).astype(np.uint8)
        out[k]=np.array(Image.fromarray(rgb).resize((256,144),Image.Resampling.LANCZOS))
    artifact_io.save(out,ROOT)
    np.savez_compressed(ROOT/'.tmp'/'counterfactual_tracks.npz',ball=np.array(new_ball),apple=np.array([p if p else (np.nan,np.nan) for p in new_apple]))
    print('Saved:',out.shape,'ball final x:',round(new_ball[-1],2),'stop frame:',round(stop+14,1))


if __name__=='__main__':
    generate()
