+"""Source-pixel counterfactual: sixfold initial linear speed, unchanged gravity.
+
+All imagery is recovered from the supplied video.  The free-flight parabola,
+floor height, restitution, and tangential impulse are measured from that video.
+The single sphere is advanced with elementary ballistic/contact equations.
+"""
+from pathlib import Path
+import json
+import cv2
+import numpy as np
+from PIL import Image
+from scipy.optimize import least_squares
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+
+
+def read_video():
+    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
+    result = []
+    while True:
+        ok, image = cap.read()
+        if not ok:
+            break
+        result.append(cv2.cvtColor(image, cv2.COLOR_BGR2RGB))
+    cap.release()
+    return np.array(result)
+
+
+def locate_ball(video):
+    """Fit only the exposed silhouette, not the ball's floor shadow."""
+    reference = np.median(video, axis=0).astype(np.float32)
+    centers = []
+    for image in video[:47]:
+        a = image.astype(np.float32)
+        r, g, b = a.transpose(2, 0, 1)
+        difference = np.linalg.norm(a - reference, axis=2)
+        mask = ((r-g > 22) & (g-b > 7) & (r > 65) &
+                (difference > 25)).astype(np.uint8)
+        mask[500:] = 0
+        mask[:, :550] = 0
+        mask[:, 850:] = 0
+        _, labels, stats, _ = cv2.connectedComponentsWithStats(mask)
+        component = 1 + np.argmax(stats[1:, 4])
+        contours, _ = cv2.findContours((labels == component).astype(np.uint8),
+                                      cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_NONE)
+        points = max(contours, key=len)[:, 0].astype(float)
+        cx = stats[component, 0] + 56
+        cy = stats[component, 1] + 56
+        points = points[(points[:, 1] < cy+20) & (points[:, 1] > cy-57)]
+        fit = least_squares(
+            lambda v: np.linalg.norm(points-v[:2], axis=1)-v[2],
+            [cx, cy, 56],
+            bounds=([cx-15, cy-15, 53], [cx+15, cy+15, 59]),
+            loss='soft_l1', f_scale=1)
+        centers.append(fit.x / 5)
+    return np.array(centers)
+
+
+def background(frames):
+    # The camera is fixed.  The last frames supply unobscured wall pixels
+    # behind the initial ball; frame zero supplies the pristine entire floor.
+    bg = frames[0].astype(float).copy()
+    temporal = np.median(frames, axis=0)
+    yy, xx = np.mgrid[:144, :256]
+    d = np.sqrt((xx-127.8)**2 + (yy-22.1)**2)
+    replacement = np.clip((16-d)/2, 0, 1)[..., None]
+    return bg*(1-replacement) + temporal*replacement
+
+
+def make_sprites(video, centers):
+    # Remove the original wall at the antialiased sphere edge.  All interior
+    # pixels (including the seams and brand mark) are source pixels.
+    yy, xx = np.mgrid[-64:65, -64:65].astype(np.float32)
+    distance = np.sqrt(xx*xx + yy*yy)
+    sprites = []
+    for i in range(11):
+        cx, cy, radius = centers[i]*5
+        # Full silhouette is slightly larger than its saturated-color fit.
+        rad = radius + 0.55
+        patch = cv2.remap(video[i].astype(np.float32),
+                          (xx+cx).astype(np.float32),
+                          (yy+cy).astype(np.float32), cv2.INTER_LINEAR)
+        alpha = np.clip((rad + .8 - distance)/1.6, 0, 1)
+        # Replace boundary RGB by nearest fully interior color; this avoids a
+        # wall-colored halo when the ball moves in front of the wooden floor.
+        factor = np.minimum(1, (rad-1.3)/np.maximum(distance, .001))
+        interior = cv2.remap(patch, (64+xx*factor).astype(np.float32),
+                            (64+yy*factor).astype(np.float32), cv2.INTER_LINEAR)
+        edge = np.clip((distance-(rad-2))/2, 0, 1)[..., None]
+        color = patch*(1-edge) + interior*edge
+        lighting = .87 - .15*yy/rad
+        albedo = color/lighting[..., None]
+        sprites.append((albedo, alpha, rad))
+    return sprites
+
+
+def measure_spin(video, centers):
+    yy, xx = np.mgrid[-60:61, -60:61].astype(np.float32)
+    rad = 56.5
+    inside = xx*xx+yy*yy < 46**2
+    zz = np.sqrt(np.maximum(0, rad*rad-xx*xx-yy*yy))
+    matrix = np.concatenate([
+        np.stack([xx[inside]*0, zz[inside], -yy[inside]], axis=1),
+        np.stack([-zz[inside], yy[inside]*0, xx[inside]], axis=1)], axis=0)
+    estimates = []
+    for i in range(8):
+        patches = []
+        for j in [i, i+1]:
+            patches.append(cv2.cvtColor(cv2.remap(video[j],
+                (xx+centers[j, 0]*5).astype(np.float32),
+                (yy+centers[j, 1]*5).astype(np.float32),
+                cv2.INTER_LINEAR), cv2.COLOR_RGB2GRAY))
+        flow = cv2.calcOpticalFlowFarneback(patches[0], patches[1], None,
+                                           .5, 3, 15, 5, 5, 1.2, 0)
+        vec = np.r_[flow[..., 0][inside], flow[..., 1][inside]]
+        estimates.append(np.linalg.lstsq(matrix, vec, rcond=None)[0][2])
+    return float(np.median(estimates))
+
+
+def main():
+    frames = np.load(ROOT / 'source_pixels.npz')['rgb']
+    spec = json.loads((ROOT / 'task_spec.json').read_text())
+    video = read_video()
+    centers = locate_ball(video)
+    ticks = np.arange(len(centers))
+    initial_x = np.polyfit(ticks[:8], centers[:8, 0], 2)
+    initial_y = np.polyfit(ticks[:8], centers[:8, 1], 2)
+    rebound_y = np.polyfit(ticks[14:30], centers[14:30, 1], 2)
+    crossings = np.roots(initial_y-rebound_y)
+    original_hit = float(min(c for c in crossings if 10 < c < 14))
+    ground = float(np.polyval(initial_y, original_hit))
+    restitution = float(-np.polyval(np.polyder(rebound_y), original_hit)
+                        / np.polyval(np.polyder(initial_y), original_hit))
+    gravity = 2*initial_y[0]
+    vx_original, vy_original = initial_x[1], initial_y[1]
+    x0, y0 = centers[0, :2]
+    radius = float(np.median(centers[:11, 2]))
+    spin_initial = measure_spin(video, centers)
+    vx_after_original = np.polyfit(ticks[14:29], centers[14:29, 0], 1)[0]
+    slip_original = vx_original-spin_initial*radius
+    impulse_fraction = (vx_original-vx_after_original)/slip_original
+    # The measured rough-sphere impulse is a fraction of the contact slip;
+    # its resulting rolling speed is NOT sixfold timeline playback.
+    vx = 6*vx_original
+    vy = 6*vy_original
+    hit = float(np.roots([gravity/2, vy, y0-ground])[1])
+    if hit < 0:
+        hit = float(np.roots([gravity/2, vy, y0-ground])[0])
+    impact_x = x0+vx*hit
+    impact_velocity = vy+gravity*hit
+    vx_post = vx-impulse_fraction*(vx-spin_initial*radius)
+    vy_post = -restitution*impact_velocity
+    spin_post = vx_post/radius
+
+    bg = background(frames)
+    bg_big = cv2.resize(bg.astype(np.float32), (1024, 576),
+                        interpolation=cv2.INTER_CUBIC)
+    sprites = make_sprites(video, centers)
+    YY, XX = np.mgrid[:576, :1024].astype(np.float32)
+
+    def motion(t):
+        if t <= hit:
+            return x0+vx*t, y0+vy*t+gravity*t*t/2, spin_initial*t
+        dt = t-hit
+        px, py = impact_x, ground
+        tx, ty = vx_post, vy_post
+        angle = spin_initial*hit + spin_post*dt
+        # Contact events continue outside the camera too, so the video remains
+        # a complete counterfactual rather than stopping at the frame boundary.
+        while dt > -2*ty/gravity and ty < -.15:
+            duration = -2*ty/gravity
+            px += tx*duration
+            dt -= duration
+            ty *= restitution
+            tx *= .96
+        if ty >= -.15:
+            return px+tx*dt, ground, angle
+        return px+tx*dt, py+ty*dt+gravity*dt*dt/2, angle
+
+    def render(t):
+        cx, cy, angle = motion(t)
+        image = bg_big.copy()
+        if cx-radius > 258 or cx+radius < -2:
+            return image
+        height = max(0, ground-cy)
+        # Floor-only soft shadow, shrinking and strengthening toward contact.
+        sx = (cx+3+.08*height)*4
+        sy = (ground+radius+.015*height)*4
+        wide = (11+.13*height)*4
+        narrow = (2.5+.045*height)*4
+        shadow = (.18*np.exp(-height/27)+.008)*np.exp(
+            -.5*((XX-sx)/wide)**2-.5*((YY-sy)/narrow)**2)
+        shadow += .10*np.exp(-height/3)*np.exp(
+            -.5*((XX-cx*4)/(7*4))**2
+            -.5*((YY-(ground+radius)*4)/(1.6*4))**2)
+        image *= 1-shadow[..., None]
+        if t < hit:
+            index = min(10, max(0, int(np.floor(t))))
+            following = min(10, index+1)
+            blend = t-index
+            a, alpha, rad = sprites[index]
+            b, alpha2, rad2 = sprites[following]
+            albedo = a*(1-blend)+b*blend
+            alpha = alpha*(1-blend)+alpha2*blend
+            rad = rad*(1-blend)+rad2*blend
+            rotation = 0
+        else:
+            index = min(10, int(round(hit)))
+            albedo, alpha, rad = sprites[index]
+            # The source image already contains its original spin angle.
+            rotation = angle-spin_initial*index
+        # Image-coordinate positive rotation is clockwise. Preserve lighting
+        # in world coordinates while rotating the original texture, not a logo
+        # drawn anew or a rigidly rotating baked light gradient.
+        scale = 4/5
+        xmin = max(0, int(np.floor(cx*4-54)))
+        xmax = min(1024, int(np.ceil(cx*4+54)))
+        ymin = max(0, int(np.floor(cy*4-54)))
+        ymax = min(576, int(np.ceil(cy*4+54)))
+        if xmin >= xmax or ymin >= ymax:
+            return image
+        yy, xx = np.mgrid[ymin:ymax, xmin:xmax].astype(np.float32)
+        dx, dy = (xx-cx*4)/scale, (yy-cy*4)/scale
+        cosine, sine = np.cos(rotation), np.sin(rotation)
+        mx = (64+cosine*dx+sine*dy).astype(np.float32)
+        my = (64-sine*dx+cosine*dy).astype(np.float32)
+        color = cv2.remap(albedo, mx, my, cv2.INTER_LINEAR,
+                          borderMode=cv2.BORDER_CONSTANT)
+        matte = cv2.remap(alpha, mx, my, cv2.INTER_LINEAR,
+                          borderMode=cv2.BORDER_CONSTANT)
+        light = np.clip(.87-.15*dy/rad, .65, 1.05)
+        color *= light[..., None]
+        region = image[ymin:ymax, xmin:xmax]
+        region[:] = region*(1-matte[..., None]) + color*matte[..., None]
+        return image
+
+    output = []
+    for i in range(spec['frames']):
+        if i == 0:
+            output.append(frames[0].copy())
+            continue
+        # Small shutter integration smooths fast subpixel flight without
+        # inventing a long motion trail absent from the factual video.
+        samples = [render(i+d) for d in [-.075, 0, .075]]
+        image = sum(samples)/len(samples)
+        image = cv2.resize(image, (256, 144), interpolation=cv2.INTER_AREA)
+        output.append(np.clip(np.rint(image), 0, 255).astype(np.uint8))
+    output = np.stack(output)
+    artifact_io.save(output, ROOT)
+    print(json.dumps(dict(initial_velocity=[float(vx), float(vy)],
+                          gravity=float(gravity), original_contact=original_hit,
+                          new_contact=hit, restitution=restitution,
+                          contact_x=impact_x, post_contact_vx=vx_post,
+                          initial_spin=spin_initial, post_spin=spin_post), indent=2))
+
+
+if __name__ == '__main__':
+    main()
