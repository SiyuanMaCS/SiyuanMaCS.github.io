+"""Render the friction counterfactual using only the supplied source pixels.
+
+The target is stationary before contact.  After contact the two similar-size
+balls advance together.  Fivefold floor resistance on the target changes the
+resistance of this coupled pair from 2*f to 6*f (threefold total resistance).
+The new constant-drag trajectory has the same initial momentum, one third of
+its former stopping distance, and one third of its former stopping time.
+We recover the source image at each new physical progress coordinate; this
+retains the source lighting, contact, perspective and rolling texture.
+"""

+import cv2

+    images = []

+        images.append(cv2.resize(cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB),
+                                 (512, 288), interpolation=cv2.INTER_AREA))

+    large = decode_source_large()

+    # remain in contact during the short forward roll.
