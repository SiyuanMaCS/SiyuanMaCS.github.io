+and unequal-mass rigid disk impulses. No appearance change accompanies mass.

+def load_native():

+    frames=[]

+def locate(frames):

+# Table top from measured visible interior corners. Dimensionless world coordinates.
+P=np.array([[65,435],[607,562],[1183,276],[751,229]],np.float32)

+H=cv2.getPerspectiveTransform(P,Q)
+J=np.linalg.inv(H)

+def project(q):
+    v=J@np.r_[q,1.]
+    return v[:2]/v[2]

+def obj_mask(shape,xy,k,reflection=True):
+    m=np.zeros(shape,np.uint8)
+    x,y=np.round(xy).astype(int)
+    rx=[34,30,37][k];ry=[14,13,15][k];height=[29,27,30][k]

+    if reflection:

+def clean_background(a,tracks):
+    # Compute a temporal background only inside the affected table region.
+    # Ignore actual objects and their contact shadows/reflections in each sample.
+    y0,y1,x0,x1=240,560,270,910
+    sample_ids=list(range(0,96,3))
+    stack=[]

+def extract_sprite(a,plate,tracks,i,k):

+    x0=int(anchor[0])-60;y0=int(anchor[1])-50
+    crop=a[i,y0:y0+90,x0:x0+120].astype(np.float32)

+    contours,_=cv2.findContours(m,cv2.RETR_EXTERNAL,cv2.CHAIN_APPROX_SIMPLE)
+    cv2.drawContours(m,contours,-1,1,-1)
+    # Capture the dark cylindrical base rim using the clean temporal plate,
+    # but never include nearby printed markings or other objects.
+    near=cv2.dilate(m,np.ones((7,7),np.uint8))>0
+    diff=np.max(np.abs(crop-bg),axis=2)
+    alpha=np.maximum(m.astype(np.float32), near*np.clip((diff-4)/15,0,1))
+    alpha=cv2.GaussianBlur(alpha,(3,3),.45)
+    # Sprite anchor and source perspective for scale adaptation.
+    return crop,alpha,anchor-np.array([x0,y0]),cv2.perspectiveTransform(anchor[None,None].astype(np.float32),H)[0,0]

+def render_sprite(frame,sprite,q):
+    crop,alpha,anchor,source_q=sprite
+    pos=project(q)
+    # Perspective makes the mallet shrink toward the far edge, not with mass.
+    ds=(J@np.r_[source_q,1])[2];dt=(J@np.r_[q,1])[2]
+    scale=ds/dt
+    transform=np.array([[scale,0,pos[0]-scale*anchor[0]],
+                        [0,scale,pos[1]-scale*anchor[1]]],np.float32)
+    premult=crop*alpha[:,:,None]
+    rgb=cv2.warpAffine(premult,transform,(1280,720),flags=cv2.INTER_LINEAR)
+    al=cv2.warpAffine(alpha,transform,(1280,720),flags=cv2.INTER_LINEAR)
+    # Soft contact shadow and subtle reflection remain attached to the mallet.
+    yy,xx=np.mgrid[:720,:1280]
+    rx=[30,27,34][render_sprite.kind]*scale;ry=8*scale
+    shadow=np.exp(-.5*(((xx-pos[0]-5*scale)/rx)**2+((yy-pos[1]-3*scale)/ry)**2))*.12
+    frame*=1-shadow[:,:,None]
+    frame=frame*(1-al[:,:,None])+rgb
+    return frame
+

+def physics(tracks):
+    observed=cv2.perspectiveTransform(tracks.astype(np.float32),H)
+    # Surface center coordinates rather than height-dependent silhouette centers.
+    radii=np.array([.053,.051,.055])
+    mass=np.array([1.,1.,10.])
+    e=.93
+    q=observed[29].astype(float).copy()
+    # White was stationary up to the impact; use the unchanged initial location.
+    q[2]=observed[0,2]
+    # Source collision axis is nearly along the table length.
+    # Subpixel color tracking can wobble; fit the pre-impact translation.
+    vi=np.polyfit(np.arange(17,28),observed[17:28,0,1],1)[0]
+    velocities=np.zeros((3,2));velocities[0]=[0,vi]
+    # Resolve the intervention's first contact between frame 29 and 30.
+    q[0,0]=q[2,0]
+    q[0,1]=q[2,1]+radii[0]+radii[2]
+    history=np.zeros((96,3,2));history[:30]=observed[:30]
+    # Same normal impulse as a rigid elastic collision, with 10x target mass.
+    v=velocities[0,1]
+    velocities[0,1]=(1-e*10)/11*v
+    velocities[2,1]=(1+e)/11*v
+    # Tiny real surface drag: proportional to normal load -> mass-independent
+    # acceleration. White's much smaller acquired velocity settles sooner.
+    drag=.000105
+    for i in range(30,96):
+        for step in range(12):
+            dt=1/12
+            q+=velocities*dt
+            for j in range(3):
+                for k in range(j+1,3):
+                    delta=q[k]-q[j];d=np.linalg.norm(delta);r=radii[j]+radii[k]
+                    if d<r:
+                        normal=delta/max(d,1e-8)
+                        inverse=1/mass[j]+1/mass[k]
+                        relative=np.dot(velocities[k]-velocities[j],normal)
+                        if relative<0:
+                            impulse=-(1+e)*relative/inverse
+                            velocities[j]-=impulse/mass[j]*normal
+                            velocities[k]+=impulse/mass[k]*normal
+                        overlap=r-d
+                        q[j]-=normal*overlap*(1/mass[j])/inverse
+                        q[k]+=normal*overlap*(1/mass[k])/inverse
+            for j in range(3):
+                for axis,limit in [(0,1),(1,2)]:
+                    if q[j,axis]<radii[j]:
+                        q[j,axis]=radii[j];velocities[j,axis]=abs(velocities[j,axis])*.82
+                    elif q[j,axis]>limit-radii[j]:
+                        q[j,axis]=limit-radii[j];velocities[j,axis]=-abs(velocities[j,axis])*.82
+                speed=np.linalg.norm(velocities[j])
+                if speed:velocities[j]*=max(0,1-drag*dt/speed)
+        history[i]=q
+    return history
+
+
+def main():

+    tracks=locate(a)
+    plate=clean_background(a,tracks)
+    # Near-impact texture for red; the white/blue keep the original appearance.
+    sprites=[extract_sprite(a,plate,tracks,27,0),
+             extract_sprite(a,plate,tracks,0,1),
+             extract_sprite(a,plate,tracks,0,2)]
+    positions=physics(tracks)
+    result=source.copy()
+    for i in range(30,96):
+        frame=a[i].astype(np.float32)
+        remove=np.zeros((720,1280),np.uint8)
+        for k in range(3):remove=np.maximum(remove,obj_mask((720,1280),tracks[i,k],k))
+        feather=cv2.GaussianBlur(remove.astype(np.float32)/255,(13,13),2)
+        frame=frame*(1-feather[:,:,None])+plate.astype(float)*feather[:,:,None]
+        # Reflections reuse the actual source surface's illumination; render a
+        # restrained tint in each object's own location before the solid sprite.
+        yy,xx=np.mgrid[:720,:1280]
+        for k in range(3):
+            p=project(positions[i,k]);scale=(J@np.r_[sprites[k][3],1])[2]/(J@np.r_[positions[i,k],1])[2]
+            ref=np.exp(-.5*(((xx-p[0])/ (12*scale))**2+((yy-p[1]-20*scale)/(14*scale))**2))
+            tint=[[-3,-5,-4],[-6,-3,4],[4,4,4]][k]
+            frame+=ref[:,:,None]*np.array(tint)
+        # Farther mallets first, preserving depth at contacts.
+        for k in np.argsort(-positions[i,:,1]):
+            render_sprite.kind=k
+            frame=render_sprite(frame,sprites[k],positions[i,k])
+        img=Image.fromarray(np.clip(frame,0,255).astype(np.uint8))
+        result[i]=np.asarray(img.resize((256,144),Image.Resampling.LANCZOS))
+    artifact_io.save(result,ROOT)
+    np.save(ROOT/'.tmp/counterfactual_positions.npy',positions)
+    # Additional conventional video export of exactly the saved RGB frames.
+    writer=cv2.VideoWriter(str(ROOT/'edited_video.mp4'),cv2.VideoWriter_fourcc(*'mp4v'),24,(256,144))
+    for f in result: writer.write(cv2.cvtColor(f,cv2.COLOR_RGB2BGR))
+    writer.release()
+    print('Saved 96 RGB frames, 256x144, 24 fps.')
+
+if __name__=='__main__':main()

and unequal-mass rigid disk impulses. No appearance change accompanies mass.

from pathlib import Path
import numpy as np
import cv2
from PIL import Image
import artifact_io

ROOT = Path(__file__).resolve().parent

def load_native():
    cap = cv2.VideoCapture(str(ROOT / 'source_video.mp4'))
    frames=[]
    while True:
        ok,frame=cap.read()
        if not ok: break
        frames.append(cv2.cvtColor(frame,cv2.COLOR_BGR2RGB))
    cap.release()
    return np.asarray(frames)

def locate(frames):
    tracks=[]
    for f in frames:
        hsv=cv2.cvtColor(f,cv2.COLOR_RGB2HSV)
        masks=[(((hsv[:,:,0]<15)|(hsv[:,:,0]>172))&(hsv[:,:,1]>75)&(hsv[:,:,2]>150)),
               ((hsv[:,:,0]>94)&(hsv[:,:,0]<118)&(hsv[:,:,1]>85)),
               ((hsv[:,:,1]<13)&(hsv[:,:,2]>225))]
        bounds=[(500,770,305,425),(680,890,250,360),(270,575,375,510)]
        pts=[]
        for k,(m,(x1,x2,y1,y2)) in enumerate(zip(masks,bounds)):
            w=np.zeros(m.shape,np.uint8);w[y1:y2,x1:x2]=m[y1:y2,x1:x2]
            cs,_=cv2.findContours(w,cv2.RETR_EXTERNAL,cv2.CHAIN_APPROX_SIMPLE)
            co=max([c for c in cs if cv2.contourArea(c)>50],key=cv2.contourArea)
            x,y,ww,hh=cv2.boundingRect(co)
            pts.append([x+ww/2, y+hh-[10,11,13][k]])
        tracks.append(pts)
    return np.array(tracks)

# Table top from measured visible interior corners. Dimensionless world coordinates.
P=np.array([[65,435],[607,562],[1183,276],[751,229]],np.float32)
Q=np.array([[0,0],[1,0],[1,2],[0,2]],np.float32)
H=cv2.getPerspectiveTransform(P,Q)
J=np.linalg.inv(H)

def project(q):
    v=J@np.r_[q,1.]
    return v[:2]/v[2]

def obj_mask(shape,xy,k,reflection=True):
    m=np.zeros(shape,np.uint8)
    x,y=np.round(xy).astype(int)
    rx=[34,30,37][k];ry=[14,13,15][k];height=[29,27,30][k]
    cv2.ellipse(m,(x+2,y+1),(rx+5,ry+4),0,0,360,255,-1)
    cv2.rectangle(m,(x-14,y-height),(x+14,y+2),255,-1)
    cv2.ellipse(m,(x,y-height),(14,7),0,0,360,255,-1)
    if reflection:
        cv2.ellipse(m,(x+4,y+23),(rx+7,28),0,0,360,255,-1)
    return m

def clean_background(a,tracks):
    # Compute a temporal background only inside the affected table region.
    # Ignore actual objects and their contact shadows/reflections in each sample.
    y0,y1,x0,x1=240,560,270,910
    sample_ids=list(range(0,96,3))
    stack=[]
    for i in sample_ids:
        patch=a[i,y0:y1,x0:x1].astype(np.float32)
        m=np.zeros((720,1280),np.uint8)
        for k in range(3):
            m=np.maximum(m,obj_mask((720,1280),tracks[i,k],k))
        patch[m[y0:y1,x0:x1]>0]=np.nan
        stack.append(patch)
    import warnings
    with warnings.catch_warnings():
        warnings.simplefilter('ignore',RuntimeWarning)
        plate=np.nanmedian(np.stack(stack),axis=0)
    plate=np.where(np.isfinite(plate),plate,a[0,y0:y1,x0:x1]).astype(np.uint8)
    full=a[0].copy();full[y0:y1,x0:x1]=plate
    return full

def extract_sprite(a,plate,tracks,i,k):
    anchor=tracks[i,k]
    x0=int(anchor[0])-60;y0=int(anchor[1])-50
    crop=a[i,y0:y0+90,x0:x0+120].astype(np.float32)
    bg=plate[y0:y0+90,x0:x0+120].astype(np.float32)
    hsv=cv2.cvtColor(crop.astype(np.uint8),cv2.COLOR_RGB2HSV)
    if k==0:
        m=(((hsv[:,:,0]<17)|(hsv[:,:,0]>171))&(hsv[:,:,1]>55)).astype(np.uint8)
    elif k==1:
        m=((hsv[:,:,0]>91)&(hsv[:,:,0]<121)&(hsv[:,:,1]>65)).astype(np.uint8)
    else:
        m=((hsv[:,:,1]<19)&(hsv[:,:,2]>215)).astype(np.uint8)
    n,l,stats,_=cv2.connectedComponentsWithStats(m)
    lab=1+np.argmax(stats[1:,cv2.CC_STAT_AREA]);m=(l==lab).astype(np.uint8)
    m=cv2.morphologyEx(m,cv2.MORPH_CLOSE,np.ones((3,3),np.uint8))
    contours,_=cv2.findContours(m,cv2.RETR_EXTERNAL,cv2.CHAIN_APPROX_SIMPLE)
    cv2.drawContours(m,contours,-1,1,-1)
    # Capture the dark cylindrical base rim using the clean temporal plate,
    # but never include nearby printed markings or other objects.
    near=cv2.dilate(m,np.ones((7,7),np.uint8))>0
    diff=np.max(np.abs(crop-bg),axis=2)
    alpha=np.maximum(m.astype(np.float32), near*np.clip((diff-4)/15,0,1))
    alpha=cv2.GaussianBlur(alpha,(3,3),.45)
    # Sprite anchor and source perspective for scale adaptation.
    return crop,alpha,anchor-np.array([x0,y0]),cv2.perspectiveTransform(anchor[None,None].astype(np.float32),H)[0,0]

def render_sprite(frame,sprite,q):
    crop,alpha,anchor,source_q=sprite
    pos=project(q)
    # Perspective makes the mallet shrink toward the far edge, not with mass.
    ds=(J@np.r_[source_q,1])[2];dt=(J@np.r_[q,1])[2]
    scale=ds/dt
    transform=np.array([[scale,0,pos[0]-scale*anchor[0]],
                        [0,scale,pos[1]-scale*anchor[1]]],np.float32)
    premult=crop*alpha[:,:,None]
    rgb=cv2.warpAffine(premult,transform,(1280,720),flags=cv2.INTER_LINEAR)
    al=cv2.warpAffine(alpha,transform,(1280,720),flags=cv2.INTER_LINEAR)
    # Soft contact shadow and subtle reflection remain attached to the mallet.
    yy,xx=np.mgrid[:720,:1280]
    rx=[30,27,34][render_sprite.kind]*scale;ry=8*scale
    shadow=np.exp(-.5*(((xx-pos[0]-5*scale)/rx)**2+((yy-pos[1]-3*scale)/ry)**2))*.12
    frame*=1-shadow[:,:,None]
    frame=frame*(1-al[:,:,None])+rgb
    return frame


def physics(tracks):
    observed=cv2.perspectiveTransform(tracks.astype(np.float32),H)
    # Surface center coordinates rather than height-dependent silhouette centers.
    radii=np.array([.053,.051,.055])
    mass=np.array([1.,1.,10.])
    e=.93
    q=observed[29].astype(float).copy()
    # White was stationary up to the impact; use the unchanged initial location.
    q[2]=observed[0,2]
    # Source collision axis is nearly along the table length.
    # Subpixel color tracking can wobble; fit the pre-impact translation.
    vi=np.polyfit(np.arange(17,28),observed[17:28,0,1],1)[0]
    velocities=np.zeros((3,2));velocities[0]=[0,vi]
    # Resolve the intervention's first contact between frame 29 and 30.
    q[0,0]=q[2,0]
    q[0,1]=q[2,1]+radii[0]+radii[2]
    history=np.zeros((96,3,2));history[:30]=observed[:30]
    # Same normal impulse as a rigid elastic collision, with 10x target mass.
    v=velocities[0,1]
    velocities[0,1]=(1-e*10)/11*v
    velocities[2,1]=(1+e)/11*v
    # Tiny real surface drag: proportional to normal load -> mass-independent
    # acceleration. White's much smaller acquired velocity settles sooner.
    drag=.000105
    for i in range(30,96):
        for step in range(12):
            dt=1/12
            q+=velocities*dt
            for j in range(3):
                for k in range(j+1,3):
                    delta=q[k]-q[j];d=np.linalg.norm(delta);r=radii[j]+radii[k]
                    if d<r:
                        normal=delta/max(d,1e-8)
                        inverse=1/mass[j]+1/mass[k]
                        relative=np.dot(velocities[k]-velocities[j],normal)
                        if relative<0:
                            impulse=-(1+e)*relative/inverse
                            velocities[j]-=impulse/mass[j]*normal
                            velocities[k]+=impulse/mass[k]*normal
                        overlap=r-d
                        q[j]-=normal*overlap*(1/mass[j])/inverse
                        q[k]+=normal*overlap*(1/mass[k])/inverse
            for j in range(3):
                for axis,limit in [(0,1),(1,2)]:
                    if q[j,axis]<radii[j]:
                        q[j,axis]=radii[j];velocities[j,axis]=abs(velocities[j,axis])*.82
                    elif q[j,axis]>limit-radii[j]:
                        q[j,axis]=limit-radii[j];velocities[j,axis]=-abs(velocities[j,axis])*.82
                speed=np.linalg.norm(velocities[j])
                if speed:velocities[j]*=max(0,1-drag*dt/speed)
        history[i]=q
    return history


def main():
    a=load_native();source=np.load(ROOT/'source_pixels.npz')['rgb']
    tracks=locate(a)
    plate=clean_background(a,tracks)
    # Near-impact texture for red; the white/blue keep the original appearance.
    sprites=[extract_sprite(a,plate,tracks,27,0),
             extract_sprite(a,plate,tracks,0,1),
             extract_sprite(a,plate,tracks,0,2)]
    positions=physics(tracks)
    result=source.copy()
    for i in range(30,96):
        frame=a[i].astype(np.float32)
        remove=np.zeros((720,1280),np.uint8)
        for k in range(3):remove=np.maximum(remove,obj_mask((720,1280),tracks[i,k],k))

        # Reflections reuse the actual source surface's illumination; render a

        yy,xx=np.mgrid[:720,:1280]

            ref=np.exp(-.5*(((xx-p[0])/ (12*scale))**2+((yy-p[1]-20*scale)/(14*scale))**2))
            tint=[[-3,-5,-4],[-6,-3,4],[4,4,4]][k]
            frame+=ref[:,:,None]*np.array(tint)
        # Farther mallets first, preserving depth at contacts.
        for k in np.argsort(-positions[i,:,1]):
            render_sprite.kind=k
            frame=render_sprite(frame,sprites[k],positions[i,k])
        img=Image.fromarray(np.clip(frame,0,255).astype(np.uint8))
        result[i]=np.asarray(img.resize((256,144),Image.Resampling.LANCZOS))
    artifact_io.save(result,ROOT)
    np.save(ROOT/'.tmp/counterfactual_positions.npy',positions)
    # Additional conventional video export of exactly the saved RGB frames.
    writer=cv2.VideoWriter(str(ROOT/'edited_video.mp4'),cv2.VideoWriter_fourcc(*'mp4v'),24,(256,144))
    for f in result: writer.write(cv2.cvtColor(f,cv2.COLOR_RGB2BGR))
    writer.release()
    print('Saved 96 RGB frames, 256x144, 24 fps.')

if __name__=='__main__':main()
