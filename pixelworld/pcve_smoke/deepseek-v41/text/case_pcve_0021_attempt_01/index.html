+from pathlib import Path
+
+import cv2
+import numpy as np
+from PIL import Image, ImageDraw
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+W, H, FPS, N = 256, 144, 24, 96
+
+
+def smoothstep(value):
+    value = np.clip(value, 0.0, 1.0)
+    return value * value * (3.0 - 2.0 * value)
+
+
+def bilinear_sample(image, x, y):
+    x0 = np.floor(x).astype(np.int32)
+    y0 = np.floor(y).astype(np.int32)
+    fx = x - x0
+    fy = y - y0
+    x0 = np.clip(x0, 0, image.shape[1] - 1)
+    x1 = np.clip(x0 + 1, 0, image.shape[1] - 1)
+    y0 = np.clip(y0, 0, image.shape[0] - 1)
+    y1 = np.clip(y0 + 1, 0, image.shape[0] - 1)
+    if image.ndim == 3:
+        fx = fx[..., None]
+        fy = fy[..., None]
+    top = image[y0, x0] * (1.0 - fx) + image[y0, x1] * fx
+    bottom = image[y1, x0] * (1.0 - fx) + image[y1, x1] * fx
+    return top * (1.0 - fy) + bottom * fy
+
+
+def render_sprite(canvas, sprite, mask, cx, cy, angle=0.0, scale=1.0):
+    """Composite a premultiplied-looking sprite at subpixel position."""
+    h, w = mask.shape
+    rad = np.deg2rad(angle)
+    c, s = np.cos(rad), np.sin(rad)
+    half_w, half_h = (w - 1) / 2.0, (h - 1) / 2.0
+    yy, xx = np.mgrid[0:H, 0:W]
+    dx = (xx + 0.5 - cx) / scale
+    dy = (yy + 0.5 - cy) / scale
+    sx = c * dx + s * dy + half_w
+    sy = -s * dx + c * dy + half_h
+    valid = (sx >= 0) & (sx <= w - 1) & (sy >= 0) & (sy <= h - 1)
+    alpha = np.zeros((H, W), dtype=np.float32)
+    color = np.zeros((H, W, 3), dtype=np.float32)
+    if np.any(valid):
+        alpha[valid] = bilinear_sample(mask[..., None], sx, sy)[..., 0][valid]
+        color[valid] = bilinear_sample(sprite, sx, sy)[valid]
+    alpha = np.clip(alpha, 0.0, 1.0)
+    canvas[:] = canvas * (1.0 - alpha[..., None]) + color * alpha[..., None]
+    return alpha
+
+
+def make_ball_sprite(frame):
+    """Extract the isolated incoming ball from its source pixels."""
+    x0, y0, x1, y1 = 120, 50, 156, 92
+    patch = frame[y0:y1, x0:x1].copy()
+    bg = np.load(CASE_DIR / "source_pixels.npz")["rgb"][45, y0:y1, x0:x1]
+    dist = np.max(np.abs(patch.astype(np.int16) - bg.astype(np.int16)), axis=2)
+    # The ball is the compact warm region in this crop.
+    warm = (
+        (patch[..., 0].astype(np.int16) - patch[..., 2].astype(np.int16) > 18)
+        & (patch[..., 0].astype(np.int16) - patch[..., 1].astype(np.int16) > 4)
+    )
+    mask = ((dist > 12) & warm).astype(np.uint8)
+    mask = cv2.morphologyEx(mask, cv2.MORPH_CLOSE, np.ones((3, 3), np.uint8))
+    n, labels, stats, _ = cv2.connectedComponentsWithStats(mask, 8)
+    if n > 1:
+        index = 1 + np.argmax(stats[1:, cv2.CC_STAT_AREA])
+        mask = labels == index
+    mask = cv2.GaussianBlur(mask.astype(np.float32), (0, 0), 0.65)
+    ys, xs = np.where(mask > 0.2)
+    pad = 3
+    bx0, bx1 = max(0, xs.min() - pad), min(patch.shape[1], xs.max() + pad + 1)
+    by0, by1 = max(0, ys.min() - pad), min(patch.shape[0], ys.max() + pad + 1)
+    return patch[by0:by1, bx0:bx1], mask[by0:by1, bx0:bx1]
+
+
+def make_star_sprite(frame, bg):
+    """Extract the large foreground ball while removing background spill."""
+    x0, y0, x1, y1 = 76, 58, 132, 144
+    patch = frame[y0:y1, x0:x1].astype(np.float32)
+    base = bg[y0:y1, x0:x1].astype(np.float32)
+    dist = np.max(np.abs(patch - base), axis=2)
+    # Foreground includes the saturated warm body and its bright highlights.
+    r, g, b = patch[..., 0], patch[..., 1], patch[..., 2]
+    warm = (r - b > 18) & (r - g > 5)
+    neutral = (dist > 24) & (np.max(patch, 2) - np.min(patch, 2) < 32)
+    mask = (dist > 8) & (warm | neutral)
+    mask = cv2.morphologyEx(mask.astype(np.uint8), cv2.MORPH_CLOSE, np.ones((3, 3), np.uint8))
+    # Keep the main connected foreground mass and fill small internal holes.
+    n, labels, stats, _ = cv2.connectedComponentsWithStats(mask, 8)
+    if n > 1:
+        keep = np.zeros_like(mask)
+        for index in range(1, n):
+            if stats[index, cv2.CC_STAT_AREA] > 30:
+                keep[labels == index] = 1
+        mask = keep
+    mask = cv2.morphologyEx(mask, cv2.MORPH_CLOSE, np.ones((5, 5), np.uint8))
+    mask = cv2.GaussianBlur(mask.astype(np.float32), (0, 0), 0.75)
+    return patch, np.clip(mask, 0.0, 1.0)
+
+
+def source_ball_path():
+    # Measured centers from the source's incoming ball; retain its exact pre-contact trajectory.
+    return np.array(
+        [
+            (238.0, 70.3),
+            (222.6, 70.3),
+            (207.8, 70.6),
+            (192.9, 69.8),
+            (179.1, 69.5),
+            (164.9, 69.4),
+            (150.9, 69.7),
+            (136.9, 70.0),
+        ],
+        dtype=np.float64,
+    )
+
+
+def make_candidate(source):
+    bg = source[45].copy()
+    incoming, incoming_mask = make_ball_sprite(source[7])
+    star, star_mask = make_star_sprite(source[0], bg)
+    star_h, star_w = star_mask.shape
+    star_cx, star_cy = 103.0, 108.0
+
+    # Contact is at frame 7. A ten-times-heavier incoming ball keeps nearly all
+    # its momentum; the much lighter foreground ball receives a reduced impulse.
+    path = source_ball_path()
+    contact = path[-1]
+    vx = path[-1, 0] - path[-2, 0]
+    vy = path[-1, 1] - path[-2, 1]
+    frames = np.empty((N, H, W, 3), dtype=np.float32)
+    for i in range(N):
+        canvas = bg.astype(np.float32).copy()
+        t = max(0.0, i - 7.0)
+
+        # Incoming ball: exact source path up to impact, then a small elastic
+        # recoil. The heavier body remains almost unchanged by the collision.
+        if i < 8:
+            p = path[min(i, 7)]
+        else:
+            bounce = 1.0 - np.exp(-t / 3.0)
+            p = np.array([contact[0] - 0.10 * t, contact[1] + 0.035 * bounce])
+
+        # The target ball is displaced by the original impact and settles.
+        # Under the intervention its translational response is damped.
+        if i < 8:
+            sx, sy = star_cx, star_cy
+            tilt = 0.0
+        else:
+            u = t / 32.0
+            settle = smoothstep(u)
+            if i < 20:
+                sx = star_cx - 2.1 * settle
+                sy = star_cy + 1.1 * settle
+                tilt = -2.0 * settle
+            else:
+                v = np.clip((i - 20) / 22.0, 0.0, 1.0)
+                sx = star_cx - 2.1 + 2.1 * smoothstep(v)
+                sy = star_cy + 1.1 * (1.0 - smoothstep(v))
+                tilt = -2.0 + 2.0 * smoothstep(v)
+
+        render_sprite(canvas, star, star_mask, sx, sy, angle=tilt)
+        render_sprite(canvas, incoming, incoming_mask, p[0], p[1])
+
+        # Keep the intervention visually legible without changing object identity:
+        # a faint contact ring is visible for two frames at the first impact.
+        if 7 <= i <= 9:
+            alpha = 0.16 if i == 7 else 0.08
+            overlay = canvas.copy()
+            cv2.circle(overlay, (int(round(contact[0])), int(round(contact[1]))), 13 + (i - 7) * 2, (238, 193, 159), 1, cv2.LINE_AA)
+            canvas = canvas * (1.0 - alpha) + overlay * alpha
+
+        frames[i] = canvas
+    return np.clip(frames, 0, 255).astype(np.uint8)
+
+
+def main():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    frames = make_candidate(source)
+    artifact_io.save(frames, CASE_DIR)
+
+    # A compact diagnostic sheet makes the intervention and settle phase easy to verify.
+    checks = [0, 7, 8, 10, 14, 20, 28, 40, 56, 80, 95]
+    sheet = Image.new("RGB", (W * 4, (H + 18) * 3), (25, 28, 35))
+    draw = ImageDraw.Draw(sheet)
+    for j, index in enumerate(checks):
+        x, y = (j % 4) * W, (j // 4) * (H + 18)
+        draw.text((x + 4, y + 2), f"{index:02d}  {index / FPS:.2f}s", fill="white")
+        sheet.paste(Image.fromarray(frames[index]), (x, y + 18))
+    sheet.save(CASE_DIR / "edit_check.png")
+    print("saved", frames.shape, frames.dtype)
+
+
+if __name__ == "__main__":
+    main()
