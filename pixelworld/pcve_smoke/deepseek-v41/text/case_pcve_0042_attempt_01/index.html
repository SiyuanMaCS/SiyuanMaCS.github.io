+def largest_component(mask):
+    """Return the largest 8-connected component of a small boolean mask."""
+    height, width = mask.shape
+    seen = np.zeros_like(mask, dtype=bool)
+    best = []
+
+    for y, x in zip(*np.nonzero(mask & ~seen)):
+        if seen[y, x]:
+            continue
+        queue = deque([(y, x)])
+        seen[y, x] = True
+        points = []
+        while queue:
+            cy, cx = queue.popleft()
+            points.append((cy, cx))
+            for dy in (-1, 0, 1):
+                for dx in (-1, 0, 1):
+                    if dx == 0 and dy == 0:
+                        continue
+                    ny, nx = cy + dy, cx + dx
+                    if (
+                        0 <= ny < height
+                        and 0 <= nx < width
+                        and mask[ny, nx]
+                        and not seen[ny, nx]
+                    ):
+                        seen[ny, nx] = True
+                        queue.append((ny, nx))
+        if len(points) > len(best):
+            best = points
+
+    component = np.zeros_like(mask, dtype=bool)
+    if best:
+        ys, xs = np.array(best).T
+        component[ys, xs] = True
+    return component
+

+def extract_sprite(frame, background):
+    """Extract the moving car and its soft contact shadow from one source frame."""
+    difference = np.abs(frame.astype(np.int16) - background.astype(np.int16)).max(axis=2)
+    component = largest_component(difference > 15)
+    ys, xs = np.nonzero(component)
+    if len(xs) == 0:
+        raise RuntimeError("Could not locate the toy car in a source frame")
+
+    x0 = max(0, int(xs.min()) - 2)
+    y0 = max(0, int(ys.min()) - 2)
+    x1 = min(frame.shape[1], int(xs.max()) + 3)
+    y1 = min(frame.shape[0], int(ys.max()) + 3)
+    local_difference = difference[y0:y1, x0:x1]
+    alpha = np.clip((local_difference - 2.0) / 12.0, 0.0, 1.0)
+
+    return {
+        "rgb": frame[y0:y1, x0:x1].copy(),
+        "alpha": alpha.astype(np.float32),
+        "x": int(xs.min()),
+        "y": int(ys.min()),
+        "crop_x": x0,
+        "crop_y": y0,
+    }
+

+def paste_sprite(canvas, sprite, x, y):
+    """Alpha-composite a sprite at a subpixel destination using a bilinear shift."""
+    target_x = int(np.floor(x))
+    target_y = int(np.floor(y))
+    fx = float(x - target_x)
+    fy = float(y - target_y)
+
+    source = sprite["rgb"].astype(np.float32)
+    alpha = sprite["alpha"]
+    height, width = alpha.shape
+
+    for oy, wy in ((0, 1.0 - fy), (1, fy)):
+        if wy == 0:
+            continue
+        for ox, wx in ((0, 1.0 - fx), (1, fx)):
+            weight = wx * wy
+            if weight == 0:
+                continue
+            px = target_x + ox
+            py = target_y + oy
+            sx0 = max(0, -px)
+            sy0 = max(0, -py)
+            sx1 = min(width, canvas.shape[1] - px)
+            sy1 = min(height, canvas.shape[0] - py)
+            if sx1 <= sx0 or sy1 <= sy0:
+                continue
+
+            dx0 = max(0, px)
+            dy0 = max(0, py)
+            dx1 = dx0 + sx1 - sx0
+            dy1 = dy0 + sy1 - sy0
+            patch = source[sy0:sy1, sx0:sx1]
+            patch_alpha = alpha[sy0:sy1, sx0:sx1, None] * weight
+            base = canvas[dy0:dy1, dx0:dx1].astype(np.float32)
+            canvas[dy0:dy1, dx0:dx1] = np.clip(
+                patch * patch_alpha + base * (1.0 - patch_alpha), 0, 255
+            ).astype(np.uint8)
+    return canvas
+

+    print(
+        "Saved slowed toy-car video:",
+        output.shape,
+        "speed_scale=",
