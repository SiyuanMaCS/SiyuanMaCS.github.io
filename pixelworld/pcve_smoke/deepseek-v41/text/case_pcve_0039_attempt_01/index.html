+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+FPS = 24.0
+FRAME_COUNT = 96
+

+def clean_plate(source):
+    """Recover the static scene while retaining the original image texture."""
+    plate = np.median(source, axis=0).astype(np.uint8)
+
+    # The car occupies the first position only at the start and the final
+    # position only after it comes to rest. Those two patches therefore have
+    # cleaner samples at opposite ends of the source clip.
+    plate[20:66, 48:93] = np.median(source[24:], axis=0)[20:66, 48:93]
+    plate[34:59, 176:211] = np.median(source[:14], axis=0)[34:59, 176:211]
+    return plate
+
+
+def extract_sprite(source, plate):
+    """Use the fully settled source car as a lit, source-derived sprite."""
+    x0, y0, x1, y1 = 181, 38, 207, 54
+    rgb = source[95, y0:y1, x0:x1].astype(np.float32)
+    bg = plate[y0:y1, x0:x1].astype(np.float32)
+    diff = np.max(np.abs(rgb - bg), axis=2)
+    alpha = np.clip((diff - 4.0) / 16.0, 0.0, 1.0)
+
+    # Reject isolated texture flicker while preserving the body, wheels and
+    # soft contact shadow as one coherent object.
+    connected = (alpha > 0.08).astype(np.uint8)
+    count, labels, stats, _ = cv2.connectedComponentsWithStats(connected, 8)
+    if count > 1:
+        main = 1 + int(np.argmax(stats[1:, cv2.CC_STAT_AREA]))
+        keep = labels == main
+        near = cv2.dilate(keep.astype(np.uint8), np.ones((3, 3), np.uint8)).astype(bool)
+        alpha *= near
+    alpha = cv2.GaussianBlur(alpha, (3, 3), 0.55)
+    alpha[alpha < 0.04] = 0.0
+    return rgb, alpha
+
+
+def smoothstep(a, b, x):
+    t = np.clip((x - a) / (b - a), 0.0, 1.0)
+    return t * t * (3.0 - 2.0 * t)
+
+
+def path_position(frame):
+    """
+    Replay the source's ramp phase, then apply 30% of its measured
+    low-friction deceleration on the level run.
+    """
+    t = frame / FPS
+
+    # Launch and ramp, measured from the source body centroid.
+    if t < 0.33:
+        u = t / 0.33
+        x = 67.0 + 91.0 * (0.90 * u + 0.10 * smoothstep(0.0, 1.0, u))
+        y = 30.0 + 16.0 * smoothstep(0.06, 0.96, u)
+        angle = -1.5 * smoothstep(0.12, 0.82, u)
+        return x, y, angle
+
+    # At frame 8 the source is on the flat and travelling right at about
+    # 154.5 px/s. Only rolling resistance changes in the counterfactual.
+    dt = t - 0.33
+    x_ramp = 158.0
+    v0 = 154.5
+    decel = 0.30 * 250.96
+    x = x_ramp + v0 * dt - 0.5 * decel * dt * dt
+    y = 46.0 - 1.5 * smoothstep(0.33, 0.62, t)
+    angle = -1.5
+    return x, y, angle
+
+
+def transform_sprite(rgb, alpha, angle):
+    if abs(angle) < 1e-4:
+        return rgb, alpha
+    h, w = alpha.shape
+    center = (w * 0.5, h * 0.5)
+    matrix = cv2.getRotationMatrix2D(center, angle, 1.0)
+    warped_rgb = cv2.warpAffine(
+        rgb, matrix, (w, h), flags=cv2.INTER_LINEAR, borderMode=cv2.BORDER_REFLECT
+    )
+    warped_alpha = cv2.warpAffine(
+        alpha, matrix, (w, h), flags=cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT
+    )
+    return warped_rgb, warped_alpha
+
+
+def composite(frame, sprite_rgb, sprite_alpha, x, y):
+    h, w = sprite_alpha.shape
+    left = int(np.floor(x - w * 0.5))
+    top = int(np.floor(y - h * 0.5))
+    fx = x - w * 0.5 - left
+    fy = y - h * 0.5 - top
+
+    # Subpixel translation prevents the car from visibly stepping between
+    # 24 fps samples as it accelerates down the ramp.
+    matrix = np.float32([[1, 0, -fx], [0, 1, -fy]])
+    rgb = cv2.warpAffine(
+        sprite_rgb, matrix, (w, h), flags=cv2.INTER_LINEAR, borderMode=cv2.BORDER_REFLECT
+    )
+    alpha = cv2.warpAffine(
+        sprite_alpha, matrix, (w, h), flags=cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT
+    )
+
+    x0, y0 = max(0, left), max(0, top)
+    x1, y1 = min(frame.shape[1], left + w), min(frame.shape[0], top + h)
+    if x0 >= x1 or y0 >= y1:
+        return
+    sx0, sy0 = x0 - left, y0 - top
+    sx1, sy1 = sx0 + (x1 - x0), sy0 + (y1 - y0)
+    a = alpha[sy0:sy1, sx0:sx1, None]
+    src = rgb[sy0:sy1, sx0:sx1]
+    dst = frame[y0:y1, x0:x1].astype(np.float32)
+    frame[y0:y1, x0:x1] = np.clip(src * a + dst * (1.0 - a), 0, 255).astype(np.uint8)
+
+
+def main():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    plate = clean_plate(source)
+    sprite_rgb, sprite_alpha = extract_sprite(source, plate)
+
+    frames = np.repeat(plate[None], FRAME_COUNT, axis=0)
+    for index in range(FRAME_COUNT):
+        x, y, angle = path_position(index)
+        rotated_rgb, rotated_alpha = transform_sprite(sprite_rgb, sprite_alpha, angle)
+        composite(frames[index], rotated_rgb, rotated_alpha, x, y)
+
+    artifact_io.save(frames, CASE_DIR)
+
+
+if __name__ == "__main__":
+    main()
