+
+import argparse
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+SPEED_SCALE = 0.95
+
+
+def _flow_between(first: np.ndarray, second: np.ndarray) -> np.ndarray:
+    """Dense forward flow from first to second at source resolution."""
+    gray_first = cv2.cvtColor(first, cv2.COLOR_RGB2GRAY)
+    gray_second = cv2.cvtColor(second, cv2.COLOR_RGB2GRAY)
+    return cv2.calcOpticalFlowFarneback(
+        gray_first,
+        gray_second,
+        None,
+        pyr_scale=0.5,
+        levels=4,
+        winsize=21,
+        iterations=5,
+        poly_n=7,
+        poly_sigma=1.5,
+        flags=0,
+    )
+
+
+def _warp(frame: np.ndarray, flow: np.ndarray, amount: float) -> np.ndarray:
+    height, width = frame.shape[:2]
+    grid_x, grid_y = np.meshgrid(
+        np.arange(width, dtype=np.float32),
+        np.arange(height, dtype=np.float32),
+    )
+    sampled_x = grid_x + amount * flow[..., 0]
+    sampled_y = grid_y + amount * flow[..., 1]
+    return cv2.remap(
+        frame,
+        sampled_x,
+        sampled_y,
+        interpolation=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_REPLICATE,
+    )
+
+
+def _interpolate_pair(
+    first: np.ndarray,
+    second: np.ndarray,
+    flow: np.ndarray,
+    alpha: float,
+) -> np.ndarray:
+    if alpha <= 1e-6:
+        return first.copy()
+    if alpha >= 1.0 - 1e-6:
+        return second.copy()
+
+    forward = _warp(first, flow, alpha)
+    backward = _warp(second, -flow, 1.0 - alpha)
+    blended = (1.0 - alpha) * forward.astype(np.float32) + alpha * backward.astype(np.float32)
+    return np.clip(np.rint(blended), 0, 255).astype(np.uint8)
+
+
+def build_frames(source: np.ndarray) -> np.ndarray:
+    count = len(source)
+    flows = {}
+
+    def pair_flow(index: int) -> np.ndarray:
+        if index not in flows:
+            flows[index] = _flow_between(source[index], source[index + 1])
+        return flows[index]
+
+    frames = np.empty_like(source)
+    for output_index in range(count):
+        if output_index == 0:
+            frames[0] = source[0]
+            continue
+
+        source_time = min((count - 1) * SPEED_SCALE, output_index * SPEED_SCALE)
+        first_index = min(int(np.floor(source_time)), count - 2)
+        alpha = float(source_time - first_index)
+        frames[output_index] = _interpolate_pair(
+            source[first_index],
+            source[first_index + 1],
+            pair_flow(first_index),
+            alpha,
+        )
+    return frames
+
+
+def main() -> None:
+    parser = argparse.ArgumentParser()
+    parser.add_argument("--candidate", default="candidate_01.npz")
+    args = parser.parse_args()
+
+    source = np.load(ROOT / "source_pixels.npz")["rgb"]
+    frames = build_frames(source)
+
+    assert frames.shape == (96, 144, 256, 3)
+    assert frames.dtype == np.uint8
+    assert np.array_equal(frames[0], source[0])
+
+    np.savez_compressed(ROOT / args.candidate, rgb=frames)
+    artifact_io.save(frames, ROOT)
+
+
+if __name__ == "__main__":
+    main()
