+def _box_mask(box, shape=(HEIGHT, WIDTH)):
+    x0, y0, x1, y1 = box
+    mask = np.zeros(shape, dtype=np.uint8)
+    mask[y0:y1, x0:x1] = 255
+    return mask
+
+
+def _feather(mask, radius=2):
+    mask = np.asarray(mask, dtype=np.uint8)
+    if radius <= 0:
+        return mask.astype(np.float32) / 255.0
+    blurred = cv2.GaussianBlur(mask, (0, 0), radius)
+    return np.clip(blurred.astype(np.float32) / 255.0, 0.0, 1.0)
+
+
+def _row_background(frame, left_x=173, right_x=199):
+    """Fill a narrow upright footprint by interpolating clean flanking pixels."""
+    out = frame.astype(np.float32).copy()
+    x0, y0, x1, y1 = D4_INNER_BOX
+    for y in range(y0, y1):
+        left = frame[y, left_x].astype(np.float32)
+        right = frame[y, right_x].astype(np.float32)
+        t = np.arange(x1 - x0, dtype=np.float32) / max(1, x1 - x0 - 1)
+        out[y, x0:x1] = left[None, :] * (1.0 - t[:, None]) + right[None, :] * t[:, None]
+    return out
+
+
+def _remove_fourth(frame):
+    """Replace the upright fourth domino with the local foliage background."""
+    x0, y0, x1, y1 = D4_BOX
+    out = frame.astype(np.float32).copy()
+    plate = _row_background(frame)
+
+    # Hard interior plus a narrow feathered edge keeps the plate from bleeding
+    # across the domino's shadow and silhouette.
+    inner = _box_mask(D4_INNER_BOX).astype(np.float32) / 255.0
+    outer = _box_mask(D4_BOX).astype(np.float32) / 255.0
+    alpha = np.maximum(inner, cv2.GaussianBlur(outer, (0, 0), 1.15) * 0.52)
+    alpha = np.clip(alpha, 0.0, 1.0)[..., None]
+    out = out * (1.0 - alpha) + plate * alpha
+    return out
+
+
+def _d3_cutout(source):
+    """Extract domino three from its stable early pose with a soft silhouette."""
+    x0, y0, x1, y1 = D3_BOX
+    patch = source[y0:y1, x0:x1].astype(np.float32)
+    lum = patch.mean(axis=2)
+    sat = patch.max(axis=2) - patch.min(axis=2)
+    core = ((lum < 112.0) & (sat < 58.0)).astype(np.uint8) * 255
+    # The face has a pale top bevel that should travel with the rigid piece.
+    core[y0 + 1:y0 + 4, 5:11] = 255
+    kernel = np.ones((3, 3), np.uint8)
+    core = cv2.morphologyEx(core, cv2.MORPH_CLOSE, kernel)
+    core = cv2.dilate(core, kernel, iterations=1)
+    alpha = _feather(core, 0.75)
+    alpha = np.clip((alpha - 0.08) / 0.86, 0.0, 1.0)
+    return patch, alpha[..., None]
+
+
+def _rigid_warp(patch, alpha, angle, pivot):
+    """Rotate a cutout about a world-space pivot with subpixel sampling."""
+    h, w = patch.shape[:2]
+    matrix = cv2.getRotationMatrix2D((float(pivot[0]), float(pivot[1])), -float(angle), 1.0)
+    warped = cv2.warpAffine(
+        patch,
+        matrix,
+        (WIDTH, HEIGHT),
+        flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=0,
+    )
+    mask = cv2.warpAffine(
+        alpha,
+        matrix,
+        (WIDTH, HEIGHT),
+        flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=0,
+    )
+    return warped, np.clip(mask, 0.0, 1.0)[..., None]
+
+
+def _d3_angle(frame_index):
+    if frame_index <= D3_START_FRAME:
+        return D3_START_DEG
+    u = min(1.0, (frame_index - D3_START_FRAME) / (D3_END_FRAME - D3_START_FRAME))
+    # Accelerating rotation, followed by a short easing into the ground.
+    eased = 1.0 - (1.0 - u) ** 1.65
+    return D3_START_DEG + (D3_END_DEG - D3_START_DEG) * eased
+
+
+def _suppress_fourth_motion(frame):
+    """Replace the moving fourth domino and its shadow with nearby foliage."""
+    out = frame.astype(np.float32).copy()
+    x0, y0, x1, y1 = 174, 31, WIDTH, 111
+    strip = out[y0:y1, x0:x1]
+    # The late frames show that the right side of the scene is a continuous
+    # foliage band. Extend its row colors leftward through the moving piece.
+    source_x = min(WIDTH - 1, 232)
+    for yy in range(y0, y1):
+        target = frame[yy, source_x].astype(np.float32)
+        dist = np.abs(np.arange(x0, x1, dtype=np.float32) - source_x)
+        weight = np.clip(1.0 - dist / 58.0, 0.0, 1.0)[:, None]
+        strip[yy - y0] = strip[yy - y0] * (1.0 - weight) + target[None, :] * weight
+    out[y0:y1, x0:x1] = strip
+    return out
+
+
+def generate():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (FRAME_COUNT, HEIGHT, WIDTH, 3)
+    assert source.dtype == np.uint8
+
+    early = source[0]
+    patch, alpha = _d3_cutout(early)
+    result = np.empty_like(source)
+
+    for i in range(FRAME_COUNT):
+        base = _remove_fourth(source[i])
+
+        # Factual fourth-domino motion starts around source frame 34. Suppress
+        # that moving silhouette and then draw the unblocked third domino over it.
+        if i >= 34:
+            base = _suppress_fourth_motion(base)
+
+        if i >= D3_START_FRAME:
+            angle = _d3_angle(i)
+            warped, warped_alpha = _rigid_warp(patch, alpha, angle, D3_PIVOT)
+            base = base * (1.0 - warped_alpha) + warped * warped_alpha
+
+        result[i] = np.clip(np.rint(base), 0, 255).astype(np.uint8)
+
+    assert result.shape == source.shape
+    assert result.dtype == np.uint8
+    artifact_io.save(result, str(CASE_DIR))
+    return result
+
+

+    generate()
