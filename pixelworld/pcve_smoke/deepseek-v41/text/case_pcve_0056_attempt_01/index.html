+import cv2
+import numpy as np
+
+import artifact_io
+

+CASE_DIR = Path(__file__).resolve().parent
+FRAMES = 96

+def alpha_sprite(frame, plate, box, low=3.0, high=18.0):
+    """Lift a moving stone and its soft contact shadow from a clean plate."""
+    x0, y0, x1, y1 = box
+    diff = np.abs(frame[y0:y1, x0:x1].astype(np.float32) -
+                  plate[y0:y1, x0:x1].astype(np.float32)).max(axis=2)
+    alpha = np.clip((diff - low) / (high - low), 0.0, 1.0)
+    alpha = cv2.GaussianBlur(alpha, (3, 3), 0.55)
+    return frame[y0:y1, x0:x1].copy(), alpha, (x0, y0)
+
+
+def composite(dst, rgb, alpha, origin, center):
+    """Composite a sprite so its extracted bounding-box center lands on center."""
+    x0, y0 = origin
+    h, w = alpha.shape
+    dx = int(round(center[0] - (x0 + (w - 1) * 0.5)))
+    dy = int(round(center[1] - (y0 + (h - 1) * 0.5)))
+    xa, ya = max(0, dx), max(0, dy)
+    xb, yb = min(dst.shape[1], dx + w), min(dst.shape[0], dy + h)
+    if xa >= xb or ya >= yb:
+        return
+    sx0, sy0 = xa - dx, ya - dy
+    sx1, sy1 = sx0 + (xb - xa), sy0 + (yb - ya)
+    a = alpha[sy0:sy1, sx0:sx1, None]
+    dst[ya:yb, xa:xb] = (
+        rgb[sy0:sy1, sx0:sx1].astype(np.float32) * a
+        + dst[ya:yb, xa:xb].astype(np.float32) * (1.0 - a)
+    ).round().astype(np.uint8)
+
+
+def shifted_sprite(rgb, alpha, dx, dy):
+    """Translate an RGBA sprite with linear interpolation for subpixel motion."""
+    matrix = np.array([[1.0, 0.0, dx], [0.0, 1.0, dy]], dtype=np.float32)
+    moved = cv2.warpAffine(
+        rgb, matrix, (rgb.shape[1], rgb.shape[0]),
+        flags=cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT, borderValue=0,
+    )
+    moved_alpha = cv2.warpAffine(
+        alpha, matrix, (rgb.shape[1], rgb.shape[0]),
+        flags=cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT, borderValue=0,
+    )
+    return moved, moved_alpha
+
+
+def composite_full(dst, rgb, alpha):
+    a = alpha[:, :, None]
+    dst[:] = (
+        rgb.astype(np.float32) * a
+        + dst.astype(np.float32) * (1.0 - a)
+    ).round().astype(np.uint8)
+
+
+def red_center(frame_index):
+    """The small red stone's source trajectory, including its entry into the ring."""
+    if frame_index <= 30:
+        t = float(frame_index)
+        return 46.9 + 1.92 * t, 66.1 + 0.112 * t
+    if frame_index <= 40:
+        u = (frame_index - 30) / 10.0
+        ease = 1.0 - (1.0 - u) ** 2
+        return 104.5 + 10.0 * ease, 69.5 + 4.0 * ease
+    return 114.5 + 0.18 * min(frame_index - 40, 12) / 12.0, 73.5
+
+
+def yellow_center(frame_index):
+    """The moved blue ring blocks the yellow stone, reversing its approach."""
+    if frame_index == 0:
+        return 213.5, 76.3
+    t = float(frame_index - 1)
+    # Elastic rebound to the right, with mild damping after contact.
+    speed = 1.65
+    x = 213.5 + speed * t - 0.0038 * t * t
+    y = 76.2 + 0.0009 * t * t
+    return x, y
+

+def main():

+    cv2.ellipse(outer, (128, 80), (62, 29), 0, 0, 360, 255, -1)
+    cv2.ellipse(inner, (128, 80), (40, 20), 0, 0, 360, 255, -1)

+    ring_alpha = np.clip(ring_alpha * 1.15, 0.0, 1.0)
+    ring_rgb = first.copy()
+    ring_rgb[ring_alpha < 0.02] = 0
+
+    red_rgb, red_alpha, red_origin = alpha_sprite(
+        first, plate, (38, 62, 60, 76), low=3.0, high=16.0
+    )
+    yellow_rgb, yellow_alpha, yellow_origin = alpha_sprite(
+        first, plate, (202, 70, 224, 89), low=3.0, high=16.0
+    )
+
+    # Quarter of the way from the yellow stone to the red stone.
+    red_ref = np.array([127.5, 75.5], dtype=np.float32)
+    yellow_ref = np.array([213.5, 76.3], dtype=np.float32)
+    blue_target = yellow_ref + 0.25 * (red_ref - yellow_ref)
+    blue_source = np.array([127.5, 79.5], dtype=np.float32)
+    blue_delta = blue_target - blue_source
+
+    moved_ring_rgb, moved_ring_alpha = shifted_sprite(
+        ring_rgb, ring_alpha, float(blue_delta[0]), float(blue_delta[1])
+    )
+
+    output = np.empty_like(source)
+    for index in range(FRAMES):
+        if index == 0:
+            output[index] = first
+            continue
+
+        frame = plate.copy()
+        composite_full(frame, moved_ring_rgb, moved_ring_alpha)
+        composite(frame, red_rgb, red_alpha, red_origin, red_center(index))
+        composite(frame, yellow_rgb, yellow_alpha, yellow_origin, yellow_center(index))
+        output[index] = frame
+
+    artifact_io.save(output, CASE_DIR)
+
+
+if __name__ == "__main__":
+    main()
