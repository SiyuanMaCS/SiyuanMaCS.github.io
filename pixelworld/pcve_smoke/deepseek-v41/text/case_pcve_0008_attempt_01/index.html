+"""Remove the moving white mallet from every frame of the source clip."""
+
+from pathlib import Path
+
+import numpy as np
+from scipy import ndimage as ndi
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def moving_region(frames: np.ndarray) -> np.ndarray:
+    """Return the complete swept silhouette of the moving mallet."""
+    temporal_range = np.ptp(frames.astype(np.int16), axis=0).max(axis=2)
+    candidates = temporal_range > 10
+    labels, count = ndi.label(candidates, structure=np.ones((3, 3), dtype=bool))
+    if count == 0:
+        raise RuntimeError("No moving object found")
+
+    sizes = np.bincount(labels.ravel())
+    sizes[0] = 0
+    largest = int(np.argmax(sizes))
+    if sizes[largest] < 100:
+        raise RuntimeError(f"Moving object is unexpectedly small: {sizes[largest]}")
+
+    # The thresholded core misses a narrow antialiased fringe. Expand it and
+    # close small holes before filling the interior of the swept silhouette.
+    region = labels == largest
+    region = ndi.binary_dilation(region, iterations=2)
+    region = ndi.binary_closing(region, structure=np.ones((5, 5), dtype=bool))
+    return ndi.binary_fill_holes(region)
+
+
+def clean_plate(frames: np.ndarray, region: np.ndarray) -> np.ndarray:
+    """Recover the unobstructed background beneath the mallet's path.
+
+    The camera and remaining scene are static. Most path pixels are observed
+    cleanly in several frames. For those, use a temporal median; only pixels
+    that are contaminated at both ends of the clip need a reconstructed value.
+    """
+    work = frames.astype(np.float32)
+    ys, xs = np.where(region)
+    y0, y1 = max(0, ys.min() - 6), min(frames.shape[1], ys.max() + 7)
+    x0, x1 = max(0, xs.min() - 6), min(frames.shape[2], xs.max() + 7)
+    crop = work[:, y0:y1, x0:x1]
+    crop_region = region[y0:y1, x0:x1]
+
+    temporal_median = np.median(crop, axis=0)
+    early = crop[:12]
+    late = crop[-12:]
+    early_median = np.median(early, axis=0)
+    late_median = np.median(late, axis=0)
+    early_agreement = np.median(
+        np.abs(early - early_median[None]).max(axis=3), axis=0
+    )
+    late_agreement = np.median(
+        np.abs(late - late_median[None]).max(axis=3), axis=0
+    )
+
+    # Stable endpoint samples are directly observed background. The mallet
+    # occupies the upper-right start and lower-left finish, so combining these
+    # two observations exposes almost the entire swept path.
+    early_clean = early_agreement <= 8
+    late_clean = late_agreement <= 8
+    plate = temporal_median.copy()
+    use_early = crop_region & early_clean
+    use_late = crop_region & late_clean & ~use_early
+    plate[use_early] = early_median[use_early]
+    plate[use_late] = late_median[use_late]
+
+    # For the few pixels obscured at both endpoints, use the dominant stable
+    # color in the middle of the clip.
+    middle = crop[12:-12]
+    quantized = np.rint(middle / 4).astype(np.int16)
+    keys = (quantized[..., 0].astype(np.int32) * 4096
+            + quantized[..., 1].astype(np.int32) * 64
+            + quantized[..., 2].astype(np.int32))
+    unresolved = crop_region & ~(early_clean | late_clean)
+    for y, x in zip(*np.where(unresolved)):
+        values = keys[:, y, x]
+        counts = np.bincount(values)
+        first = int(np.flatnonzero(values == counts.max())[0])
+        plate[y, x] = middle[first, y, x]
+
+    # Repair any remaining isolated outlier using a nearest clean neighbor. The
+    # background here is a smooth tabletop, so this changes no real structure.
+    local = ndi.median_filter(plate, size=(1, 5, 5), mode="nearest")
+    difference = np.abs(plate - local).max(axis=2)
+    suspicious = crop_region & (difference > 45)
+    if suspicious.any():
+        indices = ndi.distance_transform_edt(
+            suspicious, return_distances=False, return_indices=True
+        )
+        plate[suspicious] = plate[tuple(indices)][suspicious]
+
+    result = work[0].copy()
+    result[y0:y1, x0:x1] = plate
+    return result
+
+
+def main() -> None:
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    region = moving_region(source)
+    plate = clean_plate(source, region)
+
+    output = source.copy()
+    output[:, region] = np.rint(plate[region]).clip(0, 255).astype(np.uint8)
+    artifact_io.save(output, CASE_DIR)
+
+
+if __name__ == "__main__":
+    main()
