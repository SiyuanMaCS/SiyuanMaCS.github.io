+"""Retiming edit: make the volleyball advance at 1.5x its source speed."""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+FRAME_COUNT = 96
+FPS = 24.0
+SPEED_FACTOR = 1.5
+SOURCE_MOTION_END = 18
+

+def _restore_receiving_target(background, first_frame):
+    """Remove the resting ball from the temporal median while keeping the target."""
+    x0, y0, x1, y1 = 99, 54, 126, 89
+    restored = background.copy()
+    restored[y0:y1, x0:x1] = first_frame[y0:y1, x0:x1]
+    return restored
+
+
+def _moving_matte(frame, background, center_hint, previous_center):
+    """Build a clean, feathered matte for the moving ball and its contact shadow."""
+    delta = np.abs(frame.astype(np.int16) - background.astype(np.int16)).max(axis=2)
+    raw = (delta > 8).astype(np.uint8)
+
+    # Ignore tiny codec differences away from the ball's local neighborhood.
+    local = np.zeros_like(raw)
+    cx, cy = center_hint
+    x0, x1 = max(0, int(cx) - 35), min(frame.shape[1], int(cx) + 36)
+    y0, y1 = max(0, int(cy) - 30), min(frame.shape[0], int(cy) + 31)
+    local[y0:y1, x0:x1] = raw[y0:y1, x0:x1]
+
+    count, labels, stats, centroids = cv2.connectedComponentsWithStats(local, 8)
+    if count <= 1:
+        return np.zeros(frame.shape[:2], np.float32), previous_center
+
+    candidates = []
+    for label in range(1, count):
+        area = int(stats[label, cv2.CC_STAT_AREA])
+        if area < 18:
+            continue
+        cx_i, cy_i = centroids[label]
+        distance = (cx_i - center_hint[0]) ** 2 + (cy_i - center_hint[1]) ** 2
+        candidates.append((distance - area * 0.03, label))
+    if not candidates:
+        return np.zeros(frame.shape[:2], np.float32), previous_center
+
+    label = min(candidates)[1]
+    component = (labels == label).astype(np.uint8)
+
+    # Fill internal panel gaps and bridge the small gap to the contact shadow.
+    component = cv2.morphologyEx(component, cv2.MORPH_CLOSE, np.ones((5, 5), np.uint8))
+    component = cv2.morphologyEx(component, cv2.MORPH_OPEN, np.ones((2, 2), np.uint8))
+
+    ys, xs = np.where(component)
+    center = (float(xs.mean()), float(ys.mean())) if len(xs) else previous_center
+    alpha = cv2.GaussianBlur(component.astype(np.float32), (3, 3), 0.55)
+    alpha = np.maximum(alpha, component.astype(np.float32) * 0.92)
+    return np.clip(alpha, 0.0, 1.0), center
+

+def _extract_moving_layer(source, background):
+    """Extract the original ball/shadow sprite and estimate its source-frame centers."""
+    h, w = source.shape[1:3]
+    alpha = np.zeros((len(source), h, w), np.float32)
+    centers = []
+    previous = (80.0, 121.0)
+    for index, frame in enumerate(source):
+        if index <= SOURCE_MOTION_END:
+            # The foreground follows a smooth diagonal path; these hints only isolate it.
+            progress = min(index / SOURCE_MOTION_END, 1.0)
+            hint = (
+                80.0 + 35.0 * progress,
+                121.0 - 54.0 * progress,
+            )
+        else:
+            hint = previous
+        matte, previous = _moving_matte(frame, background, hint, previous)
+        alpha[index] = matte
+        centers.append(previous)
+    return alpha, np.asarray(centers, np.float32)
+
+
+def _sprite_from_frame(frame, alpha):
+    """Crop a compact RGBA sprite around nonzero matte pixels."""
+    ys, xs = np.where(alpha > 0.025)
+    if not len(xs):
+        return None
+    pad = 3
+    x0 = max(0, int(xs.min()) - pad)
+    y0 = max(0, int(ys.min()) - pad)
+    x1 = min(frame.shape[1], int(xs.max()) + pad + 1)
+    y1 = min(frame.shape[0], int(ys.max()) + pad + 1)
+    rgb = frame[y0:y1, x0:x1].astype(np.float32)
+    a = alpha[y0:y1, x0:x1]
+    return rgb, a, np.array([x0, y0], np.float32)
+
+
+def _place_sprite(canvas, sprite, center_target):
+    rgb, alpha, origin = sprite
+    center_source = origin + np.array(
+        [alpha.shape[1] * 0.5, alpha.shape[0] * 0.5], np.float32
+    )
+    shift = np.asarray(center_target, np.float32) - center_source
+
+    matrix = np.array([[1.0, 0.0, shift[0]], [0.0, 1.0, shift[1]]], np.float32)
+    warped_rgb = cv2.warpAffine(
+        rgb,
+        matrix,
+        (canvas.shape[1], canvas.shape[0]),
+        flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=(0, 0, 0),
+    )
+    warped_a = cv2.warpAffine(
+        alpha,
+        matrix,
+        (canvas.shape[1], canvas.shape[0]),
+        flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=0,
+    )[..., None]
+    return canvas.astype(np.float32) * (1.0 - warped_a) + warped_rgb * warped_a
+
+
+def _interpolated_sprite(source, alpha, source_time):
+    """Return the sprite at a fractional source frame."""
+    low = int(np.floor(source_time))
+    high = min(low + 1, len(source) - 1)
+    mix = float(source_time - low)
+    sprite_low = _sprite_from_frame(source[low], alpha[low])
+    sprite_high = _sprite_from_frame(source[high], alpha[high])
+    if sprite_low is None:
+        return sprite_high
+    if sprite_high is None or high == low:
+        return sprite_low
+
+    # Sample each sprite at its own source location, then blend with motion
+    # compensation so intermediate frames do not contain two separated balls.
+    ys_l, xs_l = np.where(alpha[low] > 0.025)
+    ys_h, xs_h = np.where(alpha[high] > 0.025)
+    center_low = np.array([xs_l.mean(), ys_l.mean()], np.float32)
+    center_high = np.array([xs_h.mean(), ys_h.mean()], np.float32)
+
+    h, w = source.shape[1:3]
+    rgb_l = np.zeros((h, w, 3), np.float32)
+    a_l = np.zeros((h, w), np.float32)
+    rgb_h = np.zeros((h, w, 3), np.float32)
+    a_h = np.zeros((h, w), np.float32)
+
+    for sprite, center, rgb_out, a_out in (
+        (sprite_low, center_low, rgb_l, a_l),
+        (sprite_high, center_high, rgb_h, a_h),
+    ):
+        rgb, a, origin = sprite
+        sprite_center = origin + np.array([a.shape[1] * 0.5, a.shape[0] * 0.5])
+        shift = center - sprite_center
+        matrix = np.array(
+            [[1.0, 0.0, shift[0]], [0.0, 1.0, shift[1]]], np.float32
+        )
+        rgb_out[:] = cv2.warpAffine(rgb, matrix, (w, h), flags=cv2.INTER_LINEAR)
+        a_out[:] = cv2.warpAffine(a, matrix, (w, h), flags=cv2.INTER_LINEAR)
+
+    rgb = rgb_l * (1.0 - mix) + rgb_h * mix
+    a = a_l * (1.0 - mix) + a_h * mix
+    return rgb, a, np.zeros(2, np.float32)
+
+
+def generate():
+    pixels = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    source = pixels.astype(np.uint8)
+
+    background = np.median(source[24:96], axis=0).round().astype(np.uint8)
+    background = _restore_receiving_target(background, source[0])
+    alpha, centers = _extract_moving_layer(source, background)
+
+    output = np.empty((FRAME_COUNT,) + source.shape[1:], np.uint8)
+    for out_index in range(FRAME_COUNT):
+        source_time = min(out_index * SPEED_FACTOR, float(len(source) - 1))
+        low = int(np.floor(source_time))
+        high = min(low + 1, len(source) - 1)
+        mix = source_time - low
+        target_center = centers[low] * (1.0 - mix) + centers[high] * mix
+
+        sprite = _interpolated_sprite(source, alpha, source_time)
+        if sprite is None:
+            frame = background.astype(np.float32)
+        else:
+            frame = _place_sprite(background.astype(np.float32), sprite, target_center)
+        output[out_index] = np.clip(frame + 0.5, 0, 255).astype(np.uint8)
+
+    # Frame 1 is the intervention point and must retain the original initial state.
+    output[0] = source[0]
+    return output
+
+
+if __name__ == "__main__":
+    frames = generate()
+    artifact_io.save(frames, CASE_DIR)
+    print(f"saved {len(frames)} frames at {frames.shape[2]}x{frames.shape[1]}")
