+"""Render the cola-can scene at half its original motion speed."""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+OUTPUT_FRAMES = 96
+
+
+def _motion_midpoint(first, second):
+    """Interpolate two frames with forward/backward dense optical flow."""
+    first_gray = cv2.cvtColor(first, cv2.COLOR_RGB2GRAY)
+    second_gray = cv2.cvtColor(second, cv2.COLOR_RGB2GRAY)
+    flow_options = dict(
+        pyr_scale=0.5,
+        levels=5,
+        winsize=21,
+        iterations=3,
+        poly_n=5,
+        poly_sigma=1.2,
+        flags=0,
+    )
+    forward = cv2.calcOpticalFlowFarneback(
+        first_gray, second_gray, None, **flow_options
+    )
+    backward = cv2.calcOpticalFlowFarneback(
+        second_gray, first_gray, None, **flow_options
+    )
+
+    height, width = first_gray.shape
+    grid_x, grid_y = np.meshgrid(
+        np.arange(width, dtype=np.float32),
+        np.arange(height, dtype=np.float32),
+    )
+    map_x_first = grid_x + 0.5 * forward[..., 0]
+    map_y_first = grid_y + 0.5 * forward[..., 1]
+    map_x_second = grid_x + 0.5 * backward[..., 0]
+    map_y_second = grid_y + 0.5 * backward[..., 1]
+    remap_options = dict(
+        interpolation=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_REPLICATE,
+    )
+    warped_first = cv2.remap(
+        first, map_x_first, map_y_first, **remap_options
+    )
+    warped_second = cv2.remap(
+        second, map_x_second, map_y_second, **remap_options
+    )
+    return (warped_first.astype(np.float32) + warped_second) * 0.5
+
+
+def _half_speed_midpoint(first, second):
+    """Use flow only where objects move, retaining exact static pixels elsewhere."""
+    linear = (first.astype(np.float32) + second.astype(np.float32)) * 0.5
+    try:
+        warped = _motion_midpoint(first, second)
+    except cv2.error:
+        return np.clip(np.rint(linear), 0, 255).astype(np.uint8)
+
+    height = first.shape[0]
+    rows = np.arange(height, dtype=np.float32)
+    alpha = np.zeros(height, dtype=np.float32)
+    alpha[(rows >= 48.0) & (rows <= 92.0)] = 1.0
+    alpha[(rows >= 44.0) & (rows < 48.0)] = (
+        rows[(rows >= 44.0) & (rows < 48.0)] - 44.0
+    ) / 4.0
+    alpha[(rows > 92.0) & (rows <= 96.0)] = (
+        96.0 - rows[(rows > 92.0) & (rows <= 96.0)]
+    ) / 4.0
+    blended = linear + (warped - linear) * alpha[:, None, None]
+    return np.clip(np.rint(blended), 0, 255).astype(np.uint8)
+
+
+def build_video(source):
+    """Map output time t to source time 0.5*t from the intervention onward."""
+    frames = np.empty_like(source)
+    for output_index in range(OUTPUT_FRAMES):
+        source_index, phase = divmod(output_index, 2)
+        if phase == 0:
+            frames[output_index] = source[source_index]
+        else:
+            frames[output_index] = _half_speed_midpoint(
+                source[source_index], source[source_index + 1]
+            )
+    return frames
+
+
+def main():
+    with np.load(CASE_DIR / "source_pixels.npz") as source_data:
+        source = source_data["rgb"]
+
+    assert source.shape == (96, 144, 256, 3), source.shape
+    assert source.dtype == np.uint8, source.dtype
+
+    frames = build_video(source)
+    assert frames.shape == source.shape and frames.dtype == np.uint8
+    assert np.array_equal(frames[0], source[0])
+
+    artifact_io.save(frames, CASE_DIR)
+    print(f"Saved {len(frames)} frames to {CASE_DIR}")
+
+
+if __name__ == "__main__":
+    main()
