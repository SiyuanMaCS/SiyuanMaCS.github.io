+import cv2
+import numpy as np

+import artifact_io

+CASE_DIR = Path(__file__).resolve().parent

+def select_component(raw_mask, seed):
+    count, labels = cv2.connectedComponents(raw_mask.astype(np.uint8), connectivity=8)
+    seed_x, seed_y = seed
+    if count <= 1:
+        raise RuntimeError("No foreground component found")
+
+    label = labels[seed_y, seed_x]
+    if label == 0:
+        ys, xs = np.nonzero(raw_mask)
+        distance = (xs - seed_x) ** 2 + (ys - seed_y) ** 2
+        label = labels[ys[np.argmin(distance)], xs[np.argmin(distance)]]
+    return labels == label
+
+
+def extract_sprite(source, clean_plate, roi, seed):
+    x0, y0, x1, y1 = roi
+    source_patch = source[y0:y1, x0:x1].astype(np.float32)
+    clean_patch = clean_plate[y0:y1, x0:x1].astype(np.float32)
+    difference = np.max(np.abs(source_patch - clean_patch), axis=2)
+
+    raw_mask = difference > 3.0
+    mask = select_component(raw_mask, (seed[0] - x0, seed[1] - y0))
+
+    # A soft edge retains the source anti-aliasing without importing background noise.
+    alpha = np.clip((difference - 2.0) / 9.0, 0.0, 1.0)
+    alpha *= mask.astype(np.float32)
+    alpha = cv2.GaussianBlur(alpha, (3, 3), 0.45)
+    alpha = np.clip(alpha * 1.08, 0.0, 1.0)
+    return {
+        "rgb": source_patch,
+        "alpha": alpha,
+        "x": x0,
+        "y": y0,
+    }
+
+
+def paste(canvas, sprite, dx=0.0, dy=0.0):
+    x = int(round(sprite["x"] + dx))
+    y = int(round(sprite["y"] + dy))
+    height, width = sprite["alpha"].shape
+    x0 = max(0, x)
+    y0 = max(0, y)
+    x1 = min(canvas.shape[1], x + width)
+    y1 = min(canvas.shape[0], y + height)
+    if x0 >= x1 or y0 >= y1:
+        return
+
+    sx0 = x0 - x
+    sy0 = y0 - y
+    sx1 = sx0 + (x1 - x0)
+    sy1 = sy0 + (y1 - y0)
+    alpha = sprite["alpha"][sy0:sy1, sx0:sx1, None]
+    color = sprite["rgb"][sy0:sy1, sx0:sx1]
+    destination = canvas[y0:y1, x0:x1]
+    canvas[y0:y1, x0:x1] = destination * (1.0 - alpha) + color * alpha
+
+
+def mallet_offset(frame):
+    elapsed = max(0, frame - CONTACT_FRAME)
+    drift = 1.0 - np.exp(-elapsed / 15.0)
+    vibration = np.exp(-elapsed / 8.0)
+    dx = -2.7 * drift + 0.28 * vibration * np.sin(1.45 * elapsed)
+    dy = -1.0 * drift + 0.16 * vibration * np.cos(1.35 * elapsed)
+    return dx, dy
+
+
+def red_piece_offset(frame):
+    elapsed = max(0, frame - CONTACT_FRAME)
+    rebound = 1.0 - np.exp(-elapsed / 7.5)
+    vibration = np.exp(-elapsed / 8.5)
+    dx = -15.0 + 4.2 * rebound + 0.45 * vibration * np.sin(1.6 * elapsed)
+    dy = 8.0 - 1.9 * rebound + 0.32 * vibration * np.cos(1.45 * elapsed)
+    return dx, dy
+
+
+def blue_piece_offset(frame):
+    elapsed = max(0, frame - CONTACT_FRAME)
+    vibration = np.exp(-elapsed / 7.0)
+    dx = -21.0 - 0.35 * vibration * np.sin(1.5 * elapsed)
+    dy = 8.0 + 0.25 * vibration * np.cos(1.25 * elapsed)
+    return dx, dy
