+def largest_block_component(frame):
+    """Find the dark wooden block, excluding the bright orange arm."""
+    luma = frame.mean(axis=2)
+    candidate = (luma < 65).astype(np.uint8)
+    candidate = cv2.morphologyEx(
+        candidate, cv2.MORPH_OPEN, np.ones((2, 2), np.uint8)
+    )
+    count, labels, stats, centroids = cv2.connectedComponentsWithStats(
+        candidate, connectivity=8
+    )
+
+    options = []
+    for label in range(1, count):
+        x, y, width, height, area = stats[label]
+        if (
+            x > 85
+            and 30 <= width <= 85
+            and 30 <= height <= 55
+            and area > 700
+        ):
+            options.append((int(area), label))
+    if not options:
+        raise RuntimeError("Wooden block component was not found")
+
+    _, label = max(options)
+    mask = (labels == label).astype(np.uint8)
+    contours, _ = cv2.findContours(
+        mask, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE
+    )
+    cv2.drawContours(mask, contours, -1, 1, thickness=-1)
+    return mask, centroids[label].copy()
+
+
+def track_block(source):
+    """Recover the block silhouette through the arm-contact frames."""
+    masks = []
+    centers = []
+    for frame in source:
+        mask, center = largest_block_component(frame)
+        masks.append(mask)
+        centers.append(center)
+
+    # Frames 11-14 merge the block with the incoming arm. Bridge that short
+    # overlap with a translated copy of the known block silhouette.
+    for time in range(11, 15):
+        alpha = (time - 10) / 5.0
+        center = centers[10] * (1.0 - alpha) + centers[15] * alpha
+        shift = center - centers[0]
+        transform = np.float32(
+            [[1.0, 0.0, shift[0]], [0.0, 1.0, shift[1]]]
+        )
+        masks[time] = cv2.warpAffine(
+            masks[0],
+            transform,
+            (source.shape[2], source.shape[1]),
+            flags=cv2.INTER_NEAREST,
+        )
+        centers[time] = center
+
+    return np.stack(masks), np.stack(centers)
+
+
+def make_clean_plate(source):
+    """Reuse unobstructed source pixels from the two halves of the sweep."""
+    height, width = source.shape[1:3]
+    left = np.median(source[60:96, :, :168], axis=0)
+    right = np.median(source[0:10, :, 168:], axis=0)
+    plate = np.concatenate([left, right], axis=1)
+
+    # Hide the source-composite seam under a short linear transition.
+    for x in range(164, 172):
+        weight = (x - 163) / 9.0
+        left_pixel = np.median(source[60:96, :, x], axis=0)
+        right_pixel = np.median(source[0:10, :, x], axis=0)
+        plate[:, x] = (1.0 - weight) * left_pixel + weight * right_pixel
+    return plate
+
+
+def make_repair_mask(frame, plate, seed, center):
+    """Expand the block seed to its antialiased edge and contact shadow."""
+    adjusted = plate + np.median(
+        frame[55:138, 80:255] - plate[55:138, 80:255], axis=(0, 1)
+    )
+    residual = np.linalg.norm(frame - adjusted, axis=2)
+
+    radius_x = max(38, int(seed.sum() ** 0.5))
+    radius_y = max(30, int(radius_x * 0.78))
+    x0 = max(0, int(center[0]) - radius_x - 10)
+    x1 = min(frame.shape[1], int(center[0]) + radius_x + 11)
+    y0 = max(45, int(center[1]) - radius_y - 8)
+    y1 = min(frame.shape[0], int(center[1]) + radius_y + 9)
+
+    footprint = np.zeros(frame.shape[:2], np.uint8)
+    local = residual[y0:y1, x0:x1] > 8.0
+    footprint[y0:y1, x0:x1] = local.astype(np.uint8)
+    footprint = cv2.morphologyEx(
+        footprint, cv2.MORPH_CLOSE, np.ones((5, 5), np.uint8)
+    )
+    footprint = cv2.dilate(footprint, np.ones((3, 3), np.uint8))
+
+    count, labels, stats, _ = cv2.connectedComponentsWithStats(
+        footprint, connectivity=8
+    )
+    touched = np.zeros_like(footprint)
+    for label in range(1, count):
+        component = labels == label
+        if np.any(component & (seed > 0)) or stats[label, 4] >= 120:
+            touched[component] = 1
+
+    merged = ((touched > 0) | (seed > 0)).astype(np.uint8)
+    contours, _ = cv2.findContours(
+        merged, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE
+    )
+    filled = np.zeros_like(merged)
+    cv2.drawContours(filled, contours, -1, 1, thickness=-1)
+    return filled
+
+
+def render():
+    packed = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    source = packed.astype(np.float32)
+    seeds, centers = track_block(source)
+    plate = make_clean_plate(source)
+
+    output = np.empty_like(packed)
+    masks = []
+    for time, (frame, seed, center) in enumerate(zip(source, seeds, centers)):
+        mask = make_repair_mask(frame, plate, seed, center)
+        masks.append(mask)
+
+        offset = np.median(
+            frame[55:138, 80:255] - plate[55:138, 80:255], axis=(0, 1)
+        )
+        replacement = np.clip(plate + offset, 0, 255)
+        alpha = cv2.GaussianBlur(mask.astype(np.float32), (0, 0), 1.15)
+        alpha = np.clip(alpha * 1.35, 0, 1)[..., None]
+        repaired = frame * (1.0 - alpha) + replacement * alpha
+        output[time] = np.clip(repaired + 0.5, 0, 255).astype(np.uint8)
+
+    debug = output.copy()
+    for time in range(0, len(debug), 12):
+        edge = cv2.morphologyEx(
+            masks[time], cv2.MORPH_GRADIENT, np.ones((3, 3), np.uint8)
+        )
+        debug[time][edge > 0] = np.array([0, 255, 0], dtype=np.uint8)
+    Image.fromarray(debug[0]).save(CASE_DIR / ".tmp" / "mask_check.png")
+    return output
+
+
+def main():
+    (CASE_DIR / ".tmp").mkdir(exist_ok=True)
+    frames = render()
+    artifact_io.save(frames, CASE_DIR)
+
+
+if __name__ == "__main__":
+    main()
