+
+The source shows a ball rolling away from the camera and coming to rest.  The
+counterfactual triples its friction, so the same initial roll stops in about a
+third of the distance and time.  The static scene is taken from the supplied
+frames; the ball is extracted as an RGBA sprite and re-composited along the
+new trajectory.

+import cv2
+import numpy as np
+from PIL import Image
+from scipy import ndimage

+import artifact_io

+W, H, N = 256, 144, 96

+def smoothstep(x):
+    x = np.clip(x, 0.0, 1.0)
+    return x * x * (3.0 - 2.0 * x)
+

+def make_clean_plate(source):
+    """Remove the settled ball and its contact shadow from a stable source frame."""
+    plate = source[40].copy()
+
+    # The floor under the final ball is continuous with the patch immediately
+    # to its left.  Copying that patch keeps the floor grain and lighting
+    # consistent while removing the object and its small occlusion shadow.
+    src_x0, src_x1 = 92, 111
+    dst_x0, dst_x1 = 105, 124
+    y0, y1 = 55, 78
+    patch = source[40, y0:y1, src_x0:src_x1].astype(np.float32)
+    patch = cv2.GaussianBlur(patch, (0, 0), 0.55)
+    yy, xx = np.mgrid[y0:y1, dst_x0:dst_x1]
+    alpha_x = smoothstep((xx - (dst_x0 + 1.5)) / 3.0) * smoothstep(((dst_x1 - 1.5) - xx) / 3.0)
+    alpha_y = smoothstep((yy - (y0 + 2.0)) / 4.0) * smoothstep(((y1 - 2.0) - yy) / 4.0)
+    alpha = (alpha_x * alpha_y)[..., None]
+    plate[y0:y1, dst_x0:dst_x1] = (
+        alpha * patch + (1.0 - alpha) * plate[y0:y1, dst_x0:dst_x1]
+    ).astype(np.uint8)
+
+    # Remove the long, soft shadow left of the original ball.  A local
+    # unsharpened floor estimate replaces only the darker residue.
+    roi = plate[104:140, 43:101].astype(np.float32)
+    smooth = cv2.GaussianBlur(roi, (0, 0), 9.0)
+    lum = roi.mean(axis=2)
+    sm_lum = smooth.mean(axis=2)
+    darkness = np.clip((sm_lum - lum - 2.0) / 30.0, 0.0, 1.0)
+    feather = cv2.GaussianBlur(darkness, (0, 0), 1.0)
+    out = plate[104:140, 43:101].astype(np.float32)
+    plate[104:140, 43:101] = (
+        feather[..., None] * smooth + (1.0 - feather[..., None]) * out
+    ).astype(np.uint8)
+
+    return plate
+

+def extract_ball(source, plate):
+    """Extract a perspective-corrected RGBA sprite from the first source frame."""
+    # The first frame is the clearest large view of the ball.  Its silhouette
+    # is bounded by a color transition from the brown floor and dark contact.
+    x0, x1, y0, y1 = 67, 99, 105, 136
+    crop = source[0, y0:y1, x0:x1].astype(np.float32)
+    bg = plate[y0:y1, x0:x1].astype(np.float32)
+
+    diff = np.abs(crop - bg).mean(axis=2)
+    sat = crop.max(axis=2) - crop.min(axis=2)
+    lum = crop.mean(axis=2)
+    # The ball is substantially brighter/more chromatic than the floor.  A
+    # low threshold recovers the rim, while the final morphology removes
+    # isolated floor pixels.
+    mask = (diff > 12.0) | ((sat > 38.0) & (lum > 65.0))
+    mask = ndimage.binary_closing(mask, iterations=1)
+    mask = ndimage.binary_opening(mask, iterations=1)
+    labels, count = ndimage.label(mask)
+    if count:
+        sizes = ndimage.sum(mask, labels, range(1, count + 1))
+        keep = 1 + int(np.argmax(sizes))
+        mask = labels == keep
+
+    # Reconstruct the small anti-aliased rim instead of cutting it with a hard
+    # binary edge.  A slightly larger ellipse is intersected with the color
+    # evidence and then softened.
+    yy, xx = np.mgrid[y0:y1, x0:x1]
+    cx, cy = 82.0, 121.0
+    ellipse = ((xx - cx) / 14.2) ** 2 + ((yy - cy) / 15.1) ** 2 <= 1.0
+    mask = mask | (ellipse & (diff > 7.0))

+def resize_rgba(rgba, scale):
+    h, w = rgba.shape[:2]
+    nw = max(3, int(round(w * scale)))
+    nh = max(3, int(round(h * scale)))
+    resized = cv2.resize(rgba, (nw, nh), interpolation=cv2.INTER_LANCZOS4)
+    return resized

+def paste_rgba(frame, sprite, center):
+    """Alpha-composite a small RGBA sprite into an RGB uint8 frame."""
+    h, w = sprite.shape[:2]
+    cx, cy = center
+    x0 = int(round(cx - w / 2.0))
+    y0 = int(round(cy - h / 2.0))
+    x1, y1 = x0 + w, y0 + h
+    fx0, fy0 = max(0, x0), max(0, y0)
+    fx1, fy1 = min(W, x1), min(H, y1)
+    if fx0 >= fx1 or fy0 >= fy1:
+        return
+    sx0, sy0 = fx0 - x0, fy0 - y0
+    sx1, sy1 = sx0 + (fx1 - fx0), sy0 + (fy1 - fy0)
+    src = sprite[sy0:sy1, sx0:sx1].astype(np.float32)
+    a = src[..., 3:4] / 255.0
+    dst = frame[fy0:fy1, fx0:fx1].astype(np.float32)
+    frame[fy0:fy1, fx0:fx1] = np.clip(src[..., :3] * a + dst * (1.0 - a), 0, 255).astype(np.uint8)
+

+def draw_contact_shadow(frame, center, radius):
+    """Add a soft, directional contact shadow below the ball."""
+    cx, cy = center
+    pad = 18
+    x0, x1 = max(0, int(cx) - pad), min(W, int(cx) + pad + 1)
+    y0, y1 = max(0, int(cy) - 8), min(H, int(cy) + 13)
+    if x0 >= x1 or y0 >= y1:
+        return
+    yy, xx = np.mgrid[y0:y1, x0:x1]
+    # Contact shadows sit under and slightly left of the sphere, with the
+    # near-camera end wider than the far end.
+    scx = cx - 0.10 * radius
+    scy = cy + 0.58 * radius
+    sx = max(2.3, radius * 0.92)
+    sy = max(1.0, radius * 0.24)
+    q = ((xx - scx) / sx) ** 2 + ((yy - scy) / sy) ** 2
+    shadow = np.exp(-2.5 * q) * 0.42
+    patch = frame[y0:y1, x0:x1].astype(np.float32)
+    tint = np.array([0.67, 0.64, 0.60], dtype=np.float32)
+    patch = patch * (1.0 - shadow[..., None]) + patch * tint * shadow[..., None]
+    frame[y0:y1, x0:x1] = np.clip(patch, 0, 255).astype(np.uint8)
+

+def render():

+    plate = make_clean_plate(source)

+
+    frames = []

+    # frame 1, so speed and stopping distance are reduced by 3x.

+        # scales the remaining displacement toward the far end of the floor.

+            progress = 1.0 - (1.0 - u) ** 2
+        else:
+            progress = 1.0
+        center = start + distance * progress

+        perspective = np.clip((center[1] - 40.0) / (start[1] - 40.0), 0.34, 1.0)
+        scale = 0.92 * perspective
+        ball = resize_rgba(sprite, scale)
+        radius = 14.0 * scale
+

+        # early motion; retaining it keeps the unaffected scene natural.
+        if i <= 19:
+            frame = (
+                0.82 * frame.astype(np.float32)
+                + 0.18 * source[min(i, 40)].astype(np.float32)
+            ).astype(np.uint8)
+

+        paste_rgba(frame, ball, center)
+        frames.append(frame)
+
+    return np.stack(frames).astype(np.uint8)
