+def source_ball_boxes(frames: np.ndarray) -> np.ndarray:
+    """Return the red ball's bounding boxes over the source timeline."""
+    boxes = []
+    for frame in frames.astype(np.int16):
+        red = (
+            (frame[:, :, 0] - frame[:, :, 2] > 50)
+            & (frame[:, :, 1] < frame[:, :, 0] * 0.65)
+            & (frame[:, :, 0] > 40)
+        )
+        red[:, :105] = False
+        red[:, 185:] = False
+        red[110:] = False
+        ys, xs = np.where(red)
+        if not len(xs):
+            raise RuntimeError("Could not locate the red ball in a source frame")
+        boxes.append((xs.min(), ys.min(), xs.max(), ys.max()))
+    return np.asarray(boxes, dtype=np.float32)

+def estimate_background(frames: np.ndarray, centers: np.ndarray) -> np.ndarray:
+    """Recover the static scene while excluding the ball and contact shadow."""
+    height, width, _ = frames.shape[1:]
+    result = np.empty((height, width, 3), dtype=np.float32)
+    yy, xx = np.indices((height, width))
+
+    for y in range(height):
+        for x in range(width):
+            keep = np.ones(len(frames), dtype=bool)
+            for index, (cx, cy) in enumerate(centers):
+                keep[index] = (x - cx) ** 2 + (y - cy) ** 2 > 14.5**2
+                if cy > 55 and y > 85 and abs(x - cx) < 22:
+                    keep[index] = False
+            if keep.sum() < 8:
+                keep[:] = True
+            result[y, x] = np.median(frames[keep, y, x], axis=0)
+
+    return result

+def estimate_ball_template(
+    frame: np.ndarray, background: np.ndarray, box: np.ndarray
+) -> tuple[np.ndarray, np.ndarray, tuple[float, float]]:
+    """Recover the ball's foreground color and antialiased circular coverage."""
+    x0, y0, x1, y1 = box.astype(int)
+    pad = 3
+    x0, y0 = max(0, x0 - pad), max(0, y0 - pad)
+    x1, y1 = min(frame.shape[1], x1 + pad + 1), min(frame.shape[0], y1 + pad + 1)
+    patch = frame[y0:y1, x0:x1].astype(np.float32)
+    bg = background[y0:y1, x0:x1]
+
+    cx = (box[0] + box[2]) * 0.5 - x0
+    cy = (box[1] + box[3]) * 0.5 - y0
+    gy, gx = np.indices(patch.shape[:2])
+    radius = (box[2] - box[0] + box[3] - box[1]) * 0.25
+
+    best = None
+    for r in np.arange(radius - 0.8, radius + 0.9, 0.05):
+        distance = np.sqrt((gx - cx) ** 2 + (gy - cy) ** 2)
+        for width in (0.55, 0.8, 1.05, 1.3, 1.55):
+            alpha = np.clip((r - distance) / width + 0.5, 0.0, 1.0)
+            interior = alpha > 0.12
+            foreground = np.broadcast_to(np.array([139.0, 70.0, 48.0]), patch.shape)
+            recovered = bg + (patch - bg) / np.maximum(alpha[..., None], 0.12)
+            foreground = np.where(interior[..., None], recovered, foreground)
+            reconstructed = alpha[..., None] * foreground + (1 - alpha[..., None]) * bg
+            error = np.abs(reconstructed - patch)[interior].mean()

+    _, alpha, foreground, radius, width = best

+    foreground[rim] = 0.7 * foreground[rim] + 0.3 * core_color
+
+    center = (box[0] + box[2]) * 0.5, (box[1] + box[3]) * 0.5
+    return foreground, alpha.astype(np.float32), center

+def estimate_source_physics(centers: np.ndarray, boxes: np.ndarray):
+    """Fit the source launch and collision surface from the measured track."""
+    t = np.arange(9, dtype=np.float64)

+    radius = float(

+    after = np.diff(centers[14:21, 0])

+    )
+
+    return {

+        "launch_vx": float(launch_vx),
+        "launch_vy": float(launch_vy),
+        "radius": radius,
+        "floor_center": floor_center,
+        "floor_friction": floor_friction,
+        "floor_restitution": 0.80,
+        "wall_restitution": 0.78,

+def simulate_path(
+    start: tuple[float, float],
+    velocity: tuple[float, float],
+    physics: dict[str, float],
+    frame_count: int,
+    width: int,
+    height: int,
+) -> np.ndarray:
+    """Integrate a frictionless-flight, damped-collision ball path."""
+    substeps = 64
+    dt = 1.0 / substeps
+    radius = physics["radius"]
+    floor = physics["floor_center"]
+    gravity = physics["gravity"]
+    x_min, x_max = radius, width - radius
+    y_min = radius
+
+    x, y = start
+    vx, vy = velocity
+    path = np.empty((frame_count * substeps, 2), dtype=np.float64)
+    path[0] = (x, y)
+
+    for index in range(1, len(path)):
+        vy += gravity * dt
+        x += vx * dt
+        y += vy * dt
+
+        if y > floor:
+            y = 2.0 * floor - y
+            vy = -vy * physics["floor_restitution"]
+            vx *= physics["floor_friction"]
+        if y < y_min:
+            y = 2.0 * y_min - y
+            vy = -vy * physics["wall_restitution"]
+        if x > x_max:
+            x = 2.0 * x_max - x
+            vx = -vx * physics["wall_restitution"]
+        elif x < x_min:
+            x = 2.0 * x_min - x
+            vx = -vx * physics["wall_restitution"]
+
+        path[index] = (x, y)
+
+    return path[::substeps]
+

+def sample_patch(patch: np.ndarray, xs: np.ndarray, ys: np.ndarray) -> np.ndarray:
+    """Bilinearly sample a small sprite patch."""
+    h, w = patch.shape[:2]
+    xs = np.clip(xs, 0, w - 1.001)
+    ys = np.clip(ys, 0, h - 1.001)
+    ix = np.floor(xs).astype(np.int32)
+    iy = np.floor(ys).astype(np.int32)
+    fx = (xs - ix)[..., None]
+    fy = (ys - iy)[..., None]
+    a = patch[iy, ix]
+    b = patch[iy, ix + 1]
+    c = patch[iy + 1, ix]
+    d = patch[iy + 1, ix + 1]
+    if patch.ndim == 3:
+        return (
+            a * (1 - fx) * (1 - fy)
+            + b * fx * (1 - fy)
+            + c * (1 - fx) * fy
+            + d * fx * fy
+        )
+    return (
+        a * (1 - fx[..., 0]) * (1 - fy[..., 0])
+        + b * fx[..., 0] * (1 - fy[..., 0])
+        + c * (1 - fx[..., 0]) * fy[..., 0]
+        + d * fx[..., 0] * fy[..., 0]
+    )
+

+def render_frame(
+    background: np.ndarray,
+    foreground: np.ndarray,
+    alpha: np.ndarray,
+    sprite_center: tuple[float, float],
+    ball_center: tuple[float, float],
+    physics: dict[str, float],
+) -> np.ndarray:
+    """Render a shadow and the extracted ball over the clean plate."""
+    height, width, _ = background.shape
+    yy, xx = np.indices((height, width), dtype=np.float32)
+    output = background.copy()
+
+    bx, by = ball_center
+    floor = physics["floor_center"]
+    height_above_floor = max(0.0, floor - by)
+    shadow_strength = 0.58 * np.clip((28.0 - height_above_floor) / 25.0, 0.0, 1.0) ** 2
+    if shadow_strength > 0.004:
+        sx = 10.5 + 0.10 * height_above_floor
+        sy = 3.6 + 0.035 * height_above_floor
+        shadow = np.exp(
+            -0.5
+            * (((xx - (bx + 1.5)) / sx) ** 2 + ((yy - (floor + 12.0)) / sy) ** 2)
+        )
+        shadow *= shadow_strength
+        tint = np.array([0.52, 0.55, 0.58], dtype=np.float32)
+        output *= 1.0 - shadow[..., None] * tint
+
+    px, py = sprite_center
+    sprite_h, sprite_w = alpha.shape
+    x0 = max(0, int(np.floor(bx - (px - 0) - 2)))
+    y0 = max(0, int(np.floor(by - (py - 0) - 2)))
+    x1 = min(width, int(np.ceil(bx + (sprite_w - px) + 2)))
+    y1 = min(height, int(np.ceil(by + (sprite_h - py) + 2)))
+
+    dst_y, dst_x = np.indices((y1 - y0, x1 - x0), dtype=np.float32)
+    dst_x += x0
+    dst_y += y0
+    sprite_x = dst_x - bx + px
+    sprite_y = dst_y - by + py
+    coverage = sample_patch(alpha, sprite_x, sprite_y)
+    color = sample_patch(foreground, sprite_x, sprite_y)
+
+    visible = coverage > 0.002
+    patch = output[y0:y1, x0:x1]
+    patch[visible] = (
+        coverage[visible, None] * color[visible]
+        + (1.0 - coverage[visible, None]) * patch[visible]
+    )
+    output[y0:y1, x0:x1] = patch
+    return np.clip(output + 0.5, 0, 255).astype(np.uint8)
+

+def prepare_output_dir(out_dir: Path) -> None:
+    out_dir.mkdir(parents=True, exist_ok=True)
+    for name in ("task_spec.json", "prompt.txt"):
+        if not (out_dir / name).exists():
+            shutil.copy2(ROOT / name, out_dir / name)
+

+def main() -> None:
+    parser = argparse.ArgumentParser()
+    parser.add_argument("--out", type=Path, default=ROOT)
+    args = parser.parse_args()
+
+    source = np.load(ROOT / "source_pixels.npz")["rgb"]
+    boxes = source_ball_boxes(source)
+    centers = np.stack(
+        ((boxes[:, 0] + boxes[:, 2]) * 0.5, (boxes[:, 1] + boxes[:, 3]) * 0.5),
+        axis=1,
+    ).astype(np.float64)
+    background = estimate_background(source, centers)
+    foreground, alpha, sprite_center = estimate_ball_template(
+        source[0], background, boxes[0]
+    )
+    physics = estimate_source_physics(centers, boxes)
+
+    launch = (
+        SPEED_FACTOR * physics["launch_vx"],
+        SPEED_FACTOR * physics["launch_vy"],
+    )
+    path = simulate_path(
+        tuple(centers[0]),
+        launch,
+        physics,
+        frame_count=len(source),
+        width=source.shape[2],
+        height=source.shape[1],
+    )
+
+    frames = np.empty_like(source)
+    for index, ball_center in enumerate(path):
+        frames[index] = render_frame(
+            background,
+            foreground,
+            alpha,
+            sprite_center,
+            tuple(ball_center),
+            physics,
+        )
+
+    prepare_output_dir(args.out)
+    artifact_io.save(frames, args.out)
+    print(
+        "saved",
+        args.out,
+        "launch",
+        tuple(round(value, 3) for value in launch),
+        "physics",
+        {key: round(value, 3) for key, value in physics.items()},
+    )
+
+
+if __name__ == "__main__":
+    main()
