+"""Remove the moving milk carton while preserving the source timeline.
+
+The source is a short camera move over a tabletop.  The carton crosses the
+foreground during the first 1.4 seconds and then leaves the frame.  We use the
+clean, settled part of the clip as a plate for the occluded tabletop, and a
+small motion-tracked mask for the carton itself.  Only frame 0 is left factual,
+as required by the intervention timestamp.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+N, H, W = 96, 144, 256
+
+
+def smooth_plate(source):
+    """Build a temporally stable plate from the settled part of the clip."""
+    # The tabletop is static after the carton has left.  A trimmed mean avoids
+    # retaining the handful of tiny encoder fluctuations in that interval.
+    tail = source[45:88].astype(np.float32)
+    plate = np.median(tail, axis=0)
+    plate = cv2.GaussianBlur(plate, (0, 0), 0.42)
+    return np.clip(plate, 0, 255).astype(np.uint8)
+
+
+def moving_mask(source, plate, index):
+    """Return a conservative mask of the carton and its contact shadow.
+
+    A broad trajectory envelope is intersected with a color/edge test.  This
+    catches the pale lower panel and the dark contact edge without selecting
+    the stationary white and red objects behind the carton.
+    """
+    frame = source[index]
+    diff = np.max(np.abs(frame.astype(np.int16) - plate.astype(np.int16)), axis=2)
+    yy, xx = np.indices((H, W))
+
+    # The carton enters at the left and exits at the right.  The envelope is
+    # intentionally a little wider than the silhouette; the color test below
+    # trims it back to the actual object.
+    left = np.interp(index, [0, 4, 8, 12, 16, 20, 24, 28, 32, 36],
+                     [-1, 76, 113, 143, 160, 174, 185, 192, 195, 196])
+    right = np.interp(index, [0, 4, 8, 12, 16, 20, 24, 28, 32, 36],
+                      [38, 105, 143, 174, 197, 221, 239, 249, 256, 256])
+    env = (xx >= left) & (xx <= right) & (yy >= 61) & (yy <= 84)
+
+    # Pale carton body, red printed face, and the dark shadow/contact edge.
+    rgb = frame.astype(np.int16)
+    red = (rgb[:, :, 0] - rgb[:, :, 1] > 13) & (rgb[:, :, 0] - rgb[:, :, 2] > 10)
+    dark = np.max(frame, axis=2) < 170
+    changed = diff > 5
+    mask = env & (red | dark | changed)
+
+    # Remove isolated compression speckles, then bridge the narrow internal
+    # gaps of the printed carton face.  Dilation is kept small because the
+    # neighboring tabletop texture should remain visible.
+    mask = cv2.morphologyEx(mask.astype(np.uint8), cv2.MORPH_OPEN,
+                            np.ones((2, 2), np.uint8))
+    mask = cv2.morphologyEx(mask, cv2.MORPH_CLOSE,
+                            np.ones((5, 5), np.uint8))
+    mask = cv2.dilate(mask, np.ones((3, 3), np.uint8), iterations=1)
+    mask = cv2.GaussianBlur(mask.astype(np.float32), (0, 0), 1.05)
+    return np.clip(mask, 0, 1)
+
+
+def inpaint_region(frame, plate, alpha):
+    """Composite the clean plate with a feathered alpha and preserve noise."""
+    alpha3 = alpha[:, :, None]
+    result = frame.astype(np.float32) * (1.0 - alpha3)
+    result += plate.astype(np.float32) * alpha3
+
+    # The source plate is already in the same lighting and camera position.
+    # Add a tiny amount of source-frame grain only in the fill, preventing the
+    # cleaned area from looking like a flat rectangle.
+    yy, xx = np.indices((H, W))
+    grain = (
+        np.sin(xx * 1.71 + yy * 0.37) * 0.42
+        + np.cos(xx * 0.29 - yy * 1.33) * 0.31
+    )[:, :, None]
+    result += alpha3 * grain
+    return np.clip(result, 0, 255).astype(np.uint8)
+
+
+def main():
+    with np.load(CASE_DIR / "source_pixels.npz") as data:
+        source = data["rgb"].copy()
+
+    plate = smooth_plate(source)
+    output = source.copy()
+
+    # Frame 0 is the factual history.  The intervention begins at frame 1.
+    for index in range(1, N):
+        alpha = moving_mask(source, plate, index)
+        if alpha.max() > 0:
+            output[index] = inpaint_region(source[index], plate, alpha)
+
+    artifact_io.save(output, CASE_DIR)
+
+
+if __name__ == "__main__":
+    main()
