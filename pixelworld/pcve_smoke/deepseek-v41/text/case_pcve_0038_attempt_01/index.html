+def _mode_over(frames):
+    """Return the most common color at every pixel over a short stable interval."""
+    flat = frames.astype(np.uint8).reshape(len(frames), -1, 3)
+    packed = (
+        (flat[:, :, 0].astype(np.uint32) << 16)
+        | (flat[:, :, 1].astype(np.uint32) << 8)
+        | flat[:, :, 2]
+    )
+    result = np.empty(packed.shape[1], dtype=np.uint32)
+    for index in range(packed.shape[1]):
+        values, counts = np.unique(packed[:, index], return_counts=True)
+        result[index] = values[np.argmax(counts)]
+    return np.stack(
+        [(result >> 16) & 255, (result >> 8) & 255, result & 255], axis=1
+    ).reshape(frames.shape[1:]).astype(np.uint8)
+
+
+def _clean_plate(source):
+    """Recover the static scene by removing both parked positions of the car."""
+    late = _mode_over(source[24:]).astype(np.float32)
+    early = _mode_over(source[8:24]).astype(np.float32)
+    plate = late.copy()
+    plate[34:66, 174:212] = early[34:66, 174:212]
+    return plate

+def _largest_component(mask):
+    labels, count = ndimage.label(mask, structure=np.ones((3, 3)))
+    if count == 0:
+        return mask
+    sizes = ndimage.sum(mask, labels, range(1, count + 1))
+    return labels == (int(np.argmax(sizes)) + 1)
+

+def _extract_sprite(source, plate):
+    """Extract the car and its contact shadow from the first source frame."""
+    x0, x1, y0, y1 = 50, 91, 22, 65
+    patch = source[0, y0:y1, x0:x1].astype(np.float32)
+    background = plate[y0:y1, x0:x1]
+    difference = np.abs(patch - background).max(axis=2)
+
+    car = _largest_component(difference > 9)
+    car = ndimage.binary_closing(car, iterations=1)
+    car = ndimage.binary_dilation(car, iterations=1)
+
+    shadow = np.zeros_like(car)
+    shadow[32:44, 3:22] = difference[32:44, 3:22] > 6
+    shadow = ndimage.binary_closing(shadow, iterations=1)
+    shadow = ndimage.binary_dilation(shadow, iterations=1)
+    mask = car | shadow
+
+    alpha = np.clip((difference - 2.0) / 14.0, 0.0, 1.0)
+    alpha[~mask] = 0.0
+    alpha = np.asarray(
+        Image.fromarray((alpha * 255).astype(np.uint8)).filter(
+            ImageFilter.GaussianBlur(0.45)
+        ),
+        dtype=np.float32,
+    ) / 255.0
+    alpha[alpha < 0.035] = 0.0
+
+    ys, xs = np.where(alpha > 0.02)
+    bounds = (xs.min(), ys.min(), xs.max() + 1, ys.max() + 1)
+    sx0, sy0, sx1, sy1 = bounds
+    return {
+        "rgb": patch[sy0:sy1, sx0:sx1],
+        "alpha": alpha[sy0:sy1, sx0:sx1],
+        "x": x0 + sx0,
+        "y": y0 + sy0,
+    }

+def _source_centers(source, plate):
+    """Measure the car center in each factual frame from its red body."""
+    centers = []
+    for index in range(24):
+        frame = source[index].astype(np.float32)
+        difference = np.abs(frame - plate).max(axis=2)
+        mask = difference > 15
+        yy, xx = np.indices(mask.shape)
+        mask &= (yy >= 20) & (yy <= 56) & (xx >= 45) & (xx <= 215)
+        labels, count = ndimage.label(mask, structure=np.ones((3, 3)))
+        components = []
+        for label in range(1, count + 1):
+            ys, xs = np.where(labels == label)
+            if len(xs) >= 25 and xs.max() - xs.min() >= 10:
+                components.append((len(xs), xs.mean(), ys.mean()))
+        if components:
+            _, cx, cy = max(components)
+            centers.append((float(cx), float(cy)))
+        else:
+            centers.append(centers[-1])
+    return np.asarray(centers)
+

+def _interpolate_centers(source_centers, times):
+    result = np.empty((len(times), 2), dtype=np.float32)
+    source_times = np.arange(len(source_centers), dtype=np.float32)
+    for axis in range(2):
+        result[:, axis] = np.interp(
+            times,
+            source_times,
+            source_centers[:, axis],
+            left=source_centers[0, axis],
+            right=source_centers[-1, axis],
+        )
+    return result
+

+def _render(source, plate, sprite):
+    centers = _source_centers(source, plate)
+    start = centers[0]
+
+    # The source advances from 0 to 1 in normalized motion units over 23 frames.
+    # Doubling its initial speed reaches unit motion after 11.5 frames. The
+    # remaining normalized time is then integrated through the same measured
+    # deceleration curve, allowing the car to keep moving rather than stop.
+    frame = np.arange(FRAME_COUNT, dtype=np.float32)
+    t0 = 11.5
+    normalized = np.where(
+        frame <= t0,
+        2.0 * frame / 23.0,
+        (frame - t0) / 23.0,
+    )
+    # Reuse the source's cumulative displacement curve for the continuation.
+    normalized = np.clip(normalized, 0.0, 2.25)
+    source_curve = centers - start
+    source_motion = np.r_[0.0, np.cumsum(np.linalg.norm(np.diff(centers, axis=0), axis=1))]
+    source_motion /= source_motion[-1]
+    total_curve = np.interp(normalized, source_motion, source_curve[:, 0])
+    total_curve_y = np.interp(normalized, source_motion, source_curve[:, 1])
+
+    # Continue with residual forward velocity along the measured terminal direction.
+    terminal_velocity = (centers[-1] - centers[-4]) / 3.0
+    direction = terminal_velocity / max(np.linalg.norm(terminal_velocity), 1e-6)
+    continuation = np.maximum(normalized - 1.0, 0.0) * 7.5
+    target = np.stack(
+        [total_curve + direction[0] * continuation,
+         total_curve_y + direction[1] * continuation],
+        axis=1,
+    )
+
+    frames = []
+    sprite_rgb = sprite["rgb"]
+    sprite_alpha = sprite["alpha"][..., None]
+    h, w = sprite_rgb.shape[:2]
+    for index in range(FRAME_COUNT):
+        canvas = plate.copy()
+        cx, cy = start + target[index]
+        left = int(round(cx - (sprite["x"] + w / 2.0 - start[0])))
+        top = int(round(cy - (sprite["y"] + h / 2.0 - start[1])))
+        dx0, dy0 = max(0, left), max(0, top)
+        dx1, dy1 = min(256, left + w), min(144, top + h)
+        if dx0 < dx1 and dy0 < dy1:
+            sx0, sy0 = dx0 - left, dy0 - top
+            sx1, sy1 = sx0 + dx1 - dx0, sy0 + dy1 - dy0
+            a = sprite_alpha[sy0:sy1, sx0:sx1]
+            roi = canvas[dy0:dy1, dx0:dx1]
+            canvas[dy0:dy1, dx0:dx1] = sprite_rgb[sy0:sy1, sx0:sx1] * a + roi * (1 - a)
+        frames.append(np.clip(canvas, 0, 255).astype(np.uint8))
+    return np.asarray(frames)
+
+
+def main():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    plate = _clean_plate(source)
+    sprite = _extract_sprite(source, plate)
+    frames = _render(source, plate, sprite)
+    artifact_io.save(frames, CASE_DIR)
+    print(f"saved {len(frames)} frames; sprite={sprite['rgb'].shape}")
+
+
+if __name__ == "__main__":
+    main()
