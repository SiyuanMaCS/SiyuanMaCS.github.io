+def smoothstep(edge0, edge1, values):
+    t = np.clip((values - edge0) / (edge1 - edge0), 0.0, 1.0)
+    return t * t * (3.0 - 2.0 * t)
+
+
+def reconstruct_table(frame):
+    """Rebuild the unobstructed tabletop where the lead domino and its shadow sat."""
+    result = frame.copy()
+    height, width = frame.shape[:2]
+
+    # The table is smooth across the cleared region. Sampling several rows from
+    # each clean side gives stable edge colors without borrowing another object.
+    left_anchor = frame[30:129, 50:56].mean(axis=1)
+    right_anchor = frame[30:129, 98:102].mean(axis=1)
+    xs = np.arange(55, 102, dtype=np.float32)
+    blend = smoothstep(55.0, 101.0, xs)
+
+    plate = (
+        left_anchor[:, None, :] * (1.0 - blend[None, :, None])
+        + right_anchor[:, None, :] * blend[None, :, None]
+    )
+
+    # A low-amplitude, deterministic texture keeps the replacement from looking
+    # like a flat polygon when viewed at full resolution.
+    yy, xx = np.mgrid[30:129, 55:102]
+    texture = (
+        0.55 * np.sin(0.37 * xx + 0.19 * yy)
+        + 0.35 * np.sin(0.83 * yy - 0.11 * xx + 0.7)
+        + 0.22 * np.cos(1.17 * xx + 0.43 * yy)
+    )
+    plate += texture[:, :, None]
+
+    x0, x1, y0, y1 = 52, 103, 28, 133
+    alpha_x = smoothstep(52.0, 59.0, np.arange(x0, x1, dtype=np.float32))
+    alpha_x *= 1.0 - smoothstep(98.0, 103.0, np.arange(x0, x1, dtype=np.float32))
+    alpha_y = smoothstep(28.0, 36.0, np.arange(y0, y1, dtype=np.float32))
+    alpha_y *= 1.0 - smoothstep(122.0, 133.0, np.arange(y0, y1, dtype=np.float32))
+    alpha = alpha_y[:, None] * alpha_x[None, :]
+
+    target = result[y0:y1, x0:x1]
+    plate_padded = np.empty_like(target)
+    plate_padded[2:101, 3:50] = plate
+    plate_padded[:2] = plate_padded[2]
+    plate_padded[101:] = plate_padded[100]
+    plate_padded[:, :3] = plate_padded[:, 3:4]
+    plate_padded[:, 50:] = plate_padded[:, 49:50]
+    result[y0:y1, x0:x1] = (
+        target * (1.0 - alpha[:, :, None]) + plate_padded * alpha[:, :, None]
+    )
+    return np.clip(np.rint(result), 0, 255).astype(np.uint8)
