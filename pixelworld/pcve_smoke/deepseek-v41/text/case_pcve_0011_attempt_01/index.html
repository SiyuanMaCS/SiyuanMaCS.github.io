+def feathered_copy(dst, src, box, feather=3):
+    """Copy a rectangular patch with a soft border."""
+    x0, y0, x1, y1 = box
+    patch = src[y0:y1, x0:x1].astype(np.float32)
+    h, w = patch.shape[:2]
+    yy, xx = np.mgrid[:h, :w]
+    edge = np.minimum.reduce(
+        [xx + 0.5, yy + 0.5, w - 0.5 - xx, h - 0.5 - yy]
+    )
+    alpha = np.clip(edge / max(float(feather), 1.0), 0.0, 1.0)[..., None]
+    dst[y0:y1, x0:x1] = (
+        patch * alpha + dst[y0:y1, x0:x1].astype(np.float32) * (1.0 - alpha)
+    )

+def paste_sprite(frame, sprite, alpha, center, scale=1.0, opacity=1.0):
+    patch, local_center = sprite
+    h, w = patch.shape[:2]
+    sx, sy = local_center
+    matrix = np.array(
+        [[scale, 0.0, center[0] - scale * sx],
+         [0.0, scale, center[1] - scale * sy]],
+        dtype=np.float32,
+    )
+    warped = cv2.warpAffine(
+        patch.astype(np.float32),
+        matrix,
+        (frame.shape[1], frame.shape[0]),
+        flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_REFLECT,
+    )
+    warped_alpha = cv2.warpAffine(
+        alpha,
+        matrix,
+        (frame.shape[1], frame.shape[0]),
+        flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=0,
+    )
+    warped_alpha = np.clip(warped_alpha * opacity, 0.0, 1.0)[..., None]
+    return (
+        warped * warped_alpha
+        + frame.astype(np.float32) * (1.0 - warped_alpha)
+    )

+def add_ball_shadow(frame, center, radius):

+        0,
+        360,
+        0.22,
+        -1,
+        cv2.LINE_AA,
+    )
+    shadow = cv2.GaussianBlur(shadow, (0, 0), 1.5)[..., None]
+    return frame.astype(np.float32) * (1.0 - shadow)

+    body_scale = 1.0 + 0.00070 * (body_x - 136.5)

+    body_patch, body_alpha, body_center = extract_body(source)
+    ball_sprite = (ball_patch, ball_center)

+            scale=float(ts[index]),
+        )

+            frame, (float(bx[index]), float(by[index])), float(br[index])
+        )
+        frame = paste_sprite(
+            frame,
+            ball_sprite,
+            ball_alpha,
+            (float(bx[index]), float(by[index])),
+            scale=float(br[index] / ball_base_radius),
+        )
+        output[index] = np.clip(frame + 0.5, 0, 255).astype(np.uint8)
+
+    output[0] = source[0]
