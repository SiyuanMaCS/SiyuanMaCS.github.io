+def largest_red_component(frame: np.ndarray):
+    """Return the moving red ball's bounding box, or None if absent."""
+    rgb = frame.astype(np.int16)
+    red, green, blue = rgb[..., 0], rgb[..., 1], rgb[..., 2]
+    yy, xx = np.indices((HEIGHT, WIDTH))
+    mask = (
+        (red - green > 40)
+        & (red - blue > 40)
+        & (red > 90)
+        & (xx > 105)
+        & (xx < 185)
+        & (yy < 93)
+    )
+    labels, count = ndimage.label(mask)
+    best = None
+    for label in range(1, count + 1):
+        ys, xs = np.where(labels == label)
+        if len(xs) > 20 and (best is None or len(xs) > best[0]):
+            best = (len(xs), int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max()))
+    if best is None:
+        return None
+    _, x0, y0, x1, y1 = best
+    return x0, y0, x1, y1

+def source_ball_centers(frames: np.ndarray):
+    centers = []
+    for frame in frames:
+        box = largest_red_component(frame)
+        if box is None:
+            centers.append(np.nan)
+        else:
+            x0, y0, x1, y1 = box
+            centers.append(((x0 + x1) * 0.5, (y0 + y1) * 0.5))
+    return np.asarray(centers, dtype=np.float64)

+def build_clean_background(frames: np.ndarray):
+    """Use a high temporal percentile where the ball/shadow contaminates the median."""
+    median = np.median(frames, axis=0)
+    upper = np.percentile(frames, 70, axis=0)
+    contaminated = np.max(np.abs(upper - median), axis=2) > 2.0
+
+    # Include every pixel touched by the moving ball and its soft ground shadow.
+    dynamic = np.zeros((HEIGHT, WIDTH), dtype=bool)
+    for frame in frames:
+        box = largest_red_component(frame)
+        if box is None:
+            continue
+        x0, y0, x1, y1 = box
+        cx, cy = (x0 + x1) // 2, (y0 + y1) // 2
+        yy, xx = np.indices((HEIGHT, WIDTH))
+        dynamic |= ((xx - cx) / 20.0) ** 2 + ((yy - (cy + 8)) / 24.0) ** 2 < 1.0
+    dynamic = ndimage.binary_dilation(dynamic, iterations=2)
+
+    return np.where((contaminated | dynamic)[..., None], upper, median).astype(np.float64)

+def extract_ball_sprite(frame: np.ndarray):
+    """Extract the opaque, antialiased source ball from a clean airborne frame."""
+    center_x, center_y = 141.0, 46.0
+    radius = 12.4
+    reach = 13
+    rgba = np.zeros((2 * reach + 1, 2 * reach + 1, 4), dtype=np.float64)
+    for iy, dy in enumerate(range(-reach, reach + 1)):
+        for ix, dx in enumerate(range(-reach, reach + 1)):
+            x = int(round(center_x + dx))
+            y = int(round(center_y + dy))
+            distance = np.hypot(x - center_x, y - center_y)
+            alpha = np.clip((radius + 0.55 - distance) / 1.10, 0.0, 1.0)
+            if alpha > 0:
+                rgba[iy, ix, :3] = frame[y, x].astype(np.float64)
+                rgba[iy, ix, 3] = alpha
+    return rgba, reach

+def extract_shadow_map(source_frame: np.ndarray, background: np.ndarray):
+    """Measure the source contact shadow relative to its clean background."""
+    ball_x, ball_y = 158.0, 80.0
+    x_lo, x_hi = 128, 189
+    y_lo, y_hi = 87, 119
+    strength = np.zeros((y_hi - y_lo, x_hi - x_lo), dtype=np.float64)
+    for y in range(y_lo, y_hi):
+        for x in range(x_lo, x_hi):
+            if np.hypot(x - ball_x, y - ball_y) <= 13.5:
+                continue
+            ratio = np.mean(source_frame[y, x] / np.maximum(background[y, x], 1.0))
+            if y >= 90:
+                strength[y - y_lo, x - x_lo] = np.clip(1.0 - ratio, 0.0, 0.52)
+    strength = ndimage.gaussian_filter(strength, sigma=1.15)
+    return strength, x_lo - ball_x, y_lo - ball_y

+def bilinear_sample(image: np.ndarray, yy: np.ndarray, xx: np.ndarray):
+    h, w = image.shape[:2]
+    yy = np.clip(yy, 0, h - 1)
+    xx = np.clip(xx, 0, w - 1)
+    y0 = np.floor(yy).astype(int)
+    x0 = np.floor(xx).astype(int)
+    y1 = np.minimum(y0 + 1, h - 1)
+    x1 = np.minimum(x0 + 1, w - 1)
+    wy = yy - y0
+    wx = xx - x0
+    top = image[y0, x0] * (1.0 - wx)[..., None] + image[y0, x1] * wx[..., None]
+    bottom = image[y1, x0] * (1.0 - wx)[..., None] + image[y1, x1] * wx[..., None]
+    return top * (1.0 - wy)[..., None] + bottom * wy[..., None]
+
+
+def composite_shadow(canvas, shadow, offsets, cx, cy):
+    map_x = int(np.floor(cx + offsets[0]))
+    map_y = int(np.floor(cy + offsets[1]))
+    h, w = shadow.shape
+    x0, y0 = max(0, map_x), max(0, map_y)
+    x1, y1 = min(WIDTH, map_x + w), min(HEIGHT, map_y + h)
+    if x0 >= x1 or y0 >= y1:
+        return
+    sx0, sy0 = x0 - map_x, y0 - map_y
+    field = shadow[sy0 : sy0 + (y1 - y0), sx0 : sx0 + (x1 - x0), None]
+    canvas[y0:y1, x0:x1] *= 1.0 - 0.90 * field
+
+
+def composite_ball(canvas, sprite, reach, cx, cy):
+    radius = reach + 2
+    x0, x1 = max(0, int(np.floor(cx - radius))), min(WIDTH, int(np.ceil(cx + radius)) + 1)
+    y0, y1 = max(0, int(np.floor(cy - radius))), min(HEIGHT, int(np.ceil(cy + radius)) + 1)
+    yy, xx = np.mgrid[y0:y1, x0:x1]
+    sprite_y = yy - (cy - reach)
+    sprite_x = xx - (cx - reach)
+    values = bilinear_sample(sprite, sprite_y, sprite_x)
+    alpha = np.clip(values[..., 3], 0.0, 1.0)[..., None]
+    canvas[y0:y1, x0:x1] = canvas[y0:y1, x0:x1] * (1.0 - alpha) + values[..., :3] * alpha
+
+
+def low_restitution_centers(source_centers: np.ndarray):
+    """Continue the factual horizontal motion with a 0.1x restitution impact."""
+    post = source_centers[33:, 0]
+    valid = np.isfinite(post)
+    coeff = np.polyfit(np.arange(33, FRAME_COUNT)[valid], post[valid], 1)
+
+    # Source contact geometry: the sphere radius is about 11.5 px and the
+    # contact line is y ~= 89.15 + 0.018x.
+    ground_center = lambda x: 89.15 + 0.018 * x - 11.5
+    contact_y = ground_center(coeff[1] + coeff[0] * 32.0)
+    incoming_speed = 7.05
+    outgoing_speed = RESTITUTION_SCALE * 0.65 * incoming_speed
+    impact_time = 32.12
+
+    centers = source_centers.copy()
+    for index in range(33, FRAME_COUNT):
+        x = coeff[1] + coeff[0] * index
+        dt = index - impact_time
+        if dt <= 0:
+            y = contact_y
+        else:
+            bounce_height = outgoing_speed * dt - 0.5 * GRAVITY * dt * dt
+            if bounce_height < 0:
+                bounce_height = 0.0
+            y = ground_center(x) - bounce_height
+        centers[index] = (x, y)
+    return centers
+
+
+def main():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"].astype(np.float64)
+    assert source.shape == (FRAME_COUNT, HEIGHT, WIDTH, 3)
+
+    source_centers = source_ball_centers(source)
+    background = build_clean_background(source)
+    sprite, reach = extract_ball_sprite(source[20])
+    shadow, shadow_dx, shadow_dy = extract_shadow_map(source[95], background)
+    edited_centers = low_restitution_centers(source_centers)
+
+    output = np.empty((FRAME_COUNT, HEIGHT, WIDTH, 3), dtype=np.uint8)
+    output[: LAST_FACTUAL_INDEX + 1] = source[: LAST_FACTUAL_INDEX + 1].astype(np.uint8)
+
+    for index in range(LAST_FACTUAL_INDEX + 1, FRAME_COUNT):
+        canvas = background.copy()

+        composite_shadow(canvas, shadow, (shadow_dx, shadow_dy), cx, cy)
+        composite_ball(canvas, sprite, reach, cx, cy)
+        output[index] = np.clip(np.rint(canvas), 0, 255).astype(np.uint8)
+
+    artifact_io.save(output, CASE_DIR)
+

+    main()
