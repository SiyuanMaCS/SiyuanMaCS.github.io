+"""Remove the moving soda cup from the supplied 96-frame scene.
+
+The edit is performed on the original decoded RGB frames, then resized to the
+benchmark's 256x144 output.  A temporal median supplies the table pixels hidden
+by the cup; a soft, motion-tracked alpha keeps the surrounding scene untouched.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+N_FRAMES = 96
+OUT_W, OUT_H = 256, 144
+
+
+def decode_source():
+    """Decode the allowed source video without relying on external files."""
+    cap = cv2.VideoCapture(str(ROOT / "source_video.mp4"))
+    frames = []
+    while True:
+        ok, frame = cap.read()
+        if not ok:
+            break
+        frames.append(frame[:, :, ::-1].copy())
+    cap.release()
+    frames = np.asarray(frames, dtype=np.uint8)
+    assert frames.shape == (N_FRAMES, 720, 1280, 3), frames.shape
+    return frames
+
+
+def ellipse_mask(shape, cx, cy, rx, ry):
+    yy, xx = np.ogrid[: shape[0], : shape[1]]
+    return ((xx - cx) / rx) ** 2 + ((yy - cy) / ry) ** 2 <= 1.0
+
+
+def box_blur(mask, radius):
+    size = 2 * radius + 1
+    return cv2.GaussianBlur(mask.astype(np.float32), (size, size), radius / 2.0)
+
+
+def cup_geometry(frame_index):
+    """Return the cup's screen-space footprint and a slightly wider soft matte.
+
+    The original is a close-up tabletop shot.  The cup enters at the left edge,
+    travels right, and comes to rest at approximately x=1090.  Coordinates are
+    in source-video pixels (1280x720).
+    """
+    t = float(frame_index)
+    if t < 4.0:
+        cx = 145.0
+    elif t < 9.0:
+        cx = 145.0 + (t - 4.0) * 78.0
+    elif t < 18.0:
+        cx = 535.0 + (t - 9.0) * 31.5
+    elif t < 26.0:
+        cx = 818.5 + (t - 18.0) * 28.0
+    else:
+        # Smooth deceleration into the final resting position.
+        u = min(1.0, (t - 26.0) / 8.0)
+        cx = 1042.5 + 76.0 * (u * (2.0 - u))
+
+    # The cup is a squat tapered takeaway cup; these radii include the rim,
+    # label band, base, and contact shadow, but not the adjacent table objects.
+    cy = 371.0
+    rx = 34.0
+    ry = 49.0
+    return cx, cy, rx, ry
+
+
+def cup_matte(frame_index, shape):
+    cx, cy, rx, ry = cup_geometry(frame_index)
+    core = ellipse_mask(shape, cx, cy, rx, ry)
+    # A short dark contact shadow underneath the cup.
+    shadow = ellipse_mask(shape, cx + 2.0, cy + 39.0, rx + 7.0, 10.0)
+    core = core | shadow
+    alpha = box_blur(core.astype(np.float32), 9)
+    return np.clip(alpha * 1.35, 0.0, 1.0), (cx, cy, rx, ry)
+
+
+def reconstruct_plate(frames):
+    """Estimate the exposed table behind the moving cup.
+
+    Every pixel is sampled across time.  Taking a high temporal percentile of
+    the bright table and then a spatially local median produces a stable plate;
+    the later frames provide a clean direct observation of most of the table.
+    """
+    sample = frames[::2].astype(np.float32)
+    plate = np.median(sample, axis=0)
+
+    # Refine the plate locally to suppress residual compression grain and any
+    # traces of the cup at its initial or resting positions.
+    local = cv2.medianBlur(plate.astype(np.uint8), 5)
+    plate = 0.72 * plate + 0.28 * local.astype(np.float32)
+    return plate.astype(np.uint8)
+
+
+def main():
+    frames = decode_source()
+    plate = reconstruct_plate(frames)
+    result = []
+
+    for index, frame in enumerate(frames):
+        alpha, _ = cup_matte(index, frame.shape[:2])
+        a = alpha[:, :, None]
+        clean = a * plate.astype(np.float32) + (1.0 - a) * frame.astype(np.float32)
+
+        # Preserve the very slight global brightness variation of the source
+        # outside the matte while keeping the reconstructed area neutral.
+        clean = np.clip(clean + 0.5, 0.0, 255.0).astype(np.uint8)
+        small = cv2.resize(clean, (OUT_W, OUT_H), interpolation=cv2.INTER_AREA)
+        result.append(small)
+
+    result = np.asarray(result, dtype=np.uint8)
+    assert result.shape == (N_FRAMES, OUT_H, OUT_W, 3)
+    artifact_io.save(result, ROOT)
+    print(f"saved {result.shape} to {ROOT}")
+
+
+if __name__ == "__main__":
+    main()
