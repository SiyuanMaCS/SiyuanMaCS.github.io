+def ellipse_mask(shape, center, axes, feather=1.25):
+    mask = np.zeros(shape[:2], dtype=np.uint8)
+    cv2.ellipse(
+        mask,
+        (int(round(center[0])), int(round(center[1]))),
+        (int(round(axes[0])), int(round(axes[1]))),
+        0,
+        0,
+        360,
+        255,
+        -1,
+    )
+    alpha = mask.astype(np.float32) / 255.0
+    if feather > 0:
+        alpha = cv2.GaussianBlur(alpha, (0, 0), feather)
+    return alpha

+def pink_components(frame):
+    lab = cv2.cvtColor(frame, cv2.COLOR_RGB2LAB).astype(np.int16)
+    lightness, a_channel, b_channel = lab[..., 0], lab[..., 1], lab[..., 2]
+    mask = (
+        (a_channel > 128)
+        & (a_channel > b_channel)
+        & (lightness > 125)
+    ).astype(np.uint8)
+
+    count, labels, stats, centroids = cv2.connectedComponentsWithStats(mask, 8)
+    components = []
+    for index in range(1, count):
+        x, y, width, height, area = stats[index]
+        if area >= 80:
+            components.append(
+                {
+                    "area": int(area),
+                    "x": int(x),
+                    "y": int(y),
+                    "w": int(width),
+                    "h": int(height),
+                    "centroid": centroids[index],
+                    "labels": labels,
+                    "index": index,
+                }
+            )
+    return components, mask

+def find_pink_ball(frame, expected_center):
+    components, _ = pink_components(frame)
+    candidates = []
+    for component in components:
+        cx, cy = component["centroid"]
+        distance = np.hypot(cx - expected_center[0], cy - expected_center[1])
+        if distance < 45 and component["w"] >= 18 and component["h"] >= 18:
+            candidates.append((distance, component))
+    if not candidates:
+        return None
+    component = min(candidates, key=lambda item: item[0])[1]
+    return component

+def static_feature_mask(shape):
+    mask = np.full(shape[:2], 255, dtype=np.uint8)
+    mask[48:144, 72:136] = 0
+    mask[42:96, 216:256] = 0
+    return mask

+def estimate_homographies(source):
+    height, width = source.shape[1:3]
+    orb = cv2.ORB_create(nfeatures=2200, scaleFactor=1.2, nlevels=8, edgeThreshold=8)
+    base_gray = cv2.cvtColor(source[0], cv2.COLOR_RGB2GRAY)
+    feature_mask = static_feature_mask(source.shape)
+    base_points, base_descriptors = orb.detectAndCompute(base_gray, feature_mask)
+    matcher = cv2.BFMatcher(cv2.NORM_HAMMING)
+
+    homographies = [np.eye(3, dtype=np.float64)]
+    previous = np.eye(3, dtype=np.float64)
+    for index in range(1, len(source)):
+        gray = cv2.cvtColor(source[index], cv2.COLOR_RGB2GRAY)
+        points, descriptors = orb.detectAndCompute(gray, feature_mask)
+        if descriptors is None or len(points) < 12:
+            homographies.append(previous)
+            continue
+        pairs = matcher.knnMatch(descriptors, base_descriptors, k=2)
+        good = [first for first, second in pairs if first.distance < 0.75 * second.distance]
+        if len(good) < 10:
+            homographies.append(previous)
+            continue
+        current_xy = np.float32([points[match.queryIdx].pt for match in good])
+        base_xy = np.float32([base_points[match.trainIdx].pt for match in good])
+        homography, inliers = cv2.findHomography(
+            current_xy, base_xy, cv2.RANSAC, 1.75
+        )
+        if (
+            homography is None
+            or inliers is None
+            or int(inliers.sum()) < 10
+            or not np.isfinite(homography).all()
+        ):
+            homographies.append(previous)
+            continue
+        homographies.append(homography)
+        previous = homography
+
+    # Repeated source frames should not introduce tiny registration jitter.
+    for index in range(1, len(source)):
+        if np.array_equal(source[index], source[index - 1]):
+            homographies[index] = homographies[index - 1].copy()
+    return homographies

+def build_clean_plate(frame):
+    mask = np.zeros(frame.shape[:2], dtype=np.uint8)
+    cv2.ellipse(mask, (101, 91), (17, 16), 0, 0, 360, 255, -1)
+    cv2.ellipse(mask, (238, 66), (15, 15), 0, 0, 360, 255, -1)
+    mask = cv2.dilate(mask, np.ones((3, 3), dtype=np.uint8), iterations=1)
+    return cv2.inpaint(frame, mask, 5, cv2.INPAINT_TELEA)

+def football_geometry(index):
+    progress = float(np.clip((index - INTERVENTION_INDEX) / 13.0, 0.0, 1.0))
+    center_x = 103.0 + 1.5 * progress
+    center_y = 103.0 + 9.5 * progress
+    radius_x = 16.5 + 2.5 * progress
+    radius_y = 15.5 + 2.5 * progress
+    return center_x, center_y, radius_x, radius_y

+def extract_pink_template(source):
+    frame = source[0]
+    component = find_pink_ball(frame, (238.0, 66.0))
+    if component is None:
+        return None
+    center = component["centroid"]
+    radius = max(component["w"], component["h"]) * 0.5 + 1.2
+    pad = int(np.ceil(radius + 3.0))
+    cx, cy = int(round(center[0])), int(round(center[1]))
+    x0, y0 = max(0, cx - pad), max(0, cy - pad)
+    x1, y1 = min(frame.shape[1], cx + pad + 1), min(frame.shape[0], cy + pad + 1)
+    patch = frame[y0:y1, x0:x1]
+    patch_center = (center[0] - x0, center[1] - y0)
+    return {
+        "patch": patch,
+        "center": patch_center,
+        "radius": radius,
+        "mask": ellipse_mask(patch.shape, patch_center, (radius, radius), 0.9),
+    }

+def restore_pink_ball(frame, index, source_frame, template):
+    component = find_pink_ball(frame, (107.0, 84.0))
+    if component is None:
+        return frame
+
+    x, y, width = component["x"], component["y"], component["w"]
+    radius = max(11.5, width * 0.5 + 0.5)
+    center_x = x + width * 0.5
+    center_y = y + radius
+    if index >= 42:
+        center_y = 84.0
+
+    target_mask = ellipse_mask(
+        frame.shape, (center_x, center_y), (radius, radius), 0.85
+    )
+
+    if template is None:
+        return frame
+
+    scale = radius / template["radius"]
+    source_center = template["center"]
+    transform = np.array(
+        [
+            [scale, 0.0, center_x - scale * source_center[0]],
+            [0.0, scale, center_y - scale * source_center[1]],
+        ],
+        dtype=np.float32,
+    )
+    warped = cv2.warpAffine(
+        template["patch"],
+        transform,
+        (frame.shape[1], frame.shape[0]),
+        flags=cv2.INTER_CUBIC,
+        borderMode=cv2.BORDER_REFLECT_101,
+    )
+    warped_mask = cv2.warpAffine(
+        template["mask"],
+        transform,
+        (frame.shape[1], frame.shape[0]),
+        flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=0,
+    )
+
+    lab = cv2.cvtColor(source_frame, cv2.COLOR_RGB2LAB).astype(np.int16)
+    pink_visible = (
+        (lab[..., 1] > 126)
+        & (lab[..., 1] > lab[..., 2])
+        & (lab[..., 0] > 120)
+        & (target_mask > 0.20)
+    )
+    visible_alpha = cv2.GaussianBlur(
+        pink_visible.astype(np.float32), (0, 0), 0.85
+    )
+
+    template_pixels = warped[target_mask > 0.03]
+    visible_pixels = source_frame[pink_visible]
+    if len(visible_pixels) > 12 and len(template_pixels) > 12:
+        source_mean = visible_pixels.mean(axis=0)
+        template_mean = warped[warped_mask > 0.35].mean(axis=0)
+        correction = np.clip(source_mean - template_mean, -18.0, 18.0)
+        warped = np.clip(
+            warped.astype(np.float32) + correction[None, None, :], 0, 255
+        ).astype(np.uint8)
+
+    fill_alpha = warped_mask * (1.0 - visible_alpha) * target_mask
+    fill_alpha = np.clip(fill_alpha, 0.0, 1.0)
+    output = (
+        frame.astype(np.float32) * (1.0 - fill_alpha[..., None])
+        + warped.astype(np.float32) * fill_alpha[..., None]
+    )
+    return np.clip(output, 0, 255).astype(np.uint8)
+

+def main():
+    source = np.load(ROOT / "source_pixels.npz")["rgb"].astype(np.uint8)
+    homographies = estimate_homographies(source)
+    clean_plate = build_clean_plate(source[0])
+    pink_template = extract_pink_template(source)
+
+    frames = source.copy()
+    height, width = source.shape[1:3]
+
+    for index in range(INTERVENTION_INDEX, len(source)):
+        current = source[index]
+        inverse = np.linalg.inv(homographies[index])
+        warped_plate = cv2.warpPerspective(
+            clean_plate,
+            inverse,
+            (width, height),
+            flags=cv2.INTER_CUBIC,
+            borderMode=cv2.BORDER_REFLECT_101,
+        )
+
+        center_x, center_y, radius_x, radius_y = football_geometry(index)
+        alpha = ellipse_mask(
+            current.shape,
+            (center_x, center_y + 1.2),
+            (radius_x + 2.3, radius_y + 3.0),
+            feather=1.25,
+        )
+        alpha = np.clip(alpha * 1.12, 0.0, 1.0)
+        replaced = (
+            current.astype(np.float32) * (1.0 - alpha[..., None])
+            + warped_plate.astype(np.float32) * alpha[..., None]
+        )
+        frame = np.clip(replaced, 0, 255).astype(np.uint8)
+        frames[index] = restore_pink_ball(
+            frame, index, source[index], pink_template
+        )
+
