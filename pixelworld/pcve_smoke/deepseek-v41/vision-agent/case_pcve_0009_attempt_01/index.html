+def largest_component(mask):
+    count, labels, stats, _ = cv2.connectedComponentsWithStats(
+        mask.astype(np.uint8), connectivity=8
+    )

+        return mask
+    best = 1 + int(np.argmax(stats[1:, cv2.CC_STAT_AREA]))
+    return labels == best

+def object_center(mask, region):
+    local = mask & region
+    local = largest_component(local)
+    ys, xs = np.nonzero(local)
+    if len(xs) == 0:
+        return None
+    return float(xs.mean()), float(ys.mean())

+def track_mallets(frames):
+    h, w = frames.shape[1:3]
+    yy, xx = np.mgrid[:h, :w]
+
+    red_region = (xx >= 90) & (xx <= 185) & (yy >= 52) & (yy <= 96)
+    white_region = (xx >= 94) & (xx <= 114) & (yy >= 73) & (yy <= 89)
+    cyan_region = (xx >= 120) & (xx <= 180) & (yy >= 45) & (yy <= 82)
+
+    red_track = []
+    white_track = []
+    cyan_centers = []
+    cyan_masks = []
+
+    for frame in frames:
+        rgb = frame.astype(np.int16)
+        r, g, b = rgb[:, :, 0], rgb[:, :, 1], rgb[:, :, 2]
+
+        red_mask = (r - np.maximum(g, b) > 30) & (r > 160)
+        white_mask = (rgb.min(axis=2) > 228) & (
+            rgb.max(axis=2) - rgb.min(axis=2) < 28
+        )
+        cyan_mask = (b - g > 22) & (g - r > 12) & (b - r > 32) & (r < 165)
+
+        red_center = object_center(red_mask, red_region)
+        white_center = object_center(white_mask, white_region)

+
+        if red_center is None:
+            red_center = red_track[-1]
+        if white_center is None:
+            white_center = white_track[-1] if white_track else (105.2, 80.7)
+        if cyan_center is None:
+            cyan_center = cyan_centers[-1] if cyan_centers else (146.1, 65.6)
+
+        red_track.append(red_center)
+        white_track.append(white_center)
+        cyan_centers.append(cyan_center)
+
+        # Include the darker rim around the colored mallet face.
+        cyan_seed = largest_component(cyan_mask & cyan_region)
+        cyan_full = binary_dilation(cyan_seed, iterations=2)
+        cyan_full &= (xx >= 135) & (xx <= 180) & (yy >= 45) & (yy <= 82)
+        cyan_masks.append(cyan_full)
+
+    red_track = np.asarray(red_track, dtype=np.float32)
+    white_track = np.asarray(white_track, dtype=np.float32)
+
+    # The white mallet is stationary. Smooth away tiny highlight-threshold jitter.
+    white_track[:] = np.median(white_track, axis=0)
+    return red_track, white_track, cyan_centers, cyan_masks

+def make_clean_plate(frames, original_masks):
+    """Remove the cyan mallet using two source frames where each path is clear."""
+    plate_frames = frames.copy()
+    clean_a = frames[0]
+    clean_b = frames[-1]
+
+    mask_a = original_masks[0]
+    mask_b = original_masks[-1]
+
+    for index, frame in enumerate(frames):
+        source_mask = original_masks[index]
+        use_a = source_mask & ~mask_a
+        use_b = source_mask & mask_a & ~mask_b
+        plate_frames[index][use_a] = clean_a[use_a]
+        plate_frames[index][use_b] = clean_b[use_b]
+
+        # Soften the one-pixel seam left by donor replacement without touching
+        # pixels outside the removed object's neighborhood.
+        residual = source_mask & ~use_a & ~use_b
+        if residual.any():
+            plate_frames[index][residual] = clean_a[residual]
+    return plate_frames
+

+def build_sprite(frames, plate_frames, cyan_centers, cyan_masks):
+    # Use a settled source frame for a stable, clean mallet appearance.
+    source_index = 30
+    frame = frames[source_index]
+    plate = plate_frames[source_index]
+    mask = cyan_masks[source_index].copy()
+
+    cx, cy = cyan_centers[source_index]
+    x0 = int(np.floor(cx - 8))
+    x1 = int(np.ceil(cx + 9))
+    y0 = int(np.floor(cy - 7))
+    y1 = int(np.ceil(cy + 8))
+
+    crop = frame[y0:y1, x0:x1].astype(np.float32)
+    bg = plate[y0:y1, x0:x1].astype(np.float32)
+    local_mask = mask[y0:y1, x0:x1]
+
+    # Chroma difference gives a clean, feathered alpha without carrying table
+    # texture into the sprite.
+    chroma = np.linalg.norm(crop - bg, axis=2)
+    alpha = np.clip((chroma - 2.0) / 18.0, 0.0, 1.0)
+    alpha *= binary_dilation(local_mask, iterations=1)
+    alpha = gaussian_filter(alpha, sigma=0.45)
+    alpha = np.clip(alpha, 0.0, 1.0)
+
+    sprite = np.clip(crop, 0, 255).astype(np.uint8)
+    return sprite, alpha.astype(np.float32)
+
+
+def composite_sprite(frame, sprite, alpha, center):
+    cx, cy = center
+    sh, sw = sprite.shape[:2]
+    left = int(np.floor(cx - sw / 2))
+    top = int(np.floor(cy - sh / 2))
+
+    x0, x1 = max(0, left), min(frame.shape[1], left + sw)
+    y0, y1 = max(0, top), min(frame.shape[0], top + sh)
+    sx0, sy0 = x0 - left, y0 - top
+    sx1, sy1 = sx0 + (x1 - x0), sy0 + (y1 - y0)
+
+    patch = frame[y0:y1, x0:x1].astype(np.float32)
+    fg = sprite[sy0:sy1, sx0:sx1].astype(np.float32)
+    a = alpha[sy0:sy1, sx0:sx1, None]
+
+    # A small contact shadow keeps the mallet anchored to the tabletop.
+    shadow_alpha = gaussian_filter(a[:, :, 0], sigma=1.1) * 0.24
+    shadow_alpha = np.roll(shadow_alpha, (1, 1), axis=(0, 1))
+    patch = patch * (1.0 - shadow_alpha[:, :, None] * 0.34)
+    frame[y0:y1, x0:x1] = np.clip(
+        patch * (1.0 - a) + fg * a, 0, 255
+    ).astype(np.uint8)
+
+
+def main():
+    data = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    frames = data.copy()
+    red_track, white_track, cyan_centers, cyan_masks = track_mallets(frames)
+    plate_frames = make_clean_plate(frames, cyan_masks)
+    sprite, alpha = build_sprite(frames, plate_frames, cyan_centers, cyan_masks)
+
+    output = plate_frames.copy()
+    midpoint_track = (red_track + white_track) * 0.5
+    for index, center in enumerate(midpoint_track):
+        composite_sprite(output[index], sprite, alpha, center)
+
+    output = np.clip(output, 0, 255).astype(np.uint8)
+    artifact_io.save(output, CASE_DIR)
+
+    writer = cv2.VideoWriter(
+        str(CASE_DIR / "output.mp4"),
+        cv2.VideoWriter_fourcc(*"mp4v"),
+        24.0,
+        (OUT_W, OUT_H),
+    )
+    for frame in output:
+        writer.write(cv2.cvtColor(frame, cv2.COLOR_RGB2BGR))
+    writer.release()
+
+    print("red start/end:", red_track[0].round(2), red_track[-1].round(2))
+    print("white:", white_track[0].round(2))
+    print("midpoint start/end:", midpoint_track[0].round(2), midpoint_track[-1].round(2))
+    print("sprite shape:", sprite.shape, "alpha pixels:", int((alpha > 0.05).sum()))
+
+
+if __name__ == "__main__":
+    main()
