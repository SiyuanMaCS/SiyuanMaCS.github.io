+
+The source ball approaches the block from the left and loses nearly all of its
+forward speed at contact.  A 1.8x restitution therefore reverses the small
+post-impact normal velocity.  The result below keeps the pre-contact frames
+verbatim, then rolls the ball back to the left with the corresponding rebound.
+"""
+
+from __future__ import annotations
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+CONTACT_FRAME = 12
+
+
+def feather_mask(height: int, width: int, left: int, right: int, feather: int) -> np.ndarray:
+    """Return a horizontal feather mask that is one in [left, right)."""
+    x = np.arange(width, dtype=np.float32)
+    mask = np.clip((x - left) / feather, 0.0, 1.0)
+    mask *= np.clip((right - x) / feather, 0.0, 1.0)
+    return np.broadcast_to(mask[None, :], (height, width)).copy()
+
+
+def build_clean_plate(source: np.ndarray) -> np.ndarray:
+    """Remove the source ball and its shadow while retaining the static scene."""
+    clean = source[0].copy()
+    donor = source[95]
+
+    # The ball and its floor shadow occupy the left foreground in frame zero.
+    # By the final frame the left side is unobstructed, so it provides a clean
+    # donor for that area without touching the wooden block.
+    alpha = np.zeros(clean.shape[:2], dtype=np.float32)
+    alpha[30:116, 0:102] = 1.0
+    edge = feather_mask(86, 102, 0, 102, 8)
+    alpha[30:116, :102] *= edge
+    clean = clean.astype(np.float32) * (1.0 - alpha[..., None]) + donor.astype(np.float32) * alpha[..., None]
+    return np.clip(clean, 0, 255).astype(np.uint8)
+
+
+def extract_ball_sprite(source: np.ndarray) -> tuple[np.ndarray, np.ndarray, float]:
+    """Extract a clean, fully visible ball from the last unobstructed frame."""
+    center = np.array([78.5, 68.5], dtype=np.float32)
+    radius = 17.2
+    half = 25
+    x0 = int(round(center[0])) - half
+    y0 = int(round(center[1])) - half
+    x1 = x0 + 2 * half + 1
+    y1 = y0 + 2 * half + 1
+
+    rgb = source[9, y0:y1, x0:x1].copy()
+    yy, xx = np.mgrid[0 : rgb.shape[0], 0 : rgb.shape[1]].astype(np.float32)
+    cx = center[0] - x0
+    cy = center[1] - y0
+    dist = np.sqrt((xx - cx) ** 2 + (yy - cy) ** 2)
+
+    # Supersampled antialiased disk.
+    ss = 5
+    offsets = (np.arange(ss, dtype=np.float32) + 0.5) / ss - 0.5
+    coverage = np.zeros_like(dist, dtype=np.float32)
+    for oy in offsets:
+        for ox in offsets:
+            d = np.sqrt((xx + ox - cx) ** 2 + (yy + oy - cy) ** 2)
+            coverage += (d <= radius).astype(np.float32)
+    alpha = coverage / float(ss * ss)
+
+    # Suppress a faint donor-background rim while preserving the ball edge.
+    alpha = np.clip(alpha * 1.04, 0.0, 1.0)
+    return rgb, alpha, radius
+
+
+def put_rgba(
+    canvas: np.ndarray,
+    sprite_rgb: np.ndarray,
+    sprite_alpha: np.ndarray,
+    source_radius: float,
+    center: tuple[float, float],
+    radius: float,
+    angle_deg: float,
+) -> None:
+    """Warp one antialiased sprite into the destination frame."""
+    h, w = canvas.shape[:2]
+    sh, sw = sprite_alpha.shape
+    scale = radius / source_radius
+
+    rot = cv2.getRotationMatrix2D((sw * 0.5, sh * 0.5), angle_deg, scale)
+    rot[0, 2] += center[0] - sw * 0.5
+    rot[1, 2] += center[1] - sh * 0.5
+
+    rgba = np.dstack([sprite_rgb.astype(np.float32), sprite_alpha * 255.0])
+    warped = cv2.warpAffine(
+        rgba,
+        rot,
+        (w, h),
+        flags=cv2.INTER_CUBIC,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=(0, 0, 0, 0),
+    )
+    rgb = warped[..., :3]
+    alpha = np.clip(warped[..., 3] / 255.0, 0.0, 1.0)[..., None]
+    canvas[:] = np.clip(canvas.astype(np.float32) * (1.0 - alpha) + rgb * alpha, 0, 255).astype(np.uint8)
+
+
+def add_ball_shadow(canvas: np.ndarray, center: tuple[float, float], radius: float) -> None:
+    """Darken the floor/block with a soft, perspective-squashed contact shadow."""
+    h, w = canvas.shape[:2]
+    mask = np.zeros((h, w), dtype=np.float32)
+    shadow_center = (int(round(center[0] + radius * 0.35)), int(round(center[1] + radius * 0.70)))
+    axes = (max(4, int(round(radius * 0.82))), max(2, int(round(radius * 0.38))))
+    cv2.ellipse(mask, shadow_center, axes, 0, 0, 360, 1.0, -1, lineType=cv2.LINE_AA)
+    mask = cv2.GaussianBlur(mask, (0, 0), sigmaX=2.2, sigmaY=1.5)
+    mask *= 0.48
+    darkened = canvas.astype(np.float32) * 0.58
+    canvas[:] = np.clip(canvas.astype(np.float32) * (1.0 - mask[..., None]) + darkened * mask[..., None], 0, 255).astype(np.uint8)
+
+
+def counterfactual_center(index: int) -> tuple[float, float, float]:
+    """Return ball center and radius after the contact frame."""
+    dt = float(index - CONTACT_FRAME)
+
+    # Source post-impact horizontal speed is about 0.65 px/frame after an
+    # incoming normal speed of about 6.5 px/frame.  The requested 1.8x
+    # restitution gives a reflected speed near 1.18 px/frame.  Mild rolling
+    # friction lets the ball remain visible through the four-second shot.
+    vx0 = -1.18
+    damping = 0.008
+    x = 94.0 + vx0 * (1.0 - np.exp(-damping * dt)) / damping
+    y = 72.0 + 0.095 * dt
+    radius = 16.6 - 0.022 * dt
+    return float(x), float(y), float(radius)
+
+
+def main() -> None:
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (96, 144, 256, 3), source.shape
+
+    clean_plate = build_clean_plate(source)
+    sprite_rgb, sprite_alpha, sprite_radius = extract_ball_sprite(source)
+    frames = source.copy()
+
+    for index in range(CONTACT_FRAME, source.shape[0]):
+        frame = clean_plate.copy()
+        x, y, radius = counterfactual_center(index)
+        add_ball_shadow(frame, (x, y), radius)
+
+        # The rebound rolls opposite to the incoming travel direction.
+        angle = -2.15 * (index - CONTACT_FRAME + 1)
+        put_rgba(frame, sprite_rgb, sprite_alpha, sprite_radius, (x, y), radius, angle)
+        frames[index] = frame
+
+    artifact_io.save(frames, CASE_DIR)
+
+
+if __name__ == "__main__":
+    main()
