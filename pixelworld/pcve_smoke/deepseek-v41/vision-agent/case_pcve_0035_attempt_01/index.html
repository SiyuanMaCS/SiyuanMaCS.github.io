+def feather_alpha(height, width, feather=3):
+    y = np.minimum(np.arange(height), np.arange(height)[::-1])
+    x = np.minimum(np.arange(width), np.arange(width)[::-1])

+def build_plate(source):

+    # where the source ball occupied the foreground at frame 1.

+def ball_alpha(patch, center_x, center_y, radius):
+    height, width = patch.shape[:2]
+    yy, xx = np.indices((height, width), dtype=np.float64)
+    distance = np.hypot(xx - center_x, yy - center_y)
+    circle = np.clip((radius + 1.1 - distance) / 2.0, 0.0, 1.0)
+
+    values = patch.astype(np.int16)
+    blueness = values[..., 2] - values[..., 0]
+    blue = np.clip((blueness - 28.0) / 45.0, 0.0, 1.0)
+    return (circle * blue)[..., None]

+def composite_sprite(frame, source, target_y, source_index):

+    half = 6

+    # A restrained, source-coloured contact reflection helps the low sprite
+    # sit on the polished lane without adding a synthetic halo.

+    reflection_center_y = sy - top + dy
