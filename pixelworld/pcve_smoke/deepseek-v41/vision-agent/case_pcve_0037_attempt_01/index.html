+def _largest_car_component(diff, region):
+    x0, y0, x1, y1 = region
+    mask = np.zeros(diff.shape, dtype=np.uint8)
+    mask[y0:y1, x0:x1] = (diff[y0:y1, x0:x1] > 5).astype(np.uint8)
+    mask = cv2.morphologyEx(mask, cv2.MORPH_CLOSE, np.ones((3, 3), np.uint8))
+    count, labels, stats, _ = cv2.connectedComponentsWithStats(mask, 8)
+    if count <= 1:
+        raise RuntimeError("Could not locate the moving toy car")
+    index = 1 + int(np.argmax(stats[1:, cv2.CC_STAT_AREA]))
+    component = (labels == index).astype(np.uint8)
+    component = cv2.dilate(component, np.ones((3, 3), np.uint8), iterations=1)
+    return component
+

+def _extract_sprite(frame, plate, region):
+    diff = np.max(np.abs(frame.astype(np.int16) - plate.astype(np.int16)), axis=2)
+    mask = _largest_car_component(diff, region)
+    ys, xs = np.nonzero(mask)
+    x0, x1 = int(xs.min()), int(xs.max()) + 1
+    y0, y1 = int(ys.min()), int(ys.max()) + 1
+
+    # Keep a small margin so the antialiased silhouette survives the warp.
+    x0, y0 = max(0, x0 - 2), max(0, y0 - 2)
+    x1, y1 = min(frame.shape[1], x1 + 2), min(frame.shape[0], y1 + 2)
+
+    local_mask = mask[y0:y1, x0:x1].astype(np.float32)

+    rgb = frame[y0:y1, x0:x1].astype(np.float32)
+    centroid = np.array([xs.mean(), ys.mean()], dtype=np.float64)
+    return {
+        "rgb": rgb,
+        "alpha": alpha,
+        "origin": np.array([x0, y0], dtype=np.float64),
+        "centroid": centroid,

+def _render_interpolated_sprite(sprites, time, target_center, shape):
+    i0 = int(np.floor(time))
+    i1 = min(i0 + 1, len(sprites) - 1)
+    fraction = float(time - i0)
+
+    accum_rgb = np.zeros((*shape[:2], 3), dtype=np.float32)
+    accum_alpha = np.zeros(shape[:2], dtype=np.float32)
+
+    for index, weight in ((i0, 1.0 - fraction), (i1, fraction)):
+        if weight <= 1e-8:
+            continue
+        sprite = sprites[index]
+        shift = np.rint(target_center - sprite["centroid"]).astype(int)
+        ox, oy = sprite["origin"].astype(int) + shift
+        h, w = sprite["alpha"].shape
+        sx0, sy0 = max(0, -ox), max(0, -oy)
+        dx0, dy0 = max(0, ox), max(0, oy)
+        sx1, sy1 = min(w, shape[1] - ox), min(h, shape[0] - oy)
+        if sx1 <= sx0 or sy1 <= sy0:
+            continue
+        rgb = sprite["rgb"][sy0:sy1, sx0:sx1]
+        alpha = sprite["alpha"][sy0:sy1, sx0:sx1]
+        patch_rgb = accum_rgb[dy0:dy0 + (sy1 - sy0), dx0:dx0 + (sx1 - sx0)]
+        patch_alpha = accum_alpha[dy0:dy0 + (sy1 - sy0), dx0:dx0 + (sx1 - sx0)]
+        patch_rgb += weight * rgb * alpha[..., None]
+        patch_alpha += weight * alpha
+
+    safe_alpha = np.maximum(accum_alpha, 1e-6)
+    accum_rgb /= safe_alpha[..., None]
+    return accum_rgb, np.clip(accum_alpha, 0.0, 1.0)

+def build_video(source):

+    # Frame 1 has the table visible where the car eventually rests. Copy that

+    feather[-3:] *= np.linspace(1.0, 0.0, 3)[:, None]
+    feather[:, :3] *= np.linspace(0.0, 1.0, 3)[None, :]

+    plate[y0:y1, x0:x1] = np.clip(
+        plate_patch * (1.0 - feather[..., None]) + patch * feather[..., None],
+        0,
+        255,
+    ).astype(np.uint8)
+
+    sprites = [
+        _extract_sprite(source[i], plate, CAR_REGION)
+        for i in range(24)
+    ]
+

+    acceleration = float(2.0 * coefficients[0])
+    if initial_velocity <= 0 or acceleration >= 0:
+        raise RuntimeError("Unexpected source trajectory")
+    stop_time = -initial_velocity / acceleration
+
+    output = np.empty((FRAME_COUNT, *source.shape[1:]), dtype=np.uint8)
+    output[0] = source[0]

+        source_time = min(frame_index * SPEED_SCALE, stop_time)
+        i0 = int(np.floor(source_time))
+        i1 = min(i0 + 1, len(sprites) - 1)

+        target_center = centers[i0] * (1.0 - fraction) + centers[i1] * fraction
+        car_rgb, car_alpha = _render_interpolated_sprite(
+            sprites, source_time, target_center, source.shape[1:]
+        )
+        canvas = plate.astype(np.float32)
+        canvas = canvas * (1.0 - car_alpha[..., None]) + car_rgb * car_alpha[..., None]
+        output[frame_index] = np.clip(canvas, 0, 255).astype(np.uint8)
+
+    return output
+
+
+def main():
+    with np.load(CASE_DIR / "source_pixels.npz") as data:
+        source = data["rgb"]
+    frames = build_video(source)
+    artifact_io.save(frames, CASE_DIR)
+    print(f"saved {frames.shape} counterfactual video")
+

+    main()
