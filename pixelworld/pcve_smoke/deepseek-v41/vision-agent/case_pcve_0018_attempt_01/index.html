+def make_plate():
+    """Temporal median gives a clean plate without either moving ball."""
+    plate = np.median(SRC, axis=0).astype(np.float32)
+    # The median retains a faint star-ball ghost.  Fill that low-motion patch
+    # from the settled tail where the ball is already absent at that location.
+    clean = np.median(SRC[62:95], axis=0).astype(np.float32)
+    for frame in range(N):
+        # Camera is essentially fixed; use the clean late plate for the floor.
+        pass
+    plate[72:140, 70:132] = clean[72:140, 70:132]
+    return plate

+def ball_layer(frame_index=0):
+    """Return an RGBA sprite and its source center from a sharp source frame."""
+    plate = make_plate()
+    frame = SRC[frame_index].astype(np.float32)
+    cx, cy = 103.0, 101.8
+    x0, x1 = int(cx - 16), int(cx + 17)
+    y0, y1 = int(cy - 16), int(cy + 17)
+    roi = frame[y0:y1, x0:x1]
+    base = plate[y0:y1, x0:x1]
+    diff = np.max(np.abs(roi - base), axis=2)
+    yy, xx = np.mgrid[y0:y1, x0:x1]
+    radius = 13.5
+    inside = ((xx - cx) ** 2 + (yy - cy) ** 2) <= radius**2
+    alpha = np.clip((diff - 5.0) / 13.0, 0.0, 1.0)
+    alpha *= inside.astype(np.float32)
+    # Close the texture holes without changing the silhouette.
+    alpha = cv2.GaussianBlur(alpha, (0, 0), 0.65)
+    alpha = np.where(inside, np.maximum(alpha, 0.82), 0.0).astype(np.float32)
+    rgba = np.dstack([roi, alpha * 255.0])
+    return rgba, (x0, y0)
+
+
+def ease_velocity(t):
+    """Match the source's slow deceleration, with a 0.75 initial speed."""
+    t = np.asarray(t, dtype=np.float32)
+    # Source displacement is almost linear, with the final ~0.5 s nearly still.
+    speed = np.maximum(0.0, 1.0 - 0.12 * t)
+    return speed
+
+
+def counterfactual_centers(n=N):
+    # At the intervention frame the ball is already moving.  The lower speed
+    # changes all subsequent positions while retaining the same direction.
+    t = np.arange(n, dtype=np.float32) / 24.0
+    # Source starts ~0.0 and travels 26 px over four seconds.
+    src_total = np.array([1.4, 26.0])
+    speed_integral = np.cumsum(ease_velocity(t)) / 24.0
+    norm = speed_integral[-1]
+    travel = 0.75 * src_total[None, :] * (speed_integral[:, None] / norm)
+    centers = SOURCE_CENTERS[0][None, :] + travel
+    return centers
+
+
+def composite_sprite(frame, rgba, origin, center):
+    """Alpha composite an RGBA sprite at an integer pixel origin."""
+    x0, y0 = origin
+    x = int(round(center[0] - (rgba.shape[1] - 1) * 0.5))
+    y = int(round(center[1] - (rgba.shape[0] - 1) * 0.5))
+    h, w = rgba.shape[:2]
+    sx0, sy0 = max(0, -x), max(0, -y)
+    dx0, dy0 = max(0, x), max(0, y)
+    sx1, sy1 = min(w, W - x), min(h, H - y)
+    if sx1 <= sx0 or sy1 <= sy0:
+        return frame
+    src = rgba[sy0:sy1, sx0:sx1]
+    a = src[..., 3:4] / 255.0
+    dst = frame[dy0:dy0 + sy1 - sy0, dx0:dx0 + sx1 - sx0]
+    frame[dy0:dy0 + sy1 - sy0, dx0:dx0 + sx1 - sx0] = (
+        src[..., :3] * a + dst * (1.0 - a)
+    )
+    return frame
+
+
+def make_frames():
+    plate = make_plate()
+    rgba, origin = ball_layer(0)
+    centers = counterfactual_centers()
+    frames = []
+    for i in range(N):
+        frame = plate.copy()
+        # Preserve the independent pink ball and all static scene detail.
+        # Pink ball is present only in the first six frames; use the source
+        # pixels outside the star-ball region.
+        if i <= 8:
+            src = SRC[i].astype(np.float32)
+            yy, xx = np.mgrid[:H, :W]
+            keep = ~(((xx - 100) ** 2 + (yy - 105) ** 2) < 34**2)
+            frame[keep] = src[keep]
+        # The plate's floor and cabinet are already clean, but preserve the
+        # original subtle exposure drift using a broad luminance correction.
+        target_luma = float(SRC[i].mean())
+        current_luma = float(frame.mean())
+        frame *= np.clip(target_luma / max(current_luma, 1.0), 0.98, 1.02)
+        composite_sprite(frame, rgba, origin, centers[i])
+        frames.append(np.clip(frame, 0, 255).astype(np.uint8))
+    return np.stack(frames)
