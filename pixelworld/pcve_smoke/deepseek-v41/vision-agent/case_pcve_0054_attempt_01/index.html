+import cv2

+FRAMES = 96
+WIDTH = 256
+HEIGHT = 144
+FPS = 24

+# The scene is a fixed camera with three small stones on a shallow diagonal rail.
+# This line was fitted to the source stone centers.
+LINE_INTERCEPT = 63.4
+LINE_SLOPE = 0.060
+STONE_RADIUS = 5.25
+RESTITUTION = 0.10
+ROLLING_DRAG = 0.004
+
+
+def line_y(x: float) -> float:
+    return LINE_INTERCEPT + LINE_SLOPE * x
+
+
+def make_plate(source: np.ndarray) -> np.ndarray:
+    """Use the unobstructed first frame as the plate and remove both source stones."""
+    plate = source[0].copy()
+    red_fill = np.median(source[10:21, 60:82, 36:61], axis=0).astype(np.uint8)
+    yellow_fill = np.median(source[10:19, 68:89, 201:226], axis=0).astype(np.uint8)
+    plate[60:82, 36:61] = red_fill
+    plate[68:89, 201:226] = yellow_fill
+    return plate
+

+def supersampled_disc(
+    rgb: tuple[int, int, int],
+    scale: int = 6,
+    rx: float = 5.5,
+    ry: float = 4.35,
+) -> np.ndarray:
+    """Render a small shaded stone with a soft contact shadow as RGBA."""
+    width = int(np.ceil((rx * 2 + 10) * scale))
+    height = int(np.ceil((ry * 2 + 12) * scale))
+    yy, xx = np.mgrid[0:height, 0:width]
+    cx = width * 0.5
+    cy = ry * scale + 3.0 * scale
+    dx = (xx - cx) / (rx * scale)
+    dy = (yy - cy) / (ry * scale)
+    rr = dx * dx + dy * dy
+
+    # Contact shadow, biased down and slightly right as in the source footage.
+    sx = (xx - (cx + 1.1 * scale)) / (rx * 1.18 * scale)
+    sy = (yy - (cy + ry * 0.82 * scale)) / (ry * 0.52 * scale)
+    sr = sx * sx + sy * sy
+    shadow_alpha = np.clip(1.0 - sr, 0.0, 1.0) ** 1.7
+    shadow_alpha *= 0.30
+
+    edge = np.clip((1.0 - rr) * rx * scale * 0.85, 0.0, 1.0)
+    nx = np.clip(dx, -1.0, 1.0)
+    ny = np.clip(dy, -1.0, 1.0)
+    nz = np.sqrt(np.clip(1.0 - nx * nx - ny * ny, 0.0, 1.0))
+    light = np.clip(0.22 + 0.72 * (-0.48 * nx - 0.58 * ny + 0.66 * nz), 0.0, 1.0)
+    highlight = np.clip(1.0 - ((nx + 0.30) ** 2 + (ny + 0.38) ** 2) / 0.18, 0.0, 1.0)
+
+    base = np.asarray(rgb, dtype=np.float32)
+    shade = base[None, None, :] * (0.54 + 0.52 * light[..., None])
+    shade += highlight[..., None] * np.asarray((40.0, 36.0, 32.0))
+    rim = np.clip((0.20 - nz) * 3.0, 0.0, 1.0) * edge
+    shade *= (1.0 - 0.22 * rim[..., None])
+
+    rgba = np.zeros((height, width, 4), dtype=np.float32)
+    rgba[..., :3] = shade
+    rgba[..., 3] = edge
+
+    # Premultiplied dark shadow beneath the stone.
+    shadow_rgb = np.asarray((54.0, 69.0, 84.0))
+    under = shadow_alpha * (1.0 - edge)
+    rgba[..., :3] = rgba[..., :3] * edge[..., None] + shadow_rgb * under[..., None]
+    rgba[..., 3] = np.maximum(edge, under)
+    return rgba
+
+
+def downscale_rgba(rgba: np.ndarray, scale: int = 6) -> np.ndarray:
+    h, w = rgba.shape[:2]
+    return cv2.resize(rgba, (max(1, w // scale), max(1, h // scale)), interpolation=cv2.INTER_AREA)
+
+
+def paste_sprite(frame: np.ndarray, sprite: np.ndarray, center_x: float, center_y: float) -> None:
+    """Alpha-composite a premultiplied-ish RGBA sprite around a stone center."""
+    sh, sw = sprite.shape[:2]
+    # Body center in the generated sprite is 3 source pixels below the patch top.
+    left = int(round(center_x - sw / 2))
+    top = int(round(center_y - 7.35))
+    x0 = max(0, left)
+    y0 = max(0, top)
+    x1 = min(WIDTH, left + sw)
+    y1 = min(HEIGHT, top + sh)
+    if x0 >= x1 or y0 >= y1:
+        return
+    sx0, sy0 = x0 - left, y0 - top
+    patch = sprite[sy0 : sy0 + (y1 - y0), sx0 : sx0 + (x1 - x0)]
+    alpha = patch[..., 3:4].astype(np.float32)
+    src = patch[..., :3].astype(np.float32)
+    dst = frame[y0:y1, x0:x1].astype(np.float32)
+    # Sprite RGB is straight color except for the shadow contribution; a normal
+    # source-over blend preserves the original plate through soft edges.
+    out = src * alpha + dst * (1.0 - alpha)
+    frame[y0:y1, x0:x1] = np.clip(out + 0.5, 0, 255).astype(np.uint8)
+
+
+def simulate_positions() -> np.ndarray:
+    """Three equal-radius stones with a low-restition chain of collisions."""
+    names = ("red", "blue", "yellow")
+    x = np.array((47.5, 47.5 + 0.25 * (214.0 - 47.5), 214.0), dtype=np.float64)
+    v = np.array((2.00, 0.0, -2.10), dtype=np.float64)
+    positions = np.empty((FRAMES, 3), dtype=np.float64)
+    substeps = 16
+    dt = 1.0 / substeps
+
+    for frame in range(FRAMES):
+        positions[frame] = x
+        for _ in range(substeps):
+            x += v * dt
+            speed = np.abs(v)
+            v -= np.sign(v) * np.minimum(speed, ROLLING_DRAG * dt)
+
+            # Resolve the one-dimensional chain, nearest contacts first.
+            for _pass in range(3):
+                moved = False
+                for i in range(2):
+                    j = i + 1
+                    overlap = STONE_RADIUS * 2.0 - (x[j] - x[i])
+                    if overlap <= 0.0:
+                        continue
+                    moved = True
+                    x[i] -= overlap * 0.5
+                    x[j] += overlap * 0.5
+                    rel = v[i] - v[j]
+                    if rel > 0.0:
+                        vi = 0.5 * (v[i] + v[j]) + 0.5 * RESTITUTION * rel
+                        vj = 0.5 * (v[i] + v[j]) - 0.5 * RESTITUTION * rel
+                        v[i], v[j] = vi, vj
+                if not moved:
+                    break
+
+    return positions
+
+
+def main() -> None:
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (FRAMES, HEIGHT, WIDTH, 3)
+    plate = make_plate(source)
+    positions = simulate_positions()
+
+    stone_rgba = {
+        "red": downscale_rgba(supersampled_disc((211, 55, 67))),
+        "blue": downscale_rgba(supersampled_disc((65, 137, 211))),
+        "yellow": downscale_rgba(supersampled_disc((211, 216, 62))),
+    }
+
+    frames = []
+    for frame_index, xs in enumerate(positions):
+        frame = plate.copy()
+        draw_order = sorted(range(3), key=lambda i: line_y(xs[i]))
+        for i in draw_order:
+            name = ("red", "blue", "yellow")[i]
+            paste_sprite(frame, stone_rgba[name], xs[i], line_y(xs[i]))
+        frames.append(frame)
+    frames = np.stack(frames).astype(np.uint8)
+
+    artifact_io.save(frames, CASE_DIR)
+
+    # Keep a directly playable H.264 copy alongside the required frame artifacts.
+    writer = cv2.VideoWriter(
+        str(CASE_DIR / "edited_video.mp4"),
+        cv2.VideoWriter_fourcc(*"mp4v"),
+        FPS,
+        (WIDTH, HEIGHT),
+    )
+    for frame in frames:
+        writer.write(cv2.cvtColor(frame, cv2.COLOR_RGB2BGR))
+    writer.release()
+
+    np.save(CASE_DIR / ".tmp" / "simulated_positions.npy", positions)
+    (CASE_DIR / ".tmp" / "positions.json").write_text(
+        json.dumps({str(i): row.round(4).tolist() for i, row in enumerate(positions)}, indent=2)
+    )
+    Image.fromarray(plate).save(CASE_DIR / ".tmp" / "clean_plate.png")
+
+
+if __name__ == "__main__":
+    main()
