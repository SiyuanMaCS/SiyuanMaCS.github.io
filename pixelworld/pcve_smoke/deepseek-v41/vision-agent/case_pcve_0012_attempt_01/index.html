+import json
+import shutil

+import cv2
+import numpy as np
+from scipy.interpolate import PchipInterpolator
+from scipy.signal import savgol_filter
+
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+SPEED_SCALE = 0.7
+
+
+def clean_background(frames):
+    """Remove the source ball from the one place where frame 0 is occluded."""
+    background = frames[0].copy()
+    background[:, :44] = np.median(frames, axis=0)[:, :44]
+    return background
+
+
+def track_ball(frames):
+    centers = []
+    radii = []
+    for frame in frames:
+        hsv = cv2.cvtColor(frame, cv2.COLOR_RGB2HSV)
+        mask = (
+            ((hsv[:, :, 0] < 10) | (hsv[:, :, 0] > 175))
+            & (hsv[:, :, 1] > 165)
+            & (hsv[:, :, 2] > 100)
+        ).astype(np.uint8)
+        count, labels, stats, centroids = cv2.connectedComponentsWithStats(mask)
+        index = 1 + int(np.argmax(stats[1:, cv2.CC_STAT_AREA]))
+        x, y, width, height, _ = stats[index]
+        centers.append(((x + x + width - 1) / 2.0, (y + y + height - 1) / 2.0))
+        radii.append(max(width, height) / 2.0)
+    return np.asarray(centers), np.asarray(radii)
+
+
+def make_tracks(centers, radii):
+    # The hue mask loses a few edge pixels at contact. A short polynomial
+    # smooth removes that one-frame measurement notch without changing motion.
+    xs = savgol_filter(centers[:, 0], 7, 2, mode="interp")
+    ys = savgol_filter(centers[:, 1], 7, 2, mode="interp")
+    rs = savgol_filter(radii, 9, 2, mode="interp")
+    source_t = np.arange(len(centers), dtype=np.float64)
+    return (
+        PchipInterpolator(source_t, xs),
+        PchipInterpolator(source_t, ys),
+        PchipInterpolator(source_t, rs),
+    )
+
+
+def extract_sprite(frame, background, center, radius):
+    cx, cy = center
+    left = int(np.floor(cx - radius - 15))
+    right = int(np.ceil(cx + radius + 29))
+    top = int(np.floor(cy - radius - 9))
+    bottom = int(np.ceil(cy + radius + 31))
+
+    frame_h, frame_w = frame.shape[:2]
+    clip_left = max(0, left)
+    clip_top = max(0, top)
+    clip_right = min(frame_w, right)
+    clip_bottom = min(frame_h, bottom)
+
+    rgb = frame[clip_top:clip_bottom, clip_left:clip_right]
+    bg = background[clip_top:clip_bottom, clip_left:clip_right]
+    difference = np.max(np.abs(rgb.astype(np.int16) - bg.astype(np.int16)), axis=2)
+
+    # A soft difference matte retains the antialiased ball edge and translucent
+    # contact shadow while suppressing compression noise in the room.
+    alpha = np.clip((difference.astype(np.float32) - 3.0) / 10.0, 0.0, 1.0)
+    alpha[difference <= 3] = 0.0
+
+    return {
+        "rgb": rgb,
+        "alpha": alpha,
+        "left": clip_left,
+        "top": clip_top,
+        "center": (cx, cy),
+        "radius": radius,
+    }
+
+
+def composite_sprite(canvas, sprite, target_center, target_radius):
+    source_cx, source_cy = sprite["center"]
+    source_r = sprite["radius"]
+    scale = target_radius / source_r
+
+    source_anchor_x = source_cx - sprite["left"]
+    source_anchor_y = source_cy - sprite["top"]
+    height, width = sprite["rgb"].shape[:2]
+    scaled_width = max(1, int(round(width * scale)))
+    scaled_height = max(1, int(round(height * scale)))
+
+    rgb = cv2.resize(
+        sprite["rgb"],
+        (scaled_width, scaled_height),
+        interpolation=cv2.INTER_CUBIC,
+    )
+    alpha = cv2.resize(
+        sprite["alpha"],
+        (scaled_width, scaled_height),
+        interpolation=cv2.INTER_LINEAR,
+    )
+    alpha = np.clip(alpha, 0.0, 1.0)[:, :, None]
+
+    anchor_x = int(round(source_anchor_x * scale))
+    anchor_y = int(round(source_anchor_y * scale))
+    target_x = int(round(target_center[0])) - anchor_x
+    target_y = int(round(target_center[1])) - anchor_y
+
+    canvas_h, canvas_w = canvas.shape[:2]
+    dst_left = max(0, target_x)
+    dst_top = max(0, target_y)
+    dst_right = min(canvas_w, target_x + scaled_width)
+    dst_bottom = min(canvas_h, target_y + scaled_height)
+    if dst_left >= dst_right or dst_top >= dst_bottom:
+        return
+
+    src_left = dst_left - target_x
+    src_top = dst_top - target_y
+    src_right = src_left + dst_right - dst_left
+    src_bottom = src_top + dst_bottom - dst_top
+
+    foreground = rgb[src_top:src_bottom, src_left:src_right].astype(np.float32)
+    opacity = alpha[src_top:src_bottom, src_left:src_right]
+    destination = canvas[dst_top:dst_bottom, dst_left:dst_right].astype(np.float32)
+    blended = foreground * opacity + destination * (1.0 - opacity)
+    canvas[dst_top:dst_bottom, dst_left:dst_right] = np.clip(
+        np.rint(blended), 0, 255
+    ).astype(np.uint8)
+
+
+def save_candidate(frames, name):
+    candidate = ROOT / name
+    candidate.mkdir(exist_ok=True)
+    shutil.copy2(ROOT / "prompt.txt", candidate / "prompt.txt")
+    shutil.copy2(ROOT / "task_spec.json", candidate / "task_spec.json")
+    artifact_io.save(frames, candidate)
+
+
+def main():
+    source = np.load(ROOT / "source_pixels.npz")["rgb"]
+    background = clean_background(source)
+    centers, radii = track_ball(source)
+    x_track, y_track, r_track = make_tracks(centers, radii)
+
+    sprites = [
+        extract_sprite(frame, background, centers[index], radii[index])
+        for index, frame in enumerate(source)
+    ]
+
+    output = []
+    last_source_index = len(source) - 1
+    for output_index in range(len(source)):
+        source_time = min(SPEED_SCALE * output_index, float(last_source_index))
+        source_index = int(np.clip(round(source_time), 0, last_source_index))
+        target_center = (float(x_track(source_time)), float(y_track(source_time)))
+        target_radius = float(r_track(source_time))
+
+        frame = background.copy()
+        composite_sprite(frame, sprites[source_index], target_center, target_radius)
+        output.append(frame)
+
+    # Frame 1 is the untouched intervention boundary.
+    output[0] = source[0].copy()
+    frames = np.asarray(output, dtype=np.uint8)
+
+    save_candidate(frames, "candidate_v1")
+    artifact_io.save(frames, ROOT)
+
+    report = {
+        "speed_scale": SPEED_SCALE,
+        "source_frame_0_center": centers[0].tolist(),
+        "source_frame_95_center": centers[95].tolist(),
+        "output_frame_95_source_time": SPEED_SCALE * 95,
+        "output_frame_95_center": [
+            float(x_track(SPEED_SCALE * 95)),
+            float(y_track(SPEED_SCALE * 95)),
+        ],
+    }
+    (ROOT / "generation_report.json").write_text(json.dumps(report, indent=2))
+
+
+if __name__ == "__main__":
+    main()
