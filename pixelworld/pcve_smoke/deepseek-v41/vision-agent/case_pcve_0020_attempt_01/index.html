+BRANCH_FRAME = 25

+FOOT_CENTER = np.array([102.2, 103.0], dtype=np.float64)
+PINK_CENTER = np.array([101.3, 77.2], dtype=np.float64)
+
+FOOT_TEMPLATE = (105.0, 113.0, 18.0)

+def extract_sprite(image: np.ndarray, cx: float, cy: float, radius: float):
+    """Extract a round object and a softly antialiased circular alpha mask."""
+    pad = 2
+    x0 = int(np.floor(cx - radius - pad))
+    y0 = int(np.floor(cy - radius - pad))
+    x1 = int(np.ceil(cx + radius + pad))
+    y1 = int(np.ceil(cy + radius + pad))
+    x0, y0 = max(0, x0), max(0, y0)
+    x1, y1 = min(WIDTH, x1), min(HEIGHT, y1)
+
+    rgb = image[y0:y1, x0:x1].astype(np.float32)
+    yy, xx = np.mgrid[y0:y1, x0:x1].astype(np.float32)
+    distance = np.hypot(xx - cx, yy - cy)
+    alpha = np.clip(radius + 0.55 - distance, 0.0, 1.0).astype(np.float32)
+    return rgb, alpha, np.array([cx, cy], dtype=np.float32)
+
+
+def make_clean_plate(source: np.ndarray) -> np.ndarray:
+    """Remove the two source balls from the first frame and repair the floor."""
+    plate = source[0].copy()
+    mask = np.zeros((HEIGHT, WIDTH), dtype=np.uint8)
+    cv2.circle(mask, (102, 103), 22, 255, -1)
+    cv2.circle(mask, (237, 68), 22, 255, -1)
+    plate = cv2.inpaint(plate, mask, 7, cv2.INPAINT_TELEA)
+    return plate
+

+def add_soft_shadow(
+    canvas: np.ndarray,
+    center: tuple[float, float],
+    size: tuple[float, float],
+    strength: float,
+) -> None:
+    """Apply a restrained floor-contact shadow to an RGB float canvas."""
+    cx, cy = center
+    rx, ry = size
+    pad_x = int(np.ceil(rx * 2.8))
+    pad_y = int(np.ceil(ry * 3.2))
+    x0 = max(0, int(np.floor(cx)) - pad_x)
+    x1 = min(WIDTH, int(np.ceil(cx)) + pad_x + 1)
+    y0 = max(0, int(np.floor(cy)) - pad_y)
+    y1 = min(HEIGHT, int(np.ceil(cy)) + pad_y + 1)
+    yy, xx = np.mgrid[y0:y1, x0:x1].astype(np.float32)
+    falloff = np.exp(
+        -0.5 * (((xx - cx - 1.2) / rx) ** 2 + ((yy - cy - 1.0) / ry) ** 2)
+    )
+    canvas[y0:y1, x0:x1] *= (1.0 - strength * falloff[..., None])
+
+
+def composite_sprite(
+    canvas: np.ndarray,
+    sprite: tuple[np.ndarray, np.ndarray, np.ndarray],
+    target_center: np.ndarray,
+) -> None:
+    """Alpha-composite a sprite at a subpixel target location."""
+    rgb, alpha, source_center = sprite
+    tx = float(target_center[0] - source_center[0])
+    ty = float(target_center[1] - source_center[1])
+    matrix = np.array([[1.0, 0.0, tx], [0.0, 1.0, ty]], dtype=np.float32)
+    warped_rgb = cv2.warpAffine(
+        rgb,
+        matrix,
+        (WIDTH, HEIGHT),
+        flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=0,
+    )
+    warped_alpha = cv2.warpAffine(
+        alpha,
+        matrix,
+        (WIDTH, HEIGHT),
+        flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=0,
+    )[..., None]
+    canvas[:] = canvas * (1.0 - warped_alpha) + warped_rgb * warped_alpha
+
+
+def simulate_paths() -> dict[int, tuple[np.ndarray, np.ndarray]]:
+    """Return ball centers for every post-contact frame."""
+    paths: dict[int, tuple[np.ndarray, np.ndarray]] = {}
+
+    pink = PINK_CENTER.copy()
+    foot = FOOT_CENTER.copy()
+
+    # Ten-times mass means the football receives a small forward impulse. The
+    # lighter pink ball reverses its normal velocity and keeps most of its
+    # tangential motion, producing a visible rebound up and to the right.
+    pink_velocity = np.array([1.04, -0.31], dtype=np.float64)
+    foot_velocity = np.array([0.085, 0.052], dtype=np.float64)
+
+    pink_decel = 0.034
+    foot_decel = 0.006
+
+    for frame_index in range(BRANCH_FRAME, FRAME_COUNT):
+        speed = float(np.linalg.norm(pink_velocity))
+        if speed > pink_decel:
+            pink_velocity *= (speed - pink_decel) / speed
+        else:
+            pink_velocity[:] = 0.0
+        pink += pink_velocity
+
+        foot_speed = float(np.linalg.norm(foot_velocity))
+        if foot_speed > foot_decel:
+            foot_velocity *= (foot_speed - foot_decel) / foot_speed
+        else:
+            foot_velocity[:] = 0.0
+        foot += foot_velocity
+
+        paths[frame_index] = (pink.copy(), foot.copy())
+
+    return paths
+
+
+def render(source: np.ndarray) -> np.ndarray:
+    clean_plate = make_clean_plate(source)
+    foot_sprite = extract_sprite(source[-1], *FOOT_TEMPLATE)
+    pink_sprite = extract_sprite(source[-1], *PINK_TEMPLATE)
+    paths = simulate_paths()
+
+    frames = source.copy()
+    for frame_index in range(BRANCH_FRAME, FRAME_COUNT):
+        pink_center, foot_center = paths[frame_index]
+        canvas = clean_plate.astype(np.float32)
+
+        # Farther objects are composited first; the football sits lower in the
+        # image and therefore remains in front at their brief contact.
+        add_soft_shadow(
+            canvas,
+            (float(pink_center[0] + 1.5), float(pink_center[1] + 13.5)),
+            (9.5, 3.2),
+            0.11,
+        )
+        add_soft_shadow(
+            canvas,
+            (float(foot_center[0] + 1.8), float(foot_center[1] + 14.5)),
+            (11.0, 3.8),
+            0.13,
+        )
+        composite_sprite(canvas, pink_sprite, pink_center)
+        composite_sprite(canvas, foot_sprite, foot_center)
+        frames[frame_index] = np.clip(canvas + 0.5, 0, 255).astype(np.uint8)
+
+    return frames
+
+
+def prepare_output_dir(output_dir: Path) -> None:
+    output_dir.mkdir(parents=True, exist_ok=True)
+    workspace = Path(__file__).resolve().parent
+    for name in ("prompt.txt", "task_spec.json"):
+        source_path = workspace / name
+        destination = output_dir / name
+        if source_path.resolve() != destination.resolve():
+            shutil.copyfile(source_path, destination)
+
+

+    parser = argparse.ArgumentParser()
+    parser.add_argument(
+        "--output-dir",
+        type=Path,
+        default=Path(__file__).resolve().parent,
+        help="Directory for generated artifacts.",
+    )
+    args = parser.parse_args()
+

+    if source.shape != (FRAME_COUNT, HEIGHT, WIDTH, 3):
+        raise ValueError(f"Unexpected source shape: {source.shape}")
+

+    prepare_output_dir(args.output_dir)
+    artifact_io.save(frames, args.output_dir)
+
+    displacement = np.linalg.norm(
+        simulate_paths()[FRAME_COUNT - 1][1] - FOOT_CENTER
+    )
+    pink_displacement = np.linalg.norm(
+        simulate_paths()[FRAME_COUNT - 1][0] - PINK_CENTER
+    )
+    print(
+        f"Saved {FRAME_COUNT} frames to {args.output_dir} "
+        f"(foot displacement {displacement:.2f}px, "
+        f"pink displacement {pink_displacement:.2f}px)"
+    )
+
+
+if __name__ == "__main__":
+    main()
