+import numpy as np
+from PIL import Image, ImageDraw
+from scipy.ndimage import (

+    binary_fill_holes,
+    gaussian_filter,

+import artifact_io

+CASE_DIR = Path(__file__).resolve().parent
+SPEED_MULTIPLIER = 1.5
+FINAL_SOURCE_FRAME = 18

+def make_clean_plate(frames):
+    """Recover the static room by removing both positions of the moving ball."""
+    plate = np.median(frames[18:96].astype(np.float32), axis=0)
+
+    # The late median contains the resting ball. Frame 0 has the same room
+    # without it, so restore that small region from the untouched first frame.
+    plate[48:91, 95:132] = frames[0, 48:91, 95:132].astype(np.float32)
+    return np.clip(plate, 0, 255).astype(np.uint8)
+
+
+def ball_components(frame, plate):
+    difference = np.abs(frame.astype(np.int16) - plate.astype(np.int16)).max(axis=2)
+    height, width = difference.shape
+    yy, xx = np.indices(difference.shape)
+    candidates = (
+        (difference > 12)
+        & (xx >= 40)
+        & (xx < 138)
+        & (yy >= 45)
+        & (yy < height)
+    )
+    components, count = label(candidates)
+    mask = np.zeros_like(candidates)
+    boxes = []
+    for component_id in range(1, count + 1):
+        component = components == component_id
+        if component.sum() < 8:
+            continue
+        ys, xs = np.where(component)
+        boxes.append((int(xs.min()), int(xs.max()), int(ys.min()), int(ys.max())))
+        mask |= component
+
+    if not boxes:
+        raise RuntimeError("Could not locate the volleyball")
+
+    x0 = min(box[0] for box in boxes)
+    x1 = max(box[1] for box in boxes)
+    y0 = min(box[2] for box in boxes)
+    y1 = max(box[3] for box in boxes)
+    mask = binary_fill_holes(mask)
+    mask = binary_dilation(mask, iterations=1)
+    return mask, (x0, x1, y0, y1)
+
+
+def colored_ball_box(frame, box):
+    x0, x1, y0, y1 = box
+    roi = frame[y0 : y1 + 1, x0 : x1 + 1].astype(np.int16)
+    r, g, b = roi[..., 0], roi[..., 1], roi[..., 2]
+    yellow = (r > g + 18) & (g > b + 18) & (r > 80)
+    blue = (b > r + 12) & (b > g + 8) & (b > 45)
+    colored = yellow | blue
+    ys, xs = np.where(colored)
+    if len(xs) < 12:
+        return box
+    return (
+        int(xs.min() + x0),
+        int(xs.max() + x0),
+        int(ys.min() + y0),
+        int(ys.max() + y0),
+    )
+
+
+def build_sprite(frame, plate, source_index):
+    mask, mask_box = ball_components(frame, plate)
+    color_box = colored_ball_box(frame, mask_box)
+
+    pad = 3
+    height, width = frame.shape[:2]
+    x0 = max(0, mask_box[0] - pad)
+    x1 = min(width - 1, mask_box[1] + pad)
+    y0 = max(0, mask_box[2] - pad)
+    y1 = min(height - 1, mask_box[3] + pad)
+
+    alpha = mask[y0 : y1 + 1, x0 : x1 + 1].astype(np.float32)
+    alpha = gaussian_filter(alpha, sigma=0.45)
+    alpha = np.clip(alpha, 0.0, 1.0)
+    color = frame[y0 : y1 + 1, x0 : x1 + 1].astype(np.float32)
+
+    cx = (color_box[0] + color_box[1]) * 0.5
+    cy = (color_box[2] + color_box[3]) * 0.5
+    size = max(color_box[1] - color_box[0] + 1, color_box[3] - color_box[2] + 1)
+    return {
+        "color": color,
+        "alpha": alpha,
+        "origin": np.array([x0, y0], dtype=np.float32),
+        "center": np.array([cx, cy], dtype=np.float32),
+        "size": float(size),
+        "source_index": source_index,
+    }
+
+
+def place_sprite(canvas, sprite, center, size):
+    scale = size / max(sprite["size"], 1.0)
+    src_h, src_w = sprite["alpha"].shape
+    dst_w = max(1, int(round(src_w * scale)))
+    dst_h = max(1, int(round(src_h * scale)))
+
+    color_image = Image.fromarray(np.clip(sprite["color"], 0, 255).astype(np.uint8))
+    alpha_image = Image.fromarray(np.clip(sprite["alpha"] * 255.0, 0, 255).astype(np.uint8))
+    color_image = color_image.resize((dst_w, dst_h), Image.Resampling.BILINEAR)
+    alpha_image = alpha_image.resize((dst_w, dst_h), Image.Resampling.BILINEAR)
+
+    center_in_sprite = sprite["center"] - sprite["origin"]
+    dst_center = np.array(
+        [center_in_sprite[0] * scale, center_in_sprite[1] * scale], dtype=np.float32
+    )
+    left = int(round(center[0] - dst_center[0]))
+    top = int(round(center[1] - dst_center[1]))
+
+    canvas_h, canvas_w = canvas.shape[:2]
+    src_x0 = max(0, -left)
+    src_y0 = max(0, -top)
+    src_x1 = min(dst_w, canvas_w - left)
+    src_y1 = min(dst_h, canvas_h - top)
+    if src_x1 <= src_x0 or src_y1 <= src_y0:
+        return
+
+    dst_x0 = left + src_x0
+    dst_y0 = top + src_y0
+    dst_x1 = left + src_x1
+    dst_y1 = top + src_y1
+
+    patch = np.asarray(color_image)[src_y0:src_y1, src_x0:src_x1].astype(np.float32)
+    alpha = np.asarray(alpha_image)[src_y0:src_y1, src_x0:src_x1].astype(np.float32) / 255.0
+    alpha = alpha[..., None]
+    target = canvas[dst_y0:dst_y1, dst_x0:dst_x1].astype(np.float32)
+    canvas[dst_y0:dst_y1, dst_x0:dst_x1] = np.clip(
+        patch * alpha + target * (1.0 - alpha), 0, 255
+    ).astype(np.uint8)
+
+
+def save_debug_sheet(sprites, path):
+    tiles = []
+    for sprite in sprites:
+        color = np.clip(sprite["color"], 0, 255).astype(np.uint8)
+        alpha = np.clip(sprite["alpha"] * 255.0, 0, 255).astype(np.uint8)
+        rgba = np.dstack([color, alpha])
+        tiles.append(Image.fromarray(rgba, mode="RGBA"))
+
+    max_w = max(tile.width for tile in tiles)
+    max_h = max(tile.height for tile in tiles)
+    columns = 5
+    rows = int(np.ceil(len(tiles) / columns))
+    sheet = Image.new("RGB", (columns * (max_w + 8), rows * (max_h + 20)), (20, 20, 20))
+    draw = ImageDraw.Draw(sheet)
+    for index, tile in enumerate(tiles):
+        x = (index % columns) * (max_w + 8)
+        y = (index // columns) * (max_h + 20)
+        draw.text((x + 3, y + 2), f"source {sprites[index]['source_index']}", fill="white")
+        sheet.paste(tile, (x, y + 18), tile)
+    sheet.save(path)
+
+
+def main():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    plate = make_clean_plate(source)
+
+    sprites = []
+    for source_index in range(FINAL_SOURCE_FRAME + 1):
+        sprites.append(build_sprite(source[source_index], plate, source_index))
+
+    centers = np.array([sprite["center"] for sprite in sprites], dtype=np.float32)
+    sizes = np.array([sprite["size"] for sprite in sprites], dtype=np.float32)
+
+    frames = np.empty_like(source)
+    for frame_index in range(len(source)):
+        if frame_index == 0:
+            frames[frame_index] = source[0]
+            continue
+
+        source_time = min(frame_index * SPEED_MULTIPLIER, float(FINAL_SOURCE_FRAME))
+        lower = int(np.floor(source_time))
+        upper = min(lower + 1, FINAL_SOURCE_FRAME)
+        fraction = source_time - lower
+
+        center = centers[lower] * (1.0 - fraction) + centers[upper] * fraction
+        size = sizes[lower] * (1.0 - fraction) + sizes[upper] * fraction
+        sprite_index = int(round(source_time))
+        sprite = sprites[sprite_index]
+
+        canvas = plate.copy()
+        place_sprite(canvas, sprite, center, size)
+        frames[frame_index] = canvas
+
+    (CASE_DIR / ".tmp").mkdir(exist_ok=True)
+    save_debug_sheet(sprites, CASE_DIR / ".tmp" / "ball_sprites.png")
+    artifact_io.save(frames, CASE_DIR)
+    print(f"Saved {len(frames)} frames with {SPEED_MULTIPLIER}x initial ball speed")
+
+
+if __name__ == "__main__":
+    main()
