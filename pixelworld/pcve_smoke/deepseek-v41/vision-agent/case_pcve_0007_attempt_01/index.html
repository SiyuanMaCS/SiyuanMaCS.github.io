+def track_mallet(frames):
+    """Return red-cap centers and the mallet's full union footprint."""
+    centers = []
+    red_masks = []
+    for frame in frames:
+        a = frame.astype(np.int16)
+        red = (
+            (a[:, :, 0] - np.maximum(a[:, :, 1], a[:, :, 2]) > 28)
+            & (a[:, :, 0] > 115)
+        )
+        red[:58] = False
+        red[93:] = False
+        red[:, :98] = False
+        red[:, 175:] = False
+        red = ndimage.binary_closing(red, iterations=1)
+        labels, count = ndimage.label(red)
+        if count:
+            sizes = ndimage.sum(red, labels, range(1, count + 1))
+            red = labels == (1 + int(np.argmax(sizes)))
+        ys, xs = np.nonzero(red)
+        centers.append((float(xs.mean()), float(ys.mean())))
+        red_masks.append(red)
+
+    # The source shows the same rigid mallet throughout. Its broad white body
+    # has a stable offset from the red cap, unlike the separate stationary puck.
+    cap_x, cap_y = centers[0]
+    body = np.zeros((144, 256), np.float32)
+    yy, xx = np.mgrid[:144, :256]
+    for frame in frames[:16]:
+        a = frame.astype(np.int16)
+        r, g, b = a[:, :, 0], a[:, :, 1], a[:, :, 2]
+        neutral = (
+            (np.maximum.reduce((r, g, b)) - np.minimum.reduce((r, g, b)) < 20)
+            & (r > 185)
+        )
+        dark = np.maximum.reduce((r, g, b)) < 185
+        moving = (neutral | dark) & (xx < 123) & (yy > 72) & (yy < 87)
+        body += moving
+    body = body >= 6
+    body = ndimage.binary_dilation(body, iterations=1)
+    body = ndimage.binary_closing(body, iterations=2)
+
+    masks = []
+    for i, (cx, cy) in enumerate(centers):
+        dx = int(round(cx - cap_x))
+        dy = int(round(cy - cap_y))
+        shifted = np.roll(body, (dy, dx), axis=(0, 1))
+        # The stationary white puck is not part of the mallet. Its footprint
+        # overlaps the mallet body at the end, so remove it after the shift.
+        if cx < 121.0:
+            shifted[73:87, 99:121] = False
+        mask = shifted | red_masks[i]
+        masks.append(ndimage.binary_dilation(mask, iterations=1))
+    return centers, masks
+
+
+def make_clean_plate(frames, centers):
+    """Build a robust clean plate from samples where the mallet is elsewhere."""
+    height, width = frames.shape[1:3]
+    plate = np.empty((height, width, 3), np.float32)
+    for y in range(47, 105):
+        for x in range(88, 188):
+            keep = []
+            for i, (cx, cy) in enumerate(centers):
+                if (x - cx) ** 2 + (y - cy) ** 2 > 25.0**2:
+                    keep.append(frames[i, y, x])
+            if not keep:
+                keep = list(frames[:, y, x])
+            plate[y, x] = np.median(np.stack(keep), axis=0)
+
+    temporal = np.median(frames, axis=0)
+    plate[:47] = temporal[:47]
+    plate[105:] = temporal[105:]
+    plate[:, :88] = temporal[:, :88]
+    plate[:, 188:] = temporal[:, 188:]
+
+    # Remove isolated temporal noise in the synthesized plate.
+    for channel in range(3):
+        plate[..., channel] = ndimage.gaussian_filter(plate[..., channel], 0.28)
+    return plate
+
+
+def remove_mallet(frames):
+    f = frames.astype(np.float32)
+    centers, masks = track_mallet(frames)
+    plate = make_clean_plate(f, centers)
+    out = f.copy()
+    for i, mask in enumerate(masks):
+        alpha = ndimage.gaussian_filter(mask.astype(np.float32), 0.65)
+        alpha = np.clip((alpha - 0.02) / 0.96, 0.0, 1.0)[..., None]
+        # The playfield is slightly mottled. Preserve its low-frequency
+        # illumination while replacing the occluded object and contact shadow.
+        local_plate = plate.copy()
+        out[i] = out[i] * (1.0 - alpha) + local_plate * alpha
+    return np.clip(np.rint(out), 0, 255).astype(np.uint8)
+
+
+def main():
+    source = np.load(ROOT / "source_pixels.npz")["rgb"]
+    result = remove_mallet(source)
+    assert result.shape == (96, 144, 256, 3)
+    assert result.dtype == np.uint8
+    artifact_io.save(result, ROOT)
+
+
+if __name__ == "__main__":
+    main()
