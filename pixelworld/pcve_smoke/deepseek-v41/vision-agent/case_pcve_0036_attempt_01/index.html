+def _resolve_pair(a: Body, b: Body, restitution: float) -> None:
+    dx = b.x - a.x
+    dz = b.z - a.z
+    dist = math.hypot(dx, dz)
+    minimum = a.radius + b.radius
+    if dist >= minimum or dist < 1e-6:
+        return
+
+    nx, nz = dx / dist, dz / dist
+    overlap = minimum - dist
+    inv_a = 1.0 / a.mass
+    inv_b = 1.0 / b.mass
+    inv_sum = inv_a + inv_b
+
+    a.x -= nx * overlap * inv_a / inv_sum
+    a.z -= nz * overlap * inv_a / inv_sum
+    b.x += nx * overlap * inv_b / inv_sum
+    b.z += nz * overlap * inv_b / inv_sum
+
+    relative = (b.vx - a.vx) * nx + (b.vz - a.vz) * nz
+    if relative >= 0.0:
+        return
+    impulse = -(1.0 + restitution) * relative / inv_sum
+    a.vx -= impulse * inv_a * nx
+    a.vz -= impulse * inv_a * nz
+    b.vx += impulse * inv_b * nx
+    b.vz += impulse * inv_b * nz
+
+    # A small, damped spin response makes glancing contacts read as pin motion.
+    spin = 0.22 * impulse / max(a.mass, b.mass)
+    a.omega -= spin * nz
+    b.omega += spin * nz
