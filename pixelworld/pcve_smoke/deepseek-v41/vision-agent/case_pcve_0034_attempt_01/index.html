+def _blend_rgba(dst: np.ndarray, sprite: np.ndarray, cx: float, cy: float, scale: float) -> None:
+    """Composite a premultiplied-alpha sprite around a subpixel center."""
+    sh, sw = sprite.shape[:2]
+    out_w = max(1, int(round(sw * scale)))
+    out_h = max(1, int(round(sh * scale)))
+    resized = np.asarray(
+        Image.fromarray(sprite, mode="RGBA").resize(
+            (out_w, out_h), Image.Resampling.BICUBIC
+        ),
+        dtype=np.float32,
+    )
+    alpha = resized[:, :, 3:4] / 255.0
+    color = resized[:, :, :3]
+
+    left = int(np.floor(cx - out_w / 2.0))
+    top = int(np.floor(cy - out_h / 2.0))
+    x0 = max(0, left)
+    y0 = max(0, top)
+    x1 = min(WIDTH, left + out_w)
+    y1 = min(HEIGHT, top + out_h)
+    if x0 >= x1 or y0 >= y1:
+        return
+
+    sx0 = x0 - left
+    sy0 = y0 - top
+    sx1 = sx0 + (x1 - x0)
+    sy1 = sy0 + (y1 - y0)
+    a = alpha[sy0:sy1, sx0:sx1]
+    src = color[sy0:sy1, sx0:sx1]
+    dst[y0:y1, x0:x1] = dst[y0:y1, x0:x1] * (1.0 - a) + src * a
+
+
+def _build_ball_sprite(source: np.ndarray, plate: np.ndarray) -> np.ndarray:
+    """Recover the ball, its reflected light, and contact shadow from frame 1."""
+    frame = source[0].astype(np.float32)
+    diff = np.linalg.norm(frame - plate.astype(np.float32), axis=2)
+    x0, y0, x1, y1 = 118, 94, 139, 114
+    patch = frame[y0:y1, x0:x1]
+    delta = diff[y0:y1, x0:x1]
+
+    # The low threshold retains the soft reflection under the ball. The spatial
+    # lobe rejects unrelated floor texture while leaving the reflection intact.
+    yy, xx = np.mgrid[y0:y1, x0:x1]
+    ball_lobe = ((xx - 127.5) / 5.8) ** 2 + ((yy - 102.0) / 5.8) ** 2 <= 1.0
+    reflection_lobe = (
+        ((xx - 127.5) / 4.3) ** 2 + ((yy - 108.7) / 6.5) ** 2 <= 1.0
+    )
+    lobe = ball_lobe | reflection_lobe
+    alpha = np.clip((delta - 5.0) / 18.0, 0.0, 1.0) * lobe
+    alpha = np.where(alpha < 0.08, 0.0, alpha)
+
+    sprite = np.dstack([patch, alpha * 255.0]).astype(np.uint8)
+    return sprite
+
+
+def _source_ball_centers(source: np.ndarray, plate: np.ndarray) -> list[float]:
+    """Track the blue ball in the factual approach frames."""
+    centers: list[float] = []
+    previous = 102.0
+    for index in range(CONTACT_FRAME):
+        frame = source[index].astype(np.float32)
+        diff = np.linalg.norm(frame - plate.astype(np.float32), axis=2)
+        yy, xx = np.mgrid[0:HEIGHT, 0:WIDTH]
+        lane = (
+            (xx >= 122)
+            & (xx <= 133)
+            & (yy >= max(73, int(previous - 8)))
+            & (yy <= min(112, int(previous + 8)))
+            & (diff > 35)
+        )
+        ys, xs = np.where(lane)
+        if len(ys) >= 4:
+            # The brightest central pixels give the sphere center without the
+            # faint downward reflection biasing it toward the camera.
+            weights = diff[ys, xs]
+            keep = weights >= np.percentile(weights, 55)
+            previous = float(np.average(ys[keep], weights=weights[keep]))
+        centers.append(previous)
+    return centers
+
+
+def _rebound_center(frame_index: int, approach_centers: list[float]) -> float:
+    if frame_index <= CONTACT_FRAME:
+        return approach_centers[min(frame_index, CONTACT_FRAME - 1)]
+
+    # Elastic collision with a rack 25 times heavier:
+    # v_ball_after = (1 - 25) / (1 + 25) * v_ball_before.
+    rebound_speed = 24.0 / 26.0
+    source_position = (CONTACT_FRAME - 1) - rebound_speed * (
+        frame_index - CONTACT_FRAME
+    )
+    if source_position >= 0:
+        lo = int(np.floor(source_position))
+        hi = min(lo + 1, CONTACT_FRAME - 1)
+        fraction = source_position - lo
+        return approach_centers[lo] * (1.0 - fraction) + approach_centers[hi] * fraction
+
+    # Extrapolate the near-camera part of the perspective track from its first
+    # few observed samples. The ball then exits the bottom of frame naturally.
+    samples = np.asarray(approach_centers[:5], dtype=np.float64)
+    slope = float(np.polyfit(np.arange(5), samples, 1)[0])
+    return approach_centers[0] + (-source_position) * max(slope, 1.25)
+
+
+def _ball_radius(center_y: float) -> float:
+    # Perspective radius calibrated from the source ball at the near and far
+    # ends of the approach.
+    horizon_y = 54.0
+    near_y = 102.0
+    near_radius = 4.25
+    return near_radius * (center_y - horizon_y) / (near_y - horizon_y)
+
+
+def build_frames() -> np.ndarray:
+    source = np.load(ROOT / "source_pixels.npz")["rgb"]
+    plate = np.median(source[:CONTACT_FRAME], axis=0).astype(np.uint8)
+    ball_sprite = _build_ball_sprite(source, plate)
+    approach_centers = _source_ball_centers(source, plate)
+
+    frames = np.empty((FRAME_COUNT, HEIGHT, WIDTH, 3), dtype=np.uint8)
+    for index in range(FRAME_COUNT):
+        if index < CONTACT_FRAME:
+            frames[index] = source[index]
+            continue
+
+        frame = plate.astype(np.float32)
+        center_y = _rebound_center(index, approach_centers)
+        center_x = 127.5
+        scale = max(0.43, _ball_radius(center_y) / 4.25)
+        _blend_rgba(frame, ball_sprite, center_x, center_y, scale)
+        frames[index] = np.clip(frame + 0.5, 0, 255).astype(np.uint8)
+
+    return frames
+
