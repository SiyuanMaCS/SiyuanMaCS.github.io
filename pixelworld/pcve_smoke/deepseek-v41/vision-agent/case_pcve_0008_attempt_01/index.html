+def _surface_fit(frame):
+    """Fit the smooth lavender tabletop around the white-mallet footprint."""
+    y0, y1 = 68, 94
+    x0, x1 = 84, 126
+    yy, xx = np.mgrid[y0:y1, x0:x1]
+    values = frame[y0:y1, x0:x1].astype(np.float64)
+
+    repair = (xx >= 92) & (xx <= 117) & (yy >= 73) & (yy <= 88)
+    printed_line = (xx >= 116) & (yy <= 80)
+    clean = ~(repair | printed_line)
+
+    terms = [
+        np.ones_like(xx),
+        xx,
+        yy,
+        xx * yy,
+        xx**2,
+        yy**2,
+        xx**2 * yy,
+        xx * yy**2,
+    ]
+    design = np.stack([term[clean] for term in terms], axis=1)
+    full_design = np.stack([term.ravel() for term in terms], axis=1)
+
+    surface = np.empty_like(values)
+    for channel in range(3):
+        coeffs = np.linalg.lstsq(design, values[:, :, channel][clean], rcond=None)[0]
+        surface[:, :, channel] = (full_design @ coeffs).reshape(xx.shape)
+    return surface, (y0, y1, x0, x1)

+def _grain_texture(frame, y0, y1, x0, x1):
+    """Extract subtle table grain from a clean donor patch."""
+    donor = frame[88:101, 93:116].astype(np.float32)
+    smooth = cv2.GaussianBlur(donor, (0, 0), 2.2)
+    grain = donor - smooth
+
+    target_h, target_w = y1 - y0, x1 - x0
+    reps_y = int(np.ceil(target_h / grain.shape[0]))
+    reps_x = int(np.ceil(target_w / grain.shape[1]))
+    grain = np.tile(grain, (reps_y, reps_x, 1))[:target_h, :target_w]
+    return grain
+
+
+def _red_protection(frame):
+    """Cover the red mallet, its highlights, and its soft contact shadow."""
+    rgb = frame.astype(np.int16)
+    r, g, b = rgb[:, :, 0], rgb[:, :, 1], rgb[:, :, 2]
+    red_like = (r > g + 8) & (r > b - 4)
+    red_like &= (r > 105)
+    kernel = np.ones((7, 7), np.uint8)
+    return cv2.dilate(red_like.astype(np.uint8), kernel, iterations=1).astype(bool)
+

+def remove_white_mallet(frames):
+    frames = frames.copy()
+    surface, (sy0, sy1, sx0, sx1) = _surface_fit(frames[0])
+
+    yy, xx = np.mgrid[0 : frames.shape[1], 0 : frames.shape[2]]
+    base_repair = (xx >= 93) & (xx <= 116) & (yy >= 74) & (yy <= 87)
+
+    # Rounded corners and a soft edge avoid a visible rectangular patch.
+    rounded = (
+        ((xx - 104.5) / 11.5) ** 2 + ((yy - 80.5) / 6.7) ** 2 <= 1.0
+    )
+    repair_core = base_repair & rounded
+    alpha_hard = cv2.GaussianBlur(repair_core.astype(np.float32), (0, 0), 1.15)
+    alpha_hard = np.clip(alpha_hard * 1.45, 0.0, 1.0)
+
+    grain_full = np.zeros_like(frames[0], dtype=np.float32)
+    grain = _grain_texture(frames[0], sy0, sy1, sx0, sx1)
+    grain_full[sy0:sy1, sx0:sx1] = grain
+
+    background = np.zeros_like(frames[0], dtype=np.float32)
+    background[sy0:sy1, sx0:sx1] = surface
+    background = background + 0.55 * grain_full
+
+    for index in range(37):

+        alpha = np.clip(alpha, 0.0, 1.0)[:, :, None]
+
+        blended = frame.astype(np.float32) * (1.0 - alpha) + background * alpha
+        frames[index] = np.clip(blended + 0.5, 0, 255).astype(np.uint8)
+
+    return frames

+    artifact_io.save(result, CASE_DIR)

+    main()
