+import cv2
+import numpy as np

+CASE_DIR = Path(__file__).resolve().parent
+FPS = 24.0
+N_FRAMES = 96
+ONSET_FRAME = 23
+
+# The source roll ends at frame 44.  With five times the rolling friction and
+# the same launch speed, stopping time scales by 1/sqrt(5) and distance by 1/5.
+SOURCE_STOP_FRAME = 44.0
+FRICTION_MULTIPLIER = 5.0
+
+# Source-space measurements at the start and end of the roll.
+START_CENTER = np.array([101.40, 93.10], dtype=np.float64)
+START_RADIUS = 14.40
+SOURCE_STOP_CENTER = np.array([104.60, 113.90], dtype=np.float64)
+SOURCE_STOP_RADIUS = 19.10
+
+
+def make_clean_plate(frame: np.ndarray) -> np.ndarray:
+    """Remove the soccer ball from the opening frame."""
+    mask = np.zeros(frame.shape[:2], dtype=np.uint8)
+    cv2.circle(mask, tuple(np.rint(START_CENTER).astype(int)), 18, 255, -1)
+    cv2.ellipse(mask, (98, 104), (17, 7), 0, 0, 360, 255, -1)
+    return cv2.inpaint(frame, mask, 5, cv2.INPAINT_TELEA)
+
+
+def extract_ball_sprite(frames: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
+    """Extract a sharp ball crop from the source roll."""
+    source_index = 32
+    center = np.array([103.10, 106.40])
+    radius = 17.40
+    pad = 2
+    x0 = int(np.floor(center[0] - radius - pad))
+    y0 = int(np.floor(center[1] - radius - pad))
+    x1 = int(np.ceil(center[0] + radius + pad)) + 1
+    y1 = int(np.ceil(center[1] + radius + pad)) + 1
+    crop = frames[source_index, y0:y1, x0:x1].copy()
+
+    yy, xx = np.indices(crop.shape[:2], dtype=np.float32)
+    cx = center[0] - x0
+    cy = center[1] - y0
+    distance = np.hypot(xx - cx, yy - cy)
+    alpha = np.clip(radius + 0.55 - distance, 0.0, 1.0)
+    alpha = cv2.GaussianBlur(alpha, (3, 3), 0.55)
+    alpha = np.clip(alpha, 0.0, 1.0)
+    return crop, alpha
+
+
+def target_state(frame_index: int) -> tuple[np.ndarray, float, float]:
+    """Return screen center, radius, and normalized travel."""
+    if frame_index < ONSET_FRAME:
+        return START_CENTER.copy(), START_RADIUS, 0.0
+
+    duration = (SOURCE_STOP_FRAME - ONSET_FRAME) / np.sqrt(FRICTION_MULTIPLIER)
+    u = min(1.0, (frame_index - ONSET_FRAME) / duration)
+    travel = 2.0 * u - u * u
+
+    p_start = 1.0 / START_RADIUS
+    p_stop = 1.0 / SOURCE_STOP_RADIUS
+    p = p_start + (p_stop - p_start) * travel / FRICTION_MULTIPLIER
+    radius = 1.0 / p
+
+    fraction = (p - p_start) / (p_stop - p_start)
+    center = START_CENTER + (SOURCE_STOP_CENTER - START_CENTER) * fraction
+    return center, float(radius), float(travel)
+
+
+def add_contact_shadow(
+    canvas: np.ndarray, center: np.ndarray, radius: float, travel: float
+) -> None:
+    """Composite a soft shadow underneath the rolling ball."""
+    height, width = canvas.shape[:2]
+    sx = center[0] - 0.8
+    sy = center[1] + radius * 0.88
+    axes = (max(5.0, radius * 0.82), max(1.8, radius * 0.19))
+    strength = 0.24 - 0.05 * travel
+
+    pad_x = int(np.ceil(axes[0] * 2.2))
+    pad_y = int(np.ceil(axes[1] * 2.8))
+    x0, x1 = max(0, int(sx) - pad_x), min(width, int(sx) + pad_x + 1)
+    y0, y1 = max(0, int(sy) - pad_y), min(height, int(sy) + pad_y + 1)
+    if x0 >= x1 or y0 >= y1:
+        return
+
+    yy, xx = np.indices((y1 - y0, x1 - x0), dtype=np.float32)
+    nx = (xx + x0 - sx) / axes[0]
+    ny = (yy + y0 - sy) / axes[1]
+    field = np.clip(1.0 - (nx * nx + ny * ny), 0.0, 1.0)
+    shadow_alpha = strength * field**1.6
+    patch = canvas[y0:y1, x0:x1].astype(np.float32)
+    tint = np.array([0.58, 0.48, 0.42], dtype=np.float32)
+    shaded = patch * tint[None, None, :]
+    canvas[y0:y1, x0:x1] = np.clip(
+        patch * (1.0 - shadow_alpha[..., None])
+        + shaded * shadow_alpha[..., None],
+        0,
+        255,
+    ).astype(np.uint8)
+
+
+def composite_sprite(
+    canvas: np.ndarray,
+    sprite: np.ndarray,
+    alpha: np.ndarray,
+    center: np.ndarray,
+    radius: float,
+) -> None:
+    """Place the source ball crop at a new screen position and scale."""
+    source_radius = 17.40
+    scale = radius / source_radius
+    size = int(round(sprite.shape[1] * scale))
+    size = max(size, 7)
+    sprite_scaled = cv2.resize(
+        sprite, (size, size), interpolation=cv2.INTER_LANCZOS4
+    )
+    alpha_scaled = cv2.resize(
+        alpha, (size, size), interpolation=cv2.INTER_LINEAR
+    )
+    alpha_scaled = np.clip(alpha_scaled, 0.0, 1.0)
+
+    left = int(round(center[0] - size / 2.0))
+    top = int(round(center[1] - size / 2.0))
+    x0, y0 = max(0, left), max(0, top)
+    x1 = min(canvas.shape[1], left + size)
+    y1 = min(canvas.shape[0], top + size)
+    sx0, sy0 = x0 - left, y0 - top
+    sx1, sy1 = sx0 + (x1 - x0), sy0 + (y1 - y0)
+
+    dst = canvas[y0:y1, x0:x1].astype(np.float32)
+    src = sprite_scaled[sy0:sy1, sx0:sx1].astype(np.float32)
+    a = alpha_scaled[sy0:sy1, sx0:sx1, None]
+    canvas[y0:y1, x0:x1] = np.clip(
+        src * a + dst * (1.0 - a), 0, 255
+    ).astype(np.uint8)
+
+
+def build_video() -> np.ndarray:
+    with np.load(CASE_DIR / "source_pixels.npz") as archive:
+        source = archive["rgb"]
+
+    assert source.shape == (N_FRAMES, 144, 256, 3), source.shape
+    output = source.copy()
+    plate = make_clean_plate(source[0])
+    sprite, sprite_alpha = extract_ball_sprite(source)
+
+    for frame_index in range(ONSET_FRAME, N_FRAMES):
+        canvas = plate.copy()
+        center, radius, travel = target_state(frame_index)
+        add_contact_shadow(canvas, center, radius, travel)
+        composite_sprite(canvas, sprite, sprite_alpha, center, radius)
+        output[frame_index] = canvas
+
+    return output
+
+
