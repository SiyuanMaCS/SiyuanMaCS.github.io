+
+import cv2
+import numpy as np
+from PIL import Image
+from scipy import ndimage
+
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+FRAMES = 96
+WIDTH = 256
+HEIGHT = 144
+FPS = 24
+
+# Fitted from the original car centers over the first 14 frames.
+P0 = np.array([171.5, 110.0], dtype=np.float64)
+RAMP_DIR = np.array([-0.9440627, -0.3297660], dtype=np.float64)
+V0_SOURCE = 15.433
+ACCEL_SOURCE = -0.9282
+SPEED_SCALE = 0.6
+
+# The car center reaches this point at the lower end of the ramp.
+RAMP_RELEASE_S = -24.5
+
+
+def clean_background(source):
+    background = np.median(source, axis=0)
+
+    # The factual car settles at the lower left after frame 20. Reconstruct
+    # that one region from the early frames, where the pavement is unobscured.
+    clear_left = np.median(source[:9, 100:136, :52], axis=0)
+    background[100:136, :52] = clear_left
+    return background.astype(np.uint8)
+
+
+def extract_sprite(source, background, frame_index, center, radius):
+    cx, cy = center
+    rx, ry = radius
+    x0 = max(0, int(np.floor(cx - rx)))
+    x1 = min(WIDTH, int(np.ceil(cx + rx + 1)))
+    y0 = max(0, int(np.floor(cy - ry)))
+    y1 = min(HEIGHT, int(np.ceil(cy + ry + 1)))
+
+    pixels = source[frame_index, y0:y1, x0:x1].astype(np.float32)
+    plate = background[y0:y1, x0:x1].astype(np.float32)
+    difference = np.max(np.abs(pixels - plate), axis=2)
+
+    seed = difference > 6.0
+    labels, count = ndimage.label(seed, structure=np.ones((3, 3), dtype=np.uint8))
+    if count == 0:
+        raise RuntimeError(f"No foreground found for sprite in frame {frame_index}")
+
+    local_y, local_x = np.indices(seed.shape)
+    distance = (local_x - (cx - x0)) ** 2 + (local_y - (cy - y0)) ** 2
+    best_label = 0
+    best_score = float("inf")
+    for label in range(1, count + 1):
+        component = labels == label
+        area = int(component.sum())
+        if area < 8:
+            continue
+        score = float(distance[component].mean()) - area * 0.05
+        if score < best_score:
+            best_score = score
+            best_label = label
+
+    if best_label == 0:
+        raise RuntimeError(f"Could not isolate sprite in frame {frame_index}")
+
+    mask = labels == best_label
+    mask = ndimage.binary_closing(mask, structure=np.ones((2, 2), dtype=bool))
+    mask = ndimage.binary_fill_holes(mask)
+
+    # Preserve antialiased edges and the soft cast shadow while excluding
+    # isolated compression noise in the plate.
+    alpha = ndimage.gaussian_filter(mask.astype(np.float32), 0.72)
+    alpha *= np.clip((difference - 1.5) / 7.0, 0.0, 1.0)
+    alpha = np.clip(alpha * 1.35, 0.0, 1.0)
+    alpha[mask & (difference > 9.0)] = 1.0
+
+    return {
+        "rgb": source[frame_index, y0:y1, x0:x1].astype(np.float32),
+        "alpha": alpha,
+        "anchor": np.array([cx - x0, cy - y0], dtype=np.float64),
+    }
+
+
+def warp_sprite(sprite, center):
+    sx = float(center[0] - sprite["anchor"][0])
+    sy = float(center[1] - sprite["anchor"][1])
+    matrix = np.array([[1.0, 0.0, sx], [0.0, 1.0, sy]], dtype=np.float32)
+
+    rgb = sprite["rgb"]
+    alpha = sprite["alpha"]
+    premultiplied = np.dstack([rgb * alpha[:, :, None], alpha * 255.0])
+    warped = cv2.warpAffine(
+        premultiplied,
+        matrix,
+        (WIDTH, HEIGHT),
+        flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=0,
+    )
+    warped_rgb = warped[:, :, :3]
+    warped_alpha = np.clip(warped[:, :, 3] / 255.0, 0.0, 1.0)
+    return warped_rgb, warped_alpha
+
+
+def composite(background, sprite, center):
+    rgb, alpha = warp_sprite(sprite, center)
+    plate = background.astype(np.float32)
+    output = plate * (1.0 - alpha[:, :, None]) + rgb
+    return np.clip(output, 0.0, 255.0).astype(np.uint8)
+
+
+def release_time():
+    a = 0.5 * ACCEL_SOURCE
+    b = V0_SOURCE * SPEED_SCALE
+    roots = np.roots([a, b, -RAMP_RELEASE_S])
+    return float(np.max(roots))
+
+
+def render():
+    archive = np.load(ROOT / "source_pixels.npz")
+    source = archive["rgb"]
+    background = clean_background(source)
+
+    ramp_sprite = extract_sprite(
+        source,
+        background,
+        frame_index=0,
+        center=(171.5, 110.0),
+        radius=(23.0, 15.0),
+    )
+    flat_sprite = extract_sprite(
+        source,
+        background,
+        frame_index=20,
+        center=(26.5, 120.0),
+        radius=(27.0, 16.0),
+    )
+
+    t_release = release_time()
+    p_release = P0 + RAMP_DIR * RAMP_RELEASE_S
+    release_speed = V0_SOURCE * SPEED_SCALE + ACCEL_SOURCE * t_release
+    flat_velocity = RAMP_DIR * release_speed
+    flat_velocity[1] = 0.0
+
+    frames = []
+    for frame_index in range(FRAMES):
+        if frame_index == 0:
+            # The intervention starts at frame 1 (one-based), so the initial
+            # source state remains exact.
+            frames.append(source[0].copy())
+            continue
+
+        s = (
+            V0_SOURCE * SPEED_SCALE * frame_index
+            + 0.5 * ACCEL_SOURCE * frame_index * frame_index
+        )
+
+        if frame_index <= t_release:
+            center = P0 + RAMP_DIR * s
+            output = composite(background, ramp_sprite, center)
+        else:
+            elapsed = frame_index - t_release
+            center = p_release + flat_velocity * elapsed
+            landing = min(1.0, elapsed / 1.6)
+            landing = landing * landing * (3.0 - 2.0 * landing)
+            center[1] = p_release[1] + (120.0 - p_release[1]) * landing
+            output = composite(background, flat_sprite, center)
+
+        frames.append(output)
+
+    return np.asarray(frames, dtype=np.uint8)
+
+
+def write_preview_video(frames):
+    writer = cv2.VideoWriter(
+        str(ROOT / "counterfactual.mp4"),
+        cv2.VideoWriter_fourcc(*"mp4v"),
+        FPS,
+        (WIDTH, HEIGHT),
+    )
+    if not writer.isOpened():
+        raise RuntimeError("Could not open MP4 writer")
+    for frame in frames:
+        writer.write(cv2.cvtColor(frame, cv2.COLOR_RGB2BGR))
+    writer.release()
+
+
+def main():
+    frames = render()
+    artifact_io.save(frames, ROOT)
+    write_preview_video(frames)
+    print(f"Saved {len(frames)} frames; ramp release at frame {release_time():.2f}")
+
+
+if __name__ == "__main__":
+    main()
