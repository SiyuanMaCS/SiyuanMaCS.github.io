+def _largest_ball_component(frame):
+    f = frame.astype(np.int16)
+    r, g, b = f[:, :, 0], f[:, :, 1], f[:, :, 2]
+    mask = (r > 100) & (r > g * 1.5) & (g > b * 1.05) & (r - b > 40)
+    mask[:5] = False
+    mask[105:] = False
+    mask[:, :100] = False
+    mask[:, 180:] = False
+
+    labels, count = label(mask)
+    best = None
+    for index in range(1, count + 1):
+        ys, xs = np.where(labels == index)
+        if len(xs) < 200:
+            continue
+        candidate = (len(xs), xs.min(), ys.min(), xs.max(), ys.max())
+        if best is None or candidate[0] > best[0]:
+            best = candidate
+    if best is None:
+        raise RuntimeError("Could not locate the basketball")
+
+    _, x0, y0, x1, y1 = best
+    return np.array([(x0 + x1) / 2.0, (y0 + y1) / 2.0], dtype=np.float64)
+

+def _clean_plate(frames):
+    return np.median(frames, axis=0).astype(np.uint8)
+

+def _ball_sprite(frame, center, plate, radius=11.8):
+    """Extract the source ball as premultiplied-over-alpha RGBA."""
+    cx, cy = int(round(center[0])), int(round(center[1]))
+    half = 15
+    yy, xx = np.mgrid[-half : half + 1, -half : half + 1]
+    dist = np.sqrt(xx * xx + yy * yy)
+    alpha = np.clip((radius + 0.75 - dist) / 1.5, 0.0, 1.0)
+
+    y0, y1 = cy - half, cy + half + 1
+    x0, x1 = cx - half, cx + half + 1
+    source = frame[y0:y1, x0:x1].astype(np.float64)
+    background = plate[y0:y1, x0:x1].astype(np.float64)
+
+    safe_alpha = np.maximum(alpha[:, :, None], 0.18)
+    color = (source - (1.0 - alpha[:, :, None]) * background) / safe_alpha
+    color = np.clip(color, 0.0, 255.0)
+
+    # A tiny amount of source edge color is less objectionable than a hard
+    # antialiased ring, so keep low-alpha pixels only where they are orange.
+    r, g, b = source[:, :, 0], source[:, :, 1], source[:, :, 2]
+    orange = (r > 58) & (r > g * 1.28) & (r - b > 22)
+    alpha *= np.where(orange, 1.0, 0.0)
+    return color, alpha
+

+def _shadow(base, x, y, floor_y):
+    height = max(0.0, floor_y - y)
+    strength = 0.56 * np.exp(-((height / 10.5) ** 2))
+    if strength < 0.002:
+        return
+
+    center_x = x + 1.0
+    center_y = 94.0
+    x0 = max(0, int(np.floor(center_x - 18)))
+    x1 = min(WIDTH, int(np.ceil(center_x + 19)))
+    y0 = max(0, int(np.floor(center_y - 8)))
+    y1 = min(HEIGHT, int(np.ceil(center_y + 9)))
+
+    yy, xx = np.mgrid[y0:y1, x0:x1]
+    dx = (xx - center_x) / 12.8
+    dy = (yy - center_y) / 5.2
+    falloff = np.exp(-0.5 * (dx * dx + dy * dy))
+    darkening = 1.0 - strength * falloff
+    base[y0:y1, x0:x1] = np.clip(
+        base[y0:y1, x0:x1].astype(np.float64) * darkening[:, :, None], 0, 255
+    ).astype(np.uint8)
+
+
+def _composite_ball(frame, color, alpha, center):
+    cx, cy = int(round(center[0])), int(round(center[1]))
+    half = color.shape[0] // 2
+    y0, y1 = cy - half, cy + half + 1
+    x0, x1 = cx - half, cx + half + 1
+    if y0 < 0 or x0 < 0 or y1 > HEIGHT or x1 > WIDTH:
+        return frame
+
+    a = alpha[:, :, None]
+    region = frame[y0:y1, x0:x1].astype(np.float64)
+    frame[y0:y1, x0:x1] = np.clip(
+        color * a + region * (1.0 - a), 0, 255
+    ).astype(np.uint8)
+    return frame
+

+def _trajectory(centers, gravity=0.72, restitution_scale=0.4):
+    frames = np.arange(FRAMES, dtype=np.float64)
+
+    incoming_y = np.polyfit(frames[:12], centers[:12, 1], 2)
+    incoming_x = np.polyfit(frames[:12], centers[:12, 0], 1)
+    outgoing_y = np.polyfit(frames[13:21], centers[13:21, 1], 2)
+    outgoing_x = np.polyfit(frames[13:70], centers[13:70, 0], 1)
+
+    contact_time = 12.0
+    floor_y = float(np.polyval(incoming_y, contact_time))
+    incoming_vy = float(np.polyval(np.polyder(incoming_y), contact_time))
+    source_out_vy = float(np.polyval(np.polyder(outgoing_y), contact_time))
+    source_restitution = abs(source_out_vy / incoming_vy)
+    restitution = restitution_scale * source_restitution
+
+    x_contact = float(np.polyval(incoming_x, contact_time))
+    vx = float(outgoing_x[0])
+
+    dt = 1.0 / 480.0
+    y = floor_y
+    vy = -restitution * incoming_vy
+    edited_y = np.empty(FRAMES, dtype=np.float64)
+    edited_y[:13] = centers[:13, 1]
+    edited_y[12] = floor_y
+
+    t = contact_time
+    next_sample = 13
+    while next_sample < FRAMES:
+        t += dt
+        vy += gravity * dt
+        y += vy * dt
+        if y >= floor_y:
+            y = floor_y - (y - floor_y) * restitution
+            vy = -vy * restitution
+            if abs(vy) < 0.18:
+                vy = 0.0
+                y = floor_y
+        while next_sample < FRAMES and t >= next_sample - 1e-9:
+            edited_y[next_sample] = y
+            next_sample += 1
+
+    edited_x = np.empty(FRAMES, dtype=np.float64)
+    edited_x[:13] = centers[:13, 0]
+    edited_x[12] = x_contact
+    edited_x[13:] = x_contact + vx * (frames[13:] - contact_time)
+    return edited_x, edited_y, restitution, source_restitution, floor_y
+
+
+def main():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (FRAMES, HEIGHT, WIDTH, 3)
+
+    centers = np.array([_largest_ball_component(frame) for frame in source])
+    plate = _clean_plate(source)
+    edited_x, edited_y, restitution, source_restitution, floor_y = _trajectory(centers)
+
+    output = np.empty_like(source)
+    output[:13] = source[:13]
+
+    for index in range(13, FRAMES):
+        frame = plate.copy()
+        _shadow(frame, edited_x[index], edited_y[index], floor_y)
+        color, alpha = _ball_sprite(
+            source[index], centers[index], plate
+        )
+        output[index] = _composite_ball(
+            frame, color, alpha, (edited_x[index], edited_y[index])
+        )
+
+    artifact_io.save(output, CASE_DIR)
+    print(
+        "source restitution",
+        round(source_restitution, 4),
+        "edited restitution",
+        round(restitution, 4),
+        "floor center y",
+        round(floor_y, 3),
+    )
+
+
+if __name__ == "__main__":
+    main()
