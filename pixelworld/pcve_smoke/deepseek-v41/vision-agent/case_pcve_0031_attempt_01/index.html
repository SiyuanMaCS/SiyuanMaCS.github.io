+
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+
+# Scene calibration from the source trajectory (pixels per source frame).
+START_X = 127.5
+START_Y = 22.0
+SOURCE_VX = 0.90
+SOURCE_VY = 0.46
+SPEED_MULTIPLIER = 6.0
+GRAVITY = 0.80
+FLOOR_Y = 86.5
+RESTITUTION = 0.72
+
+FPS = 24
+FRAME_COUNT = 96
+WIDTH = 256
+HEIGHT = 144
+SUPERSAMPLE = 3
+SHUTTER = 0.52
+MOTION_SAMPLES = 7
+
+
+def load_source():
+    frames = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    if frames.shape != (FRAME_COUNT, HEIGHT, WIDTH, 3):
+        raise ValueError(f"Unexpected source shape: {frames.shape}")
+    return frames
+
+
+def make_clean_plate(frames):
+    # The camera is static. A temporal median removes the moving ball while
+    # retaining the source image's grain and lighting.
+    return np.median(frames, axis=0).astype(np.uint8)
+
+
+def extract_ball_sprite(frames, background):
+    # Frame 1 contains the ball centered at (127.5, 21.5). The crop is kept
+    # symmetric about that point so later rotation does not shift the center.
+    x0, y0, size = 114, 8, 28
+    patch = frames[0, y0 : y0 + size, x0 : x0 + size].astype(np.float32)
+    plate = background[y0 : y0 + size, x0 : x0 + size].astype(np.float32)
+
+    yy, xx = np.indices((size, size), dtype=np.float32)
+    distance = np.hypot(xx - 13.5, yy - 13.5)
+    radial_alpha = np.clip((12.15 - distance) / 1.35, 0.0, 1.0)
+
+    color_distance = np.linalg.norm(patch - plate, axis=2)
+    color_alpha = np.clip((color_distance - 5.0) / 26.0, 0.0, 1.0)
+    alpha = radial_alpha * color_alpha
+
+    safe_alpha = np.maximum(alpha, 0.08)[..., None]
+    foreground = (patch - (1.0 - alpha[..., None]) * plate) / safe_alpha
+    foreground = np.clip(foreground, 0, 255)
+    foreground[alpha <= 0.0] = 0
+
+    rgba = np.dstack([foreground, alpha * 255.0]).round().astype(np.uint8)
+    return Image.fromarray(rgba, "RGBA")
+
+
+def ball_position(time_frames):
+    """Return the ball center after a sixfold initial-speed intervention."""
+    vx = SOURCE_VX * SPEED_MULTIPLIER
+    vy = SOURCE_VY * SPEED_MULTIPLIER
+
+    # Solve the first floor crossing analytically.
+    a = 0.5 * GRAVITY
+    b = vy
+    c = START_Y - FLOOR_Y
+    discriminant = b * b - 4.0 * a * c
+    contact = (-b + np.sqrt(discriminant)) / (2.0 * a)
+
+    if time_frames < contact:
+        x = START_X + vx * time_frames
+        y = START_Y + vy * time_frames + 0.5 * GRAVITY * time_frames**2
+        return x, y
+
+    impact_vy = vy + GRAVITY * contact
+    rebound_vy = -RESTITUTION * impact_vy
+    dt = time_frames - contact
+    x = START_X + vx * time_frames
+    y = FLOOR_Y + rebound_vy * dt + 0.5 * GRAVITY * dt**2
+
+    # The ball leaves the right side before its second floor impact.
+    if y > FLOOR_Y:
+        y = FLOOR_Y
+    return x, y
+
+
+def rotated_sprite(sprite, time_frames):
+    # A restrained clockwise spin keeps the ball's seams alive without
+    # overstating a spin change not requested by the prompt.
+    angle = -2.5 * time_frames
+    return sprite.rotate(
+        angle,
+        resample=Image.Resampling.BICUBIC,
+        center=(13.5, 13.5),
+    )
+
+
+def render(frames, background, sprite):
+    scale = SUPERSAMPLE
+    big_size = (WIDTH * scale, HEIGHT * scale)
+    plate = Image.fromarray(background, "RGB").resize(
+        big_size, Image.Resampling.BICUBIC
+    )
+    big_sprite = sprite.resize(
+        (sprite.width * scale, sprite.height * scale),
+        Image.Resampling.LANCZOS,
+    )
+    sprite_half = big_sprite.width / 2.0
+    output = np.empty((FRAME_COUNT, HEIGHT, WIDTH, 3), dtype=np.uint8)
+
+    for frame_index in range(FRAME_COUNT):
+        if frame_index == 0:
+            sample_times = np.array([0.0])
+        else:
+            sample_times = (
+                frame_index
+                + np.linspace(-0.5, 0.5, MOTION_SAMPLES) * SHUTTER
+            )
+            sample_times = np.clip(sample_times, 0.0, None)
+
+        accumulator = np.zeros((HEIGHT * scale, WIDTH * scale, 3), np.float32)
+        for sample_index, sample_time in enumerate(sample_times):
+            center_x, center_y = ball_position(sample_time)
+            # Reuse one rotated sprite across a frame's short shutter interval.
+            frame_sprite = big_sprite if sample_index else rotated_sprite(
+                sprite, frame_index
+            ).resize(
+                (sprite.width * scale, sprite.height * scale),
+                Image.Resampling.LANCZOS,
+            )
+            canvas = plate.copy()
+            paste_x = round(center_x * scale - sprite_half)
+            paste_y = round(center_y * scale - sprite_half)
+            canvas.paste(frame_sprite, (paste_x, paste_y), frame_sprite)
+            accumulator += np.asarray(canvas, dtype=np.float32)
+
+        mean = accumulator / len(sample_times)
+        downsampled = Image.fromarray(np.clip(mean, 0, 255).astype(np.uint8)).resize(
+            (WIDTH, HEIGHT), Image.Resampling.LANCZOS
+        )
+        output[frame_index] = np.asarray(downsampled, dtype=np.uint8)
+
+    return output
+
+
+def main():
+    source = load_source()
+    background = make_clean_plate(source)
+    sprite = extract_ball_sprite(source, background)
+    frames = render(source, background, sprite)
+
+    # Keep the exact source first frame: the intervention changes velocity,
+    # not the object's position at the intervention instant.
+    frames[0] = source[0]
+
+    artifact_io.save(frames, CASE_DIR)
+    x, y = ball_position(0.0)
+    contact_x, _ = ball_position(
+        (-SOURCE_VY * SPEED_MULTIPLIER)
+        / GRAVITY
+        + np.sqrt(
+            (SOURCE_VY * SPEED_MULTIPLIER) ** 2
+            + 2.0 * GRAVITY * (FLOOR_Y - START_Y)
+        )
+        / GRAVITY
+    )
+    print(
+        f"Saved {len(frames)} frames; start=({x:.2f}, {y:.2f}), "
+        f"first bounce x={contact_x:.2f}px"
+    )
+
+
+if __name__ == "__main__":
+    main()
