+"""Generate the low-friction toy-car counterfactual.
+
+The source scene is static apart from the car, so the edit uses a clean plate
+assembled from source frames and the original car sprite.  The motion model
+keeps the source launch position and initial speed, then reduces the uphill
+deceleration to represent the requested 0.2x friction coefficient.
+"""
+
+from __future__ import annotations
+
+import argparse
+import math
+import shutil
+from pathlib import Path
+
+import cv2
+import numpy as np
+from scipy.ndimage import binary_fill_holes, distance_transform_edt
+
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+WIDTH = 256
+HEIGHT = 144
+FRAME_COUNT = 96
+
+
+def build_clean_plate(source: np.ndarray) -> np.ndarray:
+    """Remove the car by taking the clean right side from a later frame."""
+    plate = source[0].copy()
+    x0, x1 = 145, 202
+    y0, y1 = 92, 130
+    plate[y0:y1, x0:x1] = source[24, y0:y1, x0:x1]
+
+    # Feather only the outer edge of the patch so the plate has no visible box.
+    yy, xx = np.mgrid[y0:y1, x0:x1]
+    edge = np.minimum.reduce(
+        [xx - x0, x1 - 1 - xx, yy - y0, y1 - 1 - yy]
+    ).astype(np.float32)
+    weight = np.clip(edge / 3.0, 0.0, 1.0)[..., None]
+    original = source[0, y0:y1, x0:x1].astype(np.float32)
+    replacement = source[24, y0:y1, x0:x1].astype(np.float32)
+    plate[y0:y1, x0:x1] = np.rint(
+        original * (1.0 - weight) + replacement * weight
+    ).astype(np.uint8)
+    return plate
+
+
+def extract_car_sprite(source: np.ndarray, plate: np.ndarray):
+    """Extract the original car from frame 1 with a soft alpha matte."""
+    x0, y0, x1, y1 = 147, 96, 198, 128
+    original = source[0, y0:y1, x0:x1].astype(np.float32)
+    background = plate[y0:y1, x0:x1].astype(np.float32)
+    difference = np.max(np.abs(original - background), axis=2)
+
+    raw = (difference > 7.0).astype(np.uint8)
+    raw = cv2.morphologyEx(raw, cv2.MORPH_CLOSE, np.ones((3, 3), np.uint8))
+    count, labels, stats, _ = cv2.connectedComponentsWithStats(raw, 8)
+    if count <= 1:
+        raise RuntimeError("Could not isolate the source car sprite")
+
+    # The car is the large component nearest the known source position.
+    expected = (170.5 - x0, 110.5 - y0)
+    best = None
+    for component in range(1, count):
+        area = int(stats[component, cv2.CC_STAT_AREA])
+        if area < 80:
+            continue
+        ys, xs = np.where(labels == component)
+        distance = math.hypot(float(xs.mean()) - expected[0], float(ys.mean()) - expected[1])
+        score = distance - 0.015 * area
+        if best is None or score < best[0]:
+            best = (score, component)
+    if best is None:
+        raise RuntimeError("Source car component was not found")
+
+    mask = labels == best[1]
+    mask = binary_fill_holes(mask)
+    mask = cv2.morphologyEx(
+        mask.astype(np.uint8), cv2.MORPH_CLOSE, np.ones((3, 3), np.uint8)
+    )
+
+    # Slightly soften the silhouette; preserve fully opaque interior pixels.
+    distance_inside = distance_transform_edt(mask > 0)
+    alpha = np.clip(distance_inside / 1.15, 0.0, 1.0)
+    alpha = cv2.GaussianBlur(alpha.astype(np.float32), (3, 3), 0.45)
+    ys, xs = np.where(mask)
+    crop_x0 = max(0, int(xs.min()) - 1)
+    crop_x1 = min(mask.shape[1], int(xs.max()) + 2)
+    crop_y0 = max(0, int(ys.min()) - 1)
+    crop_y1 = min(mask.shape[0], int(ys.max()) + 2)
+
+    rgb = original[crop_y0:crop_y1, crop_x0:crop_x1]
+    alpha = alpha[crop_y0:crop_y1, crop_x0:crop_x1]
+    anchor = np.array([170.5 - (x0 + crop_x0), 110.5 - (y0 + crop_y0)], dtype=np.float32)
+    return rgb, alpha, anchor
+
+
+def draw_shadow(
+    frame: np.ndarray,
+    center: tuple[float, float],
+    airborne: bool,
+    opacity: float,
+) -> None:
+    """Blend a soft contact/ground shadow before drawing the car."""
+    if opacity <= 0.0:
+        return
+    shadow = np.zeros((HEIGHT, WIDTH), dtype=np.uint8)
+    if airborne:
+        axes = (22, 4)
+        angle = 8.0
+        cx, cy = center[0] + 5.0, 119.0
+    else:
+        axes = (23, 5)
+        angle = 19.4
+        cx, cy = center[0] + 1.0, center[1] + 8.0
+    cv2.ellipse(
+        shadow,
+        (int(round(cx)), int(round(cy))),
+        axes,
+        angle,
+        0,
+        360,
+        255,
+        -1,
+        cv2.LINE_AA,
+    )
+    shadow = cv2.GaussianBlur(shadow.astype(np.float32) / 255.0, (0, 0), 2.1)
+    factor = 1.0 - opacity * shadow[..., None]
+    frame[:] = np.clip(frame.astype(np.float32) * factor, 0, 255).astype(np.uint8)
+
+
+def composite_car(
+    frame: np.ndarray,
+    rgb: np.ndarray,
+    alpha: np.ndarray,
+    anchor: np.ndarray,
+    center: tuple[float, float],
+    angle_delta: float,
+    scale: float = 1.0,
+) -> None:
+    """Place the extracted sprite at a subpixel position and angle."""
+    transform = cv2.getRotationMatrix2D(
+        (float(anchor[0]), float(anchor[1])), angle_delta, scale
+    )
+    transform[0, 2] += center[0] - anchor[0]
+    transform[1, 2] += center[1] - anchor[1]
+    warped_rgb = cv2.warpAffine(
+        rgb,
+        transform,
+        (WIDTH, HEIGHT),
+        flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=0,
+    ).astype(np.float32)
+    warped_alpha = cv2.warpAffine(
+        alpha,
+        transform,
+        (WIDTH, HEIGHT),
+        flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=0,
+    )[..., None]
+    warped_alpha = np.clip(warped_alpha, 0.0, 1.0)
+    frame[:] = np.clip(
+        warped_rgb * warped_alpha + frame.astype(np.float32) * (1.0 - warped_alpha),
+        0,
+        255,
+    ).astype(np.uint8)
+
+
+def motion_state(frame_index: int):
+    """Return center, airborne state, angle, and shadow strength."""
+    if frame_index == 0:
+        return np.array([170.5, 110.5]), False, 0.0, 0.0
+
+    # Source-calibrated values in pixels/frame.  The lower friction leaves
+    # substantially more speed for the uphill run and the ramp launch.
+    start = np.array([170.5, 110.5], dtype=np.float64)
+    uphill = np.array([-0.943068, -0.332600], dtype=np.float64)
+    ramp_distance = 123.0
+    initial_speed = 14.8
+    uphill_acceleration = -0.42
+    gravity = 1.55
+
+    t = float(frame_index)
+    discriminant = initial_speed**2 + 2.0 * uphill_acceleration * ramp_distance
+    exit_time = (initial_speed - math.sqrt(max(0.0, discriminant))) / (
+        -uphill_acceleration
+    )
+
+    if t <= exit_time:
+        distance = initial_speed * t + 0.5 * uphill_acceleration * t * t
+        center = start + uphill * distance
+        return center, False, 0.0, 0.24
+
+    exit_position = start + uphill * ramp_distance
+    exit_speed = initial_speed + uphill_acceleration * exit_time
+    exit_velocity = uphill * exit_speed
+    dt = t - exit_time
+    center = exit_position + exit_velocity * dt
+    center[1] += 0.5 * gravity * dt * dt
+
+    # A restrained tumble during the short launch reads as rigid-body motion.
+    angle = -4.5 * dt
+    shadow_strength = max(0.0, 0.18 * (1.0 - dt / 8.0))
+    return center, True, angle, shadow_strength
+
+
+def generate() -> np.ndarray:
+    source = np.load(ROOT / "source_pixels.npz")["rgb"]
+    if source.shape != (FRAME_COUNT, HEIGHT, WIDTH, 3):
+        raise ValueError(f"Unexpected source shape: {source.shape}")
+
+    plate = build_clean_plate(source)
+    sprite_rgb, sprite_alpha, anchor = extract_car_sprite(source, plate)
+
+    frames = np.empty_like(source)
+    for index in range(FRAME_COUNT):
+        if index == 0:
+            frames[index] = source[0]
+            continue
+        frame = plate.copy()
+        center, airborne, angle, shadow = motion_state(index)
+        draw_shadow(frame, tuple(center), airborne, shadow)
+        composite_car(frame, sprite_rgb, sprite_alpha, anchor, tuple(center), angle)
+        frames[index] = frame
+    return frames
+
+
+def main() -> None:
+    parser = argparse.ArgumentParser()
+    parser.add_argument(
+        "--out-dir",
+        type=Path,
+        default=ROOT,
+        help="Directory receiving pixels.npz, frames, index.html, and previews.",
+    )
+    args = parser.parse_args()
+    output = args.out_dir.resolve()
+    output.mkdir(parents=True, exist_ok=True)
+    for name in ("task_spec.json", "prompt.txt"):
+        source_path = (ROOT / name).resolve()
+        destination = (output / name).resolve()
+        if source_path != destination:
+            shutil.copy2(source_path, destination)
+    frames = generate()
+    artifact_io.save(frames, output)
+    print(f"Saved {len(frames)} frames to {output}")
+
+
+if __name__ == "__main__":
+    main()
