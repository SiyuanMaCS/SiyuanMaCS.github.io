+WIDTH = 256

+def build_clean_plate(source):
+    """Use frame 1 as the exact scene plate, replacing only its moving red stone."""
+    plate = source[0].copy()
+    temporal = np.median(source, axis=0).astype(np.uint8)
+    plate[59:81, 34:66] = temporal[59:81, 34:66]
+    return plate
+

+def extract_sprite(frame, plate, box, threshold=3.0, ramp=14.0):
+    x0, y0, x1, y1 = box
+    color = frame[y0:y1, x0:x1].astype(np.float32)
+    background = plate[y0:y1, x0:x1].astype(np.float32)
+    difference = np.abs(color - background).max(axis=2)
+    alpha = np.clip((difference - threshold) / ramp, 0.0, 1.0)
+    return color, alpha, (x0, y0)
+
+
+def translate_sprite(color, alpha, origin, anchor, center):
+    dx = center[0] - anchor[0]
+    dy = center[1] - anchor[1]
+    matrix = np.array([[1.0, 0.0, dx], [0.0, 1.0, dy]], dtype=np.float32)
+
+    premultiplied = np.dstack((color * alpha[:, :, None], alpha))
+    warped = cv2.warpAffine(
+        premultiplied,
+        matrix,
+        (WIDTH, HEIGHT),
+        flags=cv2.INTER_CUBIC,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=(0.0, 0.0, 0.0, 0.0),
+    )
+    warped = np.clip(warped, 0.0, 255.0)
+    return warped[:, :, :3], np.clip(warped[:, :, 3], 0.0, 1.0)
+
+
+def composite(image, rgb, alpha):
+    return image * (1.0 - alpha[:, :, None]) + rgb
+
+
+def positions(frame_index):
+    # Measured source trajectories, in pixels per source frame.
+    red_start = np.array([47.14, 66.17])
+    yellow_start = np.array([211.75, 76.13])
+    red_velocity = np.array([1.956, 0.1192]) * 0.5
+    yellow_velocity = np.array([-2.094, -0.1286])
+
+    # Equal-mass impact with the restitution measured from the source rebound.
+    contact_distance = 12.5
+    restitution = 0.15
+    closing = red_velocity[0] - yellow_velocity[0]
+    contact_time = (
+        yellow_start[0] - red_start[0] - contact_distance
+    ) / closing
+
+    if frame_index <= contact_time:
+        red = red_start + red_velocity * frame_index
+        yellow = yellow_start + yellow_velocity * frame_index
+        return red, yellow
+
+    red_contact = red_start + red_velocity * contact_time
+    yellow_contact = yellow_start + yellow_velocity * contact_time
+    center_velocity = (red_velocity[0] + yellow_velocity[0]) * 0.5
+    red_post = center_velocity - restitution * closing * 0.5
+    yellow_post = center_velocity + restitution * closing * 0.5
+    elapsed = frame_index - contact_time
+
+    # The measured post-impact vertical drift is nearly canceled at contact.
+    post_vertical_velocity = -0.004
+    red = red_contact + np.array([red_post, post_vertical_velocity]) * elapsed
+    yellow = yellow_contact + np.array(
+        [yellow_post, post_vertical_velocity]
+    ) * elapsed
+    return red, yellow
+
+
+def main():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    assert source.shape == (N_FRAMES, HEIGHT, WIDTH, 3)
+    plate = build_clean_plate(source)
+
+    red_color, red_alpha, _ = extract_sprite(
+        source[0], plate, (34, 59, 66, 82)
+    )
+    yellow_color, yellow_alpha, _ = extract_sprite(
+        source[10], plate, (178, 64, 207, 91)
+    )
+    red_anchor = np.array([47.14, 66.17])
+    yellow_anchor = np.array([191.11, 74.83])
+
+    frames = np.empty_like(source)
+    diagnostics = []
+    for index in range(N_FRAMES):
+        if index == 0:
+            frames[index] = source[0]
+        else:
+            red_center, yellow_center = positions(index)
+            canvas = plate.astype(np.float32)
+            red_rgb, red_a = translate_sprite(
+                red_color, red_alpha, None, red_anchor, red_center
+            )
+            yellow_rgb, yellow_a = translate_sprite(
+                yellow_color, yellow_alpha, None, yellow_anchor, yellow_center
+            )
+            canvas = composite(canvas, red_rgb, red_a)
+            canvas = composite(canvas, yellow_rgb, yellow_a)
+            frames[index] = np.clip(canvas + 0.5, 0, 255).astype(np.uint8)
+        diagnostics.append((*positions(index),))
+
+    artifact_io.save(frames, CASE_DIR)
+
+    np.savez_compressed(
+        CASE_DIR / ".tmp" / "trajectory.npz",
+        red=np.array([entry[0] for entry in diagnostics]),
+        yellow=np.array([entry[1] for entry in diagnostics]),
+    )
+    print("contact frame:", int(np.ceil(np.argmax(
+        np.linalg.norm(
+            np.array([entry[0] for entry in diagnostics])
+            - np.array([entry[1] for entry in diagnostics]),
+            axis=1,
+        ) < 12.6
+    ))) + 1)
+    print("saved", frames.shape, frames.dtype)
+
+
+if __name__ == "__main__":
+    main()
