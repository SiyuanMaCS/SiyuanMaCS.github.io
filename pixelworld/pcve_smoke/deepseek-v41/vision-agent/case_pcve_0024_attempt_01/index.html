+#!/usr/bin/env python3
+"""Render the tripled-friction volleyball counterfactual."""
+
+from __future__ import annotations
+
+import argparse
+import shutil
+from pathlib import Path
+
+import numpy as np
+from PIL import Image, ImageFilter
+
+import artifact_io
+
+
+WIDTH = 256
+HEIGHT = 144
+FRAME_COUNT = 96
+FPS = 24
+
+# Measured from the source roll. The high-friction ball follows the same
+# initial velocity but reaches rest after about 4.2 source frames.
+START = np.array([79.2, 121.2], dtype=np.float64)
+VELOCITY = np.array([4.65, -6.85], dtype=np.float64)
+STOP_FRAME = 4.2
+
+# The source ball is a compact ellipse at this location in frame 1.
+SPRITE_CENTER = np.array([79.5, 121.5], dtype=np.float64)
+SPRITE_RADII = np.array([11.6, 13.0], dtype=np.float64)
+
+
+def clean_plate(source: np.ndarray) -> np.ndarray:
+    """Remove the source ball and shadow while retaining the static room."""
+    plate = source[0].astype(np.float32)
+    late = source[95].astype(np.float32)
+
+    # The late frame has a clear floor in the lower-left. Feather the patch
+    # slightly so sensor/compression noise does not create a visible seam.
+    x0, x1, y0, y1 = 8, 138, 94, 144
+    target = late[y0:y1, x0:x1]
+    alpha = np.ones((y1 - y0, x1 - x0, 1), dtype=np.float32)
+    ramp = 5
+    alpha[:ramp, :, 0] *= np.linspace(0.0, 1.0, ramp)[:, None]
+    alpha[:, :ramp, 0] *= np.linspace(0.0, 1.0, ramp)[None, :]
+    alpha[:, -ramp:, 0] *= np.linspace(1.0, 0.0, ramp)[None, :]
+    plate[y0:y1, x0:x1] = (
+        plate[y0:y1, x0:x1] * (1.0 - alpha)
+        + target * alpha
+    )
+    return plate
+
+
+def extract_ball(source: np.ndarray) -> Image.Image:
+    """Extract an antialiased RGBA sprite from the first source frame."""
+    margin = 20
+    cx, cy = SPRITE_CENTER
+    x0 = int(round(cx)) - margin
+    y0 = int(round(cy)) - margin
+    crop = source[0, y0 : y0 + 2 * margin, x0 : x0 + 2 * margin]
+
+    yy, xx = np.mgrid[0 : 2 * margin, 0 : 2 * margin]
+    px = xx + x0 + 0.5
+    py = yy + y0 + 0.5
+    radius = np.sqrt(
+        ((px - cx) / SPRITE_RADII[0]) ** 2
+        + ((py - cy) / SPRITE_RADII[1]) ** 2
+    )
+    alpha = np.clip((1.04 - radius) / 0.09, 0.0, 1.0)
+
+    rgba = np.dstack([crop, (alpha * 255.0).round().astype(np.uint8)])
+    return Image.fromarray(rgba, "RGBA")
+
+
+def ball_position(frame: int) -> tuple[np.ndarray, float, float]:
+    """Return center, rolling progress, and screen scale for a frame."""
+    t = float(frame)
+    if t <= STOP_FRAME:
+        acceleration = -VELOCITY / STOP_FRAME
+        position = START + VELOCITY * t + 0.5 * acceleration * t * t
+        progress = t / STOP_FRAME
+    else:
+        position = START + VELOCITY * STOP_FRAME / 2.0
+        progress = 1.0
+
+    scale = float(np.clip(1.0 - 0.0080 * (position[1] - START[1]), 0.76, 1.0))
+    return position, progress, scale
+
+
+def paste_ball(
+    frame: np.ndarray,
+    sprite: Image.Image,
+    center: np.ndarray,
+    progress: float,
+    scale: float,
+) -> None:
+    """Rotate, scale, and alpha-composite the ball onto a frame."""
+    angle = -102.0 * (progress ** 1.08)
+    turned = sprite.rotate(angle, resample=Image.Resampling.BICUBIC)
+
+    side = int(round(turned.width * scale))
+    resized = turned.resize((side, side), Image.Resampling.LANCZOS)
+    left = int(round(center[0] - side / 2.0))
+    top = int(round(center[1] - side / 2.0))
+
+    patch = np.asarray(resized, dtype=np.float32)
+    dst_x0 = max(0, left)
+    dst_y0 = max(0, top)
+    dst_x1 = min(WIDTH, left + side)
+    dst_y1 = min(HEIGHT, top + side)
+    src_x0 = dst_x0 - left
+    src_y0 = dst_y0 - top
+    src_x1 = src_x0 + (dst_x1 - dst_x0)
+    src_y1 = src_y0 + (dst_y1 - dst_y0)
+
+    rgba = patch[src_y0:src_y1, src_x0:src_x1]
+    alpha = rgba[:, :, 3:4] / 255.0
+    region = frame[dst_y0:dst_y1, dst_x0:dst_x1]
+    frame[dst_y0:dst_y1, dst_x0:dst_x1] = (
+        rgba[:, :, :3] * alpha + region * (1.0 - alpha)
+    )
+
+
+def add_shadow(frame: np.ndarray, center: np.ndarray, scale: float) -> None:
+    """Add a soft, directional contact shadow beneath the ball."""
+    cx = center[0] - 13.0 * scale
+    cy = center[1] + 4.5 * scale
+    rx = 17.0 * scale
+    ry = 9.0 * scale
+    angle = np.deg2rad(12.0)
+
+    x0 = max(0, int(np.floor(cx - 2.2 * rx)))
+    x1 = min(WIDTH, int(np.ceil(cx + 2.2 * rx)))
+    y0 = max(0, int(np.floor(cy - 2.2 * ry)))
+    y1 = min(HEIGHT, int(np.ceil(cy + 2.2 * ry)))
+    yy, xx = np.mgrid[y0:y1, x0:x1]
+    dx = xx + 0.5 - cx
+    dy = yy + 0.5 - cy
+    ca, sa = np.cos(angle), np.sin(angle)
+    u = ca * dx + sa * dy
+    v = -sa * dx + ca * dy
+    d2 = (u / rx) ** 2 + (v / ry) ** 2
+
+    broad = 0.24 * np.clip(1.0 - d2, 0.0, 1.0) ** 0.72
+    halo = 0.055 * np.exp(-1.8 * d2)
+    alpha = np.clip(broad + halo, 0.0, 0.30)[:, :, None]
+    frame[y0:y1, x0:x1] *= 1.0 - alpha
+
+
+def render(source: np.ndarray) -> np.ndarray:
+    plate = clean_plate(source)
+    sprite = extract_ball(source)
+    frames = np.empty((FRAME_COUNT, HEIGHT, WIDTH, 3), dtype=np.uint8)
+
+    for index in range(FRAME_COUNT):
+        frame = plate.copy()
+        center, progress, scale = ball_position(index)
+        add_shadow(frame, center, scale)
+        paste_ball(frame, sprite, center, progress, scale)
+        frames[index] = np.clip(frame, 0, 255).round().astype(np.uint8)
+
+    return frames
+
+
+def parse_args() -> argparse.Namespace:
+    parser = argparse.ArgumentParser()
+    parser.add_argument(
+        "--output",
+        type=Path,
+        default=Path(__file__).resolve().parent,
+        help="Directory that receives the rendered artifacts.",
+    )
+    return parser.parse_args()
+
+
+def main() -> None:
+    args = parse_args()
+    root = Path(__file__).resolve().parent
+    output = args.output.resolve()
+    output.mkdir(parents=True, exist_ok=True)
+
+    for name in ("prompt.txt", "task_spec.json"):

+

+    artifact_io.save(frames, output)
+
+    resting, _, resting_scale = ball_position(FRAME_COUNT - 1)
+    print(
+        "saved",
+        len(frames),
+        "frames",
+        f"{WIDTH}x{HEIGHT}@{FPS}",
+        "to",

+    )

+        "resting ball center",
+        tuple(np.round(resting, 2)),
+        "scale",
+        round(resting_scale, 3),
+    )
+
+
+if __name__ == "__main__":
+    main()
