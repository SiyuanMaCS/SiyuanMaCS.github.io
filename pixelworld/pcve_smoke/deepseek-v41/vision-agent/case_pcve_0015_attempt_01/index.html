+def _fill_holes(mask):

+    padded = np.pad(mask.astype(np.uint8), 1)
+    flood = padded.copy()

+def _track_ball(frame_index):
+    """Return the ball mask and source bounding box for one frame."""
+    frame = SOURCE[frame_index]
+    hsv = cv2.cvtColor(frame, cv2.COLOR_RGB2HSV)
+    mask = (

+        & (hsv[:, :, 0] <= 10)
+        & (hsv[:, :, 1] >= 160)
+        & (hsv[:, :, 2] >= 100)
+    ).astype(np.uint8)
+    mask[112:, :] = 0
+    mask[:, 188:] = 0
+    mask = cv2.morphologyEx(mask, cv2.MORPH_CLOSE, np.ones((3, 3), np.uint8))
+
+    count, labels, stats, centroids = cv2.connectedComponentsWithStats(mask, 8)
+    predicted = np.array(_ball_center(frame_index))
+    best = None
+    for label in range(1, count):
+        x, y, w, h, area = stats[label]
+        if area < 180 or w < 12 or h < 12:
+            continue
+        center = centroids[label]
+        distance = np.linalg.norm(center - predicted)
+        if distance > 28.0:
+            continue
+        score = distance + max(0.0, 500.0 - area) * 0.01 + max(0.0, area - 1700.0) * 0.01
+        if best is None or score < best[0]:
+            best = (score, label, (x, y, w, h, area))
+
+    if best is None:
+        raise RuntimeError(f"Ball not found in frame {frame_index}")
+
+    _, label, (x, y, w, h, _) = best
+    component = (labels == label).astype(np.uint8)
+    component = _fill_holes(component)
+    component = cv2.morphologyEx(component, cv2.MORPH_CLOSE, np.ones((3, 3), np.uint8))
+
+    # The color blob omits the darkest rim. Grow it slightly, then soften only the edge.

+def _floor_plate():
+    """Build a clean static floor with the block and its shadow removed."""
+    frames = SOURCE.astype(np.float32)
+    base = np.median(frames, axis=0).astype(np.uint8)

+    # to the right continues their perspective and grain without synthesizing a new scene.

+
+    alpha = np.ones((target_y1 - target_y0, target_x1 - target_x0), dtype=np.float32)
+    feather = 4
+    ramp = np.linspace(0.0, 1.0, feather, endpoint=True, dtype=np.float32)
+    alpha[:, :feather] *= ramp[None, :]
+    alpha[:, -feather:] *= ramp[::-1][None, :]
+    alpha[:feather, :] *= ramp[:, None]

+    base[target_y0:target_y1, target_x0:target_x1] = np.clip(
+        roi * (1.0 - alpha[:, :, None]) + patch.astype(np.float32) * alpha[:, :, None],
+        0,
+        255,
+    ).astype(np.uint8)
+    return base
+
+
+def generate():
+    plate = _floor_plate()
+    output = np.repeat(plate[None, :, :, :], len(SOURCE), axis=0)
+    for index in range(len(SOURCE)):
+        alpha, _ = _track_ball(index)
+        source = SOURCE[index].astype(np.float32)
+        result = output[index].astype(np.float32)
+        result = result * (1.0 - alpha[:, :, None]) + source * alpha[:, :, None]
+        output[index] = np.clip(result, 0, 255).astype(np.uint8)
+    return output
+
