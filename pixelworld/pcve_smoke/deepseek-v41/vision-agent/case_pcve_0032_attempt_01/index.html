+import cv2
+import numpy as np

+import artifact_io

+CASE_DIR = Path(__file__).resolve().parent
+FRAME_COUNT = 96
+HEIGHT = 144
+WIDTH = 256

+# The source collision is between frames 32 and 33 (zero based).  The first
+# post-intervention rebound speed is 10% of the source rebound speed.
+CONTACT_FRAME = 32.14
+FLOOR_CENTER_Y = 85.50
+GRAVITY = 0.779
+SOURCE_REBOUND_SPEED = 6.77
+NEW_RESTITUTION_SCALE = 0.10

+def circle_center(image, previous_center, background):
+    """Locate the orange ball from a background difference in its local ROI."""
+    px, py = previous_center
+    x0 = max(0, int(px) - 25)
+    x1 = min(WIDTH, int(px) + 26)
+    y0 = max(0, int(py) - 25)
+    y1 = min(HEIGHT, int(py) + 26)
+    diff = np.abs(image[y0:y1, x0:x1].astype(np.float32) - background[y0:y1, x0:x1]).mean(axis=2)
+    mask = (diff > 18).astype(np.uint8)
+    count, labels, stats, centroids = cv2.connectedComponentsWithStats(mask, 8)
+    if count <= 1:
+        return previous_center
+    component = 1 + int(np.argmax(stats[1:, cv2.CC_STAT_AREA]))
+    if stats[component, cv2.CC_STAT_AREA] < 120:
+        return previous_center
+    return (
+        float(centroids[component, 0] + x0),
+        float(centroids[component, 1] + y0),
+    )
+

+def build_clean_background(frames):
+    """Use clean wall and floor pixels from frames where the ball is elsewhere."""
+    background = frames[0].copy()
+    # Frame 70 has no ball in the upper wall ROI.
+    background[0:67, 103:188] = frames[70, 0:67, 103:188]
+    return background

+def build_ball_sprite(frames, background):
+    """Extract a clean ball sprite and an antialiased circular alpha mask."""
+    source = frames[22]
+    cx, cy = 141.705, 45.159
+    radius = 13
+    x0, y0 = int(round(cx)) - radius, int(round(cy)) - radius
+    size = radius * 2 + 1
+    rgb = source[y0 : y0 + size, x0 : x0 + size].copy()
+
+    yy, xx = np.mgrid[0:size, 0:size]
+    distance = np.sqrt((xx - (cx - x0)) ** 2 + (yy - (cy - y0)) ** 2)
+    alpha = np.clip((11.75 - distance) / 1.15, 0.0, 1.0).astype(np.float32)
+
+    # Remove wall contamination from partially covered boundary pixels.
+    local_bg = background[y0 : y0 + size, x0 : x0 + size].astype(np.float32)
+    observed = rgb.astype(np.float32)
+    safe_alpha = np.maximum(alpha, 0.08)[..., None]
+    foreground = (observed - (1.0 - alpha[..., None]) * local_bg) / safe_alpha
+    boundary = (alpha > 0.02) & (alpha < 0.98)
+    rgb_float = observed.copy()
+    rgb_float[boundary] = np.clip(foreground[boundary], 0, 255)
+    return rgb_float.astype(np.uint8), alpha
+

+def rotate_sprite(rgb, alpha, angle_degrees):
+    if abs(angle_degrees) < 1e-4:
+        return rgb, alpha
+    height, width = alpha.shape
+    center = ((width - 1) * 0.5, (height - 1) * 0.5)
+    matrix = cv2.getRotationMatrix2D(center, angle_degrees, 1.0)
+    rotated_rgb = cv2.warpAffine(
+        rgb,
+        matrix,
+        (width, height),
+        flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_REPLICATE,
+    )
+    rotated_alpha = cv2.warpAffine(
+        alpha,
+        matrix,
+        (width, height),
+        flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=0,
+    )
+    return rotated_rgb, rotated_alpha
+

+def composite_ball(canvas, rgb, alpha, center):
+    height, width = alpha.shape
+    cx, cy = center
+    left = int(np.floor(cx - (width - 1) * 0.5))
+    top = int(np.floor(cy - (height - 1) * 0.5))
+
+    x0, y0 = max(0, left), max(0, top)
+    x1 = min(WIDTH, left + width)
+    y1 = min(HEIGHT, top + height)
+    sx0, sy0 = x0 - left, y0 - top
+    sx1, sy1 = sx0 + (x1 - x0), sy0 + (y1 - y0)
+
+    patch = canvas[y0:y1, x0:x1].astype(np.float32)
+    ball = rgb[sy0:sy1, sx0:sx1].astype(np.float32)
+    mask = alpha[sy0:sy1, sx0:sx1, None]
+    canvas[y0:y1, x0:x1] = np.clip(patch * (1.0 - mask) + ball * mask, 0, 255).astype(np.uint8)

+def add_contact_shadow(canvas, center_x, center_y):
+    """Darken a small floor ellipse under the nearly settled ball."""
+    height_fraction = np.clip((center_y - 70.0) / (FLOOR_CENTER_Y - 70.0), 0.0, 1.0)
+    opacity = 0.12 + 0.25 * height_fraction
+    shadow_y = FLOOR_CENTER_Y + 12.0
+
+    yy, xx = np.mgrid[0:HEIGHT, 0:WIDTH]
+    ellipse = ((xx - center_x) / 13.0) ** 2 + ((yy - shadow_y) / 4.3) ** 2
+    alpha = np.clip(1.0 - ellipse, 0.0, 1.0) ** 1.35
+    alpha *= opacity
+    alpha[yy < 90] = 0.0
+    alpha[:, :105] = 0.0
+    alpha[:, 185:] = 0.0
+
+    # Contact shadows are soft and multiply the floor's warm color.
+    tint = np.array([0.60, 0.50, 0.46], dtype=np.float32)
+    original = canvas.astype(np.float32)
+    shaded = original * tint[None, None, :]
+    canvas[:] = np.clip(original * (1.0 - alpha[..., None]) + shaded * alpha[..., None], 0, 255).astype(np.uint8)
+
+
+def custom_vertical_position(frame_index):
+    """Tiny successive hops after the restitution intervention."""
+    upward_speed = SOURCE_REBOUND_SPEED * NEW_RESTITUTION_SCALE
+    time_since_contact = max(0.0, frame_index - CONTACT_FRAME)
+    cycle_time = 2.0 * upward_speed / GRAVITY
+    if cycle_time <= 1e-6:
+        return FLOOR_CENTER_Y
+    phase = time_since_contact % cycle_time
+    y = FLOOR_CENTER_Y - upward_speed * phase + 0.5 * GRAVITY * phase * phase
+    return float(np.clip(y, FLOOR_CENTER_Y - upward_speed**2 / (2.0 * GRAVITY), FLOOR_CENTER_Y))
+
+
+def source_trajectory(frames, background):
+    centers = []
+    previous = (128.0, 22.0)
+    for frame in frames:
+        previous = circle_center(frame, previous, background)
+        centers.append(previous)
+    return np.asarray(centers, dtype=np.float32)
+
+
+def feather_roi(canvas, source_frame, background, x0=104, x1=187, y0=0, y1=130, feather=4):
+    """Replace the motion ROI while matching the source outside it exactly."""
+    output = source_frame.copy()
+    for xa, xb in ((x0, x0 + feather), (x1 - feather, x1)):
+        if xb <= xa:
+            continue
+        blend = np.linspace(0.0, 1.0, feather, dtype=np.float32)
+        if xa == x0:
+            blend = blend[::-1]
+        output[y0:y1, xa:xb] = (
+            canvas[y0:y1, xa:xb].astype(np.float32) * (1.0 - blend[None, :, None])
+            + source_frame[y0:y1, xa:xb].astype(np.float32) * blend[None, :, None]
+        ).astype(np.uint8)
+    output[y0:y1, x0 + feather : x1 - feather] = canvas[y0:y1, x0 + feather : x1 - feather]
+
+    for ya, yb in ((y0, y0 + feather), (y1 - feather, y1)):
+        if yb <= ya:
+            continue
+        blend = np.linspace(0.0, 1.0, feather, dtype=np.float32)
+        if ya == y0:
+            blend = blend[::-1]
+        row = output[ya:yb, x0:x1].astype(np.float32)
+        canvas_rows = canvas[ya:yb, x0:x1].astype(np.float32)
+        output[ya:yb, x0:x1] = (
+            canvas_rows * (1.0 - blend[:, None, None])
+            + row * blend[:, None, None]
+        ).astype(np.uint8)
+    return output
+
+
+def generate():
+    with np.load(CASE_DIR / "source_pixels.npz") as data:
+        frames = data["rgb"]
+    background = build_clean_background(frames)
+    trajectory = source_trajectory(frames, background)
+    ball_rgb, ball_alpha = build_ball_sprite(frames, background)
+
+    output = frames.copy()
+    for frame_index in range(int(np.ceil(CONTACT_FRAME)) + 1, FRAME_COUNT):
+        center_x = float(trajectory[frame_index, 0])
+        center_y = custom_vertical_position(frame_index)
+
+        canvas = background.copy()
+        add_contact_shadow(canvas, center_x, center_y)
+        angle = -0.85 * (frame_index - CONTACT_FRAME)
+        rgb, alpha = rotate_sprite(ball_rgb, ball_alpha, angle)
+        composite_ball(canvas, rgb, alpha, (center_x, center_y))
+        output[frame_index] = feather_roi(canvas, frames[frame_index], background)
+
+    assert output.shape == (FRAME_COUNT, HEIGHT, WIDTH, 3)
+    assert output.dtype == np.uint8
+    artifact_io.save(output, CASE_DIR)
+    return output
+
+
+if __name__ == "__main__":
+    generate()
