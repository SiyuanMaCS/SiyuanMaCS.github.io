+CASE_DIR = Path(__file__).resolve().parent
+INTERVENTION_FRAME = 21  # Frame 22 on the 1-based source timeline.

+def red_mallet_mask(frame):
+    """Return the expanded footprint of the moving red mallet."""
+    rgb = frame.astype(np.int16)
+    red = (
+        (rgb[:, :, 0] > 100)
+        & (rgb[:, :, 0] - rgb[:, :, 1] > 30)
+        & (rgb[:, :, 0] - rgb[:, :, 2] > 20)
+    )
+
+    roi = np.zeros(red.shape, dtype=np.uint8)
+    roi[48:104, 82:175] = 1
+    red = (red & roi.astype(bool)).astype(np.uint8)
+
+    # Include the pale rim and the soft contact shadow around the red cap.
+    kernel = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (11, 11))
+    expanded = cv2.dilate(red, kernel, iterations=1)
+    return expanded.astype(bool)
+
+
+def build_clean_plate(frames, masks):
+    """Median-composite each table pixel from frames where it is unobstructed."""
+    height, width = frames.shape[1:3]
+    plate = np.empty((height, width, 3), dtype=np.uint8)
+    valid_count = np.zeros((height, width), dtype=np.int16)
+
+    for y in range(height):
+        for x in range(width):
+            values = frames[~masks[:, y, x], y, x]
+            if len(values):
+                plate[y, x] = np.median(values, axis=0).astype(np.uint8)
+                valid_count[y, x] = len(values)
+            else:
+                plate[y, x] = frames[0, y, x]
+
+    # A tiny fallback only affects pixels that are never unobstructed.
+    missing = valid_count == 0
+    if missing.any():
+        filled = cv2.inpaint(
+            cv2.cvtColor(plate, cv2.COLOR_RGB2BGR),
+            missing.astype(np.uint8),
+            3,
+            cv2.INPAINT_TELEA,
+        )
+        plate[missing] = cv2.cvtColor(filled, cv2.COLOR_BGR2RGB)[missing]
+    return plate
+
+
+def make_feather(mask):
+    alpha = cv2.GaussianBlur(mask.astype(np.float32), (0, 0), 1.15)
+    return np.clip(alpha * 2.1, 0.0, 1.0)[:, :, None]
+
+
+def main():
+    source = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    frames = source.copy()
+
+    masks = np.stack([red_mallet_mask(frame) for frame in source], axis=0)
+    clean_plate = build_clean_plate(source, masks)
+
+    for index in range(INTERVENTION_FRAME, len(frames)):
+        alpha = make_feather(masks[index])
+        frames[index] = np.clip(
+            source[index].astype(np.float32) * (1.0 - alpha)
+            + clean_plate.astype(np.float32) * alpha,
+            0,
+            255,
+        ).astype(np.uint8)
+
+    artifact_io.save(frames, CASE_DIR)
+    print(f"saved {len(frames)} frames; clean plate and masks built")
+
+
+if __name__ == "__main__":
+    main()
