+from pathlib import Path
+
+import cv2
+import numpy as np
+from scipy.ndimage import binary_fill_holes, gaussian_filter
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def feather_patch(dst, src, x0, y0, x1, y1, feather=4):
+    """Replace a rectangular region with a softly feathered source patch."""
+    h, w = dst.shape[:2]
+    x0, y0 = max(0, x0), max(0, y0)
+    x1, y1 = min(w, x1), min(h, y1)
+    yy, xx = np.mgrid[y0:y1, x0:x1]
+    edge = np.minimum.reduce(
+        [xx - x0, x1 - 1 - xx, yy - y0, y1 - 1 - yy]
+    ).astype(np.float32)
+    alpha = np.clip((edge + 0.5) / feather, 0.0, 1.0)
+    alpha = gaussian_filter(alpha, 0.8)[..., None]
+    dst[y0:y1, x0:x1] = (
+        dst[y0:y1, x0:x1].astype(np.float32) * (1.0 - alpha)
+        + src[y0:y1, x0:x1].astype(np.float32) * alpha
+    ).round().astype(np.uint8)
+
+
+def make_car_free_plate(frames):
+    """Build the static tabletop plate with the parked source car removed."""
+    plate = frames[95].copy()
+
+    # At the end of the factual clip the car is parked at x=183..203.
+    # Frame 0 shows the same tabletop with that area unobstructed.
+    feather_patch(plate, frames[0], 176, 36, 209, 57, feather=5)
+    return plate
+
+
+def extract_car_sprite(frame, plate, box):
+    """Extract a soft-edged RGB sprite and its contact shadow."""
+    x0, y0, x1, y1 = box
+    src = frame[y0:y1, x0:x1].astype(np.float32)
+    bg = plate[y0:y1, x0:x1].astype(np.float32)
+    diff = np.max(np.abs(src - bg), axis=2)
+
+    # The car is the only strong local difference in these early crops.
+    mask = diff > 5.0
+    mask = binary_fill_holes(mask)
+    mask = cv2.morphologyEx(
+        mask.astype(np.uint8), cv2.MORPH_CLOSE, np.ones((2, 2), np.uint8)
+    )
+    mask = cv2.dilate(mask, np.ones((2, 2), np.uint8), iterations=1).astype(bool)
+
+    alpha = gaussian_filter(mask.astype(np.float32), 0.55)
+    alpha = np.clip(alpha * 1.15, 0.0, 1.0)
+    rgba = np.dstack([src, alpha * 255.0]).astype(np.uint8)
+    return rgba
+
+
+def paste_sprite(frame, sprite, center_xy):
+    """Alpha-composite a sprite at a possibly subpixel center position."""
+    sh, sw = sprite.shape[:2]
+    cx, cy = center_xy
+    tx = cx - sw / 2.0
+    ty = cy - sh / 2.0
+    matrix = np.array([[1.0, 0.0, tx], [0.0, 1.0, ty]], dtype=np.float32)
+
+    warped = cv2.warpAffine(
+        sprite,
+        matrix,
+        (frame.shape[1], frame.shape[0]),
+        flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=(0, 0, 0, 0),
+    )
+    color = warped[..., :3].astype(np.float32)
+    alpha = warped[..., 3:4].astype(np.float32) / 255.0
+    return (
+        frame.astype(np.float32) * (1.0 - alpha) + color * alpha
+    ).round().astype(np.uint8)
+
+
+def main():
+    data = np.load(CASE_DIR / "source_pixels.npz")["rgb"]
+    frames = data.copy()
+    plate = make_car_free_plate(frames)
+
+    # Early factual crops provide the actual car geometry and contact shadow.
+    # A wider patch keeps the soft shadow around the body.
+    sprites = {
+        1: extract_car_sprite(frames[1], plate, (65, 22, 93, 38)),
+        2: extract_car_sprite(frames[2], plate, (76, 22, 104, 38)),
+        3: extract_car_sprite(frames[3], plate, (86, 22, 114, 39)),
+    }
+    sprite_centers = {1: (79.0, 29.5), 2: (90.0, 29.5), 3: (100.0, 30.5)}
+
+    # The source starts at about 11 px/frame and decelerates at about
+    # 0.54 px/frame^2 once rolling. Seven times that deceleration gives a
+    # stopping distance near 16 px from the frame-1 pose.
+    x_start = 78.5
+    y_start = 29.5
+    speed = 11.0
+    deceleration = 7.0 * 0.54
+    stop_time = speed / deceleration
+    stop_x = x_start + speed * stop_time - 0.5 * deceleration * stop_time**2
+    y_slope = 0.075
+
+    output = np.empty_like(frames)
+    output[0] = frames[0]
+
+    for i in range(1, len(frames)):
+        t = float(i - 1)
+        if t < stop_time:
+            x = x_start + speed * t - 0.5 * deceleration * t * t
+        else:
+            x = stop_x
+        y = y_start + y_slope * (x - x_start)
+
+        # Select the source pose closest to the edited position. The sprite
+        # centers above are measured from the corresponding factual frames.
+        if x < 84.0:
+            source_index = 1
+        elif x < 95.0:
+            source_index = 2
+        else:
+            source_index = 3
+
+        sprite = sprites[source_index]
+        scx, scy = sprite_centers[source_index]
+        dx, dy = x - scx, y - scy
+        shifted = np.zeros_like(sprite)
+        shifted[..., :3] = cv2.warpAffine(
+            sprite[..., :3],
+            np.array([[1.0, 0.0, dx], [0.0, 1.0, dy]], dtype=np.float32),
+            (sprite.shape[1], sprite.shape[0]),
+            flags=cv2.INTER_LINEAR,
+            borderMode=cv2.BORDER_CONSTANT,
+            borderValue=(0, 0, 0),
+        )
+        shifted[..., 3] = cv2.warpAffine(
+            sprite[..., 3],
+            np.array([[1.0, 0.0, dx], [0.0, 1.0, dy]], dtype=np.float32),
+            (sprite.shape[1], sprite.shape[0]),
+            flags=cv2.INTER_LINEAR,
+            borderMode=cv2.BORDER_CONSTANT,
+            borderValue=0,
+        )
+        output[i] = paste_sprite(plate, shifted, (scx + dx, scy + dy))
+
+    artifact_io.save(output, CASE_DIR)
+
+
+if __name__ == "__main__":
+    main()
