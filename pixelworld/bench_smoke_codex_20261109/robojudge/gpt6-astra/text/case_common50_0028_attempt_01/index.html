def food_layer(cx,cy,angle,roll):

    # Its side remains visible even in the edge-on frame.

    rotation=np.array([[np.cos(roll),-np.sin(roll)],[np.sin(roll),np.cos(roll)]])
    dx=X-cx;dy=Y-cy
    qx=rotation[0,0]*dx+rotation[1,0]*dy
    qy=rotation[0,1]*dx+rotation[1,1]*dy
    mirror=1 if c>=0 else -1
    sx=129+qx/(scale*mirror)
    sy=76+qy-.115*s*qx
