Physical process:
- A lit match is lowered along its axis into a clear glass filled with water.
- In the initial frames, the match flame flickers and illuminates the rim and interior of the glass.

- A realistic steam and smoke wisp rises and disperses upwards.
- The submerged portion of the matchstick exhibits Snell's law optical refraction through the water and glass cylinder.

import math
import cv2
import numpy as np

import artifact_io


def generate_frames():
    case_dir = Path(__file__).parent
    first_frame_path = case_dir / \"first_frame.png\"
    frame0_bgr = cv2.imread(str(first_frame_path))
    rgb0 = cv2.cvtColor(frame0_bgr, cv2.COLOR_BGR2RGB)
    h, w, _ = rgb0.shape

    # 1. High-fidelity background reconstruction
    # Inpaint only the matchstick and flame region from frame 0
    mask_inpaint = np.zeros((h, w), dtype=np.uint8)
    cv2.line(mask_inpaint, (73, 0), (127, 50), 255, thickness=9)
    cv2.circle(mask_inpaint, (124, 41), 12, 255, -1)
    cv2.circle(mask_inpaint, (126, 49), 7, 255, -1)
    bg = cv2.inpaint(rgb0, mask_inpaint, 5, cv2.INPAINT_TELEA)

    flame_crop0 = rgb0[32:50, 112:138].astype(float)
    bg_crop0 = bg[32:50, 112:138].astype(float)
    flame_emission0 = np.maximum(0.0, flame_crop0 - bg_crop0)

    tip_y0, tip_x0 = 49.5, 126.5
    u_y, u_x = 0.6792, 0.7341  # unit vector along stick towards tip

    water_y = 59.5  # meniscus level inside the glass

    num_frames = 20
    frames = []

    for t in range(num_frames):

            frames.append(rgb0.copy())
            continue

        tau = t / (num_frames - 1.0)

        dx = dy * (53.5 / 49.5)

        tip_x = tip_x0 + dx

        # Render matchstick along its length
        for s in np.arange(0, 110, 0.5):

            cx = tip_x - s * u_x

            in_water = cy > water_y

            eff_cx = cx
            eff_cy = cy
            if in_water:
                # Optical refraction through water and cylindrical glass
                eff_cx = cx - 0.9
                eff_cy = cy + 0.2

            for offset_w in np.arange(-2.2, 2.3, 0.4):
                py = int(round(eff_cy + offset_w * n_y))
                px = int(round(eff_cx + offset_w * n_x))

                if 0 <= py < h and 0 <= px < w:

                        # Burnt tip
                        col = np.array([32.0, 25.0, 22.0])
                    elif s < 9.0:
                        # Burnt-to-wood transition
                        alpha_head = (s - 6.0) / 3.0
                        col = (1.0 - alpha_head) * np.array([45.0, 30.0, 24.0]) + alpha_head * np.array([180.0, 148.0, 106.0])
                    else:

                        edge_factor = 1.0 - 0.25 * (abs(offset_w) / 2.2) ** 2
                        col = np.array([182.0, 150.0, 108.0]) * edge_factor

                        if abs(offset_w - 1.2) < 0.6:

            if t == 8:

                    wisp_rad = 2.5 + wisp_t * 1.6

                            for px in range(int(wisp_x - wisp_rad), int(wisp_x + wisp_rad + 1)):
                                if 0 <= py < h and 0 <= px < w:
                                    d = math.hypot(px - wisp_x, py - wisp_y)
                                    if d < wisp_rad:
                                        falloff = (1.0 - (d / wisp_rad) ** 1.5) * wisp_alpha
                                        smoke_col = np.array([215.0, 218.0, 225.0])
                                        frame[py, px] = (1.0 - falloff) * frame[py, px] + falloff * smoke_col

            # Water meniscus ripples
            ripple_center_x = quench_x

                frame[58:62, 102:140, c],
