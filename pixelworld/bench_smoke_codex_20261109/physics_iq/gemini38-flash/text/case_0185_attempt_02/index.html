+def get_glass_mask(h=144, w=256):
+    """
+    Returns a boolean mask of the inner cavity of the glass.
+    Glass rim is around y=61, bottom around y=134.
+    Inner walls are at x_left(y) and x_right(y).
+    """
+    mask = np.zeros((h, w), dtype=bool)
+    for y in range(60, 136):
+        if y < 115:
+            x0, x1 = 104, 143
+        elif y < 125:
+            x0, x1 = 105, 142
+        elif y < 132:
+            x0, x1 = 107, 140
+        else:
+            x0, x1 = 110, 137
+        mask[y, x0:x1] = True
+    return mask

+def render_stream(frame, y_start, y_end, t, x_center=121.5, radius=1.6):
+    """
+    Renders a realistic laminar water jet from the spout to the liquid surface.
+    """
+    h, w, _ = frame.shape
+    for y in range(int(y_start), int(y_end) + 1):
+        if y < 0 or y >= h:
+            continue
+        # Minor realistic ripple/sinuosity along the falling stream
+        sin_offset = 0.25 * np.sin(y * 0.45 - t * 15.0)
+        # Stream slightly narrows due to gravitational acceleration
+        dist_fall = max(0, y - y_start)
+        curr_radius = max(1.1, radius * (1.0 - 0.18 * (dist_fall / (y_end - y_start + 1e-5))**0.5))
+        
+        xc = x_center + sin_offset
+        x_min = int(np.floor(xc - curr_radius - 1.5))
+        x_max = int(np.ceil(xc + curr_radius + 1.5))
+        
+        for x in range(x_min, x_max + 1):
+            if 0 <= x < w:
+                d = abs(x - xc)
+                if d <= curr_radius + 1.2:
+                    norm_d = d / curr_radius
+                    if norm_d < 1.0:
+                        # Specular highlight near left/center of column
+                        highlight = np.exp(-((norm_d - 0.25) ** 2) / 0.15)
+                        edge = norm_d ** 3
+                        
+                        core_color = np.array([190, 195, 205], dtype=float)
+                        spec_color = np.array([250, 252, 255], dtype=float)
+                        edge_color = np.array([60, 55, 60], dtype=float)
+                        
+                        c = core_color * (1.0 - edge) + edge_color * edge
+                        c = c * (1.0 - 0.7 * highlight) + spec_color * (0.7 * highlight)
+                        
+                        alpha = 0.85
+                        frame[y, x] = np.clip((1.0 - alpha) * frame[y, x] + alpha * c, 0, 255).astype(np.uint8)
+                    else:
+                        alpha = max(0.0, 1.0 - (d - curr_radius) / 1.2) * 0.5
+                        c = np.array([120, 125, 135], dtype=float)
+                        frame[y, x] = np.clip((1.0 - alpha) * frame[y, x] + alpha * c, 0, 255).astype(np.uint8)

+def generate_video():
+    first_frame = np.array(Image.open("first_frame.png"))
+    n_frames = 20
+    fps = 4.0

+        
+        added_water = max(0.0, y_initial - curr_y_surf)

+                for idx, x in enumerate(xs):

+            if by < 60 or by >= 135:
+                continue
+            for bx in range(118, 126):
+                if np.random.rand() > 0.45:
+                    bubble_val = np.random.uniform(190, 245)
+                    frame[by, bx] = 0.5 * frame[by, bx] + 0.5 * np.array([bubble_val, bubble_val, bubble_val + 5])
+        
+        # 3. Add few small rising bubbles
+        for b_i in range(5):
+            bx = int(121.5 + np.random.uniform(-8, 8))
+            by = int(impact_y + 4 + np.random.uniform(2, 25))
+            if by < 134 and glass_mask[by, bx]:
+                frame[by, bx] = np.clip(frame[by, bx] * 0.4 + 140, 0, 255)
+                if glass_mask[by-1, bx]:
+                    frame[by-1, bx] = np.clip(frame[by-1, bx] * 0.3 + 170, 0, 255)
+        
+        # 4. Render falling water stream
+        frame_uint8 = np.clip(frame, 0, 255).astype(np.uint8)

+        combined = first_frame.copy()
+        combined[affected] = frame_uint8[affected]
+        
+        frames.append(combined)
