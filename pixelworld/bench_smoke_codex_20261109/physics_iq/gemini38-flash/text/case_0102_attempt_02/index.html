+from pathlib import Path
+import cv2
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+def generate_frames():
+    root = Path(__file__).parent
+    first_path = root / "first_frame.png"
+    first = np.array(Image.open(first_path).convert("RGB"))
+    h, w, _ = first.shape
+
+    # 1. Inpaint background beneath the wrench
+    cx, cy = 109.5, 61.5
+
+    # Identify wrench region
+    diff_init = np.zeros((h, w), dtype=float)
+    # Background estimate from surrounding table and platform
+    for y in range(48, 75):
+        plat_val = first[y, 120:128].mean(axis=0)
+        table_val = first[y, 94:100].mean(axis=0)
+        for x in range(101, 120):
+            d_p = np.linalg.norm(first[y, x] - plat_val)
+            d_t = np.linalg.norm(first[y, x] - table_val)
+            diff_init[y, x] = min(d_p, d_t)
+
+    mask = np.zeros((h, w), dtype=np.uint8)
+    mask[50:73, 103:117] = (diff_init[50:73, 103:117] > 10.0).astype(np.uint8) * 255
+
+    # Morphological close to retain solid body
+    kernel = cv2.getStructuringElement(cv2.MORPH_RECT, (3, 3))
+    mask_closed = cv2.morphologyEx(mask, cv2.MORPH_CLOSE, kernel)
+
+    # Keep only the connected component of the wrench
+    from scipy.ndimage import label
+    labeled, num_features = label(mask_closed > 0)
+    if num_features > 0:
+        counts = np.bincount(labeled.flat)[1:]
+        largest = np.argmax(counts) + 1
+        mask_clean = ((labeled == largest) * 255).astype(np.uint8)
+    else:
+        mask_clean = mask
+
+    # Inpaint background under the wrench
+    dilated_mask = cv2.dilate(mask_clean, cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (5, 5)))
+    bg = cv2.inpaint(first, dilated_mask, inpaintRadius=3, flags=cv2.INPAINT_TELEA)
+
+    # 2. Extract clean wrench sprite with color bleeding to prevent border artifacts
+    fg_rgb = first.copy()
+    fg_bleed = cv2.inpaint(fg_rgb, 255 - mask_clean, inpaintRadius=3, flags=cv2.INPAINT_TELEA)
+    alpha = cv2.GaussianBlur(mask_clean.astype(float) / 255.0, (3, 3), 0.6)
+    sprite_rgba = np.dstack([fg_bleed, (alpha * 255).astype(np.uint8)])
+
+    # 3. Render 20 frames (4 fps, 5 seconds)
+    frames = [first.copy()]
+    num_frames = 20
+    for i in range(1, num_frames):

+        M = cv2.getRotationMatrix2D((cx, cy), -theta_deg, 1.0)
+        warped = cv2.warpAffine(

+        )
+        fg_r = warped[:, :, :3]
+        fg_a = warped[:, :, 3].astype(float) / 255.0
+
