+    # 1. Clean background inpainting to remove the initial ball and shadow
+    mask = np.zeros((h, w), dtype=np.uint8)
+    cv2.circle(mask, (105, 47), 9, 255, -1)
+    cv2.ellipse(mask, (105, 56), (7, 3), 0, 0, 360, 255, -1)
+    clean_bg = cv2.inpaint(arr0, mask, 3, cv2.INPAINT_TELEA)
+
+    cx0, cy0 = 105.0, 47.0
+    R = 7.5
+
+    # Motion kinematics:
+    # Rolls from left (x0=105) across table towards the right edge (x19=248).
+    # Realistic rolling friction decelerates by ~25% over 5 seconds.
+    v0 = 8.6015
+    a = 0.11318
+
+    frames = []
+    for t in range(20):
+        if t == 0:
+            frames.append(arr0.copy())
+            continue
+
+        x_pos = cx0 + v0 * t - 0.5 * a * (t ** 2)
+        y_pos = cy0
+
+        frame = clean_bg.copy().astype(float)
+
+        # Ambient / lampshade illumination modulation
+        table_x = int(np.clip(round(x_pos), 0, w - 1))
+        local_lum = float(np.mean(clean_bg[int(y_pos), table_x]))
+        ref_lum = float(np.mean(clean_bg[int(cy0), int(cx0)]))
+        light_mod = float(np.clip(local_lum / ref_lum, 0.85, 1.25))
+
+        # 1. Contact shadow under the rolling ball
+        shadow_cy = y_pos + R + 1.5
+        dx_shadow = (x_pos - 135.0) * 0.12  # shadow cast from lampshade at x ~ 135
+        for sy in range(int(shadow_cy - 4), int(shadow_cy + 5)):
+            for sx in range(int(x_pos + dx_shadow - 9), int(x_pos + dx_shadow + 10)):
+                if 0 <= sy < h and 0 <= sx < w:
+                    d = ((sx - (x_pos + dx_shadow)) / 7.5) ** 2 + ((sy - shadow_cy) / 2.5) ** 2
+                    if d < 1.0:
+                        attenuation = (1.0 - d) * 0.65
+                        frame[sy, sx] *= (1.0 - attenuation)
+
+        # 2. Tennis ball rendering with rolling rotation
+        # Clockwise rotation in image plane as ball rolls to the right
+        theta = (x_pos - cx0) / R
+        cos_t, sin_t = np.cos(theta), np.sin(theta)
+
+        for dy in range(int(-R - 2), int(R + 3)):
+            for dx in range(int(-R - 2), int(R + 3)):
+                dist = np.sqrt(dx ** 2 + dy ** 2)
+                if dist > R + 0.8:
+                    continue
+

+                if not (0 <= py < h and 0 <= px < w):

+
+                # Sample rotated texture from ball in frame 0
+                dx_orig = cos_t * dx + sin_t * dy
+                dy_orig = -sin_t * dx + cos_t * dy
+
+                src_x = cx0 + dx_orig
+                src_y = cy0 + dy_orig
+
+                # Bilinear sampling

+                x1_s, y1_s = x0_s + 1, y0_s + 1
+                fx, fy = src_x - x0_s, src_y - y0_s
+
+                x0_s = np.clip(x0_s, 0, w - 1)
+                x1_s = np.clip(x1_s, 0, w - 1)

+
