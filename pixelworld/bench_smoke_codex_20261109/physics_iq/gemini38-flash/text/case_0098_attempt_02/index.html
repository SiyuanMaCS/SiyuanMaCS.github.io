+    delta = first_frame.astype(float) - bg.astype(float)

+    def shift_layer(delta_img, dy):
+        y_coords, x_coords = np.indices(delta_img.shape[:2])
+        src_y = y_coords - dy
+        shifted = np.zeros_like(delta_img)
+        for c in range(3):
+            shifted[:, :, c] = map_coordinates(
+                delta_img[:, :, c],
+                [src_y, x_coords],
+                order=1,
+                mode="constant",
+                cval=0.0,
+            )
+        return shifted

+    def render(dy):
+        if dy == 0.0:
