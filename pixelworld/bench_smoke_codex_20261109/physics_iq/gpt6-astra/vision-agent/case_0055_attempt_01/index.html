# These are projections of rigid rotations about each tile's lower edge.
def vertices(b, angle):
    th=math.radians(angle)
    B,D,T,U=[np.array(b[k],float) for k in ('B','D','T','U')]
    ground=np.array([.994,-.272])
    newU=U*math.cos(th)-b['height']*ground*math.sin(th)
    newT=T*math.cos(th)+(U/b['height'])*b['thick']*math.sin(th)
    return np.array([B,B+newT,B+D+newT,B+D,B+newU,B+newT+newU,B+D+newT+newU,B+D+newU])

def angle_at(b,t):
    u=(t-b['start'])/b['duration']
    if u<=0:return 0.
    if u<1:
        # The angular acceleration of a released upright tile increases as it
        # falls; the small initial velocity is supplied by the previous tile.
        return b['end']*(.24*u+.76*u*u)
    dt=t-b['start']-b['duration']
    # A short damped contact rebound, not slow motion.
    return b['end']-1.25*math.exp(-dt*12)*max(0,math.sin(dt*27))

def put_face(canvas, points, texture, gain=1.0):
    p=np.float32(points)*S
    # Avoid degenerate faces at the instant a surface turns away.
    if abs(cv2.contourArea(p))<.3:return canvas
    th,tw=texture.shape[:2]
    src=np.float32([[0,th-1],[tw-1,th-1],[tw-1,0],[0,0]])
    mat=cv2.getPerspectiveTransform(src,p)
    tex=np.uint8(np.clip(texture.astype(float)*gain,0,255))
    warp=cv2.warpPerspective(tex,mat,SZ,flags=cv2.INTER_CUBIC,borderMode=cv2.BORDER_REPLICATE)
    mask=Image.new('L',SZ,0)
    ImageDraw.Draw(mask).polygon([tuple(a) for a in p],fill=255)
    canvas.paste(Image.fromarray(warp),(0,0),mask)
    return canvas

def draw_block(canvas,b,angle):
    if angle==0:
        canvas.paste(SOURCE,(0,0),b['static_mask'])
        return canvas
    v=vertices(b,angle)
    # Visibility is determined by the signed area of each projected cuboid face.
    for name in ['right','left','top','front']:
        inds=faces[name]
        p=v[inds]
        area=cv2.contourArea(np.float32(p),oriented=True)
        if area>=0:continue
        gain=1.0
        if name=='left':gain=1-.0015*angle
        if name=='right':gain=1+.0018*angle
        if name=='top':gain=1-.0020*angle
        canvas=put_face(canvas,p,b['tex'][name],gain)
    return canvas
