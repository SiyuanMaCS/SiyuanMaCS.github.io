def signedarea(p):
    return .5*np.sum(p[:,0]*np.roll(p[:,1],-1)-p[:,1]*np.roll(p[:,0],-1))


def putface(canvas, source, destination, gain=1.0, fallback=None):
    if abs(signedarea(destination))<.035:
        return
    source=np.asarray(source,np.float32)*S
    dest=np.asarray(destination,np.float32)*S
    matrix=cv2.getPerspectiveTransform(source,dest)
    if fallback is None:
        warped=cv2.warpPerspective(src_hi,matrix,HI,flags=cv2.INTER_LINEAR,borderMode=cv2.BORDER_REPLICATE)
    else:
        warped=np.empty_like(canvas);warped[:]=fallback
    poly=mask_poly(destination,S).astype(np.float32)/255
    # Rasterize at four-times resolution for smooth, subpixel box edges.
    canvas[:]=canvas*(1-poly[...,None])+np.clip(warped*gain,0,255)*poly[...,None]


def draw_block(canvas,b,theta):
    if theta<.001:
        m=b['srcmask_hi'][...,None]
        canvas[:]=canvas*(1-m)+original_hi*m
        return
    old=b['verts']; new=corners(b,theta)
    # The projected winding selects the camera-facing surfaces of the box.
    # Broad face, thin bright edge and narrow top all retain their own shading.
    for ids in faces:
        ids=np.array(ids)
        p=new[ids]
        if signedarea(p)>0:
            continue
        src=old[ids]
        # Back faces cannot have a reliable source texture, but only become
        # visible close to horizontal; use the matching face on the other side.
        if tuple(ids)==(2,3,7,6):
            src=old[np.array([1,0,4,5])]
        elif tuple(ids)==(3,0,4,7):
            src=old[np.array([2,1,5,6])]
        gain=1+(0.13 if b['white'] else .05)*np.sin(theta)
        putface(canvas,src,p,gain)


def draw_rod(canvas,t):
    pivot=np.array([195.5,32.0])
    angle=math.radians(75.5+61*t)
    tip=pivot+np.array([22.0*math.cos(angle),10.55*math.sin(angle)])
    v=tip-pivot
    n=np.array([-v[1],v[0]])/(np.linalg.norm(v)+1e-8)
    def line(a,b,color,width):
        cv2.line(canvas,tuple(np.rint(a*S).astype(int)),tuple(np.rint(b*S).astype(int)),color,max(1,round(width*S)),cv2.LINE_AA)
    # Contact shadow and the rounded, subtly shaded wooden surface.
    line(pivot+[.25,1],tip+[.25,1],(10.,6.,5.),2.2)
    line(pivot,tip,(103.,78.,56.),2.15)
    line(pivot-n*.28,tip-n*.28,(161.,135.,110.),1.48)
    line(pivot-n*.60,tip-n*.60,(185.,159.,133.),.40)
    line(pivot+n*.42,tip+n*.42,(132.,104.,78.),.40)
    cv2.circle(canvas,tuple(np.rint(pivot*S).astype(int)),max(2,round(.67*S)),(134.,111.,92.),-1,cv2.LINE_AA)


starts=[.13,.385,.605,.865]
gaps=[10.,9.,10.8]
final_last=.46

def angles(t):
    a=[]
    for start in starts:
        q=max(0,t-start)
        a.append(min(1.53,.70*q+5.3*q*q))
    # Nonpenetration of the final block against the much larger rubber duck.
    a[3]=min(a[3],final_last)
    # Each back block comes to rest on the next.  This is the contact
    # geometry of equal-height rods pivoting about fixed bottom hinges.
    for j in range(2,-1,-1):
        lim=a[j+1]+math.asin(min(.99,gaps[j]/22.2*math.cos(a[j+1])))
        a[j]=min(a[j],lim)
    # Very small damped contact settling, without moving the heavy duck.
    if t>1.14:
        wob=.011*math.exp(-(t-1.14)*6)*math.sin((t-1.14)*28)
        a=[x+wob*(1-.14*j) for j,x in enumerate(a)]
    return a


def render(t):
    canvas=plate_hi.copy()
    aa=angles(t)
    # Moving soft contact shadows projected onto the tabletop, away from the
    # front of the raised turntable.
    shadows=np.zeros((H*S,W*S),np.float32)
    for b,a in zip(blocks,aa):
        if a<.005: continue
        v=corners(b,a)
        d=b['d'];u=b['u']
        reach=b['fall']*np.sin(a)
        p=np.array([d,d+u,d+u+reach,d+reach])+[.7,1.25]
        cv2.fillPoly(shadows,[np.rint(p*S).astype(np.int32)],.18)
    shadows=cv2.GaussianBlur(shadows,(0,0),1.1*S)
    # A cast tabletop shadow must not darken the vertical platform or the duck.
    table_gate=cv2.resize(((yy>53)&(xx>166)).astype(np.float32),HI)
    canvas*=1-shadows[...,None]*table_gate[...,None]
    draw_rod(canvas,t)
    for b,a in zip(blocks,aa):
        draw_block(canvas,b,a)
    # Foreground occlusion at the point where the cascade meets the duck.
    canvas=canvas*(1-duck_hi[...,None])+original_hi*duck_hi[...,None]
    result=cv2.resize(np.clip(canvas,0,255),(W,H),interpolation=cv2.INTER_AREA)
    return np.rint(result).clip(0,255).astype(np.uint8)

    # camera and a sharp static background are retained throughout.
