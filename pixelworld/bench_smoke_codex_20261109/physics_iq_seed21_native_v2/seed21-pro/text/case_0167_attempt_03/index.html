+"""
+PixelWorld case generator.
+
+The scene is a locked-off shot of a smiley-face tennis ball resting on a
+black platform. The only motion is the ball rotating slowly clockwise in
+place. Everything outside the ball is static, and the lighting/shading on
+the sphere stays fixed in screen space while the felt texture and the dark
+smiley markings rotate with the ball.
+"""
+
+from pathlib import Path
+
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+CASE_DIR = Path(__file__).resolve().parent
+N_FRAMES = 20
+FPS = 4
+TOTAL_TURN_DEG = 22.5  # slow clockwise turn over the 5 s clip
+
+# Ball geometry, measured from first_frame.png.
+CX, CY, RADIUS = 129.0, 57.0, 18.75
+INK_RGB = np.array([34.0, 30.0, 22.0])  # darkest smiley markings in frame 0
+
+
+def main():
+    first = np.array(Image.open(CASE_DIR / "first_frame.png").convert("RGB"))
+    height, width = first.shape[:2]
+    yy, xx = np.mgrid[0:height, 0:width]
+    dx = xx - CX
+    dy = yy - CY
+    radial = np.sqrt(dx * dx + dy * dy)
+
+    disc = radial <= RADIUS
+    r = first[..., 0].astype(float)
+    g = first[..., 1].astype(float)
+    b = first[..., 2].astype(float)
+
+    felt = (g >= 110) & (r >= 110) & (b < 115)
+    dark = (r < 85) & (g < 85) & (b < 75)
+    # Face markings are dark pixels inside the upper part of the ball;
+    # the dark platform only occludes a small cap near the bottom.
+    mark = disc & dark & (yy <= 71)
+    visible0 = disc & (felt | dark)
+
+    # Smooth per-channel shading (degree-2 polynomial over the disc),
+    # fitted on real felt pixels so the marks are excluded.
+    nx = dx / RADIUS
+    ny = dy / RADIUS
+    basis = np.stack(
+        [np.ones_like(nx), nx, ny, nx * nx, nx * ny, ny * ny], axis=-1
+    )
+    train = disc & ~mark
+    shading = np.zeros_like(first, dtype=float)
+    sigmas = []
+    for ch in range(3):
+        coeffs, *_ = np.linalg.lstsq(basis[train], first[..., ch][train], rcond=None)
+        shading[..., ch] = basis @ coeffs
+        sigmas.append((first[..., ch][train] - shading[..., ch][train]).std())
+
+    # Coherent low-frequency felt noise, generated once and rotated with the
+    # ball so the surface reads as a physical textured sphere.
+    rng = np.random.default_rng(20261113)
+    side = int(np.ceil(RADIUS * 2)) + 6
+    low = rng.normal(0.0, 1.0, (side // 2 + 1, side // 2 + 1, 3))
+    low_img = Image.fromarray(
+        np.clip((low * 40 + 128), 0, 255).astype(np.uint8), mode="RGB"
+    )
+    noise_full = np.array(
+        low_img.resize((side, side), Image.Resampling.BILINEAR)
+    ).astype(float) - 128.0
+
+    x0, y0 = int(round(CX)) - side // 2, int(round(CY)) - side // 2
+    noise = np.zeros_like(first, dtype=float)
+    noise[y0 : y0 + side, x0 : x0 + side] = noise_full
+    for ch, sigma in enumerate(sigmas):
+        noise[..., ch] *= 0.55 * sigma / 40.0
+
+    cleaned = first.astype(float).copy()
+    fill = np.clip(shading + noise, 0, 255)
+    cleaned[disc] = fill[disc]

+    cleaned[mark] = INK_RGB
+

+                resample=Image.Resampling.BILINEAR,
+            )
+        )
+

+        frame[visible0] = warped[visible0]
+        frames.append(frame)
+
