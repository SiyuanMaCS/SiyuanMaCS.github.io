+"""Generate the 20-frame RGB video for the grabber-release scene.
+
+Approach
+--------
+Frame 00 is taken verbatim from first_frame.png. The two held objects (a brown
+tennis ball on the left and an orange block on the right) are isolated with
+local color thresholds, their original footprints are inpainted from the
+surrounding pillow/background pixels, and the sprites are then animated with a
+short gravity release followed by a damped settle onto the pillows. Everything
+is analytic/numeric RGB work - no external models or renderers.
+"""
+
+from pathlib import Path
+
+import numpy as np
+from PIL import Image
+from scipy import ndimage
+
+import artifact_io
+
+CASE_DIR = Path(__file__).resolve().parent
+N_FRAMES, FPS, H, W = 20, 4, 144, 256
+DT = 1.0 / FPS
+GRAVITY = 9.0  # px / s^2
+
+
+def build_sprites(frame):
+    """Return (ball, block) dicts with rgb sprites and alpha masks."""
+    a = frame.astype(int)
+    r, g, b = a[..., 0], a[..., 1], a[..., 2]
+    warm = r - b
+
+    # --- Brown tennis ball: warm, darker pixels inside its known region.
+    region = np.zeros((H, W), dtype=bool)
+    region[17:48, 68:99] = True
+    cand = region & (warm > 4) & (r < 120) & (r > 18)
+    lab, n = ndimage.label(cand)
+    sizes = ndimage.sum(cand, lab, range(1, n + 1))
+    ball_mask = ndimage.binary_fill_holes(lab == (1 + int(np.argmax(sizes))))
+    # Regularise to a clean disc using the fitted centre / radius.
+    ys, xs = np.where(ball_mask)
+    cx, cy = xs.mean(), ys.mean()
+    radius = 0.5 * ((xs.max() - xs.min() + 1) + (ys.max() - ys.min() + 1)) / 2
+    yy, xx = np.mgrid[0:H, 0:W]
+    ball_mask = np.hypot(xx - cx, yy - cy) <= radius + 0.35
+
+    # --- Orange block: bright warm rectangle in its known region.
+    region = np.zeros((H, W), dtype=bool)
+    region[31:64, 208:224] = True
+    cand = region & (warm > 10) & (r > 55)
+    lab, n = ndimage.label(cand)
+    sizes = ndimage.sum(cand, lab, range(1, n + 1))
+    block_mask = ndimage.binary_fill_holes(lab == (1 + int(np.argmax(sizes))))
+
+    def crop(mask):
+        ys, xs = np.where(mask)
+        return xs.min(), xs.max() + 1, ys.min(), ys.max() + 1
+
+    bx0, bx1, by0, by1 = crop(ball_mask)
+    kx0, kx1, ky0, ky1 = crop(block_mask)
+    ball = {
+        "rgb": frame[by0:by1, bx0:bx1].copy(),
+        "alpha": ball_mask[by0:by1, bx0:bx1].copy(),
+        "cx": cx - bx0,
+        "cy": cy - by0,
+        "x0": bx0,
+        "y0": by0,
+        "radius": radius,
+    }
+    bys, bxs = np.where(block_mask)
+    block = {
+        "rgb": frame[ky0:ky1, kx0:kx1].copy(),
+        "alpha": block_mask[ky0:ky1, kx0:kx1].copy(),
+        "cx": bxs.mean() - kx0,
+        "cy": bys.mean() - ky0,
+        "x0": kx0,
+        "y0": ky0,
+        "half_w": (kx1 - kx0) / 2.0,
+        "half_h": (ky1 - ky0) / 2.0,
+    }
+    return ball, block, ball_mask, block_mask
+
+
+def inpaint(frame, masks):
+    """Fill masked footprints from nearest known pixels, then relax them."""
+    unknown = np.zeros((H, W), dtype=bool)
+    for mask in masks:
+        unknown |= ndimage.binary_dilation(mask, iterations=2)
+    # Nearest-neighbour fill via EDT indices.
+    _, indices = ndimage.distance_transform_edt(unknown, return_indices=True)
+    filled = frame.copy()
+    filled[unknown] = frame[tuple(indices[:, unknown])]
+    # Relax the fill with neighbour averaging so gradients stay natural.
+    work = filled.astype(np.float32)
+    relax = ndimage.binary_dilation(unknown, iterations=3)
+    for _ in range(60):
+        avg = ndimage.uniform_filter(work, size=3, mode="nearest")
+        work[relax] = avg[relax]
+    out = frame.copy().astype(np.float32)
+    out[unknown] = work[unknown]
+    return np.clip(out, 0, 255).astype(np.uint8)
+
+
+def blit(canvas, sprite, center_x, center_y):
+    """Alpha-blit a sprite so that (cx, cy) lands at the given centre."""
+    h, w = sprite["alpha"].shape
+    top = int(round(center_y - sprite["cy"]))
+    left = int(round(center_x - sprite["cx"]))
+    for sy in range(h):
+        dy = top + sy
+        if dy < 0 or dy >= H:
+            continue
+        for sx in range(w):
+            dx = left + sx
+            if dx < 0 or dx >= W or not sprite["alpha"][sy, sx]:
+                continue
+            canvas[dy, dx] = sprite["rgb"][sy, sx]
+
+
+def add_shadow(canvas, cx, ground_y, half_w, strength):
+    """A faint soft contact shadow on the pillow surface."""
+    if strength <= 0:
+        return
+    yy, xx = np.mgrid[0:H, 0:W]
+    ellipse = ((xx - cx) / max(half_w, 1.0)) ** 2 + ((yy - ground_y) / 2.2) ** 2
+    alpha = np.clip(1.0 - ellipse, 0.0, 1.0) * strength
+    alpha = alpha[..., None]
+    canvas[:] = np.clip(canvas.astype(np.float32) * (1.0 - 0.28 * alpha), 0, 255)
+
+
+def simulate(contact_center, rest_center, start_center):
+    """Gravity release then a damped spring settle once contact is reached."""
+    y = start_center
+    v = 0.0
+    contact = False
+    positions = []
+    # Frame 0 keeps the original pose; motion starts on release at frame 1.
+    for i in range(N_FRAMES):
+        if i == 0:
+            positions.append(start_center)
+            continue
+        if not contact:
+            v += GRAVITY * DT
+            y += v * DT
+            if y >= contact_center:
+                y = contact_center
+                v *= -0.35  # small cushion rebound
+                contact = True
+        else:
+            k, c = 55.0, 12.0
+            accel = -k * (y - rest_center) - c * v
+            v += accel * DT
+            y += v * DT
+            if abs(y - rest_center) < 0.05 and abs(v) < 0.4:
+                y, v = rest_center, 0.0
+        positions.append(y)
+    return positions
+
+
+def main():
+    frame0 = np.array(Image.open(CASE_DIR / "first_frame.png").convert("RGB"))
+    ball, block, ball_mask, block_mask = build_sprites(frame0)
+    background = inpaint(frame0, [ball_mask, block_mask])
+
+    # Contact / rest geometry measured from the pillow surfaces in frame 0.
+    ball_start = ball["y0"] + ball["cy"]
+    ball_contact = 46.0 - ball["radius"]          # pillow top under the ball
+    ball_rest = ball_contact + 2.0                # slight soft compression
+    block_start = block["y0"] + block["cy"]
+    block_contact = 63.0 - block["half_h"]        # pillow top under the block
+    block_rest = block_contact + 2.5
+
+    ball_y = simulate(ball_contact, ball_rest, ball_start)
+    block_y = simulate(block_contact, block_rest, block_start)
+    ball_x = ball["x0"] + ball["cx"]
+    block_x = block["x0"] + block["cx"]
+
+    frames = []
+    for i in range(N_FRAMES):
+        canvas = background.copy()
+        if i > 0:
+            add_shadow(canvas, ball_x, 46.0, ball["radius"] + 1.0, 0.55)
+            add_shadow(canvas, block_x, 63.0, block["half_w"] + 1.0, 0.55)
+        blit(canvas, ball, ball_x, ball_y[i])
+        blit(canvas, block, block_x, block_y[i])
+        if i == 0:
+            # Guarantee a pixel-identical first frame.
+            canvas = frame0.copy()
+        frames.append(canvas)
+
+    frames = np.stack(frames).astype(np.uint8)
+    artifact_io.save(frames, CASE_DIR)
+    print("saved", frames.shape)
+    print("ball y:", [round(v, 2) for v in ball_y])
+    print("block y:", [round(v, 2) for v in block_y])
+
+
+if __name__ == "__main__":
+    main()
