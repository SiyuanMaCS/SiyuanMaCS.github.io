+"""Roll a grey tennis ball out of the black pipe across the wood table.
+
+Locked-off camera: the background never changes, so every frame starts from
+first_frame.png and only the moving ball (plus its contact shadow) is painted.
+"""
+from pathlib import Path
+
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+CASE_DIR = Path(__file__).resolve().parent
+N_FRAMES, FPS = 20, 4
+WIDTH, HEIGHT = 256, 144
+
+# Ball motion (image coordinates, y grows downward).
+R = 9                       # ball radius in pixels
+CENTER_Y = 86.0            # pipe-mouth / tabletop height
+X_START = 32.0             # frame 1: just outside the pipe mouth
+X_END = 232.0              # frame 19: far right end of the table
+
+BASE_GREY = np.array([146.0, 148.0, 151.0])
+LIGHT_DIR = np.array([-0.45, -0.55, 0.70])
+LIGHT_DIR /= np.linalg.norm(LIGHT_DIR)
+
+
+def hash_noise(size, seed=7):
+    """Small deterministic per-pixel felt grain in [-1, 1]."""
+    rng = np.random.default_rng(seed)
+    return rng.uniform(-1.0, 1.0, size=(size, size))
+
+
+def render_ball(radius, rotation):
+    """Render an RGBA patch of a shaded, rotating grey tennis ball."""
+    pad = 2
+    s = 2 * radius + 2 * pad + 1
+    cx = cy = s // 2
+    yy, xx = np.mgrid[0:s, 0:s]
+    dx = xx - cx
+    dy = yy - cy
+    dist = np.hypot(dx, dy)
+    inside = dist <= radius
+    nz2 = np.clip(1.0 - (dist / radius) ** 2, 0.0, 1.0)
+    nz = np.sqrt(nz2)
+
+    # Sphere normals and simple key-light shading.
+    nx = dx / radius
+    ny = dy / radius
+    diffuse = nx * LIGHT_DIR[0] + ny * LIGHT_DIR[1] + nz * LIGHT_DIR[2]
+    shade = 0.62 + 0.55 * np.clip(diffuse, 0.0, 1.0)
+
+    rgb = BASE_GREY[None, None, :] * shade[..., None]
+    rgb += hash_noise(s)[..., None] * 6.0  # felt texture
+
+    # Curved tennis-ball seam, rotating with the roll.
+    theta = np.arctan2(ny, nx)
+    seam_wave = np.sin(2.0 * theta - rotation)
+    seam = inside & (np.abs(seam_wave) < 0.30) & (nz > 0.25)
+    seam_shade = np.where(seam_wave >= 0.0, 0.78, 1.18)
+    rgb[seam] *= seam_shade[seam, None]
+
+    # Soft fuzzy edge.
+    alpha = np.clip(radius + 0.8 - dist, 0.0, 1.0)
+    rgb = np.clip(rgb, 0, 255)
+    patch = np.dstack([rgb, alpha * 255.0]).astype(np.uint8)
+    return patch, pad
+
+
+def render_shadow(radius):
+    """Soft contact shadow rendered as an RGBA ellipse."""
+    w, h = 2 * radius + 7, 9
+    yy, xx = np.mgrid[0:h, 0:w]
+    ex = (xx - (w - 1) / 2.0) / (radius + 2.5)
+    ey = (yy - (h - 1) / 2.0) / 3.2
+    d = np.hypot(ex, ey)
+    alpha = np.clip(1.0 - d, 0.0, 1.0) ** 1.5 * 90.0
+    rgb = np.zeros((h, w, 3))
+    return np.dstack([rgb, alpha]).astype(np.uint8)
+
+
+def composite(frame, patch, top, left):
+    """Alpha-composite an RGBA patch onto an RGB frame."""
+    h, w = patch.shape[:2]
+    roi = frame[top:top + h, left:left + w].astype(np.float32)
+    a = patch[..., 3:4].astype(np.float32) / 255.0
+    blended = patch[..., :3].astype(np.float32) * a + roi * (1.0 - a)
+    frame[top:top + h, left:left + w] = blended.astype(np.uint8)
+
+
+def main():
+    background = np.array(Image.open(CASE_DIR / "first_frame.png").convert("RGB"))
+    assert background.shape == (HEIGHT, WIDTH, 3)
+    shadow_patch = render_shadow(R)
+
+    frames = [background.copy()]
+    for i in range(1, N_FRAMES):
+        t = i / (N_FRAMES - 1)
+        cx = X_START + (X_END - X_START) * t
+        travel = cx - X_START
+        rotation = -travel / R  # rolling without slipping
+
+        frame = background.copy()
+
+        # Contact shadow sits on the tabletop just under the ball.
+        sh = shadow_patch
+        s_left = int(round(cx - (sh.shape[1] - 1) / 2.0))
+        s_top = int(round(CENTER_Y + R - 3))
+        composite(frame, sh, s_top, s_left)
+
+        ball_patch, pad = render_ball(R, rotation)
+        b_left = int(round(cx)) - ball_patch.shape[1] // 2
+        b_top = int(round(CENTER_Y)) - ball_patch.shape[0] // 2
+        composite(frame, ball_patch, b_top, b_left)
+
+        frames.append(frame)
+
+    frames = np.stack(frames, axis=0)
+    assert np.array_equal(frames[0], background)
+    artifact_io.save(frames, CASE_DIR)
+    print(f"Saved {len(frames)} frames to {CASE_DIR}")
+
+
+if __name__ == "__main__":
+    main()
