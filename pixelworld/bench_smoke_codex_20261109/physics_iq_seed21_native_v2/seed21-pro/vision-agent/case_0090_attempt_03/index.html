+"""Generate the 20-frame RGB video for the pouring beverage dispenser case.
+
+Locked-off camera: frame 00 is pixel-identical to first_frame.png. Later
+frames animate the red pour (stream, impact ripples, splash droplets,
+sinking bubbles and gentle internal flow) strictly inside the silhouette
+of the red liquid already visible in the first frame.
+"""
+from collections import deque
+
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+CASE_DIR = "."
+N_FRAMES, FPS = 20, 4
+HEIGHT, WIDTH = 144, 256
+
+STREAM_X = 128
+SURFACE_Y = 18
+
+
+def flood_fill(mask, seed):
+    seen = np.zeros_like(mask, dtype=bool)
+    queue = deque([seed])
+    seen[seed] = True
+    h, w = mask.shape
+    while queue:
+        y, x = queue.popleft()
+        for dy, dx in ((1, 0), (-1, 0), (0, 1), (0, -1)):
+            ny, nx = y + dy, x + dx
+            if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
+                seen[ny, nx] = True
+                queue.append((ny, nx))
+    return seen
+
+
+def dilate(mask):
+    out = mask.copy()
+    for dy in (-1, 0, 1):
+        for dx in (-1, 0, 1):
+            out |= np.roll(np.roll(mask, dy, 0), dx, 1)
+    return out
+
+
+def erode(mask):
+    out = mask.copy()
+    for dy in (-1, 0, 1):
+        for dx in (-1, 0, 1):
+            out &= np.roll(np.roll(mask, -dy, 0), -dx, 1)
+    return out
+
+
+def build_masks(base):
+    r = base[..., 0].astype(int)
+    g = base[..., 1].astype(int)
+    b = base[..., 2].astype(int)
+
+    liquid_like = (r > 140) & (r - g > 30) & (r - b > 30)
+    region = np.zeros_like(liquid_like)
+    region[14:95, 100:160] = True
+    component = flood_fill(liquid_like & region, (50, STREAM_X))
+
+    glass_box = np.zeros_like(component)
+    glass_box[14:95, 105:156] = True
+    silhouette = dilate(component) & glass_box
+
+    blue = (b > 120) & (b - r > 40) & (b - g > 20)
+    protected = dilate(blue)
+
+    interior = erode(silhouette) & ~protected
+    interior[:15, :] = False
+    stream_zone = silhouette & ~protected
+    stream_zone[:15, :] = False
+    return interior, stream_zone, protected
+
+
+def main():
+    base = np.array(Image.open("first_frame.png").convert("RGB"))
+    interior, stream_zone, protected = build_masks(base)
+
+    yy, xx = np.mgrid[0:HEIGHT, 0:WIDTH]
+    base_f = base.astype(np.float32)
+
+    # Static specular glints stay untouched so the glass looks fixed.
+    glint = (base_f[..., 0] > 225) & (base_f[..., 1] > 225) & (base_f[..., 2] > 225)
+    editable = interior & ~glint
+
+    stream_color = np.array([255.0, 132.0, 98.0], dtype=np.float32)
+    bright_color = np.array([255.0, 165.0, 135.0], dtype=np.float32)
+    bubble_color = np.array([255.0, 190.0, 170.0], dtype=np.float32)
+
+    # Deterministic splash droplets: (phase, vx, vy) launched from the surface.
+    droplets = [(0.0, -7.0, -16.0), (0.35, 5.5, -13.0), (0.8, -4.0, -11.0),
+                (1.25, 7.5, -15.0), (1.7, -6.0, -12.0)]
+    # Deterministic entrained bubbles with staggered phases.
+    bubbles = [0.0, 0.5, 1.05, 1.6, 2.1]
+
+    frames = [base.copy()]
+    for i in range(1, N_FRAMES):
+        t = i / FPS
+        frame = base_f.copy()
+        delta = np.zeros_like(base_f)
+        blend = np.zeros((HEIGHT, WIDTH), dtype=np.float32)
+        toward = np.zeros_like(base_f)
+
+        # Wobbling centerline of the pour.
+        xc = STREAM_X + 0.7 * np.sin(2 * np.pi * 1.3 * t + 0.5)
+        dist = xx - xc
+
+        # 1) Visible stream between spout and liquid surface (y 15..18).
+        flow_phase = 2.2 * yy - 10.0 * t
+        modulation = 0.65 + 0.35 * np.sin(flow_phase)
+        above = stream_zone & (yy >= 15) & (yy <= 18)
+        core = np.exp(-(dist / 1.5) ** 2) * modulation
+        a = np.clip(core * 0.9, 0, 0.92) * above
+        toward += a[..., None] * stream_color
+        blend += a
+
+        # 2) Submerged turbulent jet, fading with depth.
+        depth_fade = np.clip(1.6 - (yy - SURFACE_Y) / 34.0, 0, 1)
+        jet = np.exp(-(dist / 2.3) ** 2) * depth_fade * editable
+        jet *= 0.55 + 0.45 * np.sin(1.6 * yy - 6.0 * t + 0.9 * np.sin(xx * 0.5))
+        a = np.clip(jet * 0.55, 0, 0.6)
+        toward += a[..., None] * stream_color
+        blend += a
+
+        # 3) Pulsing glow at the impact point.
+        d_impact = np.hypot(dist, (yy - SURFACE_Y) * 1.5)
+        pulse = 0.32 + 0.14 * np.sin(2 * np.pi * 2.0 * t)
+        glow = np.exp(-(d_impact / 4.2) ** 2) * pulse * editable
+        toward += glow[..., None] * bright_color
+        blend += glow
+
+        # 4) Expanding surface ripples.
+        d_ring = np.hypot(dist, (yy - SURFACE_Y) * 1.7)
+        ring = np.sin(d_ring * 1.9 - 7.0 * t)
+        ring_weight = np.exp(-d_ring / 15.0) * np.clip(1.25 - (yy - 16) / 16.0, 0, 1)
+        ripple = ring * ring_weight * editable
+        delta[..., 0] += ripple * 7.0
+        delta[..., 1] += ripple * 4.5
+        delta[..., 2] += ripple * 3.5
+
+        # 5) Splash droplets following ballistic arcs above the surface.
+        for phase, vx, vy in droplets:
+            tau = (t + phase) % 2.0
+            if tau <= 0.55:
+                dx = vx * tau
+                dy = vy * tau + 0.5 * 46.0 * tau * tau
+                x = int(round(STREAM_X + dx))
+                y = int(round(SURFACE_Y + dy))
+                if 0 <= x < WIDTH and 0 <= y < HEIGHT and stream_zone[y, x]:
+                    toward[y, x] = bright_color
+                    blend[y, x] = max(blend[y, x], 0.85)
+
+        # 6) Entrained air bubbles sinking along the jet.
+        for phase in bubbles:
+            q = ((t + phase) % 2.5) / 2.5
+            y = int(round(SURFACE_Y + q * 44))
+            x = int(round(STREAM_X + 3.0 * np.sin(q * 2 * np.pi + phase * 7.0)))
+            fade = np.sin(np.pi * q)
+            if 0 <= x < WIDTH and 0 <= y < HEIGHT and editable[y, x] and fade > 0.25:
+                toward[y, x] = bubble_color
+                blend[y, x] = max(blend[y, x], 0.55 * fade)
+
+        # 7) Gentle internal flow: drifting vertical streaks and slow shimmer.
+        center_weight = np.exp(-((xx - STREAM_X) ** 2) / 300.0)
+        streaks = np.sin(0.45 * yy - 2.5 * t + 0.7 * np.sin(0.6 * xx))
+        streaks *= np.sin(0.35 * xx + 1.7 * t)
+        shimmer = np.sin(0.22 * yy + 0.31 * xx - 1.3 * t)
+        noise = (streaks * 6.0 * center_weight + shimmer * 2.0) * editable
+        delta[..., 0] += noise
+        delta[..., 1] += noise * 0.7
+        delta[..., 2] += noise * 0.55
+
+        # Combine: color-target blending plus additive brightness modulation.
+        blend = np.clip(blend, 0, 0.95)[..., None]
+        frame = base_f * (1.0 - blend) + toward * blend + delta
+        frame = np.clip(frame, 0, 255)
+
+        # Everything outside the editable/stream zones stays exactly frame 0.
+        allowed = (editable | (stream_zone & (yy >= 15) & (yy <= 18))) & ~protected
+        out = base_f.copy()
+        out[allowed] = frame[allowed]
+        frames.append(out.astype(np.uint8))
+
+    frames = np.stack(frames, axis=0)
+    assert frames.shape == (N_FRAMES, HEIGHT, WIDTH, 3)
+    assert np.array_equal(frames[0], base), "frame 00 must be pixel-identical"
+    artifact_io.save(frames, CASE_DIR)
+
+
+if __name__ == "__main__":
+    main()
