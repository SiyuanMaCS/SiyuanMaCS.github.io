+"""Text-to-video: a diagonal ray of light on a glass bottle in the dark.
+
+Pure numpy/PIL rendering at 2x supersampling, downscaled to 256x144.
+The scene is static apart from light flicker, drifting dust motes and
+slow shimmer, so the bottle and its ground shadow stay firmly anchored.
+"""
+from pathlib import Path
+
+import numpy as np
+from PIL import Image, ImageDraw, ImageFilter
+
+import artifact_io
+
+SEED = 21043
+TAU = 2 * np.pi
+rng = np.random.default_rng(SEED)
+
+W, H, N, FPS = 256, 144, 20, 4.0
+S = 2                      # supersampling factor
+SW, SH = W * S, H * S
+
+CASE_DIR = Path(__file__).parent
+
+# ---------------------------------------------------------------- geometry
+# Light comes from the upper-left, travels down-right onto the bottle.
+LIGHT_ORIGIN = np.array([-110.0, -170.0])
+LIGHT_TARGET = np.array([300.0, 196.0])
+d = LIGHT_TARGET - LIGHT_ORIGIN
+d = d / np.linalg.norm(d)
+
+CX = 300.0                 # bottle center x (2x coords)
+BASE_Y = 238.0             # bottle base y
+HORIZON_Y = 212.0          # where the back wall meets the ground
+
+# Bottle profile (2x coords): key y -> half width
+PROFILE = [
+    (50.0, 23.0),    # lip top
+    (60.0, 23.0),    # lip bottom
+    (60.0, 17.0),
+    (94.0, 17.0),    # neck
+    (122.0, 45.0),   # shoulder
+    (BASE_Y, 45.0),  # body
+]
+
+
+def half_width_at(y):
+    """Piecewise-linear bottle half-width for an array of y coordinates."""
+    hw = np.zeros_like(y, dtype=np.float64)
+    for (y0, w0), (y1, w1) in zip(PROFILE[:-1], PROFILE[1:]):
+        m = (y >= y0) & (y < y1)
+        hw[m] = w0 + (w1 - w0) * (y[m] - y0) / max(y1 - y0, 1e-9)
+    hw[y >= PROFILE[-1][0]] = PROFILE[-1][1]
+    return hw
+
+
+def build_bottle_mask():
+    layer = Image.new("L", (SW, SH), 0)
+    dc = ImageDraw.Draw(layer)
+    # Body with rounded bottom corners.
+    dc.rounded_rectangle(
+        [CX - 45, 122, CX + 45, BASE_Y + 2],
+        radius=15, fill=255,
+    )
+    # Shoulder polygon.
+    dc.polygon([
+        (CX - 45, 124), (CX + 45, 124),
+        (CX + 17, 94), (CX - 17, 94),
+    ], fill=255)
+    # Neck and lip.
+    dc.rectangle([CX - 17, 60, CX + 17, 96], fill=255)
+    dc.rectangle([CX - 23, 50, CX + 23, 62], fill=255)
+    layer = layer.filter(ImageFilter.GaussianBlur(1.2))
+    return np.asarray(layer, dtype=np.float64) / 255.0
+
+
+def build_background():
+    """Dark room: faint vertical gradient, vignette, ground plane."""
+    yy, xx = np.mgrid[0:SH, 0:SW].astype(np.float64)
+    wall = np.zeros((SH, SW, 3))
+    wall[..., 0] = 5
+    wall[..., 1] = 8
+    wall[..., 2] = 13
+    glow_dir = np.clip(1.0 - (xx + yy) / 900.0, 0.0, 1.0)
+    wall += glow_dir[..., None] * np.array([7.0, 8.0, 9.0])
+
+    img = wall.copy()
+    ground = yy >= HORIZON_Y
+    depth = np.clip((yy - HORIZON_Y) / (SH - HORIZON_Y), 0, 1)
+    ground_col = np.zeros_like(img)
+    ground_col[..., 0] = 7 + 5 * depth
+    ground_col[..., 1] = 8 + 5 * depth
+    ground_col[..., 2] = 11 + 6 * depth
+    img[ground] = ground_col[ground]
+
+    # Faint horizon seam.
+    seam = np.exp(-((yy - HORIZON_Y) ** 2) / 6.0)[..., None]
+    img = img * (1 - seam * 0.5) + seam * np.array([3.0, 4.0, 7.0])
+
+    # Vignette.
+    r2 = ((xx - SW / 2) / (SW / 2)) ** 2 + ((yy - SH / 2) / (SH / 2)) ** 2
+    vig = np.clip(1.0 - 0.45 * r2, 0.45, 1.0)
+    img *= vig[..., None]
+
+    # Fine fixed sensor grain.
+    grain = rng.normal(0, 1.6, (SH, SW, 1))
+    img += grain
+    return np.clip(img, 0, 255)
+
+
+def beam_field(xx, yy):
+    """Signed perpendicular distance and longitudinal t of the beam."""
+    vx = xx - LIGHT_ORIGIN[0]
+    vy = yy - LIGHT_ORIGIN[1]
+    t = vx * d[0] + vy * d[1]
+    perp = vx * d[1] - vy * d[0]      # signed cross product
+    return t, perp
+
+
+def soft_ellipse(center, rx, ry, angle_deg=0.0, blur=0.0):
+    """Grayscale soft ellipse mask on an SW x SH grid."""
+    layer = Image.new("L", (SW, SH), 0)
+    dc = ImageDraw.Draw(layer)
+    cx, cy = center
+    dc.ellipse([cx - rx, cy - ry, cx + rx, cy + ry], fill=255)
+    if angle_deg:
+        layer = layer.rotate(angle_deg, center=(cx, cy), resample=Image.BICUBIC)
+    if blur:
+        layer = layer.filter(ImageFilter.GaussianBlur(blur))
+    return np.asarray(layer, dtype=np.float64) / 255.0
+
+
+# Precompute static grids and masks.
+YY, XX = np.mgrid[0:SH, 0:SW].astype(np.float64)
+BG = build_background()
+BOTTLE_MASK = build_bottle_mask()
+T_GRID, PERP_GRID = beam_field(XX, YY)
+HW_GRID = half_width_at(YY)
+inside = BOTTLE_MASK > 0.5
+u = np.zeros_like(HW_GRID)                       # normalized lateral position
+u[inside] = np.abs(XX[inside] - CX) / np.maximum(HW_GRID[inside], 1e-6)
+side = np.sign(XX - CX)                          # -1 left (lit), +1 right
+
+# Dust motes living inside the beam.
+MOTES = 70
+mote_t = rng.uniform(120, 620, MOTES)
+mote_p = rng.normal(0, 46, MOTES)
+mote_speed = rng.uniform(6.0, 16.0, MOTES)
+mote_phase = rng.uniform(0, TAU, MOTES)
+mote_size = rng.uniform(1.1, 2.6, MOTES)
+
+LIGHT_RGB = np.array([255.0, 232.0, 190.0])     # warm ray
+COOL_RGB = np.array([150.0, 190.0, 205.0])      # glass rim tint
+
+
+def render_frame(frame_index):
+    t_sec = frame_index / FPS
+    flick = (1.0
+             + 0.055 * np.sin(TAU * 1.3 * t_sec + 0.7)
+             + 0.035 * np.sin(TAU * 3.1 * t_sec + 2.1)
+             + 0.02 * np.sin(TAU * 5.7 * t_sec))
+
+    img = BG.copy()
+    yy, xx = YY, XX
+
+    # ------------------------------------------------------- ground light pool
+    pool = soft_ellipse((CX + 18, BASE_Y + 6), 168, 30, angle_deg=24, blur=10)
+    img += pool[..., None] * LIGHT_RGB * 0.10 * flick
+
+    # ------------------------------------------------------------- shadows
+    shadow_layer = np.zeros((SH, SW))
+    # Long cast shadow, pushed down-right away from the upper-left light.
+    cast = soft_ellipse((CX + 86, BASE_Y + 20), 108, 15, angle_deg=21, blur=6)
+    shadow_layer += cast * 165
+    # Tight contact shadow under the base.
+    contact = soft_ellipse((CX, BASE_Y + 4), 52, 11, blur=5)
+    shadow_layer += contact * 120
+    shadow_layer = np.clip(shadow_layer, 0, 235) * (0.92 + 0.08 * flick)
+    ground = yy >= HORIZON_Y
+    img[ground] *= (1 - shadow_layer[ground, None] / 255.0)
+
+    # Faint caustic glint on the lit side of the base.
+    caustic = soft_ellipse((CX - 34, BASE_Y + 7), 30, 6, angle_deg=-12, blur=3)
+    img += caustic[..., None] * LIGHT_RGB * 0.16 * flick
+
+    # ------------------------------------------------------------- light beam
+    width = 78.0
+    core = np.exp(-(PERP_GRID / width) ** 2 * 2.2)
+    # Longitudinal envelope: fade in from off-screen, soften near the floor.
+    env = np.clip(T_GRID / 140.0, 0, 1) * np.clip((720 - T_GRID) / 160.0, 0, 1)
+    # Slow moving shafts inside the beam.
+    shafts = (0.82
+              + 0.12 * np.sin(PERP_GRID * 0.060 + t_sec * 1.9)
+              + 0.08 * np.sin(PERP_GRID * 0.137 - t_sec * 1.3 + 1.4))
+    noise = rng.normal(0, 1, (SH, SW))
+    noise = np.asarray(
+        Image.fromarray(((noise + 3) / 6 * 255).astype(np.uint8))
+        .filter(ImageFilter.GaussianBlur(7)), dtype=np.float64) / 255.0
+    beam = core * env * shafts * (0.75 + 0.5 * noise) * flick
+    img += beam[..., None] * LIGHT_RGB * 0.34
+
+    # Bright source glow bleeding in from the upper-left corner.
+    src = np.exp(-(((xx + 60) / 150.0) ** 2 + ((yy + 60) / 150.0) ** 2))
+    img += src[..., None] * LIGHT_RGB * 0.30 * flick
+
+    # ------------------------------------------------------------- glass bottle
+    bm = BOTTLE_MASK
+    # Dark translucent glass body over whatever is behind it.
+    body_col = np.zeros_like(img)
+    body_col[..., 0] = 9
+    body_col[..., 1] = 20
+    body_col[..., 2] = 25
+    fill_alpha = bm * 0.62
+    img = img * (1 - fill_alpha[..., None]) + body_col * fill_alpha[..., None]
+
+    # Light transmitted through the glass: strongest in the body core.
+    transmitted = beam * bm * (1.0 - 0.55 * u ** 2)
+    img += transmitted[..., None] * LIGHT_RGB * 0.55
+
+    # Fresnel-style rims. Left edge faces the light, right edge is cooler.
+    rim = np.clip((u - 0.72) / 0.28, 0, 1) ** 1.5 * bm
+    left_rim = rim * np.clip(-side, 0, 1)
+    right_rim = rim * np.clip(side, 0, 1)
+    img += left_rim[..., None] * LIGHT_RGB * 0.85 * flick
+    img += right_rim[..., None] * COOL_RGB * 0.30
+
+    # Vertical specular streak, slightly left of center.
+    streak_x = CX - 0.42 * HW_GRID
+    streak = np.exp(-((xx - streak_x) / 7.0) ** 2) * bm
+    vfade = np.clip((BASE_Y - yy) / 26.0, 0, 1) * np.clip((yy - 56) / 18.0, 0, 1)
+    streak *= vfade * (0.85 + 0.15 * np.sin(TAU * 2.2 * t_sec))
+    img += streak[..., None] * LIGHT_RGB * 0.55 * flick
+
+    # Secondary thin glint on the right shoulder.
+    glint2 = np.exp(-((xx - (CX + 0.55 * HW_GRID)) / 3.2) ** 2) * bm * vfade
+    img += glint2[..., None] * COOL_RGB * 0.35
+
+    # Lip band and thick glass base.
+    lip_band = np.exp(-((yy - 56) ** 2) / 10.0) * bm
+    img += lip_band[..., None] * LIGHT_RGB * 0.45 * flick
+    base_ring = soft_ellipse((CX, BASE_Y - 6), 40, 9, blur=3) * bm
+    img += base_ring[..., None] * COOL_RGB * 0.35
+
+    # Simple paper label, mostly in shadow with a lit edge.
+    label = ((np.abs(xx - CX) < 30) & (yy > 150) & (yy < 196) & (bm > 0.5))
+    label_col = np.array([46.0, 40.0, 32.0])
+    img[label] = img[label] * 0.35 + label_col * 0.65
+    lit_edge = label & (xx < CX - 18)
+    img[lit_edge] += np.array([26.0, 20.0, 10.0]) * flick
+    for ly in (152, 194):
+        line = label & (np.abs(yy - ly) < 1.2)
+        img[line] += np.array([14.0, 12.0, 9.0])
+
+    # ------------------------------------------------------------- dust motes
+    for i in range(MOTES):
+        tt = (mote_t[i] + mote_speed[i] * t_sec) % 640 + 60
+        px = LIGHT_ORIGIN[0] + d[0] * tt - d[1] * mote_p[i]
+        py = LIGHT_ORIGIN[1] + d[1] * tt + d[0] * mote_p[i]
+        if not (0 <= px < SW and 0 <= py < SH):
+            continue
+        env_p = np.exp(-(mote_p[i] / width) ** 2 * 2.2)
+        tw = 0.5 + 0.5 * np.sin(TAU * 1.7 * t_sec + mote_phase[i])
+        a = env_p * tw * 0.8 * flick
+        if a < 0.05:
+            continue
+        r = mote_size[i]
+        x0, x1 = max(int(px - r - 1), 0), min(int(px + r + 2), SW)
+        y0, y1 = max(int(py - r - 1), 0), min(int(py + r + 2), SH)
+        gx, gy = np.mgrid[y0:y1, x0:x1]
+        dot = np.exp(-(((gy - px) ** 2 + (gx - py) ** 2) / (r ** 2)))
+        img[y0:y1, x0:x1] += dot[..., None] * LIGHT_RGB * a
+
+    # A little grain so shadows don't band.
+    img += rng.normal(0, 0.9, img.shape)
+    img = np.clip(img, 0, 255).astype(np.uint8)
+
+    small = Image.fromarray(img).resize((W, H), Image.Resampling.LANCZOS)
+    return np.asarray(small, dtype=np.uint8)
+
+
+def main():
+    frames = [render_frame(i) for i in range(N)]
+    frames = np.stack(frames, axis=0)
+    artifact_io.save(frames, CASE_DIR)
+    print(f"Saved {frames.shape} frames to {CASE_DIR}")
+
+
+if __name__ == "__main__":
+    main()
