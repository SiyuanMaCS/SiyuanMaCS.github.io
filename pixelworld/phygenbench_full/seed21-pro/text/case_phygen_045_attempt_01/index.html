+"""Procedural RGB video: a diagonal light ray on a cardboard box in the dark.
+
+Renders 20 frames at 256x144 (4 fps) with PIL/numpy only. The scene is a dark
+room containing a cardboard box on a floor plane. A warm volumetric beam enters
+from the upper-left, lights the box, and casts a soft shadow on the ground to
+the lower-right. Motion comes from gentle beam flicker and drifting dust motes.
+"""
+import math
+from pathlib import Path
+
+import numpy as np
+from PIL import Image, ImageDraw, ImageFilter
+
+import artifact_io
+
+WIDTH, HEIGHT = 256, 144
+FRAMES, FPS = 20, 4.0
+CASE_DIR = Path(__file__).parent
+
+# ---------- scene geometry -------------------------------------------------
+
+# Box: front face rectangle plus a depth offset for the top / right faces.
+FRONT = [(78, 118), (150, 118), (150, 66), (78, 66)]
+DEPTH = np.array([20.0, -15.0])
+TOP = [(78, 66), (150, 66), (170, 51), (98, 51)]
+RIGHT = [(150, 66), (170, 51), (170, 103), (150, 118)]
+BOX_POLYS = [FRONT, TOP, RIGHT]
+
+# Ground contact quad of the box (front-bottom + right-bottom edges).
+CONTACT = [(78, 118), (150, 118), (170, 103), (98, 103)]
+SHADOW_SHIFT = np.array([42.0, 11.0])  # light comes from upper-left
+
+# Volumetric beam: narrow origin off the upper-left corner, wide target at box.
+BEAM_SOURCE = np.array([-24.0, -18.0])
+BEAM_TARGET_NEAR = np.array([66.0, 42.0])
+BEAM_TARGET_FAR = np.array([188.0, 84.0])
+
+rng = np.random.default_rng(21)
+DUST = []
+for _ in range(26):
+    DUST.append({
+        "u": rng.uniform(0.05, 0.95),       # position along beam
+        "w": rng.uniform(-0.45, 0.45),      # position across beam
+        "speed": rng.uniform(0.008, 0.03),  # drift along beam per frame
+        "phase": rng.uniform(0, math.tau),
+        "size": rng.uniform(0.6, 1.6),
+    })
+
+
+def lerp(a, b, t):
+    return a + (b - a) * t
+
+
+def beam_point(u, w):
+    """Point inside the beam: u along it (0 source -> 1 target), w across it."""
+    center = lerp(BEAM_SOURCE, (BEAM_TARGET_NEAR + BEAM_TARGET_FAR) / 2, u)
+    half_width = (BEAM_TARGET_FAR - BEAM_TARGET_NEAR) / 2
+    return center + half_width * w
+
+
+# ---------- static layers --------------------------------------------------
+
+def make_background():
+    """Dark wall + floor with a faint perspective gradient and fine grain."""
+    img = Image.new("RGB", (WIDTH, HEIGHT), (6, 6, 9))
+    px = img.load()
+    horizon = 74
+    for y in range(HEIGHT):
+        if y < horizon:
+            # wall: almost black, slightly lighter toward the floor
+            k = y / horizon
+            col = (int(5 + 5 * k), int(5 + 5 * k), int(8 + 6 * k))
+        else:
+            # floor: faint cool sheen, darkening toward the bottom
+            k = (y - horizon) / (HEIGHT - horizon)
+            v = int(15 - 7 * k)
+            col = (v, v - 1, v + 2)
+        for x in range(WIDTH):
+            px[x, y] = col
+    arr = np.asarray(img).astype(np.float32)
+    noise = rng.normal(0, 1.4, arr.shape).astype(np.float32)
+    arr = np.clip(arr + noise, 0, 255)
+    return Image.fromarray(arr.astype(np.uint8), "RGB")
+
+
+def cardboard_base(color, size):
+    """Cardboard texture: flat base color + fiber noise + faint vertical grain."""
+    w, h = size
+    base = np.zeros((h, w, 3), dtype=np.float32)
+    base[:, :] = np.array(color, dtype=np.float32)
+    fiber = rng.normal(0, 5.0, (h, w, 1))
+    grain = (rng.normal(0, 2.2, (h, 1, 1)) * np.ones((1, w, 1))).astype(np.float32)
+    base += fiber + grain
+    return np.clip(base, 0, 255).astype(np.uint8)
+
+
+def make_shadow_layer():
+    """Soft cast shadow on the floor, projected away from the light source."""
+    layer = Image.new("L", (WIDTH, HEIGHT), 0)
+    draw = ImageDraw.Draw(layer)
+
+    shifted = [tuple(np.array(p) + SHADOW_SHIFT) for p in CONTACT]
+    hull = [
+        CONTACT[0], CONTACT[1], shifted[1], shifted[2],
+        shifted[3], CONTACT[3],
+    ]
+    draw.polygon(hull, fill=150)
+    # Tight contact occlusion directly beneath the box.
+    draw.polygon(CONTACT, fill=210)
+    layer = layer.filter(ImageFilter.GaussianBlur(5))
+    return layer
+
+
+# ---------- per-frame rendering --------------------------------------------
+
+def draw_beam(flicker):
+    """Soft volumetric beam plus a brighter core, as an RGBA layer."""
+    layer = Image.new("RGBA", (WIDTH, HEIGHT), (0, 0, 0, 0))
+    draw = ImageDraw.Draw(layer)
+
+    s = BEAM_SOURCE
+    outer = [tuple(s), tuple(BEAM_TARGET_FAR + (10, 8)),
+             tuple(BEAM_TARGET_NEAR + (-6, -10))]
+    draw.polygon(outer, fill=(255, 226, 168, int(34 * flicker)))
+
+    inner = [tuple(s), tuple(beam_point(1.0, 0.22)),
+             tuple(beam_point(1.0, -0.22))]
+    draw.polygon(inner, fill=(255, 240, 205, int(30 * flicker)))
+
+    layer = layer.filter(ImageFilter.GaussianBlur(6))
+
+    # Dust motes drift in a separate layer built per frame.
+    mote = Image.new("RGBA", (WIDTH, HEIGHT), (0, 0, 0, 0))
+    return layer, mote
+
+
+def draw_box(base_img, flicker, frame_index):
+    """Composite the cardboard box with per-face lighting, tape and edges."""
+    faces = []
+    # face brightness: top catches the beam most, front next, right side least.
+    face_specs = [
+        (FRONT, (150, 104, 62), 0.62),
+        (TOP, (176, 128, 80), 0.95),
+        (RIGHT, (112, 74, 42), 0.38),
+    ]
+    for poly, color, lit in face_specs:
+        xs = [p[0] for p in poly]
+        ys = [p[1] for p in poly]
+        x0, y0, x1, y1 = min(xs), min(ys), max(xs), max(ys)
+        tex = cardboard_base(color, (x1 - x0 + 2, y1 - y0 + 2))
+        face = Image.new("RGBA", (x1 - x0 + 2, y1 - y0 + 2), (0, 0, 0, 0))
+        fdraw = ImageDraw.Draw(face)
+        local = [(x - x0, y - y0) for x, y in poly]
+        fdraw.polygon(local, fill=(255, 255, 255, 255))
+        face.paste(Image.fromarray(tex, "RGB"), (0, 0), face)
+
+        # Lighting overlay (warm where the beam hits).
+        shade = Image.new("RGBA", face.size, (0, 0, 0, 0))
+        sdraw = ImageDraw.Draw(shade)
+        level = lit * (0.92 + 0.08 * flicker)
+        sdraw.polygon(local, fill=(255, 224, 170, int(70 * level)))
+        sdraw.polygon(local, fill=(12, 10, 8, int(150 * (1 - level))))
+        face = Image.alpha_composite(face, shade)
+        faces.append((face, (x0, y0), poly))
+
+    for face, offset, poly in faces:
+        base_img.alpha_composite(face, offset)
+
+    draw = ImageDraw.Draw(base_img)
+    # Edges and tape details, drawn after compositing so lines stay crisp.
+    edge_col = (62, 40, 22, 255)
+    draw.line(FRONT + [FRONT[0]], fill=edge_col, width=1)
+    draw.line(TOP + [TOP[0]], fill=edge_col, width=1)
+    draw.line(RIGHT + [RIGHT[0]], fill=edge_col, width=1)
+    draw.line([FRONT[3], RIGHT[0]], fill=edge_col, width=1)
+
+    # Packing tape across the top face and down the front face.
+    tape = (214, 182, 128, 170)
+    draw.polygon([(110, 66), (118, 66), (138, 51), (130, 51)], fill=tape)
+    draw.polygon([(110, 66), (118, 66), (118, 118), (110, 118)], fill=tape)
+    # A faint flap seam on the front.
+    draw.line([(82, 78), (146, 78)], fill=(96, 64, 36, 160), width=1)
+
+    # Warm highlight where the beam strikes the top edge.
+    glow = Image.new("RGBA", (WIDTH, HEIGHT), (0, 0, 0, 0))
+    gdraw = ImageDraw.Draw(glow)
+    gdraw.line(TOP[0:2] + [TOP[3]], fill=(255, 232, 180, int(90 * flicker)), width=2)
+    glow = glow.filter(ImageFilter.GaussianBlur(2))
+    base_img.alpha_composite(glow)
+
+
+def render_frame(frame_index):
+    flicker = 1.0 + 0.07 * math.sin(frame_index * 0.9) \
+        + 0.035 * math.sin(frame_index * 2.3 + 1.2)
+
+    img = make_background().convert("RGBA")
+
+    # Cast shadow sits on the floor before the box is drawn.
+    shadow = make_shadow_layer()
+    shadow_col = Image.new("RGBA", (WIDTH, HEIGHT), (0, 0, 0, 255))
+    img = Image.composite(shadow_col, img, shadow.point(lambda v: int(v * 0.85)))
+
+    # Light pool on the floor around the lit side of the box.
+    pool = Image.new("RGBA", (WIDTH, HEIGHT), (0, 0, 0, 0))
+    pdraw = ImageDraw.Draw(pool)
+    pdraw.ellipse((58, 96, 178, 132), fill=(255, 214, 150, int(26 * flicker)))
+    pool = pool.filter(ImageFilter.GaussianBlur(10))
+    img.alpha_composite(pool)
+
+    draw_box(img, flicker, frame_index)
+
+    # Volumetric beam over the scene (in front of background, softly over box).
+    beam, mote_layer = draw_beam(flicker)
+    img = Image.alpha_composite(img, beam)
+
+    mdraw = ImageDraw.Draw(mote_layer)
+    for d in DUST:
+        u = (d["u"] + frame_index * d["speed"]) % 1.0
+        w = d["w"] + 0.05 * math.sin(frame_index * 0.5 + d["phase"])
+        x, y = beam_point(u, w)
+        twinkle = 0.5 + 0.5 * math.sin(frame_index * 1.1 + d["phase"])
+        alpha = int(120 * twinkle * (0.35 + 0.65 * u) * flicker)
+        r = d["size"]
+        mdraw.ellipse((x - r, y - r, x + r, y + r),
+                      fill=(255, 238, 200, max(alpha, 0)))
+    mote_layer = mote_layer.filter(ImageFilter.GaussianBlur(0.6))
+    img = Image.alpha_composite(img, mote_layer)
+
+    # Subtle vignette to keep the room dark at the edges.
+    vig = Image.new("L", (WIDTH, HEIGHT), 0)
+    vdraw = ImageDraw.Draw(vig)
+    vdraw.ellipse((-50, -34, WIDTH + 50, HEIGHT + 34), fill=255)
+    vig = vig.filter(ImageFilter.GaussianBlur(28))
+    dark = Image.new("RGBA", (WIDTH, HEIGHT), (0, 0, 0, 255))
+    img = Image.composite(img, dark, vig.point(lambda v: int(60 + 195 * v / 255)))
+
+    return np.asarray(img.convert("RGB"), dtype=np.uint8)
+
+
+def main():
+    frames = np.stack([render_frame(i) for i in range(FRAMES)], axis=0)
+    artifact_io.save(frames, CASE_DIR)
+    print(f"saved {frames.shape[0]} frames {frames.shape[1:]} to {CASE_DIR}")
+
+
+if __name__ == "__main__":
+    main()
