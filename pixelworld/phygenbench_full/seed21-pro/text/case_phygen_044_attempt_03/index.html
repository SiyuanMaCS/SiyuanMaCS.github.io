+"""Procedural text-to-video generator.
+
+Scene (from prompt.txt):
+  "A ray of light is shining diagonally on a plastic cup in the dark,
+   with the shadow of the plastic cup appearing at the bottom."
+
+Pure numpy/PIL renderer, no external assets. Produces 20 RGB frames of

+
+import numpy as np
+from PIL import Image, ImageFilter
+
+import artifact_io
+
+W, H, N, FPS = 256, 144, 20, 4.0
+SS = 3                      # supersampling factor per axis
+CW, CH = W * SS, H * SS     # internal working resolution
+rng = np.random.default_rng(20261111)
+

+# ---------------------------------------------------------------------------
+# Geometric helpers (everything is rendered in supersampled pixel space)
+# ---------------------------------------------------------------------------
+def lerp(a, b, t):
+    return a + (b - a) * t
+
+
+def make_cup_polygon():
+    """Cup silhouette: tapered tumbler with a flared rim and bottom ring."""
+    cx = 150.0 * SS
+    top_y = 46.0 * SS
+    rim_hw = 31.0 * SS      # rim half width
+    bot_hw = 20.0 * SS      # base half width
+    bot_y = 104.0 * SS
+    # Slightly curved sides via an intermediate point.
+    mid_y = (top_y + bot_y) * 0.5
+    mid_hw = lerp(rim_hw, bot_hw, 0.42) + 1.2 * SS
+    left = [
+        (cx - rim_hw, top_y),
+        (cx - mid_hw, mid_y),
+        (cx - bot_hw, bot_y),
+        (cx + bot_hw, bot_y),
+        (cx + mid_hw, mid_y),
+        (cx + rim_hw, top_y),
+    ]
+    return left, (cx, top_y, bot_y, rim_hw, bot_hw)
+
+
+CUP_POLY, CUP_INFO = make_cup_polygon()
+CX, TOP_Y, BOT_Y, RIM_HW, BOT_HW = CUP_INFO
+
+
+def polygon_mask(poly):
+    img = Image.new("L", (CW, CH), 0)
+    from PIL import ImageDraw
+    ImageDraw.Draw(img).polygon([(int(x), int(y)) for x, y in poly], fill=255)
+    return np.asarray(img, dtype=np.float32) / 255.0
+

+def ellipse_mask(cx, cy, rx, ry, soft=0.0):

+    ImageDraw.Draw(img).ellipse(

+                ImageFilter.GaussianBlur(soft)
+            ),
+            dtype=np.float32,
+        ) / 255.0
+    return arr

+# ---------------------------------------------------------------------------
+yy, xx = np.mgrid[0:CH, 0:CW].astype(np.float32)
+

+bg = np.zeros((CH, CW, 3), np.float32)

+bg[..., 0] = 6.0 + 5.0 * grad[..., 0]

+bg[..., 2] = 10.0 + 9.0 * grad[..., 0]

+# Ground plane: horizon band near the cup base, subtly lit.

+# ---------------------------------------------------------------------------
+# Volumetric diagonal light beam (a soft wedge from the upper-left corner)
+# ---------------------------------------------------------------------------
+def beam_field(phase):

+    rel = np.stack([xx - origin[0], yy - origin[1]], -1)
+    along = rel[..., 0] * beam_dir[0] + rel[..., 1] * beam_dir[1]
+    perp = rel[..., 0] * -beam_dir[1] + rel[..., 1] * beam_dir[0]
+    # Wedge widens with distance from the source.
+    width = 26.0 * SS + along * 0.22
+    core = np.exp(-((perp / (width * 0.42)) ** 2))
+    envelope = np.exp(-((perp / width) ** 2) * 1.4)

+# ---------------------------------------------------------------------------

+# ---------------------------------------------------------------------------

+    reach = 62.0 * SS

+    dy = beam_dir[1] * reach * 0.55     # flatten onto the ground
+    tip_cx = base_cx + dx

+        (base_cx + near_hw, base_y - 4 * SS),
+        (tip_cx + far_hw, tip_y),
+        (tip_cx - far_hw, tip_y),
+    ]
+    ImageDraw.Draw(shadow).polygon(

+    )
+    shadow = shadow.filter(ImageFilter.GaussianBlur(3.2 * SS))
+    return np.asarray(shadow, np.float32) / 255.0
+
+
+SHADOW = cup_shadow()
+
+
+# ---------------------------------------------------------------------------
+# Per-frame rendering
+# ---------------------------------------------------------------------------
+def render_frame(i):
+    t = i / (N - 1)
+    phase = i / N
+    frame = bg.copy()
+
+    beam, core, perp, width = beam_field(phase)
+
+    # Warm-white beam color; brighter core, cooler soft edges.
+    beam_rgb = np.zeros_like(frame)
+    beam_rgb[..., 0] = beam * 150 + core * 55
+    beam_rgb[..., 1] = beam * 138 + core * 50
+    beam_rgb[..., 2] = beam * 105 + core * 38
+    frame += beam_rgb
+
+    # Ground shadow darkens the lit ground beneath/right of the cup.
+    ground_light = np.clip((yy - (ground_y - 6 * SS)) / (40 * SS), 0, 1)
+    frame *= (1.0 - SHADOW[..., None] * 0.78 * ground_light[..., None])
+
+    # ----- Cup shading -----------------------------------------------------
+    # Local horizontal coordinate across the cup, 0 at center.
+    local_x = (xx - CX)
+    side_frac = np.clip(local_x / RIM_HW, -1.4, 1.4)
+
+    # Light comes from the upper-left: left-facing wall is lit.
+    lit_side = np.clip(0.5 - 0.55 * side_frac, 0, 1)  # 1 on far left
+    # Facing factor around the cylinder (cosine-ish falloff).
+    facing = np.clip(np.cos(side_frac * math.pi * 0.62), 0, 1)
+
+    base_col = np.array([188, 200, 206], np.float32)   # cool translucent plastic
+    lit_col = np.array([255, 244, 214], np.float32)    # warm key light
+    shade = 0.10 + 0.85 * lit_side * facing
+    cup_col = base_col[None, None, :] * (1 - shade[..., None]) + \
+        lit_col[None, None, :] * shade[..., None]
+
+    # Vertical falloff: rim catches the most light.
+    vf = np.clip(1.0 - (yy - TOP_Y) / (BOT_Y - TOP_Y), 0, 1)
+    cup_col *= (0.62 + 0.55 * vf[..., None])
+
+    # Specular streak on the left wall where the beam grazes the plastic.
+    spec_x = -0.52
+    spec = np.exp(-((side_frac - spec_x) / 0.13) ** 2)
+    spec *= np.clip(1.0 - np.abs(yy - (TOP_Y + 18 * SS)) / (52 * SS), 0, 1)
+    cup_col += spec[..., None] * np.array([120, 115, 95], np.float32)
+
+    # Translucency: let the beam glow through the cup body.
+    inner_glow = beam * cup_mask * 0.35
+    cup_col += inner_glow[..., None] * np.array([255, 235, 190], np.float32)
+
+    frame = frame * (1 - cup_mask[..., None]) + cup_col * cup_mask[..., None]
+
+    # ----- Rim and base highlights ----------------------------------------
+    from PIL import ImageDraw
+    rim = Image.new("L", (CW, CH), 0)
+    d = ImageDraw.Draw(rim)
+    # Top rim ellipse (foreshortened opening).
+    d.ellipse(
+        [CX - RIM_HW, TOP_Y - 4.5 * SS, CX + RIM_HW, TOP_Y + 6.5 * SS],
+        outline=255, width=max(1, int(1.6 * SS)),
+    )
+    rim = np.asarray(rim, np.float32) / 255.0
+    # Left part of rim is brightest (key light side).
+    rim_gain = np.clip(0.9 - 0.9 * side_frac, 0.15, 1.0)
+    frame += (rim * rim_gain)[..., None] * np.array([70, 66, 52], np.float32)
+
+    # Thin base ring.
+    base_ring = Image.new("L", (CW, CH), 0)
+    ImageDraw.Draw(base_ring).ellipse(
+        [CX - BOT_HW, BOT_Y - 3 * SS, CX + BOT_HW, BOT_Y + 4 * SS],
+        outline=180, width=max(1, int(1.2 * SS)),
+    )
+    base_ring = np.asarray(base_ring, np.float32) / 255.0
+    frame += base_ring[..., None] * np.array([26, 28, 30], np.float32)
+
+    # Rim light streak inside the cup opening (a bright arc on lit side).
+    arc = Image.new("L", (CW, CH), 0)
+    ImageDraw.Draw(arc).arc(
+        [CX - RIM_HW, TOP_Y - 4.5 * SS, CX + RIM_HW, TOP_Y + 6.5 * SS],
+        start=200, end=330, fill=255, width=max(1, int(1.8 * SS)),
+    )
+    arc = np.asarray(arc, np.float32) / 255.0
+    frame += arc[..., None] * np.array([95, 90, 70], np.float32)
+
+    # ----- Floating dust motes inside the beam for volume ------------------
+    mote_layer = np.zeros((CH, CW), np.float32)
+    for k in range(26):
+        mx = (motes_x[k] + beam_dir[0] * motes_drift[k] * t +
+              math.sin(phase * 6.28 + k) * 1.5 * SS)
+        my = (motes_y[k] + beam_dir[1] * motes_drift[k] * t)
+        r = motes_r[k]
+        m = ellipse_mask(mx, my, r, r, soft=r * 0.6)
+        tw = 0.55 + 0.45 * math.sin(phase * 6.28 * 2 + k * 1.7)
+        mote_layer += m * tw
+    mote_layer *= beam * 1.4
+    frame += mote_layer[..., None] * np.array([200, 190, 160], np.float32)
+
+    # Subtle exposure breathing so the ray feels alive.
+    exposure = 1.0 + 0.03 * math.sin(phase * 2 * math.pi)
+    frame *= exposure
+
+    # Vignette.
+    vx = (xx / CW - 0.5) * 2
+    vy = (yy / CH - 0.5) * 2
+    vig = np.clip(1 - 0.55 * (vx ** 2 + vy ** 2), 0.25, 1)
+    frame *= vig[..., None]
+
+    # Fine sensor grain.
+    grain = rng.normal(0, 2.2, (CH, CW, 1)).astype(np.float32)
+    frame += grain
+
+    frame = np.clip(frame, 0, 255)
+    img = Image.fromarray(frame.astype(np.uint8)).resize(
+        (W, H), Image.Resampling.LANCZOS
+    )
+    return np.asarray(img, np.uint8)
+
+
+# Deterministic dust motes distributed along the beam corridor.
+motes_x, motes_y, motes_r, motes_drift = [], [], [], []
+for _k in range(26):
+    along = rng.uniform(30, 165) * SS
+    perp = rng.normal(0, 14) * SS
+    ox = -30 * SS
+    oy = -18 * SS
+    mx = ox + beam_dir[0] * along + -beam_dir[1] * perp
+    my = oy + beam_dir[1] * along + beam_dir[0] * perp
+    motes_x.append(mx)
+    motes_y.append(my)
+    motes_r.append(rng.uniform(0.7, 1.8) * SS)
+    motes_drift.append(rng.uniform(2, 9) * SS)
+
+
+def main():
+    frames = np.stack([render_frame(i) for i in range(N)], axis=0)
+    artifact_io.save(frames, ".")
+    print("saved", frames.shape, frames.dtype)
+
+
+if __name__ == "__main__":
+    main()
