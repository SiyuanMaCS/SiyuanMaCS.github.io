+"""Procedural RGB video: a flashlight beam through a bowl of murky water.
+
+Rendered entirely with numpy/PIL math: volumetric gaussian beam, murky-water
+scattering with drifting particles, animated caustics, glass bowl and bloom.
+"""
+from pathlib import Path
+
+import numpy as np
+from PIL import Image, ImageDraw, ImageFilter
+
+import artifact_io
+
+W, H, N_FRAMES, FPS = 256, 144, 20, 4.0
+CASE_DIR = Path(__file__).parent
+
+rng = np.random.default_rng(2024047)
+
+
+# --------------------------------------------------------------------------
+# Smooth value noise (periodic lattice, bilinear + smoothstep), fbm helper
+# --------------------------------------------------------------------------
+class ValueNoise:
+    def __init__(self, res, seed):
+        self.res = res
+        self.grid = np.random.default_rng(seed).random((res, res)).astype(np.float32)
+
+    def sample(self, x, y):
+        res = self.res
+        xi = np.floor(x).astype(np.int32)
+        yi = np.floor(y).astype(np.int32)
+        xf = (x - xi).astype(np.float32)
+        yf = (y - yi).astype(np.float32)
+        u = xf * xf * (3.0 - 2.0 * xf)
+        v = yf * yf * (3.0 - 2.0 * yf)
+        x0, x1 = xi % res, (xi + 1) % res
+        y0, y1 = yi % res, (yi + 1) % res
+        g = self.grid
+        a, b = g[y0, x0], g[y0, x1]
+        c, d = g[y1, x0], g[y1, x1]
+        return (a * (1 - u) + b * u) * (1 - v) + (c * (1 - u) + d * u) * v
+
+
+def fbm(noise, x, y, octaves=3):
+    value, amp, freq, total = 0.0, 0.5, 1.0, 0.0
+    for _ in range(octaves):
+        value += amp * noise.sample(x * freq, y * freq)
+        total += amp
+        amp *= 0.5
+        freq *= 2.03
+    return value / total
+
+
+def smoothstep(x, edge0, edge1):
+    z = np.clip((x - edge0) / (edge1 - edge0), 0.0, 1.0)
+    return z * z * (3.0 - 2.0 * z)
+
+
+noise_a = ValueNoise(64, 11)
+noise_b = ValueNoise(64, 23)
+noise_c = ValueNoise(48, 37)
+noise_d = ValueNoise(48, 51)
+
+ys_grid, xs_grid = np.mgrid[0:H, 0:W].astype(np.float32)
+X, Y = xs_grid, ys_grid
+
+# --------------------------------------------------------------------------
+# Static scene geometry
+# --------------------------------------------------------------------------
+ORIGIN = np.array([40.0, 64.0])       # flashlight lens
+TARGET = np.array([242.0, 86.0])      # beam aim
+dvec = TARGET - ORIGIN
+AXIS = dvec / np.linalg.norm(dvec)
+DX, DY = AXIS
+
+BOWL_C = np.array([164.0, 90.0])
+RX, RY = 58.0, 40.0
+RIM_Y, RIM_RY = 72.0, 9.0
+SURFACE_Y = 78.0
+
+t_coord = (X - ORIGIN[0]) * DX + (Y - ORIGIN[1]) * DY
+perp = -(X - ORIGIN[0]) * DY + (Y - ORIGIN[1]) * DX
+sigma = 2.0 + np.maximum(t_coord, 0.0) * 0.038
+beam_core = np.exp(-0.5 * (perp / sigma) ** 2)
+beam_env = smoothstep(t_coord, 2.0, 14.0) * (1.0 - smoothstep(t_coord, 218.0, 246.0))
+
+ellipse = ((X - BOWL_C[0]) / RX) ** 2 + ((Y - BOWL_C[1]) / RY) ** 2
+rim_ellipse = ((X - BOWL_C[0]) / RX) ** 2 + ((Y - RIM_Y) / RIM_RY) ** 2
+
+# Beam axis intersection with the bowl ellipse (entry and exit glows)
+crossings = []
+prev_inside = False
+for tv in np.linspace(0.0, 240.0, 600):
+    p = ORIGIN + AXIS * tv
+    inside = (((p[0] - BOWL_C[0]) / RX) ** 2 + ((p[1] - BOWL_C[1]) / RY) ** 2) <= 1.0
+    if inside != prev_inside:
+        crossings.append(p.copy())
+    prev_inside = inside
+
+# Suspended murk particles inside the water
+N_PARTICLES = 170
+p0 = np.zeros((N_PARTICLES, 2), np.float32)
+filled = 0
+while filled < N_PARTICLES:
+    theta = rng.uniform(0, 2 * np.pi, 400)
+    rad = np.sqrt(rng.uniform(0, 1, 400))
+    px = BOWL_C[0] + RX * rad * np.cos(theta)
+    py = BOWL_C[1] + RY * rad * np.sin(theta)
+    ok = py > SURFACE_Y + 1.0
+    take = min(ok.sum(), N_PARTICLES - filled)
+    p0[filled:filled + take, 0] = px[ok][:take]
+    p0[filled:filled + take, 1] = py[ok][:take]
+    filled += take
+p_radius = rng.uniform(0.5, 1.7, N_PARTICLES).astype(np.float32)
+p_drift = np.stack([rng.uniform(0.4, 1.4, N_PARTICLES),
+                    rng.uniform(-0.25, 0.25, N_PARTICLES)], axis=1).astype(np.float32)
+p_osc = rng.uniform(0.8, 2.6, (N_PARTICLES, 2)).astype(np.float32)
+p_omega = rng.uniform(0.5, 1.6, N_PARTICLES).astype(np.float32)
+p_phase = rng.uniform(0, 2 * np.pi, N_PARTICLES).astype(np.float32)
+p_tint = rng.uniform(0, 1, N_PARTICLES).astype(np.float32)   # warm <-> teal
+WARM = np.array([255.0, 235.0, 185.0], np.float32)
+TEAL = np.array([130.0, 225.0, 215.0], np.float32)
+GOLD = np.array([255.0, 200.0, 110.0], np.float32)
+
+# A few dust motes in the airborne beam
+N_DUST = 26
+dust_p0 = np.stack([rng.uniform(44, 108, N_DUST), rng.uniform(58, 78, N_DUST)], axis=1).astype(np.float32)
+dust_phase = rng.uniform(0, 2 * np.pi, N_DUST).astype(np.float32)
+
+
+def render_frame(frame_index):
+    t = frame_index / FPS
+    flicker = (1.0 + 0.05 * np.sin(t * 9.1)
+               + 0.035 * np.sin(t * 17.7 + 1.3)
+               + 0.02 * np.sin(t * 31.0))
+
+    # --- background: dark room, faint table, ambient spill ----------------
+    frame = np.zeros((H, W, 3), np.float32)
+    grad = np.clip(Y / H, 0, 1)[..., None]
+    frame[:] = (10.0, 14.0, 20.0) * (1 - grad) + (4.0, 6.0, 10.0) * grad
+    table = Y > 112.0
+    frame[table] = (11.0, 13.0, 17.0)
+    spill = np.exp(-(((X - BOWL_C[0]) / 95.0) ** 2 + ((Y - 118.0) / 34.0) ** 2))
+    frame += spill[..., None] * np.array([10.0, 26.0, 28.0]) * 0.5
+    vignette = np.clip(1.0 - (((X - 128) / 165.0) ** 2 + ((Y - 72) / 110.0) ** 2), 0, 1)
+    frame *= (0.55 + 0.45 * vignette)[..., None]
+
+    # --- wavy water surface and water mask --------------------------------
+    surface = (SURFACE_Y
+               + 1.4 * np.sin(0.07 * X + t * 1.3)
+               + 0.6 * np.sin(0.13 * X - t * 0.9))
+    water = (ellipse <= 1.0) & (Y >= surface)
+
+    # --- murky water body with slowly drifting haze ------------------------
+    depth = np.clip((Y - SURFACE_Y) / 52.0, 0, 1)[..., None]
+    water_col = (np.array([24.0, 46.0, 47.0]) * (1 - depth)
+                 + np.array([6.0, 15.0, 17.0]) * depth)
+    haze = fbm(noise_a, X * 0.045 + t * 0.05, Y * 0.09 - t * 0.03) - 0.5
+    water_col = water_col + haze[..., None] * 14.0
+    frame[water] = water_col[water]
+
+    # --- airborne beam (faint, dusty) --------------------------------------
+    air_intensity = (beam_core * beam_env * flicker
+                     * (0.13 + 0.55 * beam_core ** 3))
+    air_mask = ~water
+    frame += (air_intensity * air_mask)[..., None] * WARM
+
+    # --- beam inside the water: scattering, streaks, hue drift -------------
+    streak = (0.75
+              + 0.6 * fbm(noise_b, perp * 0.85 + 5.0, t_coord * 0.02 + t * 0.45)
+              + 0.2 * fbm(noise_c, t_coord * 0.05, t * 0.3 + 20.0))
+    s_hue = np.clip((t_coord - 95.0) / 150.0, 0, 1)[..., None]
+    beam_col = WARM[None, None, :] * (1 - s_hue) + TEAL[None, None, :] * s_hue
+    gold_patch = fbm(noise_d, X * 0.05 + t * 0.12, Y * 0.10 - t * 0.08)
+    beam_col = beam_col * (0.82 + 0.45 * gold_patch[..., None])
+    beam_col = (beam_col * (1 - 0.35 * gold_patch[..., None])
+                + GOLD[None, None, :] * (0.35 * gold_patch[..., None]))
+    water_intensity = beam_core * beam_env * streak * flicker * 0.6
+    frame += (water_intensity * water)[..., None] * beam_col
+
+    # --- caustic shimmer on the bowl bottom and water surface --------------
+    n1 = fbm(noise_c, X * 0.06 + t * 0.25, Y * 0.06)
+    n2 = fbm(noise_d, X * 0.05 - t * 0.2, Y * 0.07)
+    waves = (np.sin(0.35 * X + 4.0 * n1 + t * 2.0)
+             + np.sin(0.3 * Y - 3.0 * n2 - t * 1.7))
+    caustic = np.clip(waves - 1.15, 0, None) * 0.5
+    bottom = (ellipse < 0.98) & (ellipse > 0.4) & (Y > 84.0) & water
+    broad = np.exp(-0.5 * (perp / (sigma * 1.7)) ** 2) * beam_env
+    frame += (caustic * bottom * broad)[..., None] * np.array([150.0, 230.0, 225.0]) * 0.55
+    surf_glow = np.exp(-0.5 * ((Y - surface) / 1.4) ** 2) * broad * 0.55
+    frame += (surf_glow * (ellipse <= 1.0))[..., None] * np.array([185.0, 235.0, 230.0])
+
+    # --- suspended particles, lit where the beam passes --------------------
+    pos = p0 + p_drift * t + p_osc * np.sin(
+        p_omega[:, None] * t + p_phase[:, None])
+    pos[:, 0] = np.clip(pos[:, 0], BOWL_C[0] - RX + 2.0, BOWL_C[0] + RX - 2.0)
+    pos[:, 1] = np.clip(pos[:, 1], SURFACE_Y + 1.0, BOWL_C[1] + RY - 1.0)
+    for i in range(N_PARTICLES):
+        px, py = pos[i]
+        ti = (px - ORIGIN[0]) * DX + (py - ORIGIN[1]) * DY
+        pi = -(px - ORIGIN[0]) * DY + (py - ORIGIN[1]) * DX
+        si = 2.0 + max(ti, 0.0) * 0.038
+        lit = (np.exp(-0.5 * (pi / si) ** 2)
+               * smoothstep(np.array(ti), 2, 14)
+               * (1 - smoothstep(np.array(ti), 218, 246)) * flicker)
+        r = p_radius[i]
+        patch = int(np.ceil(2.5 * r)) + 1
+        x0, x1 = int(px) - patch, int(px) + patch + 1
+        y0, y1 = int(py) - patch, int(py) + patch + 1
+        if x1 <= 0 or y1 <= 0 or x0 >= W or y0 >= H:
+            continue
+        xa, xb = max(x0, 0), min(x1, W)
+        ya, yb = max(y0, 0), min(y1, H)
+        yy, xx = np.mgrid[ya:yb, xa:xb]
+        weight = np.exp(-((xx - px) ** 2 + (yy - py) ** 2) / (2 * r * r))
+        col = WARM * (1 - p_tint[i]) + TEAL * p_tint[i]
+        local_water = water[ya:yb, xa:xb]
+        illum = (0.10 + 1.6 * lit) * weight * local_water
+        frame[ya:yb, xa:xb] += illum[..., None] * col
+
+    # --- airborne dust motes -----------------------------------------------
+    for i in range(N_DUST):
+        px = dust_p0[i, 0] + 2.0 * np.sin(t * 0.6 + dust_phase[i])
+        py = dust_p0[i, 1] + 1.5 * np.cos(t * 0.45 + dust_phase[i])
+        ti = (px - ORIGIN[0]) * DX + (py - ORIGIN[1]) * DY
+        pi = -(px - ORIGIN[0]) * DY + (py - ORIGIN[1]) * DX
+        si = 2.0 + max(ti, 0.0) * 0.038
+        lit = np.exp(-0.5 * (pi / si) ** 2) * flicker
+        frame[int(py), int(px)] += WARM * lit * 0.9
+
+    # --- entry / exit glows where the beam meets the bowl ------------------
+    for idx, p in enumerate(crossings[:2]):
+        strength = 1.0 if idx == 0 else 0.4
+        glow = np.exp(-0.5 * (((X - p[0]) / 5.5) ** 2 + ((Y - p[1]) / 5.5) ** 2))
+        frame += (glow * strength)[..., None] * np.array([235.0, 245.0, 235.0])
+
+    # --- glass bowl: silhouette ring and top rim ---------------------------
+    ring = np.abs(ellipse - 1.0) < 0.03
+    ring_alpha = 0.16 + 0.3 * np.clip(1.0 - (X - BOWL_C[0]) / RX, 0, 1) ** 2
+    glass_col = np.array([165.0, 205.0, 215.0])
+    frame[ring] = frame[ring] * (1 - ring_alpha[ring, None]) + glass_col * ring_alpha[ring, None]
+    rim_band = np.abs(rim_ellipse - 1.0) < 0.07
+    rim_alpha = np.where(Y > RIM_Y, 0.5, 0.16)
+    frame[rim_band] = (frame[rim_band] * (1 - rim_alpha[rim_band, None])
+                       + glass_col * rim_alpha[rim_band, None])
+    # crisp specular streak on the front-left glass
+    streak_glass = ring & (X < BOWL_C[0] - RX * 0.35)
+    frame[streak_glass] += np.array([70.0, 95.0, 100.0])
+
+    # --- fine sensor grain --------------------------------------------------
+    frame += rng.normal(0, 1.4, frame.shape).astype(np.float32)
+    frame = np.clip(frame, 0, 255).astype(np.uint8)
+
+    # --- flashlight body (PIL) ---------------------------------------------
+    img = Image.fromarray(frame)
+    draw = ImageDraw.Draw(img)
+    draw.rounded_rectangle([0, 56, 24, 72], radius=4, fill=(34, 38, 44), outline=(78, 84, 94))
+    draw.polygon([(22, 54), (40, 58), (40, 70), (22, 74)], fill=(52, 57, 66), outline=(96, 102, 112))
+    draw.ellipse([35, 58, 44, 70], fill=(225, 238, 250), outline=(150, 170, 185))
+    draw.ellipse([38, 61, 42, 67], fill=(255, 250, 235))
+    frame = np.asarray(img).astype(np.float32)
+
+    # --- bloom: soft halo + hot core ---------------------------------------
+    base = Image.fromarray(np.clip(frame, 0, 255).astype(np.uint8))
+    hot = np.clip(frame - 140.0, 0, None).astype(np.uint8)
+    bloom_soft = np.asarray(base.filter(ImageFilter.GaussianBlur(5))).astype(np.float32)
+    bloom_hot = np.asarray(Image.fromarray(hot).filter(ImageFilter.GaussianBlur(6))).astype(np.float32)
+    frame = frame + 0.12 * bloom_soft + 0.9 * bloom_hot
+    return np.clip(frame, 0, 255).astype(np.uint8)
+
+
+def main():
+    frames = np.stack([render_frame(i) for i in range(N_FRAMES)], axis=0)
+    artifact_io.save(frames, CASE_DIR)
+    print(f"saved {frames.shape} at {FPS} fps -> {CASE_DIR}")
+
+
+if __name__ == "__main__":
+    main()
