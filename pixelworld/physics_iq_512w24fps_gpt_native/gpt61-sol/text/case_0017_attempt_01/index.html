def ball_state(t):

    dt=t-RELEASE

    q=t-CONTACT
    if q<STOP_TIME:
        u=q/STOP_TIME
        cy=CONTACT_Y+SINK*(2*u-u*u)
        compression=.022*np.sin(np.pi*u)
        return 270.1+.5*u,cy,1+compression*.45,1-compression,np.deg2rad(1.+3.*u)
    q-=STOP_TIME

    relax=1.65*(1-np.exp(-q/.035))*np.exp(-q/.12)
    cy=CONTACT_Y+SINK-relax
