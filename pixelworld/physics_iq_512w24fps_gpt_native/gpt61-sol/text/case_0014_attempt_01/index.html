    behind=np.clip(behind,0,255)
    background[hole_region]=behind[hole_region]
    return background,alpha,premul,front_visibility,holes


def trajectory(t):
    gravity=650.
    bottom=186.8
    restitution=.72
    fall_time=np.sqrt(2*(bottom-CY)/gravity)
    if t<=fall_time:
        return CX+.75*t, CY+.5*gravity*t*t, .10*t
    elapsed=t-fall_time
    velocity=np.sqrt(2*gravity*(bottom-CY))*restitution
    count=0
    while velocity>7.5:
        duration=2*velocity/gravity
        if elapsed<duration:
            height=velocity*elapsed-.5*gravity*elapsed*elapsed
            drift=2.3*(1-np.exp(-(t-fall_time)*1.4))
            angle=.10*fall_time+.24*(1-np.exp(-(t-fall_time)*1.15))
            return CX+.75*fall_time+drift,bottom-height,angle
        elapsed-=duration
        velocity*=restitution
        count+=1
    return CX+.75*fall_time+2.3,bottom,.10*fall_time+.24


def render(t, background,alpha,premul,front_visibility,holes):
    cx,cy,theta=trajectory(t)
    dx=X-cx;dy=Y-cy
    c=np.cos(theta);s=np.sin(theta)
    # Rotate normalized sphere coordinates, retaining the original perspective.
    ux=dx/RX;uy=dy/RY
    sx=CX+RX*(c*ux+s*uy)
    sy=CY+RY*(-s*ux+c*uy)
    a=map_coordinates(alpha,[sy,sx],order=1,mode='constant',cval=0)
    p=np.stack([map_coordinates(premul[:,:,k],[sy,sx],order=1,mode='constant',cval=0) for k in range(3)],-1)
    top=front_visibility
    frame=background*(1-(a*top)[:,:,None])+p*top[:,:,None]
    # Top visibility is zero over the front face. The sphere is visible there
    # only through the original open ventilation pattern.
    inside_visibility=holes*(1-top)
    frame=frame*(1-(a*inside_visibility)[:,:,None])+p*inside_visibility[:,:,None]*.68
    return np.clip(frame,0,255)


def main():
    source=np.asarray(Image.open(CASE/'first_frame.png').convert('RGB'),dtype=np.float64)
    data=setup(source)
    reference=render(0,*data)
    # A small residual holds exact photographed shading at release. It follows
    # the same translating sphere and vanishes once that surface enters the crate.
    residual=source-reference
    q=((X-CX)/RX)**2+((Y-CY)/RY)**2
    moving_residual=residual*((q<1.35)&(Y<115))[:,:,None]
    fixed_residual=residual-moving_residual
    frames=[]
    for i in range(120):
        if i==0:
            frames.append(source.astype(np.uint8));continue
        t=i/24
        # Short exposure integration adds subtle, realistic motion blur on fast falls.
        exposures=[max(0,t-1/144),t,t+1/144]
        frame=sum(render(st,*data) for st in exposures)/len(exposures)
        cx,cy,theta=trajectory(t)
        # Source-matched release correction is warped with the moving ball, not
        # left as a stationary ghost on the wall.
        if t<.36:
            corr=np.stack([map_coordinates(moving_residual[:,:,k],[Y-(cy-CY),X-(cx-CX)],order=1,mode='constant',cval=0) for k in range(3)],-1)
            frame+=corr*np.exp(-t*15)*data[3][:,:,None]
        frame+=fixed_residual*np.exp(-t*12)
        frames.append(np.rint(np.clip(frame,0,255)).astype(np.uint8))
    frames=np.stack(frames)
    artifact_io.save(frames,CASE)
    print('Saved',frames.shape,frames.dtype)
    print('Frame zero identical:',np.array_equal(frames[0],source.astype(np.uint8)))
    print('Release reconstruction MAE:',np.mean(np.abs(reference-source)))
    for i in [0,6,12,14,18,24,30,36,42,48,60,96,119]:
        print(i,tuple(round(v,3) for v in trajectory(i/24)))

if __name__=='__main__':
    main()
