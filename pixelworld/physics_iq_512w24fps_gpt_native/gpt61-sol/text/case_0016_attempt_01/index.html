def state(t):
    if t <= RELEASE:

        v0=G*FALL_TIME*SINK_TIME
        delta=FINAL_CY-CONTACT_CY

        dep=(-2*u**3+3*u*u)*delta+(u**3-2*u*u+u)*v0

        return CX+.55+.14*ease(u), CONTACT_CY+dep, .38*FALL_TIME+.023*ease(u),vel

    settle=-.72*math.exp(-q*15)*math.sin(q*31)
