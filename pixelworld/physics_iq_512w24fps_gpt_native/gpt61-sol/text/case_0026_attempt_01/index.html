    # Shadow silhouette, including the soft contact patch below the sphere.
    ex=(xx-231)/43; ey=(yy-147)/16
    shadow_hole=1-smoothstep(.70,1.13,np.sqrt(ex*ex+ey*ey))
    lower_hole=.72*(1-smoothstep(.65,1.13,np.sqrt(((xx-213)/31)**2+((yy-161)/13)**2)))
    hole=np.maximum.reduce([ball_hole,shadow_hole,lower_hole])
    background=photo*(1-hole[...,None])+fill*hole[...,None]

    # The visible original shadow is kept as a multiplicative photographic
    # layer. Within the occluded disk, a soft contact shadow completes it.
    contact=.52*np.exp(-.5*(((xx-cx-2)/14)**2+((yy-cy-20)/4.3)**2))
    cast=.30*np.exp(-.5*(((xx-cx-19)/24)**2+((yy-cy-6)/9.2)**2))
    depth=(1-contact)*(1-cast)
    ratio=np.clip((photo.mean(2)+1)/(background.mean(2)+1),.35,1)
    visibility=1-smoothstep(.93,1.16,radial)
    shadow_ratio=depth*visibility+ratio*(1-visibility)
    shadow_ratio=1+(shadow_ratio-1)*hole
    shadow_ratio=gaussian_filter(shadow_ratio,.7)
    source_ground=background*shadow_ratio[...,None]

    # Extract intrinsic ball color with edge-background decontamination.
    # A stationary light field is removed so that illumination does not turn
    # with the seams/felt, and added back after each spherical rotation.
    a=native_alpha[...,None]
    intrinsic=(photo-source_ground*(1-a))/np.maximum(a,.06)
    intrinsic=np.clip(intrinsic,0,255)
    z=np.sqrt(np.maximum(0,1-X*X-Y*Y))
    coefficients=np.array([[19.0888,3.8316,.5809,-3.9394],
                           [35.7427,-3.9873,-8.5180,33.8357],
                           [68.7124,-13.9468,-15.6288,76.0225]])
    def light(nx,ny,nz):
        v=np.stack([np.ones_like(nx),nx,ny,nz],-1)
        return v@coefficients.T
    residual=intrinsic-light(X,Y,z)
    # Don't put background contamination onto the sphere's unseen hemisphere.
    inside=(radial<.88)
    nearest=distance_transform_edt(~inside,return_distances=False,return_indices=True)
    safe=residual[nearest[0],nearest[1]]
    mix=1-smoothstep(.87,.98,radial)
    texture=residual*mix[...,None]+safe*(1-mix[...,None])
    texture=gaussian_filter(texture,(.20,.20,0))

    # Foreground tiles are evaluated at double resolution for subpixel edges.
    # A small exposure integral also gives natural, modest motion blur.
    tile=59
    gy,gx=np.mgrid[-tile:tile+1,-tile:tile+1].astype(float)/2
    axis=np.array([-.085,1,.23]); axis/=np.linalg.norm(axis)
    n=np.stack([gx/rx,gy/ry,np.sqrt(np.maximum(0,1-(gx/rx)**2-(gy/ry)**2))],-1)
    normal_len=np.sqrt((gx/rx)**2+(gy/ry)**2)
    # Surface samples at the true silhouette, retaining a photographic blur.
    nr=n.copy()
    outer=normal_len>1
    nr[outer,0]/=normal_len[outer]
    nr[outer,1]/=normal_len[outer]
    alpha=.5*(1-erf((normal_len-1)*rx/(1.25*np.sqrt(2))))
    base_light=light(nr[...,0],nr[...,1],nr[...,2])
    angle_scale=1/rx
    cross=np.cross(axis,nr)
    dot=np.sum(nr*axis,axis=-1)[...,None]

    def state(t):
        displacement=71*t-.45*t*t
        x=cx+displacement
        # The path stays on the plane, to the near side of its visible edge.
        y=cy+.085*displacement
        return x,y,displacement

    def ball_tile(displacement):
        angle=-displacement*angle_scale
        rot=nr*np.cos(angle)+cross*np.sin(angle)+dot*axis*(1-np.cos(angle))
        sx=cx+rx*rot[...,0]
        sy=cy+ry*rot[...,1]
        # Folding the photographed hemisphere completes its unseen surface
        # without inventing new objects, colors, or illumination.
        mat=sample(texture,sx,sy)
        rgb=np.clip(base_light+mat,0,255)
        rgba=np.concatenate([rgb*alpha[...,None],alpha[...,None]],-1)
        # Reduce a 2x patch through a Lanczos filter.
        out=[]
        for k in range(4):
            channel=Image.fromarray(rgba[...,k].astype('float32'),mode='F')
            out.append(np.asarray(channel.resize((60,60),Image.Resampling.LANCZOS)))
        return np.stack(out,-1)

    frames=[]
    for i in range(spec['frames']):
        if i==0:
            frames.append(ref.copy()); continue
        t=i/spec['fps']
        bx,by,dist=state(t)
        if bx-rx>W+5:
            frames.append(np.clip(background,0,255).round().astype(np.uint8));continue
        glass=smoothstep(283,314,bx)
        # Photographic shadow follows the contact point; the clear pane makes
        # it softer and weaker, not an opaque table-colored platform.
        sr=sample(shadow_ratio,xx-dist,yy-.085*dist)
        sr=1+(sr-1)*(1-.57*glass)
        frame=background*sr[...,None]

        # Subtle flattened blue reflection in the glass just below contact.
        if glass>0:
            refl=.035*glass*np.exp(-.5*(((xx-bx-1)/15.5)**2+((yy-by-25)/4.2)**2))
            tint=np.array([38.,77.,130.])
            frame=frame*(1-refl[...,None])+tint*refl[...,None]
        # 1/100 second shutter: approximately .7 pixels of travel per exposure.
        accum=np.zeros_like(frame)
        for dt in [-.004,0,.004]:
            x,y,d=state(t+dt)
            rgba=ball_tile(d)
            # Exact tile-center coordinates, bilinear translational warp.
            ix=int(np.floor(x-29.5));iy=int(np.floor(y-29.5))
            x0=max(0,ix-1);x1=min(W,ix+62)
            y0=max(0,iy-1);y1=min(H,iy+62)
            layer=frame.copy()
            if x1>x0 and y1>y0:
                sy,sx=np.mgrid[y0:y1,x0:x1].astype(float)
                coordsx=sx-(x-29.5);coordsy=sy-(y-29.5)
                # Zero beyond the tile, to avoid nearest-mode edge replication.
                pix=np.stack([map_coordinates(rgba[...,k],[coordsy,coordsx],order=1,mode='constant',cval=0)
                              for k in range(4)],-1)
                aa=np.clip(pix[...,3:4],0,1)
                layer[y0:y1,x0:x1]=frame[y0:y1,x0:x1]*(1-aa)+pix[...,:3]
            accum+=layer/3
        frames.append(np.clip(accum,0,255).round().astype(np.uint8))
    return np.stack(frames)
