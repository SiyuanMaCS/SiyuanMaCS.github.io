def bilinear(a,x,y):
    hh,ww=a.shape[:2]
    ok=(x>=0)&(x<ww-1)&(y>=0)&(y<hh-1)
    x=np.clip(x,0,ww-1.00001);y=np.clip(y,0,hh-1.00001)
    ix=np.floor(x).astype(int);iy=np.floor(y).astype(int)
    fx=x-ix;fy=y-iy
    if a.ndim==3:fx=fx[...,None];fy=fy[...,None]
    b=(a[iy,ix]*(1-fx)*(1-fy)+a[iy,ix+1]*fx*(1-fy)+a[iy+1,ix]*(1-fx)*fy+a[iy+1,ix+1]*fx*fy)
    return b*ok[...,None] if a.ndim==3 else b*ok

entry_x=339.0
contact0=164.1

a_ramp=200.

def state(t):
    if t<=te:
        d=v0*t-.5*acc_table*t*t
        x=cx0+d; foot=contact0+m_table*d
        scale=1+(entry_scale-1)*d/(entry_x-cx0)
        roll=d/(rx0*.95)
        axis=np.array([-.0-m_table,1.,0.]);axis/=np.linalg.norm(axis)
        regime=0.
    elif t<=exit_t:
        u=t-te;d=v_ramp*u-.5*a_ramp*u*u
        x=entry_x+d;foot=entry_y+ramp_slope*d
        scale=entry_scale-.00155*d
        roll=(entry_x-cx0)/(rx0*.95)+d/(rx0*.845)*1.40
        axis=np.array([-ramp_slope,1.,0.]);axis/=np.linalg.norm(axis)
        regime=np.clip(u/.10,0,1)*np.clip((exit_t-t)/.10,0,1)
    else:
        u=t-exit_t
        d=-ve*u+.5*12.*u*u
        x=entry_x+d;foot=entry_y+m_table*d
        scale=entry_scale+(1-entry_scale)*(-d)/(entry_x-cx0)
        scale=min(scale,1.24)
        roll=(entry_x-cx0)/(rx0*.95)+d/(rx0*1.04)
        axis=np.array([-m_table,1.,0.]);axis/=np.linalg.norm(axis)
        regime=0.
    y=foot-ry0*scale
    return x,y,scale,roll,axis,regime


def render(t):
    cx,cy,scale,theta,axis,regime=state(max(0,t))
    # Keep camera and all stationary materials pixel-stable.
    out=B.copy()
    left=max(0,int(np.floor(cx+(xlo-cx0)*scale))-3)
    right=min(W,int(np.ceil(cx+(xhi-cx0)*scale))+3)
    top=max(0,int(np.floor(cy+(ylo-cy0)*scale))-3)
    bottom=min(H,int(np.ceil(cy+(yhi-cy0)*scale))+3)
    if right<=left or bottom<=top:return out
    yy,xx=np.mgrid[top:bottom,left:right]
    sx=(xx-cx)/scale+cx0-xlo
    sy=(yy-cy)/scale+cy0-ylo
    # The shadow lies on the receiving plane and follows its change in slope.
    shx=sx+regime*3.5
    shy=sy+regime*.28*(sx-(cx0-xlo))
    sh=bilinear(S,shx,shy)
    a=bilinear(AF,sx,sy)
    fg=bilinear(FF,sx,sy)
    ux=((xx-cx)/scale)/rx0
    uy=((yy-cy)/scale)/ry0
    z=np.sqrt(np.maximum(0,1-ux*ux-uy*uy))
    n=np.stack([ux,uy,z],axis=-1)
    # Inverse rotation of surface normals around the no-slip rolling axis.
    ca,sa=np.cos(theta),np.sin(theta)
    rotated=n*ca-np.cross(axis,n)*sa+axis*np.sum(n*axis,axis=-1)[...,None]*(1-ca)
    tx=cx0-xlo+rx0*rotated[...,0]
    ty=cy0-ylo+ry0*rotated[...,1]
    rrot=bilinear(RR,tx,ty)
    rbase=bilinear(RR,sx,sy)
    # Restrict material motion to the well-observed part of the sphere; lighting
    # stays fixed in camera coordinates rather than spinning with the ball.
    radial=ux*ux+uy*uy
    material_weight=np.clip((.94-radial)/.21,0,1)
    fg=np.clip(fg+.50*(rrot-rbase)*material_weight[...,None],0,255)
    patch=out[top:bottom,left:right]+sh
    patch=patch*(1-a[...,None])+fg*a[...,None]
    out[top:bottom,left:right]=patch
    # The pipe occludes the returning ball at the extreme lower-left opening.
    pipe=(X<48)&(Y>193)&(Y<259)&(I.mean(axis=-1)<40)
    if cx<65:
        out[pipe]=I[pipe]
    return np.clip(out,0,255)
