+def poly_backdrop(bounds, exclusions):

def poly_backdrop(bounds, exclusions):
    x0,y0,x1,y1=bounds
    yy,xx=np.mgrid[y0:y1,x0:x1]
    valid=np.ones(xx.shape,bool)
    for a,b,c,d in exclusions:
        valid &= ~((xx>=a)&(xx<c)&(yy>=b)&(yy<d))
    # A low-order continuous surface follows the original backdrop illumination.
    u=(xx-xx.mean())/60; v=(yy-yy.mean())/60
    terms=np.stack([np.ones_like(u),u,v,u*u,u*v,v*v,u**3,u*u*v,u*v*v,v**3],-1)
    coeff=np.linalg.lstsq(terms[valid],original[y0:y1,x0:x1][valid],rcond=None)[0]
    uf=(X-xx.mean())/60;vf=(Y-yy.mean())/60
    tf=np.stack([np.ones_like(uf),uf,vf,uf*uf,uf*vf,vf*vf,uf**3,uf*uf*vf,uf*vf*vf,vf**3],-1)
    return np.clip(tf@coeff,0,255).astype(np.float32)

def sprite(alpha,box,bg):
    x0,y0,x1,y1=box
    a=alpha[y0:y1,x0:x1].copy()
    # Recover foreground radiance from the actual clean plate, so the source
    # photograph is reproduced by over compositing at its original position.

    rgb=np.clip(rgb,0,255)
    rgba=np.concatenate([rgb*a[:,:,None],a[:,:,None]],-1).astype(np.float32)
    return {'rgba':rgba,'box':box}

def sample(array,x,y):
    \"\"\"Bilinear inverse sampling with transparent out-of-bounds.\"\"\"
    hh,ww=array.shape[:2]
    good=(x>=0)&(y>=0)&(x<=ww-1)&(y<=hh-1)
    x=np.clip(x,0,ww-1); y=np.clip(y,0,hh-1)
    ix=x.astype(np.int32);iy=y.astype(np.int32)
    jx=np.minimum(ix+1,ww-1);jy=np.minimum(iy+1,hh-1)
    fx=(x-ix)[...,None];fy=(y-iy)[...,None]
    q=(array[iy,ix]*(1-fx)+array[iy,jx]*fx)*(1-fy)+(array[jy,ix]*(1-fx)+array[jy,jx]*fx)*fy
    return q*good[...,None]


def affine_over(canvas,s,center,newcenter,angle=0,sx=1,sy=1):
    # Work only in the target bounding box, keeping the rest of the image fixed.
    x0,y0,x1,y1=s['box']; cx,cy=center;nx,ny=newcenter
    co,si=math.cos(angle),math.sin(angle)
    points=np.array([[x0,y0],[x1,y0],[x0,y1],[x1,y1]],float)-[cx,cy]
    points=points*[sx,sy]
    points=points@np.array([[co,si],[-si,co]])+[nx,ny]
    ax=max(0,int(np.floor(points[:,0].min()))-2);bx=min(W,int(np.ceil(points[:,0].max()))+2)
    ay=max(0,int(np.floor(points[:,1].min()))-2);by=min(H,int(np.ceil(points[:,1].max()))+2)
    if ax>=bx or ay>=by:return
    yy,xx=np.mgrid[ay:by,ax:bx].astype(float)
    u=xx-nx;v=yy-ny
    source_x=(co*u+si*v)/sx+cx-x0
    source_y=(-si*u+co*v)/sy+cy-y0
    p=sample(s['rgba'],source_x,source_y)
    canvas[ay:by,ax:bx]=canvas[ay:by,ax:bx]*(1-p[:,:,3:4])+p[:,:,:3]


def open_finger(canvas,s,opening,direction,pivot_y=32):

    yy,xx=np.mgrid[ay:by,ax:bx].astype(float)

    p=sample(s['rgba'],xx-displacement-x0,yy-y0)

def pillow_warp(plate,bdepth,kdepth,ballx,blockx):
    # A local yielding textile surface, not a rigid table impact.
    ball_shape=np.exp(-((X-ballx)/35)**2-((Y-237)/17)**2)*smooth((Y-199)/15)
    block_shape=np.exp(-((X-blockx)/31)**2-((Y-236)/15)**2)*smooth((Y-198)/15)
    dy=bdepth*ball_shape+kdepth*block_shape
    dx=-.9*bdepth*(X-ballx)/35*ball_shape-.5*kdepth*(X-blockx)/31*block_shape
    out=sample(plate,X+dx,np.clip(Y-dy,0,H-1))
    # Very slight change in self-shadowing around the depression's rim.
    shade=1-.014*bdepth*ball_shape-.014*kdepth*block_shape
    return out*shade[:,:,None]

    q=np.exp(-.5*(((X-x)/rx)**2+((Y-y)/ry)**2))
    canvas*=1-opacity*q[:,:,None]


def ball_motion(t):
    release=.38;initial=64.6;contact=212.5;g=850
    hit=release+math.sqrt(2*(contact-initial)/g)
    if t<release:return 168.2,initial,0,hit
    if t<hit:
        dt=t-release
        return 168.2+.7*dt,initial+.5*g*dt*dt,.12*dt,hit
    u=t-hit
    # Pillow indentation followed by one low, strongly damped rebound.
    if u<.085:
        cy=contact+3.0*math.sin(math.pi*u/.17)
    elif u<.36:
        q=(u-.085)/.275
        cy=contact+3*(1-q)-8.4*math.sin(math.pi*q)
    else:
        cy=contact+1.5+.75*math.exp(-(u-.36)*7)*math.sin((u-.36)*22)
    cx=168.6+1.7*smooth(u/.5)
    angle=.075+.10*smooth(u/.7)
    return cx,cy,angle,hit


def block_motion(t):
    release=.31;cx,cy=432.5,97.0;halfheight=20.5;floor=235.5;g=850
    hit=release+math.sqrt(2*(floor-halfheight-cy)/g)
    if t<release:return cx,cy,0,hit,cx
    if t<hit:
        dt=t-release
        return cx+.30*dt,cy+.5*g*dt*dt,.035*dt,hit,cx
    u=t-hit
    # Its narrow bottom face first presses into the pillow, then the tall
    # block tips toward the right, maintaining support contact throughout.
    tip=smooth((u-.09)/.62)
    angle=math.radians(80)*tip
    settling=.65*math.exp(-max(0,u-.7)*9)*math.sin(max(0,u-.7)*19) if u>.7 else 0
    floor_y=floor+1.25+2.0*math.exp(-u*8)*math.sin(u*24)+settling
    # The same lower-right material corner remains the pivot until lying down.
    pivot_x=439.5
    nx=pivot_x-7*math.cos(angle)+halfheight*math.sin(angle)
    ny=floor_y-7*math.sin(angle)-halfheight*math.cos(angle)
    return nx,ny,angle,hit,nx


def render(t):
    bx,by,ba,bhit=ball_motion(t)
    kx,ky,ka,khit,contactx=block_motion(t)
    bu=max(0,t-bhit);ku=max(0,t-khit)
    bdepth=(1.25+2.5*math.exp(-bu*7)*max(0,math.sin(bu*23))) *smooth(bu/.065) if t>=bhit else 0
    kdepth=(1.3+1.7*math.exp(-ku*7)*max(0,math.sin(ku*25))) *smooth(ku/.055) if t>=khit else 0
    canvas=pillow_warp(clean,bdepth,kdepth,bx,contactx)
    release_fade=smooth((t-.20)/.3)
    approach=np.clip((by-64.6)/(212.5-64.6),0,1)
    shadow(canvas,bx+4,238+1.0*bdepth,25-10*approach,8-3*approach,release_fade*(.014+.18*approach**3))
    approach2=np.clip((ky-97)/(214-97),0,1) if t<khit else 1
    shadow(canvas,kx+2,237+1.0*kdepth,17+9*math.sin(ka),5.5,release_fade*(.01+.14*approach2**3))
    # A concentrated contact shadow develops only on touchdown.
    shadow(canvas,bx+1,237.2+1.4*bdepth,12,2.4,.12*smooth(bu/.08) if t>=bhit else 0)
    shadow(canvas,kx,236.5+1.3*kdepth,11+7*math.sin(ka),2.2,.10*smooth(ku/.08) if t>=khit else 0)
    opening=smooth((t-.16)/.26)
    open_finger(canvas,larm,7.2*opening,-1)
    open_finger(canvas,rarm,7.2*opening,1)
    right_open(canvas,3.3*smooth((t-.12)/.24))
    # Short exposure integration provides unobtrusive physical motion blur.
    samples=[]
    if .38<t<max(bhit,khit):
        samples=[t-.004,t,t+.004]
    else:samples=[t]
    object_views=[]
    for st in samples:
        view=canvas.copy()
        xx,yy,ang,_=ball_motion(st)
        kxx,kyy,kang,_,_=block_motion(st)
        compression=1-.025*math.exp(-bu*10)*max(0,math.sin(bu*25)) if t>=bhit else 1
        affine_over(view,ball,(168.2,64.6),(xx,yy),ang,sx=1+(.01 if compression<1 else 0),sy=compression)
        affine_over(view,block,(432.5,97),(kxx,kyy),kang)
        object_views.append(view)
    return np.clip(np.mean(object_views,axis=0)+.5,0,255).astype(np.uint8)
