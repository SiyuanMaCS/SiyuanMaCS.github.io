+
+No external assets or rendering systems: the locked scene is the input image,
+the rolling sphere is reconstructed from its own pixels, and the dynamics are
+an elementary rolling-body integration. Run this file to rebuild all outputs.

+import numpy as np
+from PIL import Image
+import cv2
+from scipy.ndimage import gaussian_filter, map_coordinates
+import artifact_io

+CX, CY = 126.35, 89.40
+RX, RY, SKEW = 16.8, 15.5, .115

+def scene_layers(image):
+    yy, xx = np.mgrid[:H, :W].astype(np.float64)
+    # The soft tabletop shadow and the sphere are the only movable pixels.
+    # Restore their small footprint by a harmonic interpolation from its
+    # unoccluded boundary, with the input's local fine-grain texture retained.
+    ex = (xx - 132.0 - .12*(yy-90.0))/48.5
+    ey = (yy-90.0)/22.8
+    dist = np.sqrt(ex*ex+ey*ey)
+    hole = dist < .94
+    work = image.copy()
+    seed = cv2.inpaint(image.astype(np.uint8), hole.astype(np.uint8)*255,
+                       5.0, cv2.INPAINT_NS).astype(float)
+    # Laplace's equation removes radial artifacts of the inpaint seed.
+    crop = (slice(64, 118), slice(78, 187))
+    patch = seed[crop].copy()
+    fixed = image[crop]
+    mh = hole[crop]
+    for _ in range(1200):
+        neighbor = (np.roll(patch,1,0)+np.roll(patch,-1,0)
+                    + np.roll(patch,1,1)+np.roll(patch,-1,1))*.25
+        patch[mh] = neighbor[mh]
+        patch[~mh] = fixed[~mh]
+    # A tiny, genuine wood-grain residual prevents an unnaturally smooth fill.
+    grain = image[64:118, 153:262] - gaussian_filter(
+        image[64:118,153:262], (1.1,1.1,0))
+    feather = np.clip((.94-dist[crop])*12, 0, 1)
+    patch += grain * (.50*feather[...,None])
+    work[crop] = patch
+    background = work
+
+    nx = (xx-CX-SKEW*(yy-CY))/RX
+    ny = (yy-CY)/RY
+    radius = np.sqrt(nx*nx+ny*ny)
+    # Felt edge: subpixel coverage of the original slightly elliptical ball.
+    alpha = np.clip((1.025-radius)*14,0,1)
+    alpha *= ((xx>105)&(xx<147)&(yy>70)&(yy<108))
+    # Use color support as well to catch the original edge's few fuzzy pixels.
+    colored = ((image[...,1]-image[...,0]>5)
+               &(image[...,1]-image[...,2]>30)
+               &(radius<1.12))
+    coverage = gaussian_filter(colored.astype(float), .48)
+    alpha = np.maximum(alpha, coverage*.93)
+    alpha = np.clip(alpha,0,1)
+
+    # Extract a multiplicative shadow, not grey tabletop pixels.  This keeps
+    # its color natural when it moves from the wood to brown cardboard.
+    ratio = image / np.maximum(background,1)
+    outside_ball = alpha < .02
+    ratio = np.clip(ratio,.53,1.018)
+    # Interpolate the part of the shadow hidden beneath the sphere.
+    phi = np.exp(-.5*(((xx-139.0)/26.0)**2+((yy-94.0)/9.2)**2))
+    under = (1-.195*phi)[...,None] * np.ones((1,1,3))
+    shadow = ratio.copy()
+    shadow[~outside_ball] = under[~outside_ball]
+    shadow = gaussian_filter(shadow,(.65,.65,0))
+    edge = np.clip((.96-dist)*15,0,1)
+    shadow = 1+(shadow-1)*edge[...,None]
+    # Foreground unmixing recovers antialiased felt against the shadowed table.
+    shadowed = background*shadow
+    foreground = (image - shadowed*(1-alpha[...,None])) / np.maximum(alpha[...,None],.08)
+    foreground = np.clip(foreground,0,255)
+
+    # Fit only low spatial frequencies of the sphere's illumination.  The
+    # residual (including its felt and seam) is a rotating material texture.
+    nz = np.sqrt(np.maximum(0,1-nx*nx-ny*ny))
+    features = np.stack([np.ones_like(nx),nx,ny,nz,nx*nx,
+                         nx*ny,ny*ny,nx*nz,ny*nz],-1)
+    inner = radius < .90
+    design = features[inner]
+    target = foreground[inner]
+    coefficients = np.linalg.lstsq(design,target,rcond=None)[0]
+    for _ in range(3):
+        residual = target-design@coefficients
+        robust = np.minimum(1,5/np.maximum(1,np.sqrt(np.mean(residual**2,axis=1))))
+        coefficients = np.linalg.lstsq(design*robust[:,None],target*robust[:,None],rcond=None)[0]
+    smooth = features@coefficients
+    texture = foreground-smooth
+    texture *= np.clip((.97-radius)/.08,0,1)[...,None]
+    texture = np.clip(texture,-17,17)
+    # Save only very small source patches for warping.
+    sy,sx = slice(68,112),slice(104,150)
+    return background, shadow[64:118,78:187].copy(), coefficients, texture[sy,sx].copy(), alpha[sy,sx].copy()
+

+def dynamics():
+    # Contact with the low end of the ramp. The direction on the table and
+    # the direction on the elevated plane are projections of the same line.
+    xedge = 250.0
+    approach_slope = .114
+    ramp_slope = -.269
+    initial_speed = 465.0
+    table_resistance = 27.0
+    ramp_acceleration = 760.0

+        # Modest perspective shrink with distance up the board.
+        size = 1.0-.00035*max(0,x-xedge)
+        # Center moves a little down as the projected radius diminishes.
+        center_y += RY*(1-size)
+        samples.append((t,x,center_y,size,arc,v))
+        acceleration = -ramp_acceleration if uphill else -table_resistance*np.sign(v)
+        dx = v*step+.5*acceleration*step*step
+        x += dx
+        v += acceleration*step
+        arc += dx*np.sqrt(1+slope*slope)/(RX*size)
+    return np.asarray(samples)
+

+def render():
+    input_rgb = np.array(Image.open(ROOT/'first_frame.png').convert('RGB'))
+    original = input_rgb.astype(float)

+        picture = background.copy()
+        if cx < -27:
+            return picture

+        # Shearing its footprint tilts it with the cardboard surface, whereas
+        # the sphere itself stays round rather than being warped with the ramp.

+        xmax = min(W,int(np.ceil(cx+21*size)))
+        ymin = max(0,int(np.floor(cy-19*size)))
+        ymax = min(H,int(np.ceil(cy+19*size)))
+        if xmin >= xmax or ymin >= ymax:
+            return picture
+        gy,gx = np.mgrid[ymin:ymax,xmin:xmax].astype(float)
+        sx = (gx-cx)/size+CX
+        sy = (gy-cy)/size+CY

+        ay = (sy-CY)/RY
+        az = np.sqrt(np.maximum(0,1-ax*ax-ay*ay))

+        shading = f@light
+        # A rightward rolling sphere has angular velocity about its vertical
+        # image axis. Inverse Rodrigues rotation samples fixed material points.
+        direction = -.269 if cx>250 else .114
+        axis = np.array([-direction,1.,0.])
+        axis /= np.linalg.norm(axis)
+        normal = np.stack([ax,ay,az],-1)
+        cos,sin = np.cos(-angle),np.sin(-angle)
+        rotated = (normal*cos + np.cross(axis,normal)*sin
+                   + np.sum(normal*axis,axis=-1)[...,None]*axis*(1-cos))
+        material_x,material_y,material_z = rotated.transpose(2,0,1)
+        tx = CX+RX*material_x+SKEW*RY*material_y-104
+        ty = CY+RY*material_y-68
+        texcoords = np.stack([ty,tx])
+        felt = np.stack([map_coordinates(texture[...,c],texcoords,order=1,
+                         mode='nearest',prefilter=False) for c in range(3)],-1)
+        # Extend the subtle pale stitched seam onto the unobserved rear half.
+        seam_distance = material_x+.32-.24*material_y*material_y
+        seam = np.exp(-.5*(seam_distance/.052)**2)
+        backside = np.clip(-material_z*5,0,1)
+        felt += (seam*backside)[...,None]*np.array([3.8,3.0,8.0])
+        ball = np.clip(shading+felt,0,255)
+        alpha = map_coordinates(source_alpha,np.stack([sy-68,sx-104]),
+                                order=1,mode='constant',cval=0,prefilter=False)
+        # On returning to the aperture, the pipe lip passes in front of the
+        # sphere; all pipe pixels remain exactly the supplied original pixels.
+        if cx < 42:
+            lip = np.clip((gx-25.0)/4.0,0,1)
+            alpha *= lip
+        roi = picture[ymin:ymax,xmin:xmax]
+        picture[ymin:ymax,xmin:xmax] = roi*(1-alpha[...,None])+ball*alpha[...,None]
+        return picture
+
+    # Finite shutter integration is important at this real-time rolling speed;
+    # it gives a few pixels of natural exposure blur instead of hard jumps.
+    for i in range(1,N):
+        t = i/FPS
+        exposure = 1/96
+        offsets = np.linspace(-exposure*.5,exposure*.5,5)
+        result = sum(instant(t+dt) for dt in offsets)/len(offsets)
+        frames[i] = np.rint(np.clip(result,0,255)).astype(np.uint8)
+    return frames, motion
+

        robust = np.minimum(1,5/np.maximum(1,np.sqrt(np.mean(residual**2,axis=1))))
        coefficients = np.linalg.lstsq(design*robust[:,None],target*robust[:,None],rcond=None)[0]
    smooth = features@coefficients
    texture = foreground-smooth
    texture *= np.clip((.97-radius)/.08,0,1)[...,None]
    texture = np.clip(texture,-17,17)
    # Save only very small source patches for warping.
    sy,sx = slice(68,112),slice(104,150)
    return background, shadow[64:118,78:187].copy(), coefficients, texture[sy,sx].copy(), alpha[sy,sx].copy()


def dynamics():
    # Contact with the low end of the ramp. The direction on the table and
    # the direction on the elevated plane are projections of the same line.
    xedge = 250.0
    approach_slope = .114
    ramp_slope = -.269
    initial_speed = 465.0
    table_resistance = 27.0
    ramp_acceleration = 760.0
    step = 1/2400
    x = CX
    v = initial_speed
    arc = 0.
    samples = []
    tt = np.arange(0, (N+1)/FPS+step*.5, step)
    cy_edge = CY+approach_slope*(xedge-CX)
    for t in tt:
        uphill = x >= xedge
        slope = ramp_slope if uphill else approach_slope
        center_y = cy_edge+ramp_slope*(x-xedge) if uphill else CY+approach_slope*(x-CX)
        # Modest perspective shrink with distance up the board.
        size = 1.0-.00035*max(0,x-xedge)
        # Center moves a little down as the projected radius diminishes.
        center_y += RY*(1-size)
        samples.append((t,x,center_y,size,arc,v))
        acceleration = -ramp_acceleration if uphill else -table_resistance*np.sign(v)
        dx = v*step+.5*acceleration*step*step
        x += dx
        v += acceleration*step
        arc += dx*np.sqrt(1+slope*slope)/(RX*size)
    return np.asarray(samples)


def render():
    input_rgb = np.array(Image.open(ROOT/'first_frame.png').convert('RGB'))
    original = input_rgb.astype(float)
    background, shadow_patch, light, texture, source_alpha = scene_layers(original)
    motion = dynamics()
    yy,xx = np.mgrid[:H,:W].astype(float)
    frames = np.empty((N,H,W,3),np.uint8)
    frames[0] = input_rgb

    def state(t):
        return np.array([np.interp(t,motion[:,0],motion[:,i]) for i in range(1,6)])

    def instant(t):
        cx,cy,size,angle,velocity = state(t)
        picture = background.copy()
        if cx < -27:
            return picture
        # Cast the original diffuse shadow onto the current supporting plane.
        # Shearing its footprint tilts it with the cardboard surface, whereas
        # the sphere itself stays round rather than being warped with the ramp.
        k = np.clip((cx-239)/23,0,1)
        shear = -.36*k
        local_x = (xx-cx)/size+CX
        local_y = (yy-cy)/size+CY-shear*(local_x-CX)
        coords = np.stack([local_y-64,local_x-78])
        for c in range(3):
            sh = map_coordinates(shadow_patch[...,c],coords,order=1,mode='constant',cval=1,prefilter=False)
            picture[...,c] *= sh

        xmin = max(0,int(np.floor(cx-21*size)))
        xmax = min(W,int(np.ceil(cx+21*size)))
        ymin = max(0,int(np.floor(cy-19*size)))
        ymax = min(H,int(np.ceil(cy+19*size)))
        if xmin >= xmax or ymin >= ymax:
            return picture
        gy,gx = np.mgrid[ymin:ymax,xmin:xmax].astype(float)
        sx = (gx-cx)/size+CX
        sy = (gy-cy)/size+CY
        ax = (sx-CX-SKEW*(sy-CY))/RX
        ay = (sy-CY)/RY
        az = np.sqrt(np.maximum(0,1-ax*ax-ay*ay))
        f = np.stack([np.ones_like(ax),ax,ay,az,ax*ax,ax*ay,ay*ay,ax*az,ay*az],-1)
        shading = f@light
        # A rightward rolling sphere has angular velocity about its vertical
        # image axis. Inverse Rodrigues rotation samples fixed material points.
        direction = -.269 if cx>250 else .114
        axis = np.array([-direction,1.,0.])
        axis /= np.linalg.norm(axis)
        normal = np.stack([ax,ay,az],-1)
        cos,sin = np.cos(-angle),np.sin(-angle)
        rotated = (normal*cos + np.cross(axis,normal)*sin
                   + np.sum(normal*axis,axis=-1)[...,None]*axis*(1-cos))
        material_x,material_y,material_z = rotated.transpose(2,0,1)
        tx = CX+RX*material_x+SKEW*RY*material_y-104
        ty = CY+RY*material_y-68
        texcoords = np.stack([ty,tx])
        felt = np.stack([map_coordinates(texture[...,c],texcoords,order=1,
                         mode='nearest',prefilter=False) for c in range(3)],-1)
        # Extend the subtle pale stitched seam onto the unobserved rear half.
        seam_distance = material_x+.32-.24*material_y*material_y
        seam = np.exp(-.5*(seam_distance/.052)**2)
        backside = np.clip(-material_z*5,0,1)
        felt += (seam*backside)[...,None]*np.array([3.8,3.0,8.0])
        ball = np.clip(shading+felt,0,255)
        alpha = map_coordinates(source_alpha,np.stack([sy-68,sx-104]),
                                order=1,mode='constant',cval=0,prefilter=False)
        # On returning to the aperture, the pipe lip passes in front of the
        # sphere; all pipe pixels remain exactly the supplied original pixels.
        if cx < 42:
            lip = np.clip((gx-25.0)/4.0,0,1)
            alpha *= lip
        roi = picture[ymin:ymax,xmin:xmax]
        picture[ymin:ymax,xmin:xmax] = roi*(1-alpha[...,None])+ball*alpha[...,None]
        return picture

    # Finite shutter integration is important at this real-time rolling speed;
    # it gives a few pixels of natural exposure blur instead of hard jumps.
    for i in range(1,N):
        t = i/FPS
        exposure = 1/96
        offsets = np.linspace(-exposure*.5,exposure*.5,5)
        result = sum(instant(t+dt) for dt in offsets)/len(offsets)
        frames[i] = np.rint(np.clip(result,0,255)).astype(np.uint8)
    return frames, motion


if __name__ == '__main__':
    video, motion = render()
    artifact_io.save(video,ROOT)
    print('Saved',video.shape,video.dtype)
    for t in [0,.25,.5,.75,1,1.25,1.5,1.75,2,2.25,3,4.96]:
