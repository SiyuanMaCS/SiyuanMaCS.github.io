    # ball; this provides a strong visual cue that it rolls rather than slides.
    frames=np.empty((N,H,W,3),np.uint8)
    shadow_rng=np.random.default_rng(24)
    for k in range(N):
        if k==0:
            frames[k]=first
            continue
        t=k/(N-1)
        # Begin in the pipe during the first moments, then cross almost the full table.
        # Slight ease-in corresponds to acceleration down/out of the pipe; afterwards
        # rolling friction causes gentle deceleration.
        p=cosine_blend(min(1,t/.10))*.055 if t<.10 else .055+(.945)*((t-.10)/.90)**.90
        cx=-8+508*p
        # tabletop's far edge rises gently toward the left; ball stays on its plane
        cy=150.0+25.0*p+0.65*np.sin(np.pi*p)
        radius=14.2+2.0*p

        frame=clean.copy()
        # Contact shadow: narrow at the contact point, offset right from upper-left light.
        sx=cx+2.5; sy=cy+radius*.79
        rx=radius*1.12; ry=radius*.27
        sh=np.exp(-2.75*(((xx-sx)/rx)**2+((yy-sy)/ry)**2))
        # prevent darkening beyond table edge at far right
        frame*= (1-0.30*sh[...,None])

        # Sub-pixel warp from the source ball to current size, rotating according to
        # no-slip rolling: theta = distance / radius.
        angle=(cx-(-8))/15.2
        ca,si=np.cos(angle),np.sin(angle)
        u=(xx-cx)/(radius/14.5)
        v=(yy-cy)/(radius/14.5)
        # reverse map into the original crop centered at (20,20)
        mapx=(ca*u+si*v+20.2).astype(np.float32)
        mapy=(-si*u+ca*v+19.5).astype(np.float32)
        tex=cv2.remap(src,mapx,mapy,cv2.INTER_CUBIC,borderMode=cv2.BORDER_REFLECT_101)
        al=cv2.remap(sa,mapx,mapy,cv2.INTER_LINEAR,borderMode=cv2.BORDER_CONSTANT)
        radial=np.hypot(xx-cx,yy-cy)
        al*=np.clip((radius+.45-radial)/1.05,0,1)

        # Draw two fuzzy curved tennis-ball seams in object coordinates.
        seam=np.zeros((H,W),np.float32)
        m=radial<radius*.91
        # Curves on opposite hemispheres, rotated in map coordinates.
        ou=ca*u+si*v; ov=-si*u+ca*v
        curve1=np.abs(ou-(5.7-0.027*ov*ov))
        curve2=np.abs(ou-(-5.7+0.027*ov*ov))
        seam[m]=np.maximum(np.exp(-(curve1[m]/.70)**2),np.exp(-(curve2[m]/.70)**2))
        seam*=.22
        tex=tex*(1-seam[...,None])+np.array([84,87,89],np.float32)*seam[...,None]

        # Very subtle directional blur at peak speed, below one pixel at 24 fps.
        if 4<k<112:
            tex=(tex*.74+np.roll(tex,-1,axis=1)*.26)
            al=(al*.74+np.roll(al,-1,axis=1)*.26)
        a=al[...,None]
        frame=frame*(1-a)+tex*a
        frames[k]=np.clip(frame,0,255).astype(np.uint8)

    frames[0]=first # explicit contract guarantee
    artifact_io.save(frames, CASE)
