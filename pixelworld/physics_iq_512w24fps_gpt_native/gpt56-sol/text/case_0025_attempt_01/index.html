+"""Generate the requested locked-camera scientific demonstration.
+
+The photograph itself supplies all appearance.  The script removes the ball from
+the stationary plate with a small, texture-aware inpaint, extracts the original
+ball as a premultiplied sprite, and rolls that sprite across the photographed
+surface.  After the far edge of the glass it follows a short ballistic arc.
+"""
+
+from pathlib import Path
+import json
+
+import cv2
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+
+
+def ease_accelerating_position(t: float) -> tuple[float, float, float, bool]:
+    """Return center x/y, travelled distance, and whether the ball is supported.
+
+    Times are in frames.  The first twelve moving frames model a gentle push;
+    thereafter rolling is uniform.  At frame 104 the center clears the glass
+    and ordinary image-plane gravity takes over.
+    """
+    cx0, cy0 = 262.15, 151.85
+    start, ramp_end, edge = 6.0, 18.0, 104.0
+    speed = 1.82                 # pixels/frame once rolling steadily
+    accel = speed / (ramp_end - start)
+
+    if t <= start:
+        return cx0, cy0, 0.0, True
+    if t <= ramp_end:
+        dt = t - start
+        distance = 0.5 * accel * dt * dt
+        return cx0 + distance, cy0, distance, True
+
+    ramp_distance = 0.5 * speed * (ramp_end - start)
+    if t <= edge:
+        distance = ramp_distance + speed * (t - ramp_end)
+        return cx0 + distance, cy0, distance, True
+
+    edge_distance = ramp_distance + speed * (edge - ramp_end)
+    dt = t - edge
+    # The ball preserves its horizontal velocity.  1.30 px/frame^2 gives a
+    # convincing real-time fall at this photographic scale (y = 1/2*g*t^2).
+    x = cx0 + edge_distance + speed * dt
+    y = cy0 + 0.65 * dt * dt
+    distance = edge_distance + speed * dt
+    return x, y, distance, False
+
+
+def make_clean_plate_and_ball(source: np.ndarray):
+    """Build the stationary clean plate and a premultiplied photographed ball."""
+    h, w = source.shape[:2]
+    yy, xx = np.mgrid[:h, :w]
+    cx, cy = 262.15, 151.85
+
+    # The ball is the only compact, strongly blue component in this part of the
+    # photograph.  Keeping its measured silhouette retains hairs and soft edges.
+    r = source[:, :, 0].astype(np.int16)
+    g = source[:, :, 1].astype(np.int16)
+    b = source[:, :, 2].astype(np.int16)
+    blue = ((b - r > 10) & (b - g > 5) & (xx > 225) & (xx < 300)
+            & (yy > 115) & (yy < 190)).astype(np.uint8)
+    count, labels, stats, _ = cv2.connectedComponentsWithStats(blue, 8)
+    if count < 2:
+        raise RuntimeError("Could not isolate the photographed blue ball")
+    component = 1 + int(np.argmax(stats[1:, cv2.CC_STAT_AREA]))
+    silhouette = (labels == component).astype(np.uint8)
+    silhouette = cv2.morphologyEx(
+        silhouette, cv2.MORPH_CLOSE,
+        cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (5, 5)),
+    )
+    # Fill any enclosed texture holes without changing the external contour.
+    contours, _ = cv2.findContours(silhouette, cv2.RETR_EXTERNAL,
+                                   cv2.CHAIN_APPROX_SIMPLE)
+    filled = np.zeros_like(silhouette)
+    cv2.drawContours(filled, contours, -1, 1, thickness=cv2.FILLED)
+    alpha = cv2.GaussianBlur(filled.astype(np.float32), (0, 0), 0.72)
+    alpha = np.clip(alpha, 0.0, 1.0)
+
+    # Remove both the ball and its very small contact shadow.  The wider soft
+    # mask allows Telea's interpolation to continue the subtle wood grain.
+    distance = np.sqrt(((xx - cx) / 1.04) ** 2 + ((yy - cy) / 1.00) ** 2)
+    removal = (distance < 28.0).astype(np.uint8) * 255
+    removal[((xx - (cx + 1.5)) / 25.0) ** 2
+            + ((yy - (cy + 22.0)) / 8.0) ** 2 < 1.0] = 255
+    clean = cv2.inpaint(source, removal, 5.0, cv2.INPAINT_TELEA)
+
+    # Premultiplied color chosen so that at angle zero and the source location,
+    # clean*(1-alpha)+premul reconstructs the photograph nearly exactly.
+    a3 = alpha[:, :, None]
+    premul = source.astype(np.float32) - clean.astype(np.float32) * (1.0 - a3)
+    # Zeroing the numerically empty area makes affine interpolation clean.
+    premul[alpha < 1e-4] = 0.0
+    return clean, alpha, premul
+
+
+def transformed_sprite(alpha: np.ndarray, premul: np.ndarray,
+                       x: float, y: float, angle: float):
+    """Rotate around the photographed center and translate to (x,y)."""
+    h, w = alpha.shape
+    center = (262.15, 151.85)
+    matrix = cv2.getRotationMatrix2D(center, angle, 1.0)
+    matrix[0, 2] += x - center[0]
+    matrix[1, 2] += y - center[1]
+    a = cv2.warpAffine(alpha, matrix, (w, h), flags=cv2.INTER_CUBIC,
+                       borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+    p = cv2.warpAffine(premul, matrix, (w, h), flags=cv2.INTER_CUBIC,
+                       borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+    return np.clip(a, 0.0, 1.0), p
+
+
+def apply_contact_shadow(plate: np.ndarray, x: float, y: float,
+                         supported: bool) -> np.ndarray:
+    """Add a restrained contact shadow that fades while traversing clear glass."""
+    if not supported:
+        return plate
+    h, w = plate.shape[:2]
+    yy, xx = np.mgrid[:h, :w]
+    # A wooden surface gives a slightly stronger shadow than transparent glass.
+    glass_mix = np.clip((x - 322.0) / 42.0, 0.0, 1.0)
+    opacity = 0.105 * (1.0 - glass_mix) + 0.047 * glass_mix
+    sx, sy = x + 1.5, y + 22.0
+    shadow = np.exp(-0.5 * (((xx - sx) / 14.5) ** 2
+                            + ((yy - sy) / 3.0) ** 2))
+    factor = (1.0 - opacity * shadow)[:, :, None]
+    return plate * factor
+
+
+def render_sample(clean: np.ndarray, alpha: np.ndarray, premul: np.ndarray,
+                  t: float) -> np.ndarray:
+    x, y, distance, supported = ease_accelerating_position(t)
+    # Rolling right means clockwise spin.  The physical radius measured from
+    # the photograph is about 22 pixels; keep the angular speed after takeoff.
+    angle = np.degrees(distance / 22.0)
+    base = apply_contact_shadow(clean.astype(np.float32), x, y, supported)
+    a, p = transformed_sprite(alpha, premul, x, y, angle)
+    result = base * (1.0 - a[:, :, None]) + p
+    return np.clip(result, 0.0, 255.0)
+
+
+def generate() -> np.ndarray:
+    spec = json.loads((ROOT / "task_spec.json").read_text())
+    source = np.asarray(Image.open(ROOT / "first_frame.png").convert("RGB"),
+                        dtype=np.uint8)
+    expected = (spec["height"], spec["width"], 3)
+    if source.shape != expected:
+        raise ValueError(f"first_frame shape {source.shape}, expected {expected}")
+
+    clean, alpha, premul = make_clean_plate_and_ball(source)
+    frames = np.empty((spec["frames"], *source.shape), dtype=np.uint8)
+
+    # Preserve the contractual first image exactly.  The short initial hold is
+    # natural for a demonstration and also makes the onset of motion legible.
+    for index in range(spec["frames"]):
+        if index <= 6:
+            frames[index] = source
+            continue
+        # Five temporal samples give a photographic 270-degree shutter.  During
+        # the slow roll this is almost imperceptible; it becomes useful on the
+        # gravity-driven drop near the end.
+        samples = []
+        for offset in np.linspace(-0.375, 0.375, 5):
+            samples.append(render_sample(clean, alpha, premul, index + offset))
+        frames[index] = np.clip(np.mean(samples, axis=0) + 0.5,
+                                0, 255).astype(np.uint8)
+
+    # Contractual invariant, asserted rather than merely assumed.
+    if not np.array_equal(frames[0], source):
+        raise AssertionError("frame 00 changed")
+    return frames
+
+
+if __name__ == "__main__":
+    output = generate()
+    artifact_io.save(output, ROOT)
+    print(f"saved {len(output)} RGB frames at {output.shape[2]}x{output.shape[1]}")
+
