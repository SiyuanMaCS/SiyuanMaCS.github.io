+"""Generate the requested fixed-camera RGB sequence from first_frame.png.
+
+The animation is deliberately composited from pixels in the supplied frame.  The
+white ceramic tile is isolated, the area behind it is reconstructed locally,
+and the tile is then dropped with gravitational acceleration into the dark
+liquid.  Small physically-timed splashes and decaying surface ripples complete
+the demonstration.
+"""
+from pathlib import Path
+import cv2
+import numpy as np
+from PIL import Image
+import artifact_io
+
+CASE_DIR = Path(__file__).resolve().parent
+W, H, N = 512, 288, 120
+
+
+def over(dst, src, alpha):
+    """Alpha composite float/uint8 RGB arrays, alpha in [0,1]."""
+    a = np.asarray(alpha, np.float32)
+    if a.ndim == 2:
+        a = a[..., None]
+    return dst.astype(np.float32) * (1.0 - a) + src.astype(np.float32) * a
+
+
+def line_layer(shape, items):
+    """Return antialiased RGB/alpha planes for (ellipse/polyline) drawing."""
+    rgb = np.zeros(shape, np.uint8)
+    aa = np.zeros(shape[:2], np.uint8)
+    for kind, pts, color, opacity, width in items:
+        if kind == "ellipse":
+            center, axes, angle, start, end = pts
+            cv2.ellipse(rgb, center, axes, angle, start, end, color, width, cv2.LINE_AA)
+            cv2.ellipse(aa, center, axes, angle, start, end, opacity, width, cv2.LINE_AA)
+        elif kind == "line":
+            p0, p1 = pts
+            cv2.line(rgb, p0, p1, color, width, cv2.LINE_AA)
+            cv2.line(aa, p0, p1, opacity, width, cv2.LINE_AA)
+    return rgb, aa.astype(np.float32) / 255.0
+
+
+def build_assets(first):
+    # The supplied scene's ceramic domino is the bright connected object held
+    # vertically at center-right.  A luminance component is preferable to a
+    # hard-coded rectangle because it retains the actual photographed contour.
+    lum = cv2.cvtColor(first, cv2.COLOR_RGB2GRAY)
+    seed = ((lum > 137).astype(np.uint8))
+    n, labels, stats, _ = cv2.connectedComponentsWithStats(seed, 8)
+    target = int(labels[72, 335])
+    if target == 0:
+        # Defensive fallback: choose the bright component near the expected ROI.
+        choices = []
+        for i in range(1, n):
+            x, y, w, h, area = stats[i]
+            if x < 340 < x + w and y < 80 < y + h:
+                choices.append((area, i))
+        target = max(choices)[1]
+    component = (labels == target).astype(np.uint8)
+
+    # Fill the exterior silhouette, including any dark markings and narrow
+    # strips hidden in the luminance threshold, then retain a photographic edge.
+    contours, _ = cv2.findContours(component, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
+    solid = np.zeros((H, W), np.uint8)
+    cv2.drawContours(solid, contours, -1, 255, cv2.FILLED)
+    solid = cv2.morphologyEx(solid, cv2.MORPH_CLOSE, np.ones((5, 5), np.uint8))
+    solid = cv2.dilate(solid, np.ones((3, 3), np.uint8), iterations=1)
+    alpha = cv2.GaussianBlur(solid, (0, 0), 0.7).astype(np.float32) / 255.0
+
+    # Reconstruct what is behind the tile from immediate neighboring pixels.
+    # TELEA follows the surrounding blue-black grabber on its sides and the
+    # warm wood behind its upper portion, avoiding synthetic flat fills.
+    erase = cv2.dilate(solid, np.ones((7, 7), np.uint8), iterations=1)
+    clean = cv2.inpaint(first, erase, 7, cv2.INPAINT_TELEA)
+    # A second, gentle texture pass suppresses the characteristic inpaint seam.
+    blurred = cv2.GaussianBlur(clean, (0, 0), 1.0)
+    edge_band = cv2.GaussianBlur(erase, (0, 0), 3).astype(np.float32) / 255.0
+    edge_band *= (1.0 - solid.astype(np.float32) / 255.0)
+    clean = np.clip(over(clean, blurred, edge_band * 0.24), 0, 255).astype(np.uint8)
+    return clean, first.copy(), alpha
+
+
+def transformed_domino(sprite, alpha, dy, angle):
+    # Rotate about the ceramic object's visual center, then translate down.
+    center = (337.0, 83.0)
+    M = cv2.getRotationMatrix2D(center, angle, 1.0)
+    M[1, 2] += dy
+    src = cv2.warpAffine(sprite, M, (W, H), flags=cv2.INTER_CUBIC,
+                         borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+    a = cv2.warpAffine(alpha, M, (W, H), flags=cv2.INTER_CUBIC,
+                       borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+    return src, np.clip(a, 0.0, 1.0)
+
+
+def add_water(frame, f, center_x=339, surface_y=150):
+    """Dark-liquid splash and damped ripples, all locked to the mug opening."""
+    out = frame.astype(np.float32)
+    contact = 36
+    age = f - contact
+
+    # Contact depression/crown: very short-lived, small compared with the mug.
+    if 0 <= age <= 11:
+        strength = max(0.0, 1.0 - age / 12.0)
+        lay = np.zeros((H, W, 3), np.uint8)
+        al = np.zeros((H, W), np.uint8)
+        # A dark crown lip plus muted specular rim.
+        rx = 12 + age * 2
+        ry = max(2, 6 - age // 3)
+        cv2.ellipse(lay, (center_x, surface_y), (rx, ry), 0, 180, 360,
+                    (27, 24, 25), 2, cv2.LINE_AA)
+        cv2.ellipse(al, (center_x, surface_y), (rx, ry), 0, 180, 360,
+                    int(190 * strength), 2, cv2.LINE_AA)
+        # Brief crown fingers rising from the impact cavity.
+        for j, (dx, ht) in enumerate([(-12, 7), (-7, 11), (8, 10), (14, 6)]):
+            shrink = age * 0.55
+            hnow = max(0, int(ht - shrink))
+            if hnow:
+                p0 = (center_x + dx, surface_y + 1)
+                p1 = (center_x + int(dx * .82), surface_y - hnow)
+                cv2.line(lay, p0, p1, (36, 30, 28), 2, cv2.LINE_AA)
+                cv2.line(al, p0, p1, int(180 * strength), 2, cv2.LINE_AA)
+        out = over(out, lay, al.astype(np.float32) / 255.0)
+
+    # Deterministic ballistic droplets. They rise, decelerate, and return to the
+    # surface rather than flickering randomly from frame to frame.
+    if 0 <= age <= 19:
+        rng = np.random.default_rng(4602)
+        drops = []
+        for _ in range(17):
+            vx = rng.uniform(-2.15, 2.15)
+            vy = rng.uniform(-4.25, -1.75)
+            delay = int(rng.integers(0, 3))
+            radius = int(rng.choice([1, 1, 1, 2]))
+            drops.append((vx, vy, delay, radius))
+        lay = np.zeros((H, W, 3), np.uint8)
+        al = np.zeros((H, W), np.uint8)
+        for vx, vy, delay, radius in drops:
+            t = age - delay
+            if t < 0:
+                continue
+            x = center_x + vx * t
+            y = surface_y - 1 + vy * t + 0.255 * t * t
+            if y > surface_y + 3 or x < 276 or x > 407:
+                continue
+            fade = np.clip(1.0 - max(0.0, t - 12) / 7.0, 0, 1)
+            cv2.circle(lay, (int(round(x)), int(round(y))), radius,
+                       (39, 32, 29), -1, cv2.LINE_AA)
+            cv2.circle(al, (int(round(x)), int(round(y))), radius,
+                       int(215 * fade), -1, cv2.LINE_AA)
+            if radius == 2:
+                cv2.circle(lay, (int(round(x - .7)), int(round(y - .7))), 1,
+                           (119, 109, 101), -1, cv2.LINE_AA)
+                cv2.circle(al, (int(round(x - .7)), int(round(y - .7))), 1,
+                           int(115 * fade), -1, cv2.LINE_AA)
+        out = over(out, lay, al.astype(np.float32) / 255.0)
+
+    # Multiple expanding, highly flattened rings imply a horizontal liquid
+    # surface. Later rings are broad and faint, with physically plausible decay.
+    if age >= 1:
+        items = []
+        for delay, gain in [(0, 1.0), (7, .72), (15, .48)]:
+            a = age - delay
+            if a < 0:
+                continue
+            fade = np.exp(-a / 25.0) * gain
+            if fade < .035:
+                continue
+            rx = int(10 + 1.45 * a)
+            ry = max(2, int(2 + .12 * a))
+            rx = min(rx, 70)
+            col = (105, 96, 91) if delay else (128, 117, 109)
+            items.append(("ellipse", ((center_x, surface_y + 1), (rx, ry), 0, 188, 352),
+                          col, int(132 * fade), 1))
+            items.append(("ellipse", ((center_x, surface_y + 1), (rx, ry), 0, 8, 169),
+                          (39, 34, 34), int(90 * fade), 1))
+        if items:
+            rgb, a = line_layer((H, W, 3), items)
+            out = over(out, rgb, a)
+    return np.clip(out, 0, 255).astype(np.uint8)
+
+
+def generate():
+    first = np.asarray(Image.open(CASE_DIR / "first_frame.png").convert("RGB"), dtype=np.uint8)
+    assert first.shape == (H, W, 3)
+    clean, sprite, alpha = build_assets(first)
+    yy, xx = np.mgrid[0:H, 0:W]
+    frames = []
+
+    for f in range(N):
+        # A short observed hold makes the initial grasp readable. Frame zero and
+        # all hold frames use the unmodified source pixels.
+        if f <= 23:
+            frame = first.copy()
+            frames.append(frame)
+            continue
+
+        # Release at 1.0 s.  Constant acceleration gives the tile increasing
+        # displacement rather than the common synthetic constant-speed slide.
+        t = (f - 24) / 29.0
+        tcap = min(max(t, 0.0), 1.0)
+        dy = 124.0 * tcap * tcap
+        # A tiny torque from the off-center grip develops during the fall.
+        angle = -7.2 * (tcap ** 1.55)
+        obj, a = transformed_domino(sprite, alpha, dy, angle)
+
+        # Dark liquid is optically opaque. Its slightly curved surface boundary
+        # lets the lower ceramic edge disappear progressively as it penetrates.
+        surface = 148.5 + 0.0016 * (xx - 339.0) ** 2
+        depth = yy - surface
+        underwater = depth > 0
+        transmission = np.where(underwater, 0.34 * np.exp(-depth / 10.5), 1.0)
+        a *= transmission.astype(np.float32)
+
+        frame = over(clean, obj, a)
+        # During the first two post-release frames blend from the exact source,
+        # preventing a one-frame edge jump caused by inpainting reconstruction.
+        if f < 27:
+            blend = (f - 24) / 3.0
+            frame = first.astype(np.float32) * (1.0 - blend) + frame * blend
+        frame = np.clip(frame, 0, 255).astype(np.uint8)
+        frame = add_water(frame, f)
+        frames.append(frame)
+
+    frames = np.stack(frames).astype(np.uint8)
+    frames[0] = first  # generation contract: byte-identical RGB first frame
+    assert frames.shape == (N, H, W, 3)
+    assert np.array_equal(frames[0], first)
+    return frames
+
+
+if __name__ == "__main__":
+    result = generate()
+    artifact_io.save(result, CASE_DIR)
+    print(f"saved {result.shape[0]} RGB frames; first frame exact: "
+          f"{np.array_equal(result[0], np.asarray(Image.open(CASE_DIR / 'first_frame.png').convert('RGB')))}")

def over(dst, src, alpha):
    \"\"\"Alpha composite float/uint8 RGB arrays, alpha in [0,1].\"\"\"
    a = np.asarray(alpha, np.float32)
    if a.ndim == 2:
        a = a[..., None]
    return dst.astype(np.float32) * (1.0 - a) + src.astype(np.float32) * a


def line_layer(shape, items):
    \"\"\"Return antialiased RGB/alpha planes for (ellipse/polyline) drawing.\"\"\"
    rgb = np.zeros(shape, np.uint8)
    aa = np.zeros(shape[:2], np.uint8)
    for kind, pts, color, opacity, width in items:
        if kind == \"ellipse\":
            center, axes, angle, start, end = pts
            cv2.ellipse(rgb, center, axes, angle, start, end, color, width, cv2.LINE_AA)
            cv2.ellipse(aa, center, axes, angle, start, end, opacity, width, cv2.LINE_AA)
        elif kind == \"line\":
            p0, p1 = pts
            cv2.line(rgb, p0, p1, color, width, cv2.LINE_AA)
            cv2.line(aa, p0, p1, opacity, width, cv2.LINE_AA)
    return rgb, aa.astype(np.float32) / 255.0


def build_assets(first):
    # The supplied scene's ceramic domino is the bright connected object held
    # vertically at center-right.  A luminance component is preferable to a
    # hard-coded rectangle because it retains the actual photographed contour.
    lum = cv2.cvtColor(first, cv2.COLOR_RGB2GRAY)
    seed = ((lum > 137).astype(np.uint8))
    n, labels, stats, _ = cv2.connectedComponentsWithStats(seed, 8)
    target = int(labels[72, 335])
    if target == 0:
        # Defensive fallback: choose the bright component near the expected ROI.
        choices = []
        for i in range(1, n):
            x, y, w, h, area = stats[i]
            if x < 340 < x + w and y < 80 < y + h:
                choices.append((area, i))
        target = max(choices)[1]
    component = (labels == target).astype(np.uint8)

    # Fill the exterior silhouette, including any dark markings and narrow
    # strips hidden in the luminance threshold, then retain a photographic edge.
    contours, _ = cv2.findContours(component, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
    solid = np.zeros((H, W), np.uint8)
    cv2.drawContours(solid, contours, -1, 255, cv2.FILLED)
    solid = cv2.morphologyEx(solid, cv2.MORPH_CLOSE, np.ones((5, 5), np.uint8))
    solid = cv2.dilate(solid, np.ones((3, 3), np.uint8), iterations=1)
    alpha = cv2.GaussianBlur(solid, (0, 0), 0.7).astype(np.float32) / 255.0

    # Reconstruct what is behind the tile from immediate neighboring pixels.
    # TELEA follows the surrounding blue-black grabber on its sides and the
    # warm wood behind its upper portion, avoiding synthetic flat fills.

    clean = cv2.inpaint(first, erase, 7, cv2.INPAINT_TELEA)

    blurred = cv2.GaussianBlur(clean, (0, 0), 1.0)

    center = (337.0, 83.0)

                       borderMode=cv2.BORDER_CONSTANT, borderValue=0)

        # all hold frames use the unmodified source pixels.

            frames.append(frame)
            continue

        # displacement rather than the common synthetic constant-speed slide.

        tcap = min(max(t, 0.0), 1.0)
        dy = 124.0 * tcap * tcap
        # A tiny torque from the off-center grip develops during the fall.
        angle = -7.2 * (tcap ** 1.55)

        # lets the lower ceramic edge disappear progressively as it penetrates.

        frame = add_water(frame, f)
        frames.append(frame)
