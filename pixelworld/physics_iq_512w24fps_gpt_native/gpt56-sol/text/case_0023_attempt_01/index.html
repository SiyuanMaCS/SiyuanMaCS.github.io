+"""Generate the requested locked-off tabletop rolling-ball demonstration.
+
+The source RGB frame remains the camera plate.  A small procedural 3-D sphere is
+rasterized numerically: its material has tennis-felt microvariation and a curved
+seam fixed in object coordinates, so both visibly rotate as the ball translates.
+No external models, assets, masks, or renderers are used.
+"""
+from pathlib import Path
+import math
+import numpy as np
+from PIL import Image
+import cv2
+import artifact_io
+
+W, H, N, FPS = 512, 288, 120, 24
+HERE = Path(__file__).resolve().parent
+
+
+def smoothstep(a, b, x):
+    q = np.clip((x - a) / (b - a), 0.0, 1.0)
+    return q * q * (3.0 - 2.0 * q)
+
+
+def motion(frame_number):
+    """Position on the visible table plane, in source-frame pixels.
+
+    The tiny cubic term models the brief release from the pipe while retaining
+    nearly uniform real-time velocity over the tabletop.
+    """
+    t = frame_number / (N - 1)
+    travel = t + 0.018 * t * (1.0 - t)  # very gentle initial release
+    x = -17.0 + 502.0 * travel
+    # The tabletop recedes gently to the left in the photograph.
+    contact_y = 168.0 + 10.0 * t
+    radius = 12.7 + 1.1 * t
+    return x, contact_y, radius
+
+
+def add_shadow(frame, cx, contact_y, radius, visibility=1.0):
+    """Soft diffuse/contact shadow directly on the wooden plane."""
+    yy, xx = np.mgrid[0:H, 0:W].astype(np.float32)
+    # A broad penumbra and a small dense contact core.
+    sx, sy = radius * 1.18, max(1.7, radius * 0.25)
+    broad = np.exp(-0.5 * (((xx - (cx + 2.2)) / sx) ** 2 +
+                           ((yy - (contact_y + 1.0)) / sy) ** 2))
+    core = np.exp(-0.5 * (((xx - (cx + 0.7)) / (radius * .42)) ** 2 +
+                          ((yy - (contact_y + .15)) / max(.72, radius * .075)) ** 2))
+    opacity = np.clip((0.24 * broad + 0.19 * core) * visibility, 0, .38)[..., None]
+    # Shadow is slightly warm/neutral rather than pure black.
+    shadow_tint = np.array([30.0, 27.0, 24.0], dtype=np.float32)
+    arr = frame.astype(np.float32)
+    arr[:] = arr * (1.0 - opacity) + shadow_tint * opacity
+
+
+def sphere_sample(cx, cy, radius, roll_angle, shutter_dx=0.0, scale=3):
+    """Return a supersampled RGBA crop containing the procedural tennis ball."""
+    pad = int(math.ceil(radius + 3.0))
+    x0 = int(math.floor(cx - pad))
+    y0 = int(math.floor(cy - pad))
+    x1 = int(math.ceil(cx + pad + 1))
+    y1 = int(math.ceil(cy + pad + 1))
+    if x1 <= 0 or x0 >= W or y1 <= 0 or y0 >= H:
+        return None
+
+    # Supersample at pixel centers in global frame coordinates.
+    xs = x0 + (np.arange((x1 - x0) * scale, dtype=np.float32) + .5) / scale
+    ys = y0 + (np.arange((y1 - y0) * scale, dtype=np.float32) + .5) / scale
+    gx, gy = np.meshgrid(xs, ys)
+    u = (gx - (cx + shutter_dx)) / radius
+    v_screen = (gy - cy) / radius
+    rr = u * u + v_screen * v_screen
+    inside = rr < 1.12
+    z = np.sqrt(np.clip(1.0 - rr, 0.0, 1.0))
+
+    # Camera-space outward normal; object y points upward.
+    nx = u
+    ny = -v_screen
+    nz = z
+
+    # Inverse y-axis rotation maps each visible point to ball/material space.
+    ca, sa = math.cos(roll_angle), math.sin(roll_angle)
+    ox = ca * nx - sa * nz
+    oy = ny
+    oz = sa * nx + ca * nz
+
+    # Tennis seam: a continuous double-curved band on the sphere.  It is fixed
+    # in object coordinates, and therefore convincingly tumbles with the ball.
+    longitude = np.arctan2(oz, ox)
+    latitude = np.arcsin(np.clip(oy, -1.0, 1.0))
+    seam_curve = 0.42 * np.sin(2.0 * longitude + 0.28)
+    d_seam = np.abs(latitude - seam_curve)
+    seam = 1.0 - smoothstep(0.035, 0.072, d_seam)
+    seam_edge = np.exp(-((d_seam - 0.066) / 0.018) ** 2)
+
+    # Soft directional studio/window light matching the source plate.
+    light = np.array([-0.48, 0.62, 0.62], np.float32)
+    light /= np.linalg.norm(light)
+    ndotl = np.maximum(0.0, nx * light[0] + ny * light[1] + nz * light[2])
+    halfv = np.array([-0.27, 0.38, 0.885], np.float32)
+    halfv /= np.linalg.norm(halfv)
+    spec = np.maximum(0.0, nx * halfv[0] + ny * halfv[1] + nz * halfv[2]) ** 24
+    shade = 0.43 + 0.62 * ndotl + 0.13 * spec
+    # Slight lower-edge bounce from the warm table.
+    shade += 0.045 * np.clip(-ny, 0, 1) * np.clip(z, 0, 1)
+
+    # Analytic, object-locked felt mottling.  Several incommensurate waves avoid
+    # a synthetic checker while keeping every regeneration deterministic.
+    felt = (2.6 * np.sin(37.0 * ox + 19.0 * oy + 13.0 * oz) +
+            1.8 * np.sin(53.0 * oy - 29.0 * oz + 7.0 * ox) +
+            1.2 * np.sin(91.0 * ox + 47.0 * oz))
+    base = np.array([143.0, 148.0, 145.0], np.float32)
+    rgb = base[None, None, :] * shade[..., None] + felt[..., None]
+
+    # The recessed, pale seam has a narrow darker boundary and still receives
+    # the scene lighting, keeping it integrated rather than painted on top.
+    seam_rgb = np.array([190.0, 193.0, 188.0], np.float32) * (0.74 + 0.36 * ndotl[..., None])
+    rgb = rgb * (1.0 - 0.82 * seam[..., None]) + seam_rgb * (0.82 * seam[..., None])
+    rgb -= (5.0 * seam_edge)[..., None]
+
+    # Fuzzy silhouette, perturbed subtly by object-locked felt fibers.
+    edge_noise = (0.007 * np.sin(71 * ox + 43 * oy) +
+                  0.006 * np.sin(97 * oy - 31 * oz))
+    radial = np.sqrt(np.maximum(rr, 0))
+    alpha = 1.0 - smoothstep(0.965 + edge_noise, 1.035 + edge_noise, radial)
+    alpha *= inside
+
+    rgba = np.concatenate([np.clip(rgb, 0, 255), (255.0 * alpha)[..., None]], axis=2)
+    # Area-average supersampling back to the camera raster.
+    rgba = rgba.reshape(y1-y0, scale, x1-x0, scale, 4).mean(axis=(1, 3))
+    return x0, y0, rgba
+
+
+def composite(frame, layer):
+    if layer is None:
+        return
+    x0, y0, rgba = layer
+    h, w = rgba.shape[:2]
+    xa, ya = max(0, x0), max(0, y0)
+    xb, yb = min(W, x0 + w), min(H, y0 + h)
+    if xa >= xb or ya >= yb:
+        return
+    src = rgba[ya-y0:yb-y0, xa-x0:xb-x0]
+    a = src[..., 3:4] / 255.0
+    dst = frame[ya:yb, xa:xb].astype(np.float32)
+    frame[ya:yb, xa:xb] = np.clip(dst * (1-a) + src[..., :3] * a, 0, 255).astype(np.uint8)
+
+
+def generate():
+    plate = np.asarray(Image.open(HERE / "first_frame.png").convert("RGB"), dtype=np.uint8)
+    assert plate.shape == (H, W, 3)
+    frames = np.repeat(plate[None, ...], N, axis=0)
+
+    # Frame zero is contractually exact and depicts the ball still hidden in the
+    # left pipe.  At subsequent frames it enters continuously from beyond frame.
+    start_x, _, start_r = motion(0)
+    for i in range(1, N):
+        x, contact_y, radius = motion(i)
+        cy = contact_y - radius
+        # Occlusion/visibility naturally ramps as the sphere clears the left edge.
+        visibility = float(np.clip((x + radius + 2.0) / (2.0 * radius), 0, 1))
+        add_shadow(frames[i], x, contact_y, radius, visibility)
+
+        # Rolling without slipping. Positive translation gives the appropriate
+        # y-axis rotation; a small phase places a recognizable seam on entry.
+        angle = -((x - start_x) / ((radius + start_r) * 0.5)) + 0.65
+        # Three short exposure samples create subpixel camera motion blur without
+        # ghosting. The seam remains object-consistent at the middle exposure.
+        speed = motion(min(N-1, i+1))[0] - motion(max(0, i-1))[0]
+        for frac, weight in [(-0.10, 0.27), (0.0, 0.46), (0.10, 0.27)]:
+            layer = sphere_sample(x, cy, radius, angle - frac * speed/radius,
+                                  shutter_dx=frac * speed, scale=3)
+            if layer is not None:
+                # Weight exposure samples by scaling source alpha.
+                layer[2][..., 3] *= weight
+                composite(frames[i], layer)
+
+    # Explicitly enforce the identity invariant even if future edits touch frame 0.
+    frames[0] = plate
+    return frames
+
+
+if __name__ == "__main__":
+    result = generate()
+    artifact_io.save(result, HERE)
+    print(f"saved {result.shape[0]} RGB frames at {W}x{H}, {FPS} fps")
+    print("frame00 identical:", np.array_equal(result[0], np.asarray(Image.open(HERE/'first_frame.png'))))

\"\"\"Generate the requested locked-off tabletop rolling-ball demonstration.

The source RGB frame remains the camera plate.  A small procedural 3-D sphere is
rasterized numerically: its material has tennis-felt microvariation and a curved
seam fixed in object coordinates, so both visibly rotate as the ball translates.
No external models, assets, masks, or renderers are used.

from pathlib import Path
import math
import numpy as np
from PIL import Image
import cv2
import artifact_io

W, H, N, FPS = 512, 288, 120, 24
HERE = Path(__file__).resolve().parent


def smoothstep(a, b, x):
    q = np.clip((x - a) / (b - a), 0.0, 1.0)
    return q * q * (3.0 - 2.0 * q)


def motion(frame_number):
    \"\"\"Position on the visible table plane, in source-frame pixels.

    The tiny cubic term models the brief release from the pipe while retaining
    nearly uniform real-time velocity over the tabletop.
    \"\"\"
    t = frame_number / (N - 1)
    travel = t + 0.018 * t * (1.0 - t)  # very gentle initial release
    x = -17.0 + 502.0 * travel
    # The tabletop recedes gently to the left in the photograph.
    contact_y = 168.0 + 10.0 * t
    radius = 12.7 + 1.1 * t
    return x, contact_y, radius


def add_shadow(frame, cx, contact_y, radius, visibility=1.0):
    \"\"\"Soft diffuse/contact shadow directly on the wooden plane.\"\"\"
    yy, xx = np.mgrid[0:H, 0:W].astype(np.float32)
    # A broad penumbra and a small dense contact core.
    sx, sy = radius * 1.18, max(1.7, radius * 0.25)
    broad = np.exp(-0.5 * (((xx - (cx + 2.2)) / sx) ** 2 +
                           ((yy - (contact_y + 1.0)) / sy) ** 2))
    core = np.exp(-0.5 * (((xx - (cx + 0.7)) / (radius * .42)) ** 2 +
                          ((yy - (contact_y + .15)) / max(.72, radius * .075)) ** 2))
    opacity = np.clip((0.24 * broad + 0.19 * core) * visibility, 0, .38)[..., None]
    # Shadow is slightly warm/neutral rather than pure black.
    shadow_tint = np.array([30.0, 27.0, 24.0], dtype=np.float32)
    arr = frame.astype(np.float32)
    arr[:] = arr * (1.0 - opacity) + shadow_tint * opacity


def sphere_sample(cx, cy, radius, roll_angle, shutter_dx=0.0, scale=3):
    \"\"\"Return a supersampled RGBA crop containing the procedural tennis ball.\"\"\"
    pad = int(math.ceil(radius + 3.0))
    x0 = int(math.floor(cx - pad))
    y0 = int(math.floor(cy - pad))
    x1 = int(math.ceil(cx + pad + 1))
    y1 = int(math.ceil(cy + pad + 1))
    if x1 <= 0 or x0 >= W or y1 <= 0 or y0 >= H:
        return None

    # Supersample at pixel centers in global frame coordinates.
    xs = x0 + (np.arange((x1 - x0) * scale, dtype=np.float32) + .5) / scale
    ys = y0 + (np.arange((y1 - y0) * scale, dtype=np.float32) + .5) / scale
    gx, gy = np.meshgrid(xs, ys)
    u = (gx - (cx + shutter_dx)) / radius
    v_screen = (gy - cy) / radius
    rr = u * u + v_screen * v_screen
    inside = rr < 1.12
    z = np.sqrt(np.clip(1.0 - rr, 0.0, 1.0))

    # Camera-space outward normal; object y points upward.
    nx = u
    ny = -v_screen
    nz = z

    # Inverse y-axis rotation maps each visible point to ball/material space.
    ca, sa = math.cos(roll_angle), math.sin(roll_angle)
    ox = ca * nx - sa * nz
    oy = ny
    oz = sa * nx + ca * nz

    # Tennis seam: a continuous double-curved band on the sphere.  It is fixed
    # in object coordinates, and therefore convincingly tumbles with the ball.
    longitude = np.arctan2(oz, ox)
    latitude = np.arcsin(np.clip(oy, -1.0, 1.0))
    seam_curve = 0.42 * np.sin(2.0 * longitude + 0.28)
    d_seam = np.abs(latitude - seam_curve)
    seam = 1.0 - smoothstep(0.035, 0.072, d_seam)
    seam_edge = np.exp(-((d_seam - 0.066) / 0.018) ** 2)

    # Soft directional studio/window light matching the source plate.
    light = np.array([-0.48, 0.62, 0.62], np.float32)
    light /= np.linalg.norm(light)
    ndotl = np.maximum(0.0, nx * light[0] + ny * light[1] + nz * light[2])
    halfv = np.array([-0.27, 0.38, 0.885], np.float32)
    halfv /= np.linalg.norm(halfv)
    spec = np.maximum(0.0, nx * halfv[0] + ny * halfv[1] + nz * halfv[2]) ** 24
    shade = 0.43 + 0.62 * ndotl + 0.13 * spec
    # Slight lower-edge bounce from the warm table.
    shade += 0.045 * np.clip(-ny, 0, 1) * np.clip(z, 0, 1)

    # Analytic, object-locked felt mottling.  Several incommensurate waves avoid
    # a synthetic checker while keeping every regeneration deterministic.
    felt = (2.6 * np.sin(37.0 * ox + 19.0 * oy + 13.0 * oz) +
            1.8 * np.sin(53.0 * oy - 29.0 * oz + 7.0 * ox) +
            1.2 * np.sin(91.0 * ox + 47.0 * oz))
    base = np.array([143.0, 148.0, 145.0], np.float32)
    rgb = base[None, None, :] * shade[..., None] + felt[..., None]

    # The recessed, pale seam has a narrow darker boundary and still receives
    # the scene lighting, keeping it integrated rather than painted on top.
    seam_rgb = np.array([190.0, 193.0, 188.0], np.float32) * (0.74 + 0.36 * ndotl[..., None])
    rgb = rgb * (1.0 - 0.82 * seam[..., None]) + seam_rgb * (0.82 * seam[..., None])
    rgb -= (5.0 * seam_edge)[..., None]

    # Fuzzy silhouette, perturbed subtly by object-locked felt fibers.
    edge_noise = (0.007 * np.sin(71 * ox + 43 * oy) +
                  0.006 * np.sin(97 * oy - 31 * oz))
    radial = np.sqrt(np.maximum(rr, 0))
    alpha = 1.0 - smoothstep(0.965 + edge_noise, 1.035 + edge_noise, radial)
    alpha *= inside

    rgba = np.concatenate([np.clip(rgb, 0, 255), (255.0 * alpha)[..., None]], axis=2)
    # Area-average supersampling back to the camera raster.
    rgba = rgba.reshape(y1-y0, scale, x1-x0, scale, 4).mean(axis=(1, 3))
    return x0, y0, rgba


def composite(frame, layer):
    if layer is None:
        return
    x0, y0, rgba = layer
    h, w = rgba.shape[:2]
    xa, ya = max(0, x0), max(0, y0)
    xb, yb = min(W, x0 + w), min(H, y0 + h)
    if xa >= xb or ya >= yb:
        return
    src = rgba[ya-y0:yb-y0, xa-x0:xb-x0]
    a = src[..., 3:4] / 255.0
    dst = frame[ya:yb, xa:xb].astype(np.float32)
    frame[ya:yb, xa:xb] = np.clip(dst * (1-a) + src[..., :3] * a, 0, 255).astype(np.uint8)


def generate():
    plate = np.asarray(Image.open(HERE / \"first_frame.png\").convert(\"RGB\"), dtype=np.uint8)
    assert plate.shape == (H, W, 3)
    frames = np.repeat(plate[None, ...], N, axis=0)

    # Frame zero is contractually exact and depicts the ball still hidden in the
    # left pipe.  At subsequent frames it enters continuously from beyond frame.
    start_x, _, start_r = motion(0)
    for i in range(1, N):
        x, contact_y, radius = motion(i)
        cy = contact_y - radius
        # Occlusion/visibility naturally ramps as the sphere clears the left edge.
        visibility = float(np.clip((x + radius + 2.0) / (2.0 * radius), 0, 1))
        add_shadow(frames[i], x, contact_y, radius, visibility)

        # Rolling without slipping. Positive translation gives the appropriate
        # y-axis rotation; a small phase places a recognizable seam on entry.
        angle = -((x - start_x) / ((radius + start_r) * 0.5)) + 0.65
        # Three short exposure samples create subpixel camera motion blur without
        # ghosting. The seam remains object-consistent at the middle exposure.
        speed = motion(min(N-1, i+1))[0] - motion(max(0, i-1))[0]
        for frac, weight in [(-0.10, 0.27), (0.0, 0.46), (0.10, 0.27)]:
            layer = sphere_sample(x, cy, radius, angle - frac * speed/radius,
                                  shutter_dx=frac * speed, scale=3)
            if layer is not None:
                # Weight exposure samples by scaling source alpha.
                layer[2][..., 3] *= weight
                composite(frames[i], layer)

    # Explicitly enforce the identity invariant even if future edits touch frame 0.
    frames[0] = plate
    return frames
