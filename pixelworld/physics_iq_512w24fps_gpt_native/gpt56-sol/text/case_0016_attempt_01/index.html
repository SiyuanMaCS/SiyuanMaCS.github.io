+"""Generate the requested locked-off kinetic-sand drop demonstration.
+
+Only the supplied first frame is used.  The photographed ball is isolated with a
+soft geometric matte, the small area behind it is reconstructed from the same
+backdrop, and physically timed translation / contact deformation is applied.
+"""

+import math
+import numpy as np
+from PIL import Image
+import artifact_io

+CASE_DIR = Path(__file__).resolve().parent
+W, H, N, FPS = 512, 288, 120, 24
+CX, CY, BALL_R = 272.5, 78.5, 33.7

+def smoothstep(edge0, edge1, x):
+    t = np.clip((x - edge0) / (edge1 - edge0), 0.0, 1.0)
+    return t * t * (3.0 - 2.0 * t)
+

+def backdrop_model(src):
+    """Fit the smooth photographed cyclorama around the held ball.
+
+    A low-order surface retains the real vignetting. Fine residual texture is
+    borrowed from an unobstructed nearby strip of the same frame.
+    """
+    yy, xx = np.mgrid[20:133, 170:371]
+    pix = src[20:133, 170:371].astype(np.float32)
+    # Neutral, moderately bright pixels are wall rather than blue apparatus.
+    spread = pix.max(2) - pix.min(2)
+    good = (spread < 28) & (pix.mean(2) > 105)
+    xn = (xx - 270.0) / 120.0
+    yn = (yy - 78.0) / 80.0
+    terms = np.stack([np.ones_like(xn), xn, yn, xn*xn, xn*yn,
+                      yn*yn, xn*xn*xn, yn*yn*yn], axis=-1)
+    A = terms[good]
+    coef = [np.linalg.lstsq(A, pix[..., c][good], rcond=None)[0]
+            for c in range(3)]
+
+    Y, X = np.mgrid[0:H, 0:W]
+    Xn = (X - 270.0) / 120.0
+    Yn = (Y - 78.0) / 80.0
+    T = np.stack([np.ones_like(Xn), Xn, Yn, Xn*Xn, Xn*Yn,
+                  Yn*Yn, Xn*Xn*Xn, Yn*Yn*Yn], axis=-1)
+    fit = np.stack([T @ c for c in coef], axis=-1)
+
+    # Preserve subtle sensor/backdrop texture by borrowing residuals at x+92.
+    sx = np.clip(X + 92, 0, W-1)
+    borrowed = src[Y, sx].astype(np.float32)
+    borrowed_fit = fit[Y, sx]
+    residual = np.clip(borrowed - borrowed_fit, -4.0, 4.0)
+    return np.clip(fit + 0.55 * residual, 0, 255)
+
+
+def ball_matte():
+    Y, X = np.mgrid[0:H, 0:W]
+    radius = np.hypot(X - CX, Y - CY)
+    # Fully opaque through the photographed sphere, 2px anti-aliased rim.
+    return 1.0 - smoothstep(BALL_R - 1.7, BALL_R + 0.9, radius)
+

+def rgba_ball(src, alpha_full):
+    pad = 3
+    x0 = int(math.floor(CX - BALL_R - pad))
+    y0 = int(math.floor(CY - BALL_R - pad))
+    x1 = int(math.ceil(CX + BALL_R + pad + 1))
+    y1 = int(math.ceil(CY + BALL_R + pad + 1))
+    rgba = np.zeros((y1-y0, x1-x0, 4), np.uint8)
+    rgba[..., :3] = src[y0:y1, x0:x1]
+    rgba[..., 3] = np.round(alpha_full[y0:y1, x0:x1] * 255).astype(np.uint8)
+    return Image.fromarray(rgba, 'RGBA'), (x0, y0)
+

+def alpha_composite_np(base, rgba, x, y):
+    """Composite a small uint8 RGBA image, clipped to the frame."""
+    out = base.copy()
+    h, w = rgba.shape[:2]
+    xa, ya = max(0, x), max(0, y)
+    xb, yb = min(W, x+w), min(H, y+h)
+    if xa >= xb or ya >= yb:
+        return out
+    fg = rgba[ya-y:yb-y, xa-x:xb-x].astype(np.float32)
+    a = fg[..., 3:4] / 255.0
+    bg = out[ya:yb, xa:xb].astype(np.float32)
+    out[ya:yb, xa:xb] = np.clip(fg[..., :3]*a + bg*(1-a), 0, 255).astype(np.uint8)
+    return out

+def rotated_sprite(sprite, angle):

+        return np.asarray(sprite), (0, 0)
+    old_w, old_h = sprite.size

+    # Offset needed to keep rotation centered at the original ball center.
+    return np.asarray(rot), ((old_w-rot.size[0])//2, (old_h-rot.size[1])//2)

+def trajectory(f):
+    """Center position and rotation; frame 24 is the physical release."""
+    if f <= 24:
+        return CY, 0.0
+    n = f - 24.0
+    free_y = CY + 0.5 * 0.785 * n*n
+    contact_y = 135.0
+    if free_y <= contact_y:
+        return free_y, -0.62*n
+    # Impact begins at n ~= 12.05. Kinetic sand gives a short, heavily damped
+    # penetration rather than a conventional elastic bounce.
+    q = n - math.sqrt(2.0*(contact_y-CY)/0.785)
+    keys_t = np.array([0, 1, 2, 3, 4, 5, 7, 9, 12, 16, 22], np.float32)
+    keys_y = np.array([135, 144, 151, 155, 156, 155, 152.7, 151.3,
+                       152.2, 152.0, 152.0], np.float32)
+    y = float(np.interp(q, keys_t, keys_y))
+    angle = -7.5 - min(q, 14.0)*0.72
+    return y, angle

+def green_confidence(rgb):
+    a = rgb.astype(np.float32)
+    r, g, b = a[..., 0], a[..., 1], a[..., 2]
+    c1 = smoothstep(12, 34, g-r)
+    c2 = smoothstep(5, 22, g-b)
+    return c1*c2*smoothstep(35, 75, g)

+def add_contact_shadow(frame, ball_y):
+    # Soft shadow on green mound; it tightens and darkens as the ball descends.
+    if ball_y < 88:
+        return frame
+    Y, X = np.mgrid[0:H, 0:W]
+    distance = max(0.0, 136.0-ball_y)
+    sx = 16.0 + 0.29*distance
+    sy = 4.5 + 0.09*distance
+    e = ((X-CX)/sx)**2 + ((Y-174.0)/sy)**2
+    soft = np.exp(-1.7*e)
+    strength = 0.25 - min(distance, 48.0)*0.0027
+    green = green_confidence(frame)
+    a = (soft*strength*green)[..., None]
+    shade = np.array([0.52, 0.58, 0.54], np.float32)
+    return np.clip(frame.astype(np.float32)*(1-a+a*shade), 0, 255).astype(np.uint8)
+

+def impact_progress(f):
+    return float(smoothstep(36.0, 45.0, np.array(f, np.float32)))

+def add_sand_ridges(frame, src, p):
+    """Raise two restrained cohesive shoulders beside the penetrating ball."""
+    if p <= 0:
+        return frame
+    Y, X = np.mgrid[0:H, 0:W]
+    out = frame.astype(np.float32)
+    srcf = src.astype(np.float32)
+    # Each mound shoulder borrows real grains immediately below it.
+    for side in (-1, 1):
+        cx = CX + side*(29.0 + 4.5*p)
+        cy = 178.0 - 5.0*p
+        ex = (X-cx)/(13.0+2.0*p)
+        ey = (Y-cy)/(7.5+2.5*p)
+        shape = np.clip(1.0-(ex*ex+ey*ey), 0, 1)
+        shape = smoothstep(0.0, 0.55, shape) * p
+        # Sample a nearby piece of the photographed sand surface.
+        sample_x = np.clip(np.rint(X - side*7).astype(int), 0, W-1)
+        sample_y = np.clip(np.rint(Y + 16 - 4*p).astype(int), 0, H-1)
+        tex = srcf[sample_y, sample_x]
+        valid = green_confidence(tex.astype(np.uint8))
+        a = (shape*valid)[..., None]
+        out = tex*a + out*(1-a)
+    return np.clip(out, 0, 255).astype(np.uint8)
+

+def add_front_lip(frame, src, ball_y, p):
+    """Place real green sand texture over the bottom cap of the embedded ball."""
+    if p <= 0:
+        return frame
+    Y, X = np.mgrid[0:H, 0:W]
+    dx = X-CX
+    rr = dx*dx + (Y-ball_y)*(Y-ball_y)
+    inside_ball = 1.0-smoothstep((BALL_R-0.3)**2, (BALL_R+0.8)**2, rr)
+    waviness = 0.9*np.sin(dx*0.34) + 0.45*np.sin(dx*0.83+1.2)
+    lip_top = ball_y + BALL_R - (10.5*p) - 0.085*np.abs(dx) + waviness*p
+    top_alpha = smoothstep(lip_top-1.1, lip_top+1.3, Y)
+    horizontal = 1.0-smoothstep(25.5, 33.5, np.abs(dx))
+    a = inside_ball*top_alpha*horizontal*p
+
+    # Sample coherent real mound texture lower in the supplied image.
+    sx = np.clip(np.rint(X + 2.0*np.sin(Y*0.21)).astype(int), 0, W-1)
+    sy = np.clip(np.rint(190.0 + (Y-lip_top)*0.72 + 0.12*np.abs(dx)).astype(int), 0, H-1)
+    tex = src[sy, sx].astype(np.float32)
+    valid = green_confidence(tex.astype(np.uint8))
+    a = (a*valid)[..., None]
+    out = frame.astype(np.float32)*(1-a) + tex*a
+    return np.clip(out, 0, 255).astype(np.uint8)
+

+def add_grains(frame, f, src):
+    """A few cohesive grains move at impact; no dusty explosion."""
+    if not (36 <= f <= 56):
+        return frame
+    out = frame.copy()
+    # Deterministic launch positions/speeds; all are tiny (1-2 px).
+    grains = [
+        (-22,-2.8,-4.4,1), (-13,-1.6,-5.2,1), (-7,-0.8,-4.1,2),
+        (9,0.9,-5.0,1), (16,1.7,-4.5,2), (25,2.6,-3.8,1),
+        (-30,-2.3,-3.2,1), (31,2.1,-3.1,1), (3,0.35,-5.7,1),
+    ]
+    tau = f-36
+    for i, (xoff, vx, vy, rad) in enumerate(grains):
+        # Slightly staggered impulses.
+        t = tau - (i % 3)*0.45
+        if t < 0 or t > 14:
+            continue
+        x = int(round(CX+xoff+vx*t))
+        y = int(round(174+vy*t+0.39*t*t))
+        if 1 <= x < W-1 and 1 <= y < H-1:
+            col = src[min(231, 194+(i*7)%32), min(345, max(193, int(CX+xoff)))].astype(int)
+            # Ensure visible green sampled from the original mound palette.
+            if not (col[1] > col[0]+10):
+                col = np.array([31, 102+3*(i%4), 39])
+            y0,y1=max(0,y-rad),min(H,y+rad+1)
+            x0,x1=max(0,x-rad),min(W,x+rad+1)
+            gy,gx=np.mgrid[y0:y1,x0:x1]
+            mask=(gx-x)**2+(gy-y)**2 <= rad*rad
+            patch=out[y0:y1,x0:x1]
+            patch[mask]=col.astype(np.uint8)
+    return out
+
+
+def generate():
+    src = np.asarray(Image.open(CASE_DIR/'first_frame.png').convert('RGB'))
+    assert src.shape == (H, W, 3)
+    alpha = ball_matte()
+    model = backdrop_model(src)
+
+    # Plate keeps every photographed pixel except the spherical held object.
+    clean = np.clip(src.astype(np.float32)*(1-alpha[..., None]) +
+                    model*alpha[..., None], 0, 255).astype(np.uint8)
+    sprite, (orig_x, orig_y) = rgba_ball(src, alpha)
+
+    frames = np.empty((N, H, W, 3), np.uint8)
+    frames[0] = src  # contractual pixel identity
+    for f in range(1, N):
+        ball_y, angle = trajectory(f)
+        if f <= 24:
+            # Exact static hold prevents synthetic edge shimmer before release.
+            frames[f] = src
+            continue
+        base = add_contact_shadow(clean, ball_y)
+        p = impact_progress(f)
+        base = add_sand_ridges(base, src, p)
+
+        # Exposure integration gives modest photographic motion blur during the
+        # quick free-fall, while keeping contact and rest frames crisp.
+        prev_y, prev_angle = trajectory(f-1)
+        subs = 5 if (ball_y-prev_y) > 2.0 and ball_y < 138 else 1
+        accum = np.zeros((H, W, 3), np.float32)
+        for k in range(subs):
+            u = (k+0.5)/subs
+            sy = prev_y*(1-u)+ball_y*u
+            sa = prev_angle*(1-u)+angle*u
+            arr, off = rotated_sprite(sprite, sa)
+            x = orig_x + off[0]
+            y = int(round(orig_y + (sy-CY))) + off[1]
+            shot = alpha_composite_np(base, arr, x, y)
+            accum += shot.astype(np.float32)
+        frame = np.clip(accum/subs, 0, 255).astype(np.uint8)
+        frame = add_front_lip(frame, src, ball_y, p)
+        frame = add_grains(frame, f, src)
+        frames[f] = frame
+    return frames
+
+
+if __name__ == '__main__':
+    result = generate()
+    artifact_io.save(result, CASE_DIR)
+    print('saved', result.shape, result.dtype,
+          'frame0_exact=', bool(np.array_equal(result[0], np.asarray(Image.open(CASE_DIR/'first_frame.png').convert('RGB')))))

Only the supplied first frame is used.  The photographed ball is isolated with a
soft geometric matte, the small area behind it is reconstructed from the same

from pathlib import Path
import math
import numpy as np
from PIL import Image
import artifact_io

CASE_DIR = Path(__file__).resolve().parent
W, H, N, FPS = 512, 288, 120, 24
CX, CY, BALL_R = 272.5, 78.5, 33.7


def smoothstep(edge0, edge1, x):
    t = np.clip((x - edge0) / (edge1 - edge0), 0.0, 1.0)
    return t * t * (3.0 - 2.0 * t)


def backdrop_model(src):
    \"\"\"Fit the smooth photographed cyclorama around the held ball.

    A low-order surface retains the real vignetting. Fine residual texture is
    borrowed from an unobstructed nearby strip of the same frame.
    \"\"\"
    yy, xx = np.mgrid[20:133, 170:371]
    pix = src[20:133, 170:371].astype(np.float32)
    # Neutral, moderately bright pixels are wall rather than blue apparatus.
    spread = pix.max(2) - pix.min(2)
    good = (spread < 28) & (pix.mean(2) > 105)
    xn = (xx - 270.0) / 120.0
    yn = (yy - 78.0) / 80.0
    terms = np.stack([np.ones_like(xn), xn, yn, xn*xn, xn*yn,
                      yn*yn, xn*xn*xn, yn*yn*yn], axis=-1)
    A = terms[good]
    coef = [np.linalg.lstsq(A, pix[..., c][good], rcond=None)[0]
            for c in range(3)]

    Y, X = np.mgrid[0:H, 0:W]
    Xn = (X - 270.0) / 120.0
    Yn = (Y - 78.0) / 80.0
    T = np.stack([np.ones_like(Xn), Xn, Yn, Xn*Xn, Xn*Yn,
                  Yn*Yn, Xn*Xn*Xn, Yn*Yn*Yn], axis=-1)
    fit = np.stack([T @ c for c in coef], axis=-1)

    # Preserve subtle sensor/backdrop texture by borrowing residuals at x+92.
    sx = np.clip(X + 92, 0, W-1)
    borrowed = src[Y, sx].astype(np.float32)
    borrowed_fit = fit[Y, sx]
    residual = np.clip(borrowed - borrowed_fit, -4.0, 4.0)
    return np.clip(fit + 0.55 * residual, 0, 255)


def ball_matte():
    Y, X = np.mgrid[0:H, 0:W]
    radius = np.hypot(X - CX, Y - CY)
    # Fully opaque through the photographed sphere, 2px anti-aliased rim.
    return 1.0 - smoothstep(BALL_R - 1.7, BALL_R + 0.9, radius)


def rgba_ball(src, alpha_full):
    pad = 3
    x0 = int(math.floor(CX - BALL_R - pad))
    y0 = int(math.floor(CY - BALL_R - pad))
    x1 = int(math.ceil(CX + BALL_R + pad + 1))
    y1 = int(math.ceil(CY + BALL_R + pad + 1))
    rgba = np.zeros((y1-y0, x1-x0, 4), np.uint8)
    rgba[..., :3] = src[y0:y1, x0:x1]
    rgba[..., 3] = np.round(alpha_full[y0:y1, x0:x1] * 255).astype(np.uint8)
    return Image.fromarray(rgba, 'RGBA'), (x0, y0)


def alpha_composite_np(base, rgba, x, y):
    \"\"\"Composite a small uint8 RGBA image, clipped to the frame.\"\"\"
    out = base.copy()
    h, w = rgba.shape[:2]
    xa, ya = max(0, x), max(0, y)
    xb, yb = min(W, x+w), min(H, y+h)
    if xa >= xb or ya >= yb:
        return out
    fg = rgba[ya-y:yb-y, xa-x:xb-x].astype(np.float32)
    a = fg[..., 3:4] / 255.0
    bg = out[ya:yb, xa:xb].astype(np.float32)
    out[ya:yb, xa:xb] = np.clip(fg[..., :3]*a + bg*(1-a), 0, 255).astype(np.uint8)
    return out


def rotated_sprite(sprite, angle):
    if abs(angle) < 0.02:
        return np.asarray(sprite), (0, 0)
    old_w, old_h = sprite.size
    rot = sprite.rotate(angle, resample=Image.Resampling.BICUBIC, expand=True)
    # Offset needed to keep rotation centered at the original ball center.
    return np.asarray(rot), ((old_w-rot.size[0])//2, (old_h-rot.size[1])//2)


def trajectory(f):
    \"\"\"Center position and rotation; frame 24 is the physical release.\"\"\"
    if f <= 24:
        return CY, 0.0
    n = f - 24.0
    free_y = CY + 0.5 * 0.785 * n*n
    contact_y = 135.0
    if free_y <= contact_y:
        return free_y, -0.62*n
    # Impact begins at n ~= 12.05. Kinetic sand gives a short, heavily damped
    # penetration rather than a conventional elastic bounce.
    q = n - math.sqrt(2.0*(contact_y-CY)/0.785)
    keys_t = np.array([0, 1, 2, 3, 4, 5, 7, 9, 12, 16, 22], np.float32)
    keys_y = np.array([135, 144, 151, 155, 156, 155, 152.7, 151.3,
                       152.2, 152.0, 152.0], np.float32)
    y = float(np.interp(q, keys_t, keys_y))
    angle = -7.5 - min(q, 14.0)*0.72
    return y, angle


def green_confidence(rgb):
    a = rgb.astype(np.float32)
    r, g, b = a[..., 0], a[..., 1], a[..., 2]
    c1 = smoothstep(12, 34, g-r)
    c2 = smoothstep(5, 22, g-b)
    return c1*c2*smoothstep(35, 75, g)


def add_contact_shadow(frame, ball_y):
    # Soft shadow on green mound; it tightens and darkens as the ball descends.
    if ball_y < 88:
        return frame
    Y, X = np.mgrid[0:H, 0:W]
    distance = max(0.0, 136.0-ball_y)
    sx = 16.0 + 0.29*distance
    sy = 4.5 + 0.09*distance

    soft = np.exp(-1.7*e)
    strength = 0.25 - min(distance, 48.0)*0.0027
    green = green_confidence(frame)
    a = (soft*strength*green)[..., None]
    shade = np.array([0.52, 0.58, 0.54], np.float32)
    return np.clip(frame.astype(np.float32)*(1-a+a*shade), 0, 255).astype(np.uint8)


def impact_progress(f):
    return float(smoothstep(36.0, 45.0, np.array(f, np.float32)))


def add_sand_ridges(frame, src, p):
    \"\"\"Raise two restrained cohesive shoulders beside the penetrating ball.\"\"\"
    if p <= 0:
        return frame
    Y, X = np.mgrid[0:H, 0:W]
    out = frame.astype(np.float32)
    srcf = src.astype(np.float32)
    # Each mound shoulder borrows real grains immediately below it.
    for side in (-1, 1):
        cx = CX + side*(29.0 + 4.5*p)
        cy = 178.0 - 5.0*p
        ex = (X-cx)/(13.0+2.0*p)
        ey = (Y-cy)/(7.5+2.5*p)
        shape = np.clip(1.0-(ex*ex+ey*ey), 0, 1)
        shape = smoothstep(0.0, 0.55, shape) * p
        # Sample a nearby piece of the photographed sand surface.
        sample_x = np.clip(np.rint(X - side*7).astype(int), 0, W-1)
        sample_y = np.clip(np.rint(Y + 16 - 4*p).astype(int), 0, H-1)
        tex = srcf[sample_y, sample_x]
        valid = green_confidence(tex.astype(np.uint8))
        a = (shape*valid)[..., None]
        out = tex*a + out*(1-a)
    return np.clip(out, 0, 255).astype(np.uint8)


def add_front_lip(frame, src, ball_y, p):

    if p <= 0:
        return frame

    rr = dx*dx + (Y-ball_y)*(Y-ball_y)

    waviness = 0.9*np.sin(dx*0.34) + 0.45*np.sin(dx*0.83+1.2)

    horizontal = 1.0-smoothstep(25.5, 33.5, np.abs(dx))

    sx = np.clip(np.rint(X + 2.0*np.sin(Y*0.21)).astype(int), 0, W-1)
    sy = np.clip(np.rint(190.0 + (Y-lip_top)*0.72 + 0.12*np.abs(dx)).astype(int), 0, H-1)
