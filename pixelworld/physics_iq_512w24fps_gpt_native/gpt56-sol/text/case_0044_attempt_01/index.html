+"""Generate a fixed-camera, physically motivated paper-tearing demonstration.
+
+The source photograph supplies every texture.  The green sheet and the two
+blue-black grabbers are segmented numerically, then moved as layered rigid
+bodies over an inpainted version of the original tabletop.  The sheet is
+split along its already visible diagonal stress/crease and gains a light,
+fibrous torn edge as tension increases.
+"""
+from pathlib import Path
+import numpy as np
+import cv2
+from PIL import Image
+import artifact_io
+
+CASE_DIR = Path(__file__).resolve().parent
+W, H, N = 512, 288, 120
+
+
+def smoothstep(x):
+    x = float(np.clip(x, 0.0, 1.0))
+    return x * x * (3.0 - 2.0 * x)
+
+
+def largest_component(binary):
+    n, labels, stats, _ = cv2.connectedComponentsWithStats(binary.astype(np.uint8), 8)
+    if n <= 1:
+        return binary.astype(np.uint8)
+    return (labels == (1 + np.argmax(stats[1:, cv2.CC_STAT_AREA]))).astype(np.uint8)
+
+
+def grabber_mask(hh, ss, vv, side):
+    """Obtain a complete silhouette from blue/black seed colors in a side ROI."""
+    yy, xx = np.mgrid[:H, :W]
+    if side == "left":
+        roi = (xx < 151) & (yy >= 28) & (yy < 111)
+    else:
+        roi = (xx > 377) & (yy >= 28) & (yy < 113)
+    # Dark metal/plastic plus the desaturated black portions of the tools.
+    seed = roi & ((((hh >= 84) & (hh <= 139) & (ss > 34) & (vv < 195))) |
+                  ((vv < 66) & (ss > 12)))
+    seed = cv2.morphologyEx(seed.astype(np.uint8), cv2.MORPH_CLOSE,
+                            cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (7, 7)))
+    seed = largest_component(seed)
+    contours, _ = cv2.findContours(seed, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
+    out = np.zeros((H, W), np.uint8)
+    if contours:
+        c = max(contours, key=cv2.contourArea)
+        cv2.drawContours(out, [c], -1, 255, cv2.FILLED, lineType=cv2.LINE_AA)
+    return out.astype(np.float32) / 255.0
+
+
+def affine_about(pivot, angle_deg, dx, dy, scale=1.0):
+    m = cv2.getRotationMatrix2D(tuple(map(float, pivot)), float(angle_deg), float(scale))
+    m[:, 2] += (float(dx), float(dy))
+    return m.astype(np.float32)
+
+
+def warp_premultiplied(rgb, alpha, matrix):
+    """Bilinear warp without importing dark RGB from outside a cutout."""
+    a = alpha.astype(np.float32)
+    premul = rgb.astype(np.float32) * a[..., None]
+    wa = cv2.warpAffine(a, matrix, (W, H), flags=cv2.INTER_LINEAR,
+                        borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+    wp = cv2.warpAffine(premul, matrix, (W, H), flags=cv2.INTER_LINEAR,
+                        borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+    wrgb = wp / np.maximum(wa[..., None], 1e-5)
+    return wrgb, np.clip(wa, 0, 1)
+
+
+def composite(dst, src, alpha):
+    return dst * (1.0 - alpha[..., None]) + src * alpha[..., None]
+
+
+def transformed_point(matrix, p):
+    return np.array([matrix[0, 0]*p[0] + matrix[0, 1]*p[1] + matrix[0, 2],
+                     matrix[1, 0]*p[0] + matrix[1, 1]*p[1] + matrix[1, 2]])
+
+
+def make_background(source, removal):
+    """Recover the locked-off tabletop under the movable demonstration pieces."""
+    hole = cv2.dilate((removal > .08).astype(np.uint8),
+                      cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (9, 9))) * 255
+    # Telea preserves the broad horizontal wood illumination better than a
+    # flat painted fill. A little grain from nearby clean pixels is restored.
+    bg = cv2.inpaint(source, hole, 7, cv2.INPAINT_TELEA).astype(np.float32)
+    rng = np.random.default_rng(4401)
+    noise = rng.normal(0, 1.15, (H, W, 1)).astype(np.float32)
+    noise = cv2.GaussianBlur(noise, (0, 0), .55)[..., None]
+    inside = (hole > 0)[..., None]
+    bg = np.where(inside, bg + noise, source.astype(np.float32))
+    return np.clip(bg, 0, 255)
+
+
+def main():
+    source = np.asarray(Image.open(CASE_DIR / "first_frame.png").convert("RGB"), dtype=np.uint8)
+    if source.shape != (H, W, 3):
+        raise ValueError(source.shape)
+    hsv = cv2.cvtColor(source, cv2.COLOR_RGB2HSV)
+    hh, ss, vv = [hsv[..., i] for i in range(3)]
+
+    # The only large saturated green component is the sheet of paper.
+    green_seed = (hh > 34) & (hh < 87) & (ss > 67) & (vv > 52)
+    paper_u8 = largest_component(green_seed)
+    paper_u8 = cv2.morphologyEx(paper_u8, cv2.MORPH_CLOSE,
+                                cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (3, 3)))
+    paper_alpha = cv2.GaussianBlur(paper_u8.astype(np.float32), (0, 0), .52)
+    paper_alpha = np.clip(paper_alpha, 0, 1)
+
+    left_tool = grabber_mask(hh, ss, vv, "left")
+    right_tool = grabber_mask(hh, ss, vv, "right")
+
+    # The photograph shows a diagonal stressed neck between an upper-right and
+    # lower-left wing. A deterministic irregular threshold makes a natural rip.
+    yy, xx = np.mgrid[:H, :W]
+    rng = np.random.default_rng(44)
+    knots = rng.normal(0, 1.65, 24)
+    jitter = np.interp(yy[:, 0], np.linspace(0, H-1, len(knots)), knots)[:, None]
+    q = xx - 1.45 * yy - jitter
+    left_sel = (q < 139.2).astype(np.float32)
+    right_sel = 1.0 - left_sel
+    left_paper = paper_alpha * left_sel
+    right_paper = paper_alpha * right_sel
+
+    # Fibrous, pale cross-sections only on the newly separated interface.
+    lp = (left_paper > .4).astype(np.uint8)
+    rp = (right_paper > .4).astype(np.uint8)
+    left_rip = lp & cv2.dilate(rp, np.ones((5, 5), np.uint8))
+    right_rip = rp & cv2.dilate(lp, np.ones((5, 5), np.uint8))
+    left_rip = cv2.GaussianBlur(left_rip.astype(np.float32), (0, 0), .55)
+    right_rip = cv2.GaussianBlur(right_rip.astype(np.float32), (0, 0), .55)
+
+    removal = np.maximum.reduce([paper_alpha, left_tool, right_tool])
+    background = make_background(source, removal)
+    srcf = source.astype(np.float32)
+
+    frames = []
+    for i in range(N):
+        if i == 0:
+            frames.append(source.copy())
+            continue
+
+        # The setup is momentarily steady, tension then grows; the final half
+        # second has a small damped recoil rather than unnatural constant speed.
+        pull = smoothstep((i - 12) / 63.0)
+        if i > 75:
+            pull = 1.0 - 0.055 * np.exp(-(i-75)/18.0) * np.cos((i-75)*0.31)
+        pull = float(np.clip(pull, 0, 1.02))
+        tear = smoothstep((i - 22) / 39.0)
+
+        # A restrained, deterministic physical vibration under high tension.
+        vib = tear * np.exp(-max(i-65, 0)/35.0) * np.sin(i * 0.72)
+        dx = 58.0 * pull
+        left_dy = 3.0 * tear + 0.38 * vib
+        right_dy = -2.0 * tear - 0.30 * vib
+        angle = -3.6 * tear
+
+        ml = affine_about((122, 89), angle, -dx, left_dy, 1.0 + .006*pull)
+        mr = affine_about((404, 88), angle, dx, right_dy, 1.0 + .006*pull)
+        # Tool tips and held paper corners move together.
+        mtl = affine_about((122, 89), -0.35*tear, -dx, left_dy + .12*vib)
+        mtr = affine_about((404, 88), 0.25*tear, dx, right_dy - .10*vib)
+
+        frame = background.copy()
+
+        # Contact shadows remain attached to the paper, offset down-right onto
+        # the table. They also make the widening central tear readable.
+        for alpha, mat, strength in ((left_paper, ml, .24), (right_paper, mr, .24),
+                                     (left_tool, mtl, .31), (right_tool, mtr, .31)):
+            _, wa = warp_premultiplied(np.zeros_like(srcf), alpha, mat)
+            shadow = cv2.GaussianBlur(wa, (0, 0), 2.2)
+            shadow = cv2.warpAffine(shadow, np.float32([[1, 0, 2], [0, 1, 3]]),
+                                        (W, H), flags=cv2.INTER_LINEAR,
+                                        borderMode=cv2.BORDER_CONSTANT)
+            frame *= (1.0 - strength * shadow[..., None])
+
+        # Torn edge changes gradually from the original green to exposed pale
+        # fibres, avoiding a sudden visual cut at the first moving frame.
+        pale = np.array([119, 176, 102], np.float32)
+        lrgb = srcf.copy()
+        rrgb = srcf.copy()
+        aedge_l = (0.62 * tear * left_rip)[..., None]
+        aedge_r = (0.62 * tear * right_rip)[..., None]
+        lrgb = lrgb * (1-aedge_l) + pale * aedge_l
+        rrgb = rrgb * (1-aedge_r) + pale * aedge_r
+
+        wrgb, wa = warp_premultiplied(lrgb, left_paper, ml)
+        frame = composite(frame, wrgb, wa)
+        wrgb, wa = warp_premultiplied(rrgb, right_paper, mr)
+        frame = composite(frame, wrgb, wa)
+
+        # A few hairline fibres bridge the opening briefly, then snap. Their
+        # endpoints follow the two bodies' affine transforms.
+        if tear > .10 and tear < .86:
+            overlay = np.clip(frame, 0, 255).astype(np.uint8)
+            fiber_sources = [(264, 87), (268, 89), (271, 91), (275, 92)]
+            for k, p in enumerate(fiber_sources):
+                a = transformed_point(ml, p)
+                b = transformed_point(mr, p)
+                # Different fibres fail at slightly different extensions.
+                if tear < .54 + .09*k:
+                    col = (112 + 5*k, 181, 103)
+                    cv2.line(overlay, tuple(np.rint(a).astype(int)),
+                             tuple(np.rint(b).astype(int)), col, 1, cv2.LINE_AA)
+            frame = overlay.astype(np.float32)
+
+        # Grabber jaws sit above and visibly retain their respective corners.
+        wrgb, wa = warp_premultiplied(srcf, left_tool, mtl)
+        frame = composite(frame, wrgb, wa)
+        wrgb, wa = warp_premultiplied(srcf, right_tool, mtr)
+        frame = composite(frame, wrgb, wa)
+
+        frames.append(np.clip(np.rint(frame), 0, 255).astype(np.uint8))
+
+    frames = np.stack(frames, axis=0)
+    assert np.array_equal(frames[0], source)
+    artifact_io.save(frames, CASE_DIR)
+
+
+if __name__ == "__main__":
+    main()

from pathlib import Path
import numpy as np
import cv2
from PIL import Image
import artifact_io

CASE_DIR = Path(__file__).resolve().parent
W, H, N = 512, 288, 120


def smoothstep(x):
    x = float(np.clip(x, 0.0, 1.0))
    return x * x * (3.0 - 2.0 * x)


def largest_component(binary):
    n, labels, stats, _ = cv2.connectedComponentsWithStats(binary.astype(np.uint8), 8)
    if n <= 1:
        return binary.astype(np.uint8)
    return (labels == (1 + np.argmax(stats[1:, cv2.CC_STAT_AREA]))).astype(np.uint8)


def grabber_mask(hh, ss, vv, side):
    \"\"\"Obtain a complete silhouette from blue/black seed colors in a side ROI.\"\"\"
    yy, xx = np.mgrid[:H, :W]
    if side == \"left\":
        roi = (xx < 151) & (yy >= 28) & (yy < 111)
    else:
        roi = (xx > 377) & (yy >= 28) & (yy < 113)
    # Dark metal/plastic plus the desaturated black portions of the tools.
    seed = roi & ((((hh >= 84) & (hh <= 139) & (ss > 34) & (vv < 195))) |
                  ((vv < 66) & (ss > 12)))
    seed = cv2.morphologyEx(seed.astype(np.uint8), cv2.MORPH_CLOSE,
                            cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (7, 7)))
    seed = largest_component(seed)
    contours, _ = cv2.findContours(seed, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
    out = np.zeros((H, W), np.uint8)
    if contours:
        c = max(contours, key=cv2.contourArea)
        cv2.drawContours(out, [c], -1, 255, cv2.FILLED, lineType=cv2.LINE_AA)
    return out.astype(np.float32) / 255.0


def affine_about(pivot, angle_deg, dx, dy, scale=1.0):
    m = cv2.getRotationMatrix2D(tuple(map(float, pivot)), float(angle_deg), float(scale))
    m[:, 2] += (float(dx), float(dy))
    return m.astype(np.float32)


def warp_premultiplied(rgb, alpha, matrix):
    \"\"\"Bilinear warp without importing dark RGB from outside a cutout.\"\"\"
    a = alpha.astype(np.float32)
    premul = rgb.astype(np.float32) * a[..., None]
    wa = cv2.warpAffine(a, matrix, (W, H), flags=cv2.INTER_LINEAR,
                        borderMode=cv2.BORDER_CONSTANT, borderValue=0)
    wp = cv2.warpAffine(premul, matrix, (W, H), flags=cv2.INTER_LINEAR,
                        borderMode=cv2.BORDER_CONSTANT, borderValue=0)
    wrgb = wp / np.maximum(wa[..., None], 1e-5)
    return wrgb, np.clip(wa, 0, 1)


def composite(dst, src, alpha):
    return dst * (1.0 - alpha[..., None]) + src * alpha[..., None]


def transformed_point(matrix, p):
    return np.array([matrix[0, 0]*p[0] + matrix[0, 1]*p[1] + matrix[0, 2],
                     matrix[1, 0]*p[0] + matrix[1, 1]*p[1] + matrix[1, 2]])


def make_background(source, removal):
    \"\"\"Recover the locked-off tabletop under the movable demonstration pieces.\"\"\"
    hole = cv2.dilate((removal > .08).astype(np.uint8),
                      cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (9, 9))) * 255
    # Telea preserves the broad horizontal wood illumination better than a
    # flat painted fill. A little grain from nearby clean pixels is restored.
    bg = cv2.inpaint(source, hole, 7, cv2.INPAINT_TELEA).astype(np.float32)
    rng = np.random.default_rng(4401)
    noise = rng.normal(0, 1.15, (H, W, 1)).astype(np.float32)
    noise = cv2.GaussianBlur(noise, (0, 0), .55)
    inside = (hole > 0)[..., None]
    bg = np.where(inside, bg + noise, source.astype(np.float32))
    return np.clip(bg, 0, 255)


def main():
    source = np.asarray(Image.open(CASE_DIR / \"first_frame.png\").convert(\"RGB\"), dtype=np.uint8)
    if source.shape "'!= (H, W, 3):
        raise ValueError(source.shape)
    hsv = cv2.cvtColor(source, cv2.COLOR_RGB2HSV)
    hh, ss, vv = [hsv[..., i] for i in range(3)]

    # The only large saturated green component is the sheet of paper.
    green_seed = (hh > 34) & (hh < 87) & (ss > 67) & (vv > 52)
    paper_u8 = largest_component(green_seed)
    paper_u8 = cv2.morphologyEx(paper_u8, cv2.MORPH_CLOSE,
                                cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (3, 3)))
    paper_alpha = cv2.GaussianBlur(paper_u8.astype(np.float32), (0, 0), .52)
    paper_alpha = np.clip(paper_alpha, 0, 1)

    left_tool = grabber_mask(hh, ss, vv, "left")
    right_tool = grabber_mask(hh, ss, vv, "right")

    # The photograph shows a diagonal stressed neck between an upper-right and
    # lower-left wing. A deterministic irregular threshold makes a natural rip.
    yy, xx = np.mgrid[:H, :W]
    rng = np.random.default_rng(44)
    knots = rng.normal(0, 1.65, 24)
    jitter = np.interp(yy[:, 0], np.linspace(0, H-1, len(knots)), knots)[:, None]
    q = xx - 1.45 * yy - jitter
    left_sel = (q < 139.2).astype(np.float32)
    right_sel = 1.0 - left_sel
    left_paper = paper_alpha * left_sel
    right_paper = paper_alpha * right_sel

    # Fibrous, pale cross-sections only on the newly separated interface.
    lp = (left_paper > .4).astype(np.uint8)
    rp = (right_paper > .4).astype(np.uint8)
    left_rip = lp & cv2.dilate(rp, np.ones((5, 5), np.uint8))
    right_rip = rp & cv2.dilate(lp, np.ones((5, 5), np.uint8))
    left_rip = cv2.GaussianBlur(left_rip.astype(np.float32), (0, 0), .55)
    right_rip = cv2.GaussianBlur(right_rip.astype(np.float32), (0, 0), .55)

    removal = np.maximum.reduce([paper_alpha, left_tool, right_tool])
    background = make_background(source, removal)
    srcf = source.astype(np.float32)

    frames = []
    for i in range(N):
        if i == 0:
            frames.append(source.copy())
            continue

        # The setup is momentarily steady, tension then grows; the final half
        # second has a small damped recoil rather than unnatural constant speed.
        pull = smoothstep((i - 12) / 63.0)
        if i > 75:
            pull = 1.0 - 0.055 * np.exp(-(i-75)/18.0) * np.cos((i-75)*0.31)
        pull = float(np.clip(pull, 0, 1.02))
        tear = smoothstep((i - 22) / 39.0)

        # A restrained, deterministic physical vibration under high tension.
        vib = tear * np.exp(-max(i-65, 0)/35.0) * np.sin(i * 0.72)
        dx = 58.0 * pull
        left_dy = 3.0 * tear + 0.38 * vib
        right_dy = -2.0 * tear - 0.30 * vib
        angle = -3.6 * tear

        ml = affine_about((122, 89), angle, -dx, left_dy, 1.0 + .006*pull)
        mr = affine_about((404, 88), angle, dx, right_dy, 1.0 + .006*pull)
        # Tool tips and held paper corners move together.
        mtl = affine_about((122, 89), -0.35*tear, -dx, left_dy + .12*vib)
        mtr = affine_about((404, 88), 0.25*tear, dx, right_dy - .10*vib)

        frame = background.copy()

        # Contact shadows remain attached to the paper, offset down-right onto
        # the table. They also make the widening central tear readable.
        for alpha, mat, strength in ((left_paper, ml, .24), (right_paper, mr, .24),
                                     (left_tool, mtl, .31), (right_tool, mtr, .31)):
            _, wa = warp_premultiplied(np.zeros_like(srcf), alpha, mat)
            shadow = cv2.GaussianBlur(wa, (0, 0), 2.2)
            shadow = cv2.warpAffine(shadow, np.float32([[1, 0, 2], [0, 1, 3]]),
                                        (W, H), flags=cv2.INTER_LINEAR,
                                        borderMode=cv2.BORDER_CONSTANT)
            frame *= (1.0 - strength * shadow[..., None])

        # Torn edge changes gradually from the original green to exposed pale
        # fibres, avoiding a sudden visual cut at the first moving frame.
        pale = np.array([119, 176, 102], np.float32)
        lrgb = srcf.copy()
        rrgb = srcf.copy()
        aedge_l = (0.62 * tear * left_rip)[..., None]
        aedge_r = (0.62 * tear * right_rip)[..., None]
        lrgb = lrgb * (1-aedge_l) + pale * aedge_l
        rrgb = rrgb * (1-aedge_r) + pale * aedge_r

        wrgb, wa = warp_premultiplied(lrgb, left_paper, ml)
        frame = composite(frame, wrgb, wa)
        wrgb, wa = warp_premultiplied(rrgb, right_paper, mr)
        frame = composite(frame, wrgb, wa)

        # A few hairline fibres bridge the opening briefly, then snap. Their
        # endpoints follow the two bodies'"' affine transforms.
        if tear > .10 and tear < .86:
            overlay = np.clip(frame, 0, 255).astype(np.uint8)
            fiber_sources = [(264, 87), (268, 89), (271, 91), (275, 92)]
            for k, p in enumerate(fiber_sources):
                a = transformed_point(ml, p)
                b = transformed_point(mr, p)
                # Different fibres fail at slightly different extensions.
                if tear < .54 + .09*k:
                    col = (112 + 5*k, 181, 103)
                    cv2.line(overlay, tuple(np.rint(a).astype(int)),
                             tuple(np.rint(b).astype(int)), col, 1, cv2.LINE_AA)
            frame = overlay.astype(np.float32)

        # Grabber jaws sit above and visibly retain their respective corners.
        wrgb, wa = warp_premultiplied(srcf, left_tool, mtl)
        frame = composite(frame, wrgb, wa)
        wrgb, wa = warp_premultiplied(srcf, right_tool, mtr)
        frame = composite(frame, wrgb, wa)

        frames.append(np.clip(np.rint(frame), 0, 255).astype(np.uint8))

    frames = np.stack(frames, axis=0)
    assert np.array_equal(frames[0], source)
    artifact_io.save(frames, CASE_DIR)


if __name__ == \"__main__\":
    main()
