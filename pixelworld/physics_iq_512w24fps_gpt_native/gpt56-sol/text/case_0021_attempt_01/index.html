+    """Screen-space trajectory: tube axis, followed by the cardboard incline."""
+    # The first-frame ball is already clear of the tube mouth.  It continues
+    # along the tube's axis to the foot of the cardboard at a near-constant
+    # rolling speed.
+    t_hit = 1.52
+    p0 = np.array([127.0, 90.0], np.float32)
+    p1 = np.array([235.0, 136.9], np.float32)
+    if t <= t_hit:
+        # A very small ease-in avoids a synthetic-looking instantaneous start,
+        # while retaining most of the measured straight-line speed.
+        u = t / t_hit
+        s = u*u*(2.0-u) if u < .16 else (u - .0256) / .9744
+        s = float(np.clip(s, 0, 1))
+        return p0 + (p1-p0)*s, 77.5*t

+    # Cardboard upper edge: y = 195.7 - .252*x for the ball center (normal
+    # offset by its radius from the visible ramp edge).  Momentum carries it
+    # uphill, then gravity reverses it naturally near the end of the shot.
+    tau = t-t_hit

+    q = v0*tau - 0.5*accel*tau*tau          # signed distance along incline
+    q = max(q, -5.0)
+    tangent = np.array([1.0, -0.252], np.float32)
+    tangent /= np.linalg.norm(tangent)
+    p = p1 + tangent*q
+    travelled = 77.5*t_hit + q
+    return p, travelled

+def add_shadow(frame, cx, cy, on_ramp):

+    # light in the source frame puts the soft shadow just lower-right.
+    layer = np.zeros((H, W), np.float32)

+    cv2.ellipse(seam, (c[0]+5, c[1]), (12, 20), 18, 108, 252, 110, 1, cv2.LINE_AA)

+    seam = cv2.warpAffine(seam, M, (w, h), flags=cv2.INTER_LINEAR)

+    target = np.array([188, 201, 122], np.float32)
+    out = sprite.astype(np.float32)*(1-seam_a[...,None]) + target*seam_a[...,None]
+    return np.clip(out,0,255).astype(np.uint8)

+        on_ramp = t >= 1.50
+        frame = plate.copy()

+        # Rolling without slipping. Sign matches clockwise rotation for a ball
+        # moving down-right/up-right in image coordinates.
+        angle = -np.degrees(travelled/19.5)

+        srgb = add_felt_detail(srgb, sa, angle % 360)
+        composite(frame, srgb, sa, float(pos[0]), float(pos[1]))
+        frames[i] = frame
+    return frames

    dst = frame[y0:y1, x0:x1].astype(np.float32)
    src = sprite[sy0:sy1, sx0:sx1].astype(np.float32)
    frame[y0:y1, x0:x1] = np.clip(dst*(1-a)+src*a, 0, 255).astype(np.uint8)


def add_shadow(frame, cx, cy, on_ramp):
    # Contact shadow follows the supporting surface. The fixed upper-left key
    # light in the source frame puts the soft shadow just lower-right.
    layer = np.zeros((H, W), np.float32)
    axes = (18, 5) if on_ramp else (17, 5)
    angle = -14 if on_ramp else 23
    center = (int(round(cx+4)), int(round(cy+18)))
    cv2.ellipse(layer, center, axes, angle, 0, 360, .31, -1, cv2.LINE_AA)
    layer = cv2.GaussianBlur(layer, (0, 0), 3.2)
    # Restrict ramp shadows to the warm cardboard side of its lower edge; this
    # prevents a floating shadow from leaking across the silhouette boundary.
    if on_ramp:
        yy, xx = np.mgrid[:H, :W]
        support = yy >= (212.1 - .252*xx) - 4
        layer *= support.astype(np.float32)
    frame[:] = np.clip(frame.astype(np.float32)*(1-layer[..., None]), 0, 255).astype(np.uint8)


def add_felt_detail(sprite, alpha, angle_degrees):
    # Rotate two subdued curved seam strokes with the ball. They are deliberately
    # low contrast because most seam detail is already present in source pixels.
    h, w = alpha.shape
    seam = np.zeros((h, w), np.uint8)
    c = (w//2, h//2)
    cv2.ellipse(seam, (c[0]-5, c[1]), (12, 20), 18, -72, 72, 135, 1, cv2.LINE_AA)
    cv2.ellipse(seam, (c[0]+5, c[1]), (12, 20), 18, 108, 252, 110, 1, cv2.LINE_AA)
    M = cv2.getRotationMatrix2D(c, angle_degrees, 1.0)
    seam = cv2.warpAffine(seam, M, (w, h), flags=cv2.INTER_LINEAR)
    seam_a = (seam.astype(np.float32)/255.0)*alpha*.34
    target = np.array([188, 201, 122], np.float32)
    out = sprite.astype(np.float32)*(1-seam_a[...,None]) + target*seam_a[...,None]
    return np.clip(out,0,255).astype(np.uint8)


def generate():
    first = np.asarray(Image.open(CASE_DIR/'"'first_frame.png').convert('RGB'))
    assert first.shape == (H, W, 3)
    plate = make_clean_plate(first)
    ball_rgb, ball_alpha = extract_ball(first)
    frames = np.empty((N, H, W, 3), np.uint8)
    frames[0] = first                       # contract: bit-identical first frame

    for i in range(1, N):
        t = i/FPS
        pos, travelled = ball_position(t)
        on_ramp = t >= 1.50
        frame = plate.copy()
        add_shadow(frame, float(pos[0]), float(pos[1]), on_ramp)
        # Rolling without slipping. Sign matches clockwise rotation for a ball
        # moving down-right/up-right in image coordinates.
        angle = -np.degrees(travelled/19.5)
        srgb, sa = rotate_sprite(ball_rgb, ball_alpha, angle)
        srgb = add_felt_detail(srgb, sa, angle % 360)
        composite(frame, srgb, sa, float(pos[0]), float(pos[1]))
        frames[i] = frame
    return frames


if __name__ == '__main__':
    frames = generate()
    artifact_io.save(frames, CASE_DIR)
    print('saved', frames.shape, frames.dtype)
