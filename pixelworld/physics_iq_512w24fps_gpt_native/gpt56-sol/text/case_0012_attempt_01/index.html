+"""Generate the requested five-second, locked-camera RGB sequence.
+
+The supplied photograph is kept as the immutable plate.  The moving tennis
+ball is rendered analytically (including rolling surface detail and contact
+shadow), which avoids altering any of the small stationary objects already on
+the coffee table.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+W, H = 512, 288
+N_FRAMES = 120
+
+# Geometry measured from the supplied plate.  The black cylindrical object's
+# right lip is at x ~= 221 and its centerline is y ~= 67.  Starting the ball
+# behind that lip makes the first visible sliver appear naturally at frame 1.
+PIPE_LIP_X = 220.5
+START_X = 203.9
+END_X = 525.0
+BALL_Y = 67.3
+RADIUS = 14.25
+
+
+def source_felt_texture(plate: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
+    """Obtain subtle, photographic felt grain from the stationary brown ball.
+
+    Only high-frequency chroma/luminance variation is retained.  Consequently
+    the original ball's directional lighting does not rotate with the texture;
+    the generated ball receives its own stationary scene lighting below.
+    """
+    # The stationary brown tennis ball occupies approximately x=60..91,
+    # y=77..105 and is centered at (75.5, 91.2).
+    size = 39
+    cx, cy = 75.3, 91.1
+    yy, xx = np.mgrid[:size, :size].astype(np.float32)
+    map_x = cx + xx - (size - 1) / 2
+    map_y = cy + yy - (size - 1) / 2
+    sample = cv2.remap(
+        plate.astype(np.float32), map_x, map_y,
+        interpolation=cv2.INTER_CUBIC, borderMode=cv2.BORDER_REFLECT_101,
+    )
+    smooth = cv2.GaussianBlur(sample, (0, 0), 3.2)
+    ratio = np.clip((sample + 8.0) / (smooth + 8.0), 0.82, 1.18)
+
+    # A separate tiny monochrome grain field adds felt-like breakup without
+    # making the outline shimmer from frame to frame.
+    rng = np.random.default_rng(120512)
+    grain = rng.normal(0.0, 1.0, (size, size)).astype(np.float32)
+    grain = grain - cv2.GaussianBlur(grain, (0, 0), 1.25)
+    grain /= max(float(grain.std()), 1e-6)
+    return ratio.astype(np.float32), grain
+
+
+def remap_texture(
+    texture: np.ndarray,
+    grain: np.ndarray,
+    u_obj: np.ndarray,
+    v_obj: np.ndarray,
+) -> tuple[np.ndarray, np.ndarray]:
+    """Sample the felt maps at normalized, rotating object coordinates."""
+    th, tw = texture.shape[:2]
+    # Use most of the photographed sphere while avoiding the old background.
+    mx = (tw - 1) / 2 + u_obj * 14.0
+    my = (th - 1) / 2 + v_obj * 14.0
+    tex = cv2.remap(
+        texture, mx.astype(np.float32), my.astype(np.float32),
+        interpolation=cv2.INTER_LINEAR, borderMode=cv2.BORDER_REFLECT_101,
+    )
+    grit = cv2.remap(
+        grain, mx.astype(np.float32), my.astype(np.float32),
+        interpolation=cv2.INTER_LINEAR, borderMode=cv2.BORDER_REFLECT_101,
+    )
+    return tex, grit
+
+
+def shadow_layer(cx: float, visibility: float) -> tuple[np.ndarray, np.ndarray]:
+    """Return an RGB shadow color and soft alpha matte for the current ball."""
+    yy, xx = np.mgrid[:H, :W].astype(np.float32)
+    # Upper-left illumination on the source objects puts the compact core a
+    # little below/right of the contact point and a fainter penumbra around it.
+    sx, sy = cx + 2.4, BALL_Y + RADIUS - 0.4
+    core = np.exp(-0.5 * (((xx - sx) / 10.5) ** 2 + ((yy - sy) / 2.35) ** 2))
+    pen = np.exp(-0.5 * (((xx - (sx + 1.5)) / 16.5) ** 2 + ((yy - sy) / 4.0) ** 2))
+    alpha = np.clip(0.18 * core + 0.075 * pen, 0.0, 0.235) * visibility
+    # While the ball is inside the pipe, the lip also occludes its shadow.
+    alpha *= np.clip((xx - PIPE_LIP_X + 1.0) / 2.0, 0.0, 1.0)
+    color = np.zeros((H, W, 3), np.float32)
+    color[:] = (43.0, 34.0, 27.0)
+    return color, alpha[..., None]
+
+
+def ball_layer(
+    cx: float,
+    angle: float,
+    texture: np.ndarray,
+    grain: np.ndarray,
+) -> tuple[np.ndarray, np.ndarray]:
+    """Render a softly antialiased orange tennis ball at ``cx``."""
+    # A local tile is enough, but full-frame grids make compositing and clipping
+    # transparent and remain inexpensive at this resolution.
+    yy, xx = np.mgrid[:H, :W].astype(np.float32)
+    u = (xx - cx) / RADIUS
+    v = (yy - BALL_Y) / RADIUS
+    rr = u * u + v * v
+    inside = rr <= 1.10
+    z = np.sqrt(np.clip(1.0 - rr, 0.0, 1.0))
+
+    # Convert current screen-space normals to coordinates fixed on the rolling
+    # sphere.  Horizontal travel means an in-image clockwise rotation.
+    c, s = np.cos(angle), np.sin(angle)
+    uo = c * u + s * v
+    vo = -s * u + c * v
+    tex, grit = remap_texture(texture, grain, uo, vo)
+
+    # Fixed-world lighting: broad upper-left softbox plus restrained specular.
+    light = np.array([-0.43, -0.57, 0.70], np.float32)
+    light /= np.linalg.norm(light)
+    ndotl = np.maximum(0.0, u * light[0] + v * light[1] + z * light[2])
+    shade = 0.55 + 0.47 * ndotl
+    rim = np.clip(z * 1.7, 0.0, 1.0)
+    shade *= 0.88 + 0.12 * rim
+    spec = 14.0 * np.power(np.maximum(ndotl, 0.0), 16.0)
+
+    orange = np.array([225.0, 116.0, 25.0], np.float32)
+    rgb = orange[None, None, :] * shade[..., None]
+    rgb *= tex
+    rgb *= (1.0 + 0.018 * grit[..., None])
+    rgb += spec[..., None]
+
+    # A narrow, warm cream tennis seam.  The sinusoidal great-circle-like path
+    # is anchored in object coordinates, so it visibly turns at the proper
+    # rolling rate rather than sliding across the ball.
+    seam_distance = np.abs(vo - 0.31 * np.sin(np.pi * uo))
+    seam = 1.0 - np.clip((seam_distance - 0.032) / 0.052, 0.0, 1.0)
+    seam *= np.clip((0.94 - rr) / 0.10, 0.0, 1.0)
+    seam_color = np.array([244.0, 191.0, 105.0], np.float32)
+    rgb = rgb * (1.0 - 0.58 * seam[..., None]) + seam_color * (0.58 * seam[..., None])
+
+    # Analytic edge antialiasing, with fixed angular microvariation to suggest
+    # a fuzzy tennis-ball silhouette without temporal crawling.
+    phi = np.arctan2(v, u)
+    fuzz = 0.018 * (np.sin(17.0 * phi + 0.7) + 0.45 * np.sin(31.0 * phi))
+    radial = np.sqrt(np.maximum(rr, 0.0))
+    alpha = np.clip((1.015 + fuzz - radial) / 0.055, 0.0, 1.0)
+    alpha *= inside
+
+    # The right-facing pipe mouth/lip is a vertical occluder during emergence.
+    # Once the rear of the sphere clears it this clamp has no effect.
+    lip = np.clip((xx - PIPE_LIP_X) / 1.15, 0.0, 1.0)
+    if cx - RADIUS < PIPE_LIP_X + 0.5:
+        alpha *= lip
+
+    return np.clip(rgb, 0.0, 255.0), alpha[..., None]
+
+
+def composite(dst: np.ndarray, src: np.ndarray, alpha: np.ndarray) -> np.ndarray:
+    return dst * (1.0 - alpha) + src * alpha
+
+
+def make_frames() -> np.ndarray:
+    plate_u8 = np.asarray(Image.open(CASE_DIR / "first_frame.png").convert("RGB"))
+    assert plate_u8.shape == (H, W, 3)
+    plate = plate_u8.astype(np.float32)
+    texture, grain = source_felt_texture(plate_u8)
+
+    frames = np.empty((N_FRAMES, H, W, 3), dtype=np.uint8)
+    frames[0] = plate_u8  # Contract: exactly preserve every source byte.
+
+    speed = (END_X - START_X) / (N_FRAMES - 1)
+    for i in range(1, N_FRAMES):
+        cx = START_X + speed * i
+        # No easing: the demonstration is a constant-speed, straight roll.
+        angle = (cx - START_X) / RADIUS
+
+        # Sub-frame integration gives a restrained photographic shutter blur.
+        # Three samples span 36% of a frame and preserve a crisp center image.
+        accum = np.zeros_like(plate)
+        weights = (0.22, 0.56, 0.22)
+        offsets = (-0.18, 0.0, 0.18)
+        for weight, offset in zip(weights, offsets):
+            scx = cx + speed * offset
+            sangle = angle + speed * offset / RADIUS
+            visible_width = np.clip((scx + RADIUS - PIPE_LIP_X) / (2.0 * RADIUS), 0.0, 1.0)
+
+            rendered = plate.copy()
+            sh_rgb, sh_a = shadow_layer(scx, float(visible_width))
+            rendered = composite(rendered, sh_rgb, sh_a)
+            b_rgb, b_a = ball_layer(scx, sangle, texture, grain)
+            rendered = composite(rendered, b_rgb, b_a)
+            accum += weight * rendered
+
+        frames[i] = np.clip(np.rint(accum), 0, 255).astype(np.uint8)
+
+    # An explicit guard prevents an accidental filtering or color conversion of
+    # the mandated source frame during future edits to this script.
+    assert np.array_equal(frames[0], plate_u8)
+    return frames
+
+
+if __name__ == "__main__":
+    artifact_io.save(make_frames(), CASE_DIR)
+
