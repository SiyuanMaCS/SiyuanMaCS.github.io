+"""Generate the requested locked-off demonstration from the supplied first frame.
+
+The construction deliberately keeps the photographed frame as the source of all
+scene texture.  Only the moving props are rendered: an orange tennis ball, a
+dark-blue reacher, and the photographed card, which is perspective extracted
+and smoothly rotated.  The reacher/ball layer is composited behind the card so
+the important occlusion in the prompt is physically unambiguous.
+"""
+
+from pathlib import Path
+import math
+
+import cv2
+import numpy as np
+from PIL import Image, ImageDraw, ImageFilter
+
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+W, H, N = 512, 288, 120
+SS = 3
+
+
+def smoother(a: float) -> float:
+    """C2 easing, clamped to [0, 1]."""
+    a = min(1.0, max(0.0, a))
+    return a * a * a * (a * (a * 6.0 - 15.0) + 10.0)
+
+
+def mix(a, b, t):
+    t = smoother(t)
+    return (a[0] + (b[0] - a[0]) * t,
+            a[1] + (b[1] - a[1]) * t)
+
+
+def alpha_over(base: np.ndarray, rgba: np.ndarray) -> np.ndarray:
+    a = rgba[..., 3:4].astype(np.float32) / 255.0
+    return np.clip(base.astype(np.float32) * (1.0 - a) +
+                   rgba[..., :3].astype(np.float32) * a, 0, 255).astype(np.uint8)
+
+
+def add_ball(layer: Image.Image, cx: float, cy: float, radius: float, frame: int,
+             on_table: bool = False) -> None:
+    """Paint a softly textured orange tennis ball at supersampled resolution."""
+    s = SS
+    c = (int(round(cx * s)), int(round(cy * s)))
+    r = int(round(radius * s))
+    d = ImageDraw.Draw(layer, "RGBA")
+
+    # Contact shadow is strongest as the ball reaches the platform/table.
+    if on_table:
+        shadow = Image.new("RGBA", layer.size, (0, 0, 0, 0))
+        sd = ImageDraw.Draw(shadow, "RGBA")
+        sd.ellipse((c[0] - int(1.05*r), c[1] + int(.64*r),
+                    c[0] + int(1.18*r), c[1] + int(1.06*r)),
+                   fill=(0, 0, 0, 105))
+        shadow = shadow.filter(ImageFilter.GaussianBlur(3.1 * s))
+        layer.alpha_composite(shadow)
+
+    # Many concentric offset disks form a convincing spherical light falloff.
+    d = ImageDraw.Draw(layer, "RGBA")
+    for rr in range(r, 0, -1):
+        q = rr / r
+        # Light is high and camera-left, matching the broad room illumination.
+        ox = int(-0.16 * r * (1.0 - q))
+        oy = int(-0.20 * r * (1.0 - q))
+        shade = 0.54 + 0.48 * (1.0 - q)
+        red = int(196 * shade)
+        green = int(104 * shade)
+        blue = int(27 * shade)
+        d.ellipse((c[0]-rr+ox, c[1]-rr+oy, c[0]+rr+ox, c[1]+rr+oy),
+                  fill=(red, green, blue, 255))
+
+    # Felt flecks are deterministic and move with the ball, not the camera.
+    rng = np.random.default_rng(621)
+    for _ in range(155):
+        ang = rng.uniform(0, 2 * math.pi)
+        rad = math.sqrt(rng.uniform()) * radius * .90
+        x = (cx + math.cos(ang) * rad) * s
+        y = (cy + math.sin(ang) * rad) * s
+        # Tiny light/dark fibers, all subpixel after downsampling.
+        if rng.random() < .58:
+            col = (236, 139, 48, rng.integers(38, 85))
+        else:
+            col = (88, 42, 13, rng.integers(28, 65))
+        z = rng.choice([1, 1, 2])
+        d.ellipse((x-z, y-z, x+z, y+z), fill=tuple(map(int, col)))
+
+    # Tennis-ball seam: two subtle opposing arcs, clipped visually by sphere.
+    seam = (244, 195, 112, 205)
+    lw = max(2, int(1.15 * s))
+    box1 = (c[0]-int(.77*r), c[1]-int(1.00*r),
+            c[0]+int(.23*r), c[1]+int(1.02*r))
+    box2 = (c[0]-int(.20*r), c[1]-int(1.02*r),
+            c[0]+int(.80*r), c[1]+int(1.00*r))
+    d.arc(box1, -72, 68, fill=seam, width=lw)
+    d.arc(box2, 108, 248, fill=seam, width=lw)
+    # Small specular patch.
+    d.ellipse((c[0]-.45*r, c[1]-.56*r, c[0]-.17*r, c[1]-.30*r),
+              fill=(255, 200, 104, 75))
+
+
+def add_grabber(layer: Image.Image, wrist, ball, openness: float) -> None:
+    """Draw a long, blue-black mechanical grabber ending around the ball."""
+    s = SS
+    wx, wy = wrist
+    bx, by = ball
+    # Fixed off-camera hand end gives the shaft a realistic changing angle.
+    ax, ay = 424.0, -20.0
+    vx, vy = wx - ax, wy - ay
+    length = max(1.0, math.hypot(vx, vy))
+    nx, ny = -vy / length, vx / length
+    d = ImageDraw.Draw(layer, "RGBA")
+
+    def pts(seq):
+        return [(int(x*s), int(y*s)) for x, y in seq]
+
+    # Soft shaft shadow, twin frame rails, and narrow blue edge reflections.
+    d.line(pts([(ax+3, ay+4), (wx+3, wy+4)]), fill=(0, 0, 0, 82), width=10*s)
+    for off in (-4.0, 4.0):
+        p0 = (ax + nx*off, ay + ny*off)
+        p1 = (wx + nx*off, wy + ny*off)
+        d.line(pts([p0, p1]), fill=(4, 13, 21, 255), width=7*s)
+        d.line(pts([(p0[0]-nx*.8, p0[1]-ny*.8),
+                    (p1[0]-nx*.8, p1[1]-ny*.8)]),
+               fill=(21, 55, 81, 230), width=2*s)
+
+    # Regular pivots/cross braces make it read as a reacher, not a plain rod.
+    for q in (.20, .44, .68, .90):
+        x, y = ax + vx*q, ay + vy*q
+        d.line(pts([(x-nx*6, y-ny*6), (x+nx*6, y+ny*6)]),
+               fill=(8, 22, 33, 255), width=3*s)
+        d.ellipse(((x*s-2.3*s), (y*s-2.3*s), (x*s+2.3*s), (y*s+2.3*s)),
+                  fill=(42, 69, 86, 255), outline=(2, 8, 12, 255), width=s)
+
+    # Wrist block and two articulated pincers.  Openness grows on release.
+    d.rounded_rectangle((int((wx-10)*s), int((wy-8)*s),
+                         int((wx+10)*s), int((wy+9)*s)),
+                        radius=4*s, fill=(5, 17, 27, 255),
+                        outline=(29, 66, 91, 255), width=2*s)
+    spread = 9.0 + 11.0 * openness
+    tip = 15.8 + 8.0 * openness
+    # Pincers originate at the wrist and curl to either side of the sphere.
+    left = [(wx-4, wy+4), (bx-spread, by-12), (bx-tip, by+2), (bx-13, by+10)]
+    right = [(wx+4, wy+4), (bx+spread, by-12), (bx+tip, by+2), (bx+13, by+10)]
+    for path in (left, right):
+        d.line(pts(path), fill=(2, 9, 15, 255), width=7*s, joint="curve")
+        d.line(pts(path), fill=(18, 49, 72, 255), width=3*s, joint="curve")
+    # Black rubber pads touch the ball when closed.
+    for sign in (-1, 1):
+        px = (bx + sign*(13 + 8*openness))*s
+        py = (by + 9)*s
+        d.ellipse((px-3.2*s, py-4.5*s, px+3.2*s, py+4.5*s),
+                  fill=(3, 5, 7, 255), outline=(24, 44, 57, 255), width=s)
+
+
+def build_scene_sources(first: np.ndarray):
+    # Measured corners of the photographed trapezoidal cardstock.
+    card_quad = np.float32([[126, 99], [307, 99], [310, 250], [147, 250]])
+    tw, th = 184, 152
+    rect = np.float32([[0, 0], [tw-1, 0], [tw-1, th-1], [0, th-1]])
+    h_to_rect = cv2.getPerspectiveTransform(card_quad, rect)
+    texture = cv2.warpPerspective(first, h_to_rect, (tw, th),
+                                  flags=cv2.INTER_CUBIC,
+                                  borderMode=cv2.BORDER_REPLICATE)
+
+    # Recover what is behind the original card for portions exposed by rotation.
+    mask = np.zeros((H, W), np.uint8)
+    cv2.fillConvexPoly(mask, np.int32(card_quad), 255)
+    mask = cv2.dilate(mask, np.ones((5, 5), np.uint8), iterations=1)
+    background = cv2.inpaint(first, mask, 7, cv2.INPAINT_TELEA)
+
+    # The black rotating platform continues behind the lower edge of the card.
+    # Restore only the inpainted area so all photographed pixels elsewhere stay.
+    yy, xx = np.mgrid[0:H, 0:W]
+    platform = (((xx-260)/126.0)**2 + ((yy-268)/37.0)**2 <= 1.0) & (yy >= 233)
+    restore = platform & (mask > 0)
+    grad = np.clip(23 - .18*(yy-238) - .020*np.abs(xx-260), 3, 23)
+    for c, scale in enumerate((.88, .92, .96)):
+        background[..., c][restore] = np.uint8(grad[restore] * scale)
+    return background, texture, card_quad, rect
+
+
+def render_card(base: np.ndarray, texture: np.ndarray, rect: np.ndarray,
+                source_quad: np.ndarray, angle_deg: float) -> np.ndarray:
+    pivot = np.array([218.0, 174.5], np.float32)
+    a = math.radians(angle_deg)
+    rot = np.array([[math.cos(a), -math.sin(a)],
+                    [math.sin(a),  math.cos(a)]], np.float32)
+    target = (source_quad - pivot) @ rot.T + pivot
+    hmat = cv2.getPerspectiveTransform(rect, target.astype(np.float32))
+    alpha_src = np.full(texture.shape[:2], 255, np.uint8)
+    alpha = cv2.warpPerspective(alpha_src, hmat, (W, H), flags=cv2.INTER_CUBIC,
+                                borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+    warped = cv2.warpPerspective(texture, hmat, (W, H), flags=cv2.INTER_CUBIC,
+                                 borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+
+    # Cast a slight offset soft shadow before laying down the opaque paper.
+    shadow_a = cv2.GaussianBlur(alpha, (0, 0), 3.2)
+    shadow_a = np.roll(np.roll(shadow_a, 5, axis=0), 4, axis=1)
+    sa = (shadow_a.astype(np.float32) / 255.0 * .23)[..., None]
+    out = np.clip(base.astype(np.float32) * (1.0-sa), 0, 255)
+    aa = (alpha.astype(np.float32) / 255.0)[..., None]
+    out = out * (1.0-aa) + warped.astype(np.float32) * aa
+    return np.uint8(np.clip(out, 0, 255))
+
+
+def make_frames() -> np.ndarray:
+    first = np.asarray(Image.open(ROOT / "first_frame.png").convert("RGB"),
+                       dtype=np.uint8)
+    background, card_texture, card_quad, rect = build_scene_sources(first)
+    frames = np.empty((N, H, W, 3), dtype=np.uint8)
+    frames[0] = first  # Contract: exact, not merely visually identical.
+
+    final_ball = (302.0, 224.0)
+    for f in range(1, N):
+        # The platform/card begins turning gently as the prop approaches.
+        # Positive image-plane angle is clockwise because image y points down.
+        angle = 26.0 * smoother(max(0.0, (f-7) / 112.0))
+        base = background.copy()
+
+        prop = Image.new("RGBA", (W*SS, H*SS), (0, 0, 0, 0))
+
+        if f <= 12:
+            # Enter naturally from above the frame; no pop at frame 1.
+            ball = mix((390.0, -30.0), (356.0, 45.0), f / 12.0)
+            opening = 0.0
+            holding = True
+        elif f <= 31:
+            ball = mix((356.0, 45.0), (294.0, 77.0), (f-12) / 19.0)
+            opening = 0.0
+            holding = True
+        elif f <= 72:
+            ball = mix((294.0, 77.0), (302.0, 207.0), (f-31) / 41.0)
+            opening = 0.0
+            holding = True
+        elif f <= 82:
+            # Jaws open while gravity settles the last few pixels onto the table.
+            q = (f-72) / 10.0
+            ball = mix((302.0, 207.0), final_ball, q)
+            opening = smoother(q)
+            holding = True
+        else:
+            ball = final_ball
+            opening = 1.0
+            holding = False
+
+        # Ball is painted first; jaws sit over it while held.
+        add_ball(prop, ball[0], ball[1], 18.0, f, on_table=(f >= 73))
+        if holding:
+            wrist = (ball[0], ball[1]-25.0)
+            add_grabber(prop, wrist, ball, opening)
+        else:
+            # Reacher withdraws smoothly after release, leaving the ball behind.
+            q = (f-82) / 37.0
+            ghost_ball = mix(final_ball, (374.0, -37.0), q)
+            wrist = (ghost_ball[0], ghost_ball[1]-25.0)
+            add_grabber(prop, wrist, ghost_ball, 1.0)
+
+        # Downsample all synthetic geometry together for stable antialiasing.
+        prop = prop.resize((W, H), Image.Resampling.LANCZOS)
+        composed = alpha_over(base, np.asarray(prop, dtype=np.uint8))
+
+        # This last compositing operation creates the key behind-card occlusion.
+        frames[f] = render_card(composed, card_texture, rect, card_quad, angle)
+
+    return frames
+
+
+if __name__ == "__main__":
+    result = make_frames()
+    artifact_io.save(result, ROOT)
+    print(f"saved {result.shape[0]} RGB frames; exact first frame: "
+          f"{np.array_equal(result[0], np.asarray(Image.open(ROOT/'first_frame.png')))}")
