+"""Generate the requested 5 second RGB continuation from first_frame.png.
+
+The source photograph is retained as the locked-off plate.  The blue ball is
+extracted from that photograph, its old position is locally reconstructed,
+and its surface texture is rolled over a fixed lighting field.  All operations
+are deterministic, local pixel/geometry operations.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+WIDTH, HEIGHT = 512, 288
+NFRAMES, FPS = 120, 24
+
+
+def soft_disc_alpha(size=60, center=(29.5, 29.5), radius=21.4):
+    """Antialiased circular alpha matching the ball in the source frame."""
+    yy, xx = np.mgrid[:size, :size].astype(np.float32)
+    rr = np.hypot(xx - center[0], yy - center[1])
+    # The source has a soft, fuzzy tennis-ball edge about two pixels wide.
+    return np.clip((radius + 1.15 - rr) / 2.15, 0.0, 1.0)
+
+
+def reconstruct_plate(first):
+    """Remove the initial ball and its compact contact shadow.
+
+    Telea filling is constrained to a small disk wholly on the wooden table,
+    so the photographed glass and every other part of the scene remain exact.
+    A low-frequency correction makes the fill follow the table's local colour
+    gradient rather than looking like a flat painted circle.
+    """
+    bgr = cv2.cvtColor(first, cv2.COLOR_RGB2BGR)
+    mask = np.zeros((HEIGHT, WIDTH), np.uint8)
+    cv2.ellipse(mask, (164, 116), (25, 25), 0, 0, 360, 255, -1,
+                lineType=cv2.LINE_AA)
+    cv2.ellipse(mask, (166, 135), (25, 7), 0, 0, 360, 255, -1,
+                lineType=cv2.LINE_AA)
+    plate = cv2.inpaint(bgr, mask, 6.0, cv2.INPAINT_TELEA)
+    return cv2.cvtColor(plate, cv2.COLOR_BGR2RGB)
+
+
+def prepare_ball(first):
+    """Extract source pixels and split texture from stationary illumination."""
+    # A 60 px square centered at the measured centre (163.5, 115.5).
+    x0, y0, size = 134, 86, 60
+    source = first[y0:y0 + size, x0:x0 + size].astype(np.float32)
+    alpha = soft_disc_alpha(size)
+
+    # Estimate smooth illumination only from pixels inside the ball. Rotating
+    # the ratio field gives visible roll while keeping the highlight tied to
+    # the fixed room lighting.
+    weight = np.maximum(alpha, 0.001)
+    blur_weight = cv2.GaussianBlur(weight, (0, 0), 6.0)
+    smooth = np.empty_like(source)
+    for channel in range(3):
+        numerator = cv2.GaussianBlur(source[..., channel] * weight,
+                                     (0, 0), 6.0)
+        smooth[..., channel] = numerator / np.maximum(blur_weight, 0.01)
+    texture = np.clip(source / np.maximum(smooth, 8.0), 0.55, 1.55)
+    texture[alpha < 0.02] = 1.0
+    return alpha.astype(np.float32), smooth, texture
+
+
+def warp_texture(texture, angle_degrees):
+    size = texture.shape[0]
+    matrix = cv2.getRotationMatrix2D(((size - 1) / 2, (size - 1) / 2),
+                                     angle_degrees, 1.0)
+    return cv2.warpAffine(texture, matrix, (size, size),
+                          flags=cv2.INTER_CUBIC,
+                          borderMode=cv2.BORDER_CONSTANT,
+                          borderValue=(1.0, 1.0, 1.0))
+
+
+def subpixel_patch(patch, alpha, cx, cy):
+    """Return integer top-left and a subpixel translated patch/alpha pair."""
+    size = patch.shape[0]
+    left_f = cx - (size - 1) / 2
+    top_f = cy - (size - 1) / 2
+    left, top = int(np.floor(left_f)), int(np.floor(top_f))
+    dx, dy = left_f - left, top_f - top
+    matrix = np.float32([[1, 0, dx], [0, 1, dy]])
+    shifted_patch = cv2.warpAffine(patch, matrix, (size, size),
+                                   flags=cv2.INTER_CUBIC,
+                                   borderMode=cv2.BORDER_REPLICATE)
+    shifted_alpha = cv2.warpAffine(alpha, matrix, (size, size),
+                                   flags=cv2.INTER_CUBIC,
+                                   borderMode=cv2.BORDER_CONSTANT,
+                                   borderValue=0)
+    return left, top, shifted_patch, np.clip(shifted_alpha, 0, 1)
+
+
+def composite(frame, patch, alpha, left, top):
+    """Alpha composite a possibly clipped floating point RGB patch."""
+    ph, pw = alpha.shape
+    x1, y1 = max(0, left), max(0, top)
+    x2, y2 = min(WIDTH, left + pw), min(HEIGHT, top + ph)
+    if x1 >= x2 or y1 >= y2:
+        return
+    px1, py1 = x1 - left, y1 - top
+    px2, py2 = px1 + x2 - x1, py1 + y2 - y1
+    a = alpha[py1:py2, px1:px2, None]
+    roi = frame[y1:y2, x1:x2]
+    src = patch[py1:py2, px1:px2]
+    roi[:] = roi * (1.0 - a) + src * a
+
+
+def ellipse_overlay(frame, cx, cy, rx, ry, color, opacity, blur=2.0):
+    """Composite a soft analytic ellipse; used only for cast/contact shadows."""
+    margin = int(max(rx, ry) + 4 * blur + 2)
+    x1, x2 = max(0, int(cx) - margin), min(WIDTH, int(cx) + margin + 1)
+    y1, y2 = max(0, int(cy) - margin), min(HEIGHT, int(cy) + margin + 1)
+    if x1 >= x2 or y1 >= y2:
+        return
+    yy, xx = np.mgrid[y1:y2, x1:x2].astype(np.float32)
+    q = ((xx - cx) / rx) ** 2 + ((yy - cy) / ry) ** 2
+    a = (q <= 1.0).astype(np.float32)
+    if blur:
+        a = cv2.GaussianBlur(a, (0, 0), blur)
+    a = np.clip(a * opacity, 0, 1)[..., None]
+    roi = frame[y1:y2, x1:x2]
+    rgb = np.asarray(color, np.float32)
+    roi[:] = roi * (1.0 - a) + rgb * a
+
+
+def motion(frame_index):
+    """Perspective-aware, gently accelerating roll from table onto glass."""
+    p = frame_index / (NFRAMES - 1)
+    # A nonzero initial velocity avoids an artificial staged pause. The small
+    # quadratic term represents the slight acceleration on the glass.
+    distance = 374.0 * (0.79 * p + 0.21 * p * p)
+    x = 163.5 + distance
+
+    # The photographed tabletop is nearly level along the first portion. The
+    # glass's long edges rise about 0.16 image pixels per horizontal pixel.
+    if x <= 238.0:
+        y = 115.5 + 0.045 * (x - 163.5)
+    else:
+        y0 = 115.5 + 0.045 * (238.0 - 163.5)
+        y = y0 + 0.158 * (x - 238.0)
+
+    # A subpixel, rapidly damped settling response at the pane edge.
+    if x > 246.0:
+        after = (x - 246.0) / 2.6
+        y += 0.48 * np.exp(-after / 13.0) * np.sin(after * 0.92)
+    return x, y, distance
+
+
+def make_frames():
+    first = np.asarray(Image.open(ROOT / "first_frame.png").convert("RGB"),
+                       dtype=np.uint8)
+    assert first.shape == (HEIGHT, WIDTH, 3)
+    plate = reconstruct_plate(first)
+    ball_alpha, ball_light, ball_texture = prepare_ball(first)
+
+    frames = np.empty((NFRAMES, HEIGHT, WIDTH, 3), dtype=np.uint8)
+    frames[0] = first  # generation contract: exactly pixel-identical
+
+    for index in range(1, NFRAMES):
+        cx, cy, distance = motion(index)
+        frame = plate.astype(np.float32).copy()
+
+        # Transition from an opaque tabletop shadow to a fainter transmitted
+        # shadow on clear glass as the ball footprint crosses the pane edge.
+        glass_mix = float(np.clip((cx - 235.0) / 36.0, 0.0, 1.0))
+        shadow_y = cy + 18.8 + 5.2 * glass_mix
+        shadow_x = cx + 1.5 + 3.5 * glass_mix
+        ellipse_overlay(frame, shadow_x, shadow_y,
+                        19.5 - 2.0 * glass_mix,
+                        5.1 - 1.2 * glass_mix,
+                        (35, 38, 39), 0.25 - 0.11 * glass_mix, 2.2)
+        # The narrow contact shadow remains visible through the pane.
+        ellipse_overlay(frame, cx + 0.7, cy + 20.1,
+                        10.5, 2.1, (25, 28, 30),
+                        0.10 + 0.04 * glass_mix, 1.0)
+
+        # Clockwise rotation for rightward no-slip rolling.  Low frequency
+        # illumination stays fixed while the photographed felt texture turns.
+        angle = -np.degrees(distance / 21.4)
+        turned_texture = warp_texture(ball_texture, angle)
+        ball_rgb = np.clip(ball_light * turned_texture, 0, 255)
+
+        # A very restrained blue reflection helps read the otherwise clear
+        # glass without changing any of its fixed photographed edges.
+        if glass_mix > 0.01 and cx < WIDTH + 40:
+            reflected = cv2.resize(ball_rgb, (60, 15),
+                                   interpolation=cv2.INTER_AREA)
+            reflected_alpha = cv2.resize(ball_alpha, (60, 15),
+                                         interpolation=cv2.INTER_AREA)
+            reflected = reflected * np.array([0.86, 0.94, 1.08], np.float32)
+            reflected = cv2.GaussianBlur(reflected, (0, 0), 1.0,
+                                         sigmaY=0.7)
+            reflected_alpha = cv2.GaussianBlur(reflected_alpha, (0, 0), 1.2)
+            reflection_opacity = 0.075 * glass_mix
+            left = int(np.floor(cx - 29.5 + 3.0))
+            top = int(np.floor(cy + 24.0))
+            composite(frame, reflected, reflected_alpha * reflection_opacity,
+                      left, top)
+
+        left, top, shifted_rgb, shifted_alpha = subpixel_patch(
+            ball_rgb, ball_alpha, cx, cy)
+        composite(frame, shifted_rgb, shifted_alpha, left, top)
+        frames[index] = np.clip(np.rint(frame), 0, 255).astype(np.uint8)
+
+    return frames
+
+
+if __name__ == "__main__":
+    output = make_frames()
+    artifact_io.save(output, ROOT)
+    print(f"saved {len(output)} RGB frames at {WIDTH}x{HEIGHT}, {FPS} fps")
+
