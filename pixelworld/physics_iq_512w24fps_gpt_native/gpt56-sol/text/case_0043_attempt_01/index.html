def smoothstep(a, b, x):
    if b == a:
        return float(x >= b)
    z = np.clip((x-a)/(b-a), 0.0, 1.0)
    return float(z*z*(3.0-2.0*z))


def affine_stretch(amount):
    # Along-axis affine extension; points around the two jaws move amount px.
    k = amount / 124.0
    L = np.eye(2, dtype=np.float32) + k * np.outer(e, e)
    t = center - L @ center
    return np.column_stack([L, t]).astype(np.float32)


def add_translation(M, delta):
    Q = M.copy()
    Q[:, 2] += np.asarray(delta, np.float32)
    return Q


def warp_rgba(image, alpha, M):
    # Premultiplied-alpha resampling eliminates dark fringes on torn edges.
    premul = image * alpha[:, :, None]
    wi = cv2.warpAffine(premul, M, (W, H), flags=cv2.INTER_LINEAR,
                        borderMode=cv2.BORDER_CONSTANT, borderValue=0)
    wa = cv2.warpAffine(alpha.astype(np.float32), M, (W, H), flags=cv2.INTER_LINEAR,
                        borderMode=cv2.BORDER_CONSTANT, borderValue=0)
    return wi, np.clip(wa, 0, 1)

    return premul + canvas * (1.0 - alpha[:, :, None])
