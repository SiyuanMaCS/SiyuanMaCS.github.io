+"""Generate the requested locked-camera tennis-ball / rubber-duck demonstration.
+
+The animation is deliberately derived only from first_frame.png.  The two
+moving objects are segmented from that image, the small areas behind them are
+filled numerically, and deterministic 2-D rigid-body transforms create the
+motion.  No random or learned image generation is used.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+W, H = 512, 288
+N_FRAMES, FPS = 120, 24
+
+
+def grabcut_object(rgb, rect, iterations=8):
+    """Return a deterministic foreground mask initialized by a tight box."""
+    cv2.setRNGSeed(1701)
+    labels = np.zeros(rgb.shape[:2], np.uint8)
+    bg_model = np.zeros((1, 65), np.float64)
+    fg_model = np.zeros((1, 65), np.float64)
+    bgr = cv2.cvtColor(rgb, cv2.COLOR_RGB2BGR)
+    cv2.grabCut(
+        bgr, labels, rect, bg_model, fg_model, iterations, cv2.GC_INIT_WITH_RECT
+    )
+    return ((labels == cv2.GC_FGD) | (labels == cv2.GC_PR_FGD)).astype(np.uint8)
+
+
+def soft_object_alpha(mask):
+    """A narrowly feathered matte that retains all of the photographed edge."""
+    # GrabCut follows the already anti-aliased photographic boundary.  A very
+    # small feather prevents stair steps during sub-pixel affine movement.
+    a = cv2.GaussianBlur(mask.astype(np.float32), (0, 0), 0.38)
+    a[cv2.erode(mask, np.ones((3, 3), np.uint8), iterations=1) != 0] = 1.0
+    return np.clip(a, 0.0, 1.0)
+
+
+def dilate(mask, radius):
+    k = cv2.getStructuringElement(
+        cv2.MORPH_ELLIPSE, (radius * 2 + 1, radius * 2 + 1)
+    )
+    return cv2.dilate(mask.astype(np.uint8), k)
+
+
+def inpaint_background(rgb, mask):
+    # TELEA is well suited to these small holes in the gently textured table.
+    return cv2.inpaint(rgb, (mask * 255).astype(np.uint8), 5.0, cv2.INPAINT_TELEA)
+
+
+def over(dst, src_rgb, src_alpha):
+    """Alpha composite onto a float RGB destination, in place."""
+    a = np.clip(src_alpha, 0.0, 1.0)[..., None]
+    dst *= 1.0 - a
+    dst += src_rgb * a
+
+
+def add_shadow(frame, center, axes, opacity, sigma=3.0, angle=0.0):
+    """Place a soft neutral contact shadow while preserving table texture."""
+    m = np.zeros((H, W), np.float32)
+    c = (int(round(center[0])), int(round(center[1])))
+    ax = (max(1, int(round(axes[0]))), max(1, int(round(axes[1]))))
+    cv2.ellipse(m, c, ax, angle, 0, 360, 1.0, -1, cv2.LINE_AA)
+    m = cv2.GaussianBlur(m, (0, 0), sigma) * opacity
+    # A photographic shadow mostly attenuates light; a faint warm tint keeps it
+    # consistent with the beige wood already present in the source frame.
+    shadow_color = np.empty_like(frame)
+    shadow_color[..., 0] = 49.0
+    shadow_color[..., 1] = 44.0
+    shadow_color[..., 2] = 39.0
+    over(frame, shadow_color, m)
+
+
+def affine_sprite(frame, source_rgb, source_alpha, matrix):
+    """Warp a photographed sprite and composite it with premultiplied edges."""
+    flags = cv2.INTER_LANCZOS4
+    a = cv2.warpAffine(
+        source_alpha, matrix, (W, H), flags=flags, borderMode=cv2.BORDER_CONSTANT
+    )
+    # Warping premultiplied color prevents dark/color fringes at transparency.
+    premul = source_rgb.astype(np.float32) * source_alpha[..., None]
+    p = cv2.warpAffine(
+        premul, matrix, (W, H), flags=flags, borderMode=cv2.BORDER_CONSTANT
+    )
+    valid = a > 1.0e-4
+    straight = np.zeros_like(p)
+    straight[valid] = p[valid] / a[valid, None]
+    over(frame, straight, np.clip(a, 0.0, 1.0))
+
+
+def object_matrix(source_center, destination_center, angle_deg, sx=1.0, sy=1.0):
+    """Source-to-destination affine matrix, matching OpenCV's angle convention."""
+    th = np.deg2rad(angle_deg)
+    c, s = np.cos(th), np.sin(th)
+    # This is the anisotropic counterpart of cv2.getRotationMatrix2D.
+    linear = np.array([[c * sx, s * sy], [-s * sx, c * sy]], np.float32)
+    src = np.asarray(source_center, np.float32)
+    dst = np.asarray(destination_center, np.float32)
+    translation = dst - linear @ src
+    return np.column_stack((linear, translation)).astype(np.float32)
+
+
+def smoothstep(x):
+    x = float(np.clip(x, 0.0, 1.0))
+    return x * x * (3.0 - 2.0 * x)
+
+
+def build_frames(first):
+    # Tight, source-derived silhouettes.  The boxes intentionally exclude all
+    # other described objects and background furniture.
+    ball_mask = grabcut_object(first, (110, 82, 42, 45))
+    duck_mask = grabcut_object(first, (355, 82, 48, 58))
+    ball_alpha = soft_object_alpha(ball_mask)
+    duck_alpha = soft_object_alpha(duck_mask)
+
+    # Remove the ball and its original contact shadow.  The duck is retained
+    # exactly in this plate until the instant of impact.
+    ball_hole = dilate(ball_mask, 6)
+    cv2.ellipse(ball_hole, (135, 120), (23, 7), 0, 0, 360, 1, -1)
+    ball_plate = inpaint_background(first, ball_hole)
+
+    # A second plate is used after impact; its larger lower extension removes
+    # the duck's photographed resting shadow as well as the body itself.
+    duck_hole = dilate(duck_mask, 6)
+    cv2.ellipse(duck_hole, (387, 136), (31, 8), 0, 0, 360, 1, -1)
+    clean_plate = inpaint_background(ball_plate, duck_hole)
+
+    # Geometry measured directly from the masks in first_frame.png.
+    ball_center = np.array([133.5, 105.0], np.float32)
+    duck_pivot = np.array([399.0, 134.0], np.float32)  # right contact edge
+    radius = 14.2
+
+    # The projected table path is a straight line.  Its slight downward slope
+    # is the correct perspective line from the pipe mouth to the duck's flank.
+    hit_time = 52.0 / FPS
+    hit_x = 353.7
+    hit_y = 113.0
+    acceleration_tau = 0.18
+    terminal_speed = (hit_x - ball_center[0]) / (
+        hit_time - acceleration_tau * (1.0 - np.exp(-hit_time / acceleration_tau))
+    )
+
+    frames = np.empty((N_FRAMES, H, W, 3), np.uint8)
+    frames[0] = first  # contract: pixel-identical, with no processing at all
+
+    for index in range(1, N_FRAMES):
+        t = index / FPS
+        before_impact = t <= hit_time
+        frame = (ball_plate if before_impact else clean_plate).astype(np.float32)
+
+        if before_impact:
+            distance = terminal_speed * (
+                t - acceleration_tau * (1.0 - np.exp(-t / acceleration_tau))
+            )
+            distance = min(distance, hit_x - ball_center[0])
+            progress = distance / (hit_x - ball_center[0])
+            bx = ball_center[0] + distance
+            by = ball_center[1] + (hit_y - ball_center[1]) * progress
+            traveled = distance
+            # The two contact frames show a small, physically legible elastic
+            # flattening against the duck rather than overlap/interpenetration.
+            contact = np.exp(-0.5 * ((t - hit_time) / 0.055) ** 2)
+            sx = 1.0 - 0.095 * contact
+            sy = 1.0 + 0.055 * contact
+        else:
+            u = t - hit_time
+            # The tennis ball gives up most of its momentum, recoils a short
+            # distance, and makes one very low hop before friction stops it.
+            recoil = 8.7 * (1.0 - np.exp(-u / 0.22))
+            bx = hit_x - recoil
+            by = hit_y
+            if u < 0.34:
+                by -= 2.2 * np.sin(np.pi * u / 0.34)
+            traveled = (hit_x - ball_center[0]) - recoil
+            contact = np.exp(-0.5 * ((u - 0.005) / 0.055) ** 2)
+            sx = 1.0 - 0.095 * contact
+            sy = 1.0 + 0.055 * contact
+
+        # Contact shadow tightens during the tiny post-impact hop and remains
+        # locked to the projected table rather than to the center of the ball.
+        ball_lift = max(0.0, hit_y - by) if not before_impact else 0.0
+        add_shadow(
+            frame,
+            (bx - 1.6, by + radius - 0.3),
+            (13.0 + 0.35 * ball_lift, 2.3),
+            0.17 * (1.0 - 0.10 * ball_lift),
+            sigma=2.3 + 0.25 * ball_lift,
+        )
+
+        # Rolling right is clockwise in image coordinates (negative OpenCV
+        # angle).  During recoil the accumulated angle reverses accordingly.
+        ball_angle = -np.rad2deg(traveled / radius)
+        ball_m = object_matrix(ball_center, (bx, by), ball_angle, sx, sy)
+        affine_sprite(frame, first, ball_alpha, ball_m)
+
+        if not before_impact:
+            u = t - hit_time
+            # The impulse acts left of the duck's right-hand support point.
+            # Gravity then accelerates the duck clockwise onto its side.
+            fall = smoothstep(u / 0.62)
+            angle = -82.0 * fall
+            if u > 0.62:
+                settle = u - 0.62
+                angle += 5.3 * np.exp(-4.2 * settle) * np.sin(18.0 * settle)
+
+            slide = 8.0 * (1.0 - np.exp(-u / 0.48))
+            pivot_y = duck_pivot[1]
+            if 0.58 < u < 0.94:
+                # A sub-pixel damped landing bounce, with no camera motion.
+                q = u - 0.58
+                pivot_y -= 1.4 * np.exp(-6.0 * q) * abs(np.sin(18.0 * q))
+            pivot_dst = (duck_pivot[0] + slide, pivot_y)
+
+            # The shadow broadens and moves to the right as the body lies down.
+            shadow_x = duck_pivot[0] + slide + 18.0 * fall
+            add_shadow(
+                frame,
+                (shadow_x, duck_pivot[1] + 2.0),
+                (17.0 + 10.0 * fall, 3.0 + 1.1 * fall),
+                0.16,
+                sigma=3.0,
+                angle=-3.0,
+            )
+            duck_m = object_matrix(duck_pivot, pivot_dst, angle)
+            affine_sprite(frame, first, duck_alpha, duck_m)
+
+        frames[index] = np.clip(np.rint(frame), 0, 255).astype(np.uint8)
+
+    return frames
+
+
+def main():
+    case_dir = Path(__file__).resolve().parent
+    first = np.asarray(Image.open(case_dir / "first_frame.png").convert("RGB"))
+    if first.shape != (H, W, 3):
+        raise ValueError(f"Unexpected first frame shape: {first.shape}")
+    frames = build_frames(first)
+    assert frames.shape == (N_FRAMES, H, W, 3)
+    assert frames.dtype == np.uint8
+    assert np.array_equal(frames[0], first)
+    artifact_io.save(frames, case_dir)
+
+
+if __name__ == "__main__":
+    main()
