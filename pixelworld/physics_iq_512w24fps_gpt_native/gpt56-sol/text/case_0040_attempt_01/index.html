+"""Generate a fixed-camera, five-second tangerine slicing demonstration.
+
+The animation is deliberately derived from the supplied photograph.  The broad
+steel shape already entering from the upper part of the photograph is isolated,
+translated, and relit; the citrus response is made with masks computed from its
+actual orange pixels.  No learned assets or external renderers are used.
+"""
+from pathlib import Path
+import cv2
+import numpy as np
+from PIL import Image
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+W, H, N = 512, 288, 120
+
+
+def smoothstep(a, b, x):
+    if b == a:
+        return float(x >= b)
+    q = np.clip((x-a)/(b-a), 0.0, 1.0)
+    return float(q*q*(3.0-2.0*q))
+
+
+def feather(binary, radius=2.0):
+    """Subpixel-looking alpha which retains a solid, unmodified interior."""
+    b = (binary > 0).astype(np.uint8)
+    if radius <= 0:
+        return b.astype(np.float32)
+    inside = cv2.distanceTransform(b, cv2.DIST_L2, 3)
+    return np.clip(inside / radius, 0, 1).astype(np.float32)
+
+
+def translate(a, dx, dy, interp=cv2.INTER_LINEAR):
+    M = np.float32([[1, 0, dx], [0, 1, dy]])
+    return cv2.warpAffine(a, M, (W, H), flags=interp,
+                          borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+
+
+def alpha_over(dst, src, alpha):
+    aa = alpha[..., None].astype(np.float32)
+    return np.clip(dst.astype(np.float32)*(1-aa) + src.astype(np.float32)*aa,
+                   0, 255).astype(np.uint8)
+
+
+def main():
+    first = np.array(Image.open(ROOT/'first_frame.png').convert('RGB'))
+    assert first.shape == (H, W, 3)
+    yy, xx = np.indices((H, W), dtype=np.float32)
+
+    # The long diagonal steel region in the supplied frame.  Its lower-left
+    # boundary is the cutting edge, already poised at the right shoulder of
+    # the tangerine.  Keep its photographic texture rather than drawing a
+    # synthetic, flat-colored utensil.
+    blade_bin = np.zeros((H, W), np.uint8)
+    blade_poly = np.array([[244, 0], [326, 0], [435, 108],
+                           [383, 165], [354, 135]], np.int32)
+    cv2.fillConvexPoly(blade_bin, blade_poly, 255, lineType=cv2.LINE_8)
+    blade_alpha = feather(blade_bin, 1.8)
+
+    # Inpaint what is revealed behind the moving knife.  Only a narrow trailing
+    # part is normally seen; Telea therefore continues the local board grain
+    # and glass reflections convincingly.
+    bgr = cv2.cvtColor(first, cv2.COLOR_RGB2BGR)
+    behind = cv2.cvtColor(cv2.inpaint(bgr, blade_bin, 9, cv2.INPAINT_TELEA),
+                          cv2.COLOR_BGR2RGB)
+
+    # Fruit pixels are selected from the actual photograph, limited to the
+    # compact tangerine region so similarly colored warm background stays put.
+    hsv = cv2.cvtColor(first, cv2.COLOR_RGB2HSV)
+    citrus = ((hsv[..., 0] < 27) & (hsv[..., 1] > 72) &
+              (hsv[..., 2] > 56) & (xx > 276) & (xx < 400) &
+              (yy > 84) & (yy < 225)).astype(np.uint8)*255
+    citrus = cv2.morphologyEx(citrus, cv2.MORPH_CLOSE,
+                              cv2.getStructuringElement(cv2.MORPH_ELLIPSE,(9,9)))
+    citrus = cv2.morphologyEx(citrus, cv2.MORPH_DILATE,
+                              cv2.getStructuringElement(cv2.MORPH_ELLIPSE,(5,5)))
+    # Natural silhouette constraint, softly expanded beyond brightly saturated pulp.
+    silhouette = (((xx-337)/62)**2 + ((yy-158)/75)**2 < 1.0).astype(np.uint8)*255
+    citrus = cv2.bitwise_and(citrus, silhouette)
+    citrus_soft = cv2.GaussianBlur(citrus.astype(np.float32)/255, (0,0), 1.1)
+
+    # Final incision is parallel to the blade's edge and crosses the fruit's
+    # middle.  Signed distance in pixels (normalised for line slope).
+    # x = 197 + .80*y gives (293,120) -> (357,200).
+    signed = (xx - (197.0 + 0.80*yy)) / np.sqrt(1.0 + .80**2)
+
+    frames = []
+    for i in range(N):
+        if i == 0:
+            frames.append(first.copy())
+            continue
+
+        # 0.5 s observational hold, then a continuous deliberate cutting stroke.
+        stroke = smoothstep(11, 79, i)
+        # Tiny forward/back sawing, strongest while the blade is in the fruit.
+        saw_window = smoothstep(20, 31, i) * (1.0-smoothstep(68, 80, i))
+        saw = 2.2*np.sin((i-18)*0.47)*saw_window
+        distance = 56.0*stroke
+        # Down-and-left motion is perpendicular to the diagonal cutting edge;
+        # small along-edge sawing makes the action read as real knife work.
+        dx = -0.78*distance + 0.625*saw
+        dy =  0.625*distance + 0.78*saw
+
+        # Start from the untouched locked-off plate and restore only the area
+        # uncovered by the utensil. This avoids any global camera movement.
+        if stroke > 0:
+            base = first.astype(np.float32)*(1-blade_alpha[...,None]) + \
+                   behind.astype(np.float32)*blade_alpha[...,None]
+            frame = base.astype(np.uint8)
+        else:
+            frame = first.copy()
+
+        # The peel compresses slightly and the cut grows only after contact.
+        # Contact occurs when the moving edge reaches the right side, with the
+        # final incision reached around the middle of the stroke.
+        contact = smoothstep(28, 61, i)
+        passed = smoothstep(57, 87, i)
+        if contact > 0:
+            # Narrow amber trench. During contact it follows the moving edge;
+            # after passage it remains at the final cut line.
+            moving_offset = max(0.0, 49.0 - 1.28*distance)
+            sd_now = signed - moving_offset/np.sqrt(1+.8**2)
+            cut_sd = sd_now*(1-passed) + signed*passed
+            width = 0.55 + 2.15*contact
+            core = np.clip(1.0 - np.abs(cut_sd)/width, 0, 1)
+            core *= citrus_soft
+            # Wet, deep orange interior rather than an artificial black stroke.
+            trench = np.zeros_like(frame, np.float32)
+            trench[...,0] = 78 + 26*(yy/H)
+            trench[...,1] = 28 + 14*(yy/H)
+            trench[...,2] = 10
+            amount = core*(0.34 + .50*contact)
+            frame = alpha_over(frame, trench.astype(np.uint8), amount)
+
+            # Moist pulp rims on both sides catch the fixed upper light.
+            rim1 = np.exp(-((cut_sd-2.7)/1.05)**2)*citrus_soft*contact
+            rim2 = np.exp(-((cut_sd+3.0)/1.25)**2)*citrus_soft*contact
+            light = np.zeros_like(frame); light[:] = (248, 151, 63)
+            frame = alpha_over(frame, light, .30*rim1 + .17*rim2)
+
+            # A believable tiny opening after the edge has passed: opposite
+            # sides darken asymmetrically as their pulp relaxes.
+            relax = passed*citrus_soft*np.exp(-(signed/5.2)**2)
+            shade = np.zeros_like(frame); shade[:] = (95, 42, 18)
+            frame = alpha_over(frame, shade, .13*relax*(signed < 0))
+
+        # Soft contact shadow below the translated blade, strongest over fruit.
+        moved_mask = translate(blade_alpha, dx, dy)
+        shadow_mask = translate(blade_alpha, dx+3.5, dy+5.0)
+        shadow_ring = np.clip(shadow_mask - moved_mask, 0, 1)
+        shadow_ring *= (0.20 + 0.55*citrus_soft)
+        dark = np.zeros_like(frame); dark[:] = (39, 31, 27)
+        frame = alpha_over(frame, dark, .38*shadow_ring*stroke)
+
+        # Move the original blade pixels. Modest relighting gives its steel a
+        # directional brushed sheen but leaves photographic grain/reflections.
+        moved_blade = translate(first, dx, dy)
+        steel_pos = np.clip((xx - (244 + .80*yy) - dx + .80*dy)/82, 0, 1)
+        sheen = 7*np.sin(steel_pos*np.pi) - 3*steel_pos
+        moved_blade = np.clip(moved_blade.astype(np.float32) +
+                              sheen[...,None]*moved_mask[...,None], 0,255).astype(np.uint8)
+        frame = alpha_over(frame, moved_blade, moved_mask)
+
+        # Crisp bright bevel and dark micro-edge make the moving object
+        # unmistakably a sharpened knife. Clip to the translated knife body.
+        # Original edge: x=246+.806*y; transform the line with (dx,dy).
+        edge_sd = (xx-dx - (246.0 + .806*(yy-dy)))/np.sqrt(1+.806**2)
+        bevel = np.exp(-((edge_sd-1.0)/1.15)**2)*moved_mask
+        edge = np.exp(-((edge_sd+0.7)/0.85)**2)*moved_mask
+        hi = np.zeros_like(frame); hi[:] = (225, 230, 231)
+        lo = np.zeros_like(frame); lo[:] = (49, 48, 47)
+        frame = alpha_over(frame, hi, .34*bevel*stroke)
+        frame = alpha_over(frame, lo, .26*edge*stroke)
+
+        # Two minute juice highlights appear once the blade clears the incision.
+        if passed > 0:
+            juice = np.zeros((H,W), np.float32)
+            for cx,cy,rx,ry in [(326,183,2.2,1.0),(340,197,1.5,.75)]:
+                juice = np.maximum(juice, np.exp(-(((xx-cx)/rx)**2+((yy-cy)/ry)**2)*2.2))
+            juice *= passed*citrus_soft
+            wet = np.zeros_like(frame); wet[:] = (255, 193, 103)
+            frame = alpha_over(frame, wet, .48*juice)
+
+        frames.append(frame)
+
+    frames = np.stack(frames).astype(np.uint8)
+    assert np.array_equal(frames[0], first), 'frame zero changed'
+    assert frames.shape == (N,H,W,3)
+    artifact_io.save(frames, ROOT)
+
+
+if __name__ == '__main__':
+    main()

\"\"\"Generate a fixed-camera, five-second tangerine slicing demonstration.

The animation is deliberately derived from the supplied photograph.  The broad
steel shape already entering from the upper part of the photograph is isolated,
translated, and relit; the citrus response is made with masks computed from its
actual orange pixels.  No learned assets or external renderers are used.
\"\"\"
from pathlib import Path
import cv2
import numpy as np
from PIL import Image
import artifact_io

ROOT = Path(__file__).resolve().parent
W, H, N = 512, 288, 120


def smoothstep(a, b, x):
    if b == a:
        return float(x >= b)
    q = np.clip((x-a)/(b-a), 0.0, 1.0)
    return float(q*q*(3.0-2.0*q))


def feather(binary, radius=2.0):
    \"\"\"Subpixel-looking alpha which retains a solid, unmodified interior.\"\"\"
    b = (binary > 0).astype(np.uint8)
    if radius <= 0:
        return b.astype(np.float32)
    inside = cv2.distanceTransform(b, cv2.DIST_L2, 3)
    return np.clip(inside / radius, 0, 1).astype(np.float32)


def translate(a, dx, dy, interp=cv2.INTER_LINEAR):
    M = np.float32([[1, 0, dx], [0, 1, dy]])
    return cv2.warpAffine(a, M, (W, H), flags=interp,
                          borderMode=cv2.BORDER_CONSTANT, borderValue=0)


def alpha_over(dst, src, alpha):
    aa = alpha[..., None].astype(np.float32)
    return np.clip(dst.astype(np.float32)*(1-aa) + src.astype(np.float32)*aa,
                   0, 255).astype(np.uint8)


def main():
    first = np.array(Image.open(ROOT/'first_frame.png').convert('RGB'))
    assert first.shape == (H, W, 3)
    yy, xx = np.indices((H, W), dtype=np.float32)

    # The long diagonal steel region in the supplied frame.  Its lower-left
    # boundary is the cutting edge, already poised at the right shoulder of
    # the tangerine.  Keep its photographic texture rather than drawing a
    # synthetic, flat-colored utensil.
    blade_bin = np.zeros((H, W), np.uint8)
    blade_poly = np.array([[244, 0], [326, 0], [435, 108],
                           [383, 165], [354, 135]], np.int32)
    cv2.fillConvexPoly(blade_bin, blade_poly, 255, lineType=cv2.LINE_8)
    blade_alpha = feather(blade_bin, 1.8)

    # Inpaint what is revealed behind the moving knife.  Only a narrow trailing
    # part is normally seen; Telea therefore continues the local board grain
    # and glass reflections convincingly.
    bgr = cv2.cvtColor(first, cv2.COLOR_RGB2BGR)
    behind = cv2.cvtColor(cv2.inpaint(bgr, blade_bin, 9, cv2.INPAINT_TELEA),
                          cv2.COLOR_BGR2RGB)

    # Fruit pixels are selected from the actual photograph, limited to the
    # compact tangerine region so similarly colored warm background stays put.
    hsv = cv2.cvtColor(first, cv2.COLOR_RGB2HSV)
    citrus = ((hsv[..., 0] < 27) & (hsv[..., 1] > 72) &
              (hsv[..., 2] > 56) & (xx > 276) & (xx < 400) &
              (yy > 84) & (yy < 225)).astype(np.uint8)*255
    citrus = cv2.morphologyEx(citrus, cv2.MORPH_CLOSE,
                              cv2.getStructuringElement(cv2.MORPH_ELLIPSE,(9,9)))
    citrus = cv2.morphologyEx(citrus, cv2.MORPH_DILATE,
                              cv2.getStructuringElement(cv2.MORPH_ELLIPSE,(5,5)))
    # Natural silhouette constraint, softly expanded beyond brightly saturated pulp.
    silhouette = (((xx-337)/62)**2 + ((yy-158)/75)**2 < 1.0).astype(np.uint8)*255
    citrus = cv2.bitwise_and(citrus, silhouette)
    citrus_soft = cv2.GaussianBlur(citrus.astype(np.float32)/255, (0,0), 1.1)

    # Final incision is parallel to the blade's edge and crosses the fruit's
    # middle.  Signed distance in pixels (normalised for line slope).
    # x = 197 + .80*y gives (293,120) -> (357,200).
    signed = (xx - (197.0 + 0.80*yy)) / np.sqrt(1.0 + .80**2)

    frames = []
    for i in range(N):
        if i == 0:
            frames.append(first.copy())
            continue

        # 0.5 s observational hold, then a continuous deliberate cutting stroke.
        stroke = smoothstep(11, 79, i)
        # Tiny forward/back sawing, strongest while the blade is in the fruit.
        saw_window = smoothstep(20, 31, i) * (1.0-smoothstep(68, 80, i))
        saw = 2.2*np.sin((i-18)*0.47)*saw_window
        distance = 56.0*stroke
        # Down-and-left motion is perpendicular to the diagonal cutting edge;
        # small along-edge sawing makes the action read as real knife work.
        dx = -0.78*distance + 0.625*saw
        dy =  0.625*distance + 0.78*saw

        # Start from the untouched locked-off plate and restore only the area
        # uncovered by the utensil. This avoids any global camera movement.
        if stroke > 0:
            base = first.astype(np.float32)*(1-blade_alpha[...,None]) + \\
                   behind.astype(np.float32)*blade_alpha[...,None]
            frame = base.astype(np.uint8)
        else:
            frame = first.copy()

        # The peel compresses slightly and the cut grows only after contact.
        # Contact occurs when the moving edge reaches the right side, with the
        # final incision reached around the middle of the stroke.
        contact = smoothstep(28, 61, i)
        passed = smoothstep(57, 87, i)
        if contact > 0:
            # Narrow amber trench. During contact it follows the moving edge;
            # after passage it remains at the final cut line.
            moving_offset = max(0.0, 49.0 - 1.28*distance)
            sd_now = signed - moving_offset/np.sqrt(1+.8**2)
            cut_sd = sd_now*(1-passed) + signed*passed
            width = 0.55 + 2.15*contact
            core = np.clip(1.0 - np.abs(cut_sd)/width, 0, 1)
            core *= citrus_soft
            # Wet, deep orange interior rather than an artificial black stroke.
            trench = np.zeros_like(frame, np.float32)
            trench[...,0] = 78 + 26*(yy/H)
            trench[...,1] = 28 + 14*(yy/H)
            trench[...,2] = 10
            amount = core*(0.34 + .50*contact)
            frame = alpha_over(frame, trench.astype(np.uint8), amount)

            # Moist pulp rims on both sides catch the fixed upper light.
            rim1 = np.exp(-((cut_sd-2.7)/1.05)**2)*citrus_soft*contact
            rim2 = np.exp(-((cut_sd+3.0)/1.25)**2)*citrus_soft*contact
            light = np.zeros_like(frame); light[:] = (248, 151, 63)
            frame = alpha_over(frame, light, .30*rim1 + .17*rim2)

            # A believable tiny opening after the edge has passed: opposite
            # sides darken asymmetrically as their pulp relaxes.
            relax = passed*citrus_soft*np.exp(-(signed/5.2)**2)
            shade = np.zeros_like(frame); shade[:] = (95, 42, 18)
            frame = alpha_over(frame, shade, .13*relax*(signed < 0))

        # Soft contact shadow below the translated blade, strongest over fruit.
        moved_mask = translate(blade_alpha, dx, dy)
        shadow_mask = translate(blade_alpha, dx+3.5, dy+5.0)
        shadow_ring = np.clip(shadow_mask - moved_mask, 0, 1)
        shadow_ring *= (0.20 + 0.55*citrus_soft)
        dark = np.zeros_like(frame); dark[:] = (39, 31, 27)
        frame = alpha_over(frame, dark, .38*shadow_ring*stroke)

        # Move the original blade pixels. Modest relighting gives its steel a
        # directional brushed sheen but leaves photographic grain/reflections.
        moved_blade = translate(first, dx, dy)
        steel_pos = np.clip((xx - (244 + .80*yy) - dx + .80*dy)/82, 0, 1)
        sheen = 7*np.sin(steel_pos*np.pi) - 3*steel_pos
        moved_blade = np.clip(moved_blade.astype(np.float32) +
                              sheen[...,None]*moved_mask[...,None], 0,255).astype(np.uint8)
        frame = alpha_over(frame, moved_blade, moved_mask)

        # Crisp bright bevel and dark micro-edge make the moving object
        # unmistakably a sharpened knife. Clip to the translated knife body.
        # Original edge: x=246+.806*y; transform the line with (dx,dy).
        edge_sd = (xx-dx - (246.0 + .806*(yy-dy)))/np.sqrt(1+.806**2)
        bevel = np.exp(-((edge_sd-1.0)/1.15)**2)*moved_mask
        edge = np.exp(-((edge_sd+0.7)/0.85)**2)*moved_mask
        hi = np.zeros_like(frame); hi[:] = (225, 230, 231)
        lo = np.zeros_like(frame); lo[:] = (49, 48, 47)
        frame = alpha_over(frame, hi, .34*bevel*stroke)
        frame = alpha_over(frame, lo, .26*edge*stroke)

        # Two minute juice highlights appear once the blade clears the incision.
        if passed > 0:
            juice = np.zeros((H,W), np.float32)
            for cx,cy,rx,ry in [(326,183,2.2,1.0),(340,197,1.5,.75)]:
                juice = np.maximum(juice, np.exp(-(((xx-cx)/rx)**2+((yy-cy)/ry)**2)*2.2))
            juice *= passed*citrus_soft
            wet = np.zeros_like(frame); wet[:] = (255, 193, 103)
            frame = alpha_over(frame, wet, .48*juice)

        frames.append(frame)

    frames = np.stack(frames).astype(np.uint8)
    assert np.array_equal(frames[0], first), 'frame zero changed'
    assert frames.shape == (N,H,W,3)
    artifact_io.save(frames, ROOT)


if __name__ == '__main__':
    main()
