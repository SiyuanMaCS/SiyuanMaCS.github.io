+"""Generate the requested locked-camera tennis-ball collision video.
+
+The animation is constructed only from first_frame.png.  Object sprites retain
+photographic pixels from that frame; concealed table areas are reconstructed by
+local inpainting, and motion/shadows are composited at subpixel precision.
+"""

+import json

+import cv2

+import artifact_io

+def soft_ellipse(h, w, cx, cy, rx, ry, feather=1.25):
+    """Antialiased ellipse coverage in [0,1]."""
+    yy, xx = np.mgrid[:h, :w].astype(np.float32)
+    # Approximate signed distance in pixels for an ellipse.
+    q = np.sqrt(((xx - cx) / rx) ** 2 + ((yy - cy) / ry) ** 2)
+    px_scale = min(rx, ry)
+    return np.clip((1.0 - q) * px_scale / feather + 0.5, 0.0, 1.0)
+
+
+def hard_ellipse(h, w, cx, cy, rx, ry):
+    yy, xx = np.ogrid[:h, :w]
+    return ((((xx - cx) / rx) ** 2 + ((yy - cy) / ry) ** 2) <= 1.0).astype(np.uint8)
+
+
+def warp_premultiplied(src_rgb, alpha, matrix):
+    """Warp an RGB sprite as premultiplied color to avoid dark edge halos."""
+    h, w = alpha.shape
+    premul = src_rgb.astype(np.float32) * alpha[..., None]
+    wp = cv2.warpAffine(
+        premul, matrix, (w, h), flags=cv2.INTER_CUBIC,
+        borderMode=cv2.BORDER_CONSTANT, borderValue=0,
+    )
+    wa = cv2.warpAffine(
+        alpha, matrix, (w, h), flags=cv2.INTER_CUBIC,
+        borderMode=cv2.BORDER_CONSTANT, borderValue=0,
+    )
+    return wp, np.clip(wa, 0.0, 1.0)
+
+
+def composite_premul(dst, premul, alpha):
+    return dst * (1.0 - alpha[..., None]) + premul
+
+
+def translated_shadow(h, w, cx, cy, rx, ry, opacity):
+    """Soft contact shadow, with a small camera-right cast."""
+    m = hard_ellipse(h, w, cx + 2.2, cy, rx, ry).astype(np.float32)
+    m = cv2.GaussianBlur(m, (0, 0), sigmaX=2.5, sigmaY=1.35)
+    if m.max() > 0:
+        m /= m.max()
+    return np.clip(m * opacity, 0.0, 0.45)
+
+
+def ease_smooth(x):
+    x = np.clip(x, 0.0, 1.0)
+    return x * x * (3.0 - 2.0 * x)
+
+
+def build_frames(first):

+    src = first.astype(np.float32)

+    # viewing angle and the slightly fuzzy tennis-ball silhouettes.

+    # fuzz, color, and mutual occlusion as they translate after impact.

+    # into the inpaint result.

+        ps2 = translated_shadow(h, w, ox + pair_dx, oy + 16.1, 16.2, 3.7, 0.17)
+        shadow = np.clip(gs + ps1 + ps2, 0.0, 0.38)
+        canvas *= (1.0 - shadow[..., None])
+
+        # Draw target balls first.  Their shading remains tied to the room light
+        # while the photographic fuzz/occlusion boundary stays intact.
+        mp = np.array([[1.0, 0.0, pair_dx], [0.0, 1.0, 0.0]], np.float32)
+        pp, pa = warp_premultiplied(src, pair_a, mp)
+        canvas = composite_premul(canvas, pp, pa)

+        # A sphere rolling right turns clockwise.  Rotate its photographic
+        # texture by arc length / radius while translating its center.
+        angle_deg = -np.degrees(gray_dx / ((grx + gry) * 0.5))
+        mg = cv2.getRotationMatrix2D((gx, gy), angle_deg, 1.0).astype(np.float32)
+        mg[0, 2] += gray_dx
+        gp, ga = warp_premultiplied(src, gray_a, mg)
+        canvas = composite_premul(canvas, gp, ga)
+
+        frames.append(np.clip(np.rint(canvas), 0, 255).astype(np.uint8))
+
+    # Contract guarantee: do not rely on reconstruction for the first frame.
+    frames[0] = first.copy()
+    return np.stack(frames, axis=0)
+

+def write_preview_mp4(frames, fps, path):
+    """Create a convenient RGB-video preview in addition to contract outputs."""
+    h, w = frames.shape[1:3]
+    writer = cv2.VideoWriter(str(path), cv2.VideoWriter_fourcc(*"mp4v"), fps, (w, h))
+    if not writer.isOpened():
+        return False
+    for frame in frames:
+        writer.write(cv2.cvtColor(frame, cv2.COLOR_RGB2BGR))
+    writer.release()
+    return path.is_file() and path.stat().st_size > 0
+
+
+def main():
+    spec = json.loads((CASE_DIR / "task_spec.json").read_text())
+    first = np.asarray(Image.open(CASE_DIR / "first_frame.png").convert("RGB"), dtype=np.uint8)
+    assert first.shape == (spec["height"], spec["width"], 3)
+    frames = build_frames(first)
+    assert frames.shape == (spec["frames"], spec["height"], spec["width"], 3)
+    assert frames.dtype == np.uint8
+    assert np.array_equal(frames[0], first)
+
+    # Required PixelWorld artifacts.
+    artifact_io.save(frames, CASE_DIR)
+    # Optional standard video container for direct playback.
+    write_preview_mp4(frames, int(spec["fps"]), CASE_DIR / "output.mp4")
+
+
+if __name__ == "__main__":
+    main()

+def rolling_ball(src_low, src_detail, alpha, cx, cy, dx, radius):
+    """Move a ball while rotating detail but keeping illumination world-fixed.
+
+    Rotating an entire photographic crop would make its highlight rotate with
+    the ball.  Instead the broad, low-frequency shading is only translated,
+    while felt/seam detail rotates by arc length, as it does on a real sphere.
+    """
+    h, w = alpha.shape
+    move = np.array([[1.0, 0.0, dx], [0.0, 1.0, 0.0]], np.float32)
+    base, moved_alpha = warp_premultiplied(src_low, alpha, move)
+    angle_deg = -np.degrees(dx / radius)
+    spin = cv2.getRotationMatrix2D((cx, cy), angle_deg, 1.0).astype(np.float32)
+    spin[0, 2] += dx
+    detail, _ = warp_premultiplied(src_detail, alpha, spin)
+    return base + detail, moved_alpha

#"'!/usr/bin/env python3
"""Generate the requested locked-camera tennis-ball collision video.

The animation is constructed only from first_frame.png.  Object sprites retain
photographic pixels from that frame; concealed table areas are reconstructed by
local inpainting, and motion/shadows are composited at subpixel precision.
"""
from pathlib import Path
import json

import cv2
import numpy as np
from PIL import Image

import artifact_io

CASE_DIR = Path(__file__).resolve().parent


def soft_ellipse(h, w, cx, cy, rx, ry, feather=1.25):
    """Antialiased ellipse coverage in [0,1]."""
    yy, xx = np.mgrid[:h, :w].astype(np.float32)
    # Approximate signed distance in pixels for an ellipse.
    q = np.sqrt(((xx - cx) / rx) ** 2 + ((yy - cy) / ry) ** 2)
    px_scale = min(rx, ry)
    return np.clip((1.0 - q) * px_scale / feather + 0.5, 0.0, 1.0)


def hard_ellipse(h, w, cx, cy, rx, ry):
    yy, xx = np.ogrid[:h, :w]
    return ((((xx - cx) / rx) ** 2 + ((yy - cy) / ry) ** 2) <= 1.0).astype(np.uint8)


def warp_premultiplied(src_rgb, alpha, matrix):
    """Warp an RGB sprite as premultiplied color to avoid dark edge halos."""
    h, w = alpha.shape
    premul = src_rgb.astype(np.float32) * alpha[..., None]
    wp = cv2.warpAffine(
        premul, matrix, (w, h), flags=cv2.INTER_CUBIC,
        borderMode=cv2.BORDER_CONSTANT, borderValue=0,
    )
    wa = cv2.warpAffine(
        alpha, matrix, (w, h), flags=cv2.INTER_CUBIC,
        borderMode=cv2.BORDER_CONSTANT, borderValue=0,
    )
    return wp, np.clip(wa, 0.0, 1.0)


def composite_premul(dst, premul, alpha):
    return dst * (1.0 - alpha[..., None]) + premul


def translated_shadow(h, w, cx, cy, rx, ry, opacity):
    """Soft contact shadow, with a small camera-right cast."""
    m = hard_ellipse(h, w, cx + 2.2, cy, rx, ry).astype(np.float32)
    m = cv2.GaussianBlur(m, (0, 0), sigmaX=2.5, sigmaY=1.35)
    if m.max() > 0:
        m /= m.max()
    return np.clip(m * opacity, 0.0, 0.45)


def ease_smooth(x):
    x = np.clip(x, 0.0, 1.0)
    return x * x * (3.0 - 2.0 * x)


def build_frames(first):
    h, w = first.shape[:2]
    src = first.astype(np.float32)

    # Measured from the supplied frame.  Ellipticity accounts for the shallow
    # viewing angle and the slightly fuzzy tennis-ball silhouettes.
    gx, gy, grx, gry = 176.5, 104.5, 17.5, 17.8
    bx, by, brx, bry = 243.0, 110.2, 17.2, 16.8
    ox, oy, orx, ory = 271.0, 113.2, 17.1, 16.9

    gray_a = soft_ellipse(h, w, gx, gy, grx, gry, 1.15).astype(np.float32)
    # The two target balls are already touching/occluding one another.  Keeping
    # their union as one photographic sprite preserves their exact boundary,
    # fuzz, color, and mutual occlusion as they translate after impact.
    pair_a = np.maximum(
        soft_ellipse(h, w, bx, by, brx, bry, 1.15),
        soft_ellipse(h, w, ox, oy, orx, ory, 1.15),
    ).astype(np.float32)

    # Remove balls and their contact shadows to recover a clean table plate.
    # A generous binary margin prevents dark/orange edge colors from bleeding
    # into the inpaint result.
    remove = np.zeros((h, w), np.uint8)
    remove |= hard_ellipse(h, w, gx, gy, grx + 4.5, gry + 4.2)
    remove |= hard_ellipse(h, w, bx, by, brx + 4.4, bry + 4.2)
    remove |= hard_ellipse(h, w, ox, oy, orx + 4.4, ory + 4.2)
    remove |= hard_ellipse(h, w, gx + 2, gy + 17, 21, 6)
    remove |= hard_ellipse(h, w, (bx + ox) / 2 + 2, 129.0, 38, 7)
    remove = cv2.dilate(remove, np.ones((3, 3), np.uint8), iterations=1)
    plate = cv2.inpaint(first, remove * 255, 6.0, cv2.INPAINT_TELEA).astype(np.float32)

    # A very light local smoothing only inside the concealed region suppresses
    # radial inpaint streaks while preserving all untouched source pixels.
    smooth = cv2.GaussianBlur(plate, (0, 0), 1.15)
    inner = cv2.GaussianBlur(remove.astype(np.float32), (0, 0), 1.0)[..., None]
    plate = plate * (1.0 - inner) + smooth * inner

    frames = []
    for f in range(120):
        # Exact still lead-in makes frame 00 (and the initial setup) faithful.
        if f < 8:
            frames.append(first.copy())
            continue

        # Grey ball: near-constant-speed roll from the pipe direction, followed
        # by a tiny compression/settling motion at collision.
        if f <= 28:
            p = (f - 8) / 20.0
            gray_dx = 34.2 * np.clip(p, 0.0, 1.0)
        else:
            # It advances only slightly after contact, then settles.
            gray_dx = 34.2 + 2.0 * ease_smooth((f - 28) / 8.0)

        # Target pair starts moving on the first frame after impact.  Constant
        # deceleration models rolling friction and brings it almost to rest by
        # the end of the five-second shot.
        t = max(0.0, float(f - 28))
        pair_dx = max(0.0, 1.30 * t - 0.0070 * t * t)

        canvas = plate.copy()

        # Shadows are cast before the sprites.  Their fixed offsets and scale
        # reinforce a locked camera and continuous tabletop contact.
        gs = translated_shadow(h, w, gx + gray_dx, gy + 17.3, 17.0, 3.6, 0.18)
        ps1 = translated_shadow(h, w, bx + pair_dx, by + 16.4, 16.2, 3.7, 0.19)
        ps2 = translated_shadow(h, w, ox + pair_dx, oy + 16.1, 16.2, 3.7, 0.17)
        shadow = np.clip(gs + ps1 + ps2, 0.0, 0.38)
        canvas *= (1.0 - shadow[..., None])

        # Draw target balls first.  Their shading remains tied to the room light
        # while the photographic fuzz/occlusion boundary stays intact.
        mp = np.array([[1.0, 0.0, pair_dx], [0.0, 1.0, 0.0]], np.float32)
        pp, pa = warp_premultiplied(src, pair_a, mp)
        canvas = composite_premul(canvas, pp, pa)

        # A sphere rolling right turns clockwise.  Rotate its photographic
        # texture by arc length / radius while translating its center.
        angle_deg = -np.degrees(gray_dx / ((grx + gry) * 0.5))
        mg = cv2.getRotationMatrix2D((gx, gy), angle_deg, 1.0).astype(np.float32)
        mg[0, 2] += gray_dx
        gp, ga = warp_premultiplied(src, gray_a, mg)
        canvas = composite_premul(canvas, gp, ga)

        frames.append(np.clip(np.rint(canvas), 0, 255).astype(np.uint8))

    # Contract guarantee: do not rely on reconstruction for the first frame.
    frames[0] = first.copy()
    return np.stack(frames, axis=0)


def write_preview_mp4(frames, fps, path):
    """Create a convenient RGB-video preview in addition to contract outputs."""
    h, w = frames.shape[1:3]
    writer = cv2.VideoWriter(str(path), cv2.VideoWriter_fourcc(*"mp4v"), fps, (w, h))
    if not writer.isOpened():
        return False
    for frame in frames:
        writer.write(cv2.cvtColor(frame, cv2.COLOR_RGB2BGR))
    writer.release()
    return path.is_file() and path.stat().st_size > 0


def main():
    spec = json.loads((CASE_DIR / "task_spec.json").read_text())
    first = np.asarray(Image.open(CASE_DIR / "first_frame.png").convert("RGB"), dtype=np.uint8)
    assert first.shape == (spec["height"], spec["width"], 3)
    frames = build_frames(first)
    assert frames.shape == (spec["frames"], spec["height"], spec["width"], 3)
    assert frames.dtype == np.uint8
    assert np.array_equal(frames[0], first)

    # Required PixelWorld artifacts.
    artifact_io.save(frames, CASE_DIR)
    # Optional standard video container for direct playback.
    write_preview_mp4(frames, int(spec["fps"]), CASE_DIR / "output.mp4")


if __name__ == "__main__":
    main()
