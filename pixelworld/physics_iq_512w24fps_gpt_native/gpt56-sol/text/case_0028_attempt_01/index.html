    base = np.array([126., 132., 130.], np.float32)
    rgb = base[None,None,:] * br[...,None]
    rgb = rgb*(1-seam[...,None]*.42) + np.array([178.,181.,174.])[None,None,:]*(seam[...,None]*.42)

    spec = np.exp(-(((x-5.0)/3.0)**2 + ((y+5.8)/2.7)**2))*0.17
    rgb += 255*spec[...,None]

    premul = rgb * alpha[...,None]

    return np.dstack([np.clip(rgb_small,0,255), np.clip(a*255,0,255)]).astype(np.uint8)


def place_rgba(layer_rgb, layer_a, sprite, cx, cy, weight=1.0, clip_plane=None):
    \"\"\"Accumulate premultiplied RGBA into a full-frame motion layer.\"\"\"
    sh, sw = sprite.shape[:2]
    x0f, y0f = cx-sw/2, cy-sh/2
    x0, y0 = math.floor(x0f), math.floor(y0f)
    fx, fy = x0f-x0, y0f-y0
    # subpixel translation within the sprite
    M = np.float32([[1,0,fx],[0,1,fy]])
    spr = cv2.warpAffine(sprite, M, (sw,sh), flags=cv2.INTER_LINEAR,
                         borderMode=cv2.BORDER_CONSTANT, borderValue=(0,0,0,0))
    x1,y1,x2,y2 = max(0,x0),max(0,y0),min(W,x0+sw),min(H,y0+sh)
    if x2<=x1 or y2<=y1: return
    q = spr[y1-y0:y2-y0, x1-x0:x2-x0].astype(np.float32)
    a = (q[...,3]/255.0)*weight
    if clip_plane is not None:
        # Only the portion beyond the pipe's mouth is visible while emerging.
        px, py, ux, uy = clip_plane
        gy,gx = np.mgrid[y1:y2,x1:x2]
        edge = (gx-px)*ux + (gy-py)*uy
        a *= np.clip(edge/1.15 + .55, 0, 1)
    layer_rgb[y1:y2,x1:x2] += q[...,:3] * a[...,None]
    layer_a[y1:y2,x1:x2] += a


def composite_accum(frame, rgb_sum, a_sum):
    a = np.clip(a_sum,0,1)
    # Samples use weights summing to one; overlap is an exposure-weighted average.
    color = rgb_sum / np.maximum(a_sum[...,None],1e-6)
    return alpha_over(frame, color, a)
