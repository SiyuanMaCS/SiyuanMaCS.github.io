+CASE_DIR = Path(__file__).resolve().parent

+src = np.asarray(Image.open(CASE_DIR / "first_frame.png").convert("RGB"), dtype=np.uint8)
+assert src.shape == (H, W, 3)

+# The supplied photograph contains the released tennis ball at this ellipse.
+BALL_CX, BALL_CY = 237.0, 123.0
+RX, RY = 30.8, 33.0
+Y, X = np.mgrid[0:H, 0:W]
+ell = np.sqrt(((X - BALL_CX) / RX) ** 2 + ((Y - BALL_CY) / RY) ** 2)
+# Feather only the final 1-2 pixels, preserving all photographed ball texture.
+ball_alpha_full = np.clip((1.025 - ell) / 0.055, 0.0, 1.0).astype(np.float32)
+
+# Remove the source ball numerically. Telea inpainting draws only from the
+# surrounding photographed wall/table and leaves the grabber and sand intact.
+remove_mask = (ell < 1.075).astype(np.uint8) * 255
+clean = cv2.inpaint(src, remove_mask, 7.0, cv2.INPAINT_TELEA)
+
+# Crop a compact RGBA sprite from the actual ball, retaining real texture,
+# lighting, seam, edge antialiasing and source-camera resolution.
+x0, x1, y0, y1 = 201, 274, 86, 161

+ball_a = ball_alpha_full[y0:y1, x0:x1]

+# A fixed pool of cohesive kinetic-sand crumbs. Values are sampled from the
+# photographed mound, so no foreign colors are introduced.

+particles = []
+for i in range(22):
+    side = -1 if i % 2 == 0 else 1
+    speed = rng.uniform(0.75, 2.15)
+    vx = side * speed + rng.uniform(-0.18, 0.18)
+    vy = rng.uniform(-3.25, -1.15)
+    delay = int(rng.integers(0, 4))
+    life = int(rng.integers(10, 20))
+    rad = rng.uniform(0.7, 1.65)
+    sy = int(rng.integers(192, 238))
+    sx = int(rng.integers(180, 310))
+    col = src[sy, sx].astype(np.float32)
+    # Guarantee a green sand sample if random texture lands in a dark crevice.
+    if not (col[1] > col[0] * 1.35 and col[1] > col[2] * 1.15):
+        col = np.array([27, 101 + rng.integers(-9, 10), 36], np.float32)
+    particles.append((vx, vy, delay, life, rad, col))
+

+def smoothstep(a, b, x):
+    if a == b:
+        return np.asarray(x >= b, dtype=np.float32)
+    t = np.clip((x - a) / (b - a), 0.0, 1.0)
+    return t * t * (3.0 - 2.0 * t)

+def ball_state(f):
+    """Position follows release, free fall, strongly damped sand penetration."""
+    if f <= 5:
+        return BALL_CX, BALL_CY, 0.0, 1.0, 0.0
+    if f <= 17:
+        u = (f - 5) / 12.0
+        # Constant acceleration in image space.
+        return BALL_CX, BALL_CY + 27.0 * u * u, 24.0 * u, 1.0, 0.0
+    if f <= 28:
+        u = (f - 17) / 11.0
+        ease = 1.0 - (1.0 - u) ** 3
+        squash = 1.0 - 0.055 * np.sin(np.pi * np.clip(u / 0.62, 0, 1))
+        return BALL_CX + 2.0 * ease, 150.0 + 37.0 * ease, 24.0 + 17.0 * ease, squash, ease
+    if f <= 42:
+        u = (f - 28) / 14.0
+        # Very small, quickly extinguished rebound in cohesive kinetic sand.
+        bob = -2.2 * np.sin(np.pi * u) * (1.0 - u)
+        return 239.0, 187.0 + bob, 41.0 + 3.0 * u, 1.0, 1.0
+    return 239.0, 187.0, 44.0, 1.0, 1.0
+
+
+def composite_sprite(frame, cx, cy, angle, yscale):
+    h, w = ball_rgb.shape[:2]
+    # Rotate the real photographed texture as the ball falls. Slight impact
+    # flattening is centered, not anchored, avoiding a visible jump.
+    M = cv2.getRotationMatrix2D(local_center, angle, 1.0)
+    M[1, 0] *= yscale
+    M[1, 1] *= yscale
+    M[1, 2] = local_center[1] - yscale * (local_center[1] - M[1, 2])
+    rgb = cv2.warpAffine(ball_rgb, M, (w, h), flags=cv2.INTER_CUBIC,
+                         borderMode=cv2.BORDER_REFLECT_101)
+    a = cv2.warpAffine(ball_a, M, (w, h), flags=cv2.INTER_LINEAR,
+                       borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+    dx = int(round(cx - BALL_CX))
+    dy = int(round(cy - BALL_CY))
+    ox0, oy0 = x0 + dx, y0 + dy
+    ox1, oy1 = ox0 + w, oy0 + h
+    # Crop safely (all actual states remain on-screen).
+    tx0, ty0 = max(0, ox0), max(0, oy0)
+    tx1, ty1 = min(W, ox1), min(H, oy1)
+    sx0, sy0 = tx0 - ox0, ty0 - oy0
+    sx1, sy1 = sx0 + tx1 - tx0, sy0 + ty1 - ty0
+    aa = a[sy0:sy1, sx0:sx1, None]
+    dst = frame[ty0:ty1, tx0:tx1].astype(np.float32)
+    frame[ty0:ty1, tx0:tx1] = np.clip(
+        dst * (1.0 - aa) + rgb[sy0:sy1, sx0:sx1] * aa, 0, 255
+    ).astype(np.uint8)
+
+
+def add_contact_shadow(frame, cy, f):
+    # Fixed light: the soft shadow stays on the mound while the ball approaches.
+    yy, xx = np.mgrid[0:H, 0:W]
+    proximity = smoothstep(126.0, 153.0, cy)
+    if proximity <= 0:
+        return
+    sigx = 18.0 + 7.0 * (1.0 - proximity)
+    sigy = 3.3 + 2.0 * (1.0 - proximity)
+    sh = np.exp(-0.5 * (((xx - 238.0) / sigx) ** 2 + ((yy - 180.0) / sigy) ** 2))
+    # Never shadow the distant wall; restrict to tabletop/mound level.
+    sh *= smoothstep(169, 180, yy)
+    amount = sh[..., None] * (0.055 + 0.12 * proximity)
+    frame[:] = np.clip(frame.astype(np.float32) * (1.0 - amount), 0, 255).astype(np.uint8)
+
+
+def add_sand_occlusion(frame, cx, penetration):
+    """Place photographed sand in front of the lower ball and form a crater."""
+    if penetration <= 0:
+        return
+    yy, xx = np.mgrid[0:H, 0:W]
+    dx = xx - cx
+    # Undisturbed mound is low at center (apex) and slopes down at the sides.
+    original_surface = 179.0 + 0.0060 * dx * dx
+    # Impact yields a bowl: depressed center, slightly raised cohesive lips.
+    crater_surface = 201.0 - 0.0135 * dx * dx
+    surface = original_surface * (1.0 - penetration) + crater_surface * penetration
+    near = np.abs(dx) < 35.0
+    # Feathered foreground edge; constrain to pixels where photographed mound
+    # texture exists, except for a narrow raised lip made from adjacent texture.
+    alpha = np.clip((yy - surface + 0.9) / 2.0, 0.0, 1.0) * near
+    green = (src[..., 1] > src[..., 0] * 1.25) & (src[..., 1] > src[..., 2] * 1.08)
+    alpha *= cv2.GaussianBlur(green.astype(np.float32), (0, 0), 1.0)
+    # Occlusion is needed only around the ball, not across the whole mound.
+    alpha *= (yy < 225) & (yy > 172)
+    a = alpha[..., None]
+    frame[:] = np.clip(frame.astype(np.float32) * (1.0 - a) + src.astype(np.float32) * a,
+                       0, 255).astype(np.uint8)
+
+    # A narrow shaded inner rim communicates indentation without synthetic lines.
+    rim = np.exp(-0.5 * ((yy - surface) / 1.45) ** 2) * near * penetration
+    rim *= cv2.GaussianBlur(green.astype(np.float32), (0, 0), 0.8)
+    darken = (0.13 * rim)[..., None]
+    frame[:] = np.clip(frame.astype(np.float32) * (1.0 - darken), 0, 255).astype(np.uint8)
+
+
+def add_particles(frame, f):
+    # Contact is frame 17. Kinetic sand sheds a few short-lived cohesive crumbs,
+    # much less spray than dry sand would produce.
+    tbase = f - 17
+    if tbase < 0 or tbase > 25:
+        return
+    overlay = frame.copy()
+    for vx, vy, delay, life, rad, col in particles:
+        t = tbase - delay
+        if t < 0 or t > life:
+            continue
+        px = 238.0 + vx * t
+        py = 181.0 + vy * t + 0.255 * t * t
+        if not (1 <= px < W - 1 and 1 <= py < H - 1):
+            continue
+        fade = np.clip((life - t) / 4.0, 0.0, 1.0)
+        c = tuple(int(v) for v in np.clip(col * (0.82 + 0.18 * fade), 0, 255))
+        cv2.circle(overlay, (int(round(px)), int(round(py))), max(1, int(round(rad))),
+                   c, -1, lineType=cv2.LINE_AA)
+    # Tiny motion-integrated blend keeps one-pixel grains photographic.
+    frame[:] = cv2.addWeighted(overlay, 0.92, frame, 0.08, 0)
+
+
+frames = np.empty((N, H, W, 3), dtype=np.uint8)
+for f in range(N):
+    # Preserve the supplied first frame exactly (and a short natural release hold).
+    if f <= 5:
+        frames[f] = src
+        continue
+    frame = clean.copy()
+    cx, cy, angle, yscale, penetration = ball_state(f)
+    add_contact_shadow(frame, cy, f)
+    composite_sprite(frame, cx, cy, angle, yscale)
+    add_sand_occlusion(frame, cx, penetration)
+    add_particles(frame, f)
+    frames[f] = frame
+
+assert frames.shape == (120, 288, 512, 3)
+assert frames.dtype == np.uint8
+assert np.array_equal(frames[0], src), "Frame 00 must match first_frame.png exactly"
+artifact_io.save(frames, CASE_DIR)
+print("saved", frames.shape, frames.dtype, "frame0_exact=", np.array_equal(frames[0], src))

                         borderMode=cv2.BORDER_REFLECT_101)

    tx1, ty1 = min(W, ox1), min(H, oy1)
    sx0, sy0 = tx0 - ox0, ty0 - oy0
    sx1, sy1 = sx0 + tx1 - tx0, sy0 + ty1 - ty0

def add_contact_shadow(frame, cy, f):
    # Fixed light: the soft shadow stays on the mound while the ball approaches.
    yy, xx = np.mgrid[0:H, 0:W]
    proximity = smoothstep(126.0, 153.0, cy)
    if proximity <= 0:
        return
    sigx = 18.0 + 7.0 * (1.0 - proximity)
    sigy = 3.3 + 2.0 * (1.0 - proximity)
    sh = np.exp(-0.5 * (((xx - 238.0) / sigx) ** 2 + ((yy - 180.0) / sigy) ** 2))
    # Never shadow the distant wall; restrict to tabletop/mound level.
    sh *= smoothstep(169, 180, yy)
    amount = sh[..., None] * (0.055 + 0.12 * proximity)
    frame[:] = np.clip(frame.astype(np.float32) * (1.0 - amount), 0, 255).astype(np.uint8)


def add_sand_occlusion(frame, cx, penetration):
    \"\"\"Place photographed sand in front of the lower ball and form a crater.\"\"\"
    if penetration <= 0:
        return
    yy, xx = np.mgrid[0:H, 0:W]

    # Undisturbed mound is low at center (apex) and slopes down at the sides.
    original_surface = 179.0 + 0.0060 * dx * dx
    # Impact yields a bowl: depressed center, slightly raised cohesive lips.
    crater_surface = 201.0 - 0.0135 * dx * dx
    surface = original_surface * (1.0 - penetration) + crater_surface * penetration
    near = np.abs(dx) < 35.0
    # Feathered foreground edge; constrain to pixels where photographed mound
    # texture exists, except for a narrow raised lip made from adjacent texture.
    alpha = np.clip((yy - surface + 0.9) / 2.0, 0.0, 1.0) * near
    green = (src[..., 1] > src[..., 0] * 1.25) & (src[..., 1] > src[..., 2] * 1.08)
    alpha *= cv2.GaussianBlur(green.astype(np.float32), (0, 0), 1.0)
    # Occlusion is needed only around the ball, not across the whole mound.
    alpha *= (yy < 225) & (yy > 172)
    a = alpha[..., None]
    frame[:] = np.clip(frame.astype(np.float32) * (1.0 - a) + src.astype(np.float32) * a,
                       0, 255).astype(np.uint8)

    rim = np.exp(-0.5 * ((yy - surface) / 1.45) ** 2) * near * penetration
    rim *= cv2.GaussianBlur(green.astype(np.float32), (0, 0), 0.8)
    darken = (0.13 * rim)[..., None]
