+#!/usr/bin/env python3
+"""Generate the requested fixed-camera ball-on-glass demonstration.
+
+The source photograph is treated as the locked background.  The blue ball is
+segmented numerically, its hidden patch of table is reconstructed locally, and
+its fine surface texture is rotated while its image-space illumination remains
+fixed.  This gives rolling (rather than sliding) without making the specular
+lighting rotate with the object.
+"""
+from pathlib import Path
+import json
+import numpy as np
+import cv2
+from PIL import Image
+import artifact_io
+
+HERE = Path(__file__).resolve().parent
+
+
+def smoothstep(a, b, x):
+    q = np.clip((x-a)/(b-a), 0.0, 1.0)
+    return q*q*(3.0-2.0*q)
+
+
+def extract_scene(src):
+    """Return clean plate, matte, foreground ball, and source centre."""
+    h, w = src.shape[:2]
+    # The only saturated-blue connected object in the supplied setup is the ball.
+    rgb = src.astype(np.int16)
+    chroma = rgb[..., 2] - (rgb[..., 0] + rgb[..., 1]) / 2.0
+    seed = ((rgb[..., 2] - rgb[..., 0] > 25) &
+            (rgb[..., 2] - rgb[..., 1] > 8) & (chroma > 18)).astype(np.uint8)
+    n, labels, stats, centers = cv2.connectedComponentsWithStats(seed, 8)
+    candidates = [i for i in range(1, n) if 300 < stats[i, cv2.CC_STAT_AREA] < 5000]
+    if not candidates:
+        raise RuntimeError("Could not locate blue tennis ball")
+    component = max(candidates, key=lambda i: stats[i, cv2.CC_STAT_AREA])
+    hard_seed = (labels == component).astype(np.uint8)
+    cx0, cy0 = centers[component]
+
+    # GrabCut recovers dark blue felt near the silhouette that strict chroma misses.
+    yy, xx = np.mgrid[:h, :w]
+    dist = np.hypot(xx-cx0, yy-cy0)
+    gc = np.full((h, w), cv2.GC_BGD, np.uint8)
+    gc[(dist < 26)] = cv2.GC_PR_BGD
+    gc[(dist < 22)] = cv2.GC_PR_FGD
+    gc[hard_seed.astype(bool)] = cv2.GC_FGD
+    bg_model = np.zeros((1, 65), np.float64)
+    fg_model = np.zeros((1, 65), np.float64)
+    cv2.grabCut(src, gc, None, bg_model, fg_model, 7, cv2.GC_INIT_WITH_MASK)
+    obj = ((gc == cv2.GC_FGD) | (gc == cv2.GC_PR_FGD)).astype(np.uint8)
+    # Keep only the component containing the blue seed.
+    n2, lab2, st2, _ = cv2.connectedComponentsWithStats(obj, 8)
+    ids = np.unique(lab2[hard_seed.astype(bool)])
+    ids = ids[ids != 0]
+    if len(ids):
+        keep = ids[np.argmax([st2[i, cv2.CC_STAT_AREA] for i in ids])]
+        obj = (lab2 == keep).astype(np.uint8)
+
+    ys, xs = np.where(obj)
+    cx = (xs.min() + xs.max()) / 2.0
+    cy = (ys.min() + ys.max()) / 2.0
+
+    # A signed-distance matte gives a clean one-pixel photographed silhouette.
+    din = cv2.distanceTransform(obj, cv2.DIST_L2, 5)
+    dout = cv2.distanceTransform(1-obj, cv2.DIST_L2, 5)
+    alpha = np.clip(0.5 + (din-dout)/1.8, 0, 1).astype(np.float32)
+    alpha = cv2.GaussianBlur(alpha, (0, 0), 0.42)
+    alpha[dist > 28] = 0
+
+    # Remove ball and its immediate cast shadow. Telea interpolation is particularly
+    # stable here because the table has a broad, gently varying wood grain.
+    remove = (np.hypot(xx-cx, yy-cy) < 29).astype(np.uint8) * 255
+    plate = cv2.inpaint(src, remove, 7, cv2.INPAINT_TELEA)
+    # Blend the inpainting at its outer edge, preserving every untouched source pixel.
+    blend = cv2.GaussianBlur((remove > 0).astype(np.float32), (0, 0), 2.0)[..., None]
+    blend[blend < 0.015] = 0
+    plate = np.clip(src*(1-blend) + plate*blend, 0, 255).astype(np.uint8)
+
+    # Unmix edge pixels from the clean plate to avoid carrying a wooden fringe.
+    a = alpha[..., None]
+    fore = np.zeros_like(src, np.float32)
+    valid = a[..., 0] > 0.035
+    fore[valid] = ((src.astype(np.float32)[valid] -
+                    (1-a[valid])*plate.astype(np.float32)[valid]) /
+                   np.maximum(a[valid], 0.035))
+    fore = np.clip(fore, 0, 255)
+    # Fill numerically unstable translucent edge colors from their nearest interior.
+    interior = alpha > 0.82
+    missing = (alpha > 0) & ~interior
+    if missing.any():
+        # distanceTransform labels point each zero pixel to the closest zero (interior).
+        inv = (~interior).astype(np.uint8)
+        _, labels_dt = cv2.distanceTransformWithLabels(inv, cv2.DIST_L2, 5,
+                                                       labelType=cv2.DIST_LABEL_PIXEL)
+        coords = np.argwhere(interior)
+        # OpenCV labels zero pixels in scan order, matching argwhere for this mask.
+        lut = np.vstack(([0, 0], coords))
+        labs = labels_dt[missing]
+        near = lut[np.clip(labs, 0, len(lut)-1)]
+        fore[missing] = fore[near[:, 0], near[:, 1]]
+    return plate, alpha, fore, (cx, cy)
+
+
+def make_generator(src):
+    h, w = src.shape[:2]
+    plate, alpha_full, fore_full, (cx0, cy0) = extract_scene(src)
+
+    # Crop a generous sprite. Coordinates deliberately remain odd-sized so its
+    # geometric center falls on the measured half-pixel source center.
+    rbox = 29
+    x0, x1 = int(np.floor(cx0-rbox)), int(np.ceil(cx0+rbox))+1
+    y0, y1 = int(np.floor(cy0-rbox)), int(np.ceil(cy0+rbox))+1
+    alpha = alpha_full[y0:y1, x0:x1].copy()
+    fore = fore_full[y0:y1, x0:x1].copy()
+    ph, pw = alpha.shape
+    local_cx, local_cy = cx0-x0, cy0-y0
+
+    # Separate stationary illumination from rotating felt detail. At zero angle,
+    # smooth + detail is exactly the extracted source ball.
+    smooth = cv2.GaussianBlur(fore, (0, 0), 3.2)
+    detail = (fore-smooth) * alpha[..., None]
+
+    yy, xx = np.mgrid[:h, :w].astype(np.float32)
+
+    def center_at(frame_float):
+        # A short, natural acceleration from rest followed by nearly uniform rolling.
+        t = np.clip(frame_float, 0, 119) / 24.0
+        T = 119/24.0
+        tau = 0.28
+        q = t - tau*(1-np.exp(-t/tau))
+        qT = T - tau*(1-np.exp(-T/tau))
+        x = cx0 + (486.0-cx0)*(q/qT)
+        dx = x-cx0
+        # The pane's visible edges descend gently to the right. The ball follows
+        # that projective slope after crossing the wooden table edge at x ~= 298.
+        soft = 8.0*np.logaddexp(0.0, (x-298.0)/8.0)
+        soft0 = 8.0*np.logaddexp(0.0, (cx0-298.0)/8.0)
+        y = cy0 + 0.040*dx + 0.060*(soft-soft0)
+        return float(x), float(y)
+
+    def rotated_fore(angle_deg):
+        M = cv2.getRotationMatrix2D((local_cx, local_cy), angle_deg, 1.0)
+        rd = cv2.warpAffine(detail, M, (pw, ph), flags=cv2.INTER_CUBIC,
+                            borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+        return np.clip(smooth + rd, 0, 255)
+
+    def overlay_patch(canvas, rgb, matte, center):
+        """Premultiplied alpha composite, including subpixel translation."""
+        tx = center[0] - cx0
+        ty = center[1] - cy0
+        M = np.float32([[1, 0, x0+tx], [0, 1, y0+ty]])
+        # Warp cropped arrays directly into the frame to retain fractional motion.
+        wrgb = cv2.warpAffine(rgb.astype(np.float32), M, (w, h),
+                             flags=cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)
+        wa = cv2.warpAffine(matte.astype(np.float32), M, (w, h),
+                            flags=cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)
+        wa = np.clip(wa, 0, 1)[..., None]
+        return canvas*(1-wa) + wrgb*wa
+
+    def render_instant(frame_float):
+        x, y = center_at(frame_float)
+        canvas = plate.astype(np.float32).copy()
+
+        # The contact shadow remains attached to the supporting surface. It becomes
+        # a little softer/fainter over transparent glass than over opaque wood.
+        on_glass = float(smoothstep(278.0, 316.0, x))
+        strength = 1.0 - 0.34*on_glass
+        cast = np.exp(-0.5*(((xx-(x-4.0))/14.5)**2 + ((yy-(y+22.0))/4.8)**2))
+        contact = np.exp(-0.5*(((xx-x)/8.0)**2 + ((yy-(y+20.2))/1.85)**2))
+        sh = np.clip(strength*(0.105*cast + 0.16*contact), 0, 0.26)[..., None]
+        # Neutral shadow for glass; slightly warm absorption on the table.
+        shadow_tint = np.array([0.73, 0.72, 0.70], np.float32)
+        canvas *= (1-sh) + sh*shadow_tint
+
+        distance = np.hypot(x-cx0, y-cy0)
+        # Clockwise screen rotation for a sphere rolling to the right.
+        angle = -np.degrees(distance/21.6)
+        ball = rotated_fore(angle)
+        canvas = overlay_patch(canvas, ball, alpha, (x, y))
+        return canvas
+
+    def frame(index):
+        if index == 0:
+            return src.copy()  # contractual pixel identity
+        # 65%-frame shutter sampled symmetrically: subtle real 24 fps motion blur.
+        samples = 5
+        offsets = np.linspace(-0.325, 0.325, samples)
+        acc = np.zeros_like(src, np.float32)
+        for off in offsets:
+            acc += render_instant(np.clip(index+off, 0, 119))
+        out = acc/samples
+
+        # During the first few frames, preserve residual source edge/shadow detail as
+        # it begins to move; this prevents an artificial one-frame segmentation pop.
+        # The contribution vanishes well before the ball approaches the glass.
+        if index < 10:
+            start_model = render_instant(0.0)
+            residual = src.astype(np.float32)-start_model
+            x, y = center_at(float(index))
+            M = np.float32([[1, 0, x-cx0], [0, 1, y-cy0]])
+            moved = cv2.warpAffine(residual, M, (w, h), flags=cv2.INTER_LINEAR,
+                                   borderMode=cv2.BORDER_CONSTANT)
+            fade = (1-smoothstep(1, 10, index))
+            out += moved*fade
+        return np.clip(np.rint(out), 0, 255).astype(np.uint8)
+
+    return frame, center_at
+
+
+def main():
+    spec = json.loads((HERE/'task_spec.json').read_text())
+    src = np.asarray(Image.open(HERE/'first_frame.png').convert('RGB'))
+    assert src.shape == (spec['height'], spec['width'], 3)
+    render, center_at = make_generator(src)
+    frames = np.stack([render(i) for i in range(spec['frames'])])
+    assert frames.dtype == np.uint8
+    assert np.array_equal(frames[0], src)
+
+    # Required benchmark artifacts.
+    artifact_io.save(frames, HERE)
+
+    # A conventional playback copy is useful outside the HTML tensor player.
+    try:
+        import imageio.v2 as imageio
+        imageio.mimwrite(HERE/'video.mp4', frames, fps=spec['fps'], codec='libx264',
+                         quality=8, pixelformat='yuv420p', macro_block_size=1,
+                         ffmpeg_log_level='error')
+    except Exception as exc:
+        print('MP4 convenience export skipped:', exc)
+
+    # Concise deterministic diagnostics.
+    print('saved', frames.shape, frames.dtype)
+    print('frame0_exact', bool(np.array_equal(frames[0], src)))
+    for i in (0, 24, 48, 72, 96, 119):
+        print('center', i, tuple(round(v, 2) for v in center_at(i)))
+
+
+if __name__ == '__main__':
+    main()

    ys, xs = np.where(obj)
    cx = (xs.min() + xs.max()) / 2.0
    cy = (ys.min() + ys.max()) / 2.0

    # A signed-distance matte gives a clean one-pixel photographed silhouette.
    din = cv2.distanceTransform(obj, cv2.DIST_L2, 5)
    dout = cv2.distanceTransform(1-obj, cv2.DIST_L2, 5)
    alpha = np.clip(0.5 + (din-dout)/1.8, 0, 1).astype(np.float32)
    alpha = cv2.GaussianBlur(alpha, (0, 0), 0.42)
    alpha[dist > 28] = 0

    # Remove ball and its immediate cast shadow. Telea interpolation is particularly
    # stable here because the table has a broad, gently varying wood grain.
    remove = (np.hypot(xx-cx, yy-cy) < 29).astype(np.uint8) * 255
    plate = cv2.inpaint(src, remove, 7, cv2.INPAINT_TELEA)
    # Blend the inpainting at its outer edge, preserving every untouched source pixel.
    blend = cv2.GaussianBlur((remove > 0).astype(np.float32), (0, 0), 2.0)[..., None]
    blend[blend < 0.015] = 0
    plate = np.clip(src*(1-blend) + plate*blend, 0, 255).astype(np.uint8)

    # Unmix edge pixels from the clean plate to avoid carrying a wooden fringe.
    a = alpha[..., None]
    fore = np.zeros_like(src, np.float32)
    valid = a[..., 0] > 0.035
    fore[valid] = ((src.astype(np.float32)[valid] -
                    (1-a[valid])*plate.astype(np.float32)[valid]) /
                   np.maximum(a[valid], 0.035))
    fore = np.clip(fore, 0, 255)
    # Fill numerically unstable translucent edge colors from their nearest interior.
    interior = alpha > 0.82
    missing = (alpha > 0) & ~interior
    if missing.any():
        # distanceTransform labels point each zero pixel to the closest zero (interior).
        inv = (~interior).astype(np.uint8)
        _, labels_dt = cv2.distanceTransformWithLabels(inv, cv2.DIST_L2, 5,
                                                       labelType=cv2.DIST_LABEL_PIXEL)
        coords = np.argwhere(interior)
        # OpenCV labels zero pixels in scan order, matching argwhere for this mask.
        lut = np.vstack(([0, 0], coords))
        labs = labels_dt[missing]
        near = lut[np.clip(labs, 0, len(lut)-1)]
        fore[missing] = fore[near[:, 0], near[:, 1]]
    return plate, alpha, fore, (cx, cy)


def make_generator(src):
    h, w = src.shape[:2]
    plate, alpha_full, fore_full, (cx0, cy0) = extract_scene(src)

    # Crop a generous sprite. Coordinates deliberately remain odd-sized so its
    # geometric center falls on the measured half-pixel source center.
    rbox = 29
    x0, x1 = int(np.floor(cx0-rbox)), int(np.ceil(cx0+rbox))+1
    y0, y1 = int(np.floor(cy0-rbox)), int(np.ceil(cy0+rbox))+1
    alpha = alpha_full[y0:y1, x0:x1].copy()
    fore = fore_full[y0:y1, x0:x1].copy()
    ph, pw = alpha.shape
    local_cx, local_cy = cx0-x0, cy0-y0

    # Separate stationary illumination from rotating felt detail. At zero angle,
    # smooth + detail is exactly the extracted source ball.
    smooth = cv2.GaussianBlur(fore, (0, 0), 3.2)
    detail = (fore-smooth) * alpha[..., None]

    yy, xx = np.mgrid[:h, :w].astype(np.float32)

    def center_at(frame_float):
        # A short, natural acceleration from rest followed by nearly uniform rolling.
        t = np.clip(frame_float, 0, 119) / 24.0
        T = 119/24.0
        tau = 0.28
        q = t - tau*(1-np.exp(-t/tau))
        qT = T - tau*(1-np.exp(-T/tau))
        x = cx0 + (486.0-cx0)*(q/qT)
        dx = x-cx0
        # The pane'"'s visible edges descend gently to the right. The ball follows
        # that projective slope after crossing the wooden table edge at x ~= 298.
        soft = 8.0*np.logaddexp(0.0, (x-298.0)/8.0)
        soft0 = 8.0*np.logaddexp(0.0, (cx0-298.0)/8.0)
        y = cy0 + 0.040*dx + 0.060*(soft-soft0)
        return float(x), float(y)

    def rotated_fore(angle_deg):
        M = cv2.getRotationMatrix2D((local_cx, local_cy), angle_deg, 1.0)
        rd = cv2.warpAffine(detail, M, (pw, ph), flags=cv2.INTER_CUBIC,
                            borderMode=cv2.BORDER_CONSTANT, borderValue=0)
        return np.clip(smooth + rd, 0, 255)

    def overlay_patch(canvas, rgb, matte, center):
        \"\"\"Premultiplied alpha composite, including subpixel translation.\"\"\"
        tx = center[0] - cx0
        ty = center[1] - cy0
        M = np.float32([[1, 0, x0+tx], [0, 1, y0+ty]])
        # Warp cropped arrays directly into the frame to retain fractional motion.
        wrgb = cv2.warpAffine(rgb.astype(np.float32), M, (w, h),
                             flags=cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)
        wa = cv2.warpAffine(matte.astype(np.float32), M, (w, h),
                            flags=cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT)
        wa = np.clip(wa, 0, 1)[..., None]
        return canvas*(1-wa) + wrgb*wa

    def render_instant(frame_float):
        x, y = center_at(frame_float)
        canvas = plate.astype(np.float32).copy()

        # The contact shadow remains attached to the supporting surface. It becomes
        # a little softer/fainter over transparent glass than over opaque wood.
        on_glass = float(smoothstep(278.0, 316.0, x))
        strength = 1.0 - 0.34*on_glass
        cast = np.exp(-0.5*(((xx-(x-4.0))/14.5)**2 + ((yy-(y+22.0))/4.8)**2))
        contact = np.exp(-0.5*(((xx-x)/8.0)**2 + ((yy-(y+20.2))/1.85)**2))
        sh = np.clip(strength*(0.105*cast + 0.16*contact), 0, 0.26)[..., None]
        # Neutral shadow for glass; slightly warm absorption on the table.
        shadow_tint = np.array([0.73, 0.72, 0.70], np.float32)
        canvas *= (1-sh) + sh*shadow_tint

        distance = np.hypot(x-cx0, y-cy0)
        # Clockwise screen rotation for a sphere rolling to the right.
        angle = -np.degrees(distance/21.6)
        ball = rotated_fore(angle)
        canvas = overlay_patch(canvas, ball, alpha, (x, y))
        return canvas

    def frame(index):
        if index == 0:
            return src.copy()  # contractual pixel identity
        # 65%-frame shutter sampled symmetrically: subtle real 24 fps motion blur.
        samples = 5
        offsets = np.linspace(-0.325, 0.325, samples)
        acc = np.zeros_like(src, np.float32)
        for off in offsets:
            acc += render_instant(np.clip(index+off, 0, 119))
        out = acc/samples

        # During the first few frames, preserve residual source edge/shadow detail as
        # it begins to move; this prevents an artificial one-frame segmentation pop.
        # The contribution vanishes well before the ball approaches the glass.
        if index < 10:
            start_model = render_instant(0.0)
            residual = src.astype(np.float32)-start_model
            x, y = center_at(float(index))
            M = np.float32([[1, 0, x-cx0], [0, 1, y-cy0]])
            moved = cv2.warpAffine(residual, M, (w, h), flags=cv2.INTER_LINEAR,
                                   borderMode=cv2.BORDER_CONSTANT)
            fade = (1-smoothstep(1, 10, index))
            out += moved*fade
        return np.clip(np.rint(out), 0, 255).astype(np.uint8)

    return frame, center_at
