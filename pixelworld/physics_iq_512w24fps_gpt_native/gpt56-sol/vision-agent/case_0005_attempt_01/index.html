+"""Generate the 5 second, locked-camera cardstock/grabber demonstration.
+
+The animation deliberately uses the photographed pixels as its texture source.  The
+only synthesized portions are surfaces that become uncovered by the rotating card
+and the orange ball / blue grabber described by the prompt.
+"""
+from pathlib import Path
+import math
+
+import cv2
+import numpy as np
+from PIL import Image, ImageDraw, ImageFilter
+
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+W, H, N = 512, 288, 120
+PIVOT = np.array([258.0, 251.0], np.float32)
+CARD = np.array([[164, 108], [342, 106], [351, 248], [168, 263]], np.float32)
+
+
+def smoothstep(x):
+    x = np.clip(x, 0.0, 1.0)
+    return x * x * (3.0 - 2.0 * x)
+
+
+def rotated_points(points, degrees):
+    """Positive degrees means clockwise in image coordinates."""
+    a = math.radians(degrees)
+    r = np.array([[math.cos(a), -math.sin(a)],
+                  [math.sin(a),  math.cos(a)]], np.float32)
+    return (points - PIVOT) @ r.T + PIVOT
+
+
+def make_clean_plate(src):
+    """Plausibly restore wall/table hidden in the source by the card."""
+    plate = src.copy().astype(np.float32)
+    yy, xx = np.mgrid[0:H, 0:W]
+    card_mask = np.zeros((H, W), np.uint8)
+    cv2.fillConvexPoly(card_mask, CARD.astype(np.int32), 255, lineType=cv2.LINE_AA)
+
+    # Wall: interpolate two broad, actually photographed wall samples.  This keeps
+    # the real illumination falloff and the slight warm/cool cast across the wall.
+    left = src[:, 145:158].astype(np.float32).mean(axis=1)
+    right = src[:, 360:373].astype(np.float32).mean(axis=1)
+    for y in range(103, 184):
+        xs = np.arange(W, dtype=np.float32)
+        t = np.clip((xs - 153.0) / 213.0, 0, 1)[:, None]
+        row = left[y][None, :] * (1-t) + right[y][None, :] * t
+        # Broad central lamp falloff, derived from the visible wall's character.
+        row += (-2.8 * np.exp(-((xs-255)/115)**2))[:, None]
+        take = card_mask[y] > 0
+        plate[y, take] = row[take]
+
+    # Table: resample the unobstructed tabletop on the right.  It retains genuine
+    # wood grain instead of replacing the revealed area with a flat brown fill.
+    for y in range(182, 265):
+        for x in range(155, 359):
+            if card_mask[y, x]:
+                sx = int(355 + (x-155) * 150 / 204) % W
+                sx = min(510, max(355, sx))
+                plate[y, x] = src[y, sx]
+
+    plate = np.clip(plate, 0, 255).astype(np.uint8)
+
+    # Rebuild the parts of the black turntable and upright axle hidden by the card.
+    cv2.ellipse(plate, (256, 258), (73, 21), 0, 0, 360, (12, 12, 12), -1, cv2.LINE_AA)
+    cv2.ellipse(plate, (256, 253), (70, 17), 0, 0, 360, (26, 26, 25), -1, cv2.LINE_AA)
+    cv2.ellipse(plate, (256, 248), (62, 11), 0, 0, 180, (39, 39, 38), 2, cv2.LINE_AA)
+    # The photographed vertical mounting spindle, behind both demonstration items.
+    cv2.rectangle(plate, (232, 95), (249, 247), (10, 10, 10), -1)
+    cv2.line(plate, (239, 96), (239, 244), (31, 31, 31), 2, cv2.LINE_AA)
+    return plate
+
+
+def card_rgba(src):
+    """Premultiplied RGBA cutout of the original cardstock."""
+    scale = 4
+    m = Image.new("L", (W*scale, H*scale), 0)
+    d = ImageDraw.Draw(m)
+    d.polygon([(int(x*scale), int(y*scale)) for x, y in CARD], fill=255)
+    m = m.resize((W, H), Image.Resampling.LANCZOS)
+    alpha = np.asarray(m, dtype=np.float32) / 255.0
+    rgb = src.astype(np.float32) * alpha[..., None]
+    return rgb, alpha
+
+
+def composite_card(frame, rgb_premul, alpha, angle):
+    # OpenCV's negative angle produces clockwise visual motion.
+    mat = cv2.getRotationMatrix2D(tuple(PIVOT), -angle, 1.0)
+    wrgb = cv2.warpAffine(rgb_premul, mat, (W, H), flags=cv2.INTER_CUBIC,
+                         borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+    wa = cv2.warpAffine(alpha, mat, (W, H), flags=cv2.INTER_CUBIC,
+                        borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+    wa = np.clip(wa, 0, 1)
+    return np.clip(wrgb + frame.astype(np.float32)*(1-wa[..., None]), 0, 255).astype(np.uint8)
+
+
+def add_card_shadow(frame, angle):
+    pts = rotated_points(CARD, angle) + np.array([7.0, 6.0])
+    layer = np.zeros((H, W), np.uint8)
+    cv2.fillConvexPoly(layer, np.round(pts).astype(np.int32), 68, cv2.LINE_AA)
+    layer = cv2.GaussianBlur(layer, (0, 0), 5.0).astype(np.float32) / 255.0
+    # Very subtle: enough to seat the sheet spatially, not enough to dirty it.
+    return np.clip(frame.astype(np.float32)*(1.0-0.28*layer[..., None]), 0, 255).astype(np.uint8)
+
+
+def ball_and_grabber(frame, ball_xy, jaw_xy, jaw_open, opacity=1.0, show_ball=True):
+    """Paint a softly antialiased fuzzy tennis ball and blue-black reacher."""
+    S = 3
+    canvas = Image.new("RGBA", (W*S, H*S), (0, 0, 0, 0))
+    dr = ImageDraw.Draw(canvas, "RGBA")
+    bx, by = ball_xy
+    jx, jy = jaw_xy
+    bx, by, jx, jy = [v*S for v in (bx, by, jx, jy)]
+
+    # Telescoping dark-blue reacher comes from off camera. Two restrained highlights
+    # give the tool a rounded metal/plastic appearance.
+    top = (118*S, -25*S)
+    dr.line([top, (jx, jy-14*S)], fill=(7, 15, 24, int(255*opacity)), width=11*S)
+    dr.line([(top[0]+2*S, top[1]), (jx+2*S, jy-14*S)],
+            fill=(22, 54, 79, int(255*opacity)), width=5*S)
+    dr.line([(top[0]+3*S, top[1]), (jx+3*S, jy-14*S)],
+            fill=(54, 91, 117, int(180*opacity)), width=S)
+    # Black/blue pivot housing.
+    dr.ellipse([jx-9*S, jy-21*S, jx+9*S, jy-4*S],
+               fill=(9, 16, 23, int(255*opacity)), outline=(43, 76, 98, int(255*opacity)), width=2*S)
+
+    # Curved opposing jaws. As the ball is deposited they visibly open.
+    spread = (13 + 9*jaw_open)*S
+    grip_y = by-2*S
+    for side in (-1, 1):
+        x0 = jx + side*5*S
+        x1 = bx + side*spread
+        pts = [(x0, jy-5*S), (jx+side*10*S, jy+6*S),
+               (x1, grip_y-9*S), (bx+side*(16+7*jaw_open)*S, grip_y+6*S)]
+        dr.line(pts, fill=(5, 12, 18, int(255*opacity)), width=7*S, joint="curve")
+        dr.line(pts, fill=(24, 62, 88, int(235*opacity)), width=3*S, joint="curve")
+    # Rubber jaw pads sit against the sides of the ball.
+    for side in (-1, 1):
+        px = bx + side*(16+7*jaw_open)*S
+        dr.rounded_rectangle([px-3*S, by-7*S, px+3*S, by+7*S], radius=2*S,
+                             fill=(7, 10, 12, int(255*opacity)))
+
+    if show_ball:
+        r = 17*S
+        # Soft contact/shading and several concentric circles make the small object
+        # integrate with the slightly soft photographic source.
+        dr.ellipse([bx-r-2*S, by-r+3*S, bx+r+3*S, by+r+5*S],
+                   fill=(68, 33, 6, int(75*opacity)))
+        dr.ellipse([bx-r, by-r, bx+r, by+r], fill=(220, 91, 8, int(255*opacity)),
+                   outline=(133, 54, 5, int(255*opacity)), width=S)
+        dr.ellipse([bx-r+3*S, by-r+2*S, bx+r-5*S, by+r-6*S],
+                   fill=(239, 113, 14, int(245*opacity)))
+        dr.ellipse([bx-r+5*S, by-r+4*S, bx+2*S, by+1*S],
+                   fill=(255, 153, 44, int(170*opacity)))
+        # Tennis-ball curved cream seam.
+        seam = ImageDraw.Draw(canvas, "RGBA")
+        seam.arc([bx-14*S, by-21*S, bx+12*S, by+10*S], 286, 72,
+                 fill=(255, 197, 105, int(230*opacity)), width=2*S)
+        seam.arc([bx-11*S, by-8*S, bx+15*S, by+22*S], 105, 246,
+                 fill=(255, 190, 92, int(215*opacity)), width=2*S)
+        # Tiny deterministic fuzz glints around the silhouette.
+        for k in range(28):
+            a = k*2*math.pi/28 + 0.17
+            rr = (17.0 + (k % 3)*0.35)*S
+            x = bx + math.cos(a)*rr; y = by + math.sin(a)*rr
+            dr.ellipse([x-S*.45, y-S*.45, x+S*.45, y+S*.45],
+                       fill=(246, 143, 36, int(135*opacity)))
+
+    canvas = canvas.resize((W, H), Image.Resampling.LANCZOS)
+    rgba = np.asarray(canvas).astype(np.float32)
+    a = rgba[..., 3:4] / 255.0
+    out = rgba[..., :3]*a + frame.astype(np.float32)*(1-a)
+    return np.clip(out, 0, 255).astype(np.uint8)
+
+
+def animate(src):
+    clean = make_clean_plate(src)
+    crgb, ca = card_rgba(src)
+    frames = []
+    for i in range(N):
+        if i == 0:
+            frames.append(src.copy())
+            continue
+
+        # The rotation starts only after the initial held establishing view.  A
+        # small settling overshoot makes the mechanism read as physical rather than
+        # as a constant-speed image transform.
+        rot = 0.0 if i < 12 else 25.0*smoothstep((i-12)/92.0)
+        if i > 104:
+            rot = 25.0 + 0.7*math.exp(-(i-104)/7.0)*math.sin((i-104)*0.72)
+
+        frame = add_card_shadow(clean.copy(), rot)
+
+        # Tool enters holding the ball, pauses above the cardstock, then lowers it
+        # behind the card and releases it onto the table.
+        if i >= 7:
+            if i <= 26:                    # enter from above
+                u = smoothstep((i-7)/19.0)
+                bx, by = 167+31*u, -25+92*u
+                show_ball, jaw_open = True, 0.0
+                jx, jy = bx, by-18
+            elif i <= 40:                  # short, legible hold above the card
+                bx, by = 198, 67
+                show_ball, jaw_open = True, 0.0
+                jx, jy = bx, by-18
+            elif i <= 83:                  # controlled lowering motion
+                u = smoothstep((i-40)/43.0)
+                bx = 198 - 15*u
+                by = 67 + 149*u
+                show_ball, jaw_open = True, smoothstep((u-0.77)/0.23)
+                jx, jy = bx, by-18
+            else:                          # ball stays; unloaded grabber retracts
+                bx, by = 183, 216
+                show_ball = True
+                u = smoothstep((i-83)/30.0)
+                jx, jy = 183-34*u, 198-236*u
+                jaw_open = 1.0
+
+            # Add the placed ball first when the gripper is retracting, then overlay
+            # the empty tool. Both remain behind the cardstock in the final stack.
+            if i > 83:
+                frame = ball_and_grabber(frame, (bx, by), (jx, jy), 1.0,
+                                         opacity=0.0, show_ball=True)
+                frame = ball_and_grabber(frame, (bx, by), (jx, jy), jaw_open,
+                                         opacity=1.0, show_ball=False)
+            else:
+                frame = ball_and_grabber(frame, (bx, by), (jx, jy), jaw_open,
+                                         opacity=1.0, show_ball=show_ball)
+
+        # Foreground card occludes the descending tool and ball, which is the key
+        # depth/physics cue in this demonstration.
+        frame = composite_card(frame, crgb, ca, rot)
+        frames.append(frame)
+    return np.stack(frames).astype(np.uint8)
+
+
+if __name__ == "__main__":
+    source = np.asarray(Image.open(ROOT / "first_frame.png").convert("RGB"))
+    result = animate(source)
+    assert np.array_equal(result[0], source)
+    artifact_io.save(result, ROOT)
+    print(f"saved {result.shape[0]} RGB frames; first frame exact: {np.array_equal(result[0], source)}")
