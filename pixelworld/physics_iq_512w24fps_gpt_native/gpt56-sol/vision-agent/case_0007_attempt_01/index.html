+"""Generate the requested locked-off rubber-duck collision video.
+
+The animation is deliberately made from the supplied plate only.  The two
+moving subjects are represented as difference sprites over an inpainted copy
+of the tabletop.  This retains the photographic blur, illumination and
+surface contact shadows of the source image instead of replacing them with
+drawn approximations.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+W, H = 512, 288
+N_FRAMES, FPS = 120, 24
+
+
+def soft_inpaint(plate, binary_mask, radius=7):
+    """Inpaint a masked object and feather the repair into the source plate."""
+    mask_u8 = (binary_mask.astype(np.uint8) * 255)
+    repaired = cv2.inpaint(plate, mask_u8, radius, cv2.INPAINT_TELEA)
+
+    # A short feather avoids an artificial contour around the repaired patch.
+    soft = cv2.GaussianBlur(mask_u8, (0, 0), 2.2).astype(np.float32) / 255.0
+    soft = soft[..., None]
+    clean = plate.astype(np.float32) * (1.0 - soft) + repaired.astype(np.float32) * soft
+    return clean
+
+
+def transform_about(delta, center, dx=0.0, dy=0.0, angle=0.0, sx=1.0, sy=1.0):
+    """Transform a floating RGB difference layer about a chosen image point."""
+    px, py = center
+    theta = np.deg2rad(angle)
+    c, s = np.cos(theta), np.sin(theta)
+
+    # Source-to-destination transform in homogeneous coordinates.  In image
+    # coordinates a negative angle tips the top of the duck to the right.
+    translate_to_origin = np.array([[1, 0, -px], [0, 1, -py], [0, 0, 1]], np.float32)
+    scale = np.array([[sx, 0, 0], [0, sy, 0], [0, 0, 1]], np.float32)
+    rotate = np.array([[c, -s, 0], [s, c, 0], [0, 0, 1]], np.float32)
+    translate_back = np.array(
+        [[1, 0, px + dx], [0, 1, py + dy], [0, 0, 1]], np.float32
+    )
+    matrix = (translate_back @ rotate @ scale @ translate_to_origin)[:2]
+    return cv2.warpAffine(
+        delta,
+        matrix,
+        (W, H),
+        flags=cv2.INTER_CUBIC,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=(0, 0, 0),
+    )
+
+
+def motion_at(t):
+    """Positions for a rolling ball and an impulse-driven, damped duck."""
+    impact = 50.0
+    ball_x0 = 169.5
+    contact_x = 386.5
+    approach_speed = (contact_x - ball_x0) / impact
+
+    if t <= impact:
+        ball_x = ball_x0 + approach_speed * t
+        duck_dx = duck_dy = duck_angle = 0.0
+        duck_sx = duck_sy = 1.0
+    else:
+        u = t - impact
+
+        # The tennis ball gives up most of its speed at impact, but follows
+        # through a short distance before friction brings it to rest.
+        ball_x = contact_x + 19.0 * (1.0 - np.exp(-u / 10.0))
+
+        # A light rubber duck slides, briefly compresses at contact, and then
+        # rocks about its flat base with a quickly damped oscillation.
+        duck_dx = 30.0 * (1.0 - np.exp(-u / 13.0))
+        duck_angle = -11.5 * np.exp(-u / 18.0) * np.sin(np.pi * u / 10.0)
+        compression = np.exp(-((u - 2.0) / 2.4) ** 2)
+        duck_sx = 1.0 + 0.055 * compression
+        duck_sy = 1.0 - 0.065 * compression
+        duck_dy = 0.9 * np.exp(-u / 16.0) * np.sin(np.pi * u / 10.0) ** 2
+
+    # One full turn per circumference: the visible highlight/seam on the ball
+    # makes the rolling (rather than sliding) immediately readable.
+    travel = ball_x - ball_x0
+    ball_angle = -np.rad2deg(travel / 17.0)
+    return ball_x, ball_angle, duck_dx, duck_dy, duck_angle, duck_sx, duck_sy
+
+
+def main():
+    plate = np.asarray(Image.open(CASE_DIR / "first_frame.png").convert("RGB"))
+    assert plate.shape == (H, W, 3)
+
+    # Masks include each object's soft photographic contact shadow.  Their
+    # broad edges are harmless because the actual difference sprites naturally
+    # approach zero there.
+    ball_mask = np.zeros((H, W), np.uint8)
+    cv2.ellipse(ball_mask, (170, 103), (31, 28), 0, 0, 360, 1, -1)
+
+    duck_mask = np.zeros((H, W), np.uint8)
+    cv2.ellipse(duck_mask, (424, 83), (38, 37), 0, 0, 360, 1, -1)
+    # Long, soft shadow cast to the lower-left by the overhead-right light.
+    cv2.ellipse(duck_mask, (394, 96), (43, 18), 9, 0, 360, 1, -1)
+    duck_mask = cv2.dilate(duck_mask, np.ones((5, 5), np.uint8), iterations=1)
+
+    ball_clean = soft_inpaint(plate, ball_mask, radius=8)
+    duck_clean = soft_inpaint(plate, duck_mask, radius=9)
+
+    # The masks are disjoint, so their independently repaired patches can be
+    # combined into one clean locked-off background.
+    full_clean = plate.astype(np.float32).copy()
+    ball_soft = cv2.GaussianBlur(ball_mask * 255, (0, 0), 2.2).astype(np.float32) / 255.0
+    duck_soft = cv2.GaussianBlur(duck_mask * 255, (0, 0), 2.2).astype(np.float32) / 255.0
+    full_clean = full_clean * (1.0 - ball_soft[..., None]) + ball_clean * ball_soft[..., None]
+    full_clean = full_clean * (1.0 - duck_soft[..., None]) + duck_clean * duck_soft[..., None]
+
+    # Difference sprites preserve every supplied photographic detail.  A
+    # gentle spatial gate prevents tiny inpainting residuals at the outer mask
+    # boundary from travelling with an object.
+    ball_gate = cv2.GaussianBlur(ball_mask * 255, (0, 0), 1.25).astype(np.float32) / 255.0
+    duck_gate = cv2.GaussianBlur(duck_mask * 255, (0, 0), 1.25).astype(np.float32) / 255.0
+    ball_delta = (plate.astype(np.float32) - ball_clean) * ball_gate[..., None]
+    duck_delta = (plate.astype(np.float32) - duck_clean) * duck_gate[..., None]
+
+    frames = np.empty((N_FRAMES, H, W, 3), np.uint8)
+    frames[0] = plate  # contractual pixel-identical first frame
+
+    # Four temporal samples provide natural camera-like motion blur while all
+    # geometry and the camera itself remain perfectly fixed.
+    sample_offsets = (-0.33, -0.11, 0.11, 0.33)
+    impact = 50.0
+    for frame_index in range(1, N_FRAMES):
+        accum = np.zeros((H, W, 3), np.float32)
+        for offset in sample_offsets:
+            t = max(0.0, frame_index + offset)
+            ball_x, ball_angle, ddx, ddy, dangle, dsx, dsy = motion_at(t)
+
+            # Before collision use the untouched original duck and shadow.
+            # From collision onward use the fully clean plate plus its moving
+            # difference sprite.  At t=impact both representations coincide.
+            if t < impact:
+                sample = plate.astype(np.float32) - (plate.astype(np.float32) - ball_clean)
+            else:
+                sample = full_clean.copy()
+                sample += transform_about(
+                    duck_delta,
+                    center=(425.0, 103.0),
+                    dx=ddx,
+                    dy=ddy,
+                    angle=dangle,
+                    sx=dsx,
+                    sy=dsy,
+                )
+
+            sample += transform_about(
+                ball_delta,
+                center=(169.5, 102.5),
+                dx=ball_x - 169.5,
+                dy=0.0,
+                angle=ball_angle,
+            )
+            accum += np.clip(sample, 0, 255)
+
+        frames[frame_index] = np.clip(accum / len(sample_offsets) + 0.5, 0, 255).astype(np.uint8)
+
+    artifact_io.save(frames, CASE_DIR)
+
+
+if __name__ == "__main__":
+    main()
