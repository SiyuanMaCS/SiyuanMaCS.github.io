+"""Generate the requested locked-camera scientific demonstration.
+
+The original photograph supplies all of the scene texture.  The animation is a
+small 2.5-D reconstruction: a clean plate is made for the area hidden by the
+card, the photographed card is perspective-warped about its vertical axis, and
+a shaded tennis ball and moving grabber are composited behind it.
+"""
+from pathlib import Path
+
+import cv2
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+W, H, N = 512, 288, 120
+
+
+def smoothstep(x):
+    x = np.clip(x, 0.0, 1.0)
+    return x * x * (3.0 - 2.0 * x)
+
+
+def polygon_mask(points, blur=0.0):
+    m = np.zeros((H, W), np.uint8)
+    cv2.fillPoly(m, [np.asarray(points, np.int32)], 255)
+    if blur:
+        m = cv2.GaussianBlur(m, (0, 0), blur)
+    return m
+
+
+def alpha_over(dst, src, alpha):
+    a = alpha.astype(np.float32)[..., None] / 255.0
+    return np.clip(src.astype(np.float32) * a + dst.astype(np.float32) * (1-a), 0, 255).astype(np.uint8)
+
+
+def make_clean_plate(src):
+    """Reconstruct the smooth wall/table and the obscured part of turntable."""
+    yy, xx = np.mgrid[0:H, 0:W]
+    # Back edge of the table, inferred from the unobscured left and right sides.
+    edge = np.where(xx <= 390, 194.0 - 0.083 * xx, 162.0 + 0.205 * (xx - 390))
+
+    # Polynomial wall-lighting model, sampled only from unquestionably clear wall.
+    xn, yn = xx / W, yy / H
+    basis = np.stack([np.ones_like(xn), xn, yn, xn*xn, xn*yn, yn*yn,
+                      xn**3, xn*xn*yn, xn*yn*yn], axis=-1)
+    known = (yy < edge - 7) & ~(((xx > 190) & (xx < 390) & (yy < 265)) |
+                                ((xx > 224) & (xx < 247)))
+    take = known & ((xx % 4 == 0) & (yy % 4 == 0))
+    A = basis[take]
+    wall = np.empty_like(src, dtype=np.float32)
+    for c in range(3):
+        coef = np.linalg.lstsq(A, src[..., c][take], rcond=None)[0]
+        wall[..., c] = basis @ coef
+    wall = cv2.GaussianBlur(np.clip(wall, 0, 255).astype(np.uint8), (0, 0), 1.4)
+
+    # Clone table texture from its clear left side, preserving distance from its
+    # rear edge. This keeps genuine wood grain in newly exposed regions.
+    mapx = np.clip(35.0 + (xx - 195.0) * 0.58, 8, 185).astype(np.float32)
+    src_edge = 194.0 - 0.083 * mapx
+    mapy = np.clip(src_edge + (yy - edge), 0, H-1).astype(np.float32)
+    table = cv2.remap(src, mapx, mapy, cv2.INTER_LINEAR, borderMode=cv2.BORDER_REFLECT)
+
+    clean = np.where((yy < edge)[..., None], wall, table)
+
+    # Only replace the region occupied by the apparatus. Elsewhere retain the
+    # untouched source photograph, including the real cable and wood texture.
+    apparatus = polygon_mask([(194, 0), (250, 0), (250, 90), (370, 90),
+                              (390, 224), (370, 268), (220, 270), (195, 215)], 2.0)
+    clean = alpha_over(src, clean, apparatus)
+
+    # Rebuild the fixed black rotating platform (the card will cover its center).
+    layer = clean.copy()
+    base_alpha = np.zeros((H, W), np.uint8)
+    cv2.ellipse(base_alpha, (296, 254), (70, 28), 0, 0, 360, 245, -1, cv2.LINE_AA)
+    cv2.rectangle(base_alpha, (226, 226), (366, 255), 245, -1)
+    # Soft black cylinder with a faint warm reflection near its upper rim.
+    for y in range(216, 281):
+        shade = int(np.clip(32 - 0.28*(y-221), 10, 34))
+        layer[y, :, :] = np.where((base_alpha[y, :] > 0)[:, None], shade, layer[y, :, :])
+    cv2.ellipse(layer, (296, 227), (70, 19), -5, 0, 360, (38, 37, 36), -1, cv2.LINE_AA)
+    cv2.ellipse(layer, (296, 223), (67, 14), -5, 0, 360, (48, 47, 45), -1, cv2.LINE_AA)
+    cv2.ellipse(layer, (296, 260), (68, 21), 0, 0, 180, (13, 13, 12), -1, cv2.LINE_AA)
+    # Blend only within the platform footprint to retain photographic edges.
+    plat = np.zeros((H, W), np.uint8)
+    cv2.ellipse(plat, (296, 250), (72, 33), 0, 0, 360, 255, -1, cv2.LINE_AA)
+    plat = cv2.GaussianBlur(plat, (0, 0), .65)
+    clean = alpha_over(clean, layer, plat)
+    return clean
+
+
+def make_card_layer(src):
+    # Main square plus the small visible rear/supporting flap on its left.
+    silhouette = [(224, 98), (367, 93), (385, 219), (247, 258),
+                  (239, 215), (201, 219), (217, 139)]
+    alpha = polygon_mask(silhouette, .7)
+    rgba = np.dstack([src, alpha])
+    return rgba
+
+
+def make_tool_layer(src):
+    # Carefully follows the photographed shaft/sleeve and blue-black left jaw.
+    m = np.zeros((H, W), np.uint8)
+    cv2.rectangle(m, (230, 0), (241, 101), 255, -1)
+    cv2.fillPoly(m, [np.array([(226, 34), (243, 34), (246, 101), (237, 115),
+                              (228, 110)], np.int32)], 255)
+    cv2.fillPoly(m, [np.array([(230, 94), (222, 103), (213, 116), (204, 130),
+                              (198, 149), (198, 163), (204, 180), (211, 185),
+                              (219, 178), (218, 163), (218, 149), (223, 132),
+                              (232, 119), (239, 111), (239, 97)], np.int32)], 255)
+    # Avoid stealing any foreground cardstock pixels at their overlap.
+    card_front = polygon_mask([(224, 99), (367, 94), (385, 219), (247, 258)])
+    m[(card_front > 0) & (np.indices(m.shape)[1] > 225)] = 0
+    m = cv2.GaussianBlur(m, (0, 0), .55)
+    return np.dstack([src, m])
+
+
+def draw_polyline_layer(shape, points, color, width):
+    lay = np.zeros(shape, np.uint8)
+    cv2.polylines(lay, [np.asarray(points, np.int32)], False, color, width,
+                  lineType=cv2.LINE_AA)
+    return lay
+
+
+def composite_rgba(dst, rgba):
+    return alpha_over(dst, rgba[..., :3], rgba[..., 3])
+
+
+def ball_rgba(cx, cy, r=23):
+    """An orange, softly textured tennis ball with recognizable curved seams."""
+    yy, xx = np.mgrid[0:H, 0:W]
+    dx, dy = xx-cx, yy-cy
+    rr = np.sqrt(dx*dx + dy*dy)
+    # Soft/fuzzy silhouette.
+    alpha = np.clip((r + .75 - rr) * 190, 0, 255).astype(np.uint8)
+    nx, ny = dx/r, dy/r
+    nz = np.sqrt(np.clip(1-nx*nx-ny*ny, 0, 1))
+    light = np.clip(.38 + .62*(np.maximum(0, -.42*nx-.58*ny+.69*nz)), 0, 1)
+    rng_texture = (np.sin(xx*1.71 + yy*.73) + np.sin(xx*.39-yy*1.33))*1.7
+    rgb = np.empty((H, W, 3), np.float32)
+    base = np.array([225., 104., 22.])
+    rgb[:] = base
+    rgb *= (.62 + .48*light[..., None])
+    rgb += rng_texture[..., None]
+    # Two pale, curved seam arcs clipped to the sphere.
+    seam = np.zeros((H, W), np.uint8)
+    cv2.ellipse(seam, (int(cx-r*.50), int(cy)),
+                (int(r*.68), int(r*.97)), 0, 272, 82, 215, 2, cv2.LINE_AA)
+    cv2.ellipse(seam, (int(cx+r*.50), int(cy)),
+                (int(r*.68), int(r*.97)), 0, 92, 262, 215, 2, cv2.LINE_AA)
+    seam = cv2.bitwise_and(seam, seam, mask=(rr <= r).astype(np.uint8)*255)
+    sa = seam.astype(np.float32)[..., None]/255
+    rgb = rgb*(1-sa*.55) + np.array([245, 205, 117], np.float32)*sa*.55
+    return np.dstack([np.clip(rgb, 0, 255).astype(np.uint8), alpha])
+

+def main():
+    src = np.asarray(Image.open(ROOT / "first_frame.png").convert("RGB"))
+    clean = make_clean_plate(src)
+    card = make_card_layer(src)
+    tool = make_tool_layer(src)
+    q = np.float32([[224, 99], [367, 94], [385, 219], [247, 258]])
+    center_x = float(q[:, 0].mean())
+
+    frames = []
+    for f in range(N):
+        # Constant turntable motion after a very gentle startup.  68 degrees of
+        # yaw makes the clockwise rotation unambiguous without an implausible flip.
+        phase = f/(N-1)
+        # Essentially constant real-time rotation, with only a tiny ease at the
+        # beginning to avoid a mechanical one-frame jerk.
+        rot_phase = phase if phase >= .08 else .08*smoothstep(phase/.08)
+        theta = np.deg2rad(68.0 * rot_phase)
+
+        # Lower, hold/release, then retract the grabber while the ball stays put.
+        down = 56.0 * smoothstep((f-10)/54.0)
+        if f <= 70:
+            tool_dy = down
+            ball_y = 150.0 + down
+        else:
+            ball_y = 206.0
+            tool_dy = 56.0 - 148.0*smoothstep((f-78)/34.0)
+
+        frame = clean.copy()
+
+        # Soft contact shadow becomes visible as the ball reaches the tabletop.
+        contact = smoothstep((ball_y-181)/22)
+        sh = np.zeros((H, W), np.uint8)
+        cv2.ellipse(sh, (268, 229), (29, 8), 0, 0, 360, int(92*contact), -1, cv2.LINE_AA)
+        sh = cv2.GaussianBlur(sh, (0, 0), 4)
+        dark = (frame.astype(np.float32)*.58).astype(np.uint8)
+        frame = alpha_over(frame, dark, sh)
+
+        # Translate the photographed visible part of the grabber.
+        Mtool = np.float32([[1, 0, 0], [0, 1, tool_dy]])
+        moved_tool = cv2.warpAffine(tool, Mtool, (W, H), flags=cv2.INTER_LINEAR,
+                                    borderMode=cv2.BORDER_CONSTANT)
+        # Keep a metallic shaft connected to the top edge while lowering.
+        if tool_dy > 1:
+            shaft = np.zeros((H, W, 4), np.uint8)
+            y2 = min(H-1, int(35+tool_dy))
+            for x in range(230, 242):
+                val = int(104 + 42*np.exp(-((x-236)/3.1)**2))
+                shaft[:y2, x, :3] = (val, val, val+2)
+                shaft[:y2, x, 3] = 255
+            frame = composite_rgba(frame, shaft)
+        frame = composite_rgba(frame, moved_tool)
+
+        # Hidden/right jaw becomes visible naturally as the card turns edge-on.
+        huby = int(104 + tool_dy)
+        by = int(ball_y)
+        open_amt = 15.0*smoothstep((f-66)/15.0)
+        right_pts = [(238, huby), (257, huby+18), (282+int(open_amt), by-13),
+                     (289+int(open_amt), by+4)]
+        jaw = np.zeros((H, W, 4), np.uint8)
+        cv2.polylines(jaw, [np.asarray(right_pts, np.int32)], False, (12, 13, 14, 255), 13, cv2.LINE_AA)
+        cv2.polylines(jaw, [np.asarray(right_pts, np.int32)], False, (10, 88, 153, 255), 5, cv2.LINE_AA)
+        frame = composite_rgba(frame, jaw)
+
+        # Ball follows the jaws until deposited, then remains stationary.
+        ball = ball_rgba(268, ball_y, 23)
+        frame = composite_rgba(frame, ball)
+        # Small rubber gripping pads in front of the ball; they open on release.
+        pads = np.zeros((H, W, 4), np.uint8)
+        release = smoothstep((f-66)/15.0)
+        lx, rx = int(246-15*release), int(290+15*release)
+        cv2.ellipse(pads, (lx, by), (7, 13), -12, 0, 360, (15, 16, 17, 255), -1, cv2.LINE_AA)
+        cv2.ellipse(pads, (rx, by), (7, 13), 12, 0, 360, (15, 16, 17, 255), -1, cv2.LINE_AA)
+        frame = composite_rgba(frame, pads)
+
+        # Perspective-warp the real cardstock about a vertical axis. The slight
+        # opposite vertical shifts of its edges convey depth/yaw rather than a
+        # flat image-plane spin.
+        c, s = np.cos(theta), np.sin(theta)
+        target = q.copy()
+        for i in range(4):
+            side = -1.0 if i in (0, 3) else 1.0
+            target[i, 0] = center_x + (q[i, 0]-center_x)*c + side*2.5*s
+            target[i, 1] = q[i, 1] + side*10.0*s
+        Hm = cv2.getPerspectiveTransform(q, target.astype(np.float32))
+        warped = cv2.warpPerspective(card, Hm, (W, H), flags=cv2.INTER_CUBIC,
+                                     borderMode=cv2.BORDER_CONSTANT)
+        # A turning matte card becomes slightly dimmer at grazing angle.
+        shade = .99 - .13*(1-c)
+        warped[..., :3] = np.clip(warped[..., :3].astype(np.float32)*shade, 0, 255).astype(np.uint8)
+        frame = composite_rgba(frame, warped)
+
+        # Frame zero is contractually and visually identical to the photograph.
+        if f == 0:
+            frame = src.copy()
+        frames.append(frame)
+
+    frames = np.asarray(frames, np.uint8)
+    assert np.array_equal(frames[0], src)
+    artifact_io.save(frames, ROOT)
+
+
+if __name__ == "__main__":
+    main()
