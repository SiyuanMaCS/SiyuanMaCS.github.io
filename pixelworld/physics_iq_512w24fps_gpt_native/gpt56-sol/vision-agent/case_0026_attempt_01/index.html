+#!/usr/bin/env python3
+"""Generate the locked-camera ball-on-glass demonstration.
+
+The photographed setup is kept as the plate.  Only the ball and its contact
+shadow are moved.  A simple intrinsic decomposition of the photographed ball
+keeps the broad illumination fixed while rotating its seam/surface detail.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+W, H, N = 512, 288, 120
+
+
+def smoothstep(edge0, edge1, value):
+    value = np.clip((value - edge0) / (edge1 - edge0), 0.0, 1.0)
+    return value * value * (3.0 - 2.0 * value)
+
+
+def reconstruct_plate(source):
+    """Remove the original ball and shadow without disturbing scene geometry."""
+    mask = np.zeros((H, W), np.uint8)
+    # The broad mask includes the soft out-of-focus rim and cast shadow.
+    cv2.circle(mask, (209, 136), 29, 255, -1, lineType=cv2.LINE_AA)
+    cv2.ellipse(mask, (219, 159), (43, 13), 0, 0, 360, 255, -1,
+                lineType=cv2.LINE_AA)
+    plate = cv2.inpaint(source, mask, 8, cv2.INPAINT_TELEA)
+
+    # Inpainting can be over-smooth. Add back a tiny amount of local wood grain
+    # inferred from the undisturbed bands directly above and below the object.
+    # This correction is confined to the wooden tabletop.
+    soft = cv2.GaussianBlur((mask > 0).astype(np.float32), (0, 0), 3.0)
+    wood_region = np.zeros_like(soft)
+    wood_region[104:179, 175:266] = 1.0
+    soft *= wood_region
+    low = cv2.GaussianBlur(plate, (0, 0), 2.2)
+    grain = plate.astype(np.float32) - low.astype(np.float32)
+    # Grain already remains around the edge; extending it is not necessary.
+    plate = np.clip(plate.astype(np.float32) + grain * soft[..., None] * 0.12,
+                    0, 255).astype(np.uint8)
+    return plate
+
+
+def make_ball_assets(source, plate):
+    """Matte the photographed ball and split fixed shading from rotating detail."""
+    size = 72
+    half = size // 2
+    cx, cy = 209, 136
+    observed = source[cy-half:cy+half, cx-half:cx+half].astype(np.float32)
+    backing = plate[cy-half:cy+half, cx-half:cx+half].astype(np.float32)
+
+    yy, xx = np.mgrid[0:size, 0:size].astype(np.float32)
+    # Slight horizontal elongation matches the photographed, gently blurred rim.
+    rx = (xx - half) / 1.045
+    ry = yy - half
+    radius = np.sqrt(rx * rx + ry * ry)
+    geometric = 1.0 - smoothstep(20.7, 25.7, radius)
+
+    # The chroma difference supplies additional evidence at the naturally soft rim.
+    difference = observed - backing
+    chroma = np.sqrt(np.sum(difference * difference, axis=2))
+    blue_evidence = np.clip((chroma - 4.0) / 67.0, 0.0, 1.0)
+    blue_evidence *= (observed[..., 2] > observed[..., 0] + 14).astype(np.float32)
+    alpha = np.maximum(geometric, blue_evidence * (radius < 26.2))
+    alpha *= 1.0 - smoothstep(25.0, 27.0, radius)
+    alpha = cv2.GaussianBlur(alpha, (0, 0), 0.38)
+    alpha = np.clip(alpha, 0.0, 1.0)
+
+    # Solve C = alpha*F + (1-alpha)*B for foreground colour.
+    safe = np.maximum(alpha[..., None], 0.10)
+    foreground = (observed - (1.0 - alpha[..., None]) * backing) / safe
+    foreground = np.clip(foreground, 0.0, 255.0)
+
+    # Broad colour/shading does not rotate with the ball relative to the lamps.
+    # High-frequency texture (especially the curved pale seam) does rotate.
+    weighted = foreground * alpha[..., None]
+    blur_alpha = cv2.GaussianBlur(alpha, (0, 0), 6.2)
+    blur_weighted = cv2.GaussianBlur(weighted, (0, 0), 6.2)
+    base = blur_weighted / np.maximum(blur_alpha[..., None], 0.035)
+    detail = (foreground - base) * alpha[..., None]
+
+    # Keep exact local colour at zero rotation while avoiding unstable edge detail.
+    base = np.clip(base, 0.0, 255.0).astype(np.float32)
+    detail = np.clip(detail, -115.0, 115.0).astype(np.float32)
+    return alpha.astype(np.float32), base, detail
+
+
+def make_shadow_asset(source, plate):
+    """Extract the real contact-shadow residual from the photographed plate."""
+    delta = source.astype(np.float32) - plate.astype(np.float32)
+    yy, xx = np.mgrid[0:H, 0:W].astype(np.float32)
+    ell = ((xx - 220.0) / 45.0) ** 2 + ((yy - 159.0) / 13.5) ** 2
+    matte = 1.0 - smoothstep(0.52, 1.10, ell)
+    ball_r = np.sqrt(((xx - 209.0) / 1.045) ** 2 + (yy - 136.0) ** 2)
+    matte *= smoothstep(22.0, 26.5, ball_r)
+    residual = delta * matte[..., None]
+    # Tight crop around the source contact shadow.
+    return residual[140:180, 169:270].copy(), (169, 140)
+
+
+def warp_to_canvas(array, center, scale, angle_degrees=0.0,
+                   interpolation=cv2.INTER_LINEAR):
+    """Rotate/scale a 72x72 canonical asset onto a full-frame canvas."""
+    h, w = array.shape[:2]
+    canonical_center = np.array([w / 2.0, h / 2.0], np.float32)
+    theta = np.deg2rad(angle_degrees)
+    c, s = np.cos(theta) * scale, np.sin(theta) * scale
+    matrix = np.array([[c, s, center[0] - c * canonical_center[0] - s * canonical_center[1]],
+                       [-s, c, center[1] + s * canonical_center[0] - c * canonical_center[1]]],
+                      dtype=np.float32)
+    return cv2.warpAffine(array, matrix, (W, H), flags=interpolation,
+                          borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+
+
+def build_frames():
+    source = np.asarray(Image.open(CASE_DIR / "first_frame.png").convert("RGB"))
+    assert source.shape == (H, W, 3)
+    plate = reconstruct_plate(source)
+    alpha, fixed_light, rotating_detail = make_ball_assets(source, plate)
+    shadow_crop, shadow_origin = make_shadow_asset(source, plate)
+
+    frames = np.empty((N, H, W, 3), dtype=np.uint8)
+    frames[0] = source  # Contract: the first frame is pixel-identical.
+
+    x0, y0 = 209.0, 136.0
+    x1 = 469.0
+    canonical_radius = 23.0
+
+    for index in range(1, N):
+        u = index / (N - 1.0)
+        # Mostly constant real-time rolling, with imperceptibly soft release/catch.
+        travel = u - 0.018 * np.sin(2.0 * np.pi * u)
+        x = x0 + (x1 - x0) * travel
+
+        # The glass slopes subtly away in the image. The broad transition starts
+        # as the leading edge of the ball reaches the table/glass boundary.
+        glass_amount = smoothstep(276.0, 326.0, x)
+        y_glass = y0 + 0.078 * max(0.0, x - 281.0)
+        y = (1.0 - glass_amount) * y0 + glass_amount * y_glass
+
+        # Perspective change is slight in this locked, medium-distance view.
+        scale = 1.0 - 0.075 * u
+        distance = x - x0
+        rotation = -np.degrees(distance / (canonical_radius * scale))
+
+        frame = plate.astype(np.float32)
+
+        # Move the photographed shadow residual. On glass it becomes fainter and
+        # slightly more diffuse, as expected from a transparent support.
+        sh_h, sh_w = shadow_crop.shape[:2]
+        sh_center0 = (shadow_origin[0] + sh_w / 2.0,
+                      shadow_origin[1] + sh_h / 2.0)
+        sh_center = (sh_center0[0] + (x - x0),
+                     sh_center0[1] + (y - y0))
+        shadow_scale = scale * (1.0 + 0.07 * glass_amount)
+        shadow_canvas = warp_to_canvas(shadow_crop, sh_center, shadow_scale, 0.0)
+        shadow_strength = 1.0 - 0.52 * glass_amount
+        if glass_amount > 0.0:
+            softened = cv2.GaussianBlur(shadow_canvas, (0, 0), 1.25 * glass_amount)
+            shadow_canvas = (1.0 - glass_amount) * shadow_canvas + glass_amount * softened
+        frame += shadow_canvas * shadow_strength
+
+        # Rotate detail in canonical coordinates, then place it with camera-fixed
+        # broad lighting. This is more convincing than rotating the highlight.
+        detail_center = (rotating_detail.shape[1] / 2.0,
+                         rotating_detail.shape[0] / 2.0)
+        rot_m = cv2.getRotationMatrix2D(detail_center, rotation, 1.0)
+        spun_detail = cv2.warpAffine(rotating_detail, rot_m,
+                                     (rotating_detail.shape[1], rotating_detail.shape[0]),
+                                     flags=cv2.INTER_LINEAR,
+                                     borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+        canonical_rgb = np.clip(fixed_light + spun_detail, 0.0, 255.0)
+        premultiplied = canonical_rgb * alpha[..., None]
+
+        ball_pm = warp_to_canvas(premultiplied, (x, y), scale, 0.0)
+        ball_alpha = warp_to_canvas(alpha, (x, y), scale, 0.0)
+        ball_alpha = np.clip(ball_alpha, 0.0, 1.0)
+        frame = frame * (1.0 - ball_alpha[..., None]) + ball_pm
+
+        frames[index] = np.clip(frame + 0.5, 0, 255).astype(np.uint8)
+
+    return frames
+
+
+def main():
+    frames = build_frames()
+    artifact_io.save(frames, CASE_DIR)
+    print(f"saved {len(frames)} RGB frames at {W}x{H}")
+
+
+if __name__ == "__main__":
+    main()
