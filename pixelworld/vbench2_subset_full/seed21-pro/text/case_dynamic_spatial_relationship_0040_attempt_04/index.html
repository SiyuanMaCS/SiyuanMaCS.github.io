+"""Procedural text-to-video generator (no external assets/models).
+
+Scene from prompt.txt: a bird is on the right of an apple, then the bird
+flies to the front of the apple. Renders 20 RGB frames at 256x144 / 4 fps.
+Everything is drawn with PIL at 2x resolution and downsampled for AA.
+"""
+import math
+
+import numpy as np
+from PIL import Image, ImageDraw, ImageFilter
+
+import artifact_io
+
+W, H, N, FPS = 256, 144, 20, 4.0
+S = 2                      # supersampling factor
+SW, SH = W * S, H * S
+GROUND_Y = 127.0           # logical y of the ground line
+
+# Fixed layout (logical coordinates)
+APPLE = (104.0, 94.0, 23.0)          # center x, center y, radius
+BIRD_START = (207.0, 80.0)
+BIRD_END = (104.0, 99.0)             # overlaps apple -> bird drawn in front
+MOVE_T0, MOVE_T1 = 3, 15             # frame indices of flight window
+
+
+def smoothstep(t):
+    t = min(1.0, max(0.0, t))
+    return t * t * (3.0 - 2.0 * t)
+
+
+def lerp(a, b, t):
+    return a + (b - a) * t
+
+
+def vertical_gradient(top, bottom):
+    t = np.linspace(0.0, 1.0, SH, dtype=np.float32)[:, None, None]
+    top = np.asarray(top, dtype=np.float32)
+    bottom = np.asarray(bottom, dtype=np.float32)
+    arr = top[None, None, :] * (1.0 - t) + bottom[None, None, :] * t
+    arr = np.repeat(arr.astype(np.uint8), SW, axis=1)
+    return Image.fromarray(arr, "RGB").convert("RGBA")
+
+
+def draw_cloud(layer, x, y, scale):
+    """Soft white cloud centered at logical (x, y)."""
+    d = ImageDraw.Draw(layer)
+    puffs = [(0, 0, 13), (13, 4, 10), (-13, 5, 9), (6, -6, 9), (-5, -5, 8)]
+    for dx, dy, r in puffs:
+        cx, cy, rr = (x + dx * scale) * S, (y + dy * scale) * S, r * scale * S
+        d.ellipse([cx - rr, cy - rr, cx + rr, cy + rr], fill=(255, 255, 255, 235))
+    blur = layer.filter(ImageFilter.GaussianBlur(1.2 * S))
+    layer.alpha_composite(blur)
+
+
+def draw_apple(base, time_s):
+    """Red apple with stem, leaf, shading and a ground shadow."""
+    ax, ay, r = APPLE
+    shadow = Image.new("RGBA", base.size, (0, 0, 0, 0))
+    sd = ImageDraw.Draw(shadow)
+    sd.ellipse([(ax - r * 1.15) * S, (GROUND_Y - 5) * S,
+                (ax + r * 1.15) * S, (GROUND_Y + 5) * S], fill=(40, 60, 35, 80))
+    base.alpha_composite(shadow.filter(ImageFilter.GaussianBlur(2 * S)))
+
+    d = ImageDraw.Draw(base)
+    cx, cy, rr = ax * S, ay * S, r * S
+    # body and shaded side
+    d.ellipse([cx - rr, cy - rr, cx + rr, cy + rr], fill=(216, 49, 49, 255))
+    shade = Image.new("RGBA", base.size, (0, 0, 0, 0))
+    shd = ImageDraw.Draw(shade)
+    shd.ellipse([cx - rr * 0.15, cy - rr * 0.9, cx + rr * 1.05, cy + rr * 1.02],
+                fill=(150, 22, 34, 120))
+    base.alpha_composite(shade.filter(ImageFilter.GaussianBlur(rr * 0.25)))
+    # top dip
+    d.ellipse([cx - rr * 0.28, cy - rr * 1.08, cx + rr * 0.28, cy - rr * 0.72],
+              fill=(150, 22, 34, 255))
+    # glossy highlight
+    glow = Image.new("RGBA", base.size, (0, 0, 0, 0))
+    gd = ImageDraw.Draw(glow)
+    gd.ellipse([cx - rr * 0.72, cy - rr * 0.78, cx - rr * 0.08, cy - rr * 0.18],
+               fill=(255, 178, 178, 150))
+    base.alpha_composite(glow.filter(ImageFilter.GaussianBlur(rr * 0.18)))
+    # stem and leaf
+    d.line([(cx, cy - rr * 0.92), (cx + rr * 0.12, cy - rr * 1.35)],
+           fill=(112, 74, 40, 255), width=max(2, int(2.4 * S)))
+    leaf = Image.new("RGBA", base.size, (0, 0, 0, 0))
+    ld = ImageDraw.Draw(leaf)
+    lx, ly = cx + rr * 0.14, cy - rr * 1.32
+    ld.ellipse([lx, ly - 4 * S, lx + 13 * S, ly + 3 * S], fill=(72, 152, 62, 255))
+    leaf = leaf.rotate(-24, center=(lx, ly), resample=Image.BICUBIC)
+    base.alpha_composite(leaf)
+
+
+def build_bird_tile(wing_deg, settled):
+    """Cartoon bluebird facing left, returned as an RGBA supersampled tile."""
+    tw, th = 50 * S, 38 * S
+    tile = Image.new("RGBA", (tw, th), (0, 0, 0, 0))
+    d = ImageDraw.Draw(tile)
+    cx, cy = 25 * S, 20 * S            # bird pivot inside the tile
+
+    # tail feathers (trailing to the right)
+    for k, dy in enumerate((-7, 0, 7)):
+        d.polygon([(cx + 9 * S, cy + dy * 0.4 * S),
+                   (cx + 21 * S, cy + (dy - 3) * S),
+                   (cx + 21 * S, cy + (dy + 3) * S)],
+                  fill=(40, 88, 165, 255))
+    # body
+    d.ellipse([cx - 12 * S, cy - 9 * S, cx + 12 * S, cy + 10 * S],
+              fill=(58, 120, 205, 255))
+    # warm belly
+    d.ellipse([cx - 10 * S, cy - 1 * S, cx + 6 * S, cy + 10 * S],
+              fill=(250, 206, 132, 255))
+    # head
+    d.ellipse([cx - 17 * S, cy - 13 * S, cx - 3 * S, cy + 1 * S],
+              fill=(66, 130, 214, 255))
+    # beak (points left)
+    d.polygon([(cx - 16 * S, cy - 8 * S),
+               (cx - 22 * S, cy - 5.5 * S),
+               (cx - 16 * S, cy - 3 * S)], fill=(240, 150, 50, 255))
+    # eye with glint
+    ex, ey = cx - 11.5 * S, cy - 7.5 * S
+    d.ellipse([ex - 2.6 * S, ey - 2.6 * S, ex + 2.6 * S, ey + 2.6 * S],
+              fill=(255, 255, 255, 255))
+    d.ellipse([ex - 1.4 * S, ey - 1.4 * S, ex + 1.4 * S, ey + 1.4 * S],
+              fill=(25, 30, 40, 255))
+    d.ellipse([ex - 1.6 * S, ey - 1.8 * S, ex - 0.6 * S, ey - 0.8 * S],
+              fill=(255, 255, 255, 255))
+
+    # flapping wing, rotated around its leading edge
+    wing = Image.new("RGBA", (40 * S, 40 * S), (0, 0, 0, 0))
+    wd = ImageDraw.Draw(wing)
+    wd.ellipse([3 * S, 14 * S, 30 * S, 26 * S], fill=(40, 88, 165, 255))
+    wing = wing.rotate(wing_deg, resample=Image.BICUBIC, center=(6 * S, 20 * S))
+    tile.alpha_composite(wing, (int(cx - 6 * S), int(cy - 20 * S)))
+
+    if settled:
+        # tiny folded feet when the bird has landed in front of the apple
+        d.line([(cx - 4 * S, cy + 9 * S), (cx - 4 * S, cy + 14 * S)],
+               fill=(214, 130, 45, 255), width=max(1, S))
+        d.line([(cx + 3 * S, cy + 9 * S), (cx + 3 * S, cy + 14 * S)],
+               fill=(214, 130, 45, 255), width=max(1, S))
+    return tile
+
+
+def draw_bird(base, x, y, wing_deg, pitch_deg, settled):
+    tile = build_bird_tile(wing_deg, settled)
+    tile = tile.rotate(pitch_deg, resample=Image.BICUBIC,
+                       center=(25 * S, 20 * S), expand=True)
+    px = int(round(x * S - tile.width / 2))
+    py = int(round(y * S - tile.height / 2))
+    base.alpha_composite(tile, (px, py))
+
+
+def draw_shadow(base, x, y):
+    height = max(0.0, GROUND_Y - y)
+    closeness = math.exp(-height / 52.0)
+    layer = Image.new("RGBA", base.size, (0, 0, 0, 0))
+    d = ImageDraw.Draw(layer)
+    half_w = (8 + 20 * closeness) * S
+    d.ellipse([x * S - half_w, (GROUND_Y - 3.5) * S,
+               x * S + half_w, (GROUND_Y + 3.5) * S],
+              fill=(35, 55, 32, int(95 * closeness)))
+    base.alpha_composite(layer.filter(ImageFilter.GaussianBlur(1.6 * S)))
+
+
+def render_frame(t):
+    time_s = t / FPS
+    scene = vertical_gradient((126, 188, 230), (207, 233, 246))
+    d = ImageDraw.Draw(scene)
+
+    # soft sun
+    sun = Image.new("RGBA", scene.size, (0, 0, 0, 0))
+    sd = ImageDraw.Draw(sun)
+    sx, sy, sr = 42 * S, 28 * S, 13 * S
+    sd.ellipse([sx - sr, sy - sr, sx + sr, sy + sr], fill=(255, 238, 195, 220))
+    scene.alpha_composite(sun.filter(ImageFilter.GaussianBlur(3 * S)))
+
+    # drifting clouds (wrap around the frame)
+    for cx0, cy0, sc in ((70, 26, 1.0), (165, 18, 0.8), (225, 40, 0.65)):
+        cx = (cx0 - time_s * 2.2) % (W + 70) - 35
+        cloud_layer = Image.new("RGBA", scene.size, (0, 0, 0, 0))
+        draw_cloud(cloud_layer, cx, cy0, sc)
+        scene.alpha_composite(cloud_layer)
+
+    # distant hills and ground
+    d.ellipse([-30 * S, (GROUND_Y - 42) * S, 120 * S, (GROUND_Y + 30) * S],
+              fill=(152, 197, 142, 255))
+    d.ellipse([120 * S, (GROUND_Y - 34) * S, 290 * S, (GROUND_Y + 30) * S],
+              fill=(140, 188, 132, 255))
+    d.rectangle([0, GROUND_Y * S, SW, SH], fill=(112, 174, 92, 255))
+    d.rectangle([0, GROUND_Y * S, SW, (GROUND_Y + 5) * S], fill=(96, 156, 78, 255))
+    # grass tufts and small flowers
+    rng = np.random.default_rng(7)
+    for gx in rng.uniform(4, W - 4, 26):
+        gy = rng.uniform(GROUND_Y + 7, H - 4)
+        d.line([gx * S, gy * S, (gx - 2) * S, (gy - 4) * S],
+               fill=(80, 140, 66, 255), width=S)
+        d.line([gx * S, gy * S, (gx + 2) * S, (gy - 4) * S],
+               fill=(80, 140, 66, 255), width=S)
+    for fx, fy, col in ((30, 136, (255, 240, 150)), (188, 134, (245, 200, 230)),
+                        (150, 139, (255, 240, 150)), (232, 137, (245, 200, 230))):
+        d.ellipse([(fx - 1.6) * S, (fy - 1.6) * S, (fx + 1.6) * S, (fy + 1.6) * S],
+                  fill=col + (255,))
+
+    # flight progress: hold on the right, ease across, settle in front
+    p = smoothstep((t - MOVE_T0) / (MOVE_T1 - MOVE_T0))
+    settle = smoothstep((t - MOVE_T1) / 3.0)
+    bx = lerp(BIRD_START[0], BIRD_END[0], p)
+    by = lerp(BIRD_START[1], BIRD_END[1], p) - 12.0 * math.sin(math.pi * p)
+    if p < 1.0:
+        by += 2.2 * math.sin(2 * math.pi * 2.6 * time_s)
+    else:
+        by += 1.1 * math.sin(2 * math.pi * 1.3 * time_s)
+
+    draw_shadow(scene, bx, by)
+    draw_apple(scene, time_s)
+
+    # wing beat: strong while flying, gentle folded motion once settled
+    amp = lerp(34.0, 9.0, settle)
+    wing_deg = amp * math.sin(2 * math.pi * 3.0 * time_s)
+    pitch = 9.0 * math.sin(math.pi * p)          # nose-up arc while flying
+    draw_bird(scene, bx, by, wing_deg, pitch, settled=settle > 0.6)
+
+    rgb = scene.convert("RGB").resize((W, H), Image.LANCZOS)
+    return np.asarray(rgb, dtype=np.uint8)
+
+
+def main():
+    frames = np.stack([render_frame(t) for t in range(N)], axis=0)
+    artifact_io.save(frames, ".")
+    print(f"saved {frames.shape} frames ({N} at {W}x{H}, {FPS} fps)")
+
+
+if __name__ == "__main__":
+    main()
