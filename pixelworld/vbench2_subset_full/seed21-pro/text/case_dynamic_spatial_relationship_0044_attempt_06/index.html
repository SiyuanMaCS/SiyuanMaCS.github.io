+"""Procedural text-to-video generator.
+
+Prompt: "A bird is above a tree, then the bird flies to the left of the tree."
+
+Renders 20 RGB frames (256x144 @ 4fps) by drawing on a 4x supersampled
+PIL canvas and downsampling. No external assets, models, or network.
+"""
+import math
+from pathlib import Path
+
+import numpy as np
+from PIL import Image, ImageDraw
+
+import artifact_io
+
+CASE_DIR = Path(__file__).parent
+W, H, N, FPS, SS = 256, 144, 20, 4.0, 4
+SW, SH = W * SS, H * SS
+GROUND_Y = 112  # output-space horizon / ground line
+
+# Palette
+SKY_TOP, SKY_BOT = (122, 184, 234), (206, 232, 247)
+SUN_CORE, SUN_GLOW = (255, 240, 184), (255, 226, 158)
+CLOUD = (246, 250, 253)
+GROUND_TOP, GROUND_BOT = (108, 170, 86), (80, 142, 66)
+TRUNK, TRUNK_DARK = (126, 86, 54), (102, 68, 42)
+LEAF_DARK, LEAF_MID, LEAF_LIGHT = (52, 118, 58), (74, 146, 70), (112, 178, 92)
+BIRD_BODY, BIRD_BELLY, BIRD_WING, BIRD_TAIL = (62, 128, 206), (158, 202, 240), (40, 88, 158), (36, 64, 122)
+BEAK, EYE = (238, 172, 58), (24, 30, 44)
+
+
+def lerp(a, b, t):
+    return a + (b - a) * t
+
+
+def smoothstep(t):
+    t = max(0.0, min(1.0, t))
+    return t * t * (3 - 2 * t)
+
+
+def make_background():
+    """Sky gradient, sun, ground gradient. Static across frames."""
+    bg = Image.new("RGB", (SW, SH))
+    px = bg.load()
+    for y in range(SH):
+        t = y / (GROUND_Y * SS - 1)
+        if t <= 1.0:
+            col = tuple(int(lerp(SKY_TOP[i], SKY_BOT[i], min(1.0, t))) for i in range(3))
+        else:
+            g = (y - GROUND_Y * SS) / (SH - GROUND_Y * SS)
+            col = tuple(int(lerp(GROUND_TOP[i], GROUND_BOT[i], g)) for i in range(3))
+        for x in range(SW):
+            px[x, y] = col
+    d = ImageDraw.Draw(bg)
+    # Sun with a soft two-ring glow, upper right.
+    sx, sy = 214 * SS, 26 * SS
+    d.ellipse([sx - 22 * SS, sy - 22 * SS, sx + 22 * SS, sy + 22 * SS], fill=SUN_GLOW)
+    d.ellipse([sx - 16 * SS, sy - 16 * SS, sx + 16 * SS, sy + 16 * SS], fill=(255, 234, 170))
+    d.ellipse([sx - 11 * SS, sy - 11 * SS, sx + 11 * SS, sy + 11 * SS], fill=SUN_CORE)
+    return bg
+
+
+def draw_cloud(d, cx, cy, scale):
+    for ox, oy, r in [(-14, 2, 9), (-4, -3, 11), (8, -1, 10), (16, 3, 8), (2, 4, 10)]:
+        rr = r * scale
+        d.ellipse(
+            [(cx + ox * scale - rr) * SS, (cy + oy * scale - rr) * SS,
+             (cx + ox * scale + rr) * SS, (cy + oy * scale + rr) * SS],
+            fill=CLOUD,
+        )
+
+
+def draw_tree(d):
+    # Trunk with shaded side.
+    d.rectangle([146 * SS, 92 * SS, 154 * SS, GROUND_Y * SS], fill=TRUNK)
+    d.rectangle([154 * SS, 92 * SS, 158 * SS, GROUND_Y * SS], fill=TRUNK_DARK)
+    # A small branch reaching left.
+    d.polygon([(148 * SS, 100 * SS), (134 * SS, 90 * SS), (136 * SS, 87 * SS),
+               (150 * SS, 96 * SS)], fill=TRUNK)
+    # Foliage: layered clusters, dark back to light front.
+    clusters = [
+        (LEAF_DARK, [(152, 74, 27), (134, 82, 18), (170, 82, 18), (152, 90, 19)]),
+        (LEAF_MID, [(146, 72, 20), (162, 76, 19), (152, 84, 20), (136, 84, 13)]),
+        (LEAF_LIGHT, [(144, 68, 11), (160, 72, 10), (150, 78, 12)]),
+    ]
+    for color, blobs in clusters:
+        for cx, cy, r in blobs:
+            d.ellipse([(cx - r) * SS, (cy - r) * SS, (cx + r) * SS, (cy + r) * SS], fill=color)
+
+
+def draw_grass(d, rng):
+    for _ in range(46):
+        x = int(rng.integers(4, W - 4))
+        y = int(rng.integers(GROUND_Y + 2, H - 2))
+        h = int(rng.integers(2, 5))
+        col = (74, 132, 60)
+        d.line([x * SS, y * SS, (x - 1) * SS, (y - h) * SS], fill=col, width=SS)
+        d.line([x * SS, y * SS, (x + 1) * SS, (y - h) * SS], fill=col, width=SS)
+    for _ in range(10):
+        x = int(rng.integers(6, W - 6))
+        y = int(rng.integers(GROUND_Y + 6, H - 4))
+        flower = ((238, 214, 96), (240, 150, 170))[int(rng.integers(2))]
+        d.ellipse([(x - 1) * SS, (y - 1) * SS, (x + 1) * SS, (y + 1) * SS],
+                  fill=flower)
+
+
+def draw_bird(d, cx, cy, flap):
+    """Bird faces left; flap in [-1, 1] drives the wing pose."""
+    def p(pt):
+        return ((cx + pt[0]) * SS, (cy + pt[1]) * SS)
+
+    # Tail, body, head, beak, eye (front is the left side).
+    d.polygon([p((8, -2)), p((17, -6)), p((18, -1)), p((15, 3)), p((8, 3))], fill=BIRD_TAIL)
+    d.ellipse([(cx - 8) * SS, (cy - 5) * SS, (cx + 8) * SS, (cy + 5) * SS], fill=BIRD_BODY)
+    d.ellipse([(cx - 7) * SS, (cy - 1) * SS, (cx + 5) * SS, (cy + 5) * SS], fill=BIRD_BELLY)
+    d.ellipse([(cx - 13) * SS, (cy - 8) * SS, (cx - 3) * SS, (cy + 2) * SS], fill=BIRD_BODY)
+    d.polygon([p((-13, -5)), p((-19, -2)), p((-13, 0))], fill=BEAK)
+    d.ellipse([(cx - 11) * SS, (cy - 6) * SS, (cx - 8) * SS, (cy - 3) * SS], fill=EYE)
+    d.ellipse([(cx - 10.4) * SS, (cy - 5.6) * SS, (cx - 9.4) * SS, (cy - 4.6) * SS],
+              fill=(255, 255, 255))
+    # Flapping wing: shoulder fixed near body center, tip sweeps up/down.
+    wing = [(-1, -2), (6, -3 - 6 * flap), (8, 1 - 2 * flap), (2, 2)]
+    d.polygon([p(pt) for pt in wing], fill=BIRD_WING)
+
+
+def main():
+    rng = np.random.default_rng(20261111)
+    base = make_background()
+    static = base.copy()
+    sd = ImageDraw.Draw(static)
+    draw_grass(sd, rng)
+    draw_tree(sd)
+
+    # Bird keyframes: above the tree (over the canopy) then left of the tree.
+    start_xy, end_xy = (158.0, 40.0), (56.0, 80.0)
+    frames = []
+    for t in range(N):
+        img = static.copy()
+        d = ImageDraw.Draw(img)
+        # Slow drifting clouds.
+        drift = t * 0.18
+        draw_cloud(d, 46 + drift, 26, 1.0)
+        draw_cloud(d, 112 - drift * 0.6, 16, 0.75)
+
+        # Hold above the tree briefly, ease across, settle on the left.
+        p = smoothstep((t - 2) / 15.0)
+        cx = lerp(start_xy[0], end_xy[0], p)
+        arc = math.sin(p * math.pi)  # gentle rise and fall along the flight
+        cy = lerp(start_xy[1], end_xy[1], p) - 10 * arc + 1.2 * math.sin(t * 1.7)
+        flap = math.sin(t * 1.9)
+
+        # Faint moving shadow on the ground.
+        sh_w = 10 - 0.04 * (GROUND_Y - cy)
+        d.ellipse([(cx - sh_w) * SS, (GROUND_Y + 4) * SS,
+                   (cx + sh_w) * SS, (GROUND_Y + 7) * SS], fill=(86, 132, 64))
+        draw_bird(d, cx, cy, flap)
+
+        frame = img.resize((W, H), Image.Resampling.LANCZOS)
+        frames.append(np.asarray(frame, dtype=np.uint8))
+
+    frames = np.stack(frames)
+    artifact_io.save(frames, CASE_DIR)
+    print(f"saved {frames.shape} frames to {CASE_DIR}")
+
+
+if __name__ == "__main__":
+    main()
