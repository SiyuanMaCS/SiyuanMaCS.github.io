+"""Procedural text-to-video generator.
+
+Scene: a bird stands on the ground to the right of a tree, then takes off,
+flaps along an arcing flight path, and perches on top of (above) the tree.
+Renders at 3x supersampling and downsamples for smooth edges.
+"""
+import math
+from pathlib import Path
+
+import numpy as np
+from PIL import Image, ImageDraw
+
+import artifact_io
+
+CASE_DIR = Path(__file__).parent
+W, H, N, FPS = 256, 144, 20, 4.0
+S = 3  # supersampling factor
+
+# Key positions (native 256x144 coordinates), bird body center.
+GROUND_Y = 118.0
+TREE_X = 87.0
+START_X, START_Y = 175.0, 104.0
+END_X, END_Y = 87.0, 24.0
+FLIGHT_START, FLIGHT_END = 4, 16  # frame indices
+
+SKY_TOP = (126, 188, 232)
+SKY_BOT = (219, 240, 250)
+SUN_COLOR = (255, 236, 170)
+CLOUD_COLOR = (255, 255, 255)
+HILL_FAR = (138, 190, 128)
+HILL_NEAR = (104, 168, 96)
+GROUND_TOP = (111, 174, 78)
+GROUND_BOT = (74, 128, 52)
+TRUNK = (122, 79, 44)
+TRUNK_DARK = (92, 58, 30)
+LEAF_DARK = (40, 110, 46)
+LEAF_MID = (58, 142, 58)
+LEAF_LIGHT = (96, 178, 84)
+BIRD_BLUE = (67, 128, 206)
+BIRD_BLUE_DARK = (38, 88, 158)
+BIRD_BELLY = (214, 230, 246)
+BEAK = (242, 146, 43)
+LEG = (122, 82, 40)
+
+
+def sc(v):
+    return v * S
+
+
+def make_background():
+    """Static landscape rendered once at supersampled resolution."""
+    sw, sh = W * S, H * S
+    grad = np.zeros((sh, sw, 3), dtype=np.uint8)
+    for y in range(sh):
+        t = y / (sh - 1)
+        grad[y, :, 0] = int(SKY_TOP[0] * (1 - t) + SKY_BOT[0] * t)
+        grad[y, :, 1] = int(SKY_TOP[1] * (1 - t) + SKY_BOT[1] * t)
+        grad[y, :, 2] = int(SKY_TOP[2] * (1 - t) + SKY_BOT[2] * t)
+    img = Image.fromarray(grad, "RGB")
+    d = ImageDraw.Draw(img, "RGBA")
+
+    # Sun with soft glow.
+    sx, sy = sc(38), sc(26)
+    for r, alpha in ((26, 26), (20, 40), (14, 90)):
+        d.ellipse([sx - sc(r), sy - sc(r), sx + sc(r), sy + sc(r)],
+                  fill=SUN_COLOR + (alpha,))
+    d.ellipse([sx - sc(9), sy - sc(9), sx + sc(9), sy + sc(9)],
+              fill=(255, 244, 196, 255))
+
+    # Layered hills.
+    horizon = sc(96)
+    d.pieslice([sc(-30), horizon - sc(46), sc(120), horizon + sc(46)],
+               180, 360, fill=HILL_FAR)
+    d.pieslice([sc(120), horizon - sc(38), sc(300), horizon + sc(40)],
+               180, 360, fill=HILL_FAR)
+    d.pieslice([sc(60), horizon - sc(26), sc(210), horizon + sc(34)],
+               180, 360, fill=HILL_NEAR)
+
+    # Ground with vertical gradient.
+    gy = sc(GROUND_Y)
+    d.rectangle([0, gy, sw, sh], fill=GROUND_TOP)
+    for i in range(int(sh - gy)):
+        t = i / max(1, sh - gy)
+        col = tuple(int(GROUND_TOP[c] * (1 - t) + GROUND_BOT[c] * t) for c in range(3))
+        d.line([(0, gy + i), (sw, gy + i)], fill=col)
+
+    # Static tree shadow.
+    d.ellipse([sc(58), sc(114), sc(128), sc(124)], fill=(40, 80, 30, 70))
+
+    # Tree trunk (slight taper).
+    d.polygon([(sc(80), sc(118)), (sc(96), sc(118)),
+               (sc(91), sc(74)), (sc(85), sc(74))],
+              fill=TRUNK, outline=TRUNK_DARK)
+    d.line([(sc(84), sc(116)), (sc(86), sc(78))], fill=TRUNK_DARK, width=sc(1))
+    d.line([(sc(88), sc(82)), (sc(74), sc(66))], fill=TRUNK, width=sc(4))
+
+    # Foliage: layered clusters, dark base then mid-tone puffs.
+    clusters = [
+        (87, 64, 28), (65, 72, 18), (110, 72, 18),
+        (76, 50, 17), (100, 50, 16), (88, 44, 15),
+    ]
+    for cx, cy, r in clusters:
+        d.ellipse([sc(cx - r), sc(cy - r), sc(cx + r), sc(cy + r)],
+                  fill=LEAF_DARK)
+    for cx, cy, r in clusters:
+        d.ellipse([sc(cx - r + 1.5), sc(cy - r + 1.5),
+                   sc(cx + r - 1.5), sc(cy + r - 1.5)], fill=LEAF_MID)
+    for cx, cy, r in [(80, 52, 10), (99, 58, 8), (70, 64, 7), (92, 40, 6)]:
+        d.ellipse([sc(cx - r), sc(cy - r), sc(cx + r), sc(cy + r)],
+                  fill=LEAF_LIGHT)
+
+    # Grass tufts.
+    rng = np.random.default_rng(7)
+    for _ in range(26):
+        x = float(rng.integers(4, W - 4))
+        if 70 < x < 112:
+            continue
+        h = float(rng.integers(3, 7))
+        d.line([(sc(x), sc(GROUND_Y)), (sc(x - 1.5), sc(GROUND_Y - h))],
+               fill=(52, 110, 40), width=sc(1))
+        d.line([(sc(x), sc(GROUND_Y)), (sc(x + 1.5), sc(GROUND_Y - h))],
+               fill=(52, 110, 40), width=sc(1))
+
+    # Small flowers.
+    for fx, fy, petal in [(24, 114, (240, 120, 140)), (150, 112, (250, 210, 90)),
+                          (214, 115, (200, 140, 230)), (132, 116, (245, 150, 90)),
+                          (236, 110, (240, 120, 140))]:
+        for a in range(0, 360, 90):
+            rad = math.radians(a)
+            px, py = fx + 2.2 * math.cos(rad), fy + 2.2 * math.sin(rad)
+            d.ellipse([sc(px - 1.4), sc(py - 1.4), sc(px + 1.4), sc(py + 1.4)],
+                      fill=petal)
+        d.ellipse([sc(fx - 1.2), sc(fy - 1.2), sc(fx + 1.2), sc(fy + 1.2)],
+                  fill=(250, 230, 120))
+
+    clouds = [(48, 22), (150, 14), (214, 34)]
+    return img, clouds
+
+
+def draw_cloud(img, cx, cy):
+    d = ImageDraw.Draw(img, "RGBA")
+    for dx, dy, r in [(-9, 1, 7), (0, -3, 9), (9, 0, 7), (0, 3, 7)]:
+        d.ellipse([sc(cx + dx - r), sc(cy + dy - r),
+                   sc(cx + dx + r), sc(cy + dy + r)],
+                  fill=CLOUD_COLOR + (210,))
+
+
+def draw_bird(flap, pitch, legs_down, blink):
+    """Draw the bird facing left on a transparent layer."""
+    lw, lh = 96, 78
+    layer = Image.new("RGBA", (sc(lw), sc(lh)), (0, 0, 0, 0))
+    d = ImageDraw.Draw(layer)
+    ox, oy = lw / 2, lh / 2
+
+    def P(x, y):
+        return (sc(ox + x), sc(oy + y))
+
+    def ell(x0, y0, x1, y1, fill, outline=None):
+        d.ellipse([sc(ox + x0), sc(oy + y0), sc(ox + x1), sc(oy + y1)],
+                  fill=fill, outline=outline)
+
+    def poly(pts, fill, outline=None):
+        d.polygon([P(x, y) for x, y in pts], fill=fill, outline=outline)
+
+    def line(pts, fill, width=1):
+        d.line([P(x, y) for x, y in pts], fill=fill, width=sc(width))
+
+    # Legs (behind body).
+    if legs_down:
+        for lx in (-3, 4):
+            line([(lx, 6), (lx, 13)], LEG, 2)
+            line([(lx, 13), (lx - 3, 13)], LEG, 2)
+    else:
+        line([(-2, 6), (2, 9)], LEG, 1)
+        line([(3, 6), (6, 9)], LEG, 1)
+
+    # Tail feathers.
+    poly([(9, -3), (21, -7), (18, -1), (10, 1)], BIRD_BLUE_DARK)
+    poly([(9, 1), (20, 4), (17, 8), (10, 4)], BIRD_BLUE_DARK)
+
+    # Body and belly.
+    ell(-11, -8, 11, 8, BIRD_BLUE, BIRD_BLUE_DARK)
+    ell(-8, -1, 8, 8, BIRD_BELLY)
+
+    # Animated wing (tip lifts with flap).
+    poly([(1, -3), (15, -7 - flap * 11), (18, -1 - flap * 6), (9, 4)],
+         BIRD_BLUE_DARK)
+    poly([(3, -2), (12, -5 - flap * 8), (13, -2 - flap * 4), (8, 2)],
+         BIRD_BLUE)
+
+    # Head, beak, eye.
+    ell(-16, -14, -3, -2, BIRD_BLUE, BIRD_BLUE_DARK)
+    poly([(-15, -10), (-22, -8), (-15, -5)], BEAK, (190, 105, 25))
+    if blink:
+        line([(-12.5, -9.5), (-9, -9.5)], (30, 40, 60), 1)
+    else:
+        ell(-12.5, -11.5, -8.5, -7.5, (255, 255, 255))
+        ell(-11.6, -10.6, -9.2, -8.2, (25, 35, 55))
+        ell(-11.1, -10.1, -10.1, -9.1, (255, 255, 255))
+
+    if pitch:
+        layer = layer.rotate(pitch, resample=Image.Resampling.BICUBIC, expand=True)
+    return layer
+
+
+def bird_pose(frame):
+    """Return (center_x, center_y, flap, pitch, legs_down, blink)."""
+    blink = frame in (2, 18)
+    if frame < FLIGHT_START:
+        return START_X, START_Y + math.sin(frame * 1.4) * 0.6, -0.6, 0.0, True, blink
+    if frame >= FLIGHT_END:
+        t = frame - FLIGHT_END
+        return END_X, END_Y + math.sin(t * 1.1) * 0.5, -0.7, 0.0, True, blink
+
+    t = (frame - FLIGHT_START) / (FLIGHT_END - FLIGHT_START)
+    s = t * t * (3 - 2 * t)  # smoothstep
+    x = START_X + (END_X - START_X) * s
+    y = (START_Y + (END_Y - START_Y) * s) - 18.0 * math.sin(math.pi * t)
+
+    # Pitch from the local flight direction.
+    t2 = min(1.0, t + 0.02)
+    s2 = t2 * t2 * (3 - 2 * t2)
+    vx = (END_X - START_X) * (s2 - s)
+    vy = ((END_Y - START_Y) * (s2 - s)
+          - 18.0 * (math.sin(math.pi * t2) - math.sin(math.pi * t)))
+    pitch = max(-28.0, min(28.0, math.degrees(math.atan2(-vy, -vx))))
+
+    if t < 0.15:
+        flap = 1.0
+    elif t > 0.85:
+        flap = 0.8 - (t - 0.85) / 0.15 * 1.5
+    else:
+        flap = math.sin((t - 0.15) / 0.7 * math.pi * 3.0)
+    return x, y, flap, pitch, False, blink
+
+
+def main():
+    background, clouds = make_background()
+    frames = []
+    for f in range(N):
+        img = background.copy()
+        draw_cloud(img, clouds[0][0] + f * 0.18, clouds[0][1])
+        draw_cloud(img, clouds[1][0] + f * 0.12, clouds[1][1])
+        draw_cloud(img, clouds[2][0] + f * 0.22, clouds[2][1])
+
+        bx, by, flap, pitch, legs_down, blink = bird_pose(f)
+
+        # Bird shadow on the ground, fading as the bird gains altitude.
+        height = max(0.0, GROUND_Y - (by + 8))
+        alpha = int(max(28, 115 - height * 1.15))
+        rx = max(5.0, 13.0 - height * 0.06)
+        d = ImageDraw.Draw(img, "RGBA")
+        d.ellipse([sc(bx - rx), sc(GROUND_Y + 2), sc(bx + rx), sc(GROUND_Y + 7)],
+                  fill=(35, 70, 28, alpha))
+
+        bird = draw_bird(flap, pitch, legs_down, blink)
+        img.paste(bird, (int(sc(bx) - bird.width / 2), int(sc(by) - bird.height / 2)), bird)
+
+        frames.append(np.asarray(img.resize((W, H), Image.Resampling.LANCZOS),
+                                 dtype=np.uint8))
+
+    frames = np.stack(frames, axis=0)
+    artifact_io.save(frames, CASE_DIR)
+    print(f"Saved {frames.shape} frames to {CASE_DIR}")
+
+
+if __name__ == "__main__":
+    main()
