+"""Procedural text-to-video generation.
+
+Prompt: "A fox is in front of a shoe, then the fox runs to the right of the shoe."
+
+Renders 20 RGB frames (256x144, 4 fps) entirely with PIL geometry: a fixed
+sneaker on grassland, and a cartoon fox that starts occluding the shoe
+(in front of it) and runs with an articulated leg cycle to its right.
+"""
+
+from pathlib import Path
+import math
+
+import numpy as np
+from PIL import Image, ImageDraw
+
+import artifact_io
+
+W, H, N, FPS = 256, 144, 20, 4.0
+S = 2  # supersampling factor
+
+
+def smoothstep(t):
+    return t * t * (3.0 - 2.0 * t)
+
+
+class Canvas:
+    """Draw wrapper that works in native 256x144 coords on a 2x buffer."""
+
+    def __init__(self):
+        self.img = Image.new("RGB", (W * S, H * S), (0, 0, 0))
+        self.d = ImageDraw.Draw(self.img)
+
+    def _box(self, b):
+        return tuple(int(round(v * S)) for v in b)
+
+    def _pts(self, pts):
+        return [(int(round(x * S)), int(round(y * S))) for x, y in pts]
+
+    def ellipse(self, b, fill):
+        self.d.ellipse(self._box(b), fill=fill)
+
+    def rectangle(self, b, fill):
+        self.d.rectangle(self._box(b), fill=fill)
+
+    def rounded(self, b, radius, fill):
+        self.d.rounded_rectangle(self._box(b), radius=int(radius * S), fill=fill)
+
+    def polygon(self, pts, fill):
+        self.d.polygon(self._pts(pts), fill=fill)
+
+    def line(self, pts, fill, width):
+        self.d.line(self._pts(pts), fill=fill, width=max(1, int(width * S)),
+                    joint="curve")
+
+    def to_frame(self):
+        return np.asarray(self.img.resize((W, H), Image.Resampling.LANCZOS),
+                          dtype=np.uint8)
+
+
+# ---------------------------------------------------------------- background
+def draw_background(c, rng, decor):
+    top = np.array([122, 178, 226], dtype=float)
+    horizon = np.array([198, 224, 240], dtype=float)
+    for y in range(92):
+        t = y / 91.0
+        col = tuple((top * (1 - t) + horizon * t).astype(int))
+        c.rectangle((0, y, W, y + 1), fill=col)
+
+    # soft distant hills
+    c.ellipse((-40, 74, 130, 104), fill=(128, 178, 120))
+    c.ellipse((110, 78, 290, 102), fill=(116, 168, 112))
+
+    # clouds
+    for cx0, cy0, s0 in ((42, 26, 1.0), (150, 18, 0.8), (214, 40, 0.9)):
+        cloud = (246, 250, 255)
+        c.ellipse((cx0 - 16 * s0, cy0 - 6 * s0, cx0 + 4 * s0, cy0 + 7 * s0), fill=cloud)
+        c.ellipse((cx0 - 4 * s0, cy0 - 10 * s0, cx0 + 14 * s0, cy0 + 7 * s0), fill=cloud)
+        c.ellipse((cx0 + 8 * s0, cy0 - 4 * s0, cx0 + 24 * s0, cy0 + 7 * s0), fill=cloud)
+
+    # grass
+    g_top = np.array([104, 168, 84], dtype=float)
+    g_bot = np.array([66, 128, 58], dtype=float)
+    for y in range(92, H):
+        t = (y - 92) / (H - 93)
+        col = tuple((g_top * (1 - t) + g_bot * t).astype(int))
+        c.rectangle((0, y, W, y + 1), fill=col)
+
+    # fixed grass tufts (precomputed so the background is static)
+    for x, y, lean in decor["tufts"]:
+        col = (58, 118, 52)
+        for k in range(3):
+            dx = (k - 1) * 2 + lean
+            c.line([(x, y), (x + dx, y - (4 + k))], fill=col, width=1)
+    for x, y, petal in decor["flowers"]:
+        c.line([(x, y), (x, y - 5)], fill=(50, 100, 45), width=1)
+        c.ellipse((x - 1.6, y - 7.5, x + 1.6, y - 4.5), fill=petal)
+
+
+# -------------------------------------------------------------------- shoe
+def draw_shoe(c, cx=120, base=122):
+    # ground shadow
+    c.ellipse((cx - 34, base - 6, cx + 34, base + 4), fill=(52, 86, 48))
+
+    # sole + midsole
+    c.rounded((cx - 30, base - 14, cx + 30, base), 5, fill=(44, 47, 54))
+    c.rounded((cx - 28, base - 18, cx + 28, base - 9), 4, fill=(238, 238, 234))
+    for k in range(6):  # tread notches
+        x0 = cx - 26 + k * 10
+        c.rectangle((x0, base - 3, x0 + 5, base), fill=(30, 32, 38))
+
+    # upper body
+    c.rounded((cx - 24, base - 36, cx + 20, base - 15), 7, fill=(198, 64, 48))
+    c.ellipse((cx + 10, base - 26, cx + 31, base - 12), fill=(172, 50, 38))  # toe cap
+    c.rounded((cx - 26, base - 30, cx - 14, base - 15), 4, fill=(172, 50, 38))  # heel
+
+    # collar opening and tongue
+    c.ellipse((cx - 20, base - 40, cx - 2, base - 28), fill=(52, 40, 38))
+    c.rounded((cx - 12, base - 42, cx + 7, base - 24), 3, fill=(214, 210, 200))
+
+    # laces
+    for k in range(3):
+        y0 = base - 37 + k * 4
+        c.line([(cx - 9, y0), (cx + 5, y0 + 3)], fill=(245, 245, 240), width=1.6)
+
+    # side stripe
+    c.line([(cx - 18, base - 17), (cx - 2, base - 24), (cx + 14, base - 22)],
+           fill=(245, 245, 240), width=3)
+
+
+# --------------------------------------------------------------------- fox
+ORANGE = (216, 118, 42)
+ORANGE_D = (190, 95, 30)
+ORANGE_FAR = (172, 84, 28)
+CREAM = (245, 232, 200)
+DARK = (58, 38, 28)
+
+
+def draw_leg(c, hip_x, hip_y, phase, far, speed):
+    col = ORANGE_FAR if far else ORANGE_D
+    if speed < 0.06:  # standing
+        foot_x, foot_y = hip_x, 0
+        knee_x, knee_y = hip_x, hip_y * 0.5
+    else:
+        swing = math.sin(phase) * (3.0 + 7.0 * speed)
+        lift = max(0.0, math.cos(phase)) * 6.0 * speed
+        foot_x = hip_x + swing
+        foot_y = -lift
+        knee_x = hip_x + swing * 0.35
+        knee_y = hip_y * 0.55 - lift * 0.35
+    c.line([(hip_x, hip_y), (knee_x, knee_y), (foot_x, foot_y)], fill=col, width=3.4)
+    c.ellipse((foot_x - 3, foot_y - 2, foot_x + 3.5, foot_y + 2), fill=DARK)
+
+
+def draw_fox(c, ox, oy, phase, speed):
+    """Origin (ox, oy) is the fox's ground contact point. Fox faces right."""
+    bob = 2.2 * speed * abs(math.sin(phase))
+    oy -= bob
+
+    def P(x, y):
+        return (ox + x, oy + y)
+
+    # contact shadow
+    c.ellipse((ox - 22, oy - 3, ox + 22, oy + 4), fill=(52, 86, 48))
+
+    # bushy tail: circles along a wagging quadratic curve
+    wag = math.sin(phase * 0.9) * 3.0
+    ax, ay = P(-12, -30)
+    bx, by = P(-25, -33 + wag)
+    qx, qy = P(-30, -47 + wag * 1.6)
+    for k in range(7):
+        t = k / 6.0
+        mt = 1.0 - t
+        x = mt * mt * ax + 2 * mt * t * bx + t * t * qx
+        y = mt * mt * ay + 2 * mt * t * by + t * t * qy
+        r = 7.5 - 3.2 * t
+        fill = CREAM if t >= 0.72 else ORANGE
+        c.ellipse((x - r, y - r, x + r, y + r), fill=fill)
+
+    # far legs first, then near legs
+    draw_leg(c, *P(-9, -24), phase + math.pi, True, speed)
+    draw_leg(c, *P(9, -24), phase, True, speed)
+
+    # torso
+    c.ellipse((ox - 17, oy - 40, ox + 17, oy - 20), fill=ORANGE)
+    c.ellipse((ox - 12, oy - 33, ox + 12, oy - 21), fill=CREAM)  # belly
+    c.ellipse((ox + 6, oy - 38, ox + 17, oy - 26), fill=CREAM)   # chest
+
+    draw_leg(c, *P(-9, -24), phase, False, speed)
+    draw_leg(c, *P(9, -24), phase + math.pi, False, speed)
+
+    # ears
+    c.polygon([P(6, -52), P(9, -63), P(14, -52)], fill=ORANGE_D)
+    c.polygon([P(14, -52), P(20, -63), P(23, -51)], fill=ORANGE_D)
+    c.polygon([P(8, -53), P(10, -60), P(12.5, -53)], fill=DARK)
+    c.polygon([P(16, -53), P(19, -60), P(21, -52.5)], fill=DARK)
+
+    # head + cheeks
+    c.ellipse((ox + 3, oy - 55, ox + 27, oy - 33), fill=ORANGE)
+    c.ellipse((ox + 7, oy - 45, ox + 20, oy - 34), fill=CREAM)
+
+    # snout, nose, eye
+    c.polygon([P(20, -47), P(33, -42.5), P(20, -38)], fill=CREAM)
+    c.ellipse((ox + 30, oy - 45.5, ox + 35, oy - 40), fill=DARK)
+    c.ellipse((ox + 16.5, oy - 50, ox + 20, oy - 45.5), fill=DARK)
+    c.ellipse((ox + 17.5, oy - 49.2, ox + 18.7, oy - 47.8), fill=(255, 255, 255))
+
+
+# ------------------------------------------------------------------- dust
+def draw_dust(c, particles):
+    for x, y, life in particles:
+        t = 1.0 - life / 3.0
+        r = 2.0 + t * 4.0
+        shade = int(150 + 70 * t)
+        c.ellipse((x - r, y - r * 0.5, x + r, y + r * 0.5),
+                  fill=(shade, shade - 18, shade - 42))
+
+
+def main():
+    case_dir = Path(__file__).resolve().parent
+    rng = np.random.default_rng(7)
+
+    petals = [(240, 230, 120), (245, 245, 245), (238, 170, 200)]
+    decor = {
+        "tufts": [(float(rng.uniform(0, W)), float(rng.uniform(96, H - 4)),
+                   float(rng.uniform(-1, 1))) for _ in range(70)],
+        "flowers": [(float(rng.uniform(6, W - 6)), float(rng.uniform(104, H - 8)),
+                     petals[int(rng.integers(len(petals)))]) for _ in range(9)],
+    }
+
+    start_x, end_x, base_y = 116.0, 202.0, 122.0
+    frames = []
+    phase = 0.0
+    prev_x = start_x
+    dust = []
+
+    for i in range(N):
+        t = i / (N - 1)
+        x = start_x + (end_x - start_x) * smoothstep(t)
+        dx = x - prev_x
+        speed = min(1.0, dx / 8.0)
+        phase += dx * 0.12
+        prev_x = x
+
+        c = Canvas()
+        draw_background(c, rng, decor)
+        draw_shoe(c)
+
+        if speed > 0.35 and i % 2 == 0:
+            dust.append([x - 16, base_y - 1, 3])
+        dust = [[p[0] - 1.2, p[1], p[2] - 1] for p in dust if p[2] > 1]
+        draw_dust(c, dust)
+
+        # Fox is painted after the shoe, so it stays visually in front while
+        # overlapping it, then ends clearly to the shoe's right.
+        draw_fox(c, x, base_y, phase, speed)
+        frames.append(c.to_frame())
+
+    frames = np.stack(frames).astype(np.uint8)
+    artifact_io.save(frames, case_dir)
+    print(f"saved {frames.shape} to {case_dir}")
+
+
+if __name__ == "__main__":
+    main()
