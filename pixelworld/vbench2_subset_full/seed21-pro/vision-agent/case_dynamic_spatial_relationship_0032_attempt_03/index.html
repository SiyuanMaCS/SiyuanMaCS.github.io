+"""Text-to-video: a kangaroo jumps from the left of a small box to its right.
+
+Renders 20 RGB frames (256x144, 4 fps) with PIL only, at 2x supersampling
+for smooth edges, then downsamples with LANCZOS.
+"""
+import math
+
+import numpy as np
+from PIL import Image, ImageDraw, ImageFilter
+
+import artifact_io
+
+W, H, N, FPS = 256, 144, 20, 4.0
+S = 2  # supersampling factor
+GROUND_Y = 116
+BOX_X, BOX_W, BOX_H = 128, 30, 26
+X_START, X_END = 52, 202
+TAKEOFF, LAND = 3, 16
+ARC_H = 46
+
+
+def lerp(a, b, t):
+    return a + (b - a) * t
+
+
+def pts(shape, ax, gy, t, squash):
+    """Interpolate a polygon between standing (t=0) and tucked (t=1) poses."""
+    out = []
+    for p0, p1 in shape:
+        x = lerp(p0[0], p1[0], t)
+        y = lerp(p0[1], p1[1], t)
+        y = y * (1.0 - 0.18 * squash)
+        out.append(((ax + x) * S, (gy + y) * S))
+    return out
+
+
+# --- Kangaroo shape templates, local coords: origin at feet, y grows upward ---
+TAIL = [
+    ((-4, -24), (-3, -22)),
+    ((-9, -22), (-10, -21)),
+    ((-20, -10), (-22, -14)),
+    ((-26, -3), (-28, -10)),
+    ((-27, 0), (-27, -7)),
+    ((-21, -1), (-22, -8)),
+    ((-15, -8), (-14, -12)),
+    ((-7, -18), (-6, -17)),
+]
+LEG_FAR = [
+    ((-8, -18), (-8, -16)),
+    ((0, -16), (-2, -15)),
+    ((6, -8), (0, -10)),
+    ((12, -6), (-4, -8)),
+    ((14, -3), (-10, -10)),
+    ((8, -2), (-12, -13)),
+    ((-2, -4), (-8, -15)),
+    ((-10, -10), (-6, -16)),
+]
+LEG_NEAR = [
+    ((-5, -16), (-4, -15)),
+    ((3, -15), (4, -14)),
+    ((9, -7), (6, -9)),
+    ((17, -5), (2, -6)),
+    ((19, -2), (-6, -8)),
+    ((13, -1), (-8, -12)),
+    ((1, -3), (-4, -14)),
+    ((-7, -9), (-2, -15)),
+]
+ARM = [
+    ((3, -29), (2, -27)),
+    ((5, -28), (4, -26)),
+    ((8, -20), (5, -21)),
+    ((6, -20), (3, -21)),
+    ((3, -27), (1, -25)),
+]
+SNOUT = [
+    ((10, -40), (11, -36)),
+    ((17, -39), (18, -35)),
+    ((17, -36), (18, -32)),
+    ((10, -37), (11, -34)),
+]
+EAR_A = [
+    ((2, -47), (3, -43)),
+    ((0, -55), (2, -51)),
+    ((5, -50), (8, -46)),
+]
+EAR_B = [
+    ((7, -48), (9, -44)),
+    ((7, -56), (10, -52)),
+    ((11, -50), (14, -46)),
+]
+
+BROWN_DARK = (112, 66, 38)
+BROWN = (176, 112, 62)
+BROWN_LIGHT = (205, 148, 92)
+CREAM = (232, 205, 168)
+
+
+def ellipse_poly(cx, cy, rx, ry, angle_deg):
+    out = []
+    a = math.radians(angle_deg)
+    ca, sa = math.cos(a), math.sin(a)
+    for i in range(24):
+        t = 2 * math.pi * i / 24
+        x, y = rx * math.cos(t), ry * math.sin(t)
+        out.append((cx + x * ca - y * sa, cy + x * sa + y * ca))
+    return [(p0, p0) for p0 in out]
+
+
+def draw_kangaroo(base, ax, gy, tuck, squash):
+    """gy is the foot line; callers subtract the jump-arc height."""
+    layer = Image.new("RGBA", (W * S, H * S), (0, 0, 0, 0))
+    d = ImageDraw.Draw(layer)
+
+    body_cx = lerp(-2, -3, tuck)
+    body_cy = lerp(-27, -25, tuck) * (1 - 0.18 * squash)
+    angle = lerp(12, 28, tuck)
+    body = pts(ellipse_poly(body_cx, body_cy, 11, 14, angle), ax, gy, 0, 0)
+    belly = pts(ellipse_poly(body_cx + 2, body_cy + 3, 6, 9, angle), ax, gy, 0, 0)
+    head_cx = lerp(7, 8, tuck)
+    head_cy = lerp(-41, -37, tuck) * (1 - 0.18 * squash)
+    head = pts(ellipse_poly(head_cx, head_cy, 8.2, 8, angle * 0.4), ax, gy, 0, 0)
+
+    d.polygon(pts(TAIL, ax, gy, tuck, squash), fill=BROWN_DARK)
+    d.polygon(pts(LEG_FAR, ax, gy, tuck, squash), fill=BROWN_DARK)
+    d.polygon(body, fill=BROWN, outline=BROWN_DARK)
+    d.polygon(belly, fill=BROWN_LIGHT)
+    d.polygon(pts(LEG_NEAR, ax, gy, tuck, squash), fill=BROWN, outline=BROWN_DARK)
+    d.polygon(pts(ARM, ax, gy, tuck, squash), fill=BROWN_DARK)
+    d.polygon(pts(EAR_A, ax, gy, tuck, squash), fill=BROWN, outline=BROWN_DARK)
+    d.polygon(pts(EAR_B, ax, gy, tuck, squash), fill=BROWN, outline=BROWN_DARK)
+    d.polygon(head, fill=BROWN_LIGHT, outline=BROWN_DARK)
+    d.polygon(pts(SNOUT, ax, gy, tuck, squash), fill=CREAM, outline=BROWN_DARK)
+
+    nx = (ax + lerp(17, 18, tuck)) * S
+    ny = (gy + lerp(-38, -34, tuck) * (1 - 0.18 * squash)) * S
+    d.ellipse((nx - 1.6 * S, ny - 1.6 * S, nx + 1.6 * S, ny + 1.6 * S), fill=(40, 28, 22))
+    ex = (ax + lerp(9, 10, tuck)) * S
+    ey = (gy + lerp(-43, -39, tuck) * (1 - 0.18 * squash)) * S
+    d.ellipse((ex - 1.4 * S, ey - 1.4 * S, ex + 1.4 * S, ey + 1.4 * S), fill=(30, 22, 18))
+
+    base.alpha_composite(layer)
+
+
+def draw_shadow(base, x, gy, air_h):
+    alpha = max(40, 110 - air_h * 1.6)
+    spread = max(10, 20 - air_h * 0.18)
+    layer = Image.new("RGBA", (W * S, H * S), (0, 0, 0, 0))
+    d = ImageDraw.Draw(layer)
+    d.ellipse(
+        ((x - spread) * S, (gy - 3) * S, (x + spread) * S, (gy + 3) * S),
+        fill=(40, 55, 35, int(alpha)),
+    )
+    base.alpha_composite(layer)
+
+
+def build_background():
+    bg = Image.new("RGBA", (W * S, H * S))
+    d = ImageDraw.Draw(bg)
+    top = np.array([122, 178, 226])
+    bottom = np.array([205, 232, 240])
+    for y in range(GROUND_Y * S):
+        c = (top + (bottom - top) * y / (GROUND_Y * S)).astype(int)
+        d.line([(0, y), (W * S, y)], fill=tuple(c))
+
+    d.ellipse((206 * S, 12 * S, 230 * S, 36 * S), fill=(255, 240, 190))
+    for cx, cy, s in [(48, 26, 1.0), (150, 18, 0.8), (96, 44, 0.65)]:
+        for dx, dy, r in [(-10, 0, 8), (0, -3, 10), (10, 0, 8), (0, 3, 8)]:
+            d.ellipse(
+                ((cx + dx * s - r * s) * S, (cy + dy * s - r * s) * S,
+                 (cx + dx * s + r * s) * S, (cy + dy * s + r * s) * S),
+                fill=(255, 255, 255, 235),
+            )
+    d.ellipse((-40 * S, 84 * S, 130 * S, 132 * S), fill=(138, 182, 120))
+    d.ellipse((96 * S, 88 * S, 280 * S, 134 * S), fill=(122, 170, 108))
+    d.rectangle((0, GROUND_Y * S, W * S, H * S), fill=(106, 162, 78))
+    d.rectangle((0, GROUND_Y * S, W * S, (GROUND_Y + 3) * S), fill=(126, 182, 92))
+    rng = np.random.default_rng(7)
+    for _ in range(70):
+        gx = int(rng.integers(4, W - 4))
+        gh = int(rng.integers(3, 8))
+        col = tuple(int(v) for v in rng.integers(70, 110, size=3))
+        d.line((gx * S, (GROUND_Y + 2) * S, (gx - 1) * S, (GROUND_Y + 2 - gh) * S), fill=col)
+        d.line((gx * S, (GROUND_Y + 2) * S, (gx + 1) * S, (GROUND_Y + 2 - gh) * S), fill=col)
+    for _ in range(12):
+        px, py = int(rng.integers(6, W - 6)), int(rng.integers(GROUND_Y + 8, H - 6))
+        d.ellipse(((px - 1) * S, (py - 1) * S, (px + 1) * S, (py + 1) * S), fill=(180, 175, 150))
+    return bg.filter(ImageFilter.GaussianBlur(0.4))
+
+
+def draw_box(base):
+    x0, x1 = BOX_X - BOX_W // 2, BOX_X + BOX_W // 2
+    y0, y1 = GROUND_Y - BOX_H, GROUND_Y
+    layer = Image.new("RGBA", (W * S, H * S), (0, 0, 0, 0))
+    d = ImageDraw.Draw(layer)
+    d.ellipse(((x0 - 3) * S, (y1 - 2) * S, (x1 + 3) * S, (y1 + 3) * S), fill=(50, 60, 40, 90))
+    d.polygon(
+        [(x0 * S, y0 * S), (x1 * S, y0 * S), ((x1 - 4) * S, (y0 - 6) * S),
+         ((x0 - 4) * S, (y0 - 6) * S)],
+        fill=(196, 152, 96), outline=(120, 84, 48),
+    )
+    d.rectangle((x0 * S, y0 * S, x1 * S, y1 * S), fill=(178, 128, 74), outline=(110, 76, 42))
+    d.polygon(
+        [(x1 * S, y0 * S), ((x1 - 4) * S, (y0 - 6) * S),
+         ((x1 - 4) * S, (y1 - 6) * S), (x1 * S, y1 * S)],
+        fill=(150, 102, 58),
+    )
+    d.line(((BOX_X - 3) * S, y0 * S, (BOX_X - 3) * S, y1 * S), fill=(206, 172, 120), width=S)
+    d.line((x0 * S, (y0 + 8) * S, x1 * S, (y0 + 8) * S), fill=(130, 92, 52), width=max(1, S // 2))
+    base.alpha_composite(layer)
+
+
+def main():
+    bg = build_background()
+    frames = []
+    for f in range(N):
+        img = bg.copy()
+        draw_box(img)
+
+        if f < TAKEOFF:
+            # idle, then an anticipatory crouch on the last pre-jump frame
+            x = X_START
+            air = 0.0
+            tuck = 0.0
+            squash = 0.55 if f == TAKEOFF - 1 else 0.0
+        elif f <= LAND:
+            p = (f - TAKEOFF) / (LAND - TAKEOFF)
+            e = 0.5 - 0.5 * math.cos(math.pi * p)  # smooth horizontal ease
+            x = lerp(X_START, X_END, e)
+            air = ARC_H * math.sin(math.pi * p)
+            tuck = math.sin(math.pi * p)
+            squash = 0.7 if f == LAND else 0.0
+        else:
+            x = X_END
+            air = 0.0
+            tuck = 0.0
+            squash = max(0.0, 0.7 * (1 - (f - LAND) / 3))
+
+        # subtle breathing sway while on the ground
+        gy = GROUND_Y + (0.6 * math.sin(f * 1.7) if air == 0 else 0.0)
+        draw_shadow(img, x, GROUND_Y, air)
+        draw_kangaroo(img, x, gy - air, tuck, squash)
+
+        small = img.convert("RGB").resize((W, H), Image.Resampling.LANCZOS)
+        frames.append(np.asarray(small, dtype=np.uint8))
+
+    frames = np.stack(frames)
+    artifact_io.save(frames, ".")
+    print(f"saved {frames.shape}")
+
+
+if __name__ == "__main__":
+    main()
