+"""Generate the text-to-video RGB frames for the fox/shoe spatial relation.
+
+The scene is drawn with PIL at 4x resolution and downsampled to the required
+256x144 frames.  The fox begins lower in the frame and overlapping the shoe,
+making it visibly in front, then runs to a clearly separated position on the
+shoe's left.
+"""
+
+from pathlib import Path
+import math
+
+import numpy as np
+from PIL import Image, ImageDraw
+
+import artifact_io
+
+
+WIDTH, HEIGHT, FRAME_COUNT, FPS = 256, 144, 20, 4.0
+SS = 4  # supersampling factor
+
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def q(value, scale=1.0):
+    """Convert a target-resolution coordinate to supersampled pixels."""
+    return int(round(value * SS * scale))
+
+
+def box(x0, y0, x1, y1, scale=1.0):
+    return (q(x0, scale), q(y0, scale), q(x1, scale), q(y1, scale))
+
+
+def points(sequence, scale=1.0):
+    return [(q(x, scale), q(y, scale)) for x, y in sequence]
+
+
+def smooth_step(t):
+    t = np.clip(t, 0.0, 1.0)
+    return t * t * (3.0 - 2.0 * t)
+
+
+def make_background():
+    """Build a calm park-like background once; only actors change per frame."""
+    image = np.zeros((HEIGHT, WIDTH, 3), dtype=np.uint8)
+    sky_top = np.array([151, 202, 230], dtype=np.float32)
+    sky_low = np.array([229, 239, 221], dtype=np.float32)
+    grass_top = np.array([128, 185, 101], dtype=np.float32)
+    grass_low = np.array([87, 146, 73], dtype=np.float32)
+    horizon = 91
+
+    for y in range(HEIGHT):
+        if y < horizon:
+            color = sky_top * (1 - y / horizon) + sky_low * (y / horizon)
+        else:
+            f = (y - horizon) / (HEIGHT - horizon)
+            color = grass_top * (1 - f) + grass_low * f
+        image[y, :, :] = color
+
+    canvas = Image.fromarray(image).resize(
+        (WIDTH * SS, HEIGHT * SS), Image.Resampling.NEAREST
+    )
+    draw = ImageDraw.Draw(canvas)
+
+    # Soft sun and simple clouds.
+    draw.ellipse(box(211, 15, 239, 43), fill=(255, 231, 171))
+    for cloud in [((34, 24), 25, 8), ((104, 35), 31, 9), ((184, 52), 24, 7)]:
+        (cx, cy), rx, ry = cloud
+        draw.ellipse(box(cx - rx, cy - ry, cx + rx, cy + ry), fill=(245, 249, 247))
+        draw.ellipse(box(cx - rx * 0.45, cy - ry * 1.55, cx + rx * 0.45, cy + ry),
+                     fill=(245, 249, 247))
+
+    # Rounded distant hills, mostly visible as a low horizon.
+    draw.ellipse(box(-38, 70, 118, 122), fill=(132, 188, 124))
+    draw.ellipse(box(82, 66, 215, 120), fill=(116, 176, 113))
+    draw.ellipse(box(178, 74, 302, 124), fill=(137, 193, 127))
+
+    # Fixed foreground grass marks and tiny flowers give the motion a reference.
+    grass_color = (63, 123, 58)
+    for x, y in [(18, 121), (42, 132), (103, 126), (127, 136), (203, 128),
+                 (229, 137), (244, 119), (78, 116)]:
+        draw.line(points([(x, y), (x - 3, y - 7)]), fill=grass_color, width=q(1))
+        draw.line(points([(x + 1, y), (x + 1, y - 8)]), fill=grass_color, width=q(1))
+        draw.line(points([(x + 2, y), (x + 5, y - 6)]), fill=grass_color, width=q(1))
+
+    for x, y, color in [(28, 112, (245, 221, 92)), (118, 119, (239, 151, 180)),
+                        (221, 113, (245, 221, 92)), (246, 132, (230, 166, 231))]:
+        draw.ellipse(box(x - 2, y - 2, x + 2, y + 2), fill=color)
+        draw.line(points([(x, y + 2), (x, y + 7)]), fill=(64, 111, 55), width=q(1))
+
+    return canvas
+
+
+def draw_shadow(draw, cx, ground_y, width, height, color=(55, 91, 54)):
+    draw.ellipse(
+        box(cx - width / 2, ground_y - height / 2,
+            cx + width / 2, ground_y + height / 2),
+        fill=color,
+    )
+
+
+def draw_shoe(draw, cx=158, ground_y=112):
+    """A side-view blue sneaker, toe toward the right."""
+    # Sole.
+    draw.rounded_rectangle(
+        box(cx - 42, ground_y - 13, cx + 42, ground_y + 2),
+        radius=q(5),
+        fill=(238, 241, 243),
+        outline=(83, 90, 99),
+        width=q(1.5),
+    )
+    for dx in (-30, -20, -10, 0, 10, 20, 30):
+        draw.line(
+            points([(cx + dx, ground_y - 9), (cx + dx, ground_y - 2)]),
+            fill=(176, 183, 190),
+            width=q(1),
+        )
+
+    upper = [
+        (cx - 35, ground_y - 10),
+        (cx - 32, ground_y - 27),
+        (cx - 15, ground_y - 34),
+        (cx + 8, ground_y - 31),
+        (cx + 25, ground_y - 23),
+        (cx + 37, ground_y - 11),
+    ]
+    draw.polygon(points(upper), fill=(49, 103, 194), outline=(28, 61, 125))
+    draw.line(
+        points([(cx - 27, ground_y - 12), (cx + 20, ground_y - 12)]),
+        fill=(117, 164, 231),
+        width=q(2),
+    )
+
+    # Heel opening, laces, and a pale toe cap.
+    draw.ellipse(
+        box(cx - 31, ground_y - 29, cx - 15, ground_y - 17),
+        fill=(31, 48, 74),
+        outline=(20, 34, 55),
+    )
+    for index in range(3):
+        x = cx - 8 + index * 8
+        y = ground_y - 27 + index * 4
+        draw.line(points([(x, y), (x + 9, y + 4)]),
+                  fill=(238, 242, 246), width=q(1.6))
+        draw.line(points([(x + 1, y + 5), (x + 10, y + 1)]),
+                  fill=(238, 242, 246), width=q(1.6))
+
+    toe = [
+        (cx + 17, ground_y - 24),
+        (cx + 37, ground_y - 11),
+        (cx + 34, ground_y - 8),
+        (cx + 13, ground_y - 8),
+        (cx + 10, ground_y - 17),
+    ]
+    draw.polygon(points(toe), fill=(244, 246, 247), outline=(190, 198, 205))
+
+
+def draw_fox(draw, cx, ground_y, scale, phase, moving):
+    """Side-view red fox facing and running left."""
+    orange = (214, 103, 39)
+    orange_dark = (150, 64, 25)
+    far_leg = (139, 72, 34)
+    near_leg = (73, 42, 27)
+    cream = (247, 236, 213)
+    black = (31, 28, 25)
+
+    bob = (abs(math.sin(phase * 2.0)) * 2.2 * scale) if moving else 0.0
+    body_cy = ground_y - 15 * scale - bob
+    tail_wag = (math.sin(phase * 0.8) * 3.0 * scale) if moving else 1.5 * scale
+
+    def p(dx, dy):
+        return (cx + dx * scale, body_cy + dy * scale)
+
+    def b(x0, y0, x1, y1):
+        return box(x0, y0, x1, y1, scale)
+
+    # Far legs are drawn first and partly covered by the body.
+    for hip_dx, leg_phase in [(-11, 0.65 * math.pi), (10, 1.65 * math.pi)]:
+        foot_dx = math.sin(phase + leg_phase) * 6.5 * scale
+        hip = p(hip_dx, 5)
+        foot = (hip[0] + foot_dx, ground_y - 1)
+        draw.line(
+            points([hip, foot], scale),
+            fill=far_leg,
+            width=q(2.4, scale),
+        )
+
+    # Bushy tail, with a cream tip.
+    tail = [
+        p(13, -3),
+        p(25, -16 + tail_wag / scale),
+        p(35, -10 + tail_wag / scale),
+        p(31, 1 + tail_wag / scale),
+        p(21, 7),
+        p(13, 5),
+    ]
+    draw.polygon(points(tail, scale), fill=orange, outline=orange_dark)
+    tail_tip = [
+        p(28, -13 + tail_wag / scale),
+        p(35, -10 + tail_wag / scale),
+        p(31, 1 + tail_wag / scale),
+        p(25, -2 + tail_wag / scale),
+    ]
+    draw.polygon(points(tail_tip, scale), fill=cream)
+
+    # Body and belly.
+    draw.ellipse(
+        b(cx - 21 * scale, body_cy - 9 * scale,
+          cx + 18 * scale, body_cy + 9 * scale),
+        fill=orange,
+        outline=orange_dark,
+        width=q(1),
+    )
+    draw.ellipse(
+        b(cx - 12 * scale, body_cy - 1 * scale,
+          cx + 12 * scale, body_cy + 8 * scale),
+        fill=cream,
+    )
+
+    # Near legs have a stronger, visible gallop/walk cycle.
+    for hip_dx, offset in [(-11, 0.0), (10, math.pi)]:
+        foot_dx = math.sin(phase + offset) * 7.5 * scale
+        hip = p(hip_dx, 5)
+        foot = (hip[0] + foot_dx, ground_y - 1)
+        draw.line(points([hip, foot], scale), fill=near_leg, width=q(3, scale))
+        draw.ellipse(
+            b(foot[0] - 3 * scale, foot[1] - 1.5 * scale,
+              foot[0] + 3 * scale, foot[1] + 1.5 * scale),
+            fill=near_leg,
+        )
+
+    # Pointed ears, head, snout, and facial markings.
+    left_ear = [p(-29, -16), p(-26, -28), p(-19, -17)]
+    right_ear = [p(-21, -17), p(-17, -28), p(-11, -16)]
+    draw.polygon(points(left_ear, scale), fill=orange, outline=orange_dark)
+    draw.polygon(points(right_ear, scale), fill=orange, outline=orange_dark)
+    draw.polygon(points([p(-26, -19), p(-24, -25), p(-21, -19)], scale),
+                 fill=near_leg)
+    draw.polygon(points([p(-19, -19), p(-17, -25), p(-14, -19)], scale),
+                 fill=near_leg)
+
+    draw.ellipse(
+        b(cx - 32 * scale, body_cy - 19 * scale,
+          cx - 13 * scale, body_cy - 1 * scale),
+        fill=orange,
+        outline=orange_dark,
+        width=q(1),
+    )
+    snout = [p(-30, -12), p(-40, -7), p(-31, -4), p(-25, -7)]
+    draw.polygon(points(snout, scale), fill=orange, outline=orange_dark)
+    draw.polygon(points([p(-31, -7), p(-39, -6), p(-31, -3)], scale), fill=cream)
+    draw.ellipse(
+        b(cx - 41 * scale, body_cy - 9 * scale,
+          cx - 37 * scale, body_cy - 5 * scale),
+        fill=black,
+    )
+    draw.ellipse(
+        b(cx - 27 * scale, body_cy - 15 * scale,
+          cx - 24 * scale, body_cy - 12 * scale),
+        fill=black,
+    )
+    draw.ellipse(
+        b(cx - 25 * scale, body_cy - 9 * scale,
+          cx - 17 * scale, body_cy - 3 * scale),
+        fill=cream,
+    )
+
+
+def make_frame(background, index):
+    frame = background.copy()
+    draw = ImageDraw.Draw(frame)
+
+    # Hold briefly, run through the middle, then settle clearly to the left.
+    progress = float(smooth_step((index - 2) / 13.0))
+    fox_x = 150.0 + (70.0 - 150.0) * progress
+    fox_ground_y = 124.0 + (110.0 - 124.0) * progress
+    fox_scale = 1.12 + (0.96 - 1.12) * progress
+    moving = 0.0 < progress < 1.0
+    phase = index * 2.15
+
+    # Shadows keep both objects anchored to the grass.
+    draw_shadow(draw, 158, 115, 88, 12)
+    draw_shadow(draw, fox_x, fox_ground_y + 2, 48 * fox_scale, 9 * fox_scale)
+
+    draw_shoe(draw)
+    draw_fox(draw, fox_x, fox_ground_y, fox_scale, phase, moving)
+
+    return np.asarray(frame.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS),
+                      dtype=np.uint8)
+
+
+def main():
+    background = make_background()
+    frames = [make_frame(background, i) for i in range(FRAME_COUNT)]
+    artifact_io.save(frames, CASE_DIR)
+    print(f"Saved {FRAME_COUNT} RGB frames ({WIDTH}x{HEIGHT}) to {CASE_DIR}")
+
+
+if __name__ == "__main__":
+    main()
