+def smoothstep(edge0, edge1, value):
+    t = np.clip((value - edge0) / (edge1 - edge0), 0.0, 1.0)
+    return t * t * (3.0 - 2.0 * t)

+def normalize(vector):
+    length = np.sqrt(np.sum(vector * vector, axis=0, keepdims=True))
+    return vector / np.maximum(length, 1e-8)

+def mix(a, b, amount):
+    return a * (1.0 - amount) + b * amount

+def background_layer(xx, yy):

+    yn = yy / H

+    top = np.array([0.025, 0.075, 0.085], dtype=np.float32)

+def make_balloon(xx, yy, progress, contact_y):

+def needle_geometry(xx, yy, tip_x, tip_y):
+    angle = np.deg2rad(-4.0)
+    dx = xx - tip_x
+    dy = yy - tip_y
+    along = -(dx * np.cos(angle) + dy * np.sin(angle))
+    across = -dx * np.sin(angle) + dy * np.cos(angle)

+    radius = body_radius * (0.08 + 0.92 * cone_t)
+    radius = np.where(along > cone_length, body_radius, radius)

+    body_mask = (along >= -W * 0.01) & (along <= W * 1.2) & (np.abs(across) <= radius)
+    edge_soft = np.clip((radius - np.abs(across)) / 1.8, 0.0, 1.0)
+    body_mask = body_mask.astype(np.float32) * edge_soft
+
+    radial = np.clip(across / np.maximum(radius, 1e-5), -1.0, 1.0)
+    # Steel is lit from above-left, with a hot narrow reflection and darker underside.
+    metal_light = np.exp(-((radial + 0.33) / 0.24) ** 2)
+    metal_shadow = np.exp(-((radial - 0.47) / 0.42) ** 2)
+    cone_boost = 0.62 + 0.38 * np.clip(1.0 - cone_t, 0.0, 1.0)
+    color = np.zeros(xx.shape + (3,), dtype=np.float32)
+    color += np.array([0.54, 0.62, 0.66], dtype=np.float32)[None, None, :] * (0.25 + 0.68 * metal_light)[..., None]
+    color += np.array([0.98, 0.94, 0.80], dtype=np.float32)[None, None, :] * (metal_light ** 5)[..., None] * 0.82
+    color -= np.array([0.16, 0.19, 0.22], dtype=np.float32)[None, None, :] * metal_shadow[..., None]
+    color *= cone_boost[..., None]
+
+    # Fine machining bands remain visible along the shaft.
+    bands = 0.022 * np.sin(along * 0.16) + 0.012 * np.sin(along * 0.39 + 1.3)
+    color += bands[..., None]

+def cast_needle_shadow(background, xx, yy, tip_x, tip_y):
+    angle = np.deg2rad(-4.0)
+    dx = xx - tip_x
+    dy = yy - tip_y
+    along = -(dx * np.cos(angle) + dy * np.sin(angle))
+    across = -dx * np.sin(angle) + dy * np.cos(angle)
+    shadow = np.exp(-((across - H * 0.050) / (H * 0.030)) ** 2)
+    shadow *= smoothstep(-W * 0.02, W * 0.035, along) * (1.0 - smoothstep(W * 0.17, W * 0.33, along))
+    shadow *= 0.13 + 0.08 * smoothstep(W * 0.05, W * 0.24, along)
+    return background * (1.0 - shadow[..., None])

+def puncture_details(background, xx, yy, progress, tip_x, tip_y):
+    if progress < 0.63:
+        return background, np.zeros((H, W), dtype=np.float32)
+
+    t = smoothstep(0.63, 0.92, progress)
+    dx = (xx - tip_x) / (W * 0.040)
+    dy = (yy - tip_y) / (H * 0.090)
+    radial = np.sqrt(dx * dx + dy * dy)
+    hole = (1.0 - smoothstep(0.22, 1.0, radial)) * t
+
+    # A tiny crescent tear follows the surface tension around the puncture.
+    tear_line = np.exp(-((radial - 0.72) / 0.18) ** 2) * (0.55 + 0.45 * np.sin(np.arctan2(dy, dx) * 3.0 + 0.4))
+    tear_line *= t * smoothstep(-0.2, 0.25, dx)
+    color = background * (1.0 - 0.78 * hole[..., None])
+    color += tear_line[..., None] * np.array([0.52, 0.09, 0.045], dtype=np.float32)
+
+    # A few extremely fine droplets and an air wisp make the breach legible
+    # without turning this slow puncture into an explosion.
+    wisp = np.exp(-(((xx - tip_x - W * 0.070) / (W * 0.070)) ** 2 +
+                    ((yy - tip_y + H * 0.005 - (xx - tip_x) * 0.05) / (H * 0.015)) ** 2))
+    wisp *= smoothstep(0.72, 0.98, progress)
+    color += wisp[..., None] * np.array([0.10, 0.13, 0.14], dtype=np.float32) * 0.32
+    return np.clip(color, 0.0, 1.0), hole
+

+def render_frame(index):
+    p = index / (FRAMES - 1)
+    yy, xx = np.mgrid[0:H, 0:W].astype(np.float32)
+    contact_y = H * 0.405
+
+    background = background_layer(xx, yy)
+
+    # The tip eases forward, then slows under resistance before breaking through.
+    approach = smoothstep(0.02, 0.44, p)
+    press = smoothstep(0.34, 0.78, p)
+    breach = smoothstep(0.68, 0.94, p)
+    tip_x = W * (0.285 + 0.275 * approach + 0.034 * press + 0.020 * breach)
+    tip_y = contact_y - H * 0.004 * press + H * 0.002 * breach
+
+    background = cast_needle_shadow(background, xx, yy, tip_x, tip_y)
+    balloon, balloon_mask, edge, normal = make_balloon(xx, yy, p, contact_y)
+
+    # A soft shadow under the balloon gives the macro framing an anchor.
+    floor_shadow = np.exp(-(((xx - W * 0.76) / (W * 0.42)) ** 2 +
+                            ((yy - H * 0.92) / (H * 0.11)) ** 2))
+    background *= 1.0 - floor_shadow[..., None] * 0.38
+
+    image = mix(background, balloon, balloon_mask[..., None])
+
+    # Focus falloff: the far-right latex is slightly softer than the tip.
+    soft = Image.fromarray(np.clip(image * 255.0, 0, 255).astype(np.uint8))
+    soft_array = np.asarray(soft.filter(ImageFilter.GaussianBlur(radius=1.8)), dtype=np.float32) / 255.0
+    focus = 1.0 - smoothstep(0.54, 0.96, xx / W) * 0.18
+    image = mix(soft_array, image, focus[..., None])
+
+    body_mask, needle_color, radius, along, across = needle_geometry(xx, yy, tip_x, tip_y)
+
+    # Needle shadow on the balloon, especially across the dented region.
+    shadow_line = np.exp(-((yy - tip_y - H * 0.034) / (H * 0.024)) ** 2)
+    shadow_line *= (xx > tip_x - W * 0.24) & (xx < tip_x + W * 0.02)
+    image *= 1.0 - shadow_line[..., None] * balloon_mask[..., None] * 0.23
+
+    # The needle stays in front of the background, and its point remains visible
+    # against the latex while the tiny aperture opens around it.
+    image = mix(image, needle_color, body_mask[..., None])
+    image, hole = puncture_details(image, xx, yy, p, tip_x, tip_y)
+
+    # The hole overlays only the balloon; the needle point remains a crisp tip.
+    needle_again = (body_mask > 0.55) & (along > W * 0.010)
+    image = mix(image, needle_color, needle_again.astype(np.float32)[..., None] * 0.92)
+
+    # Very subtle atmospheric haze and a final filmic shoulder.
+    haze = np.exp(-(((xx - tip_x - W * 0.16) / (W * 0.18)) ** 2 +
+                    ((yy - tip_y) / (H * 0.18)) ** 2))
+    image += haze[..., None] * np.array([0.025, 0.028, 0.023], dtype=np.float32)
+    image = np.clip(image, 0.0, 1.0)
+    image = image ** np.array([0.94, 0.96, 0.98], dtype=np.float32)
+
+    image = Image.fromarray(np.clip(image * 255.0 + 0.5, 0, 255).astype(np.uint8))
+    return np.asarray(image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS), dtype=np.uint8)
+
+
+def main():
+    case_dir = Path(__file__).resolve().parent
+    frames = np.stack([render_frame(i) for i in range(FRAMES)])
+    artifact_io.save(frames, case_dir)
+
+
+if __name__ == "__main__":
+    main()
