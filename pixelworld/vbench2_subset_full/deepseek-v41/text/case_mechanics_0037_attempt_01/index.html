+def normalize(vector: np.ndarray) -> np.ndarray:
+    return vector / np.maximum(np.linalg.norm(vector, axis=-1, keepdims=True), 1e-8)

+def rotation_x(angle: float) -> np.ndarray:
+    c, s = np.cos(angle), np.sin(angle)
+    return np.array(((1.0, 0.0, 0.0), (0.0, c, -s), (0.0, s, c)))

+
+def rotation_y(angle: float) -> np.ndarray:
+    c, s = np.cos(angle), np.sin(angle)
+    return np.array(((c, 0.0, s), (0.0, 1.0, 0.0), (-s, 0.0, c)))

+def rotation_z(angle: float) -> np.ndarray:
+    c, s = np.cos(angle), np.sin(angle)
+    return np.array(((c, -s, 0.0), (s, c, 0.0), (0.0, 0.0, 1.0)))

+def smoothstep(edge0: float, edge1: float, x: np.ndarray) -> np.ndarray:
+    t = np.clip((x - edge0) / (edge1 - edge0), 0.0, 1.0)
+    return t * t * (3.0 - 2.0 * t)

+def trajectory(t: float) -> tuple[float, float]:
+    """Return duck-origin height and vertical velocity for the complete bounce."""
+    y0, v0 = 92.0, -75.1
+    if t <= CONTACT_TIMES[0]:
+        return y0 + v0 * t - 0.5 * GRAVITY * t * t, v0 - GRAVITY * t
+
+    velocities = np.array((95.4, 42.9, 21.5, 9.7, 4.4), dtype=np.float64)
+    x = -60.0
+    vx = 42.0
+    cursor = 0.0
+    y, vy = y0, v0
+    for index, contact in enumerate(CONTACT_TIMES):
+        dt = contact - cursor
+        y = y + vy * dt - 0.5 * GRAVITY * dt * dt
+        x += vx * dt
+        cursor = contact
+        vy = velocities[index]
+        vx *= 0.70
+        if t <= contact:
+            return y, vy
+        if index == len(CONTACT_TIMES) - 1:
+            dt = t - contact
+            y = vy * dt - 0.5 * GRAVITY * dt * dt
+            return max(0.0, y), vy - GRAVITY * dt
+        next_contact = CONTACT_TIMES[index + 1]
+        if t < next_contact:
+            dt = t - contact
+            return vy * dt - 0.5 * GRAVITY * dt * dt, vy - GRAVITY * dt
+    return 0.0, 0.0

+def pose_at(frame_index: int) -> Pose:
+    t = frame_index / FPS
+    y, vy = trajectory(t)
+    if t <= CONTACT_TIMES[0]:
+        x = -60.0 + 42.0 * t
+    else:
+        x = -60.0 + 42.0 * CONTACT_TIMES[0]
+        previous = CONTACT_TIMES[0]
+        vx = 42.0 * 0.70
+        for contact in CONTACT_TIMES[1:]:
+            dt = min(t, contact) - previous
+            x += vx * dt
+            if t <= contact:
+                break
+            previous = contact
+            vx *= 0.70
+        if t > CONTACT_TIMES[-1]:
+            x += 4.4 * (t - CONTACT_TIMES[-1])
+
+    impact_squash = 0.0
+    for contact, amplitude in zip(CONTACT_TIMES, CONTACT_AMPLITUDES):
+        delta = t - contact
+        if -0.055 <= delta <= 0.18:
+            if delta < 0.0:
+                pulse = amplitude * np.exp(delta / 0.035)
+            else:
+                pulse = amplitude * np.exp(-delta / 0.072)
+            impact_squash = max(impact_squash, float(pulse))
+
+    airborne_stretch = np.clip((abs(vy) - 42.0) / 260.0, 0.0, 0.10)
+    airborne_stretch *= smoothstep(7.0, 35.0, y)
+    sy = 1.0 - impact_squash + airborne_stretch
+    sxz = 1.0 / np.sqrt(max(sy, 0.58))
+    scale = np.array((sxz, sy, sxz), dtype=np.float64)
+
+    roll = np.deg2rad(20.0 - 34.0 * min(t, 0.75) + 4.5 * np.sin(2.2 * t + 0.2))
+    if t > 0.75:
+        roll = np.deg2rad(-5.0 + 2.8 * np.sin(4.3 * (t - 0.75)) * np.exp(-1.1 * (t - 0.75)))
+    pitch = np.deg2rad(7.0 - 12.0 * min(t / 0.75, 1.0) + 1.5 * np.sin(3.1 * t))
+    yaw = np.deg2rad(-12.0 + 9.0 * min(t / 2.0, 1.0) + 1.0 * np.sin(2.4 * t))
+    rotation = rotation_z(roll) @ rotation_y(yaw) @ rotation_x(pitch)
+    z = -17.0 + 24.0 * (1.0 - np.exp(-0.9 * t))
+    return Pose(np.array((x, y, z), dtype=np.float64), scale, rotation, vy)

+def camera_setup() -> tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray, float]:

+    up = normalize(np.cross(right, forward))
+    focal = 212.0

+def intersect_ellipsoids(
+    origins: np.ndarray,
+    directions: np.ndarray,
+    matrices: list[tuple[np.ndarray, np.ndarray, Material]],
+) -> tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray]:
+    hit_t = np.full(origins.shape[:-1], np.inf, dtype=np.float64)
+    hit_shape = np.full(origins.shape[:-1], -1, dtype=np.int16)
+    hit_normal = np.zeros_like(origins)
+    hit_point = np.zeros_like(origins)
+
+    for index, (linear, center, _) in enumerate(matrices):
+        inverse = np.linalg.inv(linear)
+        local_origin = (origins - center) @ inverse.T
+        local_direction = directions @ inverse.T
+        a = np.sum(local_direction * local_direction, axis=-1)
+        b = np.sum(local_origin * local_direction, axis=-1)
+        c = np.sum(local_origin * local_origin, axis=-1) - 1.0
+        discriminant = b * b - a * c
+        valid = discriminant > 0.0
+        root = np.sqrt(np.maximum(discriminant, 0.0))
+        t = (-b - root) / np.maximum(a, 1e-12)
+        valid &= t > 1e-5
+        better = valid & (t < hit_t)
+        if not np.any(better):
+            continue
+
+        local_hit = local_origin + t[..., None] * local_direction
+        normal = local_hit @ np.linalg.inv(linear)
+        normal = normalize(normal)
+        points = origins + t[..., None] * directions
+
+        hit_t = np.where(better, t, hit_t)
+        hit_shape = np.where(better, index, hit_shape)
+        hit_normal = np.where(better[..., None], normal, hit_normal)
+        hit_point = np.where(better[..., None], points, hit_point)
+
+    return hit_t, hit_shape, hit_normal, hit_point

+def shadowed_by_duck(
+    origins: np.ndarray,
+    matrices: list[tuple[np.ndarray, np.ndarray, Material]],
+    max_distance: float = 400.0,
+) -> np.ndarray:
+    shadow = np.zeros(origins.shape[:-1], dtype=bool)
+    for linear, center, _ in matrices:
+        inverse = np.linalg.inv(linear)
+        local_origin = (origins - center) @ inverse.T
+        local_direction = KEY_LIGHT @ inverse.T
+        a = float(np.sum(local_direction * local_direction))
+        b = np.sum(local_origin * local_direction, axis=-1)
+        c = np.sum(local_origin * local_origin, axis=-1) - 1.0
+        discriminant = b * b - a * c
+        valid = discriminant > 0.0
+        root = np.sqrt(np.maximum(discriminant, 0.0))
+        t = (-b - root) / max(a, 1e-12)
+        shadow |= valid & (t > 0.12) & (t < max_distance)
+    return shadow

+def floor_and_wall(origins: np.ndarray, directions: np.ndarray) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
+    height, width = origins.shape[:2]
+    floor_t = np.full((height, width), np.inf)
+    wall_t = np.full((height, width), np.inf)
+    down = directions[..., 1] < -1e-5
+    back = directions[..., 2] < -1e-5
+    floor_t[down] = -origins[..., 1][down] / directions[..., 1][down]
+    wall_t[back] = (-148.0 - origins[..., 2][back]) / directions[..., 2][back]
+    floor_t[floor_t <= 0.0] = np.inf
+    wall_t[wall_t <= 0.0] = np.inf
+    is_floor = floor_t < wall_t
+    surface_t = np.where(is_floor, floor_t, wall_t)
+    points = origins + surface_t[..., None] * directions
+
+    # Wooden floor with long boards and subtle figure.
+    px, pz = points[..., 0], points[..., 2]
+    board_coord = px / 31.0 + 0.045 * np.sin(0.035 * pz)
+    board_id = np.floor(board_coord).astype(np.int32)
+    board_frac = board_coord - board_id
+    board_tone = 0.94 + 0.065 * np.sin(board_id * 2.17 + 0.7)
+    grain = (
+        0.035 * np.sin(0.16 * pz + 1.2 * np.sin(0.32 * px))
+        + 0.018 * np.sin(0.52 * pz + 0.22 * px)
+        + 0.010 * np.sin(1.38 * pz)
+    )
+    seam = smoothstep(0.0, 0.035, np.minimum(board_frac, 1.0 - board_frac))
+    seam = 0.63 + 0.37 * seam
+    joint_coord = (pz + 48.0 * (board_id % 2)) / 82.0
+    joint_frac = joint_coord - np.floor(joint_coord)
+    joint = 0.70 + 0.30 * smoothstep(0.0, 0.014, np.minimum(joint_frac, 1.0 - joint_frac))
+    floor_base = np.array((0.42, 0.305, 0.19), dtype=np.float64)
+    floor_color = floor_base[None, None, :] * (board_tone * (1.0 + grain) * seam * joint)[..., None]
+
+    # A broad pool of warm window light crosses the back of the floor.
+    light_shape = (
+        smoothstep(-110.0, -75.0, px)
+        * (1.0 - smoothstep(55.0, 105.0, px))
+        * smoothstep(-145.0, -115.0, pz)
+        * (1.0 - smoothstep(-10.0, 42.0, pz))
+    )
+    floor_color += light_shape[..., None] * np.array((0.10, 0.082, 0.045))
+
+    # Muted blue-green plaster wall, baseboard, and a gentle wall wash.
+    wall_world_y = points[..., 1]
+    plaster_noise = (
+        0.018 * np.sin(0.19 * points[..., 0] + 0.11 * wall_world_y)
+        + 0.012 * np.sin(0.37 * wall_world_y)
+    )
+    wall_color = (
+        np.array((0.29, 0.40, 0.40), dtype=np.float64)[None, None, :]
+        * (0.94 + plaster_noise)[..., None]
+    )
+    wall_color *= (0.92 + 0.08 * smoothstep(0.0, 95.0, wall_world_y))[..., None]
+    baseboard = (wall_world_y > 0.0) & (wall_world_y < 7.2)
+    wall_color = np.where(
+        baseboard[..., None],
+        np.array((0.61, 0.56, 0.44), dtype=np.float64),
+        wall_color,
+    )
+    wall_color *= np.where((wall_world_y >= 7.2) & (wall_world_y < 8.4), 0.68, 1.0)[..., None]
+
+    color = np.where(is_floor[..., None], floor_color, wall_color)
+    # Fade distant surfaces into the wall color and add a fine screen-space vignette.
+    distance_fade = 0.08 * smoothstep(85.0, 145.0, pz)
+    color = color * (1.0 - distance_fade[..., None]) + wall_color * distance_fade[..., None]
+    return color, is_floor, points

+def shade_duck(
+    ray_directions: np.ndarray,
+    shape_indices: np.ndarray,
+    normals: np.ndarray,
+    points: np.ndarray,
+    matrices: list[tuple[np.ndarray, np.ndarray, Material]],
+) -> np.ndarray:
+    result = np.zeros_like(points)
+    view = -ray_directions
+    key_half = normalize(KEY_LIGHT + view)
+    fill_half = normalize(FILL_LIGHT + view)
+    occluded = shadowed_by_duck(points, matrices)
+
+    for index, (_, _, material) in enumerate(matrices):
+        mask = shape_indices == index
+        if not np.any(mask):
+            continue
+        normal = normals[mask]
+        hit = points[mask]
+        diffuse = np.clip(np.sum(normal * KEY_LIGHT, axis=-1), 0.0, 1.0)
+        diffuse *= np.where(occluded[mask], 0.44, 1.0)
+        fill = np.clip(np.sum(normal * FILL_LIGHT, axis=-1), 0.0, 1.0)
+        hemisphere = 0.46 + 0.20 * np.clip(normal[..., 1], 0.0, 1.0)
+        floor_ao = 0.77 + 0.23 * smoothstep(0.0, 24.0, hit[..., 1])
+        illumination = (0.20 + 0.78 * diffuse + 0.18 * fill) * hemisphere * floor_ao
+
+        spec_power = 18.0 + 92.0 * (1.0 - material.roughness)
+        specular = np.power(
+            np.clip(np.sum(normal * key_half[mask], axis=-1), 0.0, 1.0),
+            spec_power,
+        )
+        specular *= (0.28 + 0.42 * (1.0 - material.roughness)) * np.where(occluded[mask], 0.28, 1.0)
+        fill_spec = np.power(
+            np.clip(np.sum(normal * fill_half[mask], axis=-1), 0.0, 1.0),
+            22.0,
+        ) * 0.07
+        rim = np.power(
+            1.0 - np.clip(np.abs(np.sum(normal * view[mask], axis=-1)), 0.0, 1.0),
+            3.2,
+        )
+        rim *= 0.13
+
+        color = material.base[None, :] * illumination[:, None]
+        color += (specular + fill_spec + rim)[:, None] * np.array((1.0, 0.97, 0.82))
+        result[mask] = color
+    return result
+

+def render_frame(frame_index: int) -> np.ndarray:
+    render_width = WIDTH * SUPERSAMPLE
+    render_height = HEIGHT * SUPERSAMPLE
+    origins, directions = make_rays(render_width, render_height)
+    background, floor_mask, background_points = floor_and_wall(origins, directions)
+
+    pose = pose_at(frame_index)
+    matrices = shape_matrices(pose)
+    hit_t, hit_shape, hit_normal, hit_point = intersect_ellipsoids(origins, directions, matrices)
+    duck_mask = hit_shape >= 0
+
+    # A ray-traced key-light shadow, softened before it is mixed into the floor.
+    floor_origins = background_points[floor_mask]
+    raw_shadow = np.zeros(floor_mask.shape, dtype=np.float32)
+    if floor_origins.size:
+        shadow_values = shadowed_by_duck(floor_origins, matrices).astype(np.float32)
+        raw_shadow[floor_mask] = shadow_values
+    blurred_shadow = cv2.GaussianBlur(
+        raw_shadow,
+        (0, 0),
+        sigmaX=1.55 * SUPERSAMPLE,
+        borderType=cv2.BORDER_REFLECT,
+    )
+    background *= (1.0 - 0.52 * blurred_shadow[..., None] * floor_mask[..., None])
+
+    # A compact contact shadow grounds the duck even when the key shadow is long.
+    contact_dx = background_points[..., 0] - pose.position[0]
+    contact_dz = background_points[..., 2] - pose.position[2]
+    height_softness = 4.0 + 0.13 * max(pose.position[1], 0.0)
+    contact = np.exp(
+        -0.5 * ((contact_dx / (27.0 + height_softness)) ** 2 + (contact_dz / (17.0 + 0.5 * height_softness)) ** 2)
+    )
+    contact *= (0.24 + 0.36 * np.exp(-max(pose.position[1], 0.0) / 24.0)) * floor_mask
+    background *= (1.0 - contact[..., None])
+
+    duck_color = shade_duck(directions, hit_shape, hit_normal, hit_point, matrices)
+    image = np.where(duck_mask[..., None], duck_color, background)
+
+    # Atmospheric highlight and screen-space finish.
+    yy = np.linspace(-1.0, 1.0, render_height, dtype=np.float64)[:, None]
+    xx = np.linspace(-1.0, 1.0, render_width, dtype=np.float64)[None, :]
+    vignette = 1.0 - 0.16 * (0.58 * xx * xx + 0.42 * yy * yy)
+    image *= vignette[..., None]
+    image = np.clip(image, 0.0, 1.0) ** (1.0 / 1.07)
+    image = (image * 255.0 + 0.5).astype(np.uint8)
+    image = cv2.resize(image, (WIDTH, HEIGHT), interpolation=cv2.INTER_AREA)
+
+    # Very light film grain keeps the flat areas from banding after 8-bit packing.
+    rng = np.random.default_rng(8000 + frame_index)
+    grain = rng.normal(0.0, 0.85, (HEIGHT, WIDTH, 1))
+    image = np.clip(image.astype(np.float32) + grain, 0.0, 255.0).astype(np.uint8)
+    return image
+
