+def ball_state(time_s: float) -> dict[str, float]:
+    if time_s < IMPACT_TIME:
+        x = 49.0 + 25.0 * time_s
+        y = 22.0 + 5.0 * time_s + 12.5 * time_s * time_s

+    phase = time_s - IMPACT_TIME

+    else:

+    scale_x = 1.0 / math.sqrt(scale_y)
+    radius_y = BALL_RADIUS * scale_y
+    radius_x = BALL_RADIUS * scale_x
+
+    # Hold the compressed ball against the floor during its brief contact.
+    if 0.0 <= phase < 0.17:

+        "ry": radius_y,

+def add_ball_motion_blur(
+    scene: Image.Image, frame_index: int, shutter: float = 0.065, samples: int = 6
+) -> None:
+    center_t = frame_index / FPS
+    times = np.linspace(center_t - shutter / 2.0, center_t + shutter / 2.0, samples)
+    weight = 1.0 / samples
+
+    shadow_acc = np.zeros((H, W), dtype=np.float32)
+    ball_premul = np.zeros((H, W, 3), dtype=np.float32)
+    ball_alpha = np.zeros((H, W), dtype=np.float32)
+
+    for sample_t in times:
+        state = ball_state(float(sample_t))
+        height = max(0.0, GROUND_Y - (state["y"] + state["ry"]))

+        shadow_cy = (GROUND_Y - 2.0 + height * 0.005) * SUPERSAMPLE
+        shadow_w = (2.0 * BALL_RADIUS * (1.13 + height / 100.0)) * SUPERSAMPLE
+        shadow_h = (4.4 * (1.0 + height / 135.0)) * SUPERSAMPLE
+        shadow_opacity = 0.43 / (1.0 + height / 52.0)
+
+        pad = int((shadow_w * 0.55 + 12 * SUPERSAMPLE))
+        x0 = max(0, int(shadow_cx - shadow_w * 0.65 - pad))
+        x1 = min(W, int(shadow_cx + shadow_w * 0.65 + pad))
+        y0 = max(0, int(shadow_cy - shadow_h - pad))
+        y1 = min(H, int(shadow_cy + shadow_h + pad))
+        if x1 > x0 and y1 > y0:
+            mask = Image.new("L", (W, H), 0)
+            draw = ImageDraw.Draw(mask)
+            draw.ellipse(

+                    shadow_cy - shadow_h / 2.0,
+                    shadow_cx + shadow_w / 2.0,
+                    shadow_cy + shadow_h / 2.0,
+                ),
+                fill=int(255 * shadow_opacity),
+            )
+            blur_radius = (1.7 + height / 18.0) * SUPERSAMPLE
+            mask = mask.filter(ImageFilter.GaussianBlur(blur_radius))
+            shadow_acc += np.asarray(mask, dtype=np.float32) * weight / 255.0
+
+        layer = render_ball(state)
+        left = 0
+        top = 0
+        rgba = np.asarray(layer, dtype=np.float32)
+        # render_ball is cropped; recover its placement from the state bounds.
+        cx = state["x"] * SUPERSAMPLE
+        cy = state["y"] * SUPERSAMPLE
+        rx = state["rx"] * SUPERSAMPLE
+        ry = state["ry"] * SUPERSAMPLE
+        margin = 3 * SUPERSAMPLE
+        left = max(0, int(math.floor(cx - rx - margin)))
+        top = max(0, int(math.floor(cy - ry - margin)))
+        alpha = rgba[..., 3:4] / 255.0 * weight
+        region_alpha = ball_alpha[top : top + layer.height, left : left + layer.width]
+        region_rgb = ball_premul[top : top + layer.height, left : left + layer.width]
+        region_alpha += alpha[..., 0]
+        region_rgb += rgba[..., :3] * alpha
+
+    if shadow_acc.max() > 0.0:
+        alpha = np.clip(shadow_acc, 0.0, 1.0)
+        scene_np = np.asarray(scene, dtype=np.float32)
+        shadow_color = np.array([17.0, 24.0, 25.0], dtype=np.float32)
+        scene_np = scene_np * (1.0 - alpha[..., None]) + shadow_color * alpha[..., None]
+        scene.paste(Image.fromarray(np.clip(scene_np, 0, 255).astype(np.uint8)))
+
+    if ball_alpha.max() > 0.0:
+        alpha = np.clip(ball_alpha, 0.0, 1.0)
+        rgb = ball_premul / np.maximum(ball_alpha[..., None], 1e-5)
+        scene_np = np.asarray(scene, dtype=np.float32)
+        scene_np = scene_np * (1.0 - alpha[..., None]) + rgb * alpha[..., None]
+        scene.paste(Image.fromarray(np.clip(scene_np, 0, 255).astype(np.uint8)))

+def add_atmosphere(image: Image.Image, frame_index: int) -> Image.Image:
+    array = np.asarray(image, dtype=np.float32).copy()
+    yy, xx = np.indices((H, W), dtype=np.float32)
+    dx = (xx - W / 2.0) / (W / 2.0)
+    dy = (yy - H / 2.0) / (H / 2.0)
+    vignette = np.clip(1.0 - 0.14 * (dx * dx + dy * dy), 0.0, 1.0)
+    array *= vignette[..., None]
+
+    rng = np.random.default_rng(1000 + frame_index)
+    grain = rng.normal(0.0, 1.0, (H, W)).astype(np.float32)
+    array += grain[..., None] * 0.65
+
+    # Tiny lift in the cool shadows, restrained warm highlights.
+    luminance = array.mean(axis=2, keepdims=True) / 255.0
+    array += np.array([2.0, 3.0, 4.0], dtype=np.float32) * (1.0 - luminance) * 0.28
+    array += np.array([2.5, 1.2, -1.0], dtype=np.float32) * luminance
+    return Image.fromarray(np.clip(array, 0, 255).astype(np.uint8), mode="RGB")
+
+
+def generate_frames() -> np.ndarray:
+    background = make_background()
+    frames = []
+    for frame_index in range(FRAME_COUNT):
+        frame = background.copy()
+        add_ball_motion_blur(frame, frame_index)
+        frame = add_atmosphere(frame, frame_index)
+        frame = frame.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
+        frames.append(np.asarray(frame, dtype=np.uint8))
+    return np.stack(frames, axis=0)
+
+
+def main() -> None:
+    case_dir = Path(__file__).resolve().parent
+    frames = generate_frames()
+    artifact_io.save(frames, case_dir)
+    print(f"Saved {len(frames)} RGB frames to {case_dir}")
+
+
+if __name__ == "__main__":
+    main()
