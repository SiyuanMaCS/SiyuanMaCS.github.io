+def add_glove_lighting(im, phase):
+    # Cool reflected sheen and a soft vignette; keep the image low-noise and legible.
+    y, x = np.mgrid[0:CH, 0:CW].astype(np.float32)
+    u = x / (CW - 1)
+    v = y / (CH - 1)
+    rgb = np.asarray(im, dtype=np.float32)
+    sheen = np.exp(-(((u - 0.37) / 0.19) ** 2 + ((v - 0.60) / 0.10) ** 2))
+    rgb += np.array([19, 45, 48], np.float32)[None, None, :] * sheen[:, :, None]
+    warm = np.exp(-(((u - 0.61) / 0.24) ** 2 + ((v - 0.22) / 0.22) ** 2))
+    rgb += np.array([15, 10, 5], np.float32)[None, None, :] * warm[:, :, None]
+    vig = 1.0 - 0.18 * (((u - 0.5) / 0.7) ** 2 + ((v - 0.5) / 0.8) ** 2)
+    rgb *= vig[:, :, None]
+    return Image.fromarray(np.clip(rgb, 0, 255).astype(np.uint8), "RGB")

+def render_frame(index):
+    t = index / (N - 1)
+    # Slow approach, decisive indentation, then a held penetration beat.
+    approach = ease((t - 0.06) / 0.50)
+    press = ease((t - 0.50) / 0.25)
+    deep = ease((t - 0.75) / 0.20)
+    tip_y = CH * (0.145 + 0.285 * approach + 0.050 * press + 0.032 * deep)
+    cx = CW * (0.535 + 0.003 * math.sin(t * math.pi * 2.0))
+    # Keep contact at a stable point once it is reached.
+    contact_point = CH * 0.435
+    if tip_y > contact_point:
+        tip_y = contact_point + (tip_y - contact_point) * 0.30
+
+    im = gradient_background()
+    im = draw_table_contact(im, 0.65 + 0.35 * approach)
+    im = glove_surface(t)
+
+    # Slight handheld/table drift, applied only to foreground elements.
+    fg = Image.new("RGBA", (CW, CH), (0, 0, 0, 0))
+    draw_dimple(fg, max(0.0, press * 0.85 + deep * 0.15))
+    draw_penetration(fg, max(0.0, (press - 0.40) / 0.60))
+
+    # Pencil and hand share the same anchor and motion.
+    draw_pencil(fg, cx, tip_y, 255, max(0.0, press))
+    draw_hand(fg, cx, tip_y, t)
+    # Tiny graphite dust / reflected glints at the puncture.
+    if press > 0.15:
+        pd = ImageDraw.Draw(fg, "RGBA")
+        for k in range(4):
+            ang = 2.4 + k * 1.4
+            rr = SS * (8 + 9 * press + 2.5 * k)
+            x = cx + math.cos(ang) * rr
+            y = contact_point + math.sin(ang) * rr * 0.45

+    fg = fg.filter(ImageFilter.GaussianBlur(SS * 0.45))
+    im = Image.alpha_composite(im.convert("RGBA"), fg).convert("RGB")
+    im = add_glove_lighting(im, t)

+    # 4x supersampling yields crisp but natural small-screen silhouettes.
