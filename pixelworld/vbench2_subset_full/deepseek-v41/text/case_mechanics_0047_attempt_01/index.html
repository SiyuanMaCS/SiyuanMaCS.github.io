+def cubic_points(p0, p1, p2, p3, count=70):
+    t = np.linspace(0.0, 1.0, count, endpoint=True, dtype=np.float32)
+    omt = 1.0 - t
+    points = (
+        omt[:, None] ** 3 * np.asarray(p0)
+        + 3.0 * omt[:, None] ** 2 * t[:, None] * np.asarray(p1)
+        + 3.0 * omt[:, None] * t[:, None] ** 2 * np.asarray(p2)
+        + t[:, None] ** 3 * np.asarray(p3)
+    )
+    return points

+def point_at_progress(path, distance, progress):

+    tangent /= max(float(np.linalg.norm(tangent)), 1e-8)
