+def chaikin(points, rounds=2):
+    values = [(float(x), float(y)) for x, y in points]

+def path_metrics(points):
+    lengths = [0.0]
+    for a, b in zip(points[:-1], points[1:]):
+        lengths.append(lengths[-1] + math.hypot(b[0] - a[0], b[1] - a[1]))

+def point_on_path(points, lengths, fraction):
+    target = min(1.0, max(0.0, fraction)) * lengths[-1]
+    for index in range(1, len(lengths)):
+        if lengths[index] >= target:
+            span = max(1e-6, lengths[index] - lengths[index - 1])
+            amount = (target - lengths[index - 1]) / span
+            a, b = points[index - 1], points[index]
+            return (
+                a[0] + (b[0] - a[0]) * amount,
+                a[1] + (b[1] - a[1]) * amount,
+            )
+    return points[-1]
