+def smoothstep(edge0, edge1, values):
+    t = np.clip((values - edge0) / (edge1 - edge0), 0.0, 1.0)
+    return t * t * (3.0 - 2.0 * t)
+

+def blur_u8(array, radius):
+    image = Image.fromarray(np.clip(array, 0, 255).astype(np.uint8), "L")
+    return np.asarray(image.filter(ImageFilter.GaussianBlur(radius)), dtype=np.float32)
+

+def resize_float(array, size, resample=Image.Resampling.BICUBIC):
+    image = Image.fromarray(np.asarray(array, dtype=np.float32), mode="F")
+    return np.asarray(image.resize(size, resample), dtype=np.float32)

+def quadratic_curve(start, control, end, count=90):
+    t = np.linspace(0.0, 1.0, count, dtype=np.float32)[:, None]
+    p0 = np.asarray(start, dtype=np.float32)
+    p1 = np.asarray(control, dtype=np.float32)
+    p2 = np.asarray(end, dtype=np.float32)
+    return (1.0 - t) ** 2 * p0 + 2.0 * (1.0 - t) * t * p1 + t**2 * p2
+
+
+def bowed_stroke(start, end, bend=0.0, count=100):
+    start = np.asarray(start, dtype=np.float32)
+    end = np.asarray(end, dtype=np.float32)
+    midpoint = (start + end) * 0.5
+    direction = end - start
+    normal = np.array([-direction[1], direction[0]], dtype=np.float32)
+    normal /= max(float(np.linalg.norm(normal)), 1e-6)
+    control = midpoint + normal * bend
+    return quadratic_curve(start, control, end, count)

+def polyline_lengths(points):
+    deltas = np.diff(points, axis=0)
+    lengths = np.sqrt(np.sum(deltas * deltas, axis=1))
+    return np.concatenate(([0.0], np.cumsum(lengths)))
+
+
+def point_at_progress(points, lengths, progress):
+    target = float(np.clip(progress, 0.0, 1.0)) * lengths[-1]
+    index = int(np.searchsorted(lengths, target, side="right") - 1)
+    index = min(max(index, 0), len(points) - 2)
+    span = max(lengths[index + 1] - lengths[index], 1e-6)
+    amount = (target - lengths[index]) / span
+    return points[index] * (1.0 - amount) + points[index + 1] * amount
+
+
+def points_through_progress(points, lengths, progress):
+    target = float(np.clip(progress, 0.0, 1.0)) * lengths[-1]
+    if target <= 0.05:
+        return points[:1]
+    index = int(np.searchsorted(lengths, target, side="right"))
+    index = min(max(index, 1), len(points))
+    partial = points[:index].copy()
+    if index < len(points) and target < lengths[index]:
+        span = max(lengths[index] - lengths[index - 1], 1e-6)
+        amount = (target - lengths[index - 1]) / span
+        final = points[index - 1] * (1.0 - amount) + points[index] * amount
+        partial = np.vstack((partial, final))
+    return partial

+def rgba_composite(base, overlay):
+    base_rgba = Image.fromarray(np.clip(base, 0, 255).astype(np.uint8), "RGB").convert("RGBA")
+    return np.asarray(Image.alpha_composite(base_rgba, overlay).convert("RGB"), dtype=np.float32)

+def polygon_mask(points):
+    image = Image.new("L", (HRW, HRH), 0)
+    draw = ImageDraw.Draw(image)
+    draw.polygon([(float(x) * SS, float(y) * SS) for x, y in points], fill=255)
+    return np.asarray(image, dtype=np.float32) / 255.0

+def make_surface():
+    x = np.linspace(0.0, W, HRW, dtype=np.float32)[None, :]
+    y = np.linspace(0.0, H, HRH, dtype=np.float32)[:, None]
+    nx = x / W
+    ny = y / H
+
+    # A cool, soft window reflection travels across the glazed ceramic.
+    sheen = (
+        0.55 * np.exp(-(((x - 58.0) / 58.0) ** 2 + ((y - 28.0) / 39.0) ** 2))
+        + 0.36 * np.exp(-(((x - 197.0) / 70.0) ** 2 + ((y - 96.0) / 53.0) ** 2))
+        + 0.20 * np.exp(-(((x - 145.0) / 18.0) ** 2 + ((y - 70.0) / 93.0) ** 2))
+    )
+    vertical = 0.985 + 0.030 * (1.0 - ny) - 0.012 * nx
+    edge_falloff = 1.0 - 0.018 * (np.abs(nx - 0.5) * 2.0) ** 2
+    luminance = vertical * edge_falloff + sheen * 0.028
+
+    base = np.empty((HRH, HRW, 3), dtype=np.float32)
+    base[..., 0] = 241.0 * luminance + sheen * 1.2
+    base[..., 1] = 243.0 * luminance + sheen * 1.6
+    base[..., 2] = 237.0 * luminance + sheen * 2.2
+
+    # Faint glaze undulation and ceramic micro-grain.
+    low = np.random.default_rng(71).normal(0.0, 1.0, (18, 32)).astype(np.float32)
+    low = blur_u8(low * 28.0 + 128.0, 1.2) - 128.0
+    low = resize_float(low, (HRW, HRH), Image.Resampling.BICUBIC) * 0.070
+    micro = np.random.default_rng(72).normal(0.0, 0.62, (HRH, HRW)).astype(np.float32)
+    grain = low + micro
+    base += grain[..., None] * np.array([0.75, 0.82, 0.95], dtype=np.float32)
+
+    # Very restrained optical falloff keeps the white board from looking flat.
+    vignette = 1.0 - 0.030 * ((nx - 0.5) ** 2 * 1.3 + (ny - 0.5) ** 2)
+    return base * vignette[..., None]

+def make_chalk_texture():
+    rng = np.random.default_rng(431)
+    texture = rng.normal(0.0, 1.0, (192, 192)).astype(np.float32)
+    texture = blur_u8(texture * 22.0 + 128.0, 0.72) - 128.0
+    fine = rng.normal(0.0, 0.85, (192, 192)).astype(np.float32)
+    return texture / 22.0 + fine

+def chalk_geometry(tip, angle_deg, lift=0.0):
+    angle = np.deg2rad(angle_deg)
+    direction = np.array([np.cos(angle), -np.sin(angle)], dtype=np.float32)
+    normal = np.array([-direction[1], direction[0]], dtype=np.float32)
+    tip = np.asarray(tip, dtype=np.float32) + np.array([0.0, -lift * 0.28], dtype=np.float32)
+    half_width = 5.9
+    length = 85.0
+
+    silhouette = np.array(
+        [
+            tip + direction * 0.6 + normal * 2.1,
+            tip + direction * 7.0 + normal * half_width,
+            tip + direction * 78.0 + normal * half_width,
+            tip + direction * 83.0 + normal * 4.8,
+            tip + direction * length,
+            tip + direction * 83.0 - normal * 4.8,
+            tip + direction * 78.0 - normal * half_width,
+            tip + direction * 7.0 - normal * half_width,
+            tip + direction * 0.6 - normal * 2.1,
+        ],
+        dtype=np.float32,
+    )
+    tip_face = np.array(
+        [
+            tip + direction * 0.4,
+            tip + direction * 7.3 + normal * half_width,
+            tip + direction * 7.3 - normal * half_width,
+        ],
+        dtype=np.float32,
+    )
+    top = np.array(
+        [
+            tip + direction * 7.2 + normal * half_width,
+            tip + direction * 78.0 + normal * half_width,
+            tip + direction * 83.0 + normal * 4.8,
+            tip + direction * length,
+            tip + direction * 83.0 - normal * 4.8,
+            tip + direction * 78.0 - normal * half_width,
+            tip + direction * 7.2 - normal * half_width,
+        ],
+        dtype=np.float32,
+    )
+    return tip, direction, normal, silhouette, top, tip_face

+def draw_chalk(image, tip, angle_deg, lift, chalk_texture):
+    tip, direction, normal, silhouette, top, tip_face = chalk_geometry(tip, angle_deg, lift)
+    silhouette_mask = polygon_mask(silhouette)
+    top_mask = polygon_mask(top)
+    tip_mask = polygon_mask(tip_face)
+
+    # Soft cast shadow, with more separation and blur as the chalk is lifted.
+    shadow_shift = np.array([6.0 + lift * 0.9, 7.0 + lift * 1.25], dtype=np.float32)
+    shifted = silhouette + shadow_shift
+    shadow = polygon_mask(shifted)
+    shadow = np.asarray(
+        Image.fromarray((shadow * 255.0).astype(np.uint8), "L").filter(
+            ImageFilter.GaussianBlur((5.0 + lift * 0.9) * SS)
+        ),
+        dtype=np.float32,
+    ) / 255.0
+    shadow_alpha = (0.145 - lift * 0.008) * shadow
+    shadow_color = np.array([92.0, 98.0, 94.0], dtype=np.float32)
+    image *= 1.0 - shadow_alpha[..., None]
+    image += shadow_color * shadow_alpha[..., None]
+
+    # The lower side plane gives the rectangular chalk real thickness.
+    side_shift = np.array([0.0, 2.9], dtype=np.float32)
+    side_mask = np.clip(polygon_mask(silhouette + side_shift) - top_mask, 0.0, 1.0)
+    side_alpha = side_mask * 0.95
+    side_color = np.array([198.0, 151.0, 17.0], dtype=np.float32)
+    image *= 1.0 - side_alpha[..., None]
+    image += side_color * side_alpha[..., None]
+
+    x = np.linspace(0.0, W, HRW, dtype=np.float32)[None, :]
+    y = np.linspace(0.0, H, HRH, dtype=np.float32)[:, None]
+    u = (x - tip[0]) * direction[0] + (y - tip[1]) * direction[1]
+    v = (x - tip[0]) * normal[0] + (y - tip[1]) * normal[1]
+
+    tx = np.mod((u * 3.7 + 21.0).astype(np.int32), 192)
+    ty = np.mod((v * 4.8 + 43.0).astype(np.int32), 192)
+    texture = chalk_texture[ty, tx]
+    cross_highlight = np.cos(np.clip(v / 7.0, -1.0, 1.0) * np.pi * 0.5) ** 2
+    end_shade = smoothstep(58.0, 86.0, u)
+    shade = 0.94 + 0.075 * cross_highlight - 0.045 * end_shade + texture * 0.018
+    body_color = np.empty_like(image)
+    body_color[..., 0] = 248.0 * shade
+    body_color[..., 1] = 211.0 * shade
+    body_color[..., 2] = 34.0 * shade
+    body_alpha = top_mask * 0.985
+    image *= 1.0 - body_alpha[..., None]
+    image += body_color * body_alpha[..., None]
+
+    # Chalky, slightly brighter bevel at the exposed tip.
+    tip_shade = 1.0 + texture * 0.025
+    tip_color = np.stack(
+        (
+            255.0 * tip_shade,
+            232.0 * tip_shade,
+            103.0 * tip_shade,
+        ),
+        axis=-1,
+    )
+    image *= 1.0 - tip_mask[..., None] * 0.94
+    image += tip_color * tip_mask[..., None] * 0.94
+
+    # Thin upper edge highlight.
+    edge_a = tip + direction * 8.0 + normal * 5.15
+    edge_b = tip + direction * 76.5 + normal * 5.15
+    edge_overlay = Image.new("RGBA", (HRW, HRH), (0, 0, 0, 0))
+    edge_draw = ImageDraw.Draw(edge_overlay)
+    edge_draw.line(
+        [(edge_a[0] * SS, edge_a[1] * SS), (edge_b[0] * SS, edge_b[1] * SS)],
+        fill=(255, 244, 153, 150),
+        width=max(1, int(0.62 * SS)),
+    )
+    image = rgba_composite(image, edge_overlay)
+
+    # A tiny dark contact crease anchors the tip on the board.
+    if lift < 0.8:
+        contact = Image.new("RGBA", (HRW, HRH), (0, 0, 0, 0))
+        contact_draw = ImageDraw.Draw(contact)
+        cx, cy = tip[0] * SS, tip[1] * SS
+        radius = 2.3 * SS
+        contact_draw.ellipse(
+            (cx - radius, cy - radius * 0.55, cx + radius, cy + radius * 0.62),
+            fill=(121, 103, 31, int(64 * (1.0 - lift))),
+        )
+        contact = contact.filter(ImageFilter.GaussianBlur(0.55 * SS))
+        image = rgba_composite(image, contact)
+
+    return image, silhouette_mask

+def draw_stroke_layer(image, rendered_strokes, frame_no, base_noise):
+    mask_image = Image.new("L", (HRW, HRH), 0)
+    draw = ImageDraw.Draw(mask_image)
+    width = int(round(5.05 * SS))
+
+    for points, progress in rendered_strokes:
+        visible = points_through_progress(points, polyline_lengths(points), progress)
+        if len(visible) >= 2:
+            draw.line(
+                [(float(p[0]) * SS, float(p[1]) * SS) for p in visible],
+                fill=255,
+                width=width,
+                joint="curve",
+            )
+            radius = width * 0.5
+            for point in (visible[0], visible[-1]):
+                cx, cy = float(point[0]) * SS, float(point[1]) * SS
+                draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=255)
+
+    mask_hr = np.asarray(mask_image, dtype=np.float32) / 255.0
+    mask = np.asarray(
+        mask_image.resize((W, H), Image.Resampling.LANCZOS),
+        dtype=np.float32,
+    ) / 255.0
+    if float(mask.max()) < 0.01:
+        return image
+
+    edge_noise = np.roll(base_noise, frame_no * 3, axis=1)
+    alpha = mask * (0.80 + 0.20 * edge_noise)
+    alpha = np.clip(alpha, 0.0, 0.97)
+
+    core = smoothstep(0.34, 0.92, mask)
+    deposit = np.empty((H, W, 3), dtype=np.float32)
+    deposit[..., 0] = 247.0 - 10.0 * core
+    deposit[..., 1] = 220.0 - 23.0 * core
+    deposit[..., 2] = 76.0 - 55.0 * core
+
+    # A little extra gold in the densest, most recently pressed center.
+    alpha = np.clip(alpha * (0.94 + 0.07 * core), 0.0, 0.98)
+    alpha_hr = resize_float(alpha, (HRW, HRH), Image.Resampling.BICUBIC)
+    deposit_hr = np.empty_like(image)
+    for channel in range(3):
+        deposit_hr[..., channel] = resize_float(
+            deposit[..., channel], (HRW, HRH), Image.Resampling.BICUBIC
+        )
+    image *= 1.0 - alpha_hr[..., None]
+    image += deposit_hr * alpha_hr[..., None]
+
+    # Small grains escaping the stroke edge make the deposit read as chalk.
+    grains = Image.new("RGBA", (HRW, HRH), (0, 0, 0, 0))
+    grain_draw = ImageDraw.Draw(grains)
+    rng = np.random.default_rng(900 + frame_no)
+    count = int(mask.sum() * 0.55)
+    ys, xs = np.nonzero(mask > 0.50)
+    if len(xs):
+        picks = rng.choice(len(xs), size=min(count, 26), replace=False)
+        for pick in picks:
+            bx = xs[pick] + rng.normal(0.0, 3.0)
+            by = ys[pick] + rng.normal(0.0, 3.0)
+            radius = rng.uniform(0.22, 0.62) * SS
+            alpha_grain = int(rng.uniform(45, 112))
+            grain_draw.ellipse(
+                (bx * SS - radius, by * SS - radius, bx * SS + radius, by * SS + radius),
+                fill=(245, 213, 55, alpha_grain),
+            )
+    image = rgba_composite(image, grains)
+    return image

+def draw_dust(image, tip, contact, lift, frame_no, angle_deg):
+    x = np.linspace(0.0, W, HRW, dtype=np.float32)[None, :]
+    y = np.linspace(0.0, H, HRH, dtype=np.float32)[:, None]
+
+    # Fresh powder catches the light at the point of contact.
+    if contact > 0.01:
+        sigma = 3.1
+        halo = np.exp(-((x - tip[0]) ** 2 + (y - tip[1]) ** 2) / (2.0 * sigma**2))
+        halo *= contact * 0.105
+        image += halo[..., None] * np.array([255.0, 238.0, 117.0], dtype=np.float32)
+        glow = Image.new("RGBA", (HRW, HRH), (0, 0, 0, 0))
+        glow_draw = ImageDraw.Draw(glow)
+        cx, cy = tip[0] * SS, tip[1] * SS
+        radius = 0.85 * SS
+        glow_draw.ellipse(
+            (cx - radius, cy - radius, cx + radius, cy + radius),
+            fill=(255, 247, 181, int(175 * contact)),
+        )
+        image = rgba_composite(image, glow.filter(ImageFilter.GaussianBlur(0.28 * SS)))
+
+    rng = np.random.default_rng(1551)
+    particles = Image.new("RGBA", (HRW, HRH), (0, 0, 0, 0))
+    draw = ImageDraw.Draw(particles)
+    for birth in range(max(0, frame_no - 3), frame_no + 1):
+        age = (frame_no - birth) / FPS
+        if age > 0.95:
+            continue
+        birth_tip = np.asarray(frame_tip(birth), dtype=np.float32)
+        for _ in range(6):
+            offset = rng.normal(0.0, 3.0, 2).astype(np.float32)
+            velocity = np.array([rng.uniform(-4.5, 6.5), rng.uniform(1.0, 10.0)], dtype=np.float32)
+            position = birth_tip + offset + velocity * age
+            position[0] += 1.4 * age * age
+            fade = max(0.0, 1.0 - age / 0.95) ** 1.3
+            radius = rng.uniform(0.25, 0.70) * SS
+            alpha = int(np.clip(rng.uniform(40.0, 95.0) * fade, 0.0, 95.0))
+            color = (247, 220, 77, alpha) if rng.random() > 0.30 else (255, 248, 203, alpha)
+            px, py = position[0] * SS, position[1] * SS
+            draw.ellipse((px - radius, py - radius, px + radius, py + radius), fill=color)
+    image = rgba_composite(image, particles)
+    return image

+# Stroke geometry, in output pixels.  The hand-lettered bow avoids a vector look.
+STROKES = [
+    bowed_stroke((71.5, 108.5), (72.0, 52.2), bend=-0.75, count=120),
+    bowed_stroke((108.0, 52.0), (107.3, 108.5), bend=0.62, count=120),
+    bowed_stroke((72.2, 81.5), (107.5, 81.7), bend=0.70, count=90),
+    bowed_stroke((146.0, 108.0), (146.2, 52.0), bend=-0.40, count=110),
+]
+STROKE_LENGTHS = [polyline_lengths(points) for points in STROKES]
+
+
+# (kind, start, end, stroke_index or destination)
+TIMELINE = [
+    ("write", 0.00, 1.24, 0),
+    ("move", 1.24, 1.47, (108.0, 52.0)),
+    ("write", 1.47, 2.56, 1),
+    ("move", 2.56, 2.80, (72.2, 81.5)),
+    ("write", 2.80, 3.43, 2),
+    ("move", 3.43, 3.67, (146.0, 108.0)),
+    ("write", 3.67, 4.56, 3),
+    ("hold", 4.56, 5.01, (146.2, 52.0)),
+]

+def frame_state(time_seconds):
+    rendered = []
+    tip = STROKES[0][0].copy()
+    contact = 0.0
+    lift = 0.0
+    movement_direction = np.array([0.0, -1.0], dtype=np.float32)
+
+    for kind, start, end, payload in TIMELINE:
+        if time_seconds < start:
+            break
+        amount = float(np.clip((time_seconds - start) / (end - start), 0.0, 1.0))
+        if kind == "write":
+            index = int(payload)
+            points = STROKES[index]
+            lengths = STROKE_LENGTHS[index]
+            progress = amount
+            rendered.append((points, progress))
+            tip = point_at_progress(points, lengths, progress)
+            contact = 1.0
+            lift = 0.0
+            movement_direction = points[-1] - points[0]
+        elif kind == "move":
+            previous_tip = np.asarray(tip, dtype=np.float32)
+            destination = np.asarray(payload, dtype=np.float32)
+            eased = amount * amount * (3.0 - 2.0 * amount)
+            tip = previous_tip * (1.0 - eased) + destination * eased
+            lift = 4.9 * np.sin(np.pi * amount)
+            contact = 0.0
+            movement_direction = destination - previous_tip
+        else:
+            destination = np.asarray(payload, dtype=np.float32)
+            tip = destination + np.array([0.0, -0.16 * np.sin(np.pi * amount)], dtype=np.float32)
+            contact = 1.0
+            lift = 0.0
+            movement_direction = np.array([0.0, -1.0], dtype=np.float32)
+
+    if time_seconds >= TIMELINE[-1][1]:
+        for index in range(len(STROKES)):
+            rendered.append((STROKES[index], 1.0))
+        tip = np.asarray(TIMELINE[-1][3], dtype=np.float32)
+        contact = 1.0
+    return rendered, np.asarray(tip, dtype=np.float32), contact, lift, movement_direction
+

+_STATE_CACHE = {}
+
+
+def frame_tip(frame_no):
+    if frame_no not in _STATE_CACHE:
+        _STATE_CACHE[frame_no] = frame_state(frame_no / FPS)
+    return _STATE_CACHE[frame_no][1]
+

+def render_frame(surface, chalk_texture, base_noise, frame_no):
+    time_seconds = frame_no / FPS
+    rendered, tip, contact, lift, movement = frame_state(time_seconds)
+    image = surface.copy()
+
+    image = draw_stroke_layer(image, rendered, frame_no, base_noise)
+
+    # The held angle stays near 35 degrees with a very small organic wobble.
+    angle = 34.0 + 1.6 * np.sin(time_seconds * 2.3 + 0.4)
+    if np.linalg.norm(movement) > 1e-3:
