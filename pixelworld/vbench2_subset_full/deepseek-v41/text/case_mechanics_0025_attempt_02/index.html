+def smoothstep(a: float, b: float, x: float) -> float:
+    t = np.clip((x - a) / (b - a), 0.0, 1.0)
+    return t * t * (3.0 - 2.0 * t)

+def bezier(p0, p1, p2, p3, n=90):
+    t = np.linspace(0.0, 1.0, n)[:, None]
+    p0, p1, p2, p3 = map(lambda p: np.asarray(p, dtype=float), (p0, p1, p2, p3))
+    return ((1 - t) ** 3) * p0 + 3 * ((1 - t) ** 2) * t * p1 + 3 * (1 - t) * t * t * p2 + t**3 * p3

+def pipe_geometry(t: float):
+    pressure = float(smoothstep(0.8, 18.2, t))
+    collapse = float(smoothstep(6.0, 19.0, t))
+    x = np.arange(SW, dtype=np.float32) / S

+    pinches = (
+        0.55 * np.exp(-((x - 105.0) / 18.0) ** 2)
+        + 1.0 * np.exp(-((x - 137.0) / 30.0) ** 2)
+        + 0.34 * np.exp(-((x - 166.0) / 17.0) ** 2)
+    )
+    envelope = smoothstep(67.0, 83.0, x) * (1.0 - smoothstep(186.0, 207.0, x))

+    return pressure, collapse, x, center, radius

+    draw.ellipse((ss(246.5), ss(112.5), ss(247.5), ss(113.5)), fill=(255, 229, 166, 145))
+
+    # Exhaust wisps make the removed air visible above the pump outlet.
+    q = float(smoothstep(1.5, 18.0, t))
+    if q > 0.02:
+        puff = Image.new("RGBA", (SW, SH), (0, 0, 0, 0))
+        pd = ImageDraw.Draw(puff, "RGBA")
+        for k in range(9):
+            birth = k / 9.5
+            age = max(0.0, q - birth)
+            if age <= 0:
+                continue
+            px = 246.0 + age * 10.0 + 2.0 * math.sin(k * 1.9 + t * 0.25)
+            py = 107.0 - age * 33.0
+            rr = 1.5 + age * 4.2
+            a = int(58 * (1.0 - min(1.0, age / 0.72)))
+            pd.ellipse((ss(px - rr), ss(py - rr * 0.58), ss(px + rr), ss(py + rr * 0.58)),
+                       fill=(167, 201, 201, max(0, a)))
+        puff = puff.filter(ImageFilter.GaussianBlur(ss(1.2)))
+        image.alpha_composite(puff)
+
+    # A short, bright reflected streak on the upper pump edge.
+    draw.line((ss(219), ss(106.8), ss(239), ss(106.8)), fill=(208, 220, 210, 55), width=ss(0.45))
+
+
+def finish_image(image: Image.Image, frame_index: int) -> np.ndarray:
+    # Gentle lens falloff and warm/cool color grade.
+    y, x = np.mgrid[0:H, 0:W].astype(np.float32)
+    xn = (x - (W - 1) / 2) / (W * 0.55)
+    yn = (y - (H - 1) / 2) / (H * 0.55)
+    vignette = np.clip(1.0 - 0.18 * (xn * xn + yn * yn), 0.74, 1.0)[..., None]
+    arr = np.asarray(image.convert("RGB").resize((W, H), Image.Resampling.LANCZOS), dtype=np.float32)
+    arr *= vignette
+    arr += np.array([2.5, 1.2, -1.5], dtype=np.float32)
+
+    # Very light sensor grain, held back so the low-resolution forms stay clean.
+    rng = np.random.default_rng(900 + frame_index)
+    arr += rng.normal(0.0, 1.1, (H, W, 1)).astype(np.float32)
+    return np.clip(arr, 0, 255).astype(np.uint8)
+
+
+def render_frame(frame_index: int) -> np.ndarray:
+    rng = np.random.default_rng(2200 + frame_index * 13)
+    image = make_background(rng)
+    pressure, collapse, center, radius = draw_metal_pipe(image, float(frame_index))
+    draw_window(image, float(frame_index), pressure, collapse, center, radius)
+    draw_hose_and_pump(image, float(frame_index), pressure, collapse)
+    draw_gauge(image, float(frame_index), pressure)
+    return finish_image(image, frame_index)
+
+
+def main():
+    frames = np.stack([render_frame(i) for i in range(FRAMES)])
+    save(frames, CASE_DIR)
+    print(f"saved {frames.shape} RGB frames to {CASE_DIR}")
+
+
+if __name__ == "__main__":
+    main()
