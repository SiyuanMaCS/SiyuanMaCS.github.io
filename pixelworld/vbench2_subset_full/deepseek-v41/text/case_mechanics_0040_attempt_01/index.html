+def trajectory(t):
+    """Piecewise ballistic path with a high, long rebound."""
+    x = START[0] + HORIZONTAL_SPEED * t

+def ball_state(t):
+    x, y = trajectory(t)
+    impact = math.exp(-((t - IMPACT_TIME) / 0.115) ** 2)

+        "y": y,
+        "sx": sx,
+        "sy": sy,
+        "angle": angle,
+        "spin": spin,

+def render_ball(state):
+    """Render a shaded sphere with rotating beach-ball panels."""
+    cx = state["x"] * S
+    cy = state["y"] * S
+    r = BALL_RADIUS * S
+    sx, sy = state["sx"], state["sy"]
+    ca, sa = math.cos(state["angle"]), math.sin(state["angle"])
+
+    # Bounding box of the rotated, deformed sphere.
+    half_x = BALL_RADIUS * math.hypot(sx * ca, sy * sa) * S
+    half_y = BALL_RADIUS * math.hypot(sx * sa, sy * ca) * S
+    pad = 3 * S
+    left = int(math.floor(cx - half_x - pad))
+    top = int(math.floor(cy - half_y - pad))
+    right = int(math.ceil(cx + half_x + pad))
+    bottom = int(math.ceil(cy + half_y + pad))
+
+    yy, xx = np.mgrid[top:bottom, left:right].astype(np.float32)
+    u0 = (xx + 0.5 - cx) / r
+    v0 = (yy + 0.5 - cy) / r
+    u = (u0 * ca + v0 * sa) / sx
+    v = (-u0 * sa + v0 * ca) / sy

+    inside = q <= 1.0
+    z = np.sqrt(np.clip(1.0 - q, 0.0, 1.0))
+
+    # Six curved longitude panels.
+    longitude = np.arctan2(u, z) + state["spin"]

+    panel = np.floor(panel_phase).astype(np.int32) % 6
+    palette = np.array(

+            [248, 241, 221],
+            [38, 174, 193],
+            [248, 241, 221],
+            [247, 190, 54],
+            [248, 241, 221],

+        dtype=np.float32,
+    )
+    color = palette[panel]

+    # Light from the upper-left, with a darker lower edge and panel seams.
+    nx, ny, nz = u, -v, z
+    light = np.array([-0.52, 0.74, 0.43], dtype=np.float32)
+    light /= np.linalg.norm(light)

+    shade = 0.60 + 0.48 * diffuse + 0.055 * nz
+    shade *= 1.0 - 0.11 * np.clip(v, 0.0, 1.0)
+    shade *= 0.88 + 0.12 * nz
+
+    seam_dist = np.abs(panel_phase - np.round(panel_phase))
+    seam = np.clip(1.0 - seam_dist / 0.035, 0.0, 1.0)
+    shade *= 1.0 - 0.16 * seam

+    # Broad rubber sheen and a compact white specular highlight.
+    broad = np.exp(-(((u + 0.25) / 0.47) ** 2 + ((v + 0.35) / 0.42) ** 2) * 1.4)
+    spec = np.exp(-(((u + 0.39) / 0.145) ** 2 + ((v + 0.43) / 0.12) ** 2) * 0.5)
+    spec *= np.clip(0.25 + diffuse, 0.0, 1.0)
+    color = color * shade[..., None]

+    color += spec[..., None] * np.array([168.0, 173.0, 158.0])

+    # A thin, softly antialiased silhouette.

+def add_motion_blurred_ball(frame, t):
+    """Accumulate three sub-frame ball samples to avoid harsh 4 fps strobing."""
+    samples = [
+        (max(0.0, t - 0.028), 0.22),
+        (t, 0.56),
+        (min((N - 1) / FPS, t + 0.028), 0.22),
+    ]
+    acc_rgb = np.zeros((HH, HW, 3), dtype=np.float32)
+    acc_alpha = np.zeros((HH, HW), dtype=np.float32)
+    min_x, min_y, max_x, max_y = HW, HH, 0, 0
+
+    for sample_t, weight in samples:
+        state = ball_state(sample_t)
+        rgba, left, top = render_ball(state)
+        bottom, right = top + rgba.shape[0], left + rgba.shape[1]
+        src_rgb = rgba[..., :3].astype(np.float32)
+        src_a = rgba[..., 3].astype(np.float32) / 255.0
+        acc_rgb[top:bottom, left:right] += src_rgb * src_a[..., None] * weight
+        acc_alpha[top:bottom, left:right] += src_a * weight
+        min_x, min_y = min(min_x, left), min(min_y, top)
+        max_x, max_y = max(max_x, right), max(max_y, bottom)
+
+    crop_alpha = acc_alpha[min_y:max_y, min_x:max_x]
+    crop_rgb = acc_rgb[min_y:max_y, min_x:max_x]
+    safe_alpha = np.maximum(crop_alpha, 1e-5)
+    crop_rgb = np.clip(crop_rgb / safe_alpha[..., None], 0.0, 255.0)
+    overlay = np.zeros((HH, HW, 4), dtype=np.uint8)
+    overlay[min_y:max_y, min_x:max_x, :3] = crop_rgb.astype(np.uint8)
+    overlay[min_y:max_y, min_x:max_x, 3] = np.uint8(np.clip(crop_alpha, 0.0, 1.0) * 255.0)
+    ball_layer = Image.fromarray(overlay)
+    return Image.alpha_composite(frame.convert("RGBA"), ball_layer), ball_state(t)
+
+
+def render_frame(index):
+    t = index / FPS
+    frame = BACKGROUND.copy()
+    frame = add_water_sparkle(frame, t)
+
+    state = ball_state(t)
+    frame = add_shadow(frame, t, state["x"], state["y"])
+    frame = add_impact_effects(frame, t)
+    frame, state = add_motion_blurred_ball(frame, t)
+
+    # Very light vignette and warm color finishing.
+    frame = frame.convert("RGB").resize((W, H), Image.Resampling.LANCZOS)
+    rgb = np.asarray(frame, dtype=np.float32)
+    yy, xx = np.mgrid[0:H, 0:W].astype(np.float32)
+    nx = (xx - (W - 1) / 2.0) / (W * 0.62)
+    ny = (yy - (H - 1) / 2.0) / (H * 0.62)
+    vignette = 1.0 - 0.055 * np.clip(nx * nx + ny * ny, 0.0, 1.0)
+    rgb *= vignette[..., None]
+    rgb[..., 0] *= 1.012
+    rgb[..., 2] *= 0.993
+    return np.uint8(np.clip(rgb, 0, 255))
+
+
+def main():
+    frames = np.stack([render_frame(i) for i in range(N)])
+    artifact_io.save(frames, CASE_DIR)
+    print(f"Rendered {N} frames at {W}x{H}, {FPS:g} fps into {CASE_DIR}")
+

+if __name__ == "__main__":
+    main()
