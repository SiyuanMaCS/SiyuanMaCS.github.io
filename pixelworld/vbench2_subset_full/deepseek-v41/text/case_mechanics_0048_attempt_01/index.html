+def _catmull_rom(points: np.ndarray, samples_per_segment: int = 34) -> np.ndarray:
+    extended = np.vstack((points[0], points, points[-1]))
+    result = []
+    for index in range(1, len(extended) - 2):
+        p0, p1, p2, p3 = extended[index - 1 : index + 3]
+        for sample in range(samples_per_segment):
+            u = sample / samples_per_segment
+            u2 = u * u
+            u3 = u2 * u
+            point = 0.5 * (
+                (2.0 * p1)
+                + (-p0 + p2) * u
+                + (2.0 * p0 - 5.0 * p1 + 4.0 * p2 - p3) * u2
+                + (-p0 + 3.0 * p1 - 3.0 * p2 + p3) * u3
+            )
+            result.append(point)
+    result.append(points[-1])
+    return np.asarray(result, dtype=np.float32)

+def _point_at_progress(
+    path: np.ndarray,
+    cumulative: np.ndarray,
+    progress: float,
+) -> tuple[np.ndarray, np.ndarray]:
+    target = float(cumulative[-1]) * min(1.0, max(0.0, progress))
+    index = int(np.searchsorted(cumulative, target, side="right") - 1)
+    index = min(index, len(path) - 2)
+    segment = cumulative[index + 1] - cumulative[index]
+    fraction = 0.0 if segment <= 1e-6 else (target - cumulative[index]) / segment
+    point = path[index] + (path[index + 1] - path[index]) * fraction
+    tangent = path[index + 1] - path[index]
+    tangent /= max(1e-6, float(np.linalg.norm(tangent)))
+    return point.astype(np.float32), tangent.astype(np.float32)

+    normal = np.array([-math.sin(angle), math.cos(angle)], dtype=np.float32)

+    radius = np.zeros_like(along)

+    radius[nib] = 1.0 + 3.15 * np.power(along[nib] / 13.0, 0.72)
+    cone = (along >= 13.0) & (along < 20.0)
+    radius[cone] = 4.15 + 2.0 * (along[cone] - 13.0) / 7.0
+    grip = (along >= 20.0) & (along < 43.0)
+    radius[grip] = 6.15 + 0.18 * np.sin(along[grip] * 1.1)

+    tail = (along >= 128.0) & (along < 135.0)

+def _apply_ink(

+def _pen_shadow(pen_mask: np.ndarray) -> np.ndarray:
+    silhouette = Image.fromarray((pen_mask * 255).astype(np.uint8), "L")
+    shifted = Image.new("L", (CW, CH), 0)
+    shifted.paste(silhouette, (int(3.2 * SS), int(4.0 * SS)))
+    shifted = shifted.filter(ImageFilter.GaussianBlur(2.0 * SS))
+    return np.asarray(shifted, dtype=np.float32) / 255.0
+
+
+def _contact_details(
+    frame: np.ndarray,
+    tip: np.ndarray,
+    tangent: np.ndarray,
+    frame_index: int,
+) -> np.ndarray:
+    # Small contact shadow and displaced fibers sell pressure against the cloth.
+    shadow = Image.new("L", (CW, CH), 0)
+    draw = ImageDraw.Draw(shadow)
+    sx, sy = tip[0] * SS + 1.3 * SS, tip[1] * SS + 1.9 * SS
+    draw.ellipse(
+        (sx - 7.2 * SS, sy - 3.2 * SS, sx + 7.2 * SS, sy + 3.2 * SS),
+        fill=95,
+    )
+    shadow = shadow.filter(ImageFilter.GaussianBlur(1.8 * SS))
+    shade = np.asarray(shadow, dtype=np.float32) / 255.0
+    frame *= (1.0 - 0.23 * shade[..., None])
+
+    detail = Image.new("RGBA", (CW, CH), (0, 0, 0, 0))
+    detail_draw = ImageDraw.Draw(detail)
+    normal = np.array([-tangent[1], tangent[0]], dtype=np.float32)
+    for index in range(7):
+        side = -1.0 if index % 2 == 0 else 1.0
+        offset = (1.1 + 0.58 * index) * side
+        root = tip + normal * offset
+        angle = math.atan2(tangent[1], tangent[0]) + side * (0.52 + 0.11 * index)
+        length = 2.2 + 0.52 * ((index * 3 + frame_index) % 5)
+        end = root + np.array([math.cos(angle), math.sin(angle)], dtype=np.float32) * length
+        detail_draw.line(
+            (float(root[0] * SS), float(root[1] * SS),
+             float(end[0] * SS), float(end[1] * SS)),
+            fill=(196, 199, 177, 45),
+            width=max(1, SS // 2),
+        )
+
+    for index in range(5):
+        angle = (index * 2.31 + frame_index * 0.41) % math.tau
+        radius = 2.2 + (index % 3) * 1.15
+        px = tip[0] + math.cos(angle) * radius
+        py = tip[1] + math.sin(angle) * radius
+        size = 0.45 + 0.12 * (index % 2)
+        detail_draw.ellipse(
+            ((px - size) * SS, (py - size) * SS,
+             (px + size) * SS, (py + size) * SS),
+            fill=(116, 24, 66, 70),
+        )
+
+    detail_image = Image.fromarray(frame.astype(np.uint8), "RGB").convert("RGBA")
+    detail_image.alpha_composite(detail)
+    return np.asarray(detail_image.convert("RGB"), dtype=np.float32)
+
+
+def _finish_frame(frame: np.ndarray, frame_index: int) -> np.ndarray:
+    image = Image.fromarray(np.clip(frame, 0.0, 255.0).astype(np.uint8), "RGB")
+    image = image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
+    image = image.filter(ImageFilter.UnsharpMask(radius=0.55, percent=48, threshold=2))
+    out = np.asarray(image, dtype=np.float32)
+
+    xs = np.arange(WIDTH, dtype=np.float32)[None, :] / WIDTH
+    ys = np.arange(HEIGHT, dtype=np.float32)[:, None] / HEIGHT
+    vignette = 1.0 - 0.14 * ((xs - 0.47) ** 2 * 1.35 + (ys - 0.50) ** 2)
+    out *= vignette[..., None]
+    out += np.array([2.5, 1.0, -0.5], dtype=np.float32)
+
+    rng = np.random.default_rng(1200 + frame_index)
+    grain = rng.normal(0.0, 0.72, size=(HEIGHT, WIDTH, 1)).astype(np.float32)
+    out += grain
+    return np.clip(out, 0.0, 255.0).astype(np.uint8)
+
+
+def main() -> None:
+    fabric, height_norm = _make_fabric()
+    path, cumulative = _path_geometry()
+    label = _make_label()
+    frames = []
+
+    for frame_index in range(FRAME_COUNT):
+        time_value = frame_index / (FRAME_COUNT - 1)
+        progress = time_value ** 0.90
+        tip, tangent = _point_at_progress(path, cumulative, progress)
+
+        frame = _apply_ink(fabric.copy(), path, cumulative, progress, height_norm)
+        frame = _contact_details(frame, tip, tangent, frame_index)
+
+        angle = math.radians(-68.0 + 1.2 * math.sin(math.tau * time_value + 0.35))
+        pen_rgb, pen_mask = _render_pen(tip, angle, label)
+        shadow = _pen_shadow(pen_mask)
+        frame *= (1.0 - 0.34 * shadow[..., None])
+
+        pen_alpha = pen_mask[..., None]
+        frame = frame * (1.0 - pen_alpha) + pen_rgb * pen_alpha
+        frames.append(_finish_frame(frame, frame_index))
+
+    artifact_io.save(np.asarray(frames, dtype=np.uint8), CASE_DIR)
+
+
+if __name__ == "__main__":
+    main()
