+def blur_noise(shape: tuple[int, int], sigma: float) -> np.ndarray:
+    field = gaussian_filter(RNG.normal(size=shape).astype(np.float32), sigma, mode="reflect")

+def normalized(field: np.ndarray) -> np.ndarray:
+    lo, hi = np.percentile(field, (1.0, 99.0))
+    return np.clip((field - lo) / (hi - lo + 1e-6), 0.0, 1.0)

+def chalk_tip(t: float) -> tuple[float, float]:
+    u = t / (FRAME_COUNT - 1)

+def draw_dust(wall: np.ndarray, t: float) -> np.ndarray:

+def render_frame(index: int, wall_base: np.ndarray, coarse: np.ndarray, micro: np.ndarray) -> np.ndarray:
+    t = index / FPS
+    tip = chalk_tip(t)

+    exposure = 1.0 + 0.009 * math.sin(2.0 * math.pi * t / 4.8 + 0.4)
+    frame *= exposure

+

+    small = np.clip((small - 0.5) * 1.055 + 0.5, 0.0, 1.0)
+
+    frame_noise = RNG.normal(0.0, 0.0045, (HEIGHT, WIDTH, 1)).astype(np.float32)
+    small += frame_noise
+    return np.clip(small * 255.0 + 0.5, 0, 255).astype(np.uint8)

+def save_mp4(frames: np.ndarray) -> None:
