+def smoothstep(edge0: float, edge1: float, value):
+    x = np.clip((value - edge0) / (edge1 - edge0), 0.0, 1.0)
+    return x * x * (3.0 - 2.0 * x)

+def rotate_points(x, y, angle_deg):
+    a = np.deg2rad(angle_deg)
+    c, s = np.cos(a), np.sin(a)
+    return c * x - s * y, s * x + c * y

+def rounded_rect_sdf(x, y, hw, hh, radius):
+    qx = np.abs(x) - hw + radius
+    qy = np.abs(y) - hh + radius
+    outside = np.hypot(np.maximum(qx, 0.0), np.maximum(qy, 0.0))
+    inside = np.minimum(np.maximum(qx, qy), 0.0)
+    return outside + inside - radius

+def rounded_rect_outline(cx, cy, hw, hh, radius, angle, samples=48):
+    pts = []
+    corners = [
+        (hw - radius, hh - radius, 0.0),
+        (-hw + radius, hh - radius, 90.0),
+        (-hw + radius, -hh + radius, 180.0),
+        (hw - radius, -hh + radius, 270.0),
+    ]
+    for ox, oy, start in corners:
+        for k in range(samples):
+            a = np.deg2rad(start + 90.0 * k / (samples - 1))
+            pts.append((ox + radius * np.cos(a), oy + radius * np.sin(a)))
+    x, y = rotate_points(
+        np.array([p[0] for p in pts]), np.array([p[1] for p in pts]), angle
+    )
+    return [(float(cx + px), float(cy + py)) for px, py in zip(x, y)]

+def render_film(background, shadow, cx, cy, hw, hh, angle, q, p):
+    yy, xx = np.mgrid[0:H, 0:W].astype(np.float32)
+    xb = xx / SS
+    yb = yy / SS
+    lx, ly = rotate_points(xb - cx, yb - cy, -angle)
+
+    radius = 13.5 - 2.0 * q
+    outer_d = rounded_rect_sdf(lx, ly, hw, hh, radius)
+    inner_d = rounded_rect_sdf(lx, ly, hw - 6.3, hh - 6.3, max(5.0, radius - 2.2))
+
+    outer_mask = smoothstep(0.75, -0.75, outer_d)
+    inner_mask = smoothstep(0.60, -0.60, inner_d)
+    seam_mask = np.clip(outer_mask - inner_mask, 0.0, 1.0)
+
+    u = lx / max(hw, 1.0)
+    v = ly / max(hh, 1.0)
+    rho = np.sqrt((u / 0.92) ** 2 + (v / 0.97) ** 2)
+    dome = np.clip(1.0 - rho**1.65, 0.0, 1.0) ** 0.58
+
+    # The center starts as a taut air pillow and flattens as the pump runs.
+    z = (1.0 - q) * (10.8 * dome + 1.7 * np.clip(1.0 - v**2, 0.0, 1.0))
+
+    edge_env = (
+        0.28
+        + 0.95 * np.clip((np.abs(v) - 0.22) / 0.78, 0.0, 1.0)

+    w3 = np.sin(0.13 * (lx + ly) + 1.75 * np.sin(0.045 * (lx - ly) + 1.30) + 0.9 * q)
+    z += wrinkle_env * (0.72 * w1 + 0.48 * w2 + 0.34 * w3)
+
+    # Long folds converge toward the valve, reinforcing the extraction direction.
+    dxv = lx - (hw - 0.5)
+    dyv = ly

+    ny = -0.72 * dzdy

+    norm = np.sqrt(nx * nx + ny * ny + nz * nz)
+    nx, ny, nz = nx / norm, ny / norm, nz / norm
+
+    light = np.array([-0.52, -0.64, 0.56], dtype=np.float32)
+    light /= np.linalg.norm(light)
+    diffuse = np.clip(nx * light[0] + ny * light[1] + nz * light[2], 0.0, 1.0)
+
+    half = light + np.array([0.0, 0.0, 1.0], dtype=np.float32)
+    half /= np.linalg.norm(half)

+    spec = spec**31
+
+    # Two long reflections describe the curvature of the clear plastic.
+    band1 = np.exp(
+        -(
+            (
+                (lx - 10.0) * 0.56
+                + (ly + 3.0) * 0.83

+    band2 = np.exp(

+    film_rgb += (118.0 * fine_glint + 52.0 * stripe + 20.0 * dome * (1.0 - q))[..., None]
+    film_rgb = np.clip(film_rgb, 0, 255)

+    crease_energy = np.clip(

+    body_alpha = inner_mask * np.clip(
+        0.18 + 0.23 * crease_energy + 0.17 * q * wrinkle_env, 0.0, 0.68

+    seam_col = np.array([204.0, 205.0, 193.0], dtype=np.float32)

+    # A faint transparent edge glow helps the clear bag survive downsampling.

+def render_frame(index):
+    p = index / (FRAMES - 1)
+    # A brief pause, a strong evacuation, then a slower final tightening.
+    q = smoothstep(0.02, 0.93, p) ** 0.92

+    cx = 121.0 - 1.5 * q
+    cy = 70.5 + 0.8 * q
+    hw = 79.5 - 6.2 * q
+    hh = 43.5 - 5.8 * q

+    valve_x = cx + hw * np.cos(np.deg2rad(angle)) - 1.0
+    valve_y = cy + hw * np.sin(np.deg2rad(angle)) + 0.4

+    shadow = make_shadow(cx, cy, hw, hh, angle, q)

+    image = Image.fromarray(np.uint8(np.clip(film, 0, 255)), "RGB").convert("RGBA")
+
+    hose = make_hose(cx, valve_x, valve_y, p)
+    # The hose begins behind the clear film and emerges at the valve.
+    image = Image.alpha_composite(hose, image)
+    image = draw_seal_texture(image, cx, cy, hw, hh, angle, q)
+    image = draw_valve(image, valve_x, valve_y, q, p)
+
+    # Final studio bloom, kept subtle so creases remain crisp.
