+def fbm(shape, scales, weights, seed):
+    rng = np.random.default_rng(seed)
+    field = np.zeros(shape, dtype=np.float32)
+    for scale, weight in zip(scales, weights):
+        small = rng.normal(0.0, 1.0, (max(2, shape[0] // scale), max(2, shape[1] // scale)))
+        image = Image.fromarray(np.uint8(np.clip(0.5 + small * 0.13, 0.0, 1.0) * 255))
+        image = image.resize((shape[1], shape[0]), Image.Resampling.BICUBIC)
+        field += weight * (np.asarray(image, dtype=np.float32) / 255.0)
+    field -= field.mean()
+    field /= max(field.std(), 1e-5)
+    return field

+def catmull_rom_path(points, samples_per_segment=80):
+    points = np.asarray(points, dtype=np.float64)
+    padded = np.vstack([points[0], points, points[-1]])
+    curve = []
+    for index in range(1, len(padded) - 2):
+        p0, p1, p2, p3 = padded[index - 1:index + 3]
+        for t in np.linspace(0.0, 1.0, samples_per_segment, endpoint=False):
+            t2, t3 = t * t, t * t * t
+            point = 0.5 * (
+                2.0 * p1
+                + (-p0 + p2) * t
+                + (2.0 * p0 - 5.0 * p1 + 4.0 * p2 - p3) * t2
+                + (-p0 + 3.0 * p1 - 3.0 * p2 + p3) * t3
+            )
+            curve.append(point)
+    curve.append(points[-1])
+    curve = np.asarray(curve)
+    lengths = np.r_[0.0, np.cumsum(np.linalg.norm(np.diff(curve, axis=0), axis=1))]
+    return curve, lengths / lengths[-1]
