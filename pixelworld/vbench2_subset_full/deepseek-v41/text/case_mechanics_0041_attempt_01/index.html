+def smoothstep(edge0, edge1, x):
+    t = np.clip((x - edge0) / (edge1 - edge0), 0.0, 1.0)
+    return t * t * (3.0 - 2.0 * t)

+def trajectory(frame):

+    t = frame / FPS
+    impacts = {4, 10, 15, 18}
+    if frame in impacts:

+def make_background():

+def render_frame(background, frame):
+    cx, cy, rx, ry, contact = trajectory(frame)
+    image = background.astype(np.float32).copy()
+    shadow, reflection = shadow_layer(cx, cy, rx, ry, contact)
+
+    reflection_color = np.array([0.58, 0.16, 0.10], dtype=np.float32)
+    image = image * (1.0 - shadow[:, :, None])
+    image += reflection[:, :, None] * reflection_color[None, None, :] * 255.0
+
+    x0, y0, x1, y1, inside, rgb = render_ball(cx, cy, rx, ry, contact)
+    patch = image[y0:y1, x0:x1]
+    mask = inside[:, :, None].astype(np.float32)
+    patch[:] = patch * (1.0 - mask) + rgb * 255.0 * mask
+    image[y0:y1, x0:x1] = patch
+
+    # A subtle contact darkening helps the flattened underside read at 144p.
+    if contact:
+        yy, xx = np.mgrid[0:H, 0:W].astype(np.float32)
+        contact_ao = np.exp(
+            -(((xx - BALL_X * SS) / (24.0 * SS)) ** 2 + ((yy - (FLOOR_Y - 1.0) * SS) / (4.0 * SS)) ** 2)
+        )
+        image *= (1.0 - 0.12 * contact_ao[:, :, None])
+
+    image = np.clip(image, 0, 255).astype(np.uint8)
+    return np.asarray(
+        Image.fromarray(image).resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS),
+        dtype=np.uint8,
+    )
+
+
+def main():
+    background = make_background()
+    frames = np.empty((FRAME_COUNT, HEIGHT, WIDTH, 3), dtype=np.uint8)
+    for index in range(FRAME_COUNT):
+        frames[index] = render_frame(background, index)
+        print(f"rendered frame {index + 1:02d}/{FRAME_COUNT}", flush=True)
+    artifact_io.save(frames, CASE_DIR)
+
+
+if __name__ == "__main__":
+    main()
