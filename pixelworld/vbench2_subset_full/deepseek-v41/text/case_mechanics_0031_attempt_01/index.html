+def smoothstep(edge0, edge1, x):
+    t = np.clip((x - edge0) / (edge1 - edge0), 0.0, 1.0)

+def render_frame(index):
+    t = index / (FRAMES - 1)
+    # Slow constant pressure, with a tiny acceleration into the puncture.
+    progress = 0.18 + 0.82 * (t ** 1.35)
+    indentation = 0.28 + 0.62 * smoothstep(0.03, 0.83, t)
+    cx, cy = W * 0.505, H * 0.61
+
+    y, x = np.mgrid[0:H, 0:W].astype(np.float32)
+    arr = make_rubber(x, y, indentation)
+
+    # Contact shadow beneath the tip, widening with pressure.
+    dx = x - cx
+    dy = y - (cy + 1.5 * SS)
+    contact = np.exp(-((dx / (11.0 * SS)) ** 2 + (dy / (6.0 * SS)) ** 2))
+    arr *= (1.0 - 0.35 * contact * (0.6 + 0.4 * progress))[..., None]
+
+    # A second, broad shadow from the pencil shaft.
+    shaft = np.exp(-((((x - cx - 46 * SS) / (32 * SS)) ** 2 + ((y - cy + 70 * SS) / (70 * SS)) ** 2)))
+    arr *= (1.0 - 0.12 * shaft)[..., None]
+
+    canvas = Image.fromarray(np.clip(arr, 0, 255).astype(np.uint8), "RGB").convert("RGBA")
+    tip_x = cx + 0.8 * SS * np.sin(t * 0.7)
+    tip_y = cy - 0.35 * SS * t
+    angle = -1.075 + 0.018 * np.sin(t * 2.0)
+    draw_pencil(canvas, tip_x, tip_y, angle, progress)
+    add_tear(canvas, indentation, smoothstep(0.62, 0.98, t))
+

+    out = canvas.convert("RGB").resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
+    frame = np.asarray(out, dtype=np.uint8).copy()
+    frame = np.clip(frame.astype(np.float32) * np.array([1.025, 1.0, 0.975]), 0, 255).astype(np.uint8)
+    return frame
+
+
+def main():
+    frames = np.stack([render_frame(i) for i in range(FRAMES)])

+    main()
