+def trajectory_at(time_s: float) -> tuple[float, float, int, float]:
+    contact_y = FLOOR_CONTACT - BALL_RADIUS
+    if time_s <= CONTACT_TIMES[0]:
+        y = BALL_START_Y + 0.5 * GRAVITY * time_s * time_s
+        vy = GRAVITY * time_s
+        return y, vy, -1, time_s
+
+    segment = 0
+    for index in range(len(CONTACT_TIMES) - 1):
+        if time_s < CONTACT_TIMES[index + 1]:
+            segment = index
+            break
+    else:
+        segment = len(CONTACT_TIMES) - 1
+
+    start = CONTACT_TIMES[segment]
+    if segment < len(CONTACT_TIMES) - 1:
+        flight = CONTACT_TIMES[segment + 1] - start
+    else:
+        flight = 0.8
+    rebound_speed = 0.5 * GRAVITY * flight
+    dt = time_s - start
+    y = contact_y - rebound_speed * dt + 0.5 * GRAVITY * dt * dt
+    vy = -rebound_speed + GRAVITY * dt
+    return y, vy, segment, dt

+def impact_response(time_s: float) -> tuple[float, float, float]:
+    compression = 0.0
+    for contact in CONTACT_TIMES:
+        delta = time_s - contact
+        if abs(delta) < 0.11:
+            compression = max(compression, math.exp(-((delta / 0.052) ** 2)))

+    # A damped wobble after impact gives the sphere a rubbery, non-rigid feel.
+    wobble = 0.0
+    for contact in CONTACT_TIMES:
+        delta = time_s - contact
+        if 0.0 < delta < 0.55:
+            wobble += 0.036 * math.exp(-delta / 0.20) * math.cos(2.0 * math.pi * delta / 0.31)
+
+    squash_y = 1.0 - 0.29 * compression + wobble
+    squash_x = 1.0 + 0.22 * compression - 0.45 * wobble
+    return squash_x, squash_y, compression

+def make_ball(

+def add_impact_ring(scene: Image.Image, compression: float) -> Image.Image:
+    if compression < 0.05:
+        return scene
+    layer = Image.new("RGBA", (W, H), (0, 0, 0, 0))

+def render_frame(frame_index: int, background: Image.Image) -> np.ndarray:
+    time_s = frame_index / FPS
+    center_y, velocity_y, segment, segment_time = trajectory_at(time_s)
+    squash_x, squash_y, compression = impact_response(time_s)
+
+    # Pin the flattened underside to the floor at impact.
+    if compression > 0.015:
+        anchored_y = FLOOR_CONTACT - BALL_RADIUS * squash_y
+        blend = smoothstep(0.015, 0.25, compression)
+        center_y = center_y * (1.0 - blend) + anchored_y * blend
+

+    angle = 0.025 * time_s + (0.17 * (segment + 1) if segment >= 0 else 0.0)
+    ball, ball_pos = make_ball(

+        center_y * SS,
+        squash_x,
+        squash_y,
+        angle,
+    )
+
+    scene = composite_reflection(scene, ball, ball_pos, center_y)
+    scene = add_impact_ring(scene, compression)
+

+        ghost = ball.copy()

+        ghost_alpha *= 0.10 * speed
+        ghost.putalpha(Image.fromarray(np.clip(ghost_alpha, 0.0, 255.0).astype(np.uint8), "L"))
+        ghost = ghost.filter(ImageFilter.GaussianBlur(0.8 * SS))
+        scene.alpha_composite(ghost, (ball_pos[0], ball_pos[1] + ghost_offset))
+
+    scene.alpha_composite(ball, ball_pos)
+
+    final = scene.convert("RGB").resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
+    arr = np.asarray(final).astype(np.float32) / 255.0
+    arr = np.clip((arr - 0.5) * 1.035 + 0.5, 0.0, 1.0)
+    luminance = arr[:, :, 0] * 0.2126 + arr[:, :, 1] * 0.7152 + arr[:, :, 2] * 0.0722
+    arr = np.clip(luminance[:, :, None] + (arr - luminance[:, :, None]) * 1.055, 0.0, 1.0)
+    return np.round(arr * 255.0).astype(np.uint8)
+
