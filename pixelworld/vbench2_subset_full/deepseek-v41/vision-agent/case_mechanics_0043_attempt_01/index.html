+def smoothstep(value):
+    value = np.clip(value, 0.0, 1.0)

+def blur_array(array, radius):
+    """Separable box blur, applied repeatedly to approximate a Gaussian."""
+    result = np.asarray(array, dtype=np.float32)
+    if radius <= 0:
+        return result
+    passes = 3
+    kernel_radius = max(1, int(round(radius / math.sqrt(passes))))
+    for _ in range(passes):
+        padded = np.pad(result, ((kernel_radius, kernel_radius), (0, 0)), mode="reflect")
+        result = np.cumsum(padded, axis=0, dtype=np.float32)
+        result = (result[2 * kernel_radius:] - result[:-2 * kernel_radius]) / (
+            2 * kernel_radius
+        )
+        padded = np.pad(result, ((0, 0), (kernel_radius, kernel_radius)), mode="reflect")
+        result = np.cumsum(padded, axis=1, dtype=np.float32)
+        result = (result[:, 2 * kernel_radius:] - result[:, :-2 * kernel_radius]) / (
+            2 * kernel_radius
+        )
+    return result

+def catmull_rom(points, samples_per_segment=56):
+    """Interpolate a smooth path through the supplied control points."""
+    pts = np.asarray(points, dtype=np.float64)
+    padded = np.vstack((pts[0], pts, pts[-1]))
+    pieces = []
+    for index in range(1, len(padded) - 2):
+        p0, p1, p2, p3 = padded[index - 1:index + 3]
+        t = np.linspace(0.0, 1.0, samples_per_segment, endpoint=False)[:, None]
+        curve = 0.5 * (
+            2.0 * p1
+            + (-p0 + p2) * t
+            + (2.0 * p0 - 5.0 * p1 + 4.0 * p2 - p3) * t**2
+            + (-p0 + 3.0 * p1 - 3.0 * p2 + p3) * t**3
+        )
+        pieces.append(curve)
+    pieces.append(pts[-1][None, :])
+    return np.vstack(pieces)

+def arc_resample(points, spacing=0.45):
+    delta = np.diff(points, axis=0)
+    lengths = np.hypot(delta[:, 0], delta[:, 1])
+    cumulative = np.r_[0.0, np.cumsum(lengths)]
+    if cumulative[-1] < 1e-6:
+        return points.copy(), cumulative
+    distances = np.arange(0.0, cumulative[-1] + spacing, spacing)
+    distances = distances[distances <= cumulative[-1]]
+    x = np.interp(distances, cumulative, points[:, 0])
+    y = np.interp(distances, cumulative, points[:, 1])
+    return np.column_stack((x, y)), distances

+def point_on_stroke(points, fraction):

+def state_at_time(seconds):
+    """Return completed strokes, active progress, and the current chalk tip."""
+    schedule = [
+        ("stroke", 0, 0.0, 1.58),
+        ("lift", 0, 1.58, 1.80),
+        ("stroke", 1, 1.80, 3.13),
+        ("lift", 1, 3.13, 3.35),
+        ("stroke", 2, 3.35, 4.54),

+    progress = [0.0, 0.0, 0.0]
+    tip = STROKES[0][0].copy()
+    lift = 0.0
+    for kind, index, start, end in schedule:
+        if seconds < start:
+            break
+        local = np.clip((seconds - start) / (end - start), 0.0, 1.0)
+        if kind == "stroke":
+            progress[index] = local
+            tip = point_on_stroke(STROKES[index], local)

+            lift = math.sin(math.pi * local) * 0.75
