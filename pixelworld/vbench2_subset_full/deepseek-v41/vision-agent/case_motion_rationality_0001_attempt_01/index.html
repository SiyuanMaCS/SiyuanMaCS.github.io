+def solve_elbow(shoulder, wrist, upper=39.0, lower=35.0):
+    sx, sy = shoulder
+    wx, wy = wrist
+    dx, dy = wx - sx, wy - sy
+    distance = max(0.001, math.hypot(dx, dy))
+    distance = min(distance, upper + lower - 0.01)
+    ux, uy = dx / math.hypot(dx, dy), dy / math.hypot(dx, dy)
+    wx2, wy2 = sx + ux * distance, sy + uy * distance
+    a = (upper * upper - lower * lower + distance * distance) / (2 * distance)
+    h = math.sqrt(max(0.0, upper * upper - a * a))
+    mx, my = sx + ux * a, sy + uy * a
+    perp1 = (-uy, ux)
+    perp2 = (uy, -ux)
+    e1 = (mx + perp1[0] * h, my + perp1[1] * h)
+    e2 = (mx + perp2[0] * h, my + perp2[1] * h)
+    return e1 if e1[1] > e2[1] else e2
