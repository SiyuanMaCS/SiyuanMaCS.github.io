+def transformed_ray(origins, directions, pose):

+def intersect_ellipsoid(origins, directions, center, radii):

+    inv_r2 = 1.0 / np.square(np.asarray(radii))
+    a = np.sum(np.square(directions) * inv_r2, axis=-1)
+    b = 2.0 * np.sum(oc * directions * inv_r2, axis=-1)

+    disc = b * b - 4.0 * a * c

+    root = np.sqrt(np.maximum(disc, 0.0))
+    t0 = (-b - root) / (2.0 * a)
+    t1 = (-b + root) / (2.0 * a)

+    valid &= t > 0.001
+    return t, valid

+def downsample_and_finish(linear):
+    image = linear.reshape(H, SS, W, SS, 3).mean(axis=(1, 3))
+    image = np.clip(image, 0.0, 1.0)
+    # Small unsharp pass after resampling restores the rubber highlight edges.

+    result = np.asarray(pil).astype(np.float64)

+    yy, xx = np.mgrid[0:H, 0:W]
+    nx = (xx - W * 0.5) / (W * 0.5)
+    ny = (yy - H * 0.5) / (H * 0.5)

+    rng = np.random.default_rng(3700 + int(xx[0, 0]))
+    grain = rng.normal(0.0, 1.45, size=(H, W, 1))
+    result = result * vignette[..., None] + grain
