+def cubic(
+    start: tuple[float, float],
+    control1: tuple[float, float],
+    control2: tuple[float, float],
+    end: tuple[float, float],
+    count: int = 30,
+) -> list[tuple[float, float]]:
+    result = []
+    for i in range(count + 1):
+        t = i / count
+        u = 1.0 - t
+        x = u**3 * start[0] + 3 * u * u * t * control1[0] + 3 * u * t * t * control2[0] + t**3 * end[0]
+        y = u**3 * start[1] + 3 * u * u * t * control1[1] + 3 * u * t * t * control2[1] + t**3 * end[1]
+        result.append((x, y))
+    return result

+def gradient(size: tuple[int, int], top: tuple[int, int, int], bottom: tuple[int, int, int]) -> Image.Image:
+    width, height = size
+    t = np.linspace(0.0, 1.0, height, dtype=np.float32)[:, None, None]
+    a = np.asarray(top, dtype=np.float32)[None, None, :]
+    b = np.asarray(bottom, dtype=np.float32)[None, None, :]
+    array = np.broadcast_to(a * (1.0 - t) + b * t, (height, width, 3)).copy()
+    return Image.fromarray(array.astype(np.uint8), "RGB")

+def smoothstep(value: float) -> float:

+def sample_keys(frame: int, keys: list[tuple[int, float, float, float]]) -> tuple[float, float, float]:
+    if frame <= keys[0][0]:
+        return keys[0][1:]
+    if frame >= keys[-1][0]:
+        return keys[-1][1:]
+    for (f0, x0, y0, a0), (f1, x1, y1, a1) in zip(keys[:-1], keys[1:]):
+        if f0 <= frame <= f1:
+            t = smoothstep((frame - f0) / max(1, f1 - f0))
+            return x0 + (x1 - x0) * t, y0 + (y1 - y0) * t, a0 + (a1 - a0) * t
+    return keys[-1][1:]
