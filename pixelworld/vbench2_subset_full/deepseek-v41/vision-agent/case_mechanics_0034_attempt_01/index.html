+def smoothstep(edge0, edge1, value):

+def capsule_field(x, y, p0, p1, radius):
+    ax, ay = p0
+    bx, by = p1
+    vx = bx - ax
+    vy = by - ay
+    length_sq = vx * vx + vy * vy
+    t = np.clip(((x - ax) * vx + (y - ay) * vy) / length_sq, 0.0, 1.0)
+    closest_x = ax + t * vx
+    closest_y = ay + t * vy
+    dx = x - closest_x
+    dy = y - closest_y
+    distance = np.sqrt(dx * dx + dy * dy)
+    return t, dx, dy, distance

+def shade_capsule(x, y, p0, p1, radius, light, view, skin_base, seed=0):
+    t, dx, dy, distance = capsule_field(x, y, p0, p1, radius)
+    radial = np.clip(distance / radius, 0.0, 1.0)
+    normal_z = np.sqrt(np.maximum(1.0 - radial * radial, 0.0))
+    safe_distance = np.maximum(distance, 1e-5)
+    normal_x = dx / safe_distance * radial
+    normal_y = dy / safe_distance * radial
+    diffuse = np.clip(
+        normal_x * light[0] + normal_y * light[1] + normal_z * light[2],
+        0.0,
+        1.0,
+    )
+    half_vector = normalize(light + view)
+    specular = np.clip(
+        normal_x * half_vector[0]
+        + normal_y * half_vector[1]
+        + normal_z * half_vector[2],
+        0.0,
+        1.0,
+    )
+    color = skin_base * (0.31 + 0.87 * diffuse[..., None])
+    color += (specular**23)[..., None] * np.array([0.31, 0.235, 0.19], np.float32)
+
+    # Fine skin relief and two soft knuckle creases.
+    texture = (
+        0.018 * np.sin(x * 1.8 + y * 0.7 + seed)
+        + 0.010 * np.sin(x * 0.55 - y * 2.1)

+    for position in crease_positions:
+        crease += np.exp(-((t - position) / 0.028) ** 2)
+    color *= (1.0 - 0.09 * crease)[..., None]
+    color += texture[..., None]
+    alpha = smoothstep(radius + 0.35, radius - 0.35, distance)
+    return color, alpha

+def draw_nail(x, y, center, direction, length, width, alpha_parent):

+    dy = y - center[1]

+    highlight = np.exp(-((across + width * 0.22) / (width * 0.25)) ** 2) * 0.08
+    color += highlight[..., None]

+def contact_lip(x, y, press, tip_x, tip_y):
+    if press <= 0.01:
+        return np.zeros(x.shape + (3,), np.float32), np.zeros_like(x)
+    rx = x - (tip_x + 0.3)
+    ry = y - (tip_y - 0.2)
+    radius = np.sqrt(rx * rx + ry * ry)
+    lower_crescent = smoothstep(tip_y - 2.2, tip_y + 0.2, y)
+    ring = np.exp(-((radius - 4.3) / 1.55) ** 2)
+    center_cover = np.exp(-(radius / 4.4) ** 4)
+    alpha = press * np.maximum(0.70 * ring, 0.44 * center_cover) * lower_crescent
+    highlight = np.exp(-((radius - 5.1) / 1.0) ** 2) * smoothstep(tip_y, tip_y + 4.0, y)
+    color = np.array([0.36, 0.35, 0.31], np.float32)
+    color = (
+        color
+        + highlight[..., None] * np.array([0.26, 0.24, 0.20], np.float32)
+    )
+    return color, alpha

+def composite(base, color, alpha):

+def add_bloom(image):

+        axis=-1,

+def render_frame(frame_index, rng):
