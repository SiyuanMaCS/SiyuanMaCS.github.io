+def smoothstep(edge0, edge1, value):
+    t = np.clip((value - edge0) / (edge1 - edge0), 0.0, 1.0)
+    return t * t * (3.0 - 2.0 * t)

+def sd_circle(x, y, cx, cy, radius):
+    return np.hypot(x - cx, y - cy) - radius

+def sd_tapered_capsule(x, y, ax, ay, bx, by, ra, rb):
+    pax = x - ax
+    pay = y - ay
+    bax = bx - ax
+    bay = by - ay
+    h = np.clip((pax * bax + pay * bay) / (bax * bax + bay * bay), 0.0, 1.0)
+    return np.hypot(pax - bax * h, pay - bay * h) - (ra + (rb - ra) * h)

+def smooth_min(a, b, k):
+    h = np.clip(0.5 + 0.5 * (b - a) / k, 0.0, 1.0)
+    return b * (1.0 - h) + a * h - k * h * (1.0 - h)
