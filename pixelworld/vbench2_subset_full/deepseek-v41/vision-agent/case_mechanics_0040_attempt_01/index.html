+def smoothstep(edge0, edge1, value):

+def ball_motion(t):
+    """Return ball center, vertical speed, and squash for a sampled instant."""
+    x0 = 52.0

+    impact_speed = vy0 + GRAVITY * IMPACT_TIME

+    tau = t - IMPACT_TIME
+    flight_time = 2.0 * rebound_speed / GRAVITY

+    if tau < flight_time:
+        x = x0 + vx * t
+        y = contact_center - rebound_speed * tau + 0.5 * GRAVITY * tau * tau
+        vy = -rebound_speed + GRAVITY * tau

+    second_speed = rebound_speed * SECOND_RESTITUTION

+def rotation_matrix(angle_x, angle_y, angle_z):
+    cx, sx = math.cos(angle_x), math.sin(angle_x)
+    cy, sy = math.cos(angle_y), math.sin(angle_y)
+    cz, sz = math.cos(angle_z), math.sin(angle_z)
+    rx = np.array([[1.0, 0.0, 0.0], [0.0, cx, -sx], [0.0, sx, cx]], dtype=np.float32)
+    ry = np.array([[cy, 0.0, sy], [0.0, 1.0, 0.0], [-sy, 0.0, cy]], dtype=np.float32)
+    rz = np.array([[cz, -sz, 0.0], [sz, cz, 0.0], [0.0, 0.0, 1.0]], dtype=np.float32)
+    return rz @ ry @ rx

+def render_ball(cx, cy, rx, ry, spin):
+    w = WIDTH * SUPERSAMPLE
+    h = HEIGHT * SUPERSAMPLE
+    pad_x = int((rx + 3.0) * SUPERSAMPLE)
+    pad_y = int((ry + 3.0) * SUPERSAMPLE)

+    x1 = min(w, int(cx * SUPERSAMPLE) + pad_x)

+    y1 = min(h, int(cy * SUPERSAMPLE) + pad_y)

+    light = np.array([-0.48, -0.58, 0.66], dtype=np.float32)
+    light /= np.linalg.norm(light)
+    diffuse = np.clip(nx * light[0] + ny * light[1] + nz * light[2], 0.0, 1.0)
+    half_vec = light + np.array([0.0, 0.0, 1.0], dtype=np.float32)
+    half_vec /= np.linalg.norm(half_vec)
+    specular = np.clip(nx * half_vec[0] + ny * half_vec[1] + nz * half_vec[2], 0.0, 1.0) ** 34.0
+    fresnel = (1.0 - np.clip(nz, 0.0, 1.0)) ** 3.0

+def composite_ball(image, sample):
+    cx, cy, rx, ry, spin, alpha_mul = sample
+    rendered = render_ball(cx, cy, rx, ry, spin)
+    if rendered is None:
+        return

+    src = rgba[..., :3]
+    a = rgba[..., 3:4] * alpha_mul
+    dst = image[y0:y1, x0:x1].astype(np.float32)
+    image[y0:y1, x0:x1] = np.clip(dst * (1.0 - a) + src * a + 0.5, 0.0, 255.0).astype(np.uint8)

+def add_impact_ring(image, t):
+    if not (0.92 <= t <= 1.38):
+        return
+    age = (t - IMPACT_TIME) / 0.38
+    if age < -0.2:

+def build_frame(background, index):
+    t = index / FPS
+    frame = background.copy()
+
+    # A long, soft cast shadow communicates height and the downward throw.
+    x, y, vx, vy, _ = ball_motion(t)
+    height = max(0.0, GROUND_Y - BALL_RADIUS - y)
+    shadow_x = x + 7.0 + height * 0.28
+    shadow_rx = 15.0 + height * 0.18
+    shadow_ry = 3.4 + height * 0.035

+    shadow = shadow_layer(shadow_x, y, shadow_rx, shadow_ry, shadow_alpha, 2.3 + height * 0.025)
+    frame_f = frame.astype(np.float32)
+    frame_f *= (1.0 - shadow[..., None] * 0.70)
+    frame_f += np.array([54.0, 39.0, 31.0], dtype=np.float32) * shadow[..., None] * 0.12
+    frame = np.clip(frame_f, 0, 255).astype(np.uint8)
+
+    # Very short exposure blur; strongest while the ball is moving quickly.
+    speed = math.hypot(vx, vy)
+    blur_gain = np.clip((speed - 20.0) / 100.0, 0.0, 1.0)
+    exposure = 0.055 * (0.45 + 0.55 * blur_gain)
+    sub_times = [t - exposure * 0.45, t, t + exposure * 0.55]
+    weights = [0.18 * blur_gain, 0.72, 0.18 * blur_gain]
+    if not any(weights):
+        weights = [0.0, 1.0, 0.0]
+    weights = np.asarray(weights, dtype=np.float32)
+    weights /= weights.sum()
+
+    spin_base = t * 2.35
+    for sub_t, weight in zip(sub_times, weights):
+        if weight <= 0.001:
+            continue
+        sx, sy, _, _, _ = ball_motion(max(0.0, sub_t))
+        rx, ry = BALL_RADIUS, BALL_RADIUS
+        for impact_t in (IMPACT_TIME, IMPACT_TIME + 2.0 * (56.0 + GRAVITY * IMPACT_TIME) * RESTITUTION / GRAVITY):
+            dt = sub_t - impact_t
+            if abs(dt) < 0.095:
+                squash = 1.0 - 0.30 * math.cos(dt / 0.095 * math.pi * 0.5) ** 2
+                ry = BALL_RADIUS * squash

+                sy = GROUND_Y - ry
+        composite_ball(frame, (sx, sy, rx, ry, spin_base + sub_t * 2.0, weight))
+
+    add_impact_ring(frame, t)
+
+    small = Image.fromarray(frame).resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
+    return np.asarray(small, dtype=np.uint8)
