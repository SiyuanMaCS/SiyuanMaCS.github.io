+def smoothstep(edge0: float, edge1: float, x: np.ndarray | float):
+    t = np.clip((x - edge0) / (edge1 - edge0), 0.0, 1.0)
+    return t * t * (3.0 - 2.0 * t)

+def make_trajectory():
+    """Return impact times and analytic height/velocity samples."""
+    events = []
+    t0, h0, v0 = 0.0, START_HEIGHT, START_VELOCITY
+    for _ in range(4):
+        root = (v0 + math.sqrt(v0 * v0 + 2.0 * GRAVITY * h0)) / GRAVITY
+        impact_t = t0 + root
+        impact_v = v0 - GRAVITY * root
+        events.append((impact_t, impact_v))
+        t0, h0, v0 = impact_t, 0.0, -RESTITUTION * impact_v
+
+    def height_at(t: float) -> tuple[float, float]:
+        if t <= events[0][0]:
+            dt = t
+            return (
+                START_HEIGHT + START_VELOCITY * dt - 0.5 * GRAVITY * dt * dt,
+                START_VELOCITY - GRAVITY * dt,
+            )
+        segment_t = events[0][0]
+        segment_v = -RESTITUTION * events[0][1]
+        for impact_t, impact_v in events[1:]:
+            if t <= impact_t:
+                dt = t - segment_t
+                return segment_v * dt - 0.5 * GRAVITY * dt * dt, segment_v - GRAVITY * dt
+            segment_t = impact_t
+            segment_v = -RESTITUTION * impact_v
+        dt = t - segment_t
+        return segment_v * dt - 0.5 * GRAVITY * dt * dt, segment_v - GRAVITY * dt
+
+    return [event[0] for event in events], height_at

+def render_ball(

+    cy: float,
+    rx: float,
+    ry: float,
+    yaw: float,
+    pitch: float,
+    roll: float,

+    """Rasterize the ellipsoidal rubber ball at high resolution."""

+    x0 = max(0, int(math.floor(cx - rx - margin)))
+    x1 = min(W, int(math.ceil(cx + rx + margin)))

+

+    teal = np.array([19.0, 66.0, 77.0], dtype=np.float32)
+

+        0.035 * mottle[..., None]
+    )
+    color = np.where((mottle[..., None] > 0.78), orange_dark[None, None, :], color)
+
+    paw, paw_highlight = paw_mask(lon, lat)

+    color += paw_highlight[..., None] * np.array([24.0, 21.0, 15.0], dtype=np.float32)
+
+    # Molded equator seam, kept subtle so it reads as part of the rubber surface.
+    seam = np.exp(-((np.abs(lat) - 0.012) / 0.012) ** 2)
+    color *= (1.0 - 0.22 * seam)[..., None]
+
+    key = np.array([-0.48, -0.60, 0.64], dtype=np.float32)
+    key /= np.linalg.norm(key)
+    fill = np.array([0.70, -0.20, 0.50], dtype=np.float32)
+    fill /= np.linalg.norm(fill)
+    half = key + np.array([0.0, 0.0, 1.0], dtype=np.float32)
+    half /= np.linalg.norm(half)

+def squash_at(t: float) -> float:
+    amount = 0.0
+    for impact in IMPACT_TIMES:
+        amount = max(amount, math.exp(-((t - impact) / 0.072) ** 2))
+    return 0.27 * amount

+def ball_state(t: float):
+    height, velocity = HEIGHT_AT(t)
+    height = max(0.0, height)
+    x = 146.0 - 4.15 * t
+    squash = squash_at(t)

+    ry = BALL_RADIUS * (1.0 - 0.27 * squash)
+    cy = FLOOR_CONTACT_Y - height + (BALL_RADIUS - ry)
+    yaw = 0.24 + 1.55 * t
+    pitch = 0.38 + 0.48 * t
+    roll = -0.18 + 0.62 * t

+def apply_shadow(frame: np.ndarray, t: float, state) -> None:
+    x, _, _, _, _, _, _, height, _ = state
+    yy, xx = np.mgrid[0:H, 0:W].astype(np.float32)
+    sx = (BALL_RADIUS * 1.12 + 0.20 * height) * SS

+    shadow_x = (x + 0.12 * height) * SS
+    shadow_y = (FLOOR_CONTACT_Y + BALL_RADIUS + 1.3) * SS
+    dx = (xx - shadow_x) / sx
+    dy = (yy - shadow_y) / sy

+    opacity = 0.34 * math.exp(-height / 58.0) + 0.025
+    broad *= opacity
+
+    contact_sx = BALL_RADIUS * 0.72 * SS
+    contact_sy = BALL_RADIUS * 0.17 * SS
+    cdx = (xx - shadow_x) / contact_sx
+    cdy = (yy - shadow_y + 1.5 * SS) / contact_sy

+    mask = np.clip(broad + contact, 0.0, 0.62)
+    shadow_color = np.array([48.0, 35.0, 25.0], dtype=np.float32)
+    frame *= (1.0 - mask[..., None])
+    frame += shadow_color[None, None, :] * mask[..., None]

+    reflection = np.exp(-((xx - shadow_x) / (BALL_RADIUS * 1.1 * SS)) ** 2)
+    reflection *= np.exp(-((yy - shadow_y - 2.2 * SS) / (BALL_RADIUS * 0.42 * SS)) ** 2)
+    reflection *= 0.045 * math.exp(-height / 14.0)
+    frame += reflection[..., None] * np.array([67.0, 24.0, 10.0], dtype=np.float32)
+

+def render_frame(index: int) -> np.ndarray:
+    t = index / FPS
+    shutter = 0.085
+    sample_count = 9
+    weights = np.exp(-0.5 * np.linspace(-1.35, 1.35, sample_count) ** 2)
+    weights /= weights.sum()
+    accumulation = np.zeros_like(BACKGROUND)
+
+    for weight, fraction in zip(weights, np.linspace(-0.5, 0.5, sample_count)):
+        sample_t = t + shutter * fraction
+        state = ball_state(sample_t)
+        x, cy, rx, ry, yaw, pitch, roll, _, _ = state
+        frame = BACKGROUND.copy()
+        apply_shadow(frame, sample_t, state)
+        x0, y0, color, alpha = render_ball(x * SS, cy * SS, rx * SS, ry * SS, yaw, pitch, roll)
+        region = frame[y0 : y0 + alpha.shape[0], x0 : x0 + alpha.shape[1]]
+        region *= (1.0 - alpha[..., None])
+        region += color * alpha[..., None]
+        accumulation += frame * weight
+
+    # Gentle highlight bloom from the ball and a touch of sensor grain.
+    bright = np.clip(accumulation - 205.0, 0.0, 50.0) / 50.0
+    bloom = cv2.GaussianBlur(bright, (0, 0), 7.0 * SS)
+    accumulation += bloom * 7.0
+    accumulation = np.clip(accumulation, 0.0, 255.0)
+
+    rng = np.random.default_rng(1000 + index)
+    grain = rng.normal(0.0, 0.60, (SPEC_H, SPEC_W)).astype(np.float32)
+    final = cv2.resize(accumulation, (SPEC_W, SPEC_H), interpolation=cv2.INTER_AREA)
+    final += grain[..., None]
+    return np.clip(final, 0.0, 255.0).astype(np.uint8)
+
+
+def save_video(frames: np.ndarray) -> None:
+    path = ROOT / "output.mp4"

+        str(path), cv2.VideoWriter_fourcc(*"mp4v"), FPS, (SPEC_W, SPEC_H)

+
+
+def main() -> None:
+    frames = np.stack([render_frame(i) for i in range(FRAMES)])

+    save_video(frames)
+    print(f"Rendered {len(frames)} frames at {SPEC_W}x{SPEC_H}, {FPS:g} fps")
+

+    main()
