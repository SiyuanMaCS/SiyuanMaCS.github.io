+def smoothstep(edge0, edge1, x):

+def normalized(noise, strength=1.0):

+def draw_stroke(base, frame, tip_x, tip_y, rng):

+def draw_chalk(base, frame, tip_x, tip_y, angle, rng):
+    y, x = np.mgrid[0:RH, 0:RW].astype(np.float32)
+    tx, ty = tip_x * S, tip_y * S
+    ux, uy = np.cos(angle), np.sin(angle)
+    px, py = -uy, ux
+    dx, dy = x - tx, y - ty
+    v = dx * ux + dy * uy
+    s = dx * px + dy * py
+    length = 290.0
+    radius = 22.5
+    width = 6.3 + (radius - 6.3) * smoothstep(-1.0, 35.0, v)
+    edge_noise = normalized(gaussian_filter(rng.normal(size=(RH, RW)).astype(np.float32), 1.1), 0.8)
+    side = smoothstep(width + 1.5, width - 1.4, np.abs(s) + edge_noise * 1.15)
+    body = side * (v >= -1.2) * (v <= length)
+    cap = smoothstep(radius + 1.4, radius - 1.4, np.hypot(v - length, s * 0.88))
+    mask = np.clip(body + cap, 0, 1)
+
+    # A cast shadow and a narrow contact shadow anchor the stick to the wall.
+    shadow = gaussian_filter(np.roll(np.roll(mask, int(5.2 * S), axis=0), int(3.2 * S), axis=1), 3.8 * S)
+    base *= 1.0 - 0.28 * shadow[..., None]
+    contact = np.exp(-(((x - tx - 1.5 * S) / (9.0 * S)) ** 2 + ((y - ty - 2.5 * S) / (5.0 * S)) ** 2))
+    base *= 1.0 - 0.23 * contact[..., None]
+
+    across = np.clip(s / np.maximum(width, 1.0), -1.0, 1.0)
+    broad_light = 0.77 + 0.24 * np.exp(-((across + 0.48) / 0.47) ** 2)
+    round_light = 0.07 * np.sqrt(np.maximum(0.0, 1.0 - across * across))
+    surface = normalized(gaussian_filter(rng.normal(size=(RH, RW)).astype(np.float32), 1.0), 1.0)
+    surface += 0.55 * normalized(gaussian_filter(rng.normal(size=(RH, RW)).astype(np.float32), 3.3), 1.0)
+    light = np.clip(broad_light + round_light + 0.022 * surface, 0.56, 1.04)
+    light *= 1.0 - 0.08 * smoothstep(0.70, 1.0, np.abs(across))
+
+    chalk_rgb = np.empty((RH, RW, 3), dtype=np.float32)
+    chalk_rgb[..., 0] = 233.0 * light + 8.0 * np.clip(surface, -1.2, 1.2)
+    chalk_rgb[..., 1] = 232.0 * light + 5.0 * np.clip(surface, -1.2, 1.2)
+    chalk_rgb[..., 2] = 222.0 * light + 2.0 * np.clip(surface, -1.2, 1.2)
+    cool_shadow = smoothstep(0.2, 1.0, across)
+    chalk_rgb[..., 0] -= 8.0 * cool_shadow
+    chalk_rgb[..., 1] -= 3.0 * cool_shadow
+
+    # The worn tip is brighter from abrasion.
+    tip_glow = np.exp(-((v - 3.0 * S) / (11.0 * S)) ** 2 - (s / (8.0 * S)) ** 2)
+    chalk_rgb += tip_glow[..., None] * np.array([10.0, 10.0, 7.0])
+
+    # Small chips and pits break up the manufactured edge.
+    pits = (surface < -1.65) * (mask > 0.55)
+    chalk_rgb *= 1.0 - 0.12 * gaussian_filter(pits.astype(np.float32), 0.8)[..., None]
+    base[:] = base * (1.0 - mask[..., None]) + np.clip(chalk_rgb, 0, 255) * mask[..., None]
+    return base, mask

+def draw_dust(base, frame, tip_x, tip_y, particles):
+    y, x = np.mgrid[0:RH, 0:RW].astype(np.float32)
+    tx, ty = tip_x * S, tip_y * S
+    haze = np.exp(-(((x - tx + 7.0 * S) / (19.0 * S)) ** 2 + ((y - ty + 2.0 * S) / (13.0 * S)) ** 2))
+    haze *= 0.10
+    base[:] = base * (1.0 - haze[..., None]) + np.array([228.0, 226.0, 216.0]) * haze[..., None]
+
+    layer = Image.new("RGBA", (RW, RH), (0, 0, 0, 0))
+    draw = ImageDraw.Draw(layer)
+    for birth, life, bx, by, vx, vy, size, opacity in particles:
+        age = frame - birth
+        if age < 0 or age >= life:
+            continue
+        seconds = age / FPS
+        px = (bx + vx * seconds) * S
+        py = (by + vy * seconds + 0.5 * 58.0 * seconds * seconds) * S
+        fade = (1.0 - age / life) ** 0.65
+        radius = max(0.8, size * S * (1.0 - 0.08 * age))
+        a = int(255 * opacity * fade)
+        color = (236, 234, 222, a) if (birth + int(size * 1000)) % 5 else (181, 181, 172, int(a * 0.72))
+        draw.ellipse((px - radius, py - radius, px + radius, py + radius), fill=color)
+    layer = layer.filter(ImageFilter.GaussianBlur(0.35 * S))
+    arr = np.asarray(layer, dtype=np.float32) / 255.0
+    a = arr[..., 3:4]
+    base[:] = base * (1.0 - a) + arr[..., :3] * 255.0 * a
+    return base
+

+def postprocess(frame, rng):
+    arr = frame.copy()
+    # Gentle highlight bloom from the freshly exposed chalk.
+    bright = np.maximum(arr - 195.0, 0.0)
+    bloom = gaussian_filter(bright, 3.4 * S)
+    arr += bloom * 0.16
+
+    y, x = np.mgrid[0:RH, 0:RW].astype(np.float32)
+    vignette = 1.0 - 0.13 * (((x / RW) - 0.49) ** 2 + ((y / RH) - 0.50) ** 2)
+    arr *= vignette[..., None]
+    grain = rng.normal(0.0, 1.65, (RH, RW, 1)).astype(np.float32)
+    arr += grain
+    arr = np.clip(arr, 0, 255)
+
+    # Slight highlight rolloff, then downsample with a good antialiasing filter.
+    arr = 255.0 * (arr / 255.0) ** 0.97
+    image = Image.fromarray(np.clip(arr, 0, 255).astype(np.uint8), "RGB")
+    return np.asarray(image.resize((W, H), Image.Resampling.LANCZOS), dtype=np.uint8)
+

+def main():
+    concrete_rng = np.random.default_rng(FRAME_SEED)
+    concrete = make_concrete(concrete_rng)
+    particles = make_particles(np.random.default_rng(FRAME_SEED + 91))
+    frames = []
+
+    for index in range(N):
+        rng = np.random.default_rng(FRAME_SEED + 1000 + index * 37)
+        tip_x, tip_y, angle = chalk_trajectory(index)
+        frame = concrete.copy()

+
+    frames = np.asarray(frames, dtype=np.uint8)
+    artifact_io.save(frames, ROOT)
+    print(f"saved {frames.shape} frames to {ROOT}")
+
+
+if __name__ == "__main__":
+    main()
