+def smooth_noise(shape, sigma, seed):
+    local = np.random.default_rng(seed)
+    field = gaussian_filter(local.standard_normal(shape), sigma=sigma, mode="reflect")
+    return field / (field.std() + 1e-6)

+def bezier_points(p0, p1, p2, p3, count=240):

+    return ((1 - t) ** 3) * p0 + 3 * ((1 - t) ** 2) * t * p1 + 3 * (1 - t) * t**2 * p2 + t**3 * p3
