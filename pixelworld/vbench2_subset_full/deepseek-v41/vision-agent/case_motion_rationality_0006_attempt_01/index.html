+def cubic(p0, p1, p2, p3, steps=28):
+    result = []
+    for i in range(steps + 1):
+        t = i / steps
+        u = 1.0 - t
+        result.append(
+            (
+                u**3 * p0[0] + 3 * u * u * t * p1[0] + 3 * u * t * t * p2[0] + t**3 * p3[0],
+                u**3 * p0[1] + 3 * u * u * t * p1[1] + 3 * u * t * t * p2[1] + t**3 * p3[1],
+            )
+        )
+    return result

+def smoothstep(t: float) -> float:
+    t = max(0.0, min(1.0, t))
+    return t * t * (3.0 - 2.0 * t)

+def sample_keys(keys, frame: int):
+    if frame <= keys[0][0]:
+        return keys[0][1:]
+    if frame >= keys[-1][0]:
+        return keys[-1][1:]
+    for (f0, *v0), (f1, *v1) in zip(keys[:-1], keys[1:]):
+        if f0 <= frame <= f1:
+            t = smoothstep((frame - f0) / max(1, f1 - f0))
+            return tuple(a + (b - a) * t for a, b in zip(v0, v1))
+    return keys[-1][1:]
