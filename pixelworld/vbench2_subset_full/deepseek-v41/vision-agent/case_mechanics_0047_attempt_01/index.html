+def smoothstep(x):
+    x = np.clip(x, 0.0, 1.0)
+    return x * x * (3.0 - 2.0 * x)

+def noise_field(h, w, sigma, seed):
+    rng = np.random.default_rng(seed)
+    field = gaussian_filter(rng.standard_normal((h, w)).astype(np.float32), sigma)
+    field -= field.min()
+    field /= max(float(field.max()), 1e-6)
+    return field

+def cardboard_texture():
+    """Build a close-up kraft surface with broad pulp variation and fine fibers."""
+    yy, xx = np.mgrid[0:SH, 0:SW].astype(np.float32)
+    x = xx / SW
+    y = yy / SH
+
+    broad = noise_field(SH, SW, 38, 11)
+    medium = noise_field(SH, SW, 8, 12)
+    pulp = noise_field(SH, SW, 1.8, 13)
+    pore = noise_field(SH, SW, 0.7, 14)
+
+    # Flattened random fields make tiny paper fibers run mostly across the sheet.
+    fiber_seed = np.random.default_rng(21).standard_normal((SH // 3, SW)).astype(np.float32)
+    fiber = gaussian_filter(fiber_seed, (0.65, 2.2))
+    fiber = np.asarray(Image.fromarray(((fiber - fiber.min()) / np.ptp(fiber) * 255).astype(np.uint8))
+                       .resize((SW, SH), Image.Resampling.BICUBIC), dtype=np.float32) / 255.0
+    fiber = (fiber - 0.5) * 2.0
+
+    # Subtle corrugation remains visible through the paper skin.
+    phase_warp = gaussian_filter(np.random.default_rng(31).standard_normal((SH, SW)).astype(np.float32), 9)
+    phase_warp /= max(float(np.max(np.abs(phase_warp))), 1e-6)

+    flutes += 0.28 * np.sin(yy * 0.91 - x * 2.0)

+def pen_basis(tip, angle):
+    axis = np.array([np.cos(angle), -np.sin(angle)], dtype=np.float32)
+    side = np.array([-axis[1], axis[0]], dtype=np.float32)
+    return axis, side

+def pen_fields(xx, yy, tip, angle):
+    axis, side = pen_basis(tip, angle)
+    dx, dy = xx - tip[0], yy - tip[1]
+    along = dx * axis[0] + dy * axis[1]
+    across = dx * side[0] + dy * side[1]
+    return along, across

+def rounded_interval_mask(value, lo, hi, radius):
+    q = np.maximum(np.maximum(lo - value, value - hi), 0.0)

+def soft_pen_mask(xx, yy, tip, angle, length=188.0):

+def composite_pen(image, tip, angle, alpha=1.0):
+    yy, xx = np.mgrid[0:SH, 0:SW].astype(np.float32)
+    mask, silver, along, across = pen_shading(xx, yy, tip, angle)

+def draw_mark(image, tip_path, progress):
+    """Rasterize a worn graphite/metal score along the pen's contact path."""
+    yy, xx = np.mgrid[0:SH, 0:SW].astype(np.float32)
+    if progress <= 0.002:
+        return image
+
+    sample_count = min(len(tip_path), max(2, int(progress * (len(tip_path) - 1)) + 2))
+    min_dist = np.full((SH, SW), 1e9, dtype=np.float32)
+    nearest_t = np.zeros((SH, SW), dtype=np.float32)
+    path_scale = float(SS)
+    for i in range(sample_count - 1):
+        a = tip_path[i] * path_scale
+        b = tip_path[i + 1] * path_scale
+        ab = b - a
+        denom = max(float(ab @ ab), 1e-5)
+        t = np.clip(((xx - a[0]) * ab[0] + (yy - a[1]) * ab[1]) / denom, 0, 1)
+        px, py = a[0] + t * ab[0], a[1] + t * ab[1]
+        dist = np.sqrt((xx - px) ** 2 + (yy - py) ** 2)
+        take = dist < min_dist
+        min_dist[take] = dist[take]
+        nearest_t[take] = (i + t[take]) / max(1, len(tip_path) - 1)
+
+    # Friction changes width and pressure; broken flecks keep it from looking painted.
+    grain = noise_field(SH, SW, 0.65, 91)
+    wide = noise_field(SH, SW, 3.2, 92)
+    center = np.sin(nearest_t * 108.0 + 0.7) * 0.10 + (wide - 0.5) * 0.36
+    half_width = (0.82 + 0.22 * wide + 0.10 * np.sin(nearest_t * 77.0)) * SS
+    core = np.clip((half_width - np.abs(min_dist - center * SS)) / (0.60 * SS), 0, 1)
+    scratch = np.clip((1.55 * SS - min_dist) / (0.42 * SS), 0, 1) * (0.22 + 0.78 * grain)
+
+    # The score darkens and burnishes the top fibers.
+    image *= 1.0 - 0.38 * core[..., None]
+    image *= 1.0 - 0.12 * scratch[..., None]
+    dark = np.array([0.23, 0.205, 0.18], dtype=np.float32)
+    image = image * (1.0 - 0.60 * core[..., None]) + dark * (0.60 * core[..., None])
+
+    fleck = core * np.clip((grain - 0.62) * 4.5, 0, 1)
+    image += fleck[..., None] * np.array([0.18, 0.17, 0.15], dtype=np.float32)
+    bright_edge = np.clip((np.abs(min_dist - center * SS) - 0.40 * SS) / (0.32 * SS), 0, 1)
+    bright_edge *= np.clip((1.15 * SS - np.abs(min_dist - center * SS)) / (0.25 * SS), 0, 1)
+    image += (bright_edge * 0.06)[..., None] * np.array([0.80, 0.76, 0.68], dtype=np.float32)
+    return np.clip(image, 0, 1)
+
+
+def finish_image(image):
+    # Gentle highlight rolloff, warm reflected bounce, and restrained optical grain.
+    image = np.clip(image, 0.0, 1.0)
+    image = image + 0.035 * image * (1.0 - image)
+    image[..., 0] *= 1.025
+    image[..., 1] *= 1.005
+    image[..., 2] *= 0.975
+
+    yy, xx = np.mgrid[0:SH, 0:SW].astype(np.float32)
+    r = np.sqrt(((xx - SW * 0.53) / SW) ** 2 + ((yy - SH * 0.48) / SH) ** 2)
+    image *= (1.0 - 0.10 * np.clip((r - 0.27) / 0.48, 0, 1) ** 1.7)[..., None]
+
+    grain = np.random.default_rng(108).normal(0, 0.0052, (SH, SW, 1)).astype(np.float32)
+    image = np.clip(image + grain, 0, 1)
+    small = Image.fromarray((image * 255.0 + 0.5).astype(np.uint8))
+    return np.asarray(small.resize((W, H), Image.Resampling.LANCZOS), dtype=np.uint8)
+

+def main():
+    case_dir = Path(__file__).resolve().parent
+    base = cardboard_texture()
+
+    # The hand writes toward the lower left; the pen axis stays diagonal in frame.
+    p0 = np.array([157.0, 83.5], dtype=np.float32)
+    p1 = np.array([103.0, 103.5], dtype=np.float32)
+    frames = []
+    for i in range(N):
+        u = i / (N - 1)
+        progress = smoothstep(u)
+        tip = p0 + (p1 - p0) * progress
+        tip[1] += 0.55 * np.sin(u * np.pi * 3.0) * np.sin(u * np.pi)
+        angle = np.deg2rad(31.2 - 2.1 * progress)
+
+        path_count = 96
+        path = []
+        for j in range(path_count):
+            v = j / (path_count - 1)
+            p = p0 + (p1 - p0) * v
+            p[1] += 0.55 * np.sin(v * np.pi * 3.0) * np.sin(v * np.pi)
+            path.append(p)
+
+        image = draw_mark(base.copy(), path, progress)
+        image = composite_pen(image, tip * SS, angle)
+        frames.append(finish_image(image))
+
+    frames = np.stack(frames, axis=0)
+    artifact_io.save(frames, case_dir)
+
+
+if __name__ == "__main__":
+    main()
