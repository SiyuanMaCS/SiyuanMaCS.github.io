+def smoothstep(edge0, edge1, x):
+    t = np.clip((x - edge0) / (edge1 - edge0), 0.0, 1.0)

+def blur(array, radius):

+    image = Image.fromarray(np.clip(array * 255.0, 0, 255).astype(np.uint8))
+    image = image.filter(ImageFilter.GaussianBlur(radius))
+    return np.asarray(image, dtype=np.float32) / 255.0

+def sample_bilinear(image, sx, sy):
+    sx = np.clip(sx, 0.0, SS_W - 1.0)
+    sy = np.clip(sy, 0.0, SS_H - 1.0)
+    x0 = np.floor(sx).astype(np.int32)
+    y0 = np.floor(sy).astype(np.int32)
+    x1 = np.minimum(x0 + 1, SS_W - 1)
+    y1 = np.minimum(y0 + 1, SS_H - 1)
+    fx = (sx - x0)[..., None]
+    fy = (sy - y0)[..., None]
+    a = image[y0, x0] * (1.0 - fx) + image[y0, x1] * fx
+    b = image[y1, x0] * (1.0 - fx) + image[y1, x1] * fx
+    return a * (1.0 - fy) + b * fy

+def soft_ellipse(u, v, cx, cy, rx, ry, angle=0.0):

+    dy = v - cy
+    xr = ca * dx + sa * dy
+    yr = -sa * dx + ca * dy

+def finish_frame(image, rng):
+    """Downsample, grade, and add the small optical cues of a real macro shot."""
+    small = cv2.resize(image, (WIDTH, HEIGHT), interpolation=cv2.INTER_AREA)
+
+    # Gentle chromatic separation toward the corners.
+    yy, xx = np.mgrid[0:HEIGHT, 0:WIDTH].astype(np.float32)
+    dx = (xx - WIDTH * 0.5) / WIDTH
+    dy = (yy - HEIGHT * 0.5) / HEIGHT
+    r = np.clip((xx + 0.42 * dx).astype(np.int32), 0, WIDTH - 1)
+    b = np.clip((xx - 0.42 * dx).astype(np.int32), 0, WIDTH - 1)
+    ca = small.copy()
+    ca[..., 0] = small[yy.astype(np.int32), r, 0]
+    ca[..., 2] = small[yy.astype(np.int32), b, 2]
+    small = 0.92 * small + 0.08 * ca
+
+    # Soft highlight bloom.
+    bright = np.clip((small - 0.74) / 0.26, 0.0, 1.0)
+    bloom = np.asarray(
+        Image.fromarray((bright * 255).astype(np.uint8)).filter(ImageFilter.GaussianBlur(2.2)),
+        dtype=np.float32,
+    ) / 255.0
+    small = np.clip(small + 0.16 * bloom, 0.0, 1.0)
+
+    # Warm highlights, cool shadows, mild vignette, and fine film grain.
+    luma = small @ np.array([0.2126, 0.7152, 0.0722], np.float32)
+    small = small + (0.018 * (luma - 0.5))[..., None] * np.array([1.0, 0.25, -0.50], np.float32)
+    radius2 = dx * dx + dy * dy
+    vignette = np.clip(1.0 - 0.17 * radius2, 0.78, 1.0)
+    small *= vignette[..., None]
+    grain = rng.normal(0.0, 0.012, (HEIGHT, WIDTH, 1)).astype(np.float32)
+    small = np.clip(small + grain, 0.0, 1.0)
+
+    # Filmic shoulder and a clean 8-bit finish.
+    small = np.where(small < 0.80, small, 0.80 + 0.20 * (1.0 - np.exp(-(small - 0.80) / 0.20)))
+    return np.clip(small * 255.0 + 0.5, 0, 255).astype(np.uint8)
+

+def make_frames():
+    rng = np.random.default_rng(SEED)
+    background = make_background(rng)
+    cup, lip_y = render_cup(background)
+    frames = []
+    for index in range(FRAMES):
+        t = index / (FRAMES - 1)
+
+        # The bead condenses quickly, then settles into an almost still sphere.
+        grow = smoothstep(0.0, 0.42, t)
+        grow = 1.0 - (1.0 - grow) ** 2.2
+        radius = 15.6 + 7.2 * grow
+        breathing = 0.13 * np.sin(2.0 * np.pi * t * 1.15)
+        radius += breathing
+        cx = 133.0 + 0.16 * np.sin(2.0 * np.pi * t * 0.72)
+        cy = lip_y[0, int(cx * SUPERSAMPLE)] - radius * 0.73 + 0.15 * np.sin(2.0 * np.pi * t + 0.3)
+        image = render_droplet(cup, rng, radius, cx, cy)
+        frames.append(finish_frame(image, rng))
+    return np.stack(frames)
+
+
+def main():
+    case_dir = Path(__file__).resolve().parent
+    frames = make_frames()
+    artifact_io.save(frames, case_dir)
+    print(f"Wrote {len(frames)} frames to {case_dir}")
+
+
+if __name__ == "__main__":
+    main()
