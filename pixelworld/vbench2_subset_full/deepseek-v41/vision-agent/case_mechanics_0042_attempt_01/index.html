+def render_frame(frame_index):
+    time_s = frame_index / FPS
+    canvas = add_floor_shadow(BACKGROUND.copy(), time_s)
+    canvas = add_contact_flash(canvas, time_s)

+    sample_offsets = np.array([-1.0, -0.5, 0.0, 0.5, 1.0], dtype=np.float32) * shutter * 0.5
+    sample_weights = np.array([0.08, 0.18, 0.48, 0.18, 0.08], dtype=np.float32)

+        sample_t = time_s + float(offset)
+        center_x, center_y = trajectory(sample_t)
+        deformation = contact_deformation(sample_t)

+            # center exposure remains opaque while fast edges blur naturally.
+            layer = render_ball(BACKGROUND.copy(), center_x, center_y, rotation, deformation)

+            canvas = canvas * (1.0 - alpha[..., None] * weight) + layer * alpha[..., None] * weight
+        else:
+            canvas = blurred

+        np.clip(bloom_source * 255.0, 0, 255).astype(np.uint8), "RGB"
+    ).filter(ImageFilter.GaussianBlur(15.0))
+    bloom = np.asarray(bloom_image, dtype=np.float32) / 255.0
+    canvas = canvas + 0.12 * bloom
+

+    final = final.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)

+    luminance = frame[..., 0] * 0.2126 + frame[..., 1] * 0.7152 + frame[..., 2] * 0.0722
+    frame = np.clip(luminance[..., None] + (frame - luminance[..., None]) * 1.07, 0.0, 1.0)
+

+    frame = np.clip(frame + grain * 0.0065, 0.0, 1.0)
+    return np.round(frame * 255.0).astype(np.uint8)
