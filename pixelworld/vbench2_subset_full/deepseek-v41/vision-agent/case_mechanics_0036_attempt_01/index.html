+def smoothstep(a, b, x):
+    t = np.clip((x - a) / (b - a), 0.0, 1.0)

+def ball_x(t):

+def simulate_ball():
+    """Return sampled height, vertical velocity, and exact impact times."""
+    dt = 1.0 / 2400.0
+    times = np.arange(-0.05, 5.05 + dt, dt)
+    y = 5.0
+    vy = 55.0
+    g = 310.0
+    values = []
+    impacts = []
+    contact_count = 0

+    for t in times:
+        vy += g * dt
+        y += vy * dt

+            restitution = 0.66 if contact_count == 1 else 0.64
+            vy = -vy * restitution
+            if abs(vy) < 13.0:
+                vy = 0.0
+        values.append((float(t), float(y), float(vy)))

+def deformation(t, y, vy):
+    """Squash around contact and stretch slightly along the velocity."""
+    distance = min((abs(t - impact[0]) for impact in IMPACTS), default=1.0)
+    impact_radius = 0.145

+        sy = 1.0 + 0.075 * speed_stretch
+        sx = 1.0 - 0.035 * speed_stretch

+def render_ball(cx, cy, sx, sy, rotation, opacity=1.0):
+    rx = BALL_R * S * sx
+    ry = BALL_R * S * sy
+    pad = int(math.ceil(max(rx, ry) * 1.10))
+    x0 = int(math.floor(cx * S - pad))
+    y0 = int(math.floor(cy * S - pad))

+    y1 = int(math.ceil(cy * S + pad))
+    pw, ph = x1 - x0, y1 - y0

+    qx = (xx + x0 - cx * S) / rx
+    qy = (yy + y0 - cy * S) / ry
+    radial = qx * qx + qy * qy
+    inside = radial <= 1.0
+    qz = np.sqrt(np.clip(1.0 - radial, 0.0, 1.0))
+
+    # Directional key light with cool sky fill.
+    lx, ly, lz = -0.48, -0.64, 0.60
+    norm = math.sqrt(lx * lx + ly * ly + lz * lz)
+    lx, ly, lz = lx / norm, ly / norm, lz / norm
+    ndotl = np.clip(qx * lx + qy * ly + qz * lz, 0.0, 1.0)
+    half_x, half_y, half_z = lx, ly, lz + 1.0
+    hnorm = math.sqrt(half_x * half_x + half_y * half_y + half_z * half_z)
+    ndoth = np.clip((qx * half_x + qy * half_y + qz * half_z) / hnorm, 0.0, 1.0)
+    spec = ndoth ** 42
+
+    # Rotating pale equator band makes the rubber ball read as a solid object.
+    ca, sa = math.cos(rotation), math.sin(rotation)
+    tu = ca * qx + sa * qy
+    tv = -sa * qx + ca * qy
+    band_center = 0.12 + 0.10 * math.sin(rotation * 0.55)
+    band = np.exp(-((tv - band_center) / 0.105) ** 2)
+    band_edge = np.exp(-((tv - band_center - 0.19) / 0.035) ** 2)
+    band *= smoothstep(0.06, 0.45, qz)
+    band_edge *= smoothstep(0.06, 0.45, qz)
+
+    red = np.array([213, 59, 68], np.float32)
+    dark_red = np.array([80, 20, 28], np.float32)
+    cream = np.array([244, 219, 154], np.float32)
+    cool_fill = np.array([44, 60, 66], np.float32)
+
+    base = red[None, None, :] * (0.31 + 0.83 * ndotl[..., None])
+    base += cool_fill[None, None, :] * (0.20 * (1.0 - ndotl[..., None]))
+    base *= (1.0 - 0.27 * (1.0 - qz[..., None]))
+    base = base * (1.0 - band[..., None]) + cream[None, None, :] * band[..., None]
+    base = base * (1.0 - 0.50 * band_edge[..., None]) + dark_red[None, None, :] * band_edge[..., None]
+    base += spec[..., None] * np.array([255, 245, 218], np.float32) * 0.80
+    base += smoothstep(0.55, 0.98, ndotl) [..., None] * np.array([24, 18, 8], np.float32)
+
+    # Contact occlusion at the lower rim.
+    lower_rim = np.clip((qy - 0.25) / 0.75, 0.0, 1.0) * (1.0 - qz)
+    base *= (1.0 - 0.24 * lower_rim[..., None])

+    coverage[~inside] = 0.0

+                        ex + ring_w * S, ey + ring_h * S),

+    return Image.alpha_composite(frame.convert("RGBA"), effects).convert("RGB")

+def render_frame(index):
+    t = index / FPS
+    cx, cy, vy = state_at(t)
+    sx, sy = deformation(t, cy, vy)
+    if sy < 0.99:
+        cy = min(cy, FLOOR_Y - BALL_R * sy)
+    rotation = -0.30 * (cx - 23.0) / BALL_R + 0.12 * t
+
+    frame = BACKGROUND.copy()
+    frame = draw_shadow(frame, cx, cy)
+    frame = draw_contact_effects(frame, t)
+
+    # Two short-exposure ghosts retain speed at 4 fps without obscuring the ball.
+    speed = math.hypot((state_at(t - 0.035)[0] - cx), (state_at(t - 0.035)[1] - cy))
+    if speed > 4.0:
+        for lag, opacity in ((0.070, 0.13), (0.035, 0.23)):
+            gx, gy, gvy = state_at(max(0.0, t - lag))
+            gsx, gsy = deformation(max(0.0, t - lag), gy, gvy)
+            sprite, pos = render_ball(gx, gy, gsx, gsy, rotation - lag * 2.2, opacity)
+            frame.paste(sprite, pos, sprite)
+
+    sprite, pos = render_ball(cx, cy, sx, sy, rotation, 1.0)
+    frame.paste(sprite, pos, sprite)
+
+    # Gentle bloom and vignette at display scale.
+    small = frame.resize((W, H), Image.Resampling.LANCZOS)
+    arr = np.asarray(small, dtype=np.float32)
+    yy, xx = np.mgrid[0:H, 0:W].astype(np.float32)
+    vx = (xx - (W - 1) * 0.5) / (W * 0.5)
+    vyv = (yy - (H - 1) * 0.5) / (H * 0.5)
+    vignette = 1.0 - 0.13 * np.clip((vx * vx * 0.72 + vyv * vyv) - 0.30, 0.0, 1.0)
+    arr *= vignette[..., None]
+
+    # Fine, deterministic sensor grain keeps flat regions from banding.
+    rng = np.random.default_rng(900 + index)
+    grain = rng.normal(0.0, 1.1, (H, W, 1)).astype(np.float32)
+    arr += grain
+    return np.clip(arr, 0, 255).astype(np.uint8)
+

+    artifact_io.save(frames, case_dir)
+    print(f"wrote {N} frames to {case_dir}")
+

+    main()
