+def normalize(v):
+    return v / np.linalg.norm(v)

+def smoothstep(edge0, edge1, x):
+    t = np.clip((x - edge0) / (edge1 - edge0), 0.0, 1.0)
+    return t * t * (3.0 - 2.0 * t)

+def make_noise(rng, size, sigma, octaves=1):
+    field = np.zeros((size, size), dtype=np.float32)
+    weight_sum = 0.0
+    for octave in range(octaves):
+        octave_sigma = sigma / (2.0**octave)
+        weight = 0.5**octave
+        raw = rng.normal(size=(size, size)).astype(np.float32)
+        layer = gaussian_filter(raw, octave_sigma, mode="reflect")
+        layer /= np.std(layer) + 1e-6
+        field += weight * layer
+        weight_sum += weight
+    field /= weight_sum
+    field -= field.mean()
+    field /= np.std(field) + 1e-6
+    return field

+def make_marble(size=1200):
+    """Build a cool-grey marble texture over a 5 x 5 world-unit slab."""
+    rng = np.random.default_rng(SEED)
+    axis = np.linspace(-2.5, 2.5, size, dtype=np.float32)
+    x, y = np.meshgrid(axis, axis)
+
+    broad = make_noise(rng, size, 115, octaves=3)
+    medium = make_noise(rng, size, 42, octaves=4)
+    detail = make_noise(rng, size, 10, octaves=3)
+
+    warp_x = x + 0.31 * medium + 0.10 * detail
+    warp_y = y + 0.24 * broad + 0.07 * detail
+

+        + 0.48 * medium

+    primary = np.abs(np.sin(primary_phase))
+    vein_main = np.exp(-np.square(primary / 0.105))
+    vein_halo = np.exp(-np.square(primary / 0.24))

+    secondary_phase = (

+        + 2.5 * broad
+        - 0.65 * medium
+        + 0.6

+    secondary = np.abs(np.sin(secondary_phase))
+    vein_secondary = np.exp(-np.square(secondary / 0.045))

+
+    base_tone = 0.805 + 0.045 * cloud + fine
+    warm = 0.018 * broad + 0.009 * medium
+    albedo = np.empty((size, size, 3), dtype=np.float32)
+    albedo[..., 0] = base_tone + warm
+    albedo[..., 1] = base_tone + 0.005 * warm
+    albedo[..., 2] = base_tone + 0.002 * broad
+
+    dark_vein = np.array([0.285, 0.305, 0.325], dtype=np.float32)
+    main_alpha = np.clip(0.58 * vein_main + 0.13 * vein_halo, 0.0, 0.78)
+    albedo = albedo * (1.0 - main_alpha[..., None]) + dark_vein * main_alpha[..., None]
+
+    warm_vein = np.array([0.48, 0.455, 0.425], dtype=np.float32)
+    secondary_alpha = 0.24 * vein_secondary
+    albedo = (

+

+    hairline_alpha = 0.20 * vein_hairline

+        albedo * (1.0 - hairline_alpha[..., None])
+        + pale_line * hairline_alpha[..., None]

+    albedo += sheen[..., None] * np.array([1.0, 0.98, 0.94], dtype=np.float32)
+    return np.clip(albedo, 0.0, 1.0)

+def bilinear_sample(image, x, y):
+    h, w = image.shape[:2]
+    x = np.clip(x, 0.0, w - 1.001)
+    y = np.clip(y, 0.0, h - 1.001)
+    x0 = np.floor(x).astype(np.int32)
+    y0 = np.floor(y).astype(np.int32)
+    x1 = np.minimum(x0 + 1, w - 1)
+    y1 = np.minimum(y0 + 1, h - 1)
+    wx = (x - x0)[..., None]
+    wy = (y - y0)[..., None]
+    top = image[y0, x0] * (1.0 - wx) + image[y0, x1] * wx
+    bottom = image[y1, x0] * (1.0 - wx) + image[y1, x1] * wx
+    return top * (1.0 - wy) + bottom * wy

+def make_camera():
+    origin = np.array([0.0, -3.30, 3.02], dtype=np.float32)
+    target = np.array([0.0, 0.18, 0.38], dtype=np.float32)
+    forward = normalize(target - origin)
+    right = normalize(np.cross(forward, np.array([0.0, 0.0, 1.0], dtype=np.float32)))
+    up = normalize(np.cross(right, forward))
+    vfov = np.deg2rad(34.0)
+    tan_v = np.tan(vfov * 0.5)
+    tan_h = tan_v * (W / H)
+    return origin, forward, right, up, tan_v, tan_h

+def project_to_pixels(points, camera):
+    origin, forward, right, up, tan_v, tan_h = camera
+    rel = points - origin
+    depth = np.sum(rel * forward, axis=-1)
+    depth = np.maximum(depth, 1e-5)
+    sx = np.sum(rel * right, axis=-1) / depth
+    sy = np.sum(rel * up, axis=-1) / depth
+    px = (0.5 + 0.5 * sx / tan_h) * (W - 1)
+    py = (0.5 - 0.5 * sy / tan_v) * (H - 1)
+    return px, py

+def environment_color(direction):
+    """A restrained studio environment with two broad reflections."""
+    z = direction[..., 2]
+    sky = 0.58 + 0.27 * np.clip(z, 0.0, 1.0)
+    color = sky[..., None] * np.array([0.90, 0.955, 1.0], dtype=np.float32)
+
+    key_dir = normalize(np.array([-0.35, -0.48, 0.80], dtype=np.float32))
+    fill_dir = normalize(np.array([0.65, -0.18, 0.74], dtype=np.float32))
+    key = np.clip(np.sum(direction * key_dir, axis=-1), 0.0, 1.0)
+    fill = np.clip(np.sum(direction * fill_dir, axis=-1), 0.0, 1.0)
+    color += (1.9 * key**95 + 0.18 * key**12)[..., None] * np.array(
+        [1.0, 0.985, 0.94], dtype=np.float32
+    )
+    color += (0.72 * fill**130 + 0.07 * fill**18)[..., None] * np.array(
+        [0.92, 0.97, 1.0], dtype=np.float32
+    )
+    return color

+def caustic_and_shadow(points, center, radius, light_dir):
+    xy = points[..., :2]
+    rel_xy = xy - center[:2]
+
+    light_h = light_dir[:2]
+    light_h = light_h / (np.linalg.norm(light_h) + 1e-6)
+    perp = np.array([-light_h[1], light_h[0]], dtype=np.float32)
+    travel = -light_h
+    caustic_center = center[:2] + travel * radius * 1.08
+    delta = xy - caustic_center
+    along = np.sum(delta * travel, axis=-1)
+    across = np.sum(delta * perp, axis=-1)
+    broad = np.exp(
+        -np.square(along / (radius * 0.22))
+        - np.square(across / (radius * 0.78))
+    )
+    core = np.exp(
+        -np.square((along + radius * 0.035) / (radius * 0.075))
+        - np.square(across / (radius * 0.38))
+    )
+    caustic = 0.18 * broad + 0.30 * core
+
+    delta3 = center[None, :] - points
+    toward_light = np.sum(delta3 * light_dir, axis=-1)
+    distance_sq = np.sum(delta3 * delta3, axis=-1)
+    perpendicular = np.sqrt(
+        np.maximum(distance_sq - toward_light * toward_light, 0.0)
+    )
+    shadow_soft = 1.0 - smoothstep(
+        radius * 0.88, radius * 1.16, perpendicular
+    )
+    shadow_soft *= smoothstep(-radius * 0.12, radius * 0.18, toward_light)
+
+    contact_dist = np.sqrt(np.sum(rel_xy * rel_xy, axis=-1))
+    ambient_occlusion = 0.17 * np.exp(
+        -np.square(contact_dist / (radius * 0.92))
+    )
+    return caustic, shadow_soft, ambient_occlusion

+def refract(direction, normal, eta):
+    cos_i = -np.sum(direction * normal, axis=-1)
+    k = 1.0 - eta * eta * (1.0 - cos_i * cos_i)
+    refracted = (
+        eta * direction
+        + (eta * cos_i - np.sqrt(np.maximum(k, 0.0)))[..., None] * normal
+    )
+    return refracted, k, cos_i
+

+def shade_plane(points, direction, camera, marble, center, radius, light_dir):
+    px, py = project_to_pixels(points, camera)
+    world_x = (points[..., 0] + 2.5) / 5.0 * (marble.shape[1] - 1)
+    world_y = (points[..., 1] + 2.5) / 5.0 * (marble.shape[0] - 1)
+    color = bilinear_sample(marble, world_x, world_y)
+
+    # Polished-stone grazing reflection.
+    normal = np.array([0.0, 0.0, 1.0], dtype=np.float32)
+    cos_view = np.clip(-np.sum(direction * normal, axis=-1), 0.0, 1.0)
+    fresnel = 0.025 + 0.975 * np.power(1.0 - cos_view, 5.0)
+    reflected = direction - 2.0 * np.sum(direction * normal, axis=-1)[..., None] * normal
+    env = environment_color(reflected)
+    color = color * (1.0 + 0.08 * fresnel[..., None]) + env * (
+        0.17 * fresnel
+    )[..., None]
+
+    caustic, shadow, ao = caustic_and_shadow(points, center, radius, light_dir)
+    color *= (1.0 - 0.30 * shadow - ao)[..., None]
+    color += caustic[..., None] * np.array([1.0, 0.96, 0.82], dtype=np.float32)
+    return color, (px, py)

+def render_frame(frame_index, marble, camera, ray_origin, ray_dir, plane_t, base_pixels):

+            inward * exit_normal, axis=-1

+            trans_hit, center, radius, light_dir

+    color = color / (1.0 + 0.10 * color)

+    color = np.clip((color - 0.5) * 1.035 + 0.5, 0.0, 1.0)
+    return (color * 255.0 + 0.5).astype(np.uint8)

+def normalize_rows(vectors):
+    lengths = np.linalg.norm(vectors, axis=-1, keepdims=True)
+    return vectors / np.maximum(lengths, 1e-8)

+def main():

+    camera = make_camera()
+    origin, forward, right, up, tan_v, tan_h = camera
+
+    u = (np.arange(W, dtype=np.float32) + 0.5) / W
+    v = (np.arange(H, dtype=np.float32) + 0.5) / H

+    main()
