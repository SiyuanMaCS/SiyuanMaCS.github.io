+def smoothstep(edge0: float, edge1: float, values: np.ndarray) -> np.ndarray:
+    t = np.clip((values - edge0) / (edge1 - edge0), 0.0, 1.0)
+    return t * t * (3.0 - 2.0 * t)

+def normalize01(values: np.ndarray) -> np.ndarray:
+    low, high = np.percentile(values, (1.0, 99.0))
+    return np.clip((values - low) / max(high - low, 1e-6), 0.0, 1.0)

+def cubic_bezier(points: np.ndarray, s: float) -> np.ndarray:
+    p0, p1, p2, p3 = points
+    u = 1.0 - s
+    return (
+        (u**3) * p0
+        + 3.0 * (u**2) * s * p1
+        + 3.0 * u * (s**2) * p2
+        + (s**3) * p3
+    )

+def make_stroke_fields() -> tuple[np.ndarray, ...]:

+    y = yy / SUPERSAMPLE

+    control = np.array(
+        [[63.0, 61.0], [102.0, 76.0], [140.0, 96.0], [188.0, 113.0]],
+        dtype=np.float32,
+    )
+    samples = np.stack([cubic_bezier(control, i / 160.0) for i in range(161)])
+
+    distance = np.full((RH, RW), 1e6, dtype=np.float32)
+    along = np.zeros((RH, RW), dtype=np.float32)
+    signed_across = np.zeros((RH, RW), dtype=np.float32)
+    tangent_x = np.zeros((RH, RW), dtype=np.float32)
+    tangent_y = np.zeros((RH, RW), dtype=np.float32)
+    segment_lengths = np.linalg.norm(np.diff(samples, axis=0), axis=1)
+    cumulative = np.r_[0.0, np.cumsum(segment_lengths)]
+    total = float(cumulative[-1])
+
+    for index in range(len(samples) - 1):
+        p0 = samples[index]
+        p1 = samples[index + 1]
+        vx = float(p1[0] - p0[0])
+        vy = float(p1[1] - p0[1])
+        length2 = vx * vx + vy * vy
+        projection = np.clip(
+            ((x - p0[0]) * vx + (y - p0[1]) * vy) / max(length2, 1e-6), 0.0, 1.0
+        )
+        qx = p0[0] + projection * vx
+        qy = p0[1] + projection * vy
+        dx = x - qx
+        dy = y - qy
+        dist = np.sqrt(dx * dx + dy * dy)
+        update = dist < distance
+        distance[update] = dist[update]
+        along[update] = (cumulative[index] + projection[update] * segment_lengths[index]) / total
+        segment_len = math.sqrt(length2)
+        cross = (dx * (-vy) + dy * vx) / max(segment_len, 1e-6)
+        signed_across[update] = cross[update]
+        tangent_x[update] = vx / max(segment_len, 1e-6)
+        tangent_y[update] = vy / max(segment_len, 1e-6)

+    tooth = smoothed_noise(rng, (RH, RW), 1.0)
+    coarse = smoothed_noise(rng, (RH, RW), 9.0)
+    return distance, along, signed_across, tangent_x, tangent_y, rough, tooth, coarse

+    color = np.clip(color, 0.0, 255.0)
