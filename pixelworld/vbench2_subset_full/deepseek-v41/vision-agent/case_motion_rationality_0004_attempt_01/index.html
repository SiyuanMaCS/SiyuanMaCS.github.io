+def smoothstep(edge0: float, edge1: float, value):
+    t = np.clip((value - edge0) / (edge1 - edge0), 0.0, 1.0)
+    return t * t * (3.0 - 2.0 * t)

+def rounded_rect_sdf(u, v, cx, cy, hx, hy, radius):
+    qx = np.abs(u - cx) - (hx - radius)
+    qy = np.abs(v - cy) - (hy - radius)
+    outside = np.hypot(np.maximum(qx, 0.0), np.maximum(qy, 0.0))
+    inside = np.minimum(np.maximum(qx, qy), 0.0)
+    return outside + inside - radius

+def make_noise(seed: int, shape: tuple[int, int], sigma: float) -> np.ndarray:
+    rng = np.random.default_rng(seed)
+    noise = rng.normal(0.0, 1.0, shape).astype(np.float32)
+    noise = cv2.GaussianBlur(noise, (0, 0), sigma)
+    scale = float(np.std(noise))
+    return noise / max(scale, 1e-6)

+def make_coarse_noise(seed: int, cells: tuple[int, int], sigma: float) -> np.ndarray:

+def distance_to_segment(u, v, a, b):
