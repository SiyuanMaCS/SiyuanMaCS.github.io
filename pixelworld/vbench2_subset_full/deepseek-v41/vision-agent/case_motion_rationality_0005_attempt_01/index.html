+def lerp_point(a: tuple[float, float], b: tuple[float, float], t: float) -> tuple[float, float]:

+def ease(t: float) -> float:

+def cubic_points(
+    p0: tuple[float, float],
+    p1: tuple[float, float],
+    p2: tuple[float, float],
+    p3: tuple[float, float],
+    count: int = 30,
+) -> list[tuple[float, float]]:
+    result = []
+    for index in range(count):
+        t = index / (count - 1)
+        u = 1.0 - t
+        x = u**3 * p0[0] + 3 * u * u * t * p1[0] + 3 * u * t * t * p2[0] + t**3 * p3[0]
+        y = u**3 * p0[1] + 3 * u * u * t * p1[1] + 3 * u * t * t * p2[1] + t**3 * p3[1]
+        result.append((x, y))
+    return result

+def arm_points(shoulder: tuple[float, float], hand: tuple[float, float]) -> tuple[tuple[float, float], tuple[float, float]]:
+    upper = 43.0
+    lower = 40.0
+    dx = hand[0] - shoulder[0]
+    dy = hand[1] - shoulder[1]
+    distance = max(1.0, min(upper + lower - 0.25, math.hypot(dx, dy)))
+    ux, uy = dx / max(1e-6, math.hypot(dx, dy)), dy / max(1e-6, math.hypot(dx, dy))
+    along = (upper * upper - lower * lower + distance * distance) / (2 * distance)
+    height = math.sqrt(max(0.0, upper * upper - along * along))
+    base = (shoulder[0] + ux * along, shoulder[1] + uy * along)
+    candidates = [
+        (base[0] - uy * height, base[1] + ux * height),
+        (base[0] + uy * height, base[1] - ux * height),
+    ]
+    elbow = max(candidates, key=lambda p: p[1])
+    return elbow, hand
