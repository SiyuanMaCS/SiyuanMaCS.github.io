+def keyed(frame, keys):
+    """Smoothly interpolate scalar or point keys: [(frame, value), ...]."""
+    if frame <= keys[0][0]:
+        return keys[0][1]
+    for (fa, va), (fb, vb) in zip(keys[:-1], keys[1:]):
+        if frame <= fb:
+            u = smoothstep((frame - fa) / max(1e-6, fb - fa))
+            if isinstance(va, tuple):
+                return tuple(va[i] + (vb[i] - va[i]) * u for i in range(len(va)))
+            return va + (vb - va) * u
+    return keys[-1][1]
