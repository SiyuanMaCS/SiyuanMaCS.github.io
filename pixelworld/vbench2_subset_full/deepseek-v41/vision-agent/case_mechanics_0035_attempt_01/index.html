+def smoothstep(edge0, edge1, x):
+    t = np.clip((x - edge0) / (edge1 - edge0), 0.0, 1.0)

+def normalize(vector):

+def fbm(rng, shape, scales, weights):
+    field = np.zeros(shape, dtype=np.float32)
+    for sigma, weight in zip(scales, weights):
+        layer = rng.standard_normal(shape).astype(np.float32)
+        layer = gaussian_filter(layer, sigma=sigma, mode="reflect")
+        layer -= layer.mean()
+        layer /= layer.std() + 1e-6
+        field += weight * layer
+    field -= field.mean()
+    field /= field.std() + 1e-6
+    return field

+def soft_segment_distance(x, y, ax, ay, bx, by):
+    vx, vy = bx - ax, by - ay

+    return np.hypot(px - t * vx, py - t * vy), t
