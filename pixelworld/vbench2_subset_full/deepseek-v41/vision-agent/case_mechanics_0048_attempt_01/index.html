+def cubic_bezier(p0, p1, p2, p3, count=90):
+    t = np.linspace(0.0, 1.0, count, dtype=np.float32)[1:]
+    u = 1.0 - t

+        u[:, None] ** 3 * np.asarray(p0, dtype=np.float32)
+        + 3.0 * u[:, None] ** 2 * t[:, None] * np.asarray(p1, dtype=np.float32)
+        + 3.0 * u[:, None] * t[:, None] ** 2 * np.asarray(p2, dtype=np.float32)
+        + t[:, None] ** 3 * np.asarray(p3, dtype=np.float32)
+    )
+    return points
