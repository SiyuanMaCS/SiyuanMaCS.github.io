# Mesh: world points, smooth normals, triangular indices, diffuse colour, gloss.
def ellipsoid(center,radii,color,nu=24,nv=14,gloss=.08):
    c=np.array(center); r=np.array(radii); vs=[]; ns=[]; fs=[]; cs=[]
    for j in range(nv+1):
        th=math.pi*j/nv
        for k in range(nu):
            ph=2*math.pi*k/nu
            n=np.array([math.sin(th)*math.cos(ph),math.sin(th)*math.sin(ph),math.cos(th)])
            vs.append(c+r*n); nn=n/r; ns.append(nn/np.linalg.norm(nn))
            variation=1+.025*math.sin(ph*6+th*19)+.012*math.sin(ph*23+th*41)
            cs.append(np.array(color)*variation)
    for j in range(nv):
        for k in range(nu):
            a=j*nu+k; b=j*nu+(k+1)%nu; d=(j+1)*nu+k; e=(j+1)*nu+(k+1)%nu
            fs.extend([(a,d,b),(b,d,e)])
    return [np.array(vs),np.array(ns),np.array(fs),np.array(cs),gloss]

def tube(points,radii,color,nu=18,gloss=.09,wrinkles=False):
    pts=np.array(points,float); rr=np.array(radii,float)
    if rr.ndim==1: rr=np.stack([rr,rr],1)
    vs=[];ns=[];cs=[];fs=[]
    for j,p in enumerate(pts):
        t=pts[min(j+1,len(pts)-1)]-pts[max(0,j-1)]; t/=np.linalg.norm(t)
        b=np.cross(t,[0,0,1.])
        if np.linalg.norm(b)<.01:b=np.cross(t,[0,1,0])
        b/=np.linalg.norm(b); n=np.cross(b,t)
        for k in range(nu):
            a=2*math.pi*k/nu
            normal=math.cos(a)*b+math.sin(a)*n
            vs.append(p+math.cos(a)*rr[j,0]*b+math.sin(a)*rr[j,1]*n)
            ns.append(normal)
            # subtle skin irregularity and transverse joint creases
            v=1+.015*math.sin(a*11+j*4.9)
            if wrinkles and j in (7,8,16,17) and math.sin(a)>.35:v*=.88
            cs.append(np.array(color)*v)
    for j in range(len(pts)-1):
        for k in range(nu):
            a=j*nu+k;b=j*nu+(k+1)%nu;c=(j+1)*nu+k;d=(j+1)*nu+(k+1)%nu
            fs.extend([(a,c,b),(b,c,d)])
    return [np.array(vs),np.array(ns),np.array(fs),np.array(cs),gloss]

def bezier(control,n=26):
    t=np.linspace(0,1,n)[:,None];p=np.array(control)
    return (1-t)**3*p[0]+3*(1-t)**2*t*p[1]+3*(1-t)*t*t*p[2]+t**3*p[3]

def poly_face(vertices,color,gloss=.14):
    v=np.array(vertices,float);normal=np.cross(v[1]-v[0],v[2]-v[0]);normal/=np.linalg.norm(normal)
    return [v,np.tile(normal,(len(v),1)),np.array([(0,i,i+1) for i in range(1,len(v)-1)]),np.tile(color,(len(v),1)),gloss]

def box_bevel(x0,x1,y0,y1,z0,z1,bevel,col):
    # Small flat top with bevelled edges gives each scored square its familiar moulded form.
    low=[(x0,y0,z0),(x1,y0,z0),(x1,y1,z0),(x0,y1,z0)]
    rim=[(x0,y0,z1-bevel),(x1,y0,z1-bevel),(x1,y1,z1-bevel),(x0,y1,z1-bevel)]
    top=[(x0+bevel,y0+bevel,z1),(x1-bevel,y0+bevel,z1),(x1-bevel,y1-bevel,z1),(x0+bevel,y1-bevel,z1)]
    out=[poly_face(top,col,.30)]
    for k in range(4):
        j=(k+1)%4
        out.append(poly_face([low[k],low[j],rim[j],rim[k]],np.array(col)*.70,.18))
        out.append(poly_face([rim[k],rim[j],top[j],top[k]],np.array(col)*.94,.28))
    return out

def chocolate(sign,broken):

    # normal direction made explicit below for all horizontal polygon faces

    radii=np.stack([np.linspace(12,13.7,26),np.linspace(10,9.2,26)],1)

    # Four fingers curve around the outer edge of the chocolate into its underside.
    for k,y in enumerate([15,3,-9,-20]):

    # Short, curved natural thumbnail, on the upper face of the distal thumb.
    nail=ellipsoid(pt(42,-16.1,44.0),(6.0,4.0,.65),[.77,.49,.38],nu=24,nv=8,gloss=.27)
    # Rotate nail to align with the thumb's diagonal rather than the frame.

    # Ribbed blue knit cuffs near the lower corners.
    cuff=bezier([pt(113,-101,-6),pt(109,-92,-1),pt(103,-82,3),pt(99,-73,6)],18)
    rr=np.stack([np.linspace(17,14.2,18),np.linspace(14.5,12.3,18)],1)

    for j in range(18):

    out.append(cuffmesh)

def transform(mesh,R,offset):
    return [mesh[0]@R.T+offset,mesh[1]@R.T,mesh[2],mesh[3],mesh[4]]

def projection(vertices,zoom=1):

                     (86-(vertices[:,1]*.80+vertices[:,2]*.60)*1.17*zoom)*S,

def render(meshes,zoom):
    image=base.copy();zbuf=np.full((RH,RW),-1e6,dtype=np.float32)
    # Soft, coherent cast shadows projected from the geometry onto the table.
    sm=Image.new('L',(RW,RH),0);sd=ImageDraw.Draw(sm)
    for m in meshes:
        v=m[0]; select=v[::max(1,len(v)//90)]
        flat=select.copy();flat[:,0]-=LIGHT[0]/LIGHT[2]*flat[:,2];flat[:,1]-=LIGHT[1]/LIGHT[2]*flat[:,2];flat[:,2]=0
        p=projection(flat,zoom)
        # Point splats are unified by a large, soft penumbra.
        for x,y,z in p:sd.ellipse((x-5*S,y-4*S,x+5*S,y+4*S),fill=100)
    shadow=np.asarray(sm.filter(ImageFilter.GaussianBlur(5*S)))/255.
    image*=1-shadow[:,:,None]*.48
    for verts,norms,faces,colors,gloss in meshes:
        proj=projection(verts,zoom)
        # Lighting evaluated at vertices and interpolated across smooth skin.
        diffuse=np.maximum(0,norms@LIGHT)
        sky=np.clip(norms[:,2]*.25+.72,.38,.97)
        spec=np.maximum(0,norms@HALF)**(42 if gloss>.2 else 23)
        lit=colors*(.38*sky[:,None]+.82*diffuse[:,None])
        lit+=spec[:,None]*gloss*np.array([.83,.72,.57])
        # A little reflected warmth from the tabletop.
        lit+=np.clip(-norms[:,2],0,1)[:,None]*np.array([.035,.017,.007])
        # Film-like display gamma, retaining rich dark chocolate.
        lit=np.clip(lit,0,1)**(1/1.75)*255
        for tri in faces:
            p=proj[tri];minx=max(0,int(np.floor(p[:,0].min())));maxx=min(RW-1,int(np.ceil(p[:,0].max())))
            miny=max(0,int(np.floor(p[:,1].min())));maxy=min(RH-1,int(np.ceil(p[:,1].max())))
            if minx>maxx or miny>maxy:continue
            a,b,c=p
            den=(b[1]-c[1])*(a[0]-c[0])+(c[0]-b[0])*(a[1]-c[1])
            if abs(den)<1e-7:continue
            gx=np.arange(minx,maxx+1)[None,:]+.5;gy=np.arange(miny,maxy+1)[:,None]+.5
            u=((b[1]-c[1])*(gx-c[0])+(c[0]-b[0])*(gy-c[1]))/den
            v=((c[1]-a[1])*(gx-c[0])+(a[0]-c[0])*(gy-c[1]))/den
            w=1-u-v
            depth=u*a[2]+v*b[2]+w*c[2]
            zb=zbuf[miny:maxy+1,minx:maxx+1]
            valid=(u>=-.0001)&(v>=-.0001)&(w>=-.0001)&(depth>zb)
            if not valid.any():continue
            rgb=u[:,:,None]*lit[tri[0]]+v[:,:,None]*lit[tri[1]]+w[:,:,None]*lit[tri[2]]
            im=image[miny:maxy+1,minx:maxx+1];im[valid]=rgb[valid];zb[valid]=depth[valid]
    # Lens vignette and subtle deterministic sensor grain.
    vign=1-.10*((xx-RW/2)/(RW*.74))**2-.07*((yy-RH/2)/(RH*.85))**2
    image=image*vign[:,:,None]+noise[:,:,None]*.33
    return np.asarray(Image.fromarray(np.uint8(np.clip(image,0,255))).resize((W,H),Image.Resampling.LANCZOS))

def smooth(t):t=np.clip(t,0,1);return t*t*(3-2*t)

def scene(frame):
    t=frame/4
    broken=frame>=7
    if not broken:
        pull=0;angle=math.radians(3.0)*smooth((frame-2)/4)
    else:
        progress=smooth((frame-6)/6)
        pull=23*progress
        angle=math.radians(3+12*math.sin(min((frame-6)/7,1)*math.pi/2))
    # Settle and reveal both cleanly separated halves without dropping the grip.
    settle=smooth((frame-13)/6)
    angle*=1-.32*settle
    meshes=[]
    for sign,hand in [(-1,LEFT),(1,RIGHT)]:
        a=sign*angle; b=sign*math.radians(5)*smooth((frame-7)/7)
        Ry=np.array([[math.cos(a),0,math.sin(a)],[0,1,0],[-math.sin(a),0,math.cos(a)]])
        Rz=np.array([[math.cos(b),-math.sin(b),0],[math.sin(b),math.cos(b),0],[0,0,1]])
        R=Rz@Ry
        offset=np.array([sign*pull,1.5*math.sin(t*1.7),29+1.2*math.sin(t*1.9)])
        # Hands were constructed in the same coordinate system as the bar, with their grip at its height.
        for m in hand:
            local=[m[0]-np.array([0,0,29]),m[1],m[2],m[3],m[4]]
            meshes.append(transform(local,R,offset))
        for m in CHOCS[broken][sign]:meshes.append(transform(m,R,offset))
    # Tiny shards release from the fracture exactly when it occurs, then fall under gravity.
    if broken:
        for k in range(9):
            dt=max(0,(frame-7)/4-k*.018)
            vx=8*math.sin(k*6.4);vy=5*math.cos(k*3.0)
            z=max(.65,30+6*dt-95*dt*dt)
            fall=min(dt,.60)
            pos=np.array([vx*fall,-24+k*6+vy*fall,z])
            size=.6+(k%3)*.32
            m=ellipsoid(pos,(size,size*.68,size*.5),CHOC*.85,nu=8,nv=5,gloss=.05)
            meshes.append(m)
    return meshes
