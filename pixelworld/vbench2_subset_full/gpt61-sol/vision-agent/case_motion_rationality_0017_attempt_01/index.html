def rotation_z(a):

class Solid:
    def __init__(self,center,scale,mat,rot=None,lo=None,hi=None,box=False,planes=None):
        self.c=np.array(center,dtype=DT); self.s=np.array(scale,dtype=DT)
        self.R=np.eye(3,dtype=DT) if rot is None else np.array(rot,dtype=DT)
        self.mat=mat;self.box=box
        self.lo=np.array(lo if lo is not None else [-np.inf]*3,dtype=DT)
        self.hi=np.array(hi if hi is not None else [np.inf]*3,dtype=DT)
        if box:self.lo=-self.s;self.hi=self.s
        self.bound=float(np.linalg.norm(self.s)) if box else float(max(self.s))
        self.planes=planes
    def intersect(self,O,D):
        # O may be one camera origin or one origin per ray.
        OC=O-self.c
        if OC.ndim==1:
            b=D@OC; c=float(OC@OC-self.bound**2)
            disc=b*b-c
        else:
            b=np.einsum('ij,ij->i',D,OC);c=np.sum(OC*OC,axis=1)-self.bound**2;disc=b*b-c
        mask=(disc>=0)&(-b+np.sqrt(np.maximum(disc,0))>0.0001)
        ix=np.flatnonzero(mask)
        if not len(ix):return ix,np.array([],dtype=DT),np.zeros((0,3),DT),np.zeros((0,3),DT),np.array([],int)
        dd=D[ix]@self.R
        oo=(OC if OC.ndim==1 else OC[ix])@self.R
        if oo.ndim==1:oo=np.broadcast_to(oo,dd.shape)
        if self.box:
            near=np.full(len(ix),-np.inf,DT);far=np.full(len(ix),np.inf,DT)
        else:
            a=np.sum((dd/self.s)**2,axis=1); b2=np.sum(dd*oo/(self.s*self.s),axis=1)
            c2=np.sum((oo/self.s)**2,axis=1)-1
            det=b2*b2-a*c2
            sq=np.sqrt(np.maximum(det,0));near=(-b2-sq)/a;far=(-b2+sq)/a
            far[det<0]=-np.inf
        surf=np.full(len(ix),-1,dtype=np.int8)
        for ax in range(3):
            if np.isfinite(self.lo[ax]) or np.isfinite(self.hi[ax]):
                div=np.where(np.abs(dd[:,ax])<1.e-9,1.e-9,dd[:,ax])
                t1=(self.lo[ax]-oo[:,ax])/div;t2=(self.hi[ax]-oo[:,ax])/div
                nn=np.minimum(t1,t2); ff=np.maximum(t1,t2)
                changed=nn>near
                surf[changed]=ax*2+(dd[changed,ax]<0)
                near=np.maximum(near,nn);far=np.minimum(far,ff)
        # Additional convex blade boundary planes, dot(p,n)<=k.
        if self.planes:
            for j,(normal,k) in enumerate(self.planes):
                normal=np.asarray(normal,DT)
                dv=dd@normal; ov=oo@normal
                tt=(k-ov)/np.where(abs(dv)<1.e-9,1.e-9,dv)
                nn=np.where(dv<0,tt,-np.inf);ff=np.where(dv>=0,tt,np.inf)
                changed=nn>near;surf[changed]=6+j
                near=np.maximum(near,nn);far=np.minimum(far,ff)
        good=(far>=near)&(near>.00008)
        ix=ix[good];tt=near[good];p=oo[good]+dd[good]*tt[:,None];sid=surf[good]
        normal=p/(self.s*self.s)
        if self.box:normal[:]=0
        for ax in range(3):
            for side in range(2):
                m=sid==ax*2+side;normal[m]=0;normal[m,ax]=1 if side else -1
        if self.planes:
            for j,(nm,k) in enumerate(self.planes):normal[sid==6+j]=nm
        normal=unit(normal)@self.R.T
        return ix,tt,p,normal,sid

def limb(a,b,r,mat,r2=None):
    a=np.array(a,DT);b=np.array(b,DT);v=b-a;L=np.linalg.norm(v);v/=L
    helper=np.array([0,0,1],DT) if abs(v[2])<.9 else np.array([1,0,0],DT)
    x=unit(np.cross(helper,v));z=unit(np.cross(x,v))
    return sphere((a+b)/2,[r,L/2+r*.4,r if r2 is None else r2],mat,np.stack([x,v,z],axis=1))

def hsh(x,y,z=0):return np.mod(np.sin(x*127.1+y*311.7+z*74.7)*43758.5453,1)
def noise(p,f):
    q=p*f;g=np.floor(q);v=q-g;v=v*v*(3-2*v)
    out=np.zeros(len(p),DT)
    for x in [0,1]:
      for y in [0,1]:
       for z in [0,1]:
        out+=hsh(g[:,0]+x,g[:,1]+y,g[:,2]+z)*(v[:,0] if x else 1-v[:,0])*(v[:,1] if y else 1-v[:,1])*(v[:,2] if z else 1-v[:,2])
    return out

def trace(objects,O,dirs,need_data=True):
    N=len(dirs);depth=np.full(N,np.inf,DT);ids=np.full(N,-1,np.int16)
    pos=np.zeros((N,3),DT);norm=np.zeros((N,3),DT);surf=np.full(N,-1,np.int8)
    for i,obj in enumerate(objects):
        ix,ts,ps,ns,ss=obj.intersect(O,dirs)
        keep=ts<depth[ix];ix=ix[keep]
        depth[ix]=ts[keep];ids[ix]=i
        if need_data:pos[ix]=ps[keep];norm[ix]=ns[keep];surf[ix]=ss[keep]
    return depth,ids,pos,norm,surf


def render(frame):
    objects=scene(frame)
    dep,ids,local,norm,surf=trace(objects,CAM,D)
    world=CAM+D*dep[:,None]
    albedo=np.zeros((W*H,3),DT);rough=np.ones(W*H,DT);spec=np.zeros(W*H,DT)
    for i,obj in enumerate(objects):
        m=ids==i
        if m.any():
            a,r,s=material(obj,local[m],norm[m],surf[m]);albedo[m]=a;rough[m]=r;spec[m]=s
    # Cast directional light shadows on every visible surface.
    sdir=np.broadcast_to(LIGHT,(W*H,3))
    shadowdep,*_=trace(objects,world+norm*.007,sdir,False)
    shadow=np.isfinite(shadowdep).reshape(H,W).astype(DT)
    shimg=Image.fromarray(np.uint8(shadow*255)).filter(ImageFilter.GaussianBlur(1.35))
    shadow=np.asarray(shimg,dtype=DT).reshape(-1)/255
    ndotl=np.clip(norm@LIGHT,0,1)
    # Broad skylight, warm key light, and gentle countertop bounce.
    ambient=.32+.11*np.clip(norm[:,1],0,1)
    direct=ndotl*(1-shadow*.87)*.73
    shade=ambient+direct
    lit=albedo*shade[:,None]*np.array([1.055,1.015,.95])
    skin=np.array([obj.mat=='skin' for obj in objects])[np.maximum(ids,0)]
    lit[skin]+=(np.clip(-norm[skin]@LIGHT,0,1)*.033)[:,None]*np.array([1,.35,.13])
    half=unit(LIGHT-D)
    phong=np.clip(np.sum(norm*half,axis=1),0,1)**(7+100*(1-rough))
    lit+=phong[:,None]*spec[:,None]*(1-shadow[:,None]*.85)*np.array([1.05,1.02,.95])
    # Blade reflections emphasize its flat, thin steel form.
    metal=np.array([obj.mat in ['metal','rivet'] for obj in objects])[np.maximum(ids,0)]
    lit[metal]+=np.clip(norm[metal,1],0,1)[:,None]*np.array([.12,.13,.14])
    lit=np.clip(lit,0,1)
    # Gentle photographic contrast and a very subtle vignette.
    lit=lit**.91
    vign=1-.075*(xx*xx+yy*yy).reshape(-1)
    lit*=vign[:,None]
    im=Image.fromarray(np.uint8(np.clip(lit.reshape(H,W,3)*255,0,255)))
    im=im.resize((256,144),Image.Resampling.LANCZOS)
    return np.array(im)
