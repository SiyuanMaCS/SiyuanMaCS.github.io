import numpy as np
from PIL import Image, ImageDraw, ImageFilter
from pathlib import Path
import math
import artifact_io

ROOT=Path(__file__).resolve().parent
W,H,S=256,144,3
RW,RH=W*S,H*S
rng=np.random.default_rng(45)
CAM=np.array([4.8,-8.4,8.5]); TARGET=np.array([0.,0.2,0.55])
fwd=(TARGET-CAM); fwd/=np.linalg.norm(fwd)
right=np.cross(fwd,[0,0,1]); right/=np.linalg.norm(right)
up=np.cross(right,fwd)
FOCAL=RW*1.17
LIGHT=np.array([-3.,-4.,9.])
class Mesh:
    def __init__(self): self.parts=[]
    def add(self,v,f,mat):
        self.parts.append((np.array(v,float),np.array(f,int),mat))

def box(m,lo,hi,mat):
    x,y,z=lo; X,Y,Z=hi
    v=[(x,y,z),(X,y,z),(X,Y,z),(x,Y,z),(x,y,Z),(X,y,Z),(X,Y,Z),(x,Y,Z)]
    f=[(0,2,1),(0,3,2),(4,5,6),(4,6,7),(0,1,5),(0,5,4),(1,2,6),(1,6,5),(2,3,7),(2,7,6),(3,0,4),(3,4,7)]
    m.add(v,f,mat)

def ellipsoid(m,c,r,mat,n=18,k=12):
    v=[]
    for a in range(k+1):
        t=math.pi*a/k
        for b in range(n):
            p=2*math.pi*b/n
            v.append(np.array(c)+np.array(r)*[math.sin(t)*math.cos(p),math.sin(t)*math.sin(p),math.cos(t)])
    f=[]
    for a in range(k):
        for b in range(n):
            i=a*n+b; j=a*n+(b+1)%n
            f.extend([(i,j,j+n),(i,j+n,i+n)])
    m.add(v,f,mat)

def capsule(m,a,b,r,mat,n=14,k=12,r2=None):
    a=np.array(a,float); b=np.array(b,float); d=b-a; length=np.linalg.norm(d); d/=length
    v1=np.cross(d,[0,0,1] if abs(d[2])<.9 else [0,1,0]); v1/=np.linalg.norm(v1); v2=np.cross(d,v1)
    v=[]
    # elliptical-ended tapered cylinder
    for q in range(k+1):
        theta=math.pi*q/k
        tt=(1-math.cos(theta))/2
        center=a+(b-a)*tt
        rr=(r*(1-tt)+(r if r2 is None else r2)*tt)*math.sin(theta)
        for j in range(n):
            ph=2*math.pi*j/n
            v.append(center+rr*(v1*math.cos(ph)+v2*math.sin(ph)))
    f=[]
    for q in range(k):
        for j in range(n):
            i=q*n+j; jj=q*n+(j+1)%n
            f.extend([(i,jj,jj+n),(i,jj+n,i+n)])
    m.add(v,f,mat)

def project(v):
    d=v-CAM
    p=np.stack([d@right,d@up,d@fwd],axis=1)
    return np.stack([RW/2+FOCAL*p[:,0]/p[:,2], RH*.53-FOCAL*p[:,1]/p[:,2],p[:,2]],axis=1)

def normals(v,f):
    face=np.cross(v[f[:,1]]-v[f[:,0]],v[f[:,2]]-v[f[:,0]])
    nn=np.zeros_like(v)
    for j in range(3):np.add.at(nn,f[:,j],face)
    nn/=np.maximum(np.linalg.norm(nn,axis=1,keepdims=True),1e-8)
    return nn

def shade(v,n,mat):
    l=LIGHT-v; l/=np.linalg.norm(l,axis=1,keepdims=True)
    view=CAM-v; view/=np.linalg.norm(view,axis=1,keepdims=True)
    half=l+view; half/=np.linalg.norm(half,axis=1,keepdims=True)
    diffuse=np.maximum(np.sum(n*l,axis=1),0)
    # sky and warm bounce
    amb=.42+.10*np.maximum(n[:,2],0)
    intensity=amb+.52*diffuse
    colors={'skin':(193,132,94),'nail':(212,162,139),'orange':(248,111,13),'cut':(252,137,27),'wood':(177,130,79),'edge':(130,87,48),'counter':(76,84,85),'apron':(43,65,69),'cloth':(202,206,192),'metal':(162,178,188),'dark':(32,31,28),'rivet':(155,161,156),'ceramic':(227,226,209),'green':(52,99,49)}
    base=np.tile(colors[mat],(len(v),1)).astype(float)
    x,y,z=v.T
    if mat in ('wood','edge'):
        grain=3.5*np.sin(22*y+1.2*np.sin(x*1.8))+2.2*np.sin(y*74+np.sin(x*2.9))
        grain+=4*np.sin(y*4+x*.08)
        base+=grain[:,None]*np.array([1,.85,.60])
    if mat=='counter':base+=((np.sin(x*28+y*23)+np.sin(x*53-y*38))*1.5)[:,None]
    if mat in ('orange','cut'):
        base+=(2*np.sin(x*22+y*31)+2*np.cos(z*38+x*5))[:,None]
    if mat=='skin':base+=(1.8*np.sin(x*8+y*12+z*6))[:,None]
    if mat=='apron':base+=(2.8*np.sin(x*21+z*8))[:,None]
    col=base*intensity[:,None]
    power=80 if mat in ('metal','rivet') else 34
    strength=115 if mat in ('metal','rivet') else (18 if mat in ('skin','orange','cut','ceramic') else 6)
    spec=np.maximum(np.sum(n*half,axis=1),0)**power*strength
    col+=spec[:,None]
    if mat=='metal':
        reflect=np.maximum(n@np.array([0,-.65,.76]),0)
        col+=reflect[:,None]*47
    return np.clip(col,0,255)

def render_parts(parts,img,depth):
    for v,f,mat in parts:
        p=project(v)
        # Flat boxes and blade stay planar; curved meshes use vertex normals.
        flat=mat in ('wood','edge','counter','metal','dark','cut')
        vn=normals(v,f)
        if not flat: vc=shade(v,vn,mat)
        for tri in f:
            pts=p[tri]; pos=v[tri]
            if np.any(pts[:,2]<=0):continue
            x0=max(0,int(np.floor(pts[:,0].min()))); x1=min(RW-1,int(np.ceil(pts[:,0].max())))
            y0=max(0,int(np.floor(pts[:,1].min()))); y1=min(RH-1,int(np.ceil(pts[:,1].max())))
            if x1<x0 or y1<y0:continue
            a,b,c=pts
            den=(b[1]-c[1])*(a[0]-c[0])+(c[0]-b[0])*(a[1]-c[1])
            if abs(den)<1e-7:continue
            yy,xx=np.mgrid[y0:y1+1,x0:x1+1]
            wa=((b[1]-c[1])*(xx-c[0])+(c[0]-b[0])*(yy-c[1]))/den
            wb=((c[1]-a[1])*(xx-c[0])+(a[0]-c[0])*(yy-c[1]))/den
            wc=1-wa-wb
            inv=wa/a[2]+wb/b[2]+wc/c[2]
            zz=1/np.maximum(inv,1e-8)
            mask=(wa>=-1e-5)&(wb>=-1e-5)&(wc>=-1e-5)&(zz<depth[y0:y1+1,x0:x1+1])
            if not np.any(mask):continue
            if flat:
                normal=np.cross(pos[1]-pos[0],pos[2]-pos[0]); normal/=max(np.linalg.norm(normal),1e-10)
                colors=shade(pos,np.tile(normal,(3,1)),mat)
            else:colors=vc[tri]
            cs=(wa[...,None]*colors[0]/a[2]+wb[...,None]*colors[1]/b[2]+wc[...,None]*colors[2]/c[2])/inv[...,None]
            img[y0:y1+1,x0:x1+1][mask]=cs[mask]
            depth[y0:y1+1,x0:x1+1][mask]=zz[mask]

def carrot_piece(m,yl,yh,shift=0,dx=0):
    v=[]; f=[]; nx=29; ny=10
    # closed solid surfaces, curved orange top and flat orange cut walls
    def radius(x): return .58*(1-.20*(x+2)/3.7)
    for i in range(nx):
        x=-2+3.65*i/(nx-1); r=radius(x)
        for j in range(ny):
            y=yl+(yh-yl)*j/(ny-1)
            yy=np.clip(y,-r+.001,r-.001)
            zz=.235+math.sqrt(max(0,r*r-yy*yy))*.80
            v.append((x+dx,yy+shift,zz))
    for i in range(nx-1):
        for j in range(ny-1):
            a=i*ny+j; f.extend([(a,a+1,a+ny+1),(a,a+ny+1,a+ny)])
    m.add(v,f,'orange')
    for j in [0,ny-1]:
        vs=[]
        for i in range(nx):
            p=v[i*ny+j]; vs.extend([p,(p[0],p[1],.235)])
        fs=[]
        for i in range(nx-1):
            a=i*2
            fs.extend([(a,a+2,a+3),(a,a+3,a+1)] if j==0 else [(a,a+3,a+2),(a,a+1,a+3)])
        m.add(vs,fs,'cut')
    for i in [0,nx-1]:
        vs=[v[i*ny+j] for j in range(ny)]+[(v[i*ny+ny-1][0],v[i*ny+ny-1][1],.235),(v[i*ny][0],v[i*ny][1],.235)]

def make_frame(frame):
    m=Mesh()
    # Three deliberately readable lengthwise cuts, six samples each.
    cut_lines=[-.30,.015,.28]
    cyc=min(frame//6,2);phase=frame%6 if frame<18 else 4+(frame-18)*.5
    y=cut_lines[cyc]
    heights=[1.02,.70,.235,.23,.60,1.02]
    z=heights[int(min(phase,5))]
    if frame>=18:z=1.10+(frame-18)*.11
    tilt=[-.025,-.010,.0,.0,-.03,-.04][int(min(phase,5))]
    completed=sum(frame>=c*6+3 for c in range(3))
    # Pieces part only after knife reaches the board.
    boundaries=[-.58]+cut_lines[:completed]+[.58]
    for i,(lo,hi) in enumerate(zip(boundaries[:-1],boundaries[1:])):
        if i<completed:
            age=frame-(i*6+3)
            shift=-(completed-i)*.12*min(1,age/2)
            dx=.025*min(age,3)
        else:shift=0;dx=0
        carrot_piece(m,lo,hi,shift,dx)
    baton(m,-2.75,-1.68,2.55,.03)
    baton(m,-2.60,-1.37,2.62,-.025)
    baton(m,-2.35,-1.99,2.35,.11)
    knife(m,y,z,tilt)
    hands(m,y,z,frame)
    # physically projected, soft contact shadows of hands, carrot and knife
    mask=Image.new('L',(RW,RH),0);d=ImageDraw.Draw(mask)
    for v,f,mat in m.parts:
        if mat=='nail':continue
        fac=(LIGHT[2]-.222)/(LIGHT[2]-v[:,2])
        sh=LIGHT[None,:]+fac[:,None]*(v-LIGHT)
        p=project(sh)
        for face in f:
            poly=[(float(p[i,0]),float(p[i,1])) for i in face]
            d.polygon(poly,fill=65)
    mask=mask.filter(ImageFilter.GaussianBlur(2.5*S))
    alpha=np.asarray(mask)/255.
    img=img0.copy()*(1-alpha[:,:,None]*.70)
    depth=depth0.copy()
    render_parts(m.parts,img,depth)
    # subtle lens vignette, warm color response, clean antialiasing
    yy,xx=np.mgrid[0:RH,0:RW]
    vignette=1-.085*((xx/RW-.5)**2+(yy/RH-.5)**2)
    img=np.clip(img*vignette[:,:,None],0,255).astype(np.uint8)
    return np.array(Image.fromarray(img).resize((W,H),Image.Resampling.LANCZOS))
