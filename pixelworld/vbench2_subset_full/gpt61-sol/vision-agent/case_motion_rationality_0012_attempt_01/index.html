    for cx,cy,st in [(2.57,1.45,0),(-2.57,1.45,90),(-2.57,-1.45,180),(2.57,-1.45,270)]:
        for a in np.linspace(st,st+90,9):
            aa=np.radians(a);outline.append((cx+rad*np.cos(aa),cy+rad*np.sin(aa)))
    # consistent perimeter ordered counterclockwise
    center=[0,0,.13];verts=[center]+[(x,y,.13) for x,y in outline]
    scene.add(verts,[(0,i+1,(i+1)%len(outline)+1) for i in range(len(outline))],np.tile((0,0,1),(len(verts),1)),'wood')

    cycle=frame//4;phase=frame%4

    z=elevations[phase]
    completed=cycle+(phase>=2)
    end=1.007-.157*completed
    cylinder_x(scene,-1.88,end,0,.427,r=.292)
    # Two earlier rounds, and the new slices separate only after a completed stroke.
    round_slice(scene,1.65,-.43,flat=True)
    round_slice(scene,1.19,-1.01,flat=True)
    for j in range(completed):
        age=completed-1-j
        xx=.93-.157*j+.026*age
        yy=-.047-.053*age
        ang=.09+.035*age
        if j==completed-1 and phase==2: ang=.035
        round_slice(scene,xx,yy,angle=ang)

    retract=-.030*cycle
    scene.tube([(-2.02,2.58,1.82),(-1.83,1.95,1.28)], [.35,.275],'cloth',nu=28)
    scene.tube([(-1.83,1.98,1.3),(-1.60,1.37,1.07),(-1.25+retract,.79,.88),(-1.00+retract,.34,.77)], [.235,.205,.151,.167],'skin',nu=28)

    scene.tube([(1.14,3.12,2.16),(1.52,2.52,1.63)], [.38,.285],'cloth',nu=28)
    scene.tube([(1.52,2.55,1.62),(1.29,2.20,1.33),(cut+.28,1.93,z+.93),(cut+.06,1.62,z+.80)],[.248,.214,.158,.157],'skin',nu=28)

        zz=z+.66+(yy-1.1)*.07
        scene.tube([(cut+.08,yy,zz+.14),(cut+.214,yy-.02,zz+.06),(cut+.184,yy-.027,zz-.062),(cut+.040,yy-.014,zz-.107),(cut-.040,yy,zz-.06)],[.074,.081,.076,.065,.05],'skin',nu=16)

    scene.tube([(cut-.095,1.51,z+.91),(cut-.151,1.29,z+.867),(cut-.101,1.045,z+.79)],[.104,.092,.077],'skin',nu=18)
    scene.ellipsoid((cut-.099,1.035,z+.843),(.056,.077,.012),'nail',nu=16,nv=8)
    return scene,cut,z,completed


def shadows_and_finish(rgb,scene):
    # Broad, soft physical contact shadows projected onto the counter and board.
    shadow=Image.new('L',(RW,RH),0);d=ImageDraw.Draw(shadow)
    for verts,faces,normals,mat in scene.mesh:
        if mat not in ['skin','rind','metal','handle']:continue
        # Project silhouettes along the key light onto the chopping surface.
        q=verts.copy();height=np.maximum(0,q[:,2]-.133)
        q[:,0]-=LIGHT[0]/LIGHT[2]*height;q[:,1]-=LIGHT[1]/LIGHT[2]*height;q[:,2]=.132
        proj=project(q)[:,:2]
        # PIL polygons, triangle silhouettes accumulate into one consistent mask.
        for f in faces:
            xy=proj[f]
            if xy[:,1].max()<0 or xy[:,1].min()>=RH:continue
            d.polygon([tuple(v) for v in xy],fill=72)
    sm=np.asarray(shadow.filter(ImageFilter.GaussianBlur(3.0*SS)),float)/255
    # Shadows apply only where the board itself is visible (using wood hue).
    arr=rgb.astype(float)
    visiblewood=(arr[:,:,0]>arr[:,:,1]*1.16)&(arr[:,:,1]>arr[:,:,2]*1.19)
    # This color criterion can include skin; restrict by geometry surface depth instead.
    # Board visible mask rasterizes its outline at its exact plane.
    # Pixel classification also requires shade and the board's lower screen region.
    yy,xx=np.mgrid[:RH,:RW]
    visiblewood &= yy>61*SS
    arr*=1-(sm*.66*visiblewood)[...,None]
    # Very subtle lens softness and a gentle warm photographic color grade.
    img=Image.fromarray(np.uint8(np.clip(arr,0,255))).resize((W,H),Image.Resampling.LANCZOS)
    a=np.array(img).astype(float)
    y,x=np.mgrid[:H,:W];vignette=1-.055*((x-128)**2/128**2+(y-78)**2/144**2)
    a*=vignette[...,None]
    a=a*np.array([1.037,1.012,.966])+np.array([1.0,.7,.35])
    return np.uint8(np.clip(a,0,255))


def main():
    import time
    t=time.time();frames=[]

if __name__=='__main__':main()
