def metal_bar(im,pivot,end,upper):
    v=end-pivot;u=v/np.linalg.norm(v);n=np.array([-u[1],u[0]])
    # Long chrome handles, thicker at the hand, and thick toothed jaws near the hinge.
    p0=pivot-u*3;p1=pivot+v*.21;p2=pivot+v*.40;p3=end
    pts=[tuple(p0+n*3),tuple(p1+n*3.3),tuple(p2+n*2.8),tuple(p3+n*3.5),
         tuple(p3+u*2),tuple(p3-n*3.5),tuple(p2-n*2.8),tuple(p1-n*3.3),tuple(p0-n*3)]
    mask=path_mask(pts);shadow(im,mask,1.2,(.9,2),.3)
    surface(im,mask,(165,173,176),height=2.2,bevel=2.6,material='metal')
    # Long reflected light and a darker machined edge.
    line(im,[tuple(pivot+n*-1.2),tuple(end+n*-1.2)],(222,228,224),.85)
    line(im,[tuple(pivot+n*2.1),tuple(end+n*2.4)],(86,95,98),.6)
    sign=1 if upper else -1
    # Opposed rows of short teeth around the claw.
    for t in np.arange(.16,.43,.052):
        p=pivot+v*t+n*(3*sign)
        tip=p+n*1.65*sign+u*.65

        surface(im,mm,(140,148,148),height=.6,bevel=.6,material='metal')
    # Fine etched/scratched steel highlights, stable with respect to tool motion.
    for t in [.52,.61,.71,.79,.89]:
        p=pivot+v*t
        line(im,[tuple(p-u*2+n*.2),tuple(p+u*2+n*.45)],(189,193,187),.25)
