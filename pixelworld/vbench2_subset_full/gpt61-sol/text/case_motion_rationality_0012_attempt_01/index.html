    def __init__(self,background):

    def wood(self,p,kind):

        if kind=='counter':
            grain=3*np.sin(y*53+2*np.sin(x*.8))+2*np.sin(y*123+x*.3)

        col+=grain[...,None]*np.array([1,.8,.6])
        # Thin scoring marks in the working area, not a repeated photographic texture.
        scratches=np.sin(x*97+y*53)*np.sin(x*73-y*88)
        mask=(np.abs(x)<1.9)&(np.abs(y)<1.1)&(scratches>.985)
        col[mask]-=[14,12,8]

            for cx,cy,rx,ry,strength in self.shadow_data:
                d=((x-cx)/rx)**2+((y-cy)/ry)**2
                shade*=1-strength*np.exp(-d*1.6)
            col*=shade[...,None]

    def render(self,mesh):
        v,n,c,f=mesh.arrays()
        if len(v)==0:return
        dv=v-CAM
        z=dv@FWD;xy=np.stack([SW/2+FOCAL*(dv@RIGHT)/z,SH/2-FOCAL*(dv@UP)/z],axis=-1)
        n=norm(n)
        diffuse=np.maximum(0,n@LIGHT)
        view=norm(CAM-v);halfvec=norm(view+LIGHT)
        spec=np.maximum(0,np.sum(n*halfvec,axis=-1))**40
        ambient=.63+.045*n[:,2]
        shade=ambient+.37*diffuse
        shaded=c*shade[:,None]+spec[:,None]*13
        for index,face in enumerate(f):
            q=xy[face];depth=z[face]
            if np.min(depth)<=.1:continue
            minx=max(0,int(np.floor(np.min(q[:,0]))));maxx=min(SW-1,int(np.ceil(np.max(q[:,0]))))
            miny=max(0,int(np.floor(np.min(q[:,1]))));maxy=min(SH-1,int(np.ceil(np.max(q[:,1]))))
            if maxx<minx or maxy<miny:continue
            xa,ya=q[0];xb,yb=q[1];xc,yc=q[2]
            den=(yb-yc)*(xa-xc)+(xc-xb)*(ya-yc)
            if abs(den)<1e-8:continue
            yy,xx=np.ogrid[miny:maxy+1,minx:maxx+1]
            a=((yb-yc)*(xx+.5-xc)+(xc-xb)*(yy+.5-yc))/den
            b=((yc-ya)*(xx+.5-xc)+(xa-xc)*(yy+.5-yc))/den
            cc=1-a-b
            inside=(a>=-1e-5)&(b>=-1e-5)&(cc>=-1e-5)
            iz=a/depth[0]+b/depth[1]+cc/depth[2];zz=1/np.maximum(iz,1e-9)
            region=self.z[miny:maxy+1,minx:maxx+1]
            valid=inside&(zz<region)
            if not np.any(valid):continue
            # Perspective-correct interpolation.
            aa=(a/depth[0]/iz)[valid];bb=(b/depth[1]/iz)[valid];ccc=(cc/depth[2]/iz)[valid]
            kind=mesh.kind[index]
            if kind in ['board','board_edge','counter']:
                p=aa[:,None]*v[face[0]]+bb[:,None]*v[face[1]]+ccc[:,None]*v[face[2]]
                rgb=self.wood(p,kind)
                light=(aa*shade[face[0]]+bb*shade[face[1]]+ccc*shade[face[2]])
                rgb*=light[:,None]
            else:
                rgb=aa[:,None]*shaded[face[0]]+bb[:,None]*shaded[face[1]]+ccc[:,None]*shaded[face[2]]
                if kind=='fabric':

                elif kind=='metal':
                    rgb+=4
            self.image[miny:maxy+1,minx:maxx+1][valid]=np.clip(rgb,0,255)
            region[valid]=zz[valid]
