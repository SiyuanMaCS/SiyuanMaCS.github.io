+def ease(a,b,t):

+def ry(a):

+def rz(a):

+class Layer:

def noise(x,y,scale=1.,i=0):
    return map_coordinates(NOISE[i],[(y*scale*23+256)%512,(x*scale*23+256)%512],order=1,mode='wrap')

def ease(a,b,t):
    u=np.clip((t-a)/(b-a),0.,1.)
    return u*u*(3-2*u)

def ry(a):
    c,s=np.cos(a),np.sin(a)
    return np.array([[c,0,s],[0,1,0],[-s,0,c]])

def rz(a):
    c,s=np.cos(a),np.sin(a)
    return np.array([[c,-s,0],[s,c,0],[0,0,1.]])

def rigid(pivot,center,rot):
    return rot,np.asarray(center)-rot@np.asarray(pivot)

def point(p,T):
    return T[0]@np.asarray(p)+T[1]

def crop(center,radius):
    px=RW/2+(center@RIGHT)*SCALE

class Layer:
    def __init__(self):
        self.d=np.full((RH,RW),-1e6,dtype=np.float32)
        self.norm=np.zeros((RH,RW,3),dtype=np.float32)
        self.col=np.zeros((RH,RW,3),dtype=np.float32)
        self.mat=np.zeros((RH,RW),dtype=np.uint8)
    def put(self,bounds,d,n,c,mat,smooth=0.):
        x0,x1,y0,y1=bounds
        if x1<=x0 or y1<=y0:return
        old=self.d[y0:y1,x0:x1]
        valid=d>-1e5
        win=valid & (d>old)
        if smooth:
            mix=valid & (old>-1e5) & (np.abs(d-old)<smooth)
            h=np.clip(.5+.5*(d-old)/smooth,0.,1.)
            w=h[...,None]
            norm=self.norm[y0:y1,x0:x1]
            col=self.col[y0:y1,x0:x1]
            norm[mix]=(norm*(1-w)+n*w)[mix]
            col[mix]=(col*(1-w)+c*w)[mix]
            old[mix]=(old*(1-h)+d*h+smooth*h*(1-h))[mix]
            win&=~mix
        old[win]=d[win]
        self.norm[y0:y1,x0:x1][win]=n[win]
        self.col[y0:y1,x0:x1][win]=c[win]
        self.mat[y0:y1,x0:x1][valid & (d>=old-.02)]=mat
    def merge(self,other):
        mask=other.d>self.d
        self.d[mask]=other.d[mask]
        self.norm[mask]=other.norm[mask]
        self.col[mask]=other.col[mask]
        self.mat[mask]=other.mat[mask]

def ellipsoid(layer,center,radii,rot=np.eye(3),mat=2,smooth=0.,color=None):

def segment(layer,a,b,width,depth=None,mat=2,T=None):
    a=np.asarray(a,dtype=float);b=np.asarray(b,dtype=float)
    direction=b-a;length=np.linalg.norm(direction)

# Convex chamfered-box ray intersection, including its diagonal bevel planes.
def bevelbox(layer,center,half,bevel,rot,mat=1,tile=False,seed=0,broken=False):
    center=np.asarray(center);half=np.asarray(half)
    b=crop(center,np.linalg.norm(half))
    x0,x1,y0,y1=b
    if x1<=x0 or y1<=y0:return
    q=(Q[y0:y1,x0:x1]-center)@rot
    v=VIEW@rot
    far=np.full(q.shape[:2],-1e5)
    near=np.full(q.shape[:2],1e5)
    normal=np.zeros(q.shape)
    valid=np.ones(q.shape[:2],bool)
    # Every nonzero vector with components -1,0,+1 gives a supporting plane.
    for nx in [-1,0,1]:
        for ny in [-1,0,1]:
            for nz in [-1,0,1]:
                if nx==ny==nz==0:continue
                n=np.array([nx,ny,nz],dtype=float)
                count=np.count_nonzero(n)
                off=np.sum(np.abs(n)*half)-bevel*(count-1)
                den=n@v
                num=off-np.sum(q*n,axis=-1)
                if abs(den)<1e-8:
                    valid &= num>=0
                elif den>0:
                    upper=num/den
                    replace=upper<near
                    near[replace]=upper[replace]
                    normal[replace]=n/np.linalg.norm(n)
                else:far=np.maximum(far,num/den)
    valid&=near>=far
    hit=q+near[...,None]*v
    normal=normal@rot.T
    nv=noise(hit[...,0]+seed*.23,hit[...,1]+seed*.18,2.6,0)
    base=np.array([.285,.142,.078]) if tile else np.array([.244,.113,.058])
    col=np.broadcast_to(base,hit.shape).copy()
    col+=nv[...,None]*.008
    if tile:
        # Fine embossed inset and central cocoa-leaf motif, moulded into each square.
        ax=np.abs(hit[...,0]);ay=np.abs(hit[...,1])
        insetx=half[0]-.145;insety=half[1]-.145
        rectdist=np.maximum(ax-insetx,ay-insety)
        line=np.exp(-(rectdist/.017)**2)
        top=(hit[...,2]>half[2]-.021)
        col-=line[...,None]*top[...,None]*np.array([.027,.019,.012])
        rim=np.exp(-((rectdist+.027)/.015)**2)
        col+=rim[...,None]*top[...,None]*np.array([.019,.013,.008])
        # Small diagonal bean imprint (a restrained design, no fake lettering).
        u=hit[...,0]*.86+hit[...,1]*.5
        v2=-hit[...,0]*.5+hit[...,1]*.86
        oval=np.sqrt((u/.15)**2+(v2/.245)**2)
        bean=np.exp(-((oval-1)/.10)**2)
        slash=np.exp(-((u-.022*np.sin(v2*15))/.012)**2)*(np.abs(v2)<.19)
        col-=(bean*.012+slash*.012)[...,None]*top[...,None]
    if broken:
        side=np.abs(hit[...,0])>half[0]-.022
        rough=noise(hit[...,1]+seed,hit[...,2],9.,0)
        col+=side[...,None]*(rough[...,None]*.026+np.array([.025,.011,.004]))
    layer.put(b,np.where(valid,near,-1e6),normal,np.clip(col,0,1),mat)

def transforms(t):

    leftR=ry(angle*(1-.8*ease(1.9,2.7,t)))@rz(-.022)
    leftshift=-.63*first
    leftz=2.68-.18*ease(1.9,2.7,t)
    L=rigid([0,0,0],[leftshift,0,leftz],leftR)
    C=rigid([0,0,0],[1.13*first,.16*first,2.68],ry(-angle)@rz(.018))
    if t>=1.8:
        u=ease(1.8,2.48,t)
        startcenter=point([1.875,0,0],C)
        dest=np.array([4.35,3.55,.255])
        cen=startcenter*(1-u)+dest*u
        cen[2]+=.18*np.sin(np.pi*u)
        rot=rz(.018*(1-u)-.19*u)@ry(-angle*(1-u))
        C=rigid([1.875,0,0],cen,rot)
    secondT=ease(3.05,3.58,t)
    second=ease(3.58,4.35,t)
    pivot=np.array([-1.25,0,0.])

    A=rigid(pivot,pivotworld+np.array([-.44*second,-.10*second,.04*second]),leftR@ry(.115*secondT))
    B=rigid(pivot,pivotworld+np.array([1.63*second,.40*second,.1*second]),leftR@ry(-.145*secondT)@rz(.055*second))
    LH=(A[0],point([-3.75,0,0],A))

    RB=(B[0],point([0,0,0],B))
    if t<2.48:

    elif t<3.05:

    dist=((TX-px)/rng.uniform(.01,.025))**2+((TY-py)/rng.uniform(.15,.65))**2
    TABLE_COLOR-=np.exp(-dist)[...,None]*.022


def shadow(t,Ts):
    A,B,C,LH,RH,opened=Ts
    im=Image.new('L',(RW,RH),0);draw=ImageDraw.Draw(im)
    def castpoly(points,opacity=140):
        points=np.asarray(points)
        dropped=points-points[:,2,None]*LIGHT/LIGHT[2]
        draw.polygon([tuple(p) for p in project(dropped)],fill=opacity)
    for cols,T in [([0,1],A),([2],B),([3,4,5],C)]:
        lo=-3.75+min(cols)*CW;hi=-3.75+(max(cols)+1)*CW
        castpoly([point([x,y,0],T) for x,y in [(lo,-2.025),(hi,-2.025),(hi,2.025),(lo,2.025)]],180)
    for side,T in [('left',LH),('right',RH)]:
        M=np.diag([-1.,1.,1.]) if side=='left' else np.eye(3)
        for cen,rad in [([1.2,-2.3,-.2],[1.15,1.5]),([1.6,-4.2,-.5],[.8,1.7]),([2.4,-7,-.7],[1.,2.3]),([.1,-1,.3],[.75,.6]),([1.1,-.4,-.2],[.85,1.25])]:
            points=[]
            for theta in np.linspace(0,2*np.pi,28):
                p=np.asarray(cen)+np.array([rad[0]*np.cos(theta),rad[1]*np.sin(theta),0.])
                points.append(point(M@p,T))
            castpoly(points,135)
    # Area-light penumbra, with a tighter component for grounded chocolate.
    broad=np.asarray(im.filter(ImageFilter.GaussianBlur(SS*2.5)),dtype=float)/255
    fine=np.asarray(im.filter(ImageFilter.GaussianBlur(SS*.6)),dtype=float)/255
    return np.clip(.18*broad+.10*fine,0,.42)


def crumbs(layer,t):
    for event,x0 in [(1.40,0),(3.60,-1.88)]:
        age=t-event
        if age<0:continue
        r=np.random.default_rng(int(event*100))
        for i in range(11):
            x=x0+r.uniform(-.07,.07)
            y=r.uniform(-1.8,1.8)
            vx=r.uniform(-.6,.75);vy=r.uniform(-.5,.8)
            launch=2.4 if event>3 else 2.68
            fall=min(age,np.sqrt(2*launch/9.8))
            p=[x+vx*fall,y+vy*fall,max(.04,launch-4.9*age*age)]
            rr=r.uniform(.025,.063)
            ellipsoid(layer,p,[rr,rr*.7,rr*.54],rz(i*1.4)@ry(i*.8),5)


def shade(layer,t,Ts):
    hit=Q+layer.d[...,None]*VIEW
    normals=layer.norm
    normals/=np.maximum(np.linalg.norm(normals,axis=-1,keepdims=True),1e-6)
    nl=np.sum(normals*LIGHT,axis=-1)
    nv=np.clip(np.sum(normals*VIEW,axis=-1),0,1)
    nh=np.clip(np.sum(normals*HALF,axis=-1),0,1)
    linear=np.clip(layer.col,0,1)**2.2
    amb=np.array([.56,.53,.49])
    diff=np.clip(nl,0,1)
    lit=linear*(amb+diff[...,None]*np.array([.57,.535,.49]))
    mat=layer.mat
    skin=mat==2; chocolate=mat==1; nails=mat==3; cloth=mat==4
    # Broad skin highlights and warm light transmitted near grazing silhouettes.
    lit[skin]+=((nh**23)*.022)[skin,None]*np.array([1.,.91,.81])
    wrap=(np.clip((nl+.36)/1.36,0,1)-diff)*.20
    lit[skin]+=(linear*wrap[...,None]*np.array([1.4,.75,.45]))[skin]
    fresnel=.018+.065*(1-nv)**5
    gloss=(nh**65)*fresnel+(nh**12)*.006
    lit[chocolate]+=(gloss[...,None]*np.array([1.,.82,.61]))[chocolate]
    lit[nails]+=((nh**62)*.045)[nails,None]
    lit[cloth]*=.93
    # Soft screen-space crevice occlusion, restrained so it does not outline objects.
    safe=layer.d
    blur=gaussian_filter(safe,SS*1.0)
    occ=np.clip(blur-safe-.03,0,.75)*.15
    lit*=1-occ[...,None]
    shade_mask=shadow(t,Ts)
    lit[mat==0]*=(1-shade_mask[...,None])[mat==0]
    color=np.clip(lit,0,1)**(1/2.2)
    # Natural window-lit exposure falloff and subtle lens vignette.
    vignette=1-.09*((sx/10)**2+((sy+.1)/6.4)**2)
    color*=np.clip(vignette,.83,1.)[...,None]
    color=np.clip(color,0,1)
    return color


def render(t):
    scene=Layer()
    scene.d[:]=dtable
    scene.norm[:]=[0,0,1]
    scene.col[:]=TABLE_COLOR
    # A discreet, softly wrinkled linen napkin far from the action, in the upper left.
    # Rendered as the same hand-built geometry, not imported scene assets.
    napkin=Layer()
    bevelbox(napkin,[-6.45,5.85,.045],[2.15,2.1,.035],.04,rz(.13),mat=6)
    mask=napkin.d>dtable
    coords=Q+napkin.d[...,None]*VIEW
    linen=.012*np.sin(coords[...,0]*112)*np.sin(coords[...,1]*105)
    napkin.col[:]=np.array([.72,.699,.64])+linen[...,None]
    napkin.col-=np.exp(-((coords[...,0]+6.6)/.11)**2)[...,None]*.025
    scene.merge(napkin)
    Ts=transforms(t)
    A,B,C,LH,RH,opened=Ts
    objects=Layer()
    chocolate(objects,[0,1],A,t>=3.60)
    chocolate(objects,[2],B,t>=1.4)
    chocolate(objects,[3,4,5],C,t>=1.4)
    # Fingers are depth tested, so opposing thumbs are on top and the curled digits below.
    objects.merge(hand(LH,'left'))
    objects.merge(hand(RH,'right',opened))
    crumbs(objects,t)
    scene.merge(objects)
    rgb=shade(scene,t,Ts)
    rgb=(rgb*255+.5).astype(np.uint8)
    image=Image.fromarray(rgb).resize((W,H),Image.Resampling.LANCZOS)
    return np.asarray(image)
