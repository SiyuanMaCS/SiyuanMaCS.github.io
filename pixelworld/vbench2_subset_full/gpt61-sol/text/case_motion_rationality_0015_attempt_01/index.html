def shadows(objects,base_ids):
    mask=Image.new('L',(WW,HH),0);dr=ImageDraw.Draw(mask)
    contact=Image.new('L',(WW,HH),0);dc=ImageDraw.Draw(contact)
    for m in objects:
        # The projection of the silhouette onto the board from an overhead area light.
        v=m.v.copy();z=np.maximum(0,v[:,2]-.124)
        v[:,:2]-=z[:,None]*(LIGHT[:2]/LIGHT[2]);v[:,2]=.122
        pp=project(v)
        target=dc if m.v[:,2].max()<.43 else dr
        for f in m.f:
            pts=pp[f,:2]
            if (pts[:,0].max()<0 or pts[:,0].min()>WW or pts[:,1].max()<0 or pts[:,1].min()>HH):continue
            target.polygon([tuple(p) for p in pts],fill=255)
    broad=np.asarray(mask.filter(ImageFilter.GaussianBlur(2.0*S)),dtype=float)/255
    tight=np.asarray(contact.filter(ImageFilter.GaussianBlur(.40*S)),dtype=float)/255
    return (1-.21*broad-.19*tight)*(base_ids==1)+(base_ids"'!=1)


def scene(i):
    # Three deliberate lengthwise slices over five seconds, with a pause at contact.
    cycle=min(i//6,2);p=(i-cycle*6)/6
    cut_y=[-.22,.015,.245][cycle]
    if i>=18:
        z=.18+(i-18)*.28;cut_y=.245
    elif p<.50:
        z=.13+.72*(.5+.5*np.cos(np.pi*p/.50))
    elif p<.68:z=.131
    else:
        q=(p-.68)/.32;z=.13+.72*(.5-.5*np.cos(np.pi*q))
    # Smooth lateral repositioning happens only while the edge is clear.
    if cycle>0 and p<.17:
        q=p/.17;cut_y=[-.22,.015,.245][cycle-1]+(cut_y-[-.22,.015,.245][cycle-1])*(q*q*(3-2*q))
    objects=[]
    boundaries=[-.44,-.22,.015,.245,.61]
    cuts_done=min(3,sum(i>=j*6+3 for j in range(3)))
    for j in range(cuts_done):
        elapsed=(i-(j*6+3))/4
        slide=.055*(1-np.exp(-elapsed*4))
        objects+=carrot_piece(-1.47,.72,boundaries[j]+.012,boundaries[j+1]-.012,.25,ang=-.012*(1-np.exp(-elapsed*3)),shift=(.018*(j+1),-slide),seed=j+2)
    start=boundaries[cuts_done]
    objects+=carrot_piece(-1.47,.72,start,.61,.25,seed=7)
    objects+=left_hand(i/4)
    objects+=knife(cut_y,z)
    objects+=right_hand(cut_y,z)
    return objects
